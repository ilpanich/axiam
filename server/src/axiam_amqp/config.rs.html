6/// Documented dev/test-only default AMQP master signing key (SECHRD-08 /
7/// D-05c). Used ONLY as a fallback in debug builds (`cfg!(debug_assertions)`
8/// — i.e. never in the `cargo build --release` binary that ships in the
9/// production container image, see `docker/Dockerfile.server`) when
10/// `AXIAM__AMQP__SIGNING_KEY` is unset, so local dev/test runs work without
11/// extra setup. This key MUST NOT be used in production — see
12/// [`AmqpConfig::resolve_signing_key`], which fails closed instead of
13/// falling back to this constant in a release build.
14const DEV_DEFAULT_SIGNING_KEY: &[u8] = b"axiam-dev-only-amqp-signing-key-DO-NOT-USE-IN-PROD";

16/// TLS material for an `amqps://` broker connection (A6).
17///
18/// Every field is optional: with none set, an `amqps://` URL still connects,
19/// verifying the broker against the **system** root store. The fields exist for
20/// the two cases the system store cannot serve — a privately-issued broker
21/// certificate, and mutual TLS toward the broker.
22///
23/// # There is deliberately no `verify_peer: false`
24///
25/// Not as an oversight, and not as something to add later behind a scary name.
26/// A verification-skip switch is the single most reliably misused TLS option
27/// there is: it appears in a dev compose file, it works, and it travels
28/// unchanged into production, where it turns TLS into an expensive no-op
29/// against exactly the attacker TLS exists to stop. `ca_cert_path` covers the
30/// legitimate reason people reach for it (a self-signed or private-CA broker
31/// certificate) without also covering the illegitimate ones.
32#[derive(Debug, Clone, Default, Deserialize, PartialEq, Eq)]
33#[serde(default)]
34pub struct AmqpTlsConfig {
35    /// PEM bundle of the CA(s) that issued the broker's certificate.
36    ///
37    /// Unset = verify against the platform root store. Set this when the broker
38    /// certificate is issued by a private CA — including one issued by AXIAM's
39    /// own `axiam-pki` org CA, which is the recommended dogfooding path.
40    ///
41    /// # This ADDS a root; it does not replace the platform store (SEC-106)
42    ///
43    /// Setting this does **not** narrow the trust set to your CA. lapin hands
44    /// the bundle to `tcp-stream`, whose rustls backend calls
45    /// `RustlsConnectorConfig::add_parsable_certificates` on top of the
46    /// platform verifier's existing roots
47    /// (`tcp-stream-0.34/src/rustls_impl.rs`). So after setting it, a
48    /// certificate for the broker's hostname issued by *any* publicly trusted
49    /// CA is still accepted — the trust set got wider, not narrower, which is
50    /// the opposite of what "pin my private broker CA" is usually meant to
51    /// achieve.
52    ///
53    /// There is no configuration here that changes that: `OwnedTLSConfig`
54    /// carries only an identity and a certificate chain, with no hook for a
55    /// rustls `ClientConfig`. If you need the trust set actually restricted to
56    /// your CA, restrict it at the platform trust store the container image
57    /// ships (`/etc/ssl/certs`) and leave this unset, or authenticate the
58    /// broker with mutual TLS via [`Self::client_cert_path`], which is a
59    /// stronger statement than root pinning anyway.
60    pub ca_cert_path: Option<String>,
61    /// PEM client certificate, for mutual TLS toward the broker.
62    ///
63    /// Must be set together with [`Self::client_key_path`]; one without the
64    /// other is a misconfiguration and fails closed rather than silently
65    /// connecting without a client certificate.
66    pub client_cert_path: Option<String>,
67    /// PEM client private key matching [`Self::client_cert_path`].
68    pub client_key_path: Option<String>,
69}
70
71impl AmqpTlsConfig {
72    /// Whether any TLS material is configured at all.
73    pub fn is_empty(&self) -> bool {
74        self.ca_cert_path.is_none()
75            && self.client_cert_path.is_none()
76            && self.client_key_path.is_none()
77    }
78
79    /// Validate the combination, without touching the filesystem.
80    ///
81    /// The one rule: a client certificate and its key travel together. Half a
82    /// client identity is never what an operator meant, and connecting anyway
83    /// would silently drop the mutual half of mutual TLS.
84    pub fn validate(&self) -> Result<(), AxiamError> {
85        match (&self.client_cert_path, &self.client_key_path) {
86            (Some(_), None) => Err(AxiamError::ServiceUnavailable(
87                "AXIAM__AMQP__TLS__CLIENT_CERT_PATH is set without \
88                 AXIAM__AMQP__TLS__CLIENT_KEY_PATH — a client certificate without its \
89                 key cannot authenticate; set both or neither"
90                    .into(),
91            )),
92            (None, Some(_)) => Err(AxiamError::ServiceUnavailable(
93                "AXIAM__AMQP__TLS__CLIENT_KEY_PATH is set without \
94                 AXIAM__AMQP__TLS__CLIENT_CERT_PATH; set both or neither"
95                    .into(),
96            )),
97            _ => Ok(()),
98        }
99    }
100}

102/// Configuration for connecting to RabbitMQ.
103#[derive(Debug, Clone, Deserialize)]
104#[serde(default)]
105pub struct AmqpConfig {
106    /// AMQP connection URI.
107    ///
108    /// `amqps://host:5671` selects TLS (see [`AmqpConfig::is_tls`]);
109    /// `amqp://host:5672` is plaintext and, in a release build, must be
110    /// explicitly permitted via [`AmqpConfig::allow_plaintext`].
111    pub url: String,
112    /// TLS material for an `amqps://` connection (A6). Ignored for `amqp://`.
113    #[serde(default)]
114    pub tls: AmqpTlsConfig,
115    /// Permit a plaintext `amqp://` broker URL in a **release** build (A6).
116    ///
117    /// Set via `AXIAM__AMQP__ALLOW_PLAINTEXT`. Default `false`, and the
118    /// failure is deliberate: broker traffic carries authorization requests,
119    /// audit events and outbound mail payloads across service boundaries, and
120    /// the AMQP layer's HMAC gives those **authenticity and replay protection
121    /// but not confidentiality** — plaintext means they cross the wire
122    /// readable. This mirrors [`DEV_DEFAULT_SIGNING_KEY`]'s posture exactly:
123    /// convenient in a debug build, an explicit operator decision in a release
124    /// one.
125    #[serde(default)]
126    pub allow_plaintext: bool,
127    /// Channel prefetch count for consumers.
128    pub prefetch_count: u16,
129    /// Delay between reconnection attempts in milliseconds.
130    pub reconnect_delay_ms: u64,
131    /// Maximum number of connection retries before giving up.
132    pub max_retries: u32,
133    /// HMAC-SHA256 master signing key for authenticating AMQP message
134    /// payloads (SEC-022/055, SECHRD-08). Set via
135    /// `AXIAM__AMQP__SIGNING_KEY` (hex-encoded key). Signing is mandatory —
136    /// there is no unsigned code path (D-05c). Use
137    /// [`AmqpConfig::resolve_signing_key`] to resolve this to the mandatory
138    /// key that consumers/producers must use: a debug-build-only dev
139    /// default is used when unset in development, but a release build
140    /// fails closed (`AxiamError::ServiceUnavailable`) if unset.
141    #[serde(default)]
142    pub signing_key: Option<String>,
143    /// Freshness skew window (seconds) for AMQP replay protection (NEW-4). An
144    /// `AuthzRequest`/`AuditEventMessage` is accepted only when its `issued_at`
145    /// lies within ±this many seconds of the consumer's clock. Set via
146    /// `AXIAM__AMQP__REPLAY_SKEW_SECS`; defaults to
147    /// [`crate::messages::DEFAULT_FRESHNESS_SKEW_SECS`] (5 minutes).
148    #[serde(default = "default_replay_skew_secs")]
149    pub replay_skew_secs: u64,
150}
151
152fn default_replay_skew_secs() -> u64 {
153    crate::messages::DEFAULT_FRESHNESS_SKEW_SECS as u64
154}
155
156impl Default for AmqpConfig {
157    fn default() -> Self {
158        Self {
159            url: "amqp://localhost:5672".into(),
160            tls: AmqpTlsConfig::default(),
161            allow_plaintext: false,
162            prefetch_count: 10,
163            reconnect_delay_ms: 5000,
164            max_retries: 5,
165            signing_key: None,
166            replay_skew_secs: default_replay_skew_secs(),
167        }
168    }
169}
170
171impl AmqpConfig {
172    /// Resolve the configured signing key to a mandatory master key
173    /// (SECHRD-08 / D-05c).
174    ///
175    /// - If `signing_key` is set, hex-decode and return it (an invalid hex
176    ///   value is a misconfiguration and fails closed).
177    /// - If unset in a debug build (`cfg!(debug_assertions)` — dev/test),
178    ///   fall back to the documented [`DEV_DEFAULT_SIGNING_KEY`] so local
179    ///   runs work without extra setup.
180    /// - If unset in a release build (the production container image builds
181    ///   with `cargo build --release`), fail closed: return
182    ///   `AxiamError::ServiceUnavailable` — AMQP signing is mandatory in
183    ///   production, there is no unsigned/zero-key code path.
184    pub fn resolve_signing_key(&self) -> Result<Vec<u8>, AxiamError> {
185        match &self.signing_key {
186            Some(hex_key) => hex::decode(hex_key).map_err(|e| {
187                AxiamError::ServiceUnavailable(format!(
188                    "AXIAM__AMQP__SIGNING_KEY is not valid hex: {e}"
189                ))
190            }),
191            None if cfg!(debug_assertions) => {
192                tracing::warn!(
193                    "AXIAM__AMQP__SIGNING_KEY not set — using dev-only default AMQP signing key \
194                     (NOT valid in a release/production build, SECHRD-08 / D-05c)"
195                );
196                Ok(DEV_DEFAULT_SIGNING_KEY.to_vec())
197            }
198            None => Err(AxiamError::ServiceUnavailable(
199                "AMQP signing key not configured (AXIAM__AMQP__SIGNING_KEY) — mandatory in \
200                 production (SECHRD-08 / D-05c)"
201                    .to_string(),
202            )),
203        }
204    }
205
206    /// Whether [`Self::url`] selects a TLS connection (A6).
207    ///
208    /// Scheme-based, case-insensitive, and deliberately strict about the
209    /// separator: `amqps://` is TLS, `amqp://` is not, and anything else is
210    /// neither — [`Self::validate_transport_security`] rejects it rather than
211    /// guessing. Matching on `amqps` without the `://` would classify a
212    /// hypothetical `amqpsomething://` as TLS.
213    pub fn is_tls(&self) -> bool {
214        self.url.trim().to_ascii_lowercase().starts_with("amqps://")
215    }
216
217    /// Enforce the transport-security posture before any connection is
218    /// attempted (A6).
219    ///
220    /// Rules, in order:
221    ///
222    /// 1. An unrecognised scheme fails — better a clear error at startup than
223    ///    an unclear one from lapin at connect time.
224    /// 2. TLS material that is internally inconsistent fails
225    ///    ([`AmqpTlsConfig::validate`]).
226    /// 3. A plaintext URL fails in a **release** build unless
227    ///    [`Self::allow_plaintext`] is set, and logs a prominent warning when
228    ///    it is. Debug builds allow plaintext silently-ish so `just dev-up`
229    ///    keeps working.
230    ///
231    /// The project standard is "TLS 1.3 minimum for all external
232    /// communication", and six SDKs consume AMQP directly, so broker traffic
233    /// crosses service boundaries by design. Failing closed here is what makes
234    /// that standard true of the broker rather than aspirational about it.
235    pub fn validate_transport_security(&self) -> Result<(), AxiamError> {
236        let url = self.url.trim().to_ascii_lowercase();
237        let is_plaintext = url.starts_with("amqp://");
238
239        if !self.is_tls() && !is_plaintext {
240            return Err(AxiamError::ServiceUnavailable(format!(
241                "AXIAM__AMQP__URL must start with amqps:// or amqp:// (got {:?})",
242                self.url
243            )));
244        }
245
246        self.tls.validate()?;
247
248        if self.is_tls() {
249            return Ok(());
250        }
251
252        // Plaintext from here down.
253        if !self.tls.is_empty() {
254            tracing::warn!(
255                "AMQP TLS material is configured but the URL is plaintext amqp:// — \
256                 the certificates will NOT be used. Switch the URL to amqps:// (port 5671)."
257            );
258        }
259
260        if cfg!(debug_assertions) {
261            tracing::debug!(
262                "AMQP transport is plaintext (debug build) — broker payloads cross the \
263                 wire readable; use amqps:// in any deployment"
264            );
265            return Ok(());
266        }
267
268        if self.allow_plaintext {
269            tracing::warn!(
270                "AMQP transport is PLAINTEXT and explicitly permitted via \
271                 AXIAM__AMQP__ALLOW_PLAINTEXT. Authorization requests, audit events and \
272                 outbound mail payloads cross the wire readable. HMAC signing gives \
273                 authenticity and replay protection, NOT confidentiality — it is not a \
274                 substitute for TLS."
275            );
276            return Ok(());
277        }
278
279        Err(AxiamError::ServiceUnavailable(
280            "AMQP transport is plaintext (amqp://) in a release build. Broker traffic \
281             carries authorization requests, audit events and mail payloads across \
282             service boundaries, and HMAC signing protects their authenticity but not \
283             their confidentiality. Use amqps:// (port 5671), or set \
284             AXIAM__AMQP__ALLOW_PLAINTEXT=true to accept the exposure deliberately."
285                .to_string(),
286        ))
287    }
288
289    /// Resolve the configured freshness skew to a `chrono::Duration` for the
290    /// AMQP replay-protection gate (NEW-4).
291    pub fn replay_skew(&self) -> chrono::Duration {
292        chrono::Duration::seconds(self.replay_skew_secs as i64)
293    }
294}
