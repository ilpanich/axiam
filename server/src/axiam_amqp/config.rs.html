6/// Documented dev/test-only default AMQP master signing key (SECHRD-08 /
7/// D-05c). Used ONLY as a fallback in debug builds (`cfg!(debug_assertions)`
8/// — i.e. never in the `cargo build --release` binary that ships in the
9/// production container image, see `docker/Dockerfile.server`) when
10/// `AXIAM__AMQP__SIGNING_KEY` is unset, so local dev/test runs work without
11/// extra setup. This key MUST NOT be used in production — see
12/// [`AmqpConfig::resolve_signing_key`], which fails closed instead of
13/// falling back to this constant in a release build.
14const DEV_DEFAULT_SIGNING_KEY: &[u8] = b"axiam-dev-only-amqp-signing-key-DO-NOT-USE-IN-PROD";

16/// TLS material for an `amqps://` broker connection (A6).
17///
18/// Every field is optional: with none set, an `amqps://` URL still connects,
19/// verifying the broker against the **system** root store. The fields exist for
20/// the two cases the system store cannot serve — a privately-issued broker
21/// certificate, and mutual TLS toward the broker.
22///
23/// # There is deliberately no `verify_peer: false`
24///
25/// Not as an oversight, and not as something to add later behind a scary name.
26/// A verification-skip switch is the single most reliably misused TLS option
27/// there is: it appears in a dev compose file, it works, and it travels
28/// unchanged into production, where it turns TLS into an expensive no-op
29/// against exactly the attacker TLS exists to stop. `ca_cert_path` covers the
30/// legitimate reason people reach for it (a self-signed or private-CA broker
31/// certificate) without also covering the illegitimate ones.
32#[derive(Debug, Clone, Default, Deserialize, PartialEq, Eq)]
33#[serde(default)]
34pub struct AmqpTlsConfig {
35    /// PEM bundle of the CA(s) that issued the broker's certificate.
36    ///
37    /// Unset = verify against the platform root store. Set this when the broker
38    /// certificate is issued by a private CA — including one issued by AXIAM's
39    /// own `axiam-pki` org CA, which is the recommended dogfooding path.
40    ///
41    /// # This ADDS a root; it does not replace the platform store (SEC-106)
42    ///
43    /// Setting this does **not** narrow the trust set to your CA. lapin hands
44    /// the bundle to `tcp-stream`, whose rustls backend calls
45    /// `RustlsConnectorConfig::add_parsable_certificates` on top of the
46    /// platform verifier's existing roots
47    /// (`tcp-stream-0.34/src/rustls_impl.rs`). So after setting it, a
48    /// certificate for the broker's hostname issued by *any* publicly trusted
49    /// CA is still accepted — the trust set got wider, not narrower, which is
50    /// the opposite of what "pin my private broker CA" is usually meant to
51    /// achieve.
52    ///
53    /// There is no configuration here that changes that: `OwnedTLSConfig`
54    /// carries only an identity and a certificate chain, with no hook for a
55    /// rustls `ClientConfig`. If you need the trust set actually restricted to
56    /// your CA, restrict it at the platform trust store the container image
57    /// ships (`/etc/ssl/certs`) and leave this unset, or authenticate the
58    /// broker with mutual TLS via [`Self::client_cert_path`], which is a
59    /// stronger statement than root pinning anyway.
60    pub ca_cert_path: Option<String>,
61    /// PEM client certificate, for mutual TLS toward the broker.
62    ///
63    /// Must be set together with [`Self::client_key_path`]; one without the
64    /// other is a misconfiguration and fails closed rather than silently
65    /// connecting without a client certificate.
66    pub client_cert_path: Option<String>,
67    /// PEM client private key matching [`Self::client_cert_path`].
68    pub client_key_path: Option<String>,
69}
70
71impl AmqpTlsConfig {
72    /// Whether any TLS material is configured at all.
73    pub fn is_empty(&self) -> bool {
74        self.ca_cert_path.is_none()
75            && self.client_cert_path.is_none()
76            && self.client_key_path.is_none()
77    }
78
79    /// Validate the combination, without touching the filesystem.
80    ///
81    /// The one rule: a client certificate and its key travel together. Half a
82    /// client identity is never what an operator meant, and connecting anyway
83    /// would silently drop the mutual half of mutual TLS.
84    pub fn validate(&self) -> Result<(), AxiamError> {
85        match (&self.client_cert_path, &self.client_key_path) {
86            (Some(_), None) => Err(AxiamError::ServiceUnavailable(
87                "AXIAM__AMQP__TLS__CLIENT_CERT_PATH is set without \
88                 AXIAM__AMQP__TLS__CLIENT_KEY_PATH — a client certificate without its \
89                 key cannot authenticate; set both or neither"
90                    .into(),
91            )),
92            (None, Some(_)) => Err(AxiamError::ServiceUnavailable(
93                "AXIAM__AMQP__TLS__CLIENT_KEY_PATH is set without \
94                 AXIAM__AMQP__TLS__CLIENT_CERT_PATH; set both or neither"
95                    .into(),
96            )),
97            _ => Ok(()),
98        }
99    }
100}

102/// Configuration for connecting to RabbitMQ.
103#[derive(Debug, Clone, Deserialize)]
104#[serde(default)]
105pub struct AmqpConfig {
106    /// AMQP connection URI. **Must** be `amqps://host:5671` (A6).
107    ///
108    /// There is no plaintext option: see
109    /// [`AmqpConfig::validate_transport_security`], which refuses every scheme
110    /// but `amqps://` in every build profile.
111    pub url: String,
112    /// TLS material for the `amqps://` connection (A6).
113    #[serde(default)]
114    pub tls: AmqpTlsConfig,
115    /// Channel prefetch count for consumers.
116    pub prefetch_count: u16,
117    /// Delay between reconnection attempts in milliseconds.
118    pub reconnect_delay_ms: u64,
119    /// Maximum number of connection retries before giving up.
120    pub max_retries: u32,
121    /// How long a single connection attempt may take before it is abandoned,
122    /// in milliseconds (A6).
123    ///
124    /// lapin has no connect timeout of its own, and the `amqps://` handshake
125    /// gave that omission teeth: a broker whose port is published but whose
126    /// TLS listener never answers — or a client-side fault in the handshake
127    /// itself — leaves `Connection::connect_with_config` pending forever.
128    /// [`AmqpManager::connect_with_retry`]'s budget cannot help there, because
129    /// the future it would retry has not resolved; the process simply stops,
130    /// with no error and no log line. Bounding each attempt turns that into a
131    /// [`AmqpError::ConnectTimeout`] the retry loop can act on and an operator
132    /// can read.
133    ///
134    /// [`AmqpManager::connect_with_retry`]: crate::AmqpManager::connect_with_retry
135    /// [`AmqpError::ConnectTimeout`]: crate::AmqpError::ConnectTimeout
136    pub connect_timeout_ms: u64,
137    /// HMAC-SHA256 master signing key for authenticating AMQP message
138    /// payloads (SEC-022/055, SECHRD-08). Set via
139    /// `AXIAM__AMQP__SIGNING_KEY` (hex-encoded key). Signing is mandatory —
140    /// there is no unsigned code path (D-05c). Use
141    /// [`AmqpConfig::resolve_signing_key`] to resolve this to the mandatory
142    /// key that consumers/producers must use: a debug-build-only dev
143    /// default is used when unset in development, but a release build
144    /// fails closed (`AxiamError::ServiceUnavailable`) if unset.
145    #[serde(default)]
146    pub signing_key: Option<String>,
147    /// Freshness skew window (seconds) for AMQP replay protection (NEW-4). An
148    /// `AuthzRequest`/`AuditEventMessage` is accepted only when its `issued_at`
149    /// lies within ±this many seconds of the consumer's clock. Set via
150    /// `AXIAM__AMQP__REPLAY_SKEW_SECS`; defaults to
151    /// [`crate::messages::DEFAULT_FRESHNESS_SKEW_SECS`] (5 minutes).
152    #[serde(default = "default_replay_skew_secs")]
153    pub replay_skew_secs: u64,
154}
155
156fn default_replay_skew_secs() -> u64 {
157    crate::messages::DEFAULT_FRESHNESS_SKEW_SECS as u64
158}
159
160impl Default for AmqpConfig {
161    fn default() -> Self {
162        Self {
163            url: "amqps://localhost:5671".into(),
164            tls: AmqpTlsConfig::default(),
165            prefetch_count: 10,
166            reconnect_delay_ms: 5000,
167            max_retries: 5,
168            // Generous enough that a loaded broker or a slow TLS handshake is
169            // never mistaken for a stall, short enough that five attempts plus
170            // their backoff stay inside any sensible startup budget.
171            connect_timeout_ms: 30_000,
172            signing_key: None,
173            replay_skew_secs: default_replay_skew_secs(),
174        }
175    }
176}
177
178impl AmqpConfig {
179    /// Resolve the configured signing key to a mandatory master key
180    /// (SECHRD-08 / D-05c).
181    ///
182    /// - If `signing_key` is set, hex-decode and return it (an invalid hex
183    ///   value is a misconfiguration and fails closed).
184    /// - If unset in a debug build (`cfg!(debug_assertions)` — dev/test),
185    ///   fall back to the documented [`DEV_DEFAULT_SIGNING_KEY`] so local
186    ///   runs work without extra setup.
187    /// - If unset in a release build (the production container image builds
188    ///   with `cargo build --release`), fail closed: return
189    ///   `AxiamError::ServiceUnavailable` — AMQP signing is mandatory in
190    ///   production, there is no unsigned/zero-key code path.
191    pub fn resolve_signing_key(&self) -> Result<Vec<u8>, AxiamError> {
192        match &self.signing_key {
193            Some(hex_key) => hex::decode(hex_key).map_err(|e| {
194                AxiamError::ServiceUnavailable(format!(
195                    "AXIAM__AMQP__SIGNING_KEY is not valid hex: {e}"
196                ))
197            }),
198            None if cfg!(debug_assertions) => {
199                tracing::warn!(
200                    "AXIAM__AMQP__SIGNING_KEY not set — using dev-only default AMQP signing key \
201                     (NOT valid in a release/production build, SECHRD-08 / D-05c)"
202                );
203                Ok(DEV_DEFAULT_SIGNING_KEY.to_vec())
204            }
205            None => Err(AxiamError::ServiceUnavailable(
206                "AMQP signing key not configured (AXIAM__AMQP__SIGNING_KEY) — mandatory in \
207                 production (SECHRD-08 / D-05c)"
208                    .to_string(),
209            )),
210        }
211    }
212
213    /// Whether [`Self::url`] names the one accepted transport (A6).
214    ///
215    /// Scheme-based, case-insensitive, and deliberately strict about the
216    /// separator: only `amqps://` is TLS, and everything else —
217    /// `amqp://` included — is not, so
218    /// [`Self::validate_transport_security`] refuses it rather than guessing.
219    /// Matching on `amqps` without the `://` would classify a hypothetical
220    /// `amqpsomething://` as TLS.
221    pub fn is_tls(&self) -> bool {
222        self.url.trim().to_ascii_lowercase().starts_with("amqps://")
223    }
224
225    /// Enforce the transport-security posture before any connection is
226    /// attempted (A6).
227    ///
228    /// Two rules, and neither has an exception:
229    ///
230    /// 1. The URL scheme **must** be `amqps://`. Every other scheme — plaintext
231    ///    `amqp://` included — is refused, in a debug build exactly as in a
232    ///    release one.
233    /// 2. TLS material that is internally inconsistent fails
234    ///    ([`AmqpTlsConfig::validate`]).
235    ///
236    /// # Why there is no plaintext escape hatch any more
237    ///
238    /// There used to be one: `AXIAM__AMQP__ALLOW_PLAINTEXT` permitted `amqp://`
239    /// in a release build, and a debug build permitted it unconditionally. Both
240    /// are gone, because the flag did what an escape hatch does. Four of the
241    /// project's own stacks reached for it — dev compose, e2e compose, the
242    /// benchmark target and CI — and every one of them recorded a locally sound
243    /// reason (throwaway data, an ephemeral CI broker, a hop the benchmark is
244    /// trying to measure). Sound reasons are what an exception collects; the net
245    /// effect was that "AMQP is TLS-only" described no deployment artifact in
246    /// the repository except the production compose file and the k8s manifests.
247    ///
248    /// The project standard is "TLS 1.3 minimum for all external
249    /// communication", and broker traffic carries authorization requests, audit
250    /// events and outbound mail payloads across service boundaries by design.
251    /// HMAC signing (§8) gives those authenticity and replay protection but
252    /// **not** confidentiality. Failing closed here — with no build profile and
253    /// no environment variable that changes the answer — is what makes the
254    /// standard true of the broker rather than aspirational about it, and it is
255    /// the same posture the SDK reactor dialers already enforce on the other end
256    /// of the same link (CONTRACT.md §8b rules 1 and 5).
257    ///
258    /// The cost is real and was accepted deliberately: every stack now needs
259    /// broker TLS material before it boots. `scripts/gen-broker-tls.sh` mints
260    /// it, `just dev-up` and `just e2e-up` call that script, and
261    /// `docs/deployment/README.md` documents bringing your own certificate
262    /// instead.
263    pub fn validate_transport_security(&self) -> Result<(), AxiamError> {
264        if !self.is_tls() {
265            return Err(AxiamError::ServiceUnavailable(format!(
266                "AXIAM__AMQP__URL must use amqps:// (port 5671) — got {:?}. AMQP is \
267                 TLS-only: broker traffic carries authorization requests, audit events \
268                 and mail payloads across service boundaries, and HMAC signing protects \
269                 their authenticity but not their confidentiality. There is no plaintext \
270                 option and no build profile in which one exists; run \
271                 `scripts/gen-broker-tls.sh` to mint broker TLS material, or point \
272                 AXIAM__AMQP__TLS__CA_CERT_PATH at your own CA bundle.",
273                self.url
274            )));
275        }
276
277        self.tls.validate()
278    }
279
280    /// Resolve the configured freshness skew to a `chrono::Duration` for the
281    /// AMQP replay-protection gate (NEW-4).
282    pub fn replay_skew(&self) -> chrono::Duration {
283        chrono::Duration::seconds(self.replay_skew_secs as i64)
284    }
285}
