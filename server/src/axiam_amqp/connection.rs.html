12/// Read the configured TLS material off disk into lapin's owned TLS config
13/// (A6).
14///
15/// Every read failure is reported with the path and the reason. A broker
16/// connection that fails because a mount is missing looks identical to one
17/// that fails because the broker is down, unless the error says which — and an
18/// operator debugging a cert mount at 3am is exactly who this message is for.
19///
20/// # No protocol-version floor is set here, and none can be (SEC-106)
21///
22/// CLAUDE.md's standard is "TLS 1.3 minimum for all external communication",
23/// and this connection crosses a service boundary by design. It is **not**
24/// pinned to 1.3, and that is a limitation of the dependency rather than a
25/// choice: lapin 4.10's only TLS-carrying entry point is
26/// `Connection::connect_with_config(uri, props, OwnedTLSConfig, runtime)`, and
27/// `OwnedTLSConfig` (`tcp-stream 0.34`) has exactly two fields — `identity`
28/// and `cert_chain`. There is no seam for a rustls `ClientConfig`, so a client
29/// -side floor would mean reimplementing `AMQPUriTcpExt::connect_with_config`
30/// against `tcp_stream::TcpStream::into_rustls` and owning lapin's handshake
31/// sequencing. rustls 0.23's default version set is TLS 1.2 + 1.3, so a broker
32/// offering only 1.2 is currently accepted.
33///
34/// **Enforce the floor on the broker**, which is where it is enforceable for
35/// this link and where it also covers every other AMQP client:
36///
37/// ```erlang
38/// %% rabbitmq.conf / advanced.config
39/// ssl_options.versions.1 = tlsv1.3
40/// ```
41///
42/// `docs/deployment/README.md` states this as a MUST-level deployment
43/// requirement. Revisit here if lapin gains a connector hook.
44fn build_tls_config(config: &AmqpConfig) -> Result<OwnedTLSConfig, AmqpError> {
45    let cert_chain = match &config.tls.ca_cert_path {
46        Some(path) => Some(std::fs::read_to_string(path).map_err(|e| {
47            AmqpError::Config(format!(
48                "AMQP TLS: cannot read CA bundle at {path:?}: {e}. The broker certificate \
49                 cannot be verified without it; connecting insecurely is not an option."
50            ))
51        })?),
52        // System roots. Correct for a publicly-issued broker certificate, and
53        // the reason a CA bundle is optional rather than required.
54        None => None,
55    };
56
57    let identity = match (&config.tls.client_cert_path, &config.tls.client_key_path) {
58        (Some(cert_path), Some(key_path)) => {
59            let pem = std::fs::read(cert_path).map_err(|e| {
60                AmqpError::Config(format!(
61                    "AMQP TLS: cannot read client certificate at {cert_path:?}: {e}"
62                ))
63            })?;
64            let key = std::fs::read(key_path).map_err(|e| {
65                AmqpError::Config(format!(
66                    "AMQP TLS: cannot read client key at {key_path:?}: {e}"
67                ))
68            })?;
69            Some(OwnedIdentity::PKCS8 { pem, key })
70        }
71        // The mismatched cases are rejected by `AmqpTlsConfig::validate`,
72        // which `validate_transport_security` runs before we get here.
73        _ => None,
74    };
75
76    Ok(OwnedTLSConfig {
77        identity,
78        cert_chain,
79    })
80}

82/// Install the process-level rustls [`CryptoProvider`] the `amqps://`
83/// handshake resolves, if nothing has installed one yet (A6 / REQ-15 AC-1).
84///
85/// # Why a library does this at all
86///
87/// rustls 0.23 refuses to pick a default provider when a process links more
88/// than one, and this workspace links two: `ring` (explicitly, for the REST
89/// listener) and `aws-lc-rs` (transitively — `rustls-platform-verifier`,
90/// `hyper-rustls`, `futures-rustls`). Any code that resolves the *default*
91/// provider then panics with "Could not automatically determine the
92/// process-level CryptoProvider from Rustls crate features".
93///
94/// `axiam-server`'s `main()` already installs `ring` for exactly this reason,
95/// and until A6 that was enough, because the only rustls consumers were
96/// server-side listeners inside that binary. The `amqps://` client is not:
97/// lapin reaches rustls through `tcp-stream` → `rustls-connector` →
98/// `ClientConfig::builder()`, which resolves the process default. So every
99/// process that dials AXIAM's broker without being `axiam-server` — every
100/// integration-test binary, and any downstream embedder of this crate — hit
101/// the panic.
102///
103/// # Why the failure was a hang rather than a crash
104///
105/// The panic happens on lapin's `lapin-io-loop` thread, not on the caller's
106/// task. The caller is awaiting a future that thread was going to resolve, and
107/// once it is gone nothing ever does: `Connection::connect_with_config` stays
108/// pending forever, [`AmqpManager::connect_with_retry`]'s budget never fires
109/// because there is no error to retry, and the process produces one panic line
110/// on a thread nobody is reading and then silence. In CI that is a job killed
111/// at its step timeout with no failing assertion to point at; the
112/// `connect_timeout_ms` bound in [`AmqpManager::connect`] exists so that shape
113/// of fault is at least reported, whatever causes it next time.
114///
115/// `ring`, to match `axiam-server` and `tls::build_rustls_server_config`.
116/// Idempotent — `Err` means someone (usually that `main()`) got there first,
117/// which is the intended outcome and not a problem.
118fn ensure_crypto_provider() {
119    use std::sync::Once;
120    static ONCE: Once = Once::new();
121    ONCE.call_once(|| {
122        if rustls::crypto::ring::default_provider()
123            .install_default()
124            .is_err()
125        {
126            tracing::debug!("rustls default CryptoProvider was already installed");
127        }
128    });
129}

131/// The `ConnectionProperties` every AXIAM AMQP connection is dialled with.
132///
133/// # Why `enable_auto_recover` rather than `default()`
134///
135/// Every long-lived consumer in `axiam-server` is supervised by the same shape
136/// of loop: on failure, call [`AmqpManager::create_channel`] on the shared
137/// manager and start again. The cache-invalidation consumer (§4.2), the
138/// webhook consumer (CORR-03) and the X1 reactor transport's reply session all
139/// do exactly this, and every one of them rebuilds a **channel** on an
140/// existing **connection**.
141///
142/// That is the right recovery for the failure those loops were written for — a
143/// channel-level exception closes one channel and leaves the connection up. It
144/// does nothing at all for a broker restart. Under
145/// `ConnectionProperties::default()` lapin does **not** reconnect (auto
146/// recovery is off) and its TCP reconnect backoff is *zero retries*, so the
147/// `Connection` stays dead, `create_channel` fails forever, and every
148/// supervisor in the process spins until someone restarts it. Measured, not
149/// argued: `tests/amqp_recovery_test.rs` polled a shared manager for 60 s
150/// after a `docker restart` and it never became usable again.
151///
152/// The symptom that produces is the expensive kind, because the server keeps
153/// serving. HTTP stays up, the decision cache goes permanently UNTRUSTED,
154/// webhooks stop, and — since R2.4 — the reactor transport reports itself down
155/// forever, which makes every `fail_closed` registration deny every login for
156/// its tenant. Nothing in that picture points at the broker having bounced
157/// hours earlier.
158///
159/// `enable_auto_recover` makes the IO loop reconnect transparently and replay
160/// the exchange/queue/binding/consumer topology on the new connection, and
161/// raises the TCP backoff to 16 attempts. The existing supervisors are
162/// unaffected and still correct: they keep handling channel-level faults, and
163/// they now do so on a connection that can actually come back.
164///
165/// Recovery applies to *recoverable* errors only — a deliberate
166/// client-side `Connection::close` is still final, which is what
167/// `reactor_containerized_test.rs`'s teardown test relies on.
168fn connection_properties() -> ConnectionProperties {
169    ConnectionProperties::default().enable_auto_recover()
170}

237/// Manages a RabbitMQ connection and channel.
238pub struct AmqpManager {
239    connection: Connection,
240    channel: Channel,
241    prefetch_count: u16,
242}
243
244impl AmqpManager {
245    /// Establish a connection to RabbitMQ and create a channel.
246    ///
247    /// # Transport security (A6)
248    ///
249    /// AMQP is TLS-only, and the posture is enforced **before** any socket is
250    /// opened ([`AmqpConfig::validate_transport_security`]): any scheme but
251    /// `amqps://`, or half a client identity, fails here rather than at connect
252    /// time with a lapin error nobody can act on.
253    ///
254    /// There is exactly one connect path below, and that is deliberate. While
255    /// there were two, the plaintext one was reachable from configuration —
256    /// which is how four of this project's own stacks came to use it. An
257    /// `amqps://` URL whose CA bundle is missing, whose certificate is expired,
258    /// or whose hostname does not match returns an error; there is no longer a
259    /// second path for it to fall back to even in principle.
260    pub async fn connect(config: &AmqpConfig) -> Result<Self, AmqpError> {
261        config
262            .validate_transport_security()
263            .map_err(|e| AmqpError::Config(e.to_string()))?;
264
265        info!(tls = true, "Connecting to RabbitMQ");
266
267        // Before anything touches rustls — see `ensure_crypto_provider`.
268        ensure_crypto_provider();
269
270        let tls_config = build_tls_config(config)?;
271        // `connect_with_config` is the only entry point that takes TLS
272        // material, and it also requires an explicit runtime — so resolve
273        // the same default runtime `Connection::connect` would have picked
274        // rather than introducing a second runtime story.
275        let runtime = lapin::runtime::default_runtime().map_err(AmqpError::Connection)?;
276        // Bounded, because lapin's is not: see `AmqpConfig::connect_timeout_ms`
277        // for what an unbounded dial costs.
278        let connection = tokio::time::timeout(
279            std::time::Duration::from_millis(config.connect_timeout_ms),
280            Connection::connect_with_config(
281                &config.url,
282                connection_properties(),
283                tls_config,
284                runtime,
285            ),
286        )
287        .await
288        .map_err(|_| AmqpError::ConnectTimeout {
289            url: config.url.clone(),
290            timeout_ms: config.connect_timeout_ms,
291        })?
292        .map_err(AmqpError::Connection)?;
293
294        let channel = connection
295            .create_channel()
296            .await
297            .map_err(AmqpError::Channel)?;
298
299        channel
300            .basic_qos(config.prefetch_count, BasicQosOptions::default())
301            .await
302            .map_err(AmqpError::Channel)?;
303
304        info!("Successfully connected to RabbitMQ");
305
306        Ok(Self {
307            connection,
308            channel,
309            prefetch_count: config.prefetch_count,
310        })
311    }

313    /// Connect with automatic retry on failure.
314    ///
315    /// Always attempts at least once. `max_retries` controls how many
316    /// additional attempts are made after the first failure.
317    pub async fn connect_with_retry(config: &AmqpConfig) -> Result<Self, AmqpError> {
318        let total_attempts = config.max_retries.saturating_add(1);
319        for attempt in 1..=total_attempts {
320            match Self::connect(config).await {
321                Ok(manager) => return Ok(manager),
322                // A6: a configuration fault is not a transient one. Retrying a
323                // missing CA mount or a plaintext `amqp://` URL
324                // `max_retries` times only delays the same error by
325                // `max_retries × reconnect_delay_ms` and buries the actionable
326                // message under identical warnings.
327                Err(e @ AmqpError::Config(_)) => {
328                    tracing::error!(error = %e, "RabbitMQ connection refused before dialling");
329                    return Err(e);
330                }
331                Err(e) => {
332                    if attempt == total_attempts {
333                        // A timeout carries no lapin error to wrap — there is
334                        // none, which is exactly what it reports — so it is
335                        // returned as itself rather than flattened into
336                        // `MaxRetriesExhausted`, whose message would then have
337                        // to invent a cause.
338                        if let AmqpError::ConnectTimeout { .. } = e {
339                            tracing::error!(
340                                error = %e,
341                                attempts = total_attempts,
342                                "Failed to connect to RabbitMQ after all retries"
343                            );
344                            return Err(e);
345                        }
346                        let lapin_err = e.into_lapin_error();
347                        tracing::error!(
348                            error = %lapin_err,
349                            attempts = total_attempts,
350                            "Failed to connect to RabbitMQ after all retries"
351                        );
352                        return Err(AmqpError::MaxRetriesExhausted(lapin_err));
353                    }
354                    warn!(
355                        error = %e,
356                        attempt,
357                        max_retries = config.max_retries,
358                        delay_ms = config.reconnect_delay_ms,
359                        "RabbitMQ connection failed, retrying"
360                    );
361                    tokio::time::sleep(std::time::Duration::from_millis(config.reconnect_delay_ms))
362                        .await;
363                }
364            }
365        }
366        unreachable!("loop always returns")
367    }

369    /// Declare all AXIAM queues as durable.
370    ///
371    /// DLQ queues are declared first (plain durable, no DLQ args of their own),
372    /// then the primary queues are declared with `x-dead-letter-exchange`
373    /// pointing at their corresponding DLQs so that exhausted/rejected messages
374    /// dead-letter rather than being silently dropped or hot-looping (CQ-B05).
375    ///
376    /// Queues with dead-letter routing:
377    /// - `AUDIT_EVENTS`  → `AUDIT_EVENTS_DLQ`   (CQ-B05 / REQ-14 AC-5)
378    /// - `AUTHZ_REQUEST` → `AUTHZ_REQUEST_DLQ`   (CQ-B05 / REQ-14 AC-5)
379    /// - `MAIL_OUTBOUND` → `MAIL_OUTBOUND_DLQ`   (D-14)
380    pub async fn declare_queues(&self) -> Result<(), AmqpError> {
381        let options = QueueDeclareOptions {
382            durable: true,
383            ..QueueDeclareOptions::default()
384        };
385
386        // Declare plain-durable queues (DLQs and non-primary queues) first.
387        for queue in ALL_QUEUES {
388            self.channel
389                .queue_declare((*queue).into(), options, FieldTable::default())
390                .await
391                .map_err(AmqpError::Declaration)?;
392            info!(queue, "Declared queue");
393        }
394
395        // Declare AUDIT_EVENTS with dead-letter routing (CQ-B05).
396        let mut audit_args = FieldTable::default();
397        audit_args.insert(
398            "x-dead-letter-exchange".into(),
399            lapin::types::AMQPValue::LongString(queues::AUDIT_EVENTS_DLQ.into()),
400        );
401        self.channel
402            .queue_declare(queues::AUDIT_EVENTS.into(), options, audit_args)
403            .await
404            .map_err(AmqpError::Declaration)?;
405        info!(
406            queue = queues::AUDIT_EVENTS,
407            "Declared queue (with dead-letter routing)"
408        );
409
410        // Declare AUTHZ_REQUEST with dead-letter routing (CQ-B05).
411        let mut authz_args = FieldTable::default();
412        authz_args.insert(
413            "x-dead-letter-exchange".into(),
414            lapin::types::AMQPValue::LongString(queues::AUTHZ_REQUEST_DLQ.into()),
415        );
416        self.channel
417            .queue_declare(queues::AUTHZ_REQUEST.into(), options, authz_args)
418            .await
419            .map_err(AmqpError::Declaration)?;
420        info!(
421            queue = queues::AUTHZ_REQUEST,
422            "Declared queue (with dead-letter routing)"
423        );
424
425        // Declare MAIL_OUTBOUND with dead-letter routing (D-14).
426        let mut mail_args = FieldTable::default();
427        mail_args.insert(
428            "x-dead-letter-exchange".into(),
429            lapin::types::AMQPValue::LongString(queues::MAIL_OUTBOUND_DLQ.into()),
430        );
431        self.channel
432            .queue_declare(queues::MAIL_OUTBOUND.into(), options, mail_args)
433            .await
434            .map_err(AmqpError::Declaration)?;
435        info!(
436            queue = queues::MAIL_OUTBOUND,
437            "Declared queue (with dead-letter routing)"
438        );
439
440        Ok(())
441    }

443    /// Declare the durable webhook AMQP topology (primary + retry + DLQ,
444    /// CORR-03/D-07).
445    ///
446    /// **Correct DLX form (Pitfall 4):** unlike [`Self::declare_queues`]'s
447    /// existing `AUDIT_EVENTS`/`AUTHZ_REQUEST`/`MAIL_OUTBOUND` wiring above
448    /// (which sets `x-dead-letter-exchange` to a queue NAME with no matching
449    /// `exchange_declare` anywhere in this crate — RabbitMQ silently drops
450    /// dead-lettered messages in that case), this topology uses the default
451    /// (nameless, `""`) exchange plus an explicit `x-dead-letter-routing-key`
452    /// for both dead-letter hops. The default exchange's implicit
453    /// per-queue-name routing makes this the well-known-correct,
454    /// minimal-surface form — never a bare undeclared exchange name.
455    ///
456    /// Declaration order (DLQ target first, same discipline as
457    /// [`Self::declare_queues`]'s `ALL_QUEUES` convention):
458    /// 1. [`queues::WEBHOOK_DLQ`] — terminal, plain durable, no DLX of its own.
459    /// 2. [`queues::WEBHOOK`] — primary; a terminal nack (requeue=false,
460    ///    attempts exhausted per `AXIAM__WEBHOOK__MAX_ATTEMPTS`) dead-letters
461    ///    to [`queues::WEBHOOK_DLQ`].
462    /// 3. [`queues::WEBHOOK_RETRY`] — no consumer ever attached; a message
463    ///    published here with a per-message TTL dead-letters back to
464    ///    [`queues::WEBHOOK`] once the TTL expires, so RabbitMQ (not an
465    ///    in-process `tokio::time::sleep`) schedules the delay
466    ///    (D-07/Pitfall 5).
467    pub async fn declare_webhook_topology(&self) -> Result<(), AmqpError> {
468        let options = QueueDeclareOptions {
469            durable: true,
470            ..QueueDeclareOptions::default()
471        };
472
473        // 1. Terminal DLQ — plain durable, no DLX args of its own.
474        self.channel
475            .queue_declare(queues::WEBHOOK_DLQ.into(), options, FieldTable::default())
476            .await
477            .map_err(AmqpError::Declaration)?;
478        info!(queue = queues::WEBHOOK_DLQ, "Declared queue");
479
480        // 2. Primary webhook queue: terminal-exhaustion nacks dead-letter to
481        // WEBHOOK_DLQ via the default exchange + explicit routing key.
482        let mut webhook_args = FieldTable::default();
483        webhook_args.insert(
484            "x-dead-letter-exchange".into(),
485            lapin::types::AMQPValue::LongString("".into()),
486        );
487        webhook_args.insert(
488            "x-dead-letter-routing-key".into(),
489            lapin::types::AMQPValue::LongString(queues::WEBHOOK_DLQ.into()),
490        );
491        self.channel
492            .queue_declare(queues::WEBHOOK.into(), options, webhook_args)
493            .await
494            .map_err(AmqpError::Declaration)?;
495        info!(
496            queue = queues::WEBHOOK,
497            "Declared queue (with dead-letter routing to WEBHOOK_DLQ)"
498        );
499
500        // 3. Retry queue: per-message TTL (set at publish time) dead-letters
501        // back to the primary WEBHOOK queue via the default exchange — no
502        // consumer attached here, no in-process sleep holding a slot.
503        let mut retry_args = FieldTable::default();
504        retry_args.insert(
505            "x-dead-letter-exchange".into(),
506            lapin::types::AMQPValue::LongString("".into()),
507        );
508        retry_args.insert(
509            "x-dead-letter-routing-key".into(),
510            lapin::types::AMQPValue::LongString(queues::WEBHOOK.into()),
511        );
512        self.channel
513            .queue_declare(queues::WEBHOOK_RETRY.into(), options, retry_args)
514            .await
515            .map_err(AmqpError::Declaration)?;
516        info!(
517            queue = queues::WEBHOOK_RETRY,
518            "Declared queue (with dead-letter routing back to WEBHOOK)"
519        );
520
521        Ok(())
522    }

562/// Lets [`crate::cache_invalidation::CacheInvalidationPublisher`] reopen its
