12/// Read the configured TLS material off disk into lapin's owned TLS config
13/// (A6).
14///
15/// Every read failure is reported with the path and the reason. A broker
16/// connection that fails because a mount is missing looks identical to one
17/// that fails because the broker is down, unless the error says which — and an
18/// operator debugging a cert mount at 3am is exactly who this message is for.
19///
20/// # No protocol-version floor is set here, and none can be (SEC-106)
21///
22/// CLAUDE.md's standard is "TLS 1.3 minimum for all external communication",
23/// and this connection crosses a service boundary by design. It is **not**
24/// pinned to 1.3, and that is a limitation of the dependency rather than a
25/// choice: lapin 4.10's only TLS-carrying entry point is
26/// `Connection::connect_with_config(uri, props, OwnedTLSConfig, runtime)`, and
27/// `OwnedTLSConfig` (`tcp-stream 0.34`) has exactly two fields — `identity`
28/// and `cert_chain`. There is no seam for a rustls `ClientConfig`, so a client
29/// -side floor would mean reimplementing `AMQPUriTcpExt::connect_with_config`
30/// against `tcp_stream::TcpStream::into_rustls` and owning lapin's handshake
31/// sequencing. rustls 0.23's default version set is TLS 1.2 + 1.3, so a broker
32/// offering only 1.2 is currently accepted.
33///
34/// **Enforce the floor on the broker**, which is where it is enforceable for
35/// this link and where it also covers every other AMQP client:
36///
37/// ```erlang
38/// %% rabbitmq.conf / advanced.config
39/// ssl_options.versions.1 = tlsv1.3
40/// ```
41///
42/// `docs/deployment/README.md` states this as a MUST-level deployment
43/// requirement. Revisit here if lapin gains a connector hook.
44fn build_tls_config(config: &AmqpConfig) -> Result<OwnedTLSConfig, AmqpError> {
45    let cert_chain = match &config.tls.ca_cert_path {
46        Some(path) => Some(std::fs::read_to_string(path).map_err(|e| {
47            AmqpError::Config(format!(
48                "AMQP TLS: cannot read CA bundle at {path:?}: {e}. The broker certificate \
49                 cannot be verified without it; connecting insecurely is not an option."
50            ))
51        })?),
52        // System roots. Correct for a publicly-issued broker certificate, and
53        // the reason a CA bundle is optional rather than required.
54        None => None,
55    };
56
57    let identity = match (&config.tls.client_cert_path, &config.tls.client_key_path) {
58        (Some(cert_path), Some(key_path)) => {
59            let pem = std::fs::read(cert_path).map_err(|e| {
60                AmqpError::Config(format!(
61                    "AMQP TLS: cannot read client certificate at {cert_path:?}: {e}"
62                ))
63            })?;
64            let key = std::fs::read(key_path).map_err(|e| {
65                AmqpError::Config(format!(
66                    "AMQP TLS: cannot read client key at {key_path:?}: {e}"
67                ))
68            })?;
69            Some(OwnedIdentity::PKCS8 { pem, key })
70        }
71        // The mismatched cases are rejected by `AmqpTlsConfig::validate`,
72        // which `validate_transport_security` runs before we get here.
73        _ => None,
74    };
75
76    Ok(OwnedTLSConfig {
77        identity,
78        cert_chain,
79    })
80}

83pub mod queues {
84    /// Inbound async authorization check requests.
85    pub const AUTHZ_REQUEST: &str = "axiam.authz.request";
86    /// Outbound authorization decision responses.
87    pub const AUTHZ_RESPONSE: &str = "axiam.authz.response";
88    /// Dead-letter queue for [`AUTHZ_REQUEST`] poison messages (CQ-B05).
89    pub const AUTHZ_REQUEST_DLQ: &str = "axiam.authz.request.dlq";
90    /// Inbound audit events from external services.
91    pub const AUDIT_EVENTS: &str = "axiam.audit.events";
92    /// Dead-letter queue for [`AUDIT_EVENTS`] poison messages (CQ-B05).
93    pub const AUDIT_EVENTS_DLQ: &str = "axiam.audit.events.dlq";
94    /// Outbound real-time event notifications.
95    pub const NOTIFICATIONS: &str = "axiam.notifications";
96    /// Outbound async mail delivery queue (D-14).
97    ///
98    /// Messages dead-letter to [`MAIL_OUTBOUND_DLQ`] when exhausted.
99    pub const MAIL_OUTBOUND: &str = "axiam.mail.outbound";
100    /// Dead-letter queue for [`MAIL_OUTBOUND`] exhausted-retry messages (D-14).
101    pub const MAIL_OUTBOUND_DLQ: &str = "axiam.mail.outbound.dlq";
102    /// Primary webhook delivery queue (CORR-03/D-07).
103    ///
104    /// A terminal nack (requeue=false, attempts exhausted) dead-letters to
105    /// [`WEBHOOK_DLQ`]. Declared and consumed via [`AmqpManager::declare_webhook_topology`].
106    pub const WEBHOOK: &str = "axiam.webhook";
107    /// Webhook retry-delay queue (CORR-03/D-07). No consumer is ever
108    /// attached to this queue — a message published here with a per-message
109    /// TTL (`x-message-ttl`/`BasicProperties::with_expiration`) dead-letters
110    /// back to [`WEBHOOK`] via the default exchange once the TTL expires,
111    /// giving RabbitMQ-native delayed retry without an in-process sleep
112    /// tying up a consumer slot (D-07/Pitfall 5).
113    pub const WEBHOOK_RETRY: &str = "axiam.webhook.retry";
114    /// Terminal dead-letter queue for exhausted webhook deliveries
115    /// (CORR-03/D-07). Messages here are real and replayable — not silently
116    /// dropped (see the DLX-wiring note on [`WEBHOOK`]/[`WEBHOOK_RETRY`]).
117    pub const WEBHOOK_DLQ: &str = "axiam.webhook.dlq";
118}

133/// Queues declared via the plain durable loop (no special arguments).

135/// These are declared first so that the DLQ targets already exist when the
136/// primary queues with `x-dead-letter-exchange` are declared below.
137const ALL_QUEUES: &[&str] = &[
138    queues::AUTHZ_RESPONSE,
139    queues::NOTIFICATIONS,
140    // DLQs are plain-durable (no DLQ args of their own).

147/// Manages a RabbitMQ connection and channel.
148pub struct AmqpManager {
149    connection: Connection,
150    channel: Channel,
151    prefetch_count: u16,
152}
153
154impl AmqpManager {
155    /// Establish a connection to RabbitMQ and create a channel.
156    ///
157    /// # Transport security (A6)
158    ///
159    /// The URL scheme selects the transport, and the posture is enforced
160    /// **before** any socket is opened
161    /// ([`AmqpConfig::validate_transport_security`]): an unrecognised scheme,
162    /// half a client identity, or a plaintext URL in a release build without
163    /// `AXIAM__AMQP__ALLOW_PLAINTEXT` all fail here rather than at connect
164    /// time with a lapin error nobody can act on.
165    ///
166    /// There is **no silent plaintext fallback**. An `amqps://` URL whose CA
167    /// bundle is missing, whose certificate is expired, or whose hostname does
168    /// not match returns an error; it does not retry without TLS. That is the
169    /// entire point — a fallback would make the failure invisible exactly when
170    /// it matters.
171    pub async fn connect(config: &AmqpConfig) -> Result<Self, AmqpError> {
172        config
173            .validate_transport_security()
174            .map_err(|e| AmqpError::Config(e.to_string()))?;
175
176        let tls = config.is_tls();
177        info!(tls, "Connecting to RabbitMQ");
178
179        let connection = if tls {
180            let tls_config = build_tls_config(config)?;
181            // `connect_with_config` is the only entry point that takes TLS
182            // material, and it also requires an explicit runtime — so resolve
183            // the same default runtime `Connection::connect` would have picked
184            // rather than introducing a second runtime story for the TLS path.
185            let runtime = lapin::runtime::default_runtime().map_err(AmqpError::Connection)?;
186            Connection::connect_with_config(
187                &config.url,
188                ConnectionProperties::default(),
189                tls_config,
190                runtime,
191            )
192            .await
193            .map_err(AmqpError::Connection)?
194        } else {
195            Connection::connect(&config.url, ConnectionProperties::default())
196                .await
197                .map_err(AmqpError::Connection)?
198        };
199
200        let channel = connection
201            .create_channel()
202            .await
203            .map_err(AmqpError::Channel)?;
204
205        channel
206            .basic_qos(config.prefetch_count, BasicQosOptions::default())
207            .await
208            .map_err(AmqpError::Channel)?;
209
210        info!("Successfully connected to RabbitMQ");
211
212        Ok(Self {
213            connection,
214            channel,
215            prefetch_count: config.prefetch_count,
216        })
217    }

219    /// Connect with automatic retry on failure.
220    ///
221    /// Always attempts at least once. `max_retries` controls how many
222    /// additional attempts are made after the first failure.
223    pub async fn connect_with_retry(config: &AmqpConfig) -> Result<Self, AmqpError> {
224        let total_attempts = config.max_retries.saturating_add(1);
225        for attempt in 1..=total_attempts {
226            match Self::connect(config).await {
227                Ok(manager) => return Ok(manager),
228                // A6: a configuration fault is not a transient one. Retrying a
229                // missing CA mount or an unset AXIAM__AMQP__ALLOW_PLAINTEXT
230                // `max_retries` times only delays the same error by
231                // `max_retries × reconnect_delay_ms` and buries the actionable
232                // message under identical warnings.
233                Err(e @ AmqpError::Config(_)) => {
234                    tracing::error!(error = %e, "RabbitMQ connection refused before dialling");
235                    return Err(e);
236                }
237                Err(e) => {
238                    if attempt == total_attempts {
239                        let lapin_err = e.into_lapin_error();
240                        tracing::error!(
241                            error = %lapin_err,
242                            attempts = total_attempts,
243                            "Failed to connect to RabbitMQ after all retries"
244                        );
245                        return Err(AmqpError::MaxRetriesExhausted(lapin_err));
246                    }
247                    warn!(
248                        error = %e,
249                        attempt,
250                        max_retries = config.max_retries,
251                        delay_ms = config.reconnect_delay_ms,
252                        "RabbitMQ connection failed, retrying"
253                    );
254                    tokio::time::sleep(std::time::Duration::from_millis(config.reconnect_delay_ms))
255                        .await;
256                }
257            }
258        }
259        unreachable!("loop always returns")
260    }

262    /// Declare all AXIAM queues as durable.
263    ///
264    /// DLQ queues are declared first (plain durable, no DLQ args of their own),
265    /// then the primary queues are declared with `x-dead-letter-exchange`
266    /// pointing at their corresponding DLQs so that exhausted/rejected messages
267    /// dead-letter rather than being silently dropped or hot-looping (CQ-B05).
268    ///
269    /// Queues with dead-letter routing:
270    /// - `AUDIT_EVENTS`  → `AUDIT_EVENTS_DLQ`   (CQ-B05 / REQ-14 AC-5)
271    /// - `AUTHZ_REQUEST` → `AUTHZ_REQUEST_DLQ`   (CQ-B05 / REQ-14 AC-5)
272    /// - `MAIL_OUTBOUND` → `MAIL_OUTBOUND_DLQ`   (D-14)
273    pub async fn declare_queues(&self) -> Result<(), AmqpError> {
274        let options = QueueDeclareOptions {
275            durable: true,
276            ..QueueDeclareOptions::default()
277        };
278
279        // Declare plain-durable queues (DLQs and non-primary queues) first.
280        for queue in ALL_QUEUES {
281            self.channel
282                .queue_declare((*queue).into(), options, FieldTable::default())
283                .await
284                .map_err(AmqpError::Declaration)?;
285            info!(queue, "Declared queue");
286        }
287
288        // Declare AUDIT_EVENTS with dead-letter routing (CQ-B05).
289        let mut audit_args = FieldTable::default();
290        audit_args.insert(
291            "x-dead-letter-exchange".into(),
292            lapin::types::AMQPValue::LongString(queues::AUDIT_EVENTS_DLQ.into()),
293        );
294        self.channel
295            .queue_declare(queues::AUDIT_EVENTS.into(), options, audit_args)
296            .await
297            .map_err(AmqpError::Declaration)?;
298        info!(
299            queue = queues::AUDIT_EVENTS,
300            "Declared queue (with dead-letter routing)"
301        );
302
303        // Declare AUTHZ_REQUEST with dead-letter routing (CQ-B05).
304        let mut authz_args = FieldTable::default();
305        authz_args.insert(
306            "x-dead-letter-exchange".into(),
307            lapin::types::AMQPValue::LongString(queues::AUTHZ_REQUEST_DLQ.into()),
308        );
309        self.channel
310            .queue_declare(queues::AUTHZ_REQUEST.into(), options, authz_args)
311            .await
312            .map_err(AmqpError::Declaration)?;
313        info!(
314            queue = queues::AUTHZ_REQUEST,
315            "Declared queue (with dead-letter routing)"
316        );
317
318        // Declare MAIL_OUTBOUND with dead-letter routing (D-14).
319        let mut mail_args = FieldTable::default();
320        mail_args.insert(
321            "x-dead-letter-exchange".into(),
322            lapin::types::AMQPValue::LongString(queues::MAIL_OUTBOUND_DLQ.into()),
323        );
324        self.channel
325            .queue_declare(queues::MAIL_OUTBOUND.into(), options, mail_args)
326            .await
327            .map_err(AmqpError::Declaration)?;
328        info!(
329            queue = queues::MAIL_OUTBOUND,
330            "Declared queue (with dead-letter routing)"
331        );
332
333        Ok(())
334    }

336    /// Declare the durable webhook AMQP topology (primary + retry + DLQ,
337    /// CORR-03/D-07).
338    ///
339    /// **Correct DLX form (Pitfall 4):** unlike [`Self::declare_queues`]'s
340    /// existing `AUDIT_EVENTS`/`AUTHZ_REQUEST`/`MAIL_OUTBOUND` wiring above
341    /// (which sets `x-dead-letter-exchange` to a queue NAME with no matching
342    /// `exchange_declare` anywhere in this crate — RabbitMQ silently drops
343    /// dead-lettered messages in that case), this topology uses the default
344    /// (nameless, `""`) exchange plus an explicit `x-dead-letter-routing-key`
345    /// for both dead-letter hops. The default exchange's implicit
346    /// per-queue-name routing makes this the well-known-correct,
347    /// minimal-surface form — never a bare undeclared exchange name.
348    ///
349    /// Declaration order (DLQ target first, same discipline as
350    /// [`Self::declare_queues`]'s `ALL_QUEUES` convention):
351    /// 1. [`queues::WEBHOOK_DLQ`] — terminal, plain durable, no DLX of its own.
352    /// 2. [`queues::WEBHOOK`] — primary; a terminal nack (requeue=false,
353    ///    attempts exhausted per `AXIAM__WEBHOOK__MAX_ATTEMPTS`) dead-letters
354    ///    to [`queues::WEBHOOK_DLQ`].
355    /// 3. [`queues::WEBHOOK_RETRY`] — no consumer ever attached; a message
356    ///    published here with a per-message TTL dead-letters back to
357    ///    [`queues::WEBHOOK`] once the TTL expires, so RabbitMQ (not an
358    ///    in-process `tokio::time::sleep`) schedules the delay
359    ///    (D-07/Pitfall 5).
360    pub async fn declare_webhook_topology(&self) -> Result<(), AmqpError> {
361        let options = QueueDeclareOptions {
362            durable: true,
363            ..QueueDeclareOptions::default()
364        };
365
366        // 1. Terminal DLQ — plain durable, no DLX args of its own.
367        self.channel
368            .queue_declare(queues::WEBHOOK_DLQ.into(), options, FieldTable::default())
369            .await
370            .map_err(AmqpError::Declaration)?;
371        info!(queue = queues::WEBHOOK_DLQ, "Declared queue");
372
373        // 2. Primary webhook queue: terminal-exhaustion nacks dead-letter to
374        // WEBHOOK_DLQ via the default exchange + explicit routing key.
375        let mut webhook_args = FieldTable::default();
376        webhook_args.insert(
377            "x-dead-letter-exchange".into(),
378            lapin::types::AMQPValue::LongString("".into()),
379        );
380        webhook_args.insert(
381            "x-dead-letter-routing-key".into(),
382            lapin::types::AMQPValue::LongString(queues::WEBHOOK_DLQ.into()),
383        );
384        self.channel
385            .queue_declare(queues::WEBHOOK.into(), options, webhook_args)
386            .await
387            .map_err(AmqpError::Declaration)?;
388        info!(
389            queue = queues::WEBHOOK,
390            "Declared queue (with dead-letter routing to WEBHOOK_DLQ)"
391        );
392
393        // 3. Retry queue: per-message TTL (set at publish time) dead-letters
394        // back to the primary WEBHOOK queue via the default exchange — no
395        // consumer attached here, no in-process sleep holding a slot.
396        let mut retry_args = FieldTable::default();
397        retry_args.insert(
398            "x-dead-letter-exchange".into(),
399            lapin::types::AMQPValue::LongString("".into()),
400        );
401        retry_args.insert(
402            "x-dead-letter-routing-key".into(),
403            lapin::types::AMQPValue::LongString(queues::WEBHOOK.into()),
404        );
405        self.channel
406            .queue_declare(queues::WEBHOOK_RETRY.into(), options, retry_args)
407            .await
408            .map_err(AmqpError::Declaration)?;
409        info!(
410            queue = queues::WEBHOOK_RETRY,
411            "Declared queue (with dead-letter routing back to WEBHOOK)"
412        );
413
414        Ok(())
415    }

422    /// Create a new channel on the existing connection with QoS applied.
423    pub async fn create_channel(&self) -> Result<Channel, AmqpError> {
424        let channel = self
425            .connection
426            .create_channel()
427            .await
428            .map_err(AmqpError::Channel)?;
429        channel
430            .basic_qos(self.prefetch_count, BasicQosOptions::default())
431            .await
432            .map_err(AmqpError::Channel)?;
433        Ok(channel)
434    }

436    /// Create a new channel with QoS and publisher confirms enabled.
437    ///
438    /// Use this for channels that will publish messages and need broker
439    /// acknowledgement (`Confirmation::Ack`/`Nack` instead of `NotRequested`).
440    pub async fn create_publisher_channel(&self) -> Result<Channel, AmqpError> {
441        let channel = self.create_channel().await?;
442        channel
443            .confirm_select(ConfirmSelectOptions::default())
444            .await
445            .map_err(AmqpError::Channel)?;
446        Ok(channel)
447    }
