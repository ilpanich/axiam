1//! The signed envelope of the server-internal queues (SA-14, N3).
2//!
3//! `axiam.mail.outbound` and the four outbound kinds' queues (`axiam.webhook`,
4//! `axiam.ssf_push`, `axiam.scim_push`, `axiam.ciba_ping` and their retry
5//! queues) are written and read only by AXIAM itself, yet what their consumers
6//! do with a message is the server's most trusted output: tenant-branded mail
7//! through the tenant's provider, webhook POSTs signed with the webhook's
8//! secret, Security Event Tokens signed with the deployment key. Until SA-14
9//! they decoded plain JSON, so broker write access to a queue was control-plane
10//! authority.
11//!
12//! Every message on those queues now travels inside an [`InternalEnvelope`]
13//! carrying the same §8 v2 protections as the authorization and audit queues —
14//! an HMAC-SHA256 over the canonical JSON, a `key_version`, a per-message
15//! `nonce` recorded in the durable replay store, and an `issued_at` freshness
16//! gate — with one difference that is the point of this module: the key is
17//! **server-only**. It is derived from the master key under a distinct HKDF
18//! label ([`derive_internal_key`]), not from a tenant's subkey, so the subkey
19//! a tenant's own producers hold for the authz and audit queues cannot sign a
20//! message here. The `purpose` (one per queue family) is mixed into the
21//! derivation as well, so a signed mail message cannot be replayed onto a
22//! webhook queue or the reverse.
23//!
24//! The body inside the envelope is exactly the JSON each queue carried before
25//! (`OutboundMailMessage`, `WebhookMessage`, `OutboundMessage`), so the
26//! envelope adds authentication without redefining any message.
27//!
28//! Three things differ from the SDK-facing queues because these are durable
29//! **work** queues, where a message waits for reasons that are not tampering:
30//!
31//! * **Age, not skew.** A message is admitted while `issued_at` is at most
32//!   `max_age` (`AXIAM__AMQP__INTERNAL_QUEUES__MAX_AGE_SECS`, 24 h by
33//!   default) in the past — a consumer outage, a backlog or head-of-line
34//!   blocking in a retry queue must not dead-letter queued mail — and at most
35//!   the replay skew in the future (QR1-2, SR1-2). Its nonce is kept for
36//!   `max_age`.
37//! * **A redelivery is not a replay.** A nonce already recorded is admitted
38//!   when the broker marks the delivery redelivered — a consumer crashed
39//!   before it settled, or requeued the original on purpose (CQ-B49) — a flag
40//!   only the broker sets (SR1-1). A republished copy is a replay.
41//! * **The received bytes are verified** ([`split_trailing_signature`]), not a
42//!   re-serialization of the parsed body, whose floats serde_json does not
43//!   guarantee to reprint identically (QR1-9).
44//!
45//! **Rolling upgrade.** A consumer verifies every envelope it receives. What it
46//! does with a message in the old, un-enveloped shape is
47//! `AXIAM__AMQP__INTERNAL_QUEUES__REQUIRE_SIGNATURE`: `true` (the default)
48//! rejects it; `false` accepts it with a warning, for the one deployment in
49//! which old and new replicas share the queues (see the upgrade notes in
50//! `docs/deployment/README.md`).

71/// HKDF `info` prefix of the server-only key. Distinct from the per-tenant
72/// tag (`axiam-amqp-v1`) from its twelfth byte on, so no tenant subkey
73/// derivation can produce the same `info`, and therefore the same key.
74const INTERNAL_DOMAIN_TAG: &[u8] = b"axiam-amqp-internal-v1";

76/// The `purpose` of the mail queue's envelope.
77pub const MAIL_PURPOSE: &str = "mail";

79/// The audit action of a message an internal-queue consumer refused.

82/// The tenant id the internal queues' nonces are recorded under. The nonce
83/// store is keyed by `(tenant_id, nonce)`; the internal queues are not a
84/// tenant's, so their nonces live in one namespace of their own, and the
85/// rejection rows go to the system (nil-tenant) audit log for the same reason.
86pub const INTERNAL_SCOPE: Uuid = Uuid::nil();

93/// Derive the server-only signing key of one internal queue family.
94///
95/// HKDF-SHA256 over the AMQP master key, with the same salt as the per-tenant
96/// derivation and `info = "axiam-amqp-internal-v1" || key_version || purpose`.
97pub fn derive_internal_key(master: &[u8], key_version: u8, purpose: &str) -> [u8; 32] {
98    let hk = Hkdf::<Sha256>::new(Some(APP_SALT), master);
99    let mut info = Vec::with_capacity(INTERNAL_DOMAIN_TAG.len() + 1 + purpose.len());
100    info.extend_from_slice(INTERNAL_DOMAIN_TAG);
101    info.push(key_version);
102    info.extend_from_slice(purpose.as_bytes());
103    let mut okm = [0u8; 32];
104    hk.expand(&info, &mut okm)
105        .expect("32 bytes is a valid HKDF-SHA256 output length");
106    okm
107}

109/// The bytes on an internal queue.
110///
111/// **Wire order is load-bearing**: the HMAC is over this struct serialized in
112/// declaration order with `hmac_signature` absent.
113#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
114#[serde(deny_unknown_fields)]
115pub struct InternalEnvelope {
116    /// The queue family: [`MAIL_PURPOSE`] or [`outbound_purpose`].
117    pub purpose: String,
118    /// Master-key rotation / envelope version; at least
119    /// [`MIN_ACCEPTED_KEY_VERSION`].
120    pub key_version: u8,
121    /// Per-message nonce, recorded in the durable replay store.
122    pub nonce: Uuid,
123    /// When the message becomes current. The publish time, except for an
124    /// outbound retry copy, which carries the time its retry-queue TTL expires
125    /// (see `AmqpOutboundPublisher::publish_retry`).
126    pub issued_at: DateTime<Utc>,
127    /// The queue's message, in the JSON it carried before the envelope.
128    pub body: serde_json::Value,
129    /// HMAC-SHA256 (hex) of the envelope with this field absent, under
130    /// [`derive_internal_key`].
131    #[serde(default, skip_serializing_if = "Option::is_none")]
132    pub hmac_signature: Option<String>,
133}

155    /// Wrap `body` in a signed [`InternalEnvelope`] for `purpose`, current
156    /// from `issued_at`, and return its bytes.
157    pub fn seal<T: Serialize + ?Sized>(
158        &self,
159        purpose: &str,
160        body: &T,
161        issued_at: DateTime<Utc>,
162    ) -> Result<Vec<u8>, serde_json::Error> {
163        let mut envelope = InternalEnvelope {
164            purpose: purpose.to_owned(),
165            key_version: CURRENT_KEY_VERSION,
166            nonce: Uuid::new_v4(),
167            issued_at,
168            body: serde_json::to_value(body)?,
169            hmac_signature: None,
170        };
171        let canonical = serde_json::to_vec(&envelope)?;
172        let key = derive_internal_key(&self.master, envelope.key_version, purpose);
173        envelope.hmac_signature = Some(sign_payload(&key, &canonical));
174        serde_json::to_vec(&envelope)
175    }
176}

178/// Why an internal-queue consumer refused a message.
179#[derive(Debug, Clone, Copy, PartialEq, Eq)]
180pub enum Rejection {
181    /// Not JSON, or an envelope that does not parse.
182    Malformed,
183    /// No envelope (the pre-SA-14 shape while signatures are required), or an
184    /// envelope without a signature.
185    Unsigned,
186    /// An envelope for another queue family.
187    WrongPurpose,
188    /// The signature does not verify under the server-only key, or the bytes
189    /// do not end in the signature member the server writes.
190    BadSignature,
191    /// A `key_version` below [`MIN_ACCEPTED_KEY_VERSION`].
192    KeyVersion,
193    /// `issued_at` more than the maximum age in the past, or more than the
194    /// replay skew in the future.
195    Stale,
196    /// The nonce was already consumed, and the broker did not mark this
197    /// delivery as a redelivery.
198    Replay,
199    /// The nonce could not be recorded; refused rather than risk a replay.
200    NonceStore,
201    /// A verified envelope whose body is not the queue's message.
202    MalformedBody,
203}
204
205impl Rejection {
206    /// The reason recorded in the audit row's metadata.
207    pub fn as_str(self) -> &'static str {
208        match self {
209            Self::Malformed => "malformed",
210            Self::Unsigned => "unsigned",
211            Self::WrongPurpose => "wrong_purpose",
212            Self::BadSignature => "bad_signature",
213            Self::KeyVersion => "key_version",
214            Self::Stale => "stale",
215            Self::Replay => "replay",
216            Self::NonceStore => "nonce_store_error",
217            Self::MalformedBody => "malformed_body",
218        }
219    }
220}

222/// What [`InternalQueueGate::open`] admitted.
223#[derive(Debug, Clone, PartialEq)]
224pub enum Opened {
225    /// A verified, fresh, first-seen envelope's body.
226    Verified(serde_json::Value),
227    /// A message in the pre-SA-14 shape, admitted only because
228    /// `require_signature` is `false`.
229    LegacyUnsigned(serde_json::Value),
230}
231
232impl Opened {
233    /// The message, however it was admitted.
234    pub fn into_body(self) -> serde_json::Value {
235        match self {
236            Self::Verified(body) | Self::LegacyUnsigned(body) => body,
237        }
238    }
239}

296impl InternalQueueGate {
297    /// A gate over the AMQP master key, the future-dating `skew`
298    /// (`AmqpConfig::replay_skew`), the maximum message age `max_age`
299    /// (`AmqpConfig::internal_queue_max_age`, raised to `skew` if lower) and
300    /// the durable nonce store. `require_signature` is
301    /// `amqp.internal_queues.require_signature`.
302    pub fn new<N>(
303        master: Vec<u8>,
304        require_signature: bool,
305        skew: chrono::Duration,
306        max_age: chrono::Duration,
307        nonces: N,
308    ) -> Self
309    where
310        N: AmqpNonceRepository + 'static,
311    {
312        Self {
313            master: Arc::new(master),
314            require_signature,
315            skew,
316            max_age: max_age.max(skew),
317            nonces: Arc::new(RepoNonceSink(nonces)),
318        }
319    }
320
321    /// Whether `issued_at` is current at `now`: at most `max_age` old, and at
322    /// most `skew` in the future.
323    fn is_current(&self, issued_at: DateTime<Utc>, now: DateTime<Utc>) -> bool {
324        let age = now.signed_duration_since(issued_at);
325        age <= self.max_age && -age <= self.skew
326    }
327
328    /// Whether a message in the pre-SA-14 shape is refused.
329    pub fn require_signature(&self) -> bool {
330        self.require_signature
331    }
332
333    /// Admit `raw`, delivered for the first time off a queue of family
334    /// `purpose`, at `now`. [`Self::open_delivery`] with `redelivered = false`.
335    pub async fn open(
336        &self,
337        purpose: &str,
338        raw: &[u8],
339        now: DateTime<Utc>,
340    ) -> Result<Opened, Rejection> {
341        self.open_delivery(purpose, raw, now, false).await
342    }
343
344    /// Admit `raw`, taken off a queue of family `purpose`, at `now`.
345    /// `redelivered` is the broker's flag on the delivery.
346    ///
347    /// The checks run in the order of the authz consumer's: parse, signature,
348    /// key version, age, then the nonce, so a message that fails any earlier
349    /// check never consumes a nonce. A nonce already recorded is a replay
350    /// unless `redelivered` is set: the broker sets it only on a message it
351    /// delivered before and got back unsettled (a consumer that crashed, or
352    /// one that requeued the original because its retry copy could not be
353    /// published), and a publisher cannot set it (SR1-1).
354    pub async fn open_delivery(
355        &self,
356        purpose: &str,
357        raw: &[u8],
358        now: DateTime<Utc>,
359        redelivered: bool,
360    ) -> Result<Opened, Rejection> {
361        let value: serde_json::Value =
362            serde_json::from_slice(raw).map_err(|_| Rejection::Malformed)?;
363
364        // Every envelope has a `body`; no pre-SA-14 message on these queues
365        // does (`OutboundMailMessage`, `WebhookMessage`, `OutboundMessage`).
366        if value.get("body").is_none() {
367            return if self.require_signature {
368                Err(Rejection::Unsigned)
369            } else {
370                tracing::warn!(
371                    purpose,
372                    "unsigned internal-queue message admitted: \
373                     AXIAM__AMQP__INTERNAL_QUEUES__REQUIRE_SIGNATURE=false (SA-14)"
374                );
375                Ok(Opened::LegacyUnsigned(value))
376            };
377        }
378
379        let mut envelope: InternalEnvelope =
380            serde_json::from_value(value).map_err(|_| Rejection::Malformed)?;
381        let Some(signature) = envelope.hmac_signature.take() else {
382            return Err(Rejection::Unsigned);
383        };
384        if envelope.purpose != purpose {
385            return Err(Rejection::WrongPurpose);
386        }
387        // QR1-9: the HMAC covers the bytes as received, minus the trailing
388        // signature member — never a re-serialization of `body`, whose floats
389        // serde_json does not promise to reprint digit for digit. The member
390        // stripped must be the one that parsed (no second, smuggled one).
391        let Some((canonical, received)) = split_trailing_signature(raw) else {
392            return Err(Rejection::BadSignature);
393        };
394        if received != signature {
395            return Err(Rejection::BadSignature);
396        }
397        let key = derive_internal_key(&self.master, envelope.key_version, purpose);
398        if !verify_payload(&key, &canonical, &signature) {
399            return Err(Rejection::BadSignature);
400        }
401        if envelope.key_version < MIN_ACCEPTED_KEY_VERSION {
402            return Err(Rejection::KeyVersion);
403        }
404        if !self.is_current(envelope.issued_at, now) {
405            return Err(Rejection::Stale);
406        }
407        match self
408            .nonces
409            .record(envelope.nonce, envelope.issued_at + self.max_age)
410            .await
411        {
412            Ok(()) => Ok(Opened::Verified(envelope.body)),
413            Err(AxiamError::ReplayDetected) if redelivered => {
414                tracing::info!(
415                    purpose,
416                    "redelivered internal-queue message admitted (nonce already recorded)"
417                );
418                Ok(Opened::Verified(envelope.body))
419            }
420            Err(AxiamError::ReplayDetected) => Err(Rejection::Replay),
421            Err(e) => {
422                tracing::error!(error = %e, purpose, "could not record an internal-queue nonce");
423                Err(Rejection::NonceStore)
424            }
425        }
426    }
427}
