1//! The outcome table: **the one place** that decides what a delivery attempt's
2//! result means, shared by the AMQP consume loop ([`super::consumer`]) and the
3//! in-process dispatcher ([`super::inprocess`], the minimal profile, G-8).
4//!
5//! A deliverer reports a [`DeliveryOutcome`] (or errors); [`decide`] turns that,
6//! the message and the kind's retry policy into a [`Verdict`] carrying the audit
7//! row to write. Neither transport re-implements any of it: they differ only in
8//! how they *carry out* the verdict (AMQP: ack, republish to the TTL retry
9//! queue, nack to the DLQ; in-process: drop, delayed re-dispatch, audit row
10//! only).
11//!
12//! | Outcome | Verdict | Audit record (`<slug>.` prefix) |
13//! |---|---|---|
14//! | `Delivered` | [`Verdict::Delivered`] | `delivery_succeeded` (success) |
15//! | `Retry`, or a deliverer `Err`, with attempts left | [`Verdict::Retry`] with `attempt + 1` and TTL [`backoff_ttl_ms`] | `delivery_attempt` (failure) |
16//! | `Retry` / `Err` with no attempts left | [`Verdict::DeadLetter`] | `delivery_failed` (failure) |
17//! | `DeadLetter` | [`Verdict::DeadLetter`], immediately | `delivery_failed` (failure) |
18//!
19//! The audit vocabulary (`<slug>.delivery_succeeded`, `.delivery_attempt`,
20//! `.delivery_failed`, their metadata keys and the system actor) is built here
21//! and nowhere else.
22//!
23//! A fourth action, `<slug>.delivery_abandoned` ([`abandoned_entry`]), is not an
24//! outcome of an attempt: the minimal profile's in-process dispatcher writes it
25//! for a delivery it lost without a verdict (queued or waiting for a retry when
26//! the process stopped, or refused at enqueue). It is deliberately **not**
27//! `delivery_failed`: that row is what a tenant's `scim_delivery_failed`
28//! notification rule matches, and a stop is not a downstream outage.

38/// What to do with a delivery after one attempt.
39#[derive(Debug, Clone)]
40pub(crate) enum Verdict {
41    /// Done. Acknowledge, and write the success row.
42    Delivered {
43        /// The `<slug>.delivery_succeeded` row.
44        audit: CreateAuditLogEntry,
45    },
46    /// Try again later: enqueue `next` (the message with `attempt + 1`) after
47    /// `ttl_ms`, then write the attempt row.
48    Retry {
49        /// The message to re-dispatch, `attempt` already incremented.
50        next: OutboundMessage,
51        /// The bounded exponential backoff before it re-enters the queue.
52        ttl_ms: u64,
53        /// The `<slug>.delivery_attempt` row.
54        audit: CreateAuditLogEntry,
55    },
56    /// Terminal. Write the failure row; the transport parks the message (AMQP:
57    /// the DLQ) or drops it (in-process: the row is the whole record).
58    DeadLetter {
59        /// The `<slug>.delivery_failed` row.
60        audit: CreateAuditLogEntry,
61    },
62}

64/// A deliverer that errors instead of classifying the attempt is a retry.
65pub(crate) fn classify(result: Result<DeliveryOutcome, OutboundError>) -> DeliveryOutcome {
66    match result {
67        Ok(outcome) => outcome,
68        Err(e) => DeliveryOutcome::Retry {
69            reason: e.to_string(),
70        },
71    }
72}

74/// Build a delivery audit entry. `actor_id` uses the `Uuid::nil()`
75/// system-actor convention: no human or service account initiated the attempt.
76pub(crate) fn audit_entry(
77    msg: &OutboundMessage,
78    action_suffix: &str,
79    outcome: AuditOutcome,
80    metadata: serde_json::Value,
81) -> CreateAuditLogEntry {
82    CreateAuditLogEntry {
83        tenant_id: msg.tenant_id,
84        actor_id: Uuid::nil(),
85        actor_type: ActorType::System,
86        action: format!("{}.{action_suffix}", msg.kind.as_str()),
87        resource_id: Some(msg.target_id),
88        outcome,
89        ip_address: None,
90        metadata: Some(metadata),
91    }
92}

94/// The terminal failure row, also written by a transport that has to give up on
95/// a delivery for a reason of its own (the in-process dispatcher's retry
96/// capacity). Same action, same metadata keys as an exhausted delivery.
97pub(crate) fn failed_entry(msg: &OutboundMessage, error_detail: &str) -> CreateAuditLogEntry {
98    audit_entry(
99        msg,
100        "delivery_failed",
101        AuditOutcome::Failure,
102        serde_json::json!({
103            "delivery_id": msg.delivery_id,
104            "attempt": msg.attempt + 1,
105            "error": error_detail,
106            "next_retry_in_ms": null,
107        }),
108    )
109}

111/// The terminal row of a delivery the in-process dispatcher lost without a
112/// verdict (P23W5-A4): `<slug>.delivery_abandoned`, outcome `Failure`, with one
113/// of the dispatcher's fixed reasons. `attempts_made` is how many attempts ran
114/// before the loss (`0` for a message never attempted).
115///
116/// Not `delivery_failed` on purpose: `NotificationEventType::from_audit_action`
117/// maps only `scim_push.delivery_failed`, so this row mails nobody.
118pub(crate) fn abandoned_entry(msg: &OutboundMessage, reason: &str) -> CreateAuditLogEntry {
119    audit_entry(
120        msg,
121        "delivery_abandoned",
122        AuditOutcome::Failure,
123        serde_json::json!({
124            "delivery_id": msg.delivery_id,
125            "attempts_made": msg.attempt,
126            "reason": reason,
127        }),
128    )
129}

131/// The one `<slug>.delivery_abandoned` row of the messages a bulk producer held
132/// back after the queue refused one of its run (R101W4-01): `refused_count` of
133/// them, none attempted, none with a row of its own. No `delivery_id`: the row
134/// stands for many deliveries that never had one queued.
135pub(crate) fn not_offered_entry(
136    kind: OutboundKind,
137    tenant_id: Uuid,
138    target_id: Uuid,
139    reason: &str,
140    refused_count: u64,
141) -> CreateAuditLogEntry {
142    CreateAuditLogEntry {
143        tenant_id,
144        actor_id: Uuid::nil(),
145        actor_type: ActorType::System,
146        action: format!("{}.delivery_abandoned", kind.as_str()),
147        resource_id: Some(target_id),
148        outcome: AuditOutcome::Failure,
149        ip_address: None,
150        metadata: Some(serde_json::json!({
151            "attempts_made": 0,
152            "reason": reason,
153            "refused_count": refused_count,
154        })),
155    }

158/// Decide what one attempt's `outcome` means for `msg` under `cfg`.
159pub(crate) fn decide(
160    msg: &OutboundMessage,
161    outcome: DeliveryOutcome,
162    cfg: &OutboundRetryConfig,
163) -> Verdict {
164    match outcome {
165        DeliveryOutcome::Delivered { response_status } => {
166            let mut metadata = serde_json::json!({
167                "delivery_id": msg.delivery_id,
168                "attempt": msg.attempt + 1,
169            });
170            if let (Some(status), Some(map)) = (response_status, metadata.as_object_mut()) {
171                map.insert("status".into(), serde_json::json!(status));
172            }
173            Verdict::Delivered {
174                audit: audit_entry(msg, "delivery_succeeded", AuditOutcome::Success, metadata),
175            }
176        }
177        DeliveryOutcome::Retry { reason } => decide_failure(msg, true, reason, cfg),
178        DeliveryOutcome::DeadLetter { reason } => decide_failure(msg, false, reason, cfg),
179    }
180}

182/// Retry-or-exhaust: shared by every "this attempt did not succeed" outcome.
183fn decide_failure(
184    msg: &OutboundMessage,
185    retryable: bool,
186    error_detail: String,
187    cfg: &OutboundRetryConfig,
188) -> Verdict {
189    let kind = msg.kind;
190    let next_attempt = msg.attempt + 1;
191
192    if retryable && next_attempt < cfg.max_attempts {
193        let ttl_ms = backoff_ttl_ms(next_attempt, cfg);
194        let mut next = msg.clone();
195        next.attempt = next_attempt;
196        Verdict::Retry {
197            next,
198            ttl_ms,
199            audit: audit_entry(
200                msg,
201                "delivery_attempt",
202                AuditOutcome::Failure,
203                serde_json::json!({
204                    "delivery_id": msg.delivery_id,
205                    "attempt": next_attempt,
206                    "error": error_detail,
207                    "next_retry_in_ms": ttl_ms,
208                }),
209            ),
210        }
211    } else {
212        warn!(
213            %kind,
214            target_id = %msg.target_id,
215            delivery_id = %msg.delivery_id,
216            attempt = next_attempt,
217            max_attempts = cfg.max_attempts,
218            error = %error_detail,
219            "Outbound delivery exhausted retries or cannot succeed; dead-lettering"
220        );
221        Verdict::DeadLetter {
222            audit: failed_entry(msg, &error_detail),
223        }
224    }
225}

229    //! The outcome table as data, so that **both** transports' tests assert
230    //! against the very same expectations (G-8: "the same outcome table, the
231    //! same audit vocabulary").

233    use super::*;
234    use axiam_core::outbound::OutboundKind;

236    /// The shape of a verdict, without its payload.
237    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
238    pub(crate) enum Shape {
239        Delivered,
240        Retry,
241        DeadLetter,
242    }
243
244    /// One row: an attempt result for a message at `attempt`, and what the
245    /// table says follows.
246    pub(crate) struct Case {
247        pub name: &'static str,
248        pub attempt: u32,
249        pub result: Result<DeliveryOutcome, OutboundError>,
250        pub shape: Shape,
251        pub action: &'static str,
252        pub success: bool,
253    }
254
255    /// The retry policy every table row is evaluated under.
256    pub(crate) const CFG: OutboundRetryConfig = OutboundRetryConfig {
257        max_attempts: 3,
258        backoff_base_ms: 100,
259        backoff_ceiling_ms: 10_000,
260    };
261
262    pub(crate) fn message(kind: OutboundKind, attempt: u32) -> OutboundMessage {
263        OutboundMessage {
264            kind,
265            tenant_id: Uuid::new_v4(),
266            target_id: Uuid::new_v4(),
267            delivery_id: Uuid::new_v4(),
268            event_type: "user.created".into(),
269            payload: serde_json::json!({"hello": "world"}),
270            attempt,
271        }
272    }
273
274    pub(crate) fn cases() -> Vec<Case> {
275        let retry = |reason: &str| {
276            Ok(DeliveryOutcome::Retry {
277                reason: reason.into(),
278            })
279        };
280        vec![
281            Case {
282                name: "delivered",
283                attempt: 0,
284                result: Ok(DeliveryOutcome::Delivered {
285                    response_status: Some(204),
286                }),
287                shape: Shape::Delivered,
288                action: "delivery_succeeded",
289                success: true,
290            },
291            Case {
292                name: "delivered after a retry",
293                attempt: 1,
294                result: Ok(DeliveryOutcome::Delivered {
295                    response_status: None,
296                }),
297                shape: Shape::Delivered,
298                action: "delivery_succeeded",
299                success: true,
300            },
301            Case {
302                name: "retry with attempts left",
303                attempt: 0,
304                result: retry("non-2xx status: 503"),
305                shape: Shape::Retry,
306                action: "delivery_attempt",
307                success: false,
308            },
309            Case {
310                name: "retry on the last affordable attempt",
311                attempt: CFG.max_attempts - 2,
312                result: retry("still down"),
313                shape: Shape::Retry,
314                action: "delivery_attempt",
315                success: false,
316            },
317            Case {
318                name: "retry with no attempts left",
319                attempt: CFG.max_attempts - 1,
320                result: retry("still down"),
321                shape: Shape::DeadLetter,
322                action: "delivery_failed",
323                success: false,
324            },
325            Case {
326                name: "deliverer error is a retry",
327                attempt: 0,
328                result: Err(OutboundError::Delivery("lookup failed".into())),
329                shape: Shape::Retry,
330                action: "delivery_attempt",
331                success: false,
332            },
333            Case {
334                name: "deliverer error on the last attempt",
335                attempt: CFG.max_attempts - 1,
336                result: Err(OutboundError::Delivery("boom".into())),
337                shape: Shape::DeadLetter,
338                action: "delivery_failed",
339                success: false,
340            },
341            Case {
342                name: "dead letter skips the remaining attempts",
343                attempt: 0,
344                result: Ok(DeliveryOutcome::DeadLetter {
345                    reason: "target gone".into(),
346                }),
347                shape: Shape::DeadLetter,
348                action: "delivery_failed",
349                success: false,
350            },
351        ]
352    }
353
354    /// The shape of the verdict `decide` returns.
355    pub(crate) fn shape_of(verdict: &Verdict) -> Shape {
356        match verdict {
357            Verdict::Delivered { .. } => Shape::Delivered,
358            Verdict::Retry { .. } => Shape::Retry,
359            Verdict::DeadLetter { .. } => Shape::DeadLetter,
360        }
361    }
362
363    /// The audit row inside a verdict.
364    pub(crate) fn audit_of(verdict: &Verdict) -> &CreateAuditLogEntry {
365        match verdict {
366            Verdict::Delivered { audit }
367            | Verdict::Retry { audit, .. }
368            | Verdict::DeadLetter { audit } => audit,
369        }
370    }
371}

379    #[test]
380    fn every_row_of_the_table_decides_as_documented() {
381        for case in cases() {
382            for kind in OutboundKind::ALL {
383                let msg = message(*kind, case.attempt);
384                let verdict = decide(&msg, classify(case.result.clone()), &CFG);
385                assert_eq!(shape_of(&verdict), case.shape, "{}", case.name);
386                let audit = audit_of(&verdict);
387                assert_eq!(
388                    audit.action,
389                    format!("{}.{}", kind.as_str(), case.action),
390                    "{}",
391                    case.name
392                );
393                assert_eq!(
394                    matches!(audit.outcome, AuditOutcome::Success),
395                    case.success,
396                    "{}",
397                    case.name
398                );
399                assert_eq!(audit.tenant_id, msg.tenant_id);
400                assert_eq!(audit.resource_id, Some(msg.target_id));
401                assert_eq!(audit.actor_id, Uuid::nil());
402                assert!(matches!(audit.actor_type, ActorType::System));
403                assert!(audit.ip_address.is_none());
404            }
405        }
406    }

408    #[test]
409    fn a_retry_carries_the_incremented_attempt_and_the_backoff_ttl() {
410        for attempt in [0u32, 1] {
411            let msg = message(OutboundKind::Webhook, attempt);
412            let verdict = decide(
413                &msg,
414                DeliveryOutcome::Retry {
415                    reason: "non-2xx status: 503".into(),
416                },
417                &CFG,
418            );
419            let Verdict::Retry {
420                next,
421                ttl_ms,
422                audit,
423            } = verdict
424            else {
425                panic!("expected a retry");
426            };
427            assert_eq!(next.attempt, attempt + 1);
428            assert_eq!(ttl_ms, backoff_ttl_ms(attempt + 1, &CFG));
429            assert_eq!(
430                audit.metadata,
431                Some(serde_json::json!({
432                    "delivery_id": msg.delivery_id,
433                    "attempt": attempt + 1,
434                    "error": "non-2xx status: 503",
435                    "next_retry_in_ms": ttl_ms,
436                }))
437            );
438            // Only `attempt` differs between the message and its retry copy.
439            assert_eq!(next.delivery_id, msg.delivery_id);
440            assert_eq!(next.payload, msg.payload);
441        }
442        assert_ne!(backoff_ttl_ms(1, &CFG), backoff_ttl_ms(2, &CFG));
443    }
