13/// Publishes outbound messages to their kind's primary or retry queue.
14///
15/// Implements the core [`OutboundPublisher`] port (enqueue to the primary
16/// queue), and additionally exposes [`Self::publish_retry`] for the consumer
17/// loop's TTL-delayed republish. Use `AmqpManager::create_publisher_channel` to
18/// obtain a channel with publisher confirms enabled before wrapping it here.
19#[derive(Clone)]
20pub struct AmqpOutboundPublisher {
21    channel: Channel,
22}
23
24/// Properties of a primary-queue publish: JSON, persistent.
25pub(crate) fn primary_properties() -> BasicProperties {
26    BasicProperties::default()
27        .with_content_type("application/json".into())
28        .with_delivery_mode(2) // persistent
29}
30
31/// Properties of a retry-queue publish: as [`primary_properties`] plus the
32/// per-message TTL after which the broker dead-letters it back to the primary.
33pub(crate) fn retry_properties(ttl_ms: u64) -> BasicProperties {
34    primary_properties().with_expiration(ttl_ms.to_string().into())
35}
36
37impl AmqpOutboundPublisher {
38    /// Wrap a confirm-enabled publisher channel.
39    pub fn new(channel: Channel) -> Self {
40        Self { channel }
41    }
42
43    async fn publish_to(
44        &self,
45        queue: &str,
46        msg: &OutboundMessage,
47        properties: BasicProperties,
48    ) -> Result<(), AmqpError> {
49        let payload = wire::encode(msg).map_err(|e| {
50            error!(kind = %msg.kind, error = %e, "Failed to serialize outbound message");
51            AmqpError::Publish(e.to_string())
52        })?;
53
54        // The default (nameless) exchange routes by queue name.
55        let confirm = self
56            .channel
57            .basic_publish(
58                "".into(),
59                queue.into(),
60                BasicPublishOptions::default(),
61                &payload,
62                properties,
63            )
64            .await
65            .map_err(|e| AmqpError::Publish(e.to_string()))?;
66
67        match confirm.await {
68            Ok(Confirmation::Nack(_)) => Err(AmqpError::Publish(format!(
69                "broker nacked {} publish",
70                msg.kind
71            ))),
72            Err(e) => {
73                error!(kind = %msg.kind, error = %e, "Outbound publish not confirmed by broker");
74                Err(AmqpError::Publish(e.to_string()))
75            }
76            Ok(_) => Ok(()),
77        }
78    }
79
80    /// Publish to the kind's primary queue (first attempt, or a message that
81    /// already dead-lettered back from the retry queue).
82    pub async fn publish(&self, msg: &OutboundMessage) -> Result<(), AmqpError> {
83        let topology = OutboundTopology::for_kind(msg.kind);
84        self.publish_to(&topology.primary, msg, primary_properties())
85            .await
86    }
87
88    /// Publish to the kind's retry queue with a per-message TTL of `ttl_ms`.
89    /// RabbitMQ dead-letters the message back to the primary queue once the TTL
90    /// expires; no consumer is ever attached to the retry queue, so no slot is
91    /// held for the delay.
92    pub async fn publish_retry(&self, msg: &OutboundMessage, ttl_ms: u64) -> Result<(), AmqpError> {
93        let topology = OutboundTopology::for_kind(msg.kind);
94        self.publish_to(&topology.retry, msg, retry_properties(ttl_ms))
95            .await
96    }
97}
98
99impl OutboundPublisher for AmqpOutboundPublisher {
100    fn enqueue<'a>(
101        &'a self,
102        msg: &'a OutboundMessage,
103    ) -> OutboundFuture<'a, Result<(), OutboundError>> {
104        Box::pin(async move {
105            self.publish(msg)
106                .await
107                .map_err(|e| OutboundError::Enqueue(e.to_string()))
108        })
109    }
110}

112#[cfg(test)]
113mod tests {
114    use super::*;
115
116    #[test]
117    fn primary_publish_is_persistent_json_without_expiration() {
118        let props = primary_properties();
119        assert_eq!(props.delivery_mode(), &Some(2));
120        assert_eq!(
121            props.content_type().as_ref().map(|s| s.to_string()),
122            Some("application/json".to_string())
123        );
124        assert!(props.expiration().is_none());
125    }
126
127    #[test]
128    fn retry_publish_carries_the_ttl_as_expiration() {
129        let props = retry_properties(30_000);
130        assert_eq!(props.delivery_mode(), &Some(2));
131        assert_eq!(
132            props.expiration().as_ref().map(|s| s.to_string()),
133            Some("30000".to_string())
134        );
135    }
