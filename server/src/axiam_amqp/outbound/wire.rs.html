1//! The bytes on the queue.
2//!
3//! SA-14 (N3): every message travels inside the server-only signed envelope
4//! of [`crate::internal_envelope`], with purpose `outbound.<slug>`. [`seal`]
5//! and the consumer's gate own the envelope; this module owns its `body`.
6//!
7//! The webhook kind's body keeps its pre-extraction format: the JSON of
8//! [`WebhookMessage`] (`webhook_id`, `delivery_id`, `tenant_id`, `event_type`,
9//! `payload`, `attempt`), which is also what an old replica wrote un-enveloped
10//! and what `AXIAM__AMQP__INTERNAL_QUEUES__REQUIRE_SIGNATURE=false` still
11//! admits during a rolling upgrade.
12//!
13//! Every other kind's body is the plain JSON of
14//! [`OutboundMessage`](axiam_core::outbound::OutboundMessage), which carries its
15//! own `kind`. A new kind needs no code here; it takes the generic path by
16//! default.
17
18use axiam_core::outbound::{OutboundError, OutboundKind, OutboundMessage};

25impl From<WebhookMessage> for OutboundMessage {
26    fn from(m: WebhookMessage) -> Self {
27        Self {
28            kind: OutboundKind::Webhook,
29            tenant_id: m.tenant_id,
30            target_id: m.webhook_id,
31            delivery_id: m.delivery_id,
32            event_type: m.event_type,
33            payload: m.payload,
34            attempt: m.attempt,
35        }
36    }
37}
38
39fn to_webhook_message(msg: &OutboundMessage) -> WebhookMessage {
40    WebhookMessage {
41        webhook_id: msg.target_id,
42        delivery_id: msg.delivery_id,
43        tenant_id: msg.tenant_id,
44        event_type: msg.event_type.clone(),
45        payload: msg.payload.clone(),
46        attempt: msg.attempt,
47    }
48}

50/// The envelope body of `msg`: its kind's JSON.
51pub fn encode_body(msg: &OutboundMessage) -> Result<serde_json::Value, OutboundError> {
52    let value = if msg.kind == OutboundKind::Webhook {
53        serde_json::to_value(to_webhook_message(msg))
54    } else {
55        serde_json::to_value(msg)
56    };
57    value.map_err(|e| OutboundError::Enqueue(format!("serialize {} message: {e}", msg.kind)))
58}
59
60/// Seal `msg` for its kind's queue, current from `issued_at`.
61pub fn seal(
62    signer: &InternalSigner,
63    msg: &OutboundMessage,
64    issued_at: DateTime<Utc>,
65) -> Result<Vec<u8>, OutboundError> {
66    let body = encode_body(msg)?;
67    signer
68        .seal(&outbound_purpose(msg.kind), &body, issued_at)
69        .map_err(|e| OutboundError::Enqueue(format!("seal {} message: {e}", msg.kind)))
70}

72/// Parse an envelope body taken off `kind`'s queue.
73///
74/// A generic-envelope message whose own `kind` differs from the queue's is
75/// refused: it was published to the wrong queue and must not reach another
76/// kind's deliverer.
77pub fn decode_body(
78    kind: OutboundKind,
79    body: serde_json::Value,
80) -> Result<OutboundMessage, OutboundError> {
81    if kind == OutboundKind::Webhook {
82        return serde_json::from_value::<WebhookMessage>(body)
83            .map(OutboundMessage::from)
84            .map_err(|e| OutboundError::Malformed(e.to_string()));
85    }
86    let msg: OutboundMessage =
87        serde_json::from_value(body).map_err(|e| OutboundError::Malformed(e.to_string()))?;
88    if msg.kind != kind {
89        return Err(OutboundError::Malformed(format!(
90            "message of kind {} on the {kind} queue",
91            msg.kind
92        )));
93    }
94    Ok(msg)
95}

97#[cfg(test)]
98mod tests {
99    use super::*;
100    use uuid::Uuid;
101
102    fn webhook_msg() -> OutboundMessage {
103        OutboundMessage {
104            kind: OutboundKind::Webhook,
105            tenant_id: Uuid::new_v4(),
106            target_id: Uuid::new_v4(),
107            delivery_id: Uuid::new_v4(),
108            event_type: "user.created".into(),
109            payload: serde_json::json!({"k": "v"}),
110            attempt: 2,
111        }
112    }
113
114    /// The webhook body is exactly what the pre-extraction code wrote.
115    #[test]
116    fn webhook_bytes_equal_the_legacy_webhook_message() {
117        let msg = webhook_msg();
118        let legacy = serde_json::to_vec(&WebhookMessage {
119            webhook_id: msg.target_id,
120            delivery_id: msg.delivery_id,
121            tenant_id: msg.tenant_id,
122            event_type: msg.event_type.clone(),
123            payload: msg.payload.clone(),
124            attempt: msg.attempt,
125        })
126        .unwrap();
127        assert_eq!(
128            serde_json::to_vec(&encode_body(&msg).unwrap()).unwrap(),
129            serde_json::to_vec(&serde_json::from_slice::<serde_json::Value>(&legacy).unwrap())
130                .unwrap()
131        );
132        let text = String::from_utf8(legacy).unwrap();
133        assert!(text.contains("\"webhook_id\""), "{text}");
134        assert!(!text.contains("target_id"), "{text}");
135        assert!(!text.contains("\"kind\""), "{text}");
136    }
137
138    /// A message written by the old code, in flight at upgrade, still parses
139    /// as a body (admitted only while signatures are optional).
140    #[test]
141    fn legacy_in_flight_webhook_message_decodes() {
142        let (w, d, t) = (Uuid::new_v4(), Uuid::new_v4(), Uuid::new_v4());
143        let raw = format!(
144            r#"{{"webhook_id":"{w}","delivery_id":"{d}","tenant_id":"{t}","event_type":"user.created","payload":{{"a":1}},"attempt":3}}"#
145        );
146        let msg = decode_body(OutboundKind::Webhook, serde_json::from_str(&raw).unwrap()).unwrap();
147        assert_eq!(msg.kind, OutboundKind::Webhook);
148        assert_eq!(msg.target_id, w);
149        assert_eq!(msg.delivery_id, d);
150        assert_eq!(msg.tenant_id, t);
151        assert_eq!(msg.attempt, 3);
152        assert_eq!(msg.payload, serde_json::json!({"a": 1}));
153    }
154
155    #[test]
156    fn webhook_round_trips() {
157        let msg = webhook_msg();
158        assert_eq!(
159            decode_body(OutboundKind::Webhook, encode_body(&msg).unwrap()).unwrap(),
160            msg
161        );
162    }
163
164    #[test]
165    fn garbage_is_malformed() {
166        let err =
167            decode_body(OutboundKind::Webhook, serde_json::json!("not a message")).unwrap_err();
168        assert!(matches!(err, OutboundError::Malformed(_)), "{err:?}");
169    }
