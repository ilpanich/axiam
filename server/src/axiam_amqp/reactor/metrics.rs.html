1//! Reactor dispatch counters (X1 / R2.2).
2//!
3//! # Why atomics and a `tracing` event rather than a metrics crate
4//!
5//! This workspace has no metrics facade — `axiam-db::metrics` is the pattern
6//! every other instrumented boundary follows: process-wide relaxed atomics,
7//! readable from tests and from a future `/metrics` endpoint, plus one
8//! structured `tracing` event per interesting transition so an operator gets
9//! the same fact in the log stream without a scrape. Introducing a second
10//! convention here would leave the reactor path measurable by different means
11//! than everything around it.
12//!
13//! # The counters that matter, and why each exists
14//!
15//! `X1.2` requires that **every timeout is audited and surfaced as a metric**.
16//! A timeout is audited by the gate and counted by [`record_failure`]; the
17//! reason both are needed is that they answer different questions. The audit
18//! record says *which* reactor failed on *which* operation and is queryable per
19//! tenant; the counter says *how often*, cheaply, without reading the audit
20//! table — which is the difference between a health panel that can refresh
21//! every few seconds and one that cannot.
22//!
23//! [`dispatches`] counts only events that actually reached a chain. An event
24//! with no registered reactor never increments anything: the whole point of the
25//! hot-path exclusion argument is that the un-hooked path stays free, and a
26//! counter increment on it would be a (tiny) measurable delta where the plan
27//! promises zero.
28
29use std::sync::atomic::{AtomicU64, Ordering};
30
31use crate::reactor::dispatcher::DispatchFailure;
32
33macro_rules! counters {
34    ($($vis:vis $name:ident => $static_name:ident, $metric:literal, $doc:literal;)*) => {
35        $(
36            static $static_name: AtomicU64 = AtomicU64::new(0);
37            #[doc = $doc]
38            #[doc = ""]
39            #[doc = concat!("Metric name: `", $metric, "`.")]
40            $vis fn $name() -> u64 {
41                $static_name.load(Ordering::Relaxed)
42            }
43        )*
44    };
45}

79/// Record that a chain ran, and how long it took.
80pub(crate) fn record_dispatch(elapsed_ms: u64) {
81    DISPATCHES.fetch_add(1, Ordering::Relaxed);
82    CHAIN_DURATION_MS_TOTAL.fetch_add(elapsed_ms, Ordering::Relaxed);
83}

109/// Count one dispatch failure, both in its own bucket and in the total.
110///
111/// The per-kind split is what makes the aggregate actionable: "reactor failures
112/// are up" is not a page anyone can act on, whereas "timeouts are up while
113/// transport failures are flat" names the reactor rather than the broker.
114pub(crate) fn record_failure(failure: &DispatchFailure) {
115    FAILURES.fetch_add(1, Ordering::Relaxed);
116    match failure {
117        DispatchFailure::Timeout => TIMEOUTS.fetch_add(1, Ordering::Relaxed),
118        DispatchFailure::BudgetExhausted => BUDGET_EXHAUSTIONS.fetch_add(1, Ordering::Relaxed),
119        DispatchFailure::Overloaded => OVERLOADS.fetch_add(1, Ordering::Relaxed),
120        DispatchFailure::Transport(_) => TRANSPORT_FAILURES.fetch_add(1, Ordering::Relaxed),
121        DispatchFailure::Rejected(_) => REJECTIONS.fetch_add(1, Ordering::Relaxed),
122    };
123}

125/// The stable metric-name string for a failure kind. Used in the audit record
126/// and the log line so the two can be joined against the counter.
127pub const fn failure_kind(failure: &DispatchFailure) -> &'static str {
128    match failure {
129        DispatchFailure::Timeout => "timeout",
130        DispatchFailure::BudgetExhausted => "budget_exhausted",
131        DispatchFailure::Overloaded => "overloaded",
132        DispatchFailure::Transport(_) => "transport",
133        DispatchFailure::Rejected(_) => "reply_rejected",
134    }
135}

137/// Every counter, as `(metric_name, value)` — what a `/metrics` endpoint or a
138/// health panel renders, and what a test asserts against without naming a
139/// static.
140pub fn snapshot() -> Vec<(&'static str, u64)> {
141    vec![
142        ("axiam_reactor_dispatch_total", dispatches()),
143        ("axiam_reactor_deny_total", denials()),
144        ("axiam_reactor_mutate_total", mutations()),
145        ("axiam_reactor_require_mfa_total", step_ups()),
146        ("axiam_reactor_timeout_total", timeouts()),
147        ("axiam_reactor_budget_exhausted_total", budget_exhaustions()),
148        ("axiam_reactor_overload_total", overloads()),
149        (
150            "axiam_reactor_transport_failure_total",
151            transport_failures(),
152        ),
153        ("axiam_reactor_reply_rejected_total", rejections()),
154        ("axiam_reactor_failure_total", failures()),
155        ("axiam_reactor_registry_error_total", registry_errors()),
156        (
157            "axiam_reactor_registry_stale_serve_total",
158            registry_stale_serves(),

162            gate_patch_rejections(),

164        (
165            "axiam_reactor_chain_duration_ms_total",
166            chain_duration_ms_total(),
167        ),
168    ]
169}

171#[cfg(test)]
172mod tests {
173    use super::*;
174    use crate::reactor::protocol::ReplyRejection;
175
176    /// Counters are process-wide, so a test that asserts an absolute value
177    /// would fail whenever another test in the same binary incremented first.
178    /// Every assertion here is a *delta*, which is also how an operator reads
179    /// a counter.
180    #[test]
181    fn every_failure_kind_lands_in_its_own_bucket_and_in_the_total() {
182        for (failure, read) in [
183            (DispatchFailure::Timeout, timeouts as fn() -> u64),
184            (DispatchFailure::BudgetExhausted, budget_exhaustions),
185            (DispatchFailure::Overloaded, overloads),
186            (
187                DispatchFailure::Transport("down".into()),
188                transport_failures,
189            ),
190            (
191                DispatchFailure::Rejected(ReplyRejection::BadSignature),
192                rejections,
193            ),
194        ] {
195            let before_kind = read();
196            let before_total = failures();
197            record_failure(&failure);
198            assert_eq!(read(), before_kind + 1, "{failure} missed its bucket");
199            assert_eq!(failures(), before_total + 1, "{failure} missed the total");
200        }
201    }
202
203    #[test]
204    fn the_failure_kind_label_is_stable_and_distinct() {
205        let labels = [
206            failure_kind(&DispatchFailure::Timeout),
207            failure_kind(&DispatchFailure::BudgetExhausted),
208            failure_kind(&DispatchFailure::Overloaded),
209            failure_kind(&DispatchFailure::Transport(String::new())),
210            failure_kind(&DispatchFailure::Rejected(ReplyRejection::Stale)),
211        ];
212        let mut sorted = labels.to_vec();
213        sorted.sort_unstable();
214        sorted.dedup();
215        assert_eq!(sorted.len(), labels.len(), "labels must be distinct");
216        assert_eq!(failure_kind(&DispatchFailure::Timeout), "timeout");
217    }
218
219    #[test]
220    fn the_snapshot_names_every_counter_exactly_once() {
221        let snap = snapshot();
222        let mut names: Vec<_> = snap.iter().map(|(n, _)| *n).collect();
223        let before = names.len();
224        names.sort_unstable();
225        names.dedup();
226        assert_eq!(before, names.len(), "duplicate metric name in the snapshot");
227        assert!(names.iter().all(|n| n.starts_with("axiam_reactor_")));
228    }
