1//! gRPC auth interceptor — validates bearer JWT on every inbound request.
2//!
3//! SEC-003: Any caller that reaches the authorization service must present a
4//! cryptographically verified bearer token.  The interceptor extracts the
5//! `authorization` metadata header, strips the `Bearer ` prefix, delegates
6//! to `axiam_auth::token::validate_access_token`, and — on success — stores
7//! the [`ValidatedClaims`] in the request extensions so downstream service
8//! handlers can derive `tenant_id`/`subject_id` from verified claims rather
9//! than trusting the request body.
10
11use axiam_auth::config::AuthConfig;
12use axiam_auth::token::{
13    AUD_M2M, AUD_USER, PresentedProofs, validate_access_token, verify_token_binding,
14};
15use tonic::service::Interceptor;
16use tonic::{Request, Status};
17
18/// Tonic interceptor that enforces bearer JWT authentication.
19///
20/// Clone-safe: `AuthConfig` derives `Clone`, so `AuthInterceptor` can be
21/// cloned for each new connection (Tonic 0.14 requirement for
22/// `with_interceptor`).
23#[derive(Clone)]
24pub struct AuthInterceptor {
25    auth_config: AuthConfig,
26}
27
28impl AuthInterceptor {
29    /// Create a new interceptor backed by the given [`AuthConfig`].
30    pub fn new(auth_config: AuthConfig) -> Self {
31        Self { auth_config }
32    }
33}
34
35impl Interceptor for AuthInterceptor {
36    fn call(&mut self, mut req: Request<()>) -> Result<Request<()>, Status> {
37        let token = req
38            .metadata()
39            .get("authorization")
40            .and_then(|v| v.to_str().ok())
41            .and_then(|s| s.strip_prefix("Bearer "))
42            .ok_or_else(|| Status::unauthenticated("missing bearer token"))?;
43
44        let claims = validate_access_token(token, &self.auth_config)
45            .map_err(|_| Status::unauthenticated("invalid or expired token"))?;
46
47        // §17.2 residual 2: the audience policy on this transport used to be
48        // entirely implicit. `decode_access_token` accepts both audiences, and
49        // deliberately **skips** the audience check altogether when the claim
50        // is absent (the D-20 pre-Phase-4 back-compat window) — so a token
51        // with no `aud` reached the authorization service with no narrowing
52        // applied and nothing recorded about it. The REST extractors gate that
53        // same case on `allow_missing_aud_as_user` and warn on every use; this
54        // transport did neither.
55        //
56        // gRPC accepts **both** audiences on purpose, and that is now stated
57        // rather than inherited: this is the low-latency check surface for a
58        // service mesh, so machine callers are the norm, while user tokens
59        // reach it through SDK callers doing the same checks. What is fixed
60        // here is the *unstated* third case.
61        match claims.0.aud.as_deref() {
62            Some(AUD_USER) | Some(AUD_M2M) => {}
63            None => {
64                if !self.auth_config.allow_missing_aud_as_user {
65                    return Err(Status::unauthenticated("aud required"));
66                }
67                tracing::warn!(
68                    token_jti = %claims.0.jti,
69                    "grpc: accepted access token without aud — backward-compat window active"
70                );
71            }
72            Some(_) => return Err(Status::unauthenticated("unknown audience")),
73        }
74
75        enforce_sender_constraint(&req, &claims.0)?;
76
77        req.extensions_mut().insert(claims);
78        Ok(req)
79    }
80}

82/// Enforce a token's `cnf` sender constraint on the gRPC transport
83/// (X5.1; RFC 8705 §3.2, RFC 9449 §7.1).
84///
85/// AXIAM's gRPC surface is a resource server for AXIAM's own tokens, so it owes
86/// the same obligation the REST extractor does and that SDK contract §10.1
87/// rule 9 places on every SDK: **a token carrying `cnf` is not a bearer token.**
88/// Before this existed, a certificate-bound or DPoP-bound token presented here
89/// was accepted exactly as an unbound one — which made the binding decorative
90/// for any deployment whose authorization checks go through the mesh.
91///
92/// # Ordinary callers are untouched
93///
94/// A token with no `cnf` — every token AXIAM issued before X5.1, and every
95/// token it issues to a client that has not asked for binding — returns `Ok`
96/// without the peer chain being read. That is asserted directly by
97/// `an_unbound_token_is_accepted_with_no_evidence`, and it is the property this
98/// function is most likely to break.
99///
100/// # What this transport can and cannot check
101///
102/// | Confirmation | Checkable here? | Why |
103/// |---|---|---|
104/// | `x5t#S256` | **yes** | `Request::peer_certs()` gives the chain rustls verified for this connection, and the leaf's thumbprint is one SHA-256 over bytes already in memory |
105/// | `jkt` (DPoP) | **no** | RFC 9449 binds a proof to an HTTP method and URI (`htm`/`htu`). A tonic interceptor sees neither — it runs on `Request<()>`, before the RPC path is bound to a request — so a proof cannot be checked against what it was minted for. Verifying it *without* `htm`/`htu` would accept a proof captured from any other endpoint, which is worse than not verifying at all |
106///
107/// The `jkt` row is a refusal, not a gap to be papered over: rule 9's rule is
108/// **reject when you cannot verify**, so a DPoP-bound token presented to this
109/// transport is refused. An operator who needs DPoP-bound tokens against the
110/// mesh should terminate them at the REST surface, which has the method and URI.
111fn enforce_sender_constraint(
112    req: &Request<()>,
113    claims: &axiam_auth::token::AccessTokenClaims,
114) -> Result<(), Status> {
115    // The fast path, and the common one: nothing claimed, nothing read.
116    if claims.cnf.is_none() {
117        return Ok(());
118    }
119
120    // The leaf is the first certificate in the chain rustls verified for this
121    // connection. Taking it from anywhere else — a metadata header, say —
122    // would make the whole mechanism decorative, which is why this reads
123    // `peer_certs()` and nothing else.
124    let certificate_thumbprint = req
125        .peer_certs()
126        .and_then(|chain| chain.first().map(|leaf| thumbprint_s256(leaf.as_ref())));
127
128    verify_token_binding(
129        claims,
130        PresentedProofs {
131            certificate_thumbprint: certificate_thumbprint.as_deref(),
132            // See the table above: a DPoP proof cannot be verified without the
133            // method and URI it was minted for, and this interceptor has
134            // neither. `None` here means a `jkt`-bound token is refused.
135            dpop_thumbprint: None,
136        },
137    )
138    .map_err(|e| {
139        tracing::debug!(
140            token_jti = %claims.jti,
141            error = %e,
142            "grpc: refused a sender-constrained token"
143        );
144        Status::unauthenticated("invalid or expired token")
145    })
146}

148/// RFC 8705 §3.1 `x5t#S256`: base64url-unpadded SHA-256 of the DER certificate.
149///
150/// Inlined rather than calling `axiam_oauth2::mtls::thumbprint_s256`: this
151/// crate does not depend on `axiam-oauth2` and should not acquire the whole
152/// OAuth2 authorization server to compute one digest. The operation is a
153/// SHA-256 plus a base64url encode either way, and
154/// `the_thumbprint_agrees_with_the_oauth2_crates_definition` pins the two
155/// against a shared vector so they cannot drift.
156fn thumbprint_s256(der: &[u8]) -> String {
157    use base64::Engine as _;
158    use sha2::{Digest, Sha256};
159    base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(Sha256::digest(der))
160}
