1//! gRPC rate limiting via tower-governor (per D-10, D-11), brought to parity
2//! with the fixed REST limiter (SECHRD-03, D-01c).
3//!
4//! Two layers cooperate here (mirroring
5//! `axiam-api-rest::middleware::rate_limit_shared` / `extractors::rate_limit`):
6//!
7//! - [`GrpcTrustedHopsKeyExtractor`] — a custom `tower_governor::KeyExtractor`
8//!   that replaces `SmartIpKeyExtractor`. `SmartIpKeyExtractor` unconditionally
9//!   trusts the LEFTMOST `X-Forwarded-For` hop and, more fundamentally, can
10//!   never find tonic's real peer address at all (it only looks for an
11//!   `axum::extract::ConnectInfo<SocketAddr>` extension or a bare `SocketAddr`
12//!   extension — tonic inserts `TcpConnectInfo`/`TlsConnectInfo<TcpConnectInfo>`
13//!   instead). This extractor mirrors the fixed REST
14//!   `XForwardedForKeyExtractor` (plan 24-03): a configured `trusted_hops`
15//!   selects the rightmost trusted XFF hop, and when there are NOT enough
16//!   hops to trust, XFF is ignored entirely and the verified tonic connection
17//!   peer address is used instead (never `hops[0]` — SECHRD-03/D-01d).
18//! - [`GrpcSharedRateLimitLayer`] — a pre-check `tower::Layer` over the
19//!   process-wide write-behind `axiam_db::SharedRateLimitCounter` (the SAME
20//!   counter type and env knobs — `AXIAM__RATE_LIMIT__SHARED`,
21//!   `AXIAM__RATE_LIMIT__SHARED_SYNC_MS` — the REST `RateLimitShared`
22//!   middleware uses), run BEFORE the existing per-replica in-memory
23//!   `GovernorLayer` (kept byte-for-byte unchanged as the fail-open fallback,
24//!   D-01b). This is deliberately NOT a `governor::StateStore` impl and never
25//!   calls `block_on` — `StateStore::measure_and_replace` is a *synchronous*
26//!   trait method (RESEARCH Pitfall 1).
27//!
28//!   Since the H2 performance fix this layer performs NO datastore round trip
29//!   on the request path (it used to `await` one `UPSERT` per call, and being
30//!   server-wide, *every* gRPC call paid it — see the layer's own docs for the
31//!   measurements). A single background flusher coalesces the in-memory
32//!   increments into one write per bucket per sync interval.
33//!
34//! HARD CONSTRAINT (D-01c coordination note): the `Quota::per_second(...)
35//! .burst_size(...)` throughput/quota math in [`build_grpc_governor_layer`]
36//! is untouched by this module — CORR-01/Phase 26 owns it.
37//!
38//! # Units and scope (run-4 fixes I1 and I2)
39//!
40//! - **I1 — units.** The gRPC ceilings are configured **per second**; the
41//!   shared cross-replica bucket runs a fixed **60-second** window shared with
42//!   the REST limiter. The conversion lives in exactly one place,
43//!   [`per_sec_to_window_limit`], and the production constructor
44//!   ([`GrpcSharedRateLimitLayer::new_method_scoped`]) takes per-second
45//!   ceilings so no caller can hand a per-second number to a per-minute
46//!   window again.
47//! - **I2 — scope.** Both layers used to be server-wide, so the *authz*
48//!   ceiling throttled identity reads and admin traffic too. Buckets are now
49//!   split per [`GrpcMethodFamily`] (authz-check / identity-read / admin /
50//!   infrastructure).
51//!
52//! # SEC-079 — the admin ceiling is absolute, not derived
53//!
54//! The admin family contains `axiam.v1.UserService/ValidateCredentials`, an
55//! **Argon2id verification**. Its ceiling is a CPU guard, so it is an absolute
56//! constant ([`ADMIN_PER_SEC_DEFAULT`]) rather than a multiple of — or the
57//! same number as — the read-sized `authz_per_sec` base. See that constant's
58//! docs for the full reasoning; the short version is that when the I1 units
59//! bug was fixed the effective per-IP ceiling on that RPC rose 60x, and
60//! deriving it from a read-sized base would let a `gateway`/`mesh` posture
61//! raise it further still.

92/// pre-check (`axiam-api-rest::middleware::rate_limit_shared::trusted_hops`)
93/// and `server.rs::build_governor` use, so the gRPC key and the REST key are

96/// `pub(crate)` so `server.rs::start_grpc_server` can pass the SAME value to

98/// [`GrpcSharedRateLimitLayer::new`] — the key-extraction logic in the

100pub(crate) fn trusted_hops_from_env() -> usize {
101    std::env::var("AXIAM__RATE_LIMIT__TRUSTED_HOPS")
102        .ok()
103        .and_then(|v| v.parse().ok())
104        .unwrap_or(0)
105}

111/// `axiam_rate_limit_xff_discarded_total{protocol="grpc"}`.
112static XFF_DISCARDED: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
113
114/// Whether the explanatory `WARN` has already been emitted this process.
115static XFF_WARNED: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
116
117/// How many gRPC requests arrived with an `X-Forwarded-For` header that
118/// `trusted_hops` discarded, so the key came from the connection peer instead.
119///
120/// Metric name: `axiam_rate_limit_xff_discarded_total{protocol="grpc"}`. The
121/// REST listener keeps the identical counter under `{protocol="rest"}`
122/// (`axiam_api_rest::extractors::xff_metrics`); the two crates are siblings and
123/// neither may depend on the other, so the label is expressed as two counters
124/// an exporter renders as one labelled series.
125///
126/// **A non-zero value is not automatically a fault.** It is a fault when it
127/// tracks total request volume: that means the header is present on every
128/// request and trusted on none, i.e. `trusted_hops` is too high for this
129/// deployment's proxy count and every client shares one bucket.
130///
131/// # Why a counter and not a metrics crate
132///
133/// This workspace has no metrics facade. `axiam_db::metrics` and
134/// `axiam_amqp::reactor::metrics` are the pattern: a process-wide relaxed
135/// atomic, readable from tests and from a future `/metrics` endpoint, with the
136/// Prometheus name it will be exported under written down beside it.
137pub fn xff_discarded_total() -> u64 {
138    XFF_DISCARDED.load(std::sync::atomic::Ordering::Relaxed)
139}
140
141/// Record that a present `X-Forwarded-For` header was discarded because
142/// `trusted_hops` exceeded the hops available, and explain it once.
143///
144/// Called only on that path — a request with **no** header does not come
145/// through here, because a client with no proxy in front of it is not a
146/// misconfiguration and counting it would bury the signal.
147///
148/// The `WARN` is once per process on purpose: the condition fires on *every*
149/// request under a misconfiguration, so one line per request would be a log
150/// flood proportional to traffic, and the second line adds nothing the first
151/// did not say.
152fn record_discarded_xff(hops_seen: usize, trusted_hops: usize) {
153    use std::sync::atomic::Ordering;
154    XFF_DISCARDED.fetch_add(1, Ordering::Relaxed);
155
156    if !XFF_WARNED.swap(true, Ordering::Relaxed) {
157        tracing::warn!(
158            protocol = "grpc",
159            hops_seen,
160            trusted_hops,
161            "X-Forwarded-For was DISCARDED and the rate-limit key fell back to the \
162             connection peer: the header carried {hops_seen} hop(s) but \
163             AXIAM__RATE_LIMIT__TRUSTED_HOPS is {trusted_hops}, which trusts them all. \
164             If this fires on every request, every client is sharing ONE bucket keyed \
165             on the proxy's address. The rule is trusted_hops = proxies - 1 \
166             (one proxy in front of the server means 0). Counted as \
167             axiam_rate_limit_xff_discarded_total{{protocol=\"grpc\"}}; logged once per \
168             process."
169        );
170    }
171}

173/// Extracts the gRPC caller's IP from `X-Forwarded-For`, falling back to the
174/// verified tonic connection peer address (`TcpConnectInfo`/
175/// `TlsConnectInfo<TcpConnectInfo>`).
176///
177/// `trusted_hops` controls how many rightmost XFF entries are trusted
178/// reverse-proxy hops to skip (mirrors
179/// `axiam-api-rest::extractors::rate_limit::XForwardedForKeyExtractor`).
180/// When `trusted_hops >= hops.len()`, the header is NOT trusted at all and
181/// XFF is ignored entirely — the key is derived from the verified peer
182/// address instead (a client cannot manufacture extra hops to force a
183/// fallback to an attacker-controlled `hops[0]` — SECHRD-03/D-01d).
184#[derive(Debug, Clone, Default)]
185pub struct GrpcTrustedHopsKeyExtractor {
186    pub trusted_hops: usize,
187}
188
189impl GrpcTrustedHopsKeyExtractor {
190    /// Create an extractor keyed by `trusted_hops` trusted reverse-proxy
191    /// hops (0 = no trusted hops; XFF is only ever used when it has more
192    /// than 0 entries beyond what's trusted).
193    pub fn new(trusted_hops: usize) -> Self {
194        Self { trusted_hops }
195    }
196}
197
198impl KeyExtractor for GrpcTrustedHopsKeyExtractor {
199    type Key = IpAddr;
200
201    fn extract<T>(&self, req: &Request<T>) -> Result<Self::Key, GovernorError> {
202        if let Some(forwarded_for) = req.headers().get("x-forwarded-for")
203            && let Ok(val) = forwarded_for.to_str()
204        {
205            let hops: Vec<&str> = val.split(',').map(str::trim).collect();
206            // Select the rightmost-untrusted hop (parity with REST).
207            if self.trusted_hops < hops.len() {
208                let idx = hops.len() - 1 - self.trusted_hops;
209                if let Ok(ip) = hops[idx].parse::<IpAddr>() {
210                    return Ok(ip);
211                }
212            }
213            // Fewer hops than trusted_hops requires: the header cannot be
214            // trusted. Fall through to the verified peer address below
215            // instead of indexing into XFF (SECHRD-03 — no `hops[0]`
216            // fallback that would let a rotating XFF mint a fresh bucket).
217            //
218            // R-4: no longer silently. Under a `trusted_hops` one too high this
219            // fires on every request and every client shares one bucket keyed on
220            // the proxy — the failure T-212 describes, which went unnoticed
221            // precisely because nothing said so.
222            if self.trusted_hops >= hops.len() {
223                record_discarded_xff(hops.len(), self.trusted_hops);
224            }
225        }
226
227        grpc_peer_addr(req)
228            .map(|addr| addr.ip())
229            .ok_or(GovernorError::UnableToExtractKey)
230    }
231}

233/// Reads the verified tonic connection peer address from request
234/// extensions, mirroring `tonic::Request::remote_addr()`'s own lookup
235/// (`TcpConnectInfo` for plaintext, `TlsConnectInfo<TcpConnectInfo>` for TLS)
236/// — the extension types tonic's `ConnectInfoLayer` actually inserts (NOT
237/// the `axum::extract::ConnectInfo<SocketAddr>` / bare `SocketAddr`
238/// extensions `SmartIpKeyExtractor` looks for, which tonic never sets).
239fn grpc_peer_addr<T>(req: &Request<T>) -> Option<SocketAddr> {
240    req.extensions()
241        .get::<TcpConnectInfo>()
242        .and_then(TcpConnectInfo::remote_addr)
243        .or_else(|| {
244            req.extensions()
245                .get::<TlsConnectInfo<TcpConnectInfo>>()
246                .and_then(|info| info.get_ref().remote_addr())
247        })
248}

254/// The rate-limit bucket family a gRPC method belongs to (I2).
255///
256/// **Why this exists.** Both gRPC rate-limit layers used to be *server-wide*:
257/// one bucket, sized from `AXIAM__GRPC__GRPC_AUTHZ_PER_SEC`, shared by every
258/// method on every service. Run 4 measured `UserInfoService/GetUserInfo` — an
259/// identity read that sustains ~12 700/s — collapsing to the *authz* ceiling
260/// under production posture, because a `check_access` sizing decision was
261/// silently also a userinfo sizing decision
262/// (`claude_dev/improvement-after-run4-benchmark.md` I2). Splitting the
263/// buckets lets an operator size the authz path (called once per protected
264/// request) independently of identity reads and of administrative traffic.
265///
266/// **Classification is by gRPC path**, i.e. `/<package>.<Service>/<Method>`,
267/// which is the only identity available to a `tower::Layer` running before
268/// tonic resolves per-RPC claims (the same structural reason the buckets key
269/// on IP — see `crate::config::GrpcRateLimitKeyMode`).
270#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
271pub enum GrpcMethodFamily {
272    /// `axiam.v1.AuthorizationService` — `CheckAccess` / `BatchCheckAccess`.
273    /// Also the family for any **unrecognized** path: an unknown method must
274    /// land in the strictest bucket rather than in the unlimited one.
275    AuthzCheck,
276    /// Identity reads: `axiam.v1.UserInfoService` (OIDC-style self lookup)
277    /// and `axiam.v1.TokenService` (validate/introspect). High-frequency,
278    /// cheap, and measured an order of magnitude faster than an authz check.
279    IdentityRead,
280    /// Administrative / user-management traffic: `axiam.v1.UserService`
281    /// (`GetUser`, `ValidateCredentials`) and `axiam.v1.ReactorAdminService`
282    /// (reactor CRUD, `ListReactorEvents`). `ValidateCredentials` hashes a
283    /// password, so this family is deliberately NOT sized from read capacity —
284    /// its ceiling is the absolute [`ADMIN_PER_SEC_DEFAULT`] (SEC-079).
285    ///
286    /// `ReactorAdminService` joined it in R-5. It used to fall through the
287    /// catch-all into [`Self::AuthzCheck`], which is the *strictest* family by
288    /// design but also the one the `gateway` and `mesh` profiles **raise** — so
289    /// an administrative surface was being sized like the hot path, at 100/s
290    /// per IP by default and more under a profile. It carries no Argon2 cost,
291    /// which is what `ADMIN_PER_SEC_DEFAULT` was sized for, but administrative
292    /// surfaces should not scale with authorization throughput either, and
293    /// 10/s per IP is generous for reactor administration. A fifth family with
294    /// its own knob is the alternative if `ListReactorEvents` ever needs more;
295    /// a benchmark should say so before it is added.
296    Admin,
297    /// Infrastructure probes: gRPC **reflection** and **health**. Their whole
298    /// job is to answer during an incident, when the limited families are most
299    /// likely to be saturated, so this family gets a deliberately generous
300    /// ceiling ([`INFRA_PER_SEC`]) — but a **finite** one.
301    ///
302    /// This family used to be a literal pass-through (`Unlimited`) that
303    /// bypassed BOTH limiter layers, selected by a prefix test on the
304    /// attacker-controlled `:path`. Neither service is registered today, so
305    /// the practical effect was unmetered HTTP/2 stream churn rather than DB
306    /// work — but registering a health service would have turned it into a
307    /// genuinely unmetered endpoint. A generous bucket keeps probes working
308    /// during an overload without leaving an unbounded surface behind
309    /// (security re-verification 2026-08-03, residual 4).
310    Infra,
311}
312
313/// gRPC reflection service package prefix (v1 and v1alpha).
314const REFLECTION_PREFIX: &str = "/grpc.reflection.";
315/// gRPC health-checking service package prefix.
316const HEALTH_PREFIX: &str = "/grpc.health.";
317
318impl GrpcMethodFamily {
319    /// Classifies a gRPC request path (`/<package>.<Service>/<Method>`).
320    ///
321    /// Unknown paths deliberately fall into [`Self::AuthzCheck`] — the
322    /// strictest limited family — so that adding a service without updating
323    /// this function fails safe (throttled) rather than open (unlimited).
324    pub fn classify(path: &str) -> Self {
325        if path.starts_with(REFLECTION_PREFIX) || path.starts_with(HEALTH_PREFIX) {
326            return Self::Infra;
327        }
328        match path.split('/').nth(1).unwrap_or_default() {
329            "axiam.v1.UserInfoService" | "axiam.v1.TokenService" => Self::IdentityRead,
330            // R-5: `ReactorAdminService` is administrative traffic and is sized
331            // as such. Before this it reached the catch-all below and was
332            // throttled like `CheckAccess` — including being *raised* by the
333            // `gateway` and `mesh` profiles, which is not a thing an admin
334            // surface should inherit from authorization throughput.
335            "axiam.v1.UserService" | "axiam.v1.ReactorAdminService" => Self::Admin,
336            // "axiam.v1.AuthorizationService" and everything else.
337            _ => Self::AuthzCheck,
338        }
339    }
340
341    /// Shared-counter bucket-name prefix for this family.
342    ///
343    /// These names are part of the persisted `rate_limit_bucket` key space
344    /// (`"{endpoint}:{ip}"`), so `grpc_authz` is kept byte-for-byte for the
345    /// authz family — an in-flight upgrade keeps counting against the same
346    /// rows.
347    ///
348    /// Every family now has one: the infrastructure family is bucketed like
349    /// the rest rather than skipping the counter entirely (residual 4).
350    pub const fn shared_endpoint(self) -> &'static str {
351        match self {
352            Self::AuthzCheck => "grpc_authz",
353            Self::IdentityRead => "grpc_identity",
354            Self::Admin => "grpc_admin",
355            Self::Infra => "grpc_infra",
356        }
357    }
358}

360/// Per-family gRPC ceilings, in requests **per second per IP** (I2).
361///
362/// The unit is per-second at every boundary of this type. The shared
363/// cross-replica layer runs a 60-second window and therefore converts with
364/// [`per_sec_to_window_limit`] internally — callers never do that arithmetic
365/// themselves, which is precisely the bug I1 fixed (the per-second number was
366/// handed to a per-minute window verbatim, making the effective ceiling
367/// 1/60th of the configured one).
368#[derive(Debug, Clone, Copy, PartialEq, Eq)]
369pub struct GrpcRateLimits {
370    /// `AXIAM__GRPC__GRPC_AUTHZ_PER_SEC` — authorization decisions.
371    pub authz_per_sec: u32,
372    /// `AXIAM__GRPC__GRPC_IDENTITY_PER_SEC` — userinfo / token reads.
373    pub identity_per_sec: u32,
374    /// `AXIAM__GRPC__GRPC_ADMIN_PER_SEC` — user management. Defaults to the
375    /// absolute [`ADMIN_PER_SEC_DEFAULT`], never to a multiple of
376    /// [`Self::authz_per_sec`] (SEC-079).
377    pub admin_per_sec: u32,
378}
379
380/// Identity reads default to this multiple of the authz ceiling — they were
381/// measured ~14x faster than an authz check (12 665/s vs 887/s, run 4) and a
382/// resource server does one per protected request just like a check. The
383/// multiplier (rather than a fixed number) is what keeps the family coherent
384/// when `AXIAM__RATE_LIMIT__PROFILE` raises the authz ceiling.
385pub const IDENTITY_PER_SEC_MULTIPLE: u32 = 5;
386
387/// Default ceiling for the [`GrpcMethodFamily::Admin`] family, in requests per
388/// second per IP — **an absolute number, deliberately not derived from
389/// [`GrpcRateLimits::authz_per_sec`]** (SEC-079).
390///
391/// **Why absolute.** The admin family contains
392/// `axiam.v1.UserService/ValidateCredentials`, which performs a real Argon2id
393/// `verify_password` (~19 MiB of memory arena per verification). Its ceiling is
394/// therefore a **CPU/memory guard on an online-password-guessing surface**, not
395/// a throughput ceiling on a read. `authz_per_sec` is a read-sized base (100/s
396/// shipped, and `AXIAM__RATE_LIMIT__PROFILE=gateway`/`mesh` raise it to
397/// 1 000/s / 5 000/s); deriving the admin ceiling from it — even 1:1 — makes a
398/// service-mesh capacity decision silently also a credential-guessing-breadth
399/// decision. Decoupling the two is the entire point of this constant.
400///
401/// **Why 10/s (600/min per IP).** The family holds exactly two RPCs, `GetUser`
402/// and `ValidateCredentials`; the high-volume identity read is `GetUserInfo` on
403/// `UserInfoService`, which lives in [`GrpcMethodFamily::IdentityRead`] and is
404/// unaffected. 600 administrative calls per minute from one client IP is well
405/// above any real admin console or M2M provisioning loop, and roughly an order
406/// of magnitude below the point at which concurrent Argon2id verifications
407/// become the server's dominant cost. It also restores the protection the I1
408/// units fix accidentally removed: before that fix the shared window enforced
409/// the per-second number against a 60-second window, so the deployed ceiling
410/// was ~100/**minute**; correcting the units silently raised it to
411/// ~6 000/minute. 600/minute is the CPU-appropriate value the doc comment
412/// always claimed this family had.
413///
414/// Operators who genuinely need more can still pin
415/// `AXIAM__GRPC__GRPC_ADMIN_PER_SEC` — explicit configuration always wins.
416pub const ADMIN_PER_SEC_DEFAULT: u32 = 10;
417
418/// Ceiling for the [`GrpcMethodFamily::Infra`] family (gRPC reflection and
419/// health), in requests per second per IP — generous, but finite.
420///
421/// Infrastructure probes are intrinsically low-rate: a Kubernetes liveness or
422/// readiness probe runs on the order of once every few seconds per prober, and
423/// even a large fleet of sidecars sharing one NAT egress IP stays orders of
424/// magnitude below 100/s. So this ceiling can never be the thing that breaks a
425/// health check during an incident — which is the property that made the family
426/// a pass-through in the first place — while still bounding the surface, so a
427/// health service registered later is not an unmetered endpoint reachable by
428/// prefix-matching an attacker-controlled `:path` (residual 4).
429///
430/// It is an absolute constant for the same reason as
431/// [`ADMIN_PER_SEC_DEFAULT`]: probe volume has nothing to do with authz
432/// capacity, so a posture preset must not move it.
433pub const INFRA_PER_SEC: u32 = 100;
434
435impl GrpcRateLimits {
436    /// Derives the whole family from the authz ceiling alone — the shape an
437    /// operator gets when only `AXIAM__GRPC__GRPC_AUTHZ_PER_SEC` (or a
438    /// posture preset) is set.
439    ///
440    /// Only [`Self::identity_per_sec`] actually scales: `ValidateCredentials`
441    /// is an Argon2id verification, so the admin ceiling is a CPU guard and
442    /// takes the absolute [`ADMIN_PER_SEC_DEFAULT`] rather than inheriting a
443    /// read-sized base (SEC-079).
444    pub const fn from_authz_per_sec(authz_per_sec: u32) -> Self {
445        Self {
446            authz_per_sec,
447            identity_per_sec: authz_per_sec.saturating_mul(IDENTITY_PER_SEC_MULTIPLE),
448            admin_per_sec: ADMIN_PER_SEC_DEFAULT,
449        }
450    }
451
452    /// The per-second ceiling for `family`.
453    ///
454    /// Every family is limited: [`GrpcMethodFamily::Infra`] takes the fixed
455    /// [`INFRA_PER_SEC`], which is why this is not configurable per instance
456    /// (residual 4).
457    pub const fn per_sec(self, family: GrpcMethodFamily) -> u32 {
458        match family {
459            GrpcMethodFamily::AuthzCheck => self.authz_per_sec,
460            GrpcMethodFamily::IdentityRead => self.identity_per_sec,
461            GrpcMethodFamily::Admin => self.admin_per_sec,
462            GrpcMethodFamily::Infra => INFRA_PER_SEC,
463        }
464    }
465
466    /// The ceiling for `family` expressed in the shared layer's 60-second
467    /// window (I1).
468    pub const fn window_limit(self, family: GrpcMethodFamily) -> u32 {
469        per_sec_to_window_limit(self.per_sec(family))
470    }
471}

486/// Build a [`GovernorLayer`] for gRPC server-wide rate limiting.
487///
488/// The layer uses token-bucket semantics (via the `governor` crate) with
489/// `authz_per_sec` tokens replenished per second and a burst allowance equal
490/// to `authz_per_sec` (one second's worth of tokens). For service-mesh
491/// patterns where the authz endpoint is called on every request, the default
492/// of 100 tokens/sec is intentionally generous.
493///
494/// # Panics
495///
496/// Panics at startup if `authz_per_sec` is 0 — the governor crate requires a
497/// non-zero burst size.
498pub fn build_grpc_governor_layer(authz_per_sec: u32) -> GrpcGovernorLayer {
499    assert!(authz_per_sec >= 1, "grpc_authz_per_sec must be >= 1");
500
501    // CORR-01 (D-01): the PREVIOUS "fix" here (CQ-B44) called
502    // `tower_governor`'s OWN `GovernorConfigBuilder::per_second(n)` /
503    // `.burst_size(authz_per_sec * 2)` as if `.per_second(n)` meant "n tokens
504    // per second." It does NOT — `tower_governor`'s builder methods set the
505    // replenish PERIOD (i.e. "one token every n seconds"), so with the
506    // default `grpc_authz_per_sec=100` the governor replenished 1 token every
507    // 100 seconds, and raising the configured rate made throughput WORSE, not
508    // better. Fixed by bypassing `tower_governor`'s confusing
509    // `.per_second()`/`.per_millisecond()` convenience methods entirely and
510    // constructing the underlying `governor::Quota` directly via
511    // `Quota::per_second`, whose `replenish_interval()` is `1s /
512    // authz_per_sec` and whose `burst_size()` is `authz_per_sec` — then
513    // feeding that quota into the builder via `.const_period()`/
514    // `.const_burst_size()`. Burst is `authz_per_sec` (not `* 2`), per D-01.
515    //
516    // SECHRD-03/D-01c: key extractor is GrpcTrustedHopsKeyExtractor for
517    // IP-spoofing resistance parity with REST — unrelated to this quota math.
518    let burst = NonZeroU32::new(authz_per_sec).expect("authz_per_sec >= 1 asserted above");
519    let quota = Quota::per_second(burst);
520
521    let config = Arc::new(
522        GovernorConfigBuilder::default()
523            .const_period(quota.replenish_interval())
524            .const_burst_size(quota.burst_size().get())
525            .key_extractor(GrpcTrustedHopsKeyExtractor::new(trusted_hops_from_env()))
526            .finish()
527            .expect("valid GovernorConfig for gRPC rate limiter"),
528    );
529
530    GovernorLayer::new(config)
531}

582/// Dispatches each request to the governor for its [`GrpcMethodFamily`].
583///

585/// routing decision over rate-limit state only — the request itself reaches
586/// the identical tonic router whichever branch it takes. There is no
587/// pass-through branch: the infrastructure family has its own (generous)
588/// governor rather than bypassing the layer (residual 4).

597impl<S> Service<Request<tonic::body::Body>> for GrpcMethodScopedGovernor<S>
598where
599    S: Service<Request<tonic::body::Body>, Response = Response<tonic::body::Body>>
600        + Clone
601        + Send
602        + 'static,
603    S::Future: Send + 'static,
604    S::Error: Send + 'static,
605{
606    type Response = Response<tonic::body::Body>;
607    type Error = S::Error;
608    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;
609
610    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
611        // Every branch must be ready before we can promise `call` will
612        // succeed, since the family is only known once the request arrives.
613        // In practice all four wrap clones of the same always-ready tonic
614        // router, so this is a formality that keeps the tower contract exact.
615        std::task::ready!(Service::poll_ready(&mut self.authz, cx))?;
616        std::task::ready!(Service::poll_ready(&mut self.identity, cx))?;
617        std::task::ready!(Service::poll_ready(&mut self.admin, cx))?;
618        Service::poll_ready(&mut self.infra, cx)
619    }
620
621    fn call(&mut self, req: Request<tonic::body::Body>) -> Self::Future {
622        // Standard clone-and-swap per branch so the returned future owns an
623        // independent, ready-to-call copy (same convention as
624        // `GrpcSharedRateLimitService::call`).
625        match GrpcMethodFamily::classify(req.uri().path()) {
626            GrpcMethodFamily::AuthzCheck => {
627                let clone = self.authz.clone();
628                let mut svc = std::mem::replace(&mut self.authz, clone);
629                Box::pin(async move { svc.call(req).await })
630            }
631            GrpcMethodFamily::IdentityRead => {
632                let clone = self.identity.clone();
633                let mut svc = std::mem::replace(&mut self.identity, clone);
634                Box::pin(async move { svc.call(req).await })
635            }
636            GrpcMethodFamily::Admin => {
637                let clone = self.admin.clone();
638                let mut svc = std::mem::replace(&mut self.admin, clone);
639                Box::pin(async move { svc.call(req).await })
640            }
641            GrpcMethodFamily::Infra => {
642                let clone = self.infra.clone();
643                let mut svc = std::mem::replace(&mut self.infra, clone);
644                Box::pin(async move { svc.call(req).await })
645            }
646        }
647    }
648}

654/// Fixed-window duration (seconds) for the shared bucket — same value and
655/// rationale as the REST shared pre-check
656/// (`axiam-api-rest::middleware::rate_limit_shared::WINDOW_SECS`): a simple
657/// fixed-window counter is acceptable here since this layer only needs to be
658/// *approximately* right (it fails open by design).
659pub const WINDOW_SECS: i64 = 60;
660
661/// Converts a per-**second** ceiling into the budget for one
662/// [`WINDOW_SECS`]-second shared window (I1).
663///
664/// **This is the units bug this constant exists to prevent.** The gRPC
665/// ceiling (`AXIAM__GRPC__GRPC_AUTHZ_PER_SEC`) is per second; the shared
666/// cross-replica bucket is a fixed 60-second window shared with the REST
667/// limiter, whose knobs are per minute. Before I1 the per-second number was
668/// handed to [`GrpcSharedRateLimitLayer::new`] verbatim, so the shared layer
669/// enforced "N per 60 s" while the operator had configured "N per second" —
670/// and since the stricter of the two cooperating layers wins, the effective
671/// ceiling was **1/60th** of the configured one. Run 4 measured exactly that:
672/// with `GRPC_AUTHZ_PER_SEC=100`, all three gRPC scenarios admitted ~100 ops
673/// per minute (`claude_dev/improvement-after-run4-benchmark.md` I1).
674///
675/// The multiply saturates: an operator who configures a per-second ceiling
676/// above `u32::MAX / 60` gets `u32::MAX` for the window (effectively
677/// unlimited, which is what they asked for) instead of a wrapped — and
678/// therefore *tiny* — budget.
679///
680/// The alternative fix, a per-second shared window, was rejected: it would
681/// fork the `rate_limit_bucket` schema away from the REST limiter's and
682/// change the write-behind flusher cadence, for no enforcement benefit.
683pub const fn per_sec_to_window_limit(per_sec: u32) -> u32 {
684    per_sec.saturating_mul(WINDOW_SECS as u32)
685}
686
687/// Truncates `now` down to the start of the current fixed
688/// [`WINDOW_SECS`]-second window.
689///
690/// No longer on the request path — [`GrpcSharedRateLimitService::call`] hands
691/// the raw `now` to [`axiam_db::SharedRateLimitCounter::check_at`] instead
692/// (run-5 J1). Kept, and asserted by a test, because [`WINDOW_SECS`] must
693/// keep agreeing with the counter's configured window.
694#[cfg_attr(not(test), allow(dead_code))]
695fn window_start(now: DateTime<Utc>) -> DateTime<Utc> {
696    let epoch = now.timestamp();
697    let start_epoch = epoch - epoch.rem_euclid(WINDOW_SECS);
698    DateTime::<Utc>::from_timestamp(start_epoch, 0).unwrap_or(now)
699}

701/// Builds a gRPC `RESOURCE_EXHAUSTED` response — the same status the

703/// (see `tower_governor::errors::GovernorError`'s `Response<tonic::body::Body>`
704/// conversion) — so clients see one consistent rate-limit contract
705/// regardless of which layer rejected the request.
706fn too_many_requests_response() -> Response<tonic::body::Body> {
707    tonic::Status::resource_exhausted("rate limit exceeded").into_http()

710/// Shared (cross-replica) rate-limit pre-check `tower::Layer` for the gRPC
711/// path (SECHRD-03 / D-01a, D-01b, D-01c, D-01d).
712///
713/// Consults the process-wide write-behind [`SharedRateLimitCounter`]
714/// (`axiam-db::rate_limit_counter`) — the SAME counter type and the SAME env
715/// knobs the REST `RateLimitShared` middleware uses, so both listeners behave
716/// identically. There is no reimplemented counter here. Wire this layer BEFORE
717/// (i.e. `.layer()` it FIRST — tower's `ServiceBuilder`/`Server::builder()`
718/// executes the FIRST-added layer with the request FIRST, the opposite of
719/// actix's last-`.wrap()`-is-outermost rule) the existing
720/// [`build_grpc_governor_layer`] `GovernorLayer` on the same
721/// `Server::builder()`, e.g.:
722///
723/// ```ignore
724/// Server::builder()
725///     .layer(GrpcSharedRateLimitLayer::new(db, "grpc_authz", grpc_config.grpc_authz_per_sec, trusted_hops))
726///     .layer(build_grpc_governor_layer(grpc_config.grpc_authz_per_sec))
727///     .add_service(authz_svc)
728/// ```
729///
730/// # No synchronous datastore write on the request path (H2 fix)
731///
732/// This layer used to `await` one SurrealDB `UPSERT` per call. Because it is a
733/// **server-wide** layer on the tonic listener, EVERY gRPC call paid that
734/// write — measured at ~39–46 ms of datastore-attributed latency, pinning the
735/// listener at ~20 ops/s regardless of concurrency
736/// (`claude_dev/postseed-transient-investigation.md` §1, §2, §7). It now pays
737/// only an in-memory sharded-map increment
738/// ([`SharedRateLimitCounter::check`], fully synchronous), while a single
739/// background flusher coalesces the increments into one datastore write per
740/// bucket per `AXIAM__RATE_LIMIT__SHARED_SYNC_MS`. Cross-replica enforcement
741/// is therefore eventual, with the overshoot bound stated precisely in the
742/// [`axiam_db::rate_limit_counter`] module docs; the in-memory governor still
743/// makes a full per-replica decision on the same traffic.
744///
745/// Everything else is unchanged: the same `{endpoint}:{ip}` bucket key, the
746/// same `GrpcTrustedHopsKeyExtractor` key derivation (D-01d), and the same
747/// `RESOURCE_EXHAUSTED` rejection.
748///
749/// **Fail-open (D-01b, T-24-73 accepted risk):** when no client IP can be
750/// extracted, or the shared layer is disabled via
751/// `AXIAM__RATE_LIMIT__SHARED=off`, this layer forwards the request unchanged
752/// so the existing in-memory governor makes the decision instead. Store
753/// outages are surfaced by the counter's flusher (a `warn` alarm that never
754/// includes the raw key) while decisions keep being served from local counts —
755/// a counter-store outage must never hard-block gRPC authz traffic.
756///
757/// **CRITICAL (RESEARCH Pitfall 1):** `governor::StateStore::measure_and_replace`
758/// is a *synchronous* trait method. This layer is deliberately NOT a
759/// `StateStore` implementation and never calls `block_on`.
760///
761/// No longer generic over the SurrealDB connection type: it holds a
762/// non-generic [`SharedRateLimitCounter`] (which owns the boxed store), which
763/// is why the SurrealDB handle only appears in [`Self::new`]'s signature.
764#[derive(Clone)]
765pub struct GrpcSharedRateLimitLayer {
766    counter: SharedRateLimitCounter,
767    scope: SharedScope,
768    trusted_hops: usize,
769}

771/// How a [`GrpcSharedRateLimitLayer`] derives `(bucket endpoint, window
772/// limit)` for a request.
773#[derive(Debug, Clone, Copy, PartialEq, Eq)]
774enum SharedScope {
775    /// One bucket for every request, whatever the method — the pre-I2 shape.
776    /// `limit` is already expressed in the 60-second window.
777    Uniform { endpoint: &'static str, limit: u32 },
778    /// One bucket per [`GrpcMethodFamily`] (I2), sized from per-second
779    /// ceilings and converted with [`per_sec_to_window_limit`].
780    PerFamily(GrpcRateLimits),
781}
782
783impl SharedScope {
784    /// `(endpoint, window limit)` for `path`.
785    ///
786    /// Every path resolves to a bucket — since residual 4 there is no family
787    /// that skips the shared counter.
788    fn resolve(self, path: &str) -> (&'static str, u32) {
789        match self {
790            Self::Uniform { endpoint, limit } => (endpoint, limit),
791            Self::PerFamily(limits) => {
792                let family = GrpcMethodFamily::classify(path);
793                (family.shared_endpoint(), limits.window_limit(family))
794            }
795        }
796    }
797}

915    scope: SharedScope,

919impl<S> Service<Request<tonic::body::Body>> for GrpcSharedRateLimitService<S>
920where
921    S: Service<Request<tonic::body::Body>, Response = Response<tonic::body::Body>>
922        + Clone
923        + Send
924        + 'static,
925    S::Future: Send + 'static,
926    S::Error: Send + 'static,
927{
928    type Response = Response<tonic::body::Body>;
929    type Error = S::Error;
930    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;
931
932    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
933        self.inner.poll_ready(cx)
934    }
935
936    fn call(&mut self, req: Request<tonic::body::Body>) -> Self::Future {
937        // Standard clone-and-swap so the returned future owns an
938        // independent, ready-to-call copy of the inner service (mirrors
939        // tower-http's convention for async-wrapping middlewares).
940        let clone = self.inner.clone();
941        let mut inner = std::mem::replace(&mut self.inner, clone);
942
943        // Cheap `Arc` clone — every request shares the ONE process-wide
944        // counter, never a fresh one (a per-request counter would count
945        // nothing).
946        let counter = self.counter.clone();
947        // I2: the bucket (and its ceiling) depend on which method family the
948        // request belongs to. Every family resolves to a bucket — reflection
949        // and health are counted too, against a generous one (residual 4).
950        let (endpoint, limit) = self.scope.resolve(req.uri().path());
951        let key_extractor = GrpcTrustedHopsKeyExtractor::new(self.trusted_hops);
952
953        // The rate-limit decision itself is now synchronous and taken BEFORE
954        // the future is even created — no datastore round trip on the path of
955        // this (server-wide!) layer. The returned future only awaits the inner
956        // service.
957        let ip = key_extractor.extract(&req).ok();
958        let allow = match ip {
959            Some(ip) => {
960                // Same bucket key shape as before — `{endpoint}:{ip}` — and
961                // the authz family keeps the `grpc_authz` endpoint name, so an
962                // in-flight upgrade keeps counting against the same
963                // `rate_limit_bucket` records.
964                let key = format!("{endpoint}:{ip}");
965                // `check_at` (not `check`): the counter derives the window
966                // itself AND sees the offset into it, which is what its
967                // sliding-window mode needs to bound every rolling minute
968                // rather than only the aligned ones (run-5 J1,
969                // `axiam_db::rate_limit_counter::WindowMode`).
970                counter.check_at(&key, Utc::now(), limit)
971            }
972            // No client-IP key available — fail open; the in-memory governor
973            // still makes the real decision. (A counter built with
974            // `AXIAM__RATE_LIMIT__SHARED=off` also always allows, from inside
975            // `check`.)
976            None => true,
977        };
978
979        Box::pin(async move {
980            if allow {
981                inner.call(req).await
982            } else {
983                Ok(too_many_requests_response())
984            }
985        })
986    }

1059    // CORR-01/D-02: sustained-throughput + monotonicity regression test
1060    // -----------------------------------------------------------------
1061    //
1062    // Drives sustained load against the SAME `governor::Quota::per_second`
1063    // construction `build_grpc_governor_layer` uses (see the production
1064    // function above) via a `FakeRelativeClock`, and asserts the permitted
1065    // count over a simulated window approximates the configured rate — the
1066    // only reliable way to distinguish a correctly-throttling governor from
1067    // one that is inverted (RESEARCH Pitfall 1: a naive "does the first
1068    // burst get through" smoke test looks identical either way).
1069    //
1070    // This test would FAIL against the old, inverted construction: with
1071    // `tower_governor`'s `.per_second(n)` treated as "n tokens/sec" (the
1072    // CQ-B44 bug), the replenish period becomes `n` SECONDS, so over a
1073    // 1-second simulated window virtually no additional tokens regenerate
1074    // past the initial burst — the sustained (post-burst) count would be
1075    // ~0, not approximately `n`.
1076    fn permitted_over_window(rate: u32, window: std::time::Duration) -> u64 {
1077        use governor::RateLimiter;
1078        use governor::clock::FakeRelativeClock;
1079
1080        let burst = NonZeroU32::new(rate).expect("rate >= 1");
1081        let quota = Quota::per_second(burst);
1082        let clock = FakeRelativeClock::default();
1083        let limiter = RateLimiter::direct_with_clock(quota, clock.clone());
1084
1085        // Drain the initial burst (the bucket starts full) — burst capacity
1086        // is a separate, already-asserted property (D-01: burst ==
1087        // authz_per_sec); it must NOT be counted as "sustained" throughput,
1088        // or a fully-inverted governor with a large burst could still look
1089        // like it's passing this test on the first drain alone.
1090        while limiter.check().is_ok() {}
1091
1092        // Now simulate sustained load: advance the fake clock in small
1093        // steps across the window, greedily draining whatever new tokens
1094        // have accumulated at each step — this proves tokens keep
1095        // replenishing throughout the window (sustained throughput), not
1096        // just once at t=0.
1097        let step = std::time::Duration::from_millis(5);
1098        let mut elapsed = std::time::Duration::ZERO;
1099        let mut permitted: u64 = 0;
1100        while elapsed < window {
1101            clock.advance(step);
1102            elapsed += step;
1103            while limiter.check().is_ok() {
1104                permitted += 1;
1105            }
1106        }
1107        permitted
1108    }
