1//! gRPC rate limiting via tower-governor (per D-10, D-11), brought to parity
2//! with the fixed REST limiter (SECHRD-03, D-01c).
3//!
4//! Two layers cooperate here (mirroring
5//! `axiam-api-rest::middleware::rate_limit_shared` / `extractors::rate_limit`):
6//!
7//! - [`GrpcTrustedHopsKeyExtractor`] — a custom `tower_governor::KeyExtractor`
8//!   that replaces `SmartIpKeyExtractor`. `SmartIpKeyExtractor` unconditionally
9//!   trusts the LEFTMOST `X-Forwarded-For` hop and, more fundamentally, can
10//!   never find tonic's real peer address at all (it only looks for an
11//!   `axum::extract::ConnectInfo<SocketAddr>` extension or a bare `SocketAddr`
12//!   extension — tonic inserts `TcpConnectInfo`/`TlsConnectInfo<TcpConnectInfo>`
13//!   instead). This extractor mirrors the fixed REST
14//!   `XForwardedForKeyExtractor` (plan 24-03): a configured `trusted_hops`
15//!   selects the rightmost trusted XFF hop, and when there are NOT enough
16//!   hops to trust, XFF is ignored entirely and the verified tonic connection
17//!   peer address is used instead (never `hops[0]` — SECHRD-03/D-01d).
18//! - [`GrpcSharedRateLimitLayer`] — a pre-check `tower::Layer` over the
19//!   process-wide write-behind `axiam_db::SharedRateLimitCounter` (the SAME
20//!   counter type and env knobs — `AXIAM__RATE_LIMIT__SHARED`,
21//!   `AXIAM__RATE_LIMIT__SHARED_SYNC_MS` — the REST `RateLimitShared`
22//!   middleware uses), run BEFORE the existing per-replica in-memory
23//!   `GovernorLayer` (kept byte-for-byte unchanged as the fail-open fallback,
24//!   D-01b). This is deliberately NOT a `governor::StateStore` impl and never
25//!   calls `block_on` — `StateStore::measure_and_replace` is a *synchronous*
26//!   trait method (RESEARCH Pitfall 1).
27//!
28//!   Since the H2 performance fix this layer performs NO datastore round trip
29//!   on the request path (it used to `await` one `UPSERT` per call, and being
30//!   server-wide, *every* gRPC call paid it — see the layer's own docs for the
31//!   measurements). A single background flusher coalesces the in-memory
32//!   increments into one write per bucket per sync interval.
33//!
34//! HARD CONSTRAINT (D-01c coordination note): the `Quota::per_second(...)
35//! .burst_size(...)` throughput/quota math in [`build_grpc_governor_layer`]
36//! is untouched by this module — CORR-01/Phase 26 owns it.
37//!
38//! # Units and scope (run-4 fixes I1 and I2)
39//!
40//! - **I1 — units.** The gRPC ceilings are configured **per second**; the
41//!   shared cross-replica bucket runs a fixed **60-second** window shared with
42//!   the REST limiter. The conversion lives in exactly one place,
43//!   [`per_sec_to_window_limit`], and the production constructor
44//!   ([`GrpcSharedRateLimitLayer::new_method_scoped`]) takes per-second
45//!   ceilings so no caller can hand a per-second number to a per-minute
46//!   window again.
47//! - **I2 — scope.** Both layers used to be server-wide, so the *authz*
48//!   ceiling throttled identity reads and admin traffic too. Buckets are now
49//!   split per [`GrpcMethodFamily`] (authz-check / identity-read / admin /
50//!   infrastructure).
51//!
52//! # SEC-079 — the admin ceiling is absolute, not derived
53//!
54//! The admin family contains `axiam.v1.UserService/ValidateCredentials`, an
55//! **Argon2id verification**. Its ceiling is a CPU guard, so it is an absolute
56//! constant ([`ADMIN_PER_SEC_DEFAULT`]) rather than a multiple of — or the
57//! same number as — the read-sized `authz_per_sec` base. See that constant's
58//! docs for the full reasoning; the short version is that when the I1 units
59//! bug was fixed the effective per-IP ceiling on that RPC rose 60x, and
60//! deriving it from a read-sized base would let a `gateway`/`mesh` posture
61//! raise it further still.

91/// Reads the SAME `AXIAM__RATE_LIMIT__TRUSTED_HOPS` env var the REST shared
92/// pre-check (`axiam-api-rest::middleware::rate_limit_shared::trusted_hops`)
93/// and `server.rs::build_governor` use, so the gRPC key and the REST key are
94/// derived identically from the same deployment-topology configuration.
95///
96/// `pub(crate)` so `server.rs::start_grpc_server` can pass the SAME value to
97/// both [`build_grpc_governor_layer`] (via this function internally) and
98/// [`GrpcSharedRateLimitLayer::new`] — the key-extraction logic in the
99/// shared-store pre-check and the in-memory governor MUST stay in lockstep.
100pub(crate) fn trusted_hops_from_env() -> usize {
101    std::env::var("AXIAM__RATE_LIMIT__TRUSTED_HOPS")
102        .ok()
103        .and_then(|v| v.parse().ok())
104        .unwrap_or(0)
105}

107/// Extracts the gRPC caller's IP from `X-Forwarded-For`, falling back to the
108/// verified tonic connection peer address (`TcpConnectInfo`/
109/// `TlsConnectInfo<TcpConnectInfo>`).
110///
111/// `trusted_hops` controls how many rightmost XFF entries are trusted
112/// reverse-proxy hops to skip (mirrors
113/// `axiam-api-rest::extractors::rate_limit::XForwardedForKeyExtractor`).
114/// When `trusted_hops >= hops.len()`, the header is NOT trusted at all and
115/// XFF is ignored entirely — the key is derived from the verified peer
116/// address instead (a client cannot manufacture extra hops to force a
117/// fallback to an attacker-controlled `hops[0]` — SECHRD-03/D-01d).
118#[derive(Debug, Clone, Default)]
119pub struct GrpcTrustedHopsKeyExtractor {
120    pub trusted_hops: usize,
121}
122
123impl GrpcTrustedHopsKeyExtractor {
124    /// Create an extractor keyed by `trusted_hops` trusted reverse-proxy
125    /// hops (0 = no trusted hops; XFF is only ever used when it has more
126    /// than 0 entries beyond what's trusted).
127    pub fn new(trusted_hops: usize) -> Self {
128        Self { trusted_hops }
129    }
130}
131
132impl KeyExtractor for GrpcTrustedHopsKeyExtractor {
133    type Key = IpAddr;
134
135    fn extract<T>(&self, req: &Request<T>) -> Result<Self::Key, GovernorError> {
136        if let Some(forwarded_for) = req.headers().get("x-forwarded-for")
137            && let Ok(val) = forwarded_for.to_str()
138        {
139            let hops: Vec<&str> = val.split(',').map(str::trim).collect();
140            // Select the rightmost-untrusted hop (parity with REST).
141            if self.trusted_hops < hops.len() {
142                let idx = hops.len() - 1 - self.trusted_hops;
143                if let Ok(ip) = hops[idx].parse::<IpAddr>() {
144                    return Ok(ip);
145                }
146            }
147            // Fewer hops than trusted_hops requires: the header cannot be
148            // trusted. Fall through to the verified peer address below
149            // instead of indexing into XFF (SECHRD-03 — no `hops[0]`
150            // fallback that would let a rotating XFF mint a fresh bucket).
151        }
152
153        grpc_peer_addr(req)
154            .map(|addr| addr.ip())
155            .ok_or(GovernorError::UnableToExtractKey)
156    }
157}
158
159/// Reads the verified tonic connection peer address from request
160/// extensions, mirroring `tonic::Request::remote_addr()`'s own lookup
161/// (`TcpConnectInfo` for plaintext, `TlsConnectInfo<TcpConnectInfo>` for TLS)
162/// — the extension types tonic's `ConnectInfoLayer` actually inserts (NOT
163/// the `axum::extract::ConnectInfo<SocketAddr>` / bare `SocketAddr`
164/// extensions `SmartIpKeyExtractor` looks for, which tonic never sets).
165fn grpc_peer_addr<T>(req: &Request<T>) -> Option<SocketAddr> {
166    req.extensions()
167        .get::<TcpConnectInfo>()
168        .and_then(TcpConnectInfo::remote_addr)
169        .or_else(|| {
170            req.extensions()
171                .get::<TlsConnectInfo<TcpConnectInfo>>()
172                .and_then(|info| info.get_ref().remote_addr())
173        })
174}

180/// The rate-limit bucket family a gRPC method belongs to (I2).
181///
182/// **Why this exists.** Both gRPC rate-limit layers used to be *server-wide*:
183/// one bucket, sized from `AXIAM__GRPC__GRPC_AUTHZ_PER_SEC`, shared by every
184/// method on every service. Run 4 measured `UserInfoService/GetUserInfo` — an
185/// identity read that sustains ~12 700/s — collapsing to the *authz* ceiling
186/// under production posture, because a `check_access` sizing decision was
187/// silently also a userinfo sizing decision
188/// (`claude_dev/improvement-after-run4-benchmark.md` I2). Splitting the
189/// buckets lets an operator size the authz path (called once per protected
190/// request) independently of identity reads and of administrative traffic.
191///
192/// **Classification is by gRPC path**, i.e. `/<package>.<Service>/<Method>`,
193/// which is the only identity available to a `tower::Layer` running before
194/// tonic resolves per-RPC claims (the same structural reason the buckets key
195/// on IP — see `crate::config::GrpcRateLimitKeyMode`).
196#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
197pub enum GrpcMethodFamily {
198    /// `axiam.v1.AuthorizationService` — `CheckAccess` / `BatchCheckAccess`.
199    /// Also the family for any **unrecognized** path: an unknown method must
200    /// land in the strictest bucket rather than in the unlimited one.
201    AuthzCheck,
202    /// Identity reads: `axiam.v1.UserInfoService` (OIDC-style self lookup)
203    /// and `axiam.v1.TokenService` (validate/introspect). High-frequency,
204    /// cheap, and measured an order of magnitude faster than an authz check.
205    IdentityRead,
206    /// Administrative / user-management traffic: `axiam.v1.UserService`
207    /// (`GetUser`, `ValidateCredentials`). `ValidateCredentials` hashes a
208    /// password, so this family is deliberately NOT sized from read capacity —
209    /// its ceiling is the absolute [`ADMIN_PER_SEC_DEFAULT`] (SEC-079).
210    Admin,
211    /// Infrastructure probes: gRPC **reflection** and **health**. Their whole
212    /// job is to answer during an incident, when the limited families are most
213    /// likely to be saturated, so this family gets a deliberately generous
214    /// ceiling ([`INFRA_PER_SEC`]) — but a **finite** one.
215    ///
216    /// This family used to be a literal pass-through (`Unlimited`) that
217    /// bypassed BOTH limiter layers, selected by a prefix test on the
218    /// attacker-controlled `:path`. Neither service is registered today, so
219    /// the practical effect was unmetered HTTP/2 stream churn rather than DB
220    /// work — but registering a health service would have turned it into a
221    /// genuinely unmetered endpoint. A generous bucket keeps probes working
222    /// during an overload without leaving an unbounded surface behind
223    /// (security re-verification 2026-08-03, residual 4).
224    Infra,
225}
226
227/// gRPC reflection service package prefix (v1 and v1alpha).
228const REFLECTION_PREFIX: &str = "/grpc.reflection.";
229/// gRPC health-checking service package prefix.
230const HEALTH_PREFIX: &str = "/grpc.health.";
231
232impl GrpcMethodFamily {
233    /// Classifies a gRPC request path (`/<package>.<Service>/<Method>`).
234    ///
235    /// Unknown paths deliberately fall into [`Self::AuthzCheck`] — the
236    /// strictest limited family — so that adding a service without updating
237    /// this function fails safe (throttled) rather than open (unlimited).
238    pub fn classify(path: &str) -> Self {
239        if path.starts_with(REFLECTION_PREFIX) || path.starts_with(HEALTH_PREFIX) {
240            return Self::Infra;
241        }
242        match path.split('/').nth(1).unwrap_or_default() {
243            "axiam.v1.UserInfoService" | "axiam.v1.TokenService" => Self::IdentityRead,
244            "axiam.v1.UserService" => Self::Admin,
245            // "axiam.v1.AuthorizationService" and everything else.
246            _ => Self::AuthzCheck,
247        }
248    }
249
250    /// Shared-counter bucket-name prefix for this family.
251    ///
252    /// These names are part of the persisted `rate_limit_bucket` key space
253    /// (`"{endpoint}:{ip}"`), so `grpc_authz` is kept byte-for-byte for the
254    /// authz family — an in-flight upgrade keeps counting against the same
255    /// rows.
256    ///
257    /// Every family now has one: the infrastructure family is bucketed like
258    /// the rest rather than skipping the counter entirely (residual 4).
259    pub const fn shared_endpoint(self) -> &'static str {
260        match self {
261            Self::AuthzCheck => "grpc_authz",
262            Self::IdentityRead => "grpc_identity",
263            Self::Admin => "grpc_admin",
264            Self::Infra => "grpc_infra",
265        }
266    }
267}

269/// Per-family gRPC ceilings, in requests **per second per IP** (I2).
270///
271/// The unit is per-second at every boundary of this type. The shared
272/// cross-replica layer runs a 60-second window and therefore converts with
273/// [`per_sec_to_window_limit`] internally — callers never do that arithmetic
274/// themselves, which is precisely the bug I1 fixed (the per-second number was
275/// handed to a per-minute window verbatim, making the effective ceiling
276/// 1/60th of the configured one).
277#[derive(Debug, Clone, Copy, PartialEq, Eq)]
278pub struct GrpcRateLimits {
279    /// `AXIAM__GRPC__GRPC_AUTHZ_PER_SEC` — authorization decisions.
280    pub authz_per_sec: u32,
281    /// `AXIAM__GRPC__GRPC_IDENTITY_PER_SEC` — userinfo / token reads.
282    pub identity_per_sec: u32,
283    /// `AXIAM__GRPC__GRPC_ADMIN_PER_SEC` — user management. Defaults to the
284    /// absolute [`ADMIN_PER_SEC_DEFAULT`], never to a multiple of
285    /// [`Self::authz_per_sec`] (SEC-079).
286    pub admin_per_sec: u32,
287}
288
289/// Identity reads default to this multiple of the authz ceiling — they were
290/// measured ~14x faster than an authz check (12 665/s vs 887/s, run 4) and a
291/// resource server does one per protected request just like a check. The
292/// multiplier (rather than a fixed number) is what keeps the family coherent
293/// when `AXIAM__RATE_LIMIT__PROFILE` raises the authz ceiling.
294pub const IDENTITY_PER_SEC_MULTIPLE: u32 = 5;
295
296/// Default ceiling for the [`GrpcMethodFamily::Admin`] family, in requests per
297/// second per IP — **an absolute number, deliberately not derived from
298/// [`GrpcRateLimits::authz_per_sec`]** (SEC-079).
299///
300/// **Why absolute.** The admin family contains
301/// `axiam.v1.UserService/ValidateCredentials`, which performs a real Argon2id
302/// `verify_password` (~19 MiB of memory arena per verification). Its ceiling is
303/// therefore a **CPU/memory guard on an online-password-guessing surface**, not
304/// a throughput ceiling on a read. `authz_per_sec` is a read-sized base (100/s
305/// shipped, and `AXIAM__RATE_LIMIT__PROFILE=gateway`/`mesh` raise it to
306/// 1 000/s / 5 000/s); deriving the admin ceiling from it — even 1:1 — makes a
307/// service-mesh capacity decision silently also a credential-guessing-breadth
308/// decision. Decoupling the two is the entire point of this constant.
309///
310/// **Why 10/s (600/min per IP).** The family holds exactly two RPCs, `GetUser`
311/// and `ValidateCredentials`; the high-volume identity read is `GetUserInfo` on
312/// `UserInfoService`, which lives in [`GrpcMethodFamily::IdentityRead`] and is
313/// unaffected. 600 administrative calls per minute from one client IP is well
314/// above any real admin console or M2M provisioning loop, and roughly an order
315/// of magnitude below the point at which concurrent Argon2id verifications
316/// become the server's dominant cost. It also restores the protection the I1
317/// units fix accidentally removed: before that fix the shared window enforced
318/// the per-second number against a 60-second window, so the deployed ceiling
319/// was ~100/**minute**; correcting the units silently raised it to
320/// ~6 000/minute. 600/minute is the CPU-appropriate value the doc comment
321/// always claimed this family had.
322///
323/// Operators who genuinely need more can still pin
324/// `AXIAM__GRPC__GRPC_ADMIN_PER_SEC` — explicit configuration always wins.
325pub const ADMIN_PER_SEC_DEFAULT: u32 = 10;
326
327/// Ceiling for the [`GrpcMethodFamily::Infra`] family (gRPC reflection and
328/// health), in requests per second per IP — generous, but finite.
329///
330/// Infrastructure probes are intrinsically low-rate: a Kubernetes liveness or
331/// readiness probe runs on the order of once every few seconds per prober, and
332/// even a large fleet of sidecars sharing one NAT egress IP stays orders of
333/// magnitude below 100/s. So this ceiling can never be the thing that breaks a
334/// health check during an incident — which is the property that made the family
335/// a pass-through in the first place — while still bounding the surface, so a
336/// health service registered later is not an unmetered endpoint reachable by
337/// prefix-matching an attacker-controlled `:path` (residual 4).
338///
339/// It is an absolute constant for the same reason as
340/// [`ADMIN_PER_SEC_DEFAULT`]: probe volume has nothing to do with authz
341/// capacity, so a posture preset must not move it.
342pub const INFRA_PER_SEC: u32 = 100;
343
344impl GrpcRateLimits {
345    /// Derives the whole family from the authz ceiling alone — the shape an
346    /// operator gets when only `AXIAM__GRPC__GRPC_AUTHZ_PER_SEC` (or a
347    /// posture preset) is set.
348    ///
349    /// Only [`Self::identity_per_sec`] actually scales: `ValidateCredentials`
350    /// is an Argon2id verification, so the admin ceiling is a CPU guard and
351    /// takes the absolute [`ADMIN_PER_SEC_DEFAULT`] rather than inheriting a
352    /// read-sized base (SEC-079).
353    pub const fn from_authz_per_sec(authz_per_sec: u32) -> Self {
354        Self {
355            authz_per_sec,
356            identity_per_sec: authz_per_sec.saturating_mul(IDENTITY_PER_SEC_MULTIPLE),
357            admin_per_sec: ADMIN_PER_SEC_DEFAULT,
358        }
359    }
360
361    /// The per-second ceiling for `family`.
362    ///
363    /// Every family is limited: [`GrpcMethodFamily::Infra`] takes the fixed
364    /// [`INFRA_PER_SEC`], which is why this is not configurable per instance
365    /// (residual 4).
366    pub const fn per_sec(self, family: GrpcMethodFamily) -> u32 {
367        match family {
368            GrpcMethodFamily::AuthzCheck => self.authz_per_sec,
369            GrpcMethodFamily::IdentityRead => self.identity_per_sec,
370            GrpcMethodFamily::Admin => self.admin_per_sec,
371            GrpcMethodFamily::Infra => INFRA_PER_SEC,
372        }
373    }
374
375    /// The ceiling for `family` expressed in the shared layer's 60-second
376    /// window (I1).
377    pub const fn window_limit(self, family: GrpcMethodFamily) -> u32 {
378        per_sec_to_window_limit(self.per_sec(family))
379    }
380}

395/// Build a [`GovernorLayer`] for gRPC server-wide rate limiting.
396///
397/// The layer uses token-bucket semantics (via the `governor` crate) with
398/// `authz_per_sec` tokens replenished per second and a burst allowance equal
399/// to `authz_per_sec` (one second's worth of tokens). For service-mesh
400/// patterns where the authz endpoint is called on every request, the default
401/// of 100 tokens/sec is intentionally generous.
402///
403/// # Panics
404///
405/// Panics at startup if `authz_per_sec` is 0 — the governor crate requires a
406/// non-zero burst size.
407pub fn build_grpc_governor_layer(authz_per_sec: u32) -> GrpcGovernorLayer {
408    assert!(authz_per_sec >= 1, "grpc_authz_per_sec must be >= 1");
409
410    // CORR-01 (D-01): the PREVIOUS "fix" here (CQ-B44) called
411    // `tower_governor`'s OWN `GovernorConfigBuilder::per_second(n)` /
412    // `.burst_size(authz_per_sec * 2)` as if `.per_second(n)` meant "n tokens
413    // per second." It does NOT — `tower_governor`'s builder methods set the
414    // replenish PERIOD (i.e. "one token every n seconds"), so with the
415    // default `grpc_authz_per_sec=100` the governor replenished 1 token every
416    // 100 seconds, and raising the configured rate made throughput WORSE, not
417    // better. Fixed by bypassing `tower_governor`'s confusing
418    // `.per_second()`/`.per_millisecond()` convenience methods entirely and
419    // constructing the underlying `governor::Quota` directly via
420    // `Quota::per_second`, whose `replenish_interval()` is `1s /
421    // authz_per_sec` and whose `burst_size()` is `authz_per_sec` — then
422    // feeding that quota into the builder via `.const_period()`/
423    // `.const_burst_size()`. Burst is `authz_per_sec` (not `* 2`), per D-01.
424    //
425    // SECHRD-03/D-01c: key extractor is GrpcTrustedHopsKeyExtractor for
426    // IP-spoofing resistance parity with REST — unrelated to this quota math.
427    let burst = NonZeroU32::new(authz_per_sec).expect("authz_per_sec >= 1 asserted above");
428    let quota = Quota::per_second(burst);
429
430    let config = Arc::new(
431        GovernorConfigBuilder::default()
432            .const_period(quota.replenish_interval())
433            .const_burst_size(quota.burst_size().get())
434            .key_extractor(GrpcTrustedHopsKeyExtractor::new(trusted_hops_from_env()))
435            .finish()
436            .expect("valid GovernorConfig for gRPC rate limiter"),
437    );
438
439    GovernorLayer::new(config)
440}

442/// Build the per-method-family in-memory governor stack (I2).
443///
444/// One independent [`GovernorLayer`] per [`GrpcMethodFamily`], each with its
445/// own token bucket keyed by client IP. Replaces the single server-wide
446/// governor, which made the authz ceiling the ceiling for every gRPC surface.
447///
448/// Since residual 4 the infrastructure family (reflection / health) gets a
449/// governor too — a generous [`INFRA_PER_SEC`] one — instead of a pass-through
450/// that bypassed both limiter layers.
451///
452/// # Panics
453///
454/// Panics at startup if any family's ceiling is 0 — the governor crate
455/// requires a non-zero burst size (same contract as
456/// [`build_grpc_governor_layer`]).
457pub fn build_grpc_method_scoped_governor_layer(
458    limits: GrpcRateLimits,
459) -> GrpcMethodScopedGovernorLayer {
460    GrpcMethodScopedGovernorLayer {
461        authz: build_grpc_governor_layer(limits.per_sec(GrpcMethodFamily::AuthzCheck)),
462        identity: build_grpc_governor_layer(limits.per_sec(GrpcMethodFamily::IdentityRead)),
463        admin: build_grpc_governor_layer(limits.per_sec(GrpcMethodFamily::Admin)),
464        infra: build_grpc_governor_layer(limits.per_sec(GrpcMethodFamily::Infra)),
465    }
466}

491/// Dispatches each request to the governor for its [`GrpcMethodFamily`].
492///
493/// All four branches wrap clones of the SAME inner service, so this is a
494/// routing decision over rate-limit state only — the request itself reaches
495/// the identical tonic router whichever branch it takes. There is no
496/// pass-through branch: the infrastructure family has its own (generous)
497/// governor rather than bypassing the layer (residual 4).
498#[derive(Clone)]
499pub struct GrpcMethodScopedGovernor<S> {
500    authz: GrpcGovernor<S>,
501    identity: GrpcGovernor<S>,
502    admin: GrpcGovernor<S>,
503    infra: GrpcGovernor<S>,
504}
505
506impl<S> Service<Request<tonic::body::Body>> for GrpcMethodScopedGovernor<S>
507where
508    S: Service<Request<tonic::body::Body>, Response = Response<tonic::body::Body>>
509        + Clone
510        + Send
511        + 'static,
512    S::Future: Send + 'static,
513    S::Error: Send + 'static,
514{
515    type Response = Response<tonic::body::Body>;
516    type Error = S::Error;
517    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;
518
519    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
520        // Every branch must be ready before we can promise `call` will
521        // succeed, since the family is only known once the request arrives.
522        // In practice all four wrap clones of the same always-ready tonic
523        // router, so this is a formality that keeps the tower contract exact.
524        std::task::ready!(Service::poll_ready(&mut self.authz, cx))?;
525        std::task::ready!(Service::poll_ready(&mut self.identity, cx))?;
526        std::task::ready!(Service::poll_ready(&mut self.admin, cx))?;
527        Service::poll_ready(&mut self.infra, cx)
528    }
529
530    fn call(&mut self, req: Request<tonic::body::Body>) -> Self::Future {
531        // Standard clone-and-swap per branch so the returned future owns an
532        // independent, ready-to-call copy (same convention as
533        // `GrpcSharedRateLimitService::call`).
534        match GrpcMethodFamily::classify(req.uri().path()) {
535            GrpcMethodFamily::AuthzCheck => {
536                let clone = self.authz.clone();
537                let mut svc = std::mem::replace(&mut self.authz, clone);
538                Box::pin(async move { svc.call(req).await })
539            }
540            GrpcMethodFamily::IdentityRead => {
541                let clone = self.identity.clone();
542                let mut svc = std::mem::replace(&mut self.identity, clone);
543                Box::pin(async move { svc.call(req).await })
544            }
545            GrpcMethodFamily::Admin => {
546                let clone = self.admin.clone();
547                let mut svc = std::mem::replace(&mut self.admin, clone);
548                Box::pin(async move { svc.call(req).await })
549            }
550            GrpcMethodFamily::Infra => {
551                let clone = self.infra.clone();
552                let mut svc = std::mem::replace(&mut self.infra, clone);
553                Box::pin(async move { svc.call(req).await })
554            }
555        }
556    }
557}

563/// Fixed-window duration (seconds) for the shared bucket — same value and
564/// rationale as the REST shared pre-check
565/// (`axiam-api-rest::middleware::rate_limit_shared::WINDOW_SECS`): a simple
566/// fixed-window counter is acceptable here since this layer only needs to be
567/// *approximately* right (it fails open by design).
568pub const WINDOW_SECS: i64 = 60;
569
570/// Converts a per-**second** ceiling into the budget for one
571/// [`WINDOW_SECS`]-second shared window (I1).
572///
573/// **This is the units bug this constant exists to prevent.** The gRPC
574/// ceiling (`AXIAM__GRPC__GRPC_AUTHZ_PER_SEC`) is per second; the shared
575/// cross-replica bucket is a fixed 60-second window shared with the REST
576/// limiter, whose knobs are per minute. Before I1 the per-second number was
577/// handed to [`GrpcSharedRateLimitLayer::new`] verbatim, so the shared layer
578/// enforced "N per 60 s" while the operator had configured "N per second" —
579/// and since the stricter of the two cooperating layers wins, the effective
580/// ceiling was **1/60th** of the configured one. Run 4 measured exactly that:
581/// with `GRPC_AUTHZ_PER_SEC=100`, all three gRPC scenarios admitted ~100 ops
582/// per minute (`claude_dev/improvement-after-run4-benchmark.md` I1).
583///
584/// The multiply saturates: an operator who configures a per-second ceiling
585/// above `u32::MAX / 60` gets `u32::MAX` for the window (effectively
586/// unlimited, which is what they asked for) instead of a wrapped — and
587/// therefore *tiny* — budget.
588///
589/// The alternative fix, a per-second shared window, was rejected: it would
590/// fork the `rate_limit_bucket` schema away from the REST limiter's and
591/// change the write-behind flusher cadence, for no enforcement benefit.
592pub const fn per_sec_to_window_limit(per_sec: u32) -> u32 {
593    per_sec.saturating_mul(WINDOW_SECS as u32)
594}
595
596/// Truncates `now` down to the start of the current fixed
597/// [`WINDOW_SECS`]-second window.
598///
599/// No longer on the request path — [`GrpcSharedRateLimitService::call`] hands
600/// the raw `now` to [`axiam_db::SharedRateLimitCounter::check_at`] instead
601/// (run-5 J1). Kept, and asserted by a test, because [`WINDOW_SECS`] must
602/// keep agreeing with the counter's configured window.
603#[cfg_attr(not(test), allow(dead_code))]
604fn window_start(now: DateTime<Utc>) -> DateTime<Utc> {
605    let epoch = now.timestamp();
606    let start_epoch = epoch - epoch.rem_euclid(WINDOW_SECS);
607    DateTime::<Utc>::from_timestamp(start_epoch, 0).unwrap_or(now)
608}

610/// Builds a gRPC `RESOURCE_EXHAUSTED` response — the same status the
611/// in-memory `GovernorLayer` returns for `GovernorError::TooManyRequests`
612/// (see `tower_governor::errors::GovernorError`'s `Response<tonic::body::Body>`
613/// conversion) — so clients see one consistent rate-limit contract
614/// regardless of which layer rejected the request.
615fn too_many_requests_response() -> Response<tonic::body::Body> {
616    tonic::Status::resource_exhausted("rate limit exceeded").into_http()
617}

619/// Shared (cross-replica) rate-limit pre-check `tower::Layer` for the gRPC
620/// path (SECHRD-03 / D-01a, D-01b, D-01c, D-01d).
621///
622/// Consults the process-wide write-behind [`SharedRateLimitCounter`]
623/// (`axiam-db::rate_limit_counter`) — the SAME counter type and the SAME env
624/// knobs the REST `RateLimitShared` middleware uses, so both listeners behave
625/// identically. There is no reimplemented counter here. Wire this layer BEFORE
626/// (i.e. `.layer()` it FIRST — tower's `ServiceBuilder`/`Server::builder()`
627/// executes the FIRST-added layer with the request FIRST, the opposite of
628/// actix's last-`.wrap()`-is-outermost rule) the existing
629/// [`build_grpc_governor_layer`] `GovernorLayer` on the same
630/// `Server::builder()`, e.g.:
631///
632/// ```ignore
633/// Server::builder()
634///     .layer(GrpcSharedRateLimitLayer::new(db, "grpc_authz", grpc_config.grpc_authz_per_sec, trusted_hops))
635///     .layer(build_grpc_governor_layer(grpc_config.grpc_authz_per_sec))
636///     .add_service(authz_svc)
637/// ```
638///
639/// # No synchronous datastore write on the request path (H2 fix)
640///
641/// This layer used to `await` one SurrealDB `UPSERT` per call. Because it is a
642/// **server-wide** layer on the tonic listener, EVERY gRPC call paid that
643/// write — measured at ~39–46 ms of datastore-attributed latency, pinning the
644/// listener at ~20 ops/s regardless of concurrency
645/// (`claude_dev/postseed-transient-investigation.md` §1, §2, §7). It now pays
646/// only an in-memory sharded-map increment
647/// ([`SharedRateLimitCounter::check`], fully synchronous), while a single
648/// background flusher coalesces the increments into one datastore write per
649/// bucket per `AXIAM__RATE_LIMIT__SHARED_SYNC_MS`. Cross-replica enforcement
650/// is therefore eventual, with the overshoot bound stated precisely in the
651/// [`axiam_db::rate_limit_counter`] module docs; the in-memory governor still
652/// makes a full per-replica decision on the same traffic.
653///
654/// Everything else is unchanged: the same `{endpoint}:{ip}` bucket key, the
655/// same `GrpcTrustedHopsKeyExtractor` key derivation (D-01d), and the same
656/// `RESOURCE_EXHAUSTED` rejection.
657///
658/// **Fail-open (D-01b, T-24-73 accepted risk):** when no client IP can be
659/// extracted, or the shared layer is disabled via
660/// `AXIAM__RATE_LIMIT__SHARED=off`, this layer forwards the request unchanged
661/// so the existing in-memory governor makes the decision instead. Store
662/// outages are surfaced by the counter's flusher (a `warn` alarm that never
663/// includes the raw key) while decisions keep being served from local counts —
664/// a counter-store outage must never hard-block gRPC authz traffic.
665///
666/// **CRITICAL (RESEARCH Pitfall 1):** `governor::StateStore::measure_and_replace`
667/// is a *synchronous* trait method. This layer is deliberately NOT a
668/// `StateStore` implementation and never calls `block_on`.
669///
670/// No longer generic over the SurrealDB connection type: it holds a
671/// non-generic [`SharedRateLimitCounter`] (which owns the boxed store), which
672/// is why the SurrealDB handle only appears in [`Self::new`]'s signature.
673#[derive(Clone)]
674pub struct GrpcSharedRateLimitLayer {
675    counter: SharedRateLimitCounter,
676    scope: SharedScope,
677    trusted_hops: usize,
678}
679
680/// How a [`GrpcSharedRateLimitLayer`] derives `(bucket endpoint, window
681/// limit)` for a request.
682#[derive(Debug, Clone, Copy, PartialEq, Eq)]
683enum SharedScope {
684    /// One bucket for every request, whatever the method — the pre-I2 shape.
685    /// `limit` is already expressed in the 60-second window.
686    Uniform { endpoint: &'static str, limit: u32 },
687    /// One bucket per [`GrpcMethodFamily`] (I2), sized from per-second
688    /// ceilings and converted with [`per_sec_to_window_limit`].
689    PerFamily(GrpcRateLimits),
690}
691
692impl SharedScope {
693    /// `(endpoint, window limit)` for `path`.
694    ///
695    /// Every path resolves to a bucket — since residual 4 there is no family
696    /// that skips the shared counter.
697    fn resolve(self, path: &str) -> (&'static str, u32) {
698        match self {
699            Self::Uniform { endpoint, limit } => (endpoint, limit),
700            Self::PerFamily(limits) => {
701                let family = GrpcMethodFamily::classify(path);
702                (family.shared_endpoint(), limits.window_limit(family))
703            }
704        }
705    }
706}

828impl<S> Service<Request<tonic::body::Body>> for GrpcSharedRateLimitService<S>
829where
830    S: Service<Request<tonic::body::Body>, Response = Response<tonic::body::Body>>
831        + Clone
832        + Send
833        + 'static,
834    S::Future: Send + 'static,
835    S::Error: Send + 'static,
836{
837    type Response = Response<tonic::body::Body>;
838    type Error = S::Error;
839    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;
840
841    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
842        self.inner.poll_ready(cx)
843    }
844
845    fn call(&mut self, req: Request<tonic::body::Body>) -> Self::Future {
846        // Standard clone-and-swap so the returned future owns an
847        // independent, ready-to-call copy of the inner service (mirrors
848        // tower-http's convention for async-wrapping middlewares).
849        let clone = self.inner.clone();
850        let mut inner = std::mem::replace(&mut self.inner, clone);
851
852        // Cheap `Arc` clone — every request shares the ONE process-wide
853        // counter, never a fresh one (a per-request counter would count
854        // nothing).
855        let counter = self.counter.clone();
856        // I2: the bucket (and its ceiling) depend on which method family the
857        // request belongs to. Every family resolves to a bucket — reflection
858        // and health are counted too, against a generous one (residual 4).
859        let (endpoint, limit) = self.scope.resolve(req.uri().path());
860        let key_extractor = GrpcTrustedHopsKeyExtractor::new(self.trusted_hops);
861
862        // The rate-limit decision itself is now synchronous and taken BEFORE
863        // the future is even created — no datastore round trip on the path of
864        // this (server-wide!) layer. The returned future only awaits the inner
865        // service.
866        let ip = key_extractor.extract(&req).ok();
867        let allow = match ip {
868            Some(ip) => {
869                // Same bucket key shape as before — `{endpoint}:{ip}` — and
870                // the authz family keeps the `grpc_authz` endpoint name, so an
871                // in-flight upgrade keeps counting against the same
872                // `rate_limit_bucket` records.
873                let key = format!("{endpoint}:{ip}");
874                // `check_at` (not `check`): the counter derives the window
875                // itself AND sees the offset into it, which is what its
876                // sliding-window mode needs to bound every rolling minute
877                // rather than only the aligned ones (run-5 J1,
878                // `axiam_db::rate_limit_counter::WindowMode`).
879                counter.check_at(&key, Utc::now(), limit)
880            }
881            // No client-IP key available — fail open; the in-memory governor
882            // still makes the real decision. (A counter built with
883            // `AXIAM__RATE_LIMIT__SHARED=off` also always allows, from inside
884            // `check`.)
885            None => true,
886        };
887
888        Box::pin(async move {
889            if allow {
890                inner.call(req).await
891            } else {
892                Ok(too_many_requests_response())
893            }
894        })
895    }

969    // -----------------------------------------------------------------
970    //
971    // Drives sustained load against the SAME `governor::Quota::per_second`
972    // construction `build_grpc_governor_layer` uses (see the production
973    // function above) via a `FakeRelativeClock`, and asserts the permitted
974    // count over a simulated window approximates the configured rate — the
975    // only reliable way to distinguish a correctly-throttling governor from
976    // one that is inverted (RESEARCH Pitfall 1: a naive "does the first
977    // burst get through" smoke test looks identical either way).
978    //
979    // This test would FAIL against the old, inverted construction: with
980    // `tower_governor`'s `.per_second(n)` treated as "n tokens/sec" (the
981    // CQ-B44 bug), the replenish period becomes `n` SECONDS, so over a
982    // 1-second simulated window virtually no additional tokens regenerate
983    // past the initial burst — the sustained (post-burst) count would be
984    // ~0, not approximately `n`.
985    fn permitted_over_window(rate: u32, window: std::time::Duration) -> u64 {
986        use governor::RateLimiter;
987        use governor::clock::FakeRelativeClock;
988
989        let burst = NonZeroU32::new(rate).expect("rate >= 1");
990        let quota = Quota::per_second(burst);
991        let clock = FakeRelativeClock::default();
992        let limiter = RateLimiter::direct_with_clock(quota, clock.clone());
993
994        // Drain the initial burst (the bucket starts full) — burst capacity
995        // is a separate, already-asserted property (D-01: burst ==
996        // authz_per_sec); it must NOT be counted as "sustained" throughput,
997        // or a fully-inverted governor with a large burst could still look
998        // like it's passing this test on the first drain alone.
999        while limiter.check().is_ok() {}
1000
1001        // Now simulate sustained load: advance the fake clock in small
1002        // steps across the window, greedily draining whatever new tokens
1003        // have accumulated at each step — this proves tokens keep
1004        // replenishing throughout the window (sustained throughput), not
1005        // just once at t=0.
1006        let step = std::time::Duration::from_millis(5);
1007        let mut elapsed = std::time::Duration::ZERO;
1008        let mut permitted: u64 = 0;
1009        while elapsed < window {
1010            clock.advance(step);
1011            elapsed += step;
1012            while limiter.check().is_ok() {
1013                permitted += 1;
1014            }
1015        }
1016        permitted
1017    }

1232    /// **SEC-079.** The admin ceiling is a CPU guard on an Argon2id RPC and
1233    /// must NOT scale with the read-sized authz base — otherwise a
1234    /// `gateway`/`mesh` posture (or any operator raising the authz ceiling for
1235    /// service-mesh throughput) silently widens online password guessing.
1236    #[test]
1237    fn admin_ceiling_never_derives_from_the_authz_ceiling() {
1238        for authz in [1, 100, 1_000, 5_000, 60_000, u32::MAX] {
1239            let limits = GrpcRateLimits::from_authz_per_sec(authz);
1240            assert_eq!(
1241                limits.admin_per_sec, ADMIN_PER_SEC_DEFAULT,
1242                "authz={authz}: the admin ceiling must stay at the absolute default \
1243                 ({ADMIN_PER_SEC_DEFAULT}/s); deriving it from a read-sized base is SEC-079"
1244            );
1245            assert_eq!(
1246                limits.per_sec(GrpcMethodFamily::Admin),
1247                ADMIN_PER_SEC_DEFAULT
1248            );
1249        }
1250
1251        // And the absolute value is the CPU-appropriate one: 10/s = 600/min
1252        // per IP through the 60-second shared window.
1253        assert_eq!(ADMIN_PER_SEC_DEFAULT, 10);
1254        assert_eq!(
1255            GrpcRateLimits::from_authz_per_sec(100).window_limit(GrpcMethodFamily::Admin),
1256            600,
1257            "600 credential checks per minute per IP — not the 6 000 the units fix \
1258             would have produced from the read-sized base"
1259        );
