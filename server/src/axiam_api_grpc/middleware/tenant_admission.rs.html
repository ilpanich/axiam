1//! Per-request tenant lifecycle enforcement for the gRPC listener (SA-13, N16,
2//! D-3).
3//!
4//! A suspended tenant's principals are refused at every authentication door,
5//! and this listener is one: a token a suspended tenant's user or service
6//! account obtained before the suspension stays cryptographically valid until
7//! it expires, and the verifying interceptor (`middleware::auth`) checks
8//! nothing else. Unlike the session check (`middleware::strict_revocation`,
9//! opt-in for latency), this one is **always on** — D-3 makes suspension
10//! refuse API requests, with no transport excepted.
11//!
12//! # Why a `tower::Layer` and not the interceptor
13//!
14//! For the reason `strict_revocation` gives: `Interceptor::call` is
15//! synchronous, and the tenant's status is a datastore read.
16//!
17//! # Why deciding on unverified claims is sound
18//!
19//! Also `strict_revocation`'s argument: this layer can only **deny**. A request
20//! it lets through still meets the verifying interceptor, so a forged token can
21//! at most be passed on here and refused there. It denies when the claims name
22//! a tenant whose stored status does not admit principals
23//! (`axiam_core::models::tenant::tenant_accepts_principals`).
24//!
25//! An undecodable token, and a tenant that is not stored, fall through: the
26//! interceptor refuses the first, and the second is a tombstoned or unknown
27//! tenant, whose sessions and clients are revoked elsewhere.
28//!
29//! # A datastore error refuses
30//!
31//! Unlike `strict_revocation`, this layer **fails closed**: when the tenant
32//! cannot be read the call is answered `UNAVAILABLE` with a generic message
33//! (the error itself is logged). No service behind the listener re-checks the
34//! tenant's status, so passing the call on would admit a possibly suspended
35//! tenant's token whenever the read failed. This matches the REST extractor
36//! and the AMQP consumer, which also refuse on "could not tell".
37//!
38//! # Cost
39//!
40//! One tenant read per authenticated call. There is no tenant-status cache to
41//! reuse (`tenant_org_cache` caches the organization mapping, which never
42//! changes; status does), and none is added: a suspension takes effect at the
43//! next call, on every replica.
44

58/// The status a suspended tenant's call is answered with.
59///
60/// `PERMISSION_DENIED`, not `UNAUTHENTICATED`: the credential is genuine, and a
61/// client told to re-authenticate would only be refused again. The message is
62/// the stable code first, so a client can match on it.
63fn suspended_response() -> Response<tonic::body::Body> {
64    tonic::Status::permission_denied(format!("tenant_suspended: {}", AxiamError::TenantSuspended))
65        .into_http()
66}
67
68/// The status a call is answered with when its tenant cannot be read.
69///
70/// `UNAVAILABLE` — transient server state, which a client retries — with a
71/// generic message: the datastore's own words belong in the log.
72fn unavailable_response() -> Response<tonic::body::Body> {
73    tonic::Status::unavailable("the tenant could not be checked; retry this call").into_http()
74}

106/// Inner service produced by [`GrpcTenantAdmissionLayer`].

114/// The tenant a request's bearer token names, or `None` when there is nothing
115/// to check (no header, no `Bearer ` prefix, undecodable token or claim).
116pub(crate) fn tenant_of_request<T>(req: &Request<T>, auth_config: &AuthConfig) -> Option<Uuid> {
117    let token = req
118        .headers()
119        .get("authorization")
120        .and_then(|v| v.to_str().ok())
121        .and_then(|s| s.strip_prefix("Bearer "))?;
122    let claims = decode_access_token(token, auth_config).ok()?;
123    Uuid::parse_str(&claims.tenant_id).ok()
124}

126impl<S> Service<Request<tonic::body::Body>> for GrpcTenantAdmissionService<S>
127where
128    S: Service<Request<tonic::body::Body>, Response = Response<tonic::body::Body>>
129        + Clone
130        + Send
131        + 'static,
132    S::Future: Send + 'static,
133    S::Error: Send + 'static,
134{
135    type Response = Response<tonic::body::Body>;
136    type Error = S::Error;
137    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;
138
139    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
140        self.inner.poll_ready(cx)
141    }
142
143    fn call(&mut self, req: Request<tonic::body::Body>) -> Self::Future {
144        let clone = self.inner.clone();
145        let mut inner = std::mem::replace(&mut self.inner, clone);
146
147        let tenant = tenant_of_request(&req, &self.auth_config);
148        let tenants = Arc::clone(&self.tenants);
149
150        Box::pin(async move {
151            // Admitted or not stored: passed on. Suspended: refused. Not
152            // readable: refused too (see the module docs).
153            if let Some(tenant_id) = tenant {
154                match tenants.admit(tenant_id).await {
155                    Ok(()) => {}
156                    Err(AxiamError::TenantSuspended) => {
157                        tracing::debug!(
158                            %tenant_id,
159                            "grpc: denying a call from a suspended tenant (SA-13)"
160                        );
161                        return Ok(suspended_response());
162                    }
163                    Err(error) => {
164                        tracing::error!(
165                            %tenant_id,
166                            %error,
167                            "grpc: the tenant of a call could not be read; refusing it (SA-13)"
168                        );
169                        return Ok(unavailable_response());
170                    }
171                }
172            }
173            inner.call(req).await
174        })
175    }
176}

203    impl Service<Request<tonic::body::Body>> for Downstream {
204        type Response = Response<tonic::body::Body>;
205        type Error = std::convert::Infallible;
206        type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;
207
208        fn poll_ready(&mut self, _cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
209            Poll::Ready(Ok(()))
210        }
211
212        fn call(&mut self, _req: Request<tonic::body::Body>) -> Self::Future {
213            self.0.fetch_add(1, Ordering::SeqCst);
214            Box::pin(async {
215                Ok(Response::builder()
216                    .status(200)
217                    .body(tonic::body::Body::empty())
218                    .expect("a 200 with an empty body always builds"))
219            })
220        }
221    }

223    fn bearer_request(
224        config: &AuthConfig,
225        tenant_id: Uuid,
226        aud: &str,
227    ) -> Request<tonic::body::Body> {
228        let token = issue_access_token(
229            Uuid::new_v4(),
230            tenant_id,
231            Uuid::new_v4(),
232            &[],
233            config,
234            Uuid::new_v4().to_string(),
235            aud,
236        )
237        .expect("issue token");
238        Request::builder()
239            .header("authorization", format!("Bearer {token}"))
240            .body(tonic::body::Body::empty())
241            .expect("request builds")
242    }

244    /// The `grpc-status` a response carries in its headers (tonic puts a
245    /// refusal's status there, with no body).
246    fn grpc_status(response: &Response<tonic::body::Body>) -> Option<String> {
247        response
248            .headers()
249            .get("grpc-status")
250            .and_then(|v| v.to_str().ok())
251            .map(str::to_owned)
252    }

254    async fn through_the_layer(
255        tenants: Arc<dyn TenantAdmission>,
256        req: Request<tonic::body::Body>,
257        config: AuthConfig,
258    ) -> (Option<String>, Option<String>, usize) {
259        let reached = Arc::new(AtomicUsize::new(0));
260        let layer = GrpcTenantAdmissionLayer::new(tenants, config);
261        let mut service = layer.layer(Downstream(Arc::clone(&reached)));
262        let response = service.call(req).await.expect("infallible downstream");
263        let message = response
264            .headers()
265            .get("grpc-message")
266            .and_then(|v| v.to_str().ok())
267            .map(str::to_owned);
268        (
269            grpc_status(&response),
270            message,
271            reached.load(Ordering::SeqCst),
272        )
273    }

303    /// SA-13 (N16): a call whose token — user or machine — names a suspended
304    /// tenant is refused `PERMISSION_DENIED` `tenant_suspended`, and the
305    /// service behind the layer never runs; un-suspending lets it through.
306    #[tokio::test]
307    async fn a_suspended_tenants_call_is_refused_and_never_reaches_the_service() {
308        let config = test_config();
309        let (repo, tenant_id) = tenant_store().await;
310        let tenants: Arc<dyn TenantAdmission> = Arc::new(repo.clone());
311
312        for aud in [AUD_USER, AUD_M2M] {
313            let (status, _, reached) = through_the_layer(
314                Arc::clone(&tenants),
315                bearer_request(&config, tenant_id, aud),
316                config.clone(),
317            )
318            .await;
319            assert_eq!(status, None, "{aud}: an active tenant's call passes");
320            assert_eq!(reached, 1, "{aud}");
321        }
322
323        repo.update(
324            tenant_id,
325            UpdateTenant {
326                status: Some(TenantStatus::Suspended),
327                ..Default::default()
328            },
329        )
330        .await
331        .unwrap();
332        for aud in [AUD_USER, AUD_M2M] {
333            let (status, message, reached) = through_the_layer(
334                Arc::clone(&tenants),
335                bearer_request(&config, tenant_id, aud),
336                config.clone(),
337            )
338            .await;
339            assert_eq!(
340                status.as_deref(),
341                Some((tonic::Code::PermissionDenied as i32).to_string().as_str()),
342                "{aud}: a suspended tenant's call is PERMISSION_DENIED"
343            );
344            assert!(
345                message.is_some_and(|m| m.starts_with("tenant_suspended")),
346                "{aud}: the message leads with the stable code"
347            );
348            assert_eq!(reached, 0, "{aud}: the service must not run");
349        }
350
351        repo.update(
352            tenant_id,
353            UpdateTenant {
354                status: Some(TenantStatus::Active),
355                ..Default::default()
356            },
357        )
358        .await
359        .unwrap();
360        let (status, _, reached) = through_the_layer(
361            tenants,
362            bearer_request(&config, tenant_id, AUD_USER),
363            config,
364        )
365        .await;
366        assert_eq!(status, None, "un-suspending restores the tenant's calls");
367        assert_eq!(reached, 1);
368    }

382    /// SA-13 review: no service re-checks the tenant's status, so a call whose
383    /// tenant cannot be read is refused `UNAVAILABLE` (generic message, never
384    /// the datastore's words) rather than admitted — "could not tell" must not
385    /// read as an admission of a possibly suspended tenant.
386    #[tokio::test]
387    async fn a_call_whose_tenant_cannot_be_read_is_refused_unavailable() {
388        let config = test_config();
389        let (status, message, reached) = through_the_layer(
390            Arc::new(Unreadable),
391            bearer_request(&config, Uuid::new_v4(), AUD_USER),
392            config,
393        )
394        .await;
395        assert_eq!(
396            status.as_deref(),
397            Some((tonic::Code::Unavailable as i32).to_string().as_str()),
398            "an unreadable tenant is UNAVAILABLE"
399        );
400        assert!(
401            message.is_some_and(|m| !m.contains("connection reset")),
402            "the datastore's words stay in the log"
403        );
404        assert_eq!(reached, 0, "the service must not run");
405    }
