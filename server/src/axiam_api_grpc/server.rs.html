1//! Tonic gRPC server setup.
2//!
3//! CQ-B20: Server builder sets message-size limits, per-connection timeout,
4//! and concurrency limit to prevent resource exhaustion.
5//! TLS (REQ-15 AC-1): When `AXIAM__GRPC_TLS_CERT_PATH` and
6//! `AXIAM__GRPC_TLS_KEY_PATH` env vars are set, the server is configured with
7//! `ServerTlsConfig`; otherwise plaintext mode is used (suitable for in-mesh
8//! mutual-TLS handled at the sidecar/service-mesh layer).
9//!
10//! D2 (benchmark plan — native gRPC TLS termination): this is the same
11//! mechanism the p2-tls13 native-TLS bench overlay
12//! (`benchmarks/targets/axiam/docker-compose.native-tls.yml`) turns on,
13//! pointed at the SAME server cert/key files the REST listener uses
14//! (`crates/axiam-server/src/tls.rs` / `AXIAM__SERVER__TLS__CERT_PATH`+
15//! `KEY_PATH`) so the two protocols present identical PKI material. No new
16//! `TlsConfig`-style struct was introduced here — the existing flat
17//! `AXIAM__GRPC_TLS_CERT_PATH`/`KEY_PATH` env-var pair (already shipped in
18//! phase 11) is reused as-is per D2's "least invasive" guidance, rather than
19//! adding a parallel `AXIAM__GRPC__TLS__*` nested config surface.
20//!
21//! Caveat vs. the REST listener: `crates/axiam-server/src/tls.rs` builds a
22//! custom rustls `ServerConfig` restricted to TLS 1.3 only
23//! (`with_protocol_versions(&[&rustls::version::TLS13])`). Tonic 0.14's
24//! `ServerTlsConfig` (see `tonic::transport::server::tls`) does not expose a
25//! protocol-version knob — it always builds `rustls::ServerConfig::builder()`
26//! with the crate's default versions (TLS 1.2 negotiable in addition to 1.3).
27//! There is no way to force TLS-1.3-only through tonic's public API without
28//! hand-rolling the accept loop, which is out of scope here; the gRPC
29//! listener is TLS 1.3-*capable* (matching REST posture) but not TLS
30//! 1.3-*exclusive*.

65/// Start the gRPC server with all registered services.
66///
67/// Applies two cooperating rate-limit layers (SECHRD-03, D-01a/b/c — gap
68/// closure for 24-07): the [`GrpcSharedRateLimitLayer`] cross-replica
69/// pre-check runs FIRST (outermost), failing OPEN to the per-replica in-memory
70/// governor stack (per D-10). Both layers derive their client-IP key from the
71/// SAME `trusted_hops` value so gRPC keying stays in lockstep across the
72/// shared counter and the in-memory fallback, and both are sized from the
73/// same per-second [`crate::middleware::rate_limit::GrpcRateLimits`] value.
74///
75/// Since I1/I2 the ceilings are **per method family**, not server-wide:
76/// `AXIAM__GRPC__GRPC_AUTHZ_PER_SEC` sizes `AuthorizationService`,
77/// `AXIAM__GRPC__GRPC_IDENTITY_PER_SEC` sizes `UserInfoService`/`TokenService`,
78/// `AXIAM__GRPC__GRPC_ADMIN_PER_SEC` sizes `UserService`, and gRPC
79/// reflection/health are never limited.
80///
81/// Since the H2 performance fix the shared pre-check performs NO synchronous
82/// datastore write: it is backed by the process-wide write-behind
83/// `axiam_db::SharedRateLimitCounter`, built ONCE here (this layer is
84/// server-wide, so every gRPC call used to pay one `UPSERT`). Knobs:
85/// `AXIAM__RATE_LIMIT__SHARED` (on|off, default on) and
86/// `AXIAM__RATE_LIMIT__SHARED_SYNC_MS` (default 1000, clamped 50..=60000) —
87/// the same two the REST listener reads.
88///
89/// Transport limits (CQ-B20):
90/// - Max message size: 4 MiB decode / 4 MiB encode
91/// - Per-connection timeout: 30 s
92/// - Concurrency limit: 256 streams per connection
93///
94/// TLS (REQ-15 AC-1): env-gated via `AXIAM__GRPC_TLS_CERT_PATH` /
95/// `AXIAM__GRPC_TLS_KEY_PATH`. When absent, TLS is disabled and a warning
96/// is logged (acceptable for in-mesh/loopback deployments).
97///
98/// # `strict_revocation_checker` (A4/J10)
99///
100/// `Some(..)` only when `AXIAM__GRPC__STRICT_REVOCATION=true`, and it MUST be
101/// the SAME session-repository instance the REST path holds. A second instance
102/// would carry a second validation cache, and a cache the REST invalidation
103/// hooks never reach is exactly the stale allow strict mode exists to prevent.
104/// `None` keeps the shipped posture: gRPC trusts JWT lifetime, so revocation
105/// takes effect at token expiry (up to 15 min) rather than immediately. See
106/// [`crate::middleware::strict_revocation`] for the full posture table.
107///
108/// # `reactor_engine`, `reactor_repo`, `reactor_audit_repo`, `reactor_routing_invalidator` (X1 / R2.3)
109///
110/// `reactor_engine` is a SECOND [`AuthorizationEngine`] instance, built from
111/// the same repositories as `engine`. `AuthorizationEngine` does not
112/// implement `Clone`, so `ReactorAdminServiceImpl` cannot share `engine`'s
113/// instance — it is given its own, mirroring `AuthorizationServiceImpl`'s.
114/// Two engines over the same repositories is correct, not wasteful: the
115/// decision cache and invalidation broadcaster (when configured) are handed
116/// to both by the caller, so the two stay coherent with each other exactly as
117/// `axiam-server`'s REST `AuthzChecker` and this gRPC listener already do.
118/// `reactor_routing_invalidator` MUST be the same closure over the same
119/// `ReactorRoutingTable` the REST admin handlers invalidate through, or a
120/// registration written over gRPC would not take effect on this replica
121/// until the routing table's TTL expired.
122#[allow(clippy::too_many_arguments)]
123pub async fn start_grpc_server<R, P, Res, S, G, U, C, Rr, A>(
124    addr: SocketAddr,
125    engine: AuthorizationEngine<R, P, Res, S, G>,
126    user_repo: U,
127    auth_config: AuthConfig,
128    grpc_config: &GrpcConfig,
129    db: impl Into<DbHandle<C>>,
130    batch_max_concurrency: usize,
131    strict_revocation_checker: Option<Arc<dyn SessionRevocationCheck>>,
132    reactor_engine: AuthorizationEngine<R, P, Res, S, G>,
133    reactor_repo: Rr,
134    reactor_audit_repo: A,
135    reactor_routing_invalidator: Arc<dyn Fn(Uuid) + Send + Sync>,
136    // SEC-101 — whether the composed reactor transport can deliver at all.
137    // MUST be read from the same gate `axiam-api-rest` holds, so the two
138    // admin surfaces refuse (or accept) the same registrations.
139    reactor_dispatch_available: bool,
140    // B1 — the process-wide Argon2id gate. MUST be a clone of the same
141    // `Arc<Semaphore>` `AppState::crypto_semaphore` holds: the permit count
142    // bounds peak concurrent ~19 MiB hash arenas for the WHOLE process, so a
143    // listener with its own instance would silently double the bound.
144    // `UserService/ValidateCredentials` is this listener's only consumer.
145    crypto_semaphore: Arc<Semaphore>,
146) -> Result<(), tonic::transport::Error>
147where
148    R: RoleRepository + 'static,
149    P: PermissionRepository + 'static,
150    Res: ResourceRepository + 'static,
151    S: ScopeRepository + 'static,
152    G: GroupRepository + 'static,
153    U: UserRepository + Clone + 'static,
154    C: Connection + 'static,
155    Rr: ReactorRepository + 'static,
156    A: AuditLogRepository + 'static,
157{
158    // I2: per-family ceilings (unset knobs derived from the authz ceiling).
159    let rate_limits = grpc_config.rate_limits();
160    tracing::info!(
161        bind = %addr,
162        grpc_authz_per_sec = rate_limits.authz_per_sec,
163        grpc_identity_per_sec = rate_limits.identity_per_sec,
164        grpc_admin_per_sec = rate_limits.admin_per_sec,
165        "Starting gRPC server",
166    );
167
168    // SECHRD-03 gap closure (24-07 follow-up): the shared-store pre-check
169    // MUST use the same trusted_hops value as the in-memory governor's key
170    // extractor (both ultimately key off GrpcTrustedHopsKeyExtractor logic)
171    // so a rotating XFF cannot mint a fresh bucket in one layer while being
172    // correctly collapsed in the other.
173    let trusted_hops = trusted_hops_from_env();
174    // Built exactly ONCE per process: this constructs the write-behind
175    // `SharedRateLimitCounter` (and spawns its single background flusher on
176    // this runtime). Cloning the LAYER is fine — the counter inside is an
177    // `Arc` handle, so every clone shares the same local counts. Building a
178    // second layer would create a second, independent counter.
179    //
180    // I1 (units) + I2 (scope): both layers are now built from the SAME
181    // per-second `GrpcRateLimits` value. `new_method_scoped` converts each
182    // per-second ceiling into the shared layer's 60-second window budget
183    // itself (`per_sec_to_window_limit`) — previously the per-second number
184    // was passed to a per-minute window verbatim, which made the effective
185    // gRPC ceiling 1/60th of the configured one.
186    let shared_rate_limit_layer =
187        GrpcSharedRateLimitLayer::new_method_scoped(db, rate_limits, trusted_hops);
188    let governor_layer = build_grpc_method_scoped_governor_layer(rate_limits);
189
190    // A4/J10: built before the interceptors take ownership of `auth_config`.
191    // The layer needs it to decode (not verify — see the module docs) the
192    // bearer token far enough to learn which session a request names.
193    let strict_revocation_layer = strict_revocation_checker.map(|checker| {
194        tracing::warn!(
195            "gRPC STRICT REVOCATION enabled — every gRPC request now re-checks \
196             session validity, matching REST. Expect the gRPC check profile to \
197             approach REST's revocation-checked one; with the session-validation \
198             cache disabled this is one datastore read per request."
199        );
200        GrpcStrictRevocationLayer::new(checker, auth_config.clone())
201    });
202
203    let authz_svc = AuthorizationServiceServer::with_interceptor(
204        AuthorizationServiceImpl::new(engine, batch_max_concurrency),
205        AuthInterceptor::new(auth_config.clone()),
206    );
207    // SECFIX-01: UserService and TokenService previously had zero auth —
208    // any unauthenticated mesh peer could call GetUser/ValidateCredentials/
209    // IntrospectToken. Wrap them with the same AuthInterceptor chokepoint
210    // as AuthorizationService so every gRPC call requires a verified bearer JWT.
211    let user_svc = UserServiceServer::with_interceptor(
212        UserServiceImpl::new(user_repo.clone(), auth_config.clone(), crypto_semaphore),
213        AuthInterceptor::new(auth_config.clone()),
214    );
215    // UserInfoService: OIDC-style self lookup — identity derived entirely from
216    // the interceptor-verified bearer token (no request body), mirroring the
217    // REST `/oauth2/userinfo` endpoint. Guarded by the same AuthInterceptor.
218    let user_info_svc = UserInfoServiceServer::with_interceptor(
219        UserInfoServiceImpl::new(user_repo),
220        AuthInterceptor::new(auth_config.clone()),
221    );
222    let token_svc = TokenServiceServer::with_interceptor(
223        TokenServiceImpl::new(auth_config.clone()),
224        AuthInterceptor::new(auth_config.clone()),
225    );
226    // X1 / R2.3 — reactor admin CRUD over gRPC, mirroring the REST
227    // `/api/v1/reactors*` surface. Same `AuthInterceptor` chokepoint as every
228    // other authenticated service here.
229    let reactor_svc = ReactorAdminServiceServer::with_interceptor(
230        ReactorAdminServiceImpl::new(
231            reactor_repo,
232            reactor_engine,
233            reactor_audit_repo,
234            reactor_routing_invalidator,
235            reactor_dispatch_available,
236        ),
237        AuthInterceptor::new(auth_config),
238    );
239
240    // CQ-B20: Apply transport limits to the gRPC server builder.
241    // Note: tonic 0.14 does not expose max_decoding_message_size / max_encoding_message_size
242    // at the Server level (added in tonic 0.12+ via the Router API). The HTTP/2 frame-size
243    // limit (max_frame_size) is the closest available per-connection cap in 0.14. Upgrade
244    // to tonic ≥0.12 to use per-service max_decoding_message_size / max_encoding_message_size.
245    // Tracked: max_decoding_message_size (CQ-B20, pending tonic upgrade in Phase 19).
246    // SECHRD-03 (D-01a/b/c) + run-5 J1: layer ORDER is load-bearing here.
247    // Tower's `Server::builder()` runs the FIRST-added layer with the request
248    // FIRST (the opposite of actix's last-`.wrap()`-is-outermost rule), so
249    // the layer named first below is the OUTERMOST one.
250    //
251    // Until run 5 the shared-store pre-check was outermost. That inverted the
252    // two layers' natural roles and starved every gRPC family: the pre-check
253    // charges its whole 60-second window budget the moment it admits a
254    // request, but the per-second `governor` behind it then rejected almost
255    // all of those requests. A flood therefore burned all 6 000 authz
256    // admissions of a window on traffic that was never served, in well under
257    // a second, and the window stayed exhausted for its remaining ~59 —
258    // measured at 181 admissions/min against a configured 6 000/min, i.e.
259    // 1/33 of the ceiling (`claude_dev/improvement-after-run5-benchmark.md`
260    // A1/J1).
261    //
262    // The governor is therefore now OUTERMOST and owns the admission
263    // decision, and the shared cross-replica counter sits behind it, counting
264    // only traffic that was actually admitted. Both roles are preserved:
265    //
266    // - single replica — the governor's per-second quota is the binding
267    //   constraint and the shared counter (sized `per_sec × 60`) never trips,
268    //   so the effective ceiling equals the configured one;
269    // - N replicas — each admits its own per-second quota, their sum reaches
270    //   the shared window budget, and the shared counter cuts in. That is
271    //   exactly the `replicas × limit` multiplication SECHRD-03 exists to
272    //   stop, and it is still stopped.
273    //
274    // Fail-open is unchanged: on a missing key, a disabled shared layer, or a
275    // SurrealDB blip the inner layer forwards the request and the governor's
276    // decision (already taken, outside it) stands.
277    //
278    // The REST listener cannot use this ordering — its shared middleware is
279    // also what parses `client_id` out of the form body for the client-aware
280    // key modes, so it must stay outermost there. It solves the same problem
281    // with `SharedRateLimitCounter::refund` instead; see that method's docs.
282    // A4/J10: strict revocation, when enabled, sits INSIDE the rate limiters
283    // (so a flood of revoked-session calls is throttled before it can touch
284    // the session cache) and OUTSIDE the services (so it applies uniformly to
285    // every RPC, including any added later). `option_layer` makes the disabled
286    // case a genuine no-op rather than a branch in the request path.
287    let mut builder = Server::builder()
288        .max_frame_size(4 * 1024 * 1024) // CQ-B20: 4 MiB frame cap (tonic-0.14 equivalent of max_decoding_message_size)
289        .timeout(Duration::from_secs(30))
290        .concurrency_limit_per_connection(256)
291        // OUTERMOST (added first = runs first): refunds h2's small-DATA-frame
292        // budget for any call the rate limiters below answer without reading
293        // its body. Without it the listener GOAWAYs the connection every ~102
294        // rejections — see `middleware::drain_rejected`.
295        .layer(DrainRejectedBodyLayer)
296        .layer(governor_layer)
297        .layer(shared_rate_limit_layer)
298        .layer(tower::util::option_layer(strict_revocation_layer));
299
300    // REQ-15 AC-1 / CQ-B20: Env-gated TLS.
301    // Set AXIAM__GRPC_TLS_CERT_PATH and AXIAM__GRPC_TLS_KEY_PATH to enable.
302    let cert_path = std::env::var("AXIAM__GRPC_TLS_CERT_PATH").ok();
303    let key_path = std::env::var("AXIAM__GRPC_TLS_KEY_PATH").ok();
304    match (cert_path, key_path) {
305        (Some(cert_path), Some(key_path)) => {
306            let cert_pem = std::fs::read(&cert_path).unwrap_or_else(|e| {
307                panic!("AXIAM__GRPC_TLS_CERT_PATH set but file not readable at '{cert_path}': {e}")
308            });
309            let key_pem = std::fs::read(&key_path).unwrap_or_else(|e| {
310                panic!("AXIAM__GRPC_TLS_KEY_PATH set but file not readable at '{key_path}': {e}")
311            });
312
313            let identity = Identity::from_pem(cert_pem, key_pem);
314            let tls_config = ServerTlsConfig::new().identity(identity);
315            tracing::info!("gRPC server TLS enabled (AXIAM__GRPC_TLS_CERT_PATH)");
316            builder
317                .tls_config(tls_config)?
318                .add_service(authz_svc)
319                .add_service(user_svc)
320                .add_service(user_info_svc)
321                .add_service(token_svc)
322                .add_service(reactor_svc)
323                .serve(addr)
324                .await
325        }
326        _ => {
327            tracing::warn!(
328                "gRPC TLS is DISABLED — set AXIAM__GRPC_TLS_CERT_PATH + \
329                 AXIAM__GRPC_TLS_KEY_PATH to enable (acceptable for in-mesh deployments)"
330            );
331            builder
332                .add_service(authz_svc)
333                .add_service(user_svc)
334                .add_service(user_info_svc)
335                .add_service(token_svc)
336                .add_service(reactor_svc)
337                .serve(addr)
338                .await
339        }
340    }
