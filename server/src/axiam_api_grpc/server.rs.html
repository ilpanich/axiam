1//! Tonic gRPC server setup.
2//!
3//! CQ-B20: Server builder sets message-size limits, per-connection timeout,
4//! and concurrency limit to prevent resource exhaustion.
5//! TLS (REQ-15 AC-1): When `AXIAM__GRPC_TLS_CERT_PATH` and
6//! `AXIAM__GRPC_TLS_KEY_PATH` env vars are set, the server is configured with
7//! `ServerTlsConfig`; otherwise plaintext mode is used (suitable for in-mesh
8//! mutual-TLS handled at the sidecar/service-mesh layer).
9//!
10//! D2 (benchmark plan — native gRPC TLS termination): this is the same
11//! mechanism the p2-tls13 native-TLS bench overlay
12//! (`benchmarks/targets/axiam/docker-compose.native-tls.yml`) turns on,
13//! pointed at the SAME server cert/key files the REST listener uses
14//! (`crates/axiam-server/src/tls.rs` / `AXIAM__SERVER__TLS__CERT_PATH`+
15//! `KEY_PATH`) so the two protocols present identical PKI material. No new
16//! `TlsConfig`-style struct was introduced here — the existing flat
17//! `AXIAM__GRPC_TLS_CERT_PATH`/`KEY_PATH` env-var pair (already shipped in
18//! phase 11) is reused as-is per D2's "least invasive" guidance, rather than
19//! adding a parallel `AXIAM__GRPC__TLS__*` nested config surface.
20//!
21//! Caveat vs. the REST listener: `crates/axiam-server/src/tls.rs` builds a
22//! custom rustls `ServerConfig` restricted to TLS 1.3 only
23//! (`with_protocol_versions(&[&rustls::version::TLS13])`). Tonic 0.14's
24//! `ServerTlsConfig` (see `tonic::transport::server::tls`) does not expose a
25//! protocol-version knob — it always builds `rustls::ServerConfig::builder()`
26//! with the crate's default versions (TLS 1.2 negotiable in addition to 1.3).
27//! There is no way to force TLS-1.3-only through tonic's public API without
28//! hand-rolling the accept loop, which is out of scope here; the gRPC
29//! listener is TLS 1.3-*capable* (matching REST posture) but not TLS
30//! 1.3-*exclusive*.

58/// Start the gRPC server with all registered services.
59///
60/// Applies two cooperating rate-limit layers (SECHRD-03, D-01a/b/c — gap
61/// closure for 24-07): the [`GrpcSharedRateLimitLayer`] cross-replica
62/// pre-check runs FIRST (outermost), failing OPEN to the per-replica in-memory
63/// `GovernorLayer` (per D-10) built via `grpc_authz_per_sec` from
64/// `GrpcConfig`. Both layers derive their client-IP key from the SAME
65/// `trusted_hops` value so gRPC keying stays in lockstep across the shared
66/// counter and the in-memory fallback.
67///
68/// Since the H2 performance fix the shared pre-check performs NO synchronous
69/// datastore write: it is backed by the process-wide write-behind
70/// `axiam_db::SharedRateLimitCounter`, built ONCE here (this layer is
71/// server-wide, so every gRPC call used to pay one `UPSERT`). Knobs:
72/// `AXIAM__RATE_LIMIT__SHARED` (on|off, default on) and
73/// `AXIAM__RATE_LIMIT__SHARED_SYNC_MS` (default 1000, clamped 50..=60000) —
74/// the same two the REST listener reads.
75///
76/// Transport limits (CQ-B20):
77/// - Max message size: 4 MiB decode / 4 MiB encode
78/// - Per-connection timeout: 30 s
79/// - Concurrency limit: 256 streams per connection
80///
81/// TLS (REQ-15 AC-1): env-gated via `AXIAM__GRPC_TLS_CERT_PATH` /
82/// `AXIAM__GRPC_TLS_KEY_PATH`. When absent, TLS is disabled and a warning
83/// is logged (acceptable for in-mesh/loopback deployments).
84pub async fn start_grpc_server<R, P, Res, S, G, U, C>(
85    addr: SocketAddr,
86    engine: AuthorizationEngine<R, P, Res, S, G>,
87    user_repo: U,
88    auth_config: AuthConfig,
89    grpc_config: &GrpcConfig,
90    db: impl Into<DbHandle<C>>,
91    batch_max_concurrency: usize,
92) -> Result<(), tonic::transport::Error>
93where
94    R: RoleRepository + 'static,
95    P: PermissionRepository + 'static,
96    Res: ResourceRepository + 'static,
97    S: ScopeRepository + 'static,
98    G: GroupRepository + 'static,
99    U: UserRepository + Clone + 'static,
100    C: Connection + 'static,
101{
102    tracing::info!(
103        bind = %addr,
104        grpc_authz_per_sec = grpc_config.grpc_authz_per_sec,
105        "Starting gRPC server",
106    );
107
108    // SECHRD-03 gap closure (24-07 follow-up): the shared-store pre-check
109    // MUST use the same trusted_hops value as the in-memory governor's key
110    // extractor (both ultimately key off GrpcTrustedHopsKeyExtractor logic)
111    // so a rotating XFF cannot mint a fresh bucket in one layer while being
112    // correctly collapsed in the other.
113    let trusted_hops = trusted_hops_from_env();
114    // Built exactly ONCE per process: this constructs the write-behind
115    // `SharedRateLimitCounter` (and spawns its single background flusher on
116    // this runtime). Cloning the LAYER is fine — the counter inside is an
117    // `Arc` handle, so every clone shares the same local counts. Building a
118    // second layer would create a second, independent counter.
119    let shared_rate_limit_layer = GrpcSharedRateLimitLayer::new(
120        db,
121        "grpc_authz",
122        grpc_config.grpc_authz_per_sec,
123        trusted_hops,
124    );
125    let governor_layer = build_grpc_governor_layer(grpc_config.grpc_authz_per_sec);
126
127    let authz_svc = AuthorizationServiceServer::with_interceptor(
128        AuthorizationServiceImpl::new(engine, batch_max_concurrency),
129        AuthInterceptor::new(auth_config.clone()),
130    );
131    // SECFIX-01: UserService and TokenService previously had zero auth —
132    // any unauthenticated mesh peer could call GetUser/ValidateCredentials/
133    // IntrospectToken. Wrap them with the same AuthInterceptor chokepoint
134    // as AuthorizationService so every gRPC call requires a verified bearer JWT.
135    let user_svc = UserServiceServer::with_interceptor(
136        UserServiceImpl::new(user_repo.clone(), auth_config.clone()),
137        AuthInterceptor::new(auth_config.clone()),
138    );
139    // UserInfoService: OIDC-style self lookup — identity derived entirely from
140    // the interceptor-verified bearer token (no request body), mirroring the
141    // REST `/oauth2/userinfo` endpoint. Guarded by the same AuthInterceptor.
142    let user_info_svc = UserInfoServiceServer::with_interceptor(
143        UserInfoServiceImpl::new(user_repo),
144        AuthInterceptor::new(auth_config.clone()),
145    );
146    let token_svc = TokenServiceServer::with_interceptor(
147        TokenServiceImpl::new(auth_config.clone()),
148        AuthInterceptor::new(auth_config),
149    );
150
151    // CQ-B20: Apply transport limits to the gRPC server builder.
152    // Note: tonic 0.14 does not expose max_decoding_message_size / max_encoding_message_size
153    // at the Server level (added in tonic 0.12+ via the Router API). The HTTP/2 frame-size
154    // limit (max_frame_size) is the closest available per-connection cap in 0.14. Upgrade
155    // to tonic ≥0.12 to use per-service max_decoding_message_size / max_encoding_message_size.
156    // Tracked: max_decoding_message_size (CQ-B20, pending tonic upgrade in Phase 19).
157    // SECHRD-03 (D-01a/b/c): the shared-store pre-check is `.layer()`'d
158    // FIRST so it is OUTERMOST (tower's ServiceBuilder/Server::builder()
159    // convention — first `.layer()` call = outermost = runs first, the
160    // opposite of actix's last-`.wrap()`-is-outermost rule). It fails OPEN
161    // to `governor_layer` on any SurrealDB error or missing key, so a DB
162    // blip degrades to the in-memory limiter rather than hard-blocking.
163    let mut builder = Server::builder()
164        .max_frame_size(4 * 1024 * 1024) // CQ-B20: 4 MiB frame cap (tonic-0.14 equivalent of max_decoding_message_size)
165        .timeout(Duration::from_secs(30))
166        .concurrency_limit_per_connection(256)
167        .layer(shared_rate_limit_layer)
168        .layer(governor_layer);
169
170    // REQ-15 AC-1 / CQ-B20: Env-gated TLS.
171    // Set AXIAM__GRPC_TLS_CERT_PATH and AXIAM__GRPC_TLS_KEY_PATH to enable.
172    let cert_path = std::env::var("AXIAM__GRPC_TLS_CERT_PATH").ok();
173    let key_path = std::env::var("AXIAM__GRPC_TLS_KEY_PATH").ok();
174    match (cert_path, key_path) {
175        (Some(cert_path), Some(key_path)) => {
176            let cert_pem = std::fs::read(&cert_path).unwrap_or_else(|e| {
177                panic!("AXIAM__GRPC_TLS_CERT_PATH set but file not readable at '{cert_path}': {e}")
178            });
179            let key_pem = std::fs::read(&key_path).unwrap_or_else(|e| {
180                panic!("AXIAM__GRPC_TLS_KEY_PATH set but file not readable at '{key_path}': {e}")
181            });
182
183            let identity = Identity::from_pem(cert_pem, key_pem);
184            let tls_config = ServerTlsConfig::new().identity(identity);
185            tracing::info!("gRPC server TLS enabled (AXIAM__GRPC_TLS_CERT_PATH)");
186            builder
187                .tls_config(tls_config)?
188                .add_service(authz_svc)
189                .add_service(user_svc)
190                .add_service(user_info_svc)
191                .add_service(token_svc)
192                .serve(addr)
193                .await
194        }
195        _ => {
196            tracing::warn!(
197                "gRPC TLS is DISABLED — set AXIAM__GRPC_TLS_CERT_PATH + \
198                 AXIAM__GRPC_TLS_KEY_PATH to enable (acceptable for in-mesh deployments)"
199            );
200            builder
201                .add_service(authz_svc)
202                .add_service(user_svc)
203                .add_service(user_info_svc)
204                .add_service(token_svc)
205                .serve(addr)
206                .await
207        }
208    }
