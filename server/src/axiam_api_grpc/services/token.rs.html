16/// Project a decoded `cnf` claim onto the wire message (X5.1).
17///
18/// `None` for an unbound token, so the field is simply absent — which is what
19/// every pre-X5.1 token produces and what keeps this response byte-compatible
20/// for a consumer that has never heard of sender-constraining.
21fn wire_cnf(claims: &axiam_auth::token::AccessTokenClaims) -> Option<CnfClaim> {
22    claims.cnf.as_ref().map(|c| CnfClaim {
23        x5t_s256: c.x5t_s256.clone().unwrap_or_default(),
24        jkt: c.jkt.clone().unwrap_or_default(),
25    })
26}
27
28/// RFC 6749 §7.1 / RFC 9449 §5 token type for a set of claims.
29///
30/// A DPoP-bound token is announced as `DPoP`; everything else — including a
31/// certificate-bound token, which RFC 8705 leaves a bearer token at the HTTP
32/// layer — is `Bearer`.
33fn wire_token_type(claims: &axiam_auth::token::AccessTokenClaims) -> String {
34    match claims.cnf.as_ref().and_then(|c| c.jkt.as_deref()) {
35        Some(_) => "DPoP".to_owned(),
36        None => "Bearer".to_owned(),
37    }
38}

51impl TokenService for TokenServiceImpl {
52    async fn validate_token(
53        &self,
54        request: Request<ValidateTokenRequest>,
55    ) -> Result<Response<ValidateTokenResponse>, Status> {
56        // SEC-068: the caller's tenant comes from the interceptor-verified JWT.
57        let caller_tenant = request
58            .extensions()
59            .get::<ValidatedClaims>()
60            .ok_or_else(|| Status::unauthenticated("missing validated claims"))?
61            .0
62            .tenant_id
63            .clone();
64        let req = request.into_inner();
65
66        match validate_access_token(&req.access_token, &self.config) {
67            // SEC-068: refuse to introspect a token belonging to a different
68            // tenant — report it as invalid (not merely denied) so a mesh peer
69            // in tenant A cannot read a tenant-B token's claims by observing the
70            // difference between "denied" and "invalid".
71            Ok(validated) if validated.0.tenant_id == caller_tenant => {
72                let claims = validated.0;
73                let cnf = wire_cnf(&claims);
74                let token_type = wire_token_type(&claims);
75                if cnf.is_some() {
76                    // Worth a line in the log rather than silence: the caller
77                    // is about to receive `valid: true` for a token that is
78                    // NOT usable as a bearer credential, and whether they act
79                    // on that is beyond this service's reach.
80                    tracing::debug!(
81                        token_jti = %claims.jti,
82                        "grpc: validated a sender-constrained token; the caller must prove \
83                         possession of the confirmed key or refuse the request"
84                    );
85                }
86                Ok(Response::new(ValidateTokenResponse {
87                    valid: true,
88                    subject_id: claims.sub,
89                    tenant_id: claims.tenant_id,
90                    org_id: claims.org_id,
91                    exp: claims.exp,
92                    cnf,
93                    token_type,
94                }))
95            }
96            Err(AuthError::Crypto(msg)) => {
97                Err(Status::internal(format!("token validation error: {msg}")))
98            }
99            // Invalid token OR a cross-tenant token (guard above failed) — both
100            // report inactive so the two are indistinguishable to the caller.
101            _ => Ok(Response::new(ValidateTokenResponse {
102                valid: false,
103                subject_id: String::new(),
104                tenant_id: String::new(),
105                org_id: String::new(),
106                exp: 0,
107                // An inactive answer discloses nothing, including whether the
108                // token that failed was bound.
109                cnf: None,
110                token_type: String::new(),
111            })),
112        }
113    }

115    async fn introspect_token(
116        &self,
117        request: Request<IntrospectTokenRequest>,
118    ) -> Result<Response<IntrospectTokenResponse>, Status> {
119        // SEC-068: the caller's tenant comes from the interceptor-verified JWT.
120        let caller_tenant = request
121            .extensions()
122            .get::<ValidatedClaims>()
123            .ok_or_else(|| Status::unauthenticated("missing validated claims"))?
124            .0
125            .tenant_id
126            .clone();
127        let req = request.into_inner();
128
129        match validate_access_token(&req.access_token, &self.config) {
130            // SEC-068: only introspect a token from the caller's own tenant; a
131            // cross-tenant token reports inactive (indistinguishable from an
132            // invalid one) so its sub/org/jti claims are not disclosed.
133            Ok(validated) if validated.0.tenant_id == caller_tenant => {
134                let claims = validated.0;
135                let cnf = wire_cnf(&claims);
136                let token_type = wire_token_type(&claims);
137                let permissions = claims
138                    .permissions
139                    .as_ref()
140                    .map(|ps| {
141                        ps.iter()
142                            .map(|p| RptPermission {
143                                resource_id: p.resource_id.to_string(),
144                                resource_scopes: p.resource_scopes.clone(),
145                                exp: p.exp,
146                            })
147                            .collect()
148                    })
149                    .unwrap_or_default();
150                let ext_exchange_iss = claims
151                    .ext_exchange
152                    .as_ref()
153                    .map(|e| e.iss.clone())
154                    .unwrap_or_default();
155                // The client id lives in `sub` for a client-credentials token,
156                // where `sub_kind` says so. Reading it off `sub_kind` rather
157                // than pattern-matching the string is what keeps this from
158                // guessing: an `oa_`-prefixed user id would otherwise be
159                // reported as a client.
160                let client_id = match claims.sub_kind {
161                    axiam_auth::token::SubjectKind::OAuth2Client
162                    | axiam_auth::token::SubjectKind::ServiceAccount => claims.sub.clone(),
163                    _ => String::new(),
164                };
165                Ok(Response::new(IntrospectTokenResponse {
166                    active: true,
167                    sub: claims.sub,
168                    tenant_id: claims.tenant_id,
169                    org_id: claims.org_id,
170                    iss: claims.iss,
171                    iat: claims.iat,
172                    exp: claims.exp,
173                    jti: claims.jti,
174                    scope: claims.scope.unwrap_or_default(),
175                    client_id,
176                    token_type,
177                    cnf,
178                    permissions,
179                    ext_exchange_iss,
180                }))
181            }
182            Err(AuthError::Crypto(msg)) => {
183                Err(Status::internal(format!("token validation error: {msg}")))
184            }
185            _ => Ok(Response::new(IntrospectTokenResponse {
186                active: false,
187                sub: String::new(),
188                tenant_id: String::new(),
189                org_id: String::new(),
190                iss: String::new(),
191                iat: 0,
192                exp: 0,
193                jti: String::new(),
194                // RFC 7662 §2.2: an inactive response carries `active: false`
195                // and nothing else. Every field below stays empty on purpose —
196                // leaking a `cnf` or a scope for a token the caller may not
197                // introspect would undo the SEC-068 guard above.
198                scope: String::new(),
199                client_id: String::new(),
200                token_type: String::new(),
201                cnf: None,
202                permissions: Vec::new(),
203                ext_exchange_iss: String::new(),
204            })),
205        }
206    }
