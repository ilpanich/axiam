21/// Project a decoded `cnf` claim onto the wire message (X5.1).
22///
23/// `None` for an unbound token, so the field is simply absent — which is what
24/// every pre-X5.1 token produces and what keeps this response byte-compatible
25/// for a consumer that has never heard of sender-constraining.
26fn wire_cnf(claims: &axiam_auth::token::AccessTokenClaims) -> Option<CnfClaim> {
27    claims.cnf.as_ref().map(|c| CnfClaim {
28        x5t_s256: c.x5t_s256.clone().unwrap_or_default(),
29        jkt: c.jkt.clone().unwrap_or_default(),
30    })
31}
32
33/// RFC 6749 §7.1 / RFC 9449 §5 token type for a set of claims.
34///
35/// A DPoP-bound token is announced as `DPoP`; everything else — including a
36/// certificate-bound token, which RFC 8705 leaves a bearer token at the HTTP
37/// layer — is `Bearer`.
38fn wire_token_type(claims: &axiam_auth::token::AccessTokenClaims) -> String {
39    match claims.cnf.as_ref().and_then(|c| c.jkt.as_deref()) {
40        Some(_) => "DPoP".to_owned(),
41        None => "Bearer".to_owned(),
42    }
43}

45pub struct TokenServiceImpl<U: UserRepository> {
46    config: AuthConfig,
47    /// Read by `IntrospectToken` only (#520, P23W1-12): introspection reports
48    /// a token whose account may no longer act as inactive. `ValidateToken`
49    /// stays a local signature-and-claims check and reads nothing.
50    user_repo: U,
51    /// Read by `IntrospectToken` only (SA-41): a token whose `sid` session is
52    /// no longer live reports `active: false`. `None` skips the check.
53    session_check: Option<Arc<dyn SessionRevocationCheck>>,
54}
55
56impl<U: UserRepository> TokenServiceImpl<U> {
57    pub fn new(config: AuthConfig, user_repo: U) -> Self {
58        Self {
59            config,
60            user_repo,
61            session_check: None,
62        }
63    }
64
65    /// Have `IntrospectToken` report a token inactive once the session its
66    /// `sid` claim names is revoked or expired (SA-41). The checker should
67    /// hold no validation cache: a negative answer must never be stale.
68    #[must_use]
69    pub fn with_session_check(mut self, check: Arc<dyn SessionRevocationCheck>) -> Self {
70        self.session_check = Some(check);
71        self
72    }
73
74    /// Whether the session a token names is still live.
75    ///
76    /// The session is named by `sid`, else by `jti`
77    /// (`AccessTokenClaims::session_reference`, the rule the REST extractor and
78    /// REST introspection apply): a first-party sign-in token carries its
79    /// session id in `jti` and no `sid`. Only a user token can name a session;
80    /// a client-credentials or service-account token, and a token of a grant
81    /// that names none (exchanged, RPT, device), is unaffected. A user token
82    /// whose reference is not a session id cannot be live.
83    async fn session_is_live(&self, claims: &axiam_auth::token::AccessTokenClaims) -> bool {
84        let Some(check) = &self.session_check else {
85            return true;
86        };
87        if claims.sub_kind != axiam_auth::token::SubjectKind::User {
88            return true;
89        }
90        let Some(reference) = claims.session_reference() else {
91            return true;
92        };
93        let (Ok(sid), Ok(tenant_id)) =
94            (Uuid::parse_str(reference), claims.tenant_id.parse::<Uuid>())
95        else {
96            return false;
97        };
98        check.is_session_active(tenant_id, sid).await
99    }
100
101    /// Whether the account a user token names may still act (#520,
102    /// P23W1-12) — `axiam_auth::service::account_may_act`, the rule REST
103    /// introspection, UserInfo, every OAuth2 grant and `/oauth2/authorize`
104    /// apply. `Ok(true)` for a token that names no account (a service account
105    /// or an OAuth2 client, whose `sub` is not a user id). A subject that no
106    /// longer exists may not act; any other read failure is INTERNAL rather
107    /// than a guess in either direction.
108    async fn subject_may_act(
109        &self,
110        claims: &axiam_auth::token::AccessTokenClaims,
111    ) -> Result<bool, Status> {
112        if claims.sub_kind != axiam_auth::token::SubjectKind::User {
113            return Ok(true);
114        }
115        let (Ok(tenant_id), Ok(user_id)) =
116            (claims.tenant_id.parse::<Uuid>(), claims.sub.parse::<Uuid>())
117        else {
118            return Ok(true);
119        };
120        // Through the record cache: a status check, ≤ TTL old (record_cache).
121        match self.user_repo.get_by_id_cached(tenant_id, user_id).await {
122            Ok(user) => Ok(axiam_auth::service::account_may_act(&user).is_ok()),
123            Err(AxiamError::NotFound { .. }) => Ok(false),
124            Err(_) => Err(Status::internal("failed to read the token's subject")),
125        }
126    }
127}

129#[tonic::async_trait]
130impl<U: UserRepository + 'static> TokenService for TokenServiceImpl<U> {
131    async fn validate_token(
132        &self,
133        request: Request<ValidateTokenRequest>,
134    ) -> Result<Response<ValidateTokenResponse>, Status> {
135        // SEC-068: the caller's tenant comes from the interceptor-verified JWT.
136        let caller_tenant = request
137            .extensions()
138            .get::<ValidatedClaims>()
139            .ok_or_else(|| Status::unauthenticated("missing validated claims"))?
140            .0
141            .tenant_id
142            .clone();
143        let req = request.into_inner();
144
145        match validate_access_token(&req.access_token, &self.config) {
146            // SEC-068: refuse to introspect a token belonging to a different
147            // tenant — report it as invalid (not merely denied) so a mesh peer
148            // in tenant A cannot read a tenant-B token's claims by observing the
149            // difference between "denied" and "invalid".
150            Ok(validated) if validated.0.tenant_id == caller_tenant => {
151                let claims = validated.0;
152                let cnf = wire_cnf(&claims);
153                let token_type = wire_token_type(&claims);
154                if cnf.is_some() {
155                    // Worth a line in the log rather than silence: the caller
156                    // is about to receive `valid: true` for a token that is
157                    // NOT usable as a bearer credential, and whether they act
158                    // on that is beyond this service's reach.
159                    tracing::debug!(
160                        token_jti = %claims.jti,
161                        "grpc: validated a sender-constrained token; the caller must prove \
162                         possession of the confirmed key or refuse the request"
163                    );
164                }
165                Ok(Response::new(ValidateTokenResponse {
166                    valid: true,
167                    subject_id: claims.sub,
168                    tenant_id: claims.tenant_id,
169                    org_id: claims.org_id,
170                    exp: claims.exp,
171                    cnf,
172                    token_type,
173                }))
174            }
175            Err(AuthError::Crypto(msg)) => {
176                Err(Status::internal(format!("token validation error: {msg}")))
177            }
178            // Invalid token OR a cross-tenant token (guard above failed) — both
179            // report inactive so the two are indistinguishable to the caller.
180            _ => Ok(Response::new(ValidateTokenResponse {
181                valid: false,
182                subject_id: String::new(),
183                tenant_id: String::new(),
184                org_id: String::new(),
185                exp: 0,
186                // An inactive answer discloses nothing, including whether the
187                // token that failed was bound.
188                cnf: None,
189                token_type: String::new(),
190            })),
191        }
192    }
193
194    async fn introspect_token(
195        &self,
196        request: Request<IntrospectTokenRequest>,
197    ) -> Result<Response<IntrospectTokenResponse>, Status> {
198        // SEC-068: the caller's tenant comes from the interceptor-verified JWT.
199        let caller_tenant = request
200            .extensions()
201            .get::<ValidatedClaims>()
202            .ok_or_else(|| Status::unauthenticated("missing validated claims"))?
203            .0
204            .tenant_id
205            .clone();
206        let req = request.into_inner();
207
208        match validate_access_token(&req.access_token, &self.config) {
209            // SEC-068: only introspect a token from the caller's own tenant; a
210            // cross-tenant token reports inactive (indistinguishable from an
211            // invalid one) so its sub/org/jti claims are not disclosed.
212            //
213            // #520, P23W1-12: and only while the account it names may still
214            // act. Introspection is the documented answer for a caller that
215            // needs immediate revocation (T-39), so a suspended user's token is
216            // inactive here at once, not at `exp`. One indexed user read per
217            // call; `ValidateToken` stays the read-free local check.
218            Ok(validated)
219                if validated.0.tenant_id == caller_tenant
220                    && self.subject_may_act(&validated.0).await?
221                    && self.session_is_live(&validated.0).await =>
222            {
223                let claims = validated.0;
224                let cnf = wire_cnf(&claims);
225                let token_type = wire_token_type(&claims);
226                let permissions = claims
227                    .permissions
228                    .as_ref()
229                    .map(|ps| {
230                        ps.iter()
231                            .map(|p| RptPermission {
232                                resource_id: p.resource_id.to_string(),
233                                resource_scopes: p.resource_scopes.clone(),
234                                exp: p.exp,
235                            })
236                            .collect()
237                    })
238                    .unwrap_or_default();
239                let ext_exchange_iss = claims
240                    .ext_exchange
241                    .as_ref()
242                    .map(|e| e.iss.clone())
243                    .unwrap_or_default();
244                // RFC 9068's `client_id` claim when the token carries one —
245                // the code, refresh, CIBA and device grants, and since #819 a
246                // token exchange (the exchanging client), whose `sub` is the
247                // user. Otherwise the client id lives in `sub` for a
248                // client-credentials token, where `sub_kind` says so. Reading
249                // it off `sub_kind` rather than pattern-matching the string is
250                // what keeps this from guessing: an `oa_`-prefixed user id
251                // would otherwise be reported as a client.
252                let client_id = match (&claims.client_id, claims.sub_kind) {
253                    (Some(client_id), _) => client_id.clone(),
254                    (
255                        None,
256                        axiam_auth::token::SubjectKind::OAuth2Client
257                        | axiam_auth::token::SubjectKind::ServiceAccount,
258                    ) => claims.sub.clone(),
259                    _ => String::new(),
260                };
261                Ok(Response::new(IntrospectTokenResponse {
262                    active: true,
263                    sub: claims.sub,
264                    tenant_id: claims.tenant_id,
265                    org_id: claims.org_id,
266                    iss: claims.iss,
267                    iat: claims.iat,
268                    exp: claims.exp,
269                    jti: claims.jti,
270                    scope: claims.scope.unwrap_or_default(),
271                    client_id,
272                    token_type,
273                    cnf,
274                    permissions,
275                    ext_exchange_iss,
276                }))
277            }
278            Err(AuthError::Crypto(msg)) => {
279                Err(Status::internal(format!("token validation error: {msg}")))
280            }
281            _ => Ok(Response::new(IntrospectTokenResponse {
282                active: false,
283                sub: String::new(),
284                tenant_id: String::new(),
285                org_id: String::new(),
286                iss: String::new(),
287                iat: 0,
288                exp: 0,
289                jti: String::new(),
290                // RFC 7662 §2.2: an inactive response carries `active: false`
291                // and nothing else. Every field below stays empty on purpose —
292                // leaking a `cnf` or a scope for a token the caller may not
293                // introspect would undo the SEC-068 guard above.
294                scope: String::new(),
295                client_id: String::new(),
296                token_type: String::new(),
297                cnf: None,
298                permissions: Vec::new(),
299                ext_exchange_iss: String::new(),
300            })),
301        }
302    }
