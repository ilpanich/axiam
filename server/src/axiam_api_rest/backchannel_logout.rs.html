1//! Back-Channel Logout 1.0 delivery — B5.
2//!
3//! When a session ends, every client that participated in it and registered a
4//! `backchannel_logout_uri` receives a POST carrying a signed logout token.
5//!
6//! # Delivery is best-effort and never blocks the logout
7//!
8//! The user's session is gone from AXIAM the moment the request returns,
9//! whether or not any RP acknowledged. That is the spec's model
10//! (Back-Channel Logout 1.0 §2.6), and the alternative is worse: making logout
11//! synchronous on N external HTTP calls would make the feature a hostage to
12//! the least reliable RP, so one RP whose endpoint hangs would make logging
13//! out of AXIAM hang for the user.
14//!
15//! # Why this doesn't ride on the webhook pipeline
16//!
17//! `claude_dev/logout-and-par-design.md` originally proposed reusing the AMQP
18//! webhook path for its retry and DLQ. It is delivered here instead, with a
19//! small bounded retry of its own, for two reasons: the webhook consumer signs
20//! every delivery with the HMAC header scheme of §13, which is exactly what a
21//! logout token must **not** carry (the signed JWT *is* the authentication,
22//! and offering a second, cheaper check invites an RP to verify that one); and
23//! routing through AMQP would make logout notification silently depend on the
24//! broker being up, which is a stronger coupling than a best-effort side
25//! effect should have. The retry below keeps the property that actually
26//! matters — a briefly-unavailable RP still gets told.
27//!
28//! # The only way out of the process (R101W2-04)
29//!
30//! The logout token is a signed, session-naming credential, and the URI it goes
31//! to was typed by a tenant administrator. Delivery therefore makes **one hop
32//! through `ssrf::guarded_fetch_no_redirect`** with `allow_private = false`: the
33//! name is resolved fresh, every address must be globally routable, the
34//! validated address is pinned into the connection and `https` is required. A
35//! `3xx` comes back as a response and is a failed attempt: the token cannot
36//! reach a host the client never registered. The URI is also validated when the
37//! client is registered (`validate_backchannel_logout_uri`), but a row written
38//! under 1.0.0 may hold anything, so the guard here is the one that counts.
39//!
40//! | The RP answers | Outcome |
41//! |---|---|
42//! | any `2xx` | delivered |
43//! | a `3xx` | failed attempt, retried, never followed |
44//! | anything else, a timeout, a refused address, no connection | failed attempt, retried |
45

53/// Whether delivery may reach a non-public address. `false` unless
54/// [`admit_private_networks_for_tests`] was called, which nothing outside the
55/// test suites does (a source pin below asserts it).
56static ADMIT_PRIVATE_FOR_TESTS: AtomicBool = AtomicBool::new(false);
57
58/// **Test seam, never called by the composition root or any production path.**
59///
60/// Lets the first hop reach a loopback receiver over plain `http`, the way
61/// `WebhookDeliverer::admitting_private_networks_for_tests` and the SSF, SCIM
62/// and CIBA-ping deliverers' seams do. Delivery here is a detached task started
63/// from a handler, so there is no deliverer value to carry the bit; a test
64/// binary that needs a loopback RP calls this once. Redirects stay unfollowed
65/// either way.
66#[doc(hidden)]
67pub fn admit_private_networks_for_tests() {
68    ADMIT_PRIVATE_FOR_TESTS.store(true, Ordering::SeqCst);
69}

71/// How many times a single client's endpoint is tried.
72///
73/// Three attempts covers a restart or a momentary blip, which is what
74/// "SHOULD retry" in §2.6 is for. It deliberately does not cover a sustained
75/// outage: an RP that is down for minutes will keep its own session until it
76/// expires, and queueing indefinitely would mean holding a session-termination
77/// command — a security-relevant instruction — for an unbounded time.
78const MAX_ATTEMPTS: usize = 3;
79
80/// Per-attempt timeout. Short, because this runs detached and a hung socket
81/// would otherwise pin the task for the client's default timeout.
82const ATTEMPT_TIMEOUT: Duration = Duration::from_secs(5);
83
84/// Backoff before attempt N+1.
85const BACKOFF: [Duration; 2] = [Duration::from_millis(500), Duration::from_millis(2000)];
86
87/// One client's delivery target: where to POST, and the token to send.
88#[derive(Debug, Clone)]
89pub struct LogoutDelivery {
90    pub client_id: String,
91    pub uri: String,
92    pub logout_token: String,
93}
94
95/// Select the clients that should be notified.
96///
97/// Two filters, both load-bearing:
98///
99/// - **Only clients that participated in the session.** Broadcasting to every
100///   registered client would tell clients that were never part of the session
101///   that one just ended, which leaks its existence.
102/// - **Only clients with a `backchannel_logout_uri`.** A client without one
103///   has not opted in; it is skipped, not retried.
104///
105/// Participation records are deduplicated here rather than constrained in the
106/// datastore, because a client legitimately re-authorizes within one session.
107pub fn select_targets(
108    participants: &[String],
109    clients: &[OAuth2Client],
110    tokens: impl Fn(&str) -> Option<String>,
111) -> Vec<LogoutDelivery> {
112    let mut seen = std::collections::HashSet::new();
113    let mut out = Vec::new();
114    for client_id in participants {
115        if !seen.insert(client_id.clone()) {
116            continue;
117        }
118        let Some(client) = clients.iter().find(|c| &c.client_id == client_id) else {
119            continue;
120        };
121        let Some(uri) = client.backchannel_logout_uri.as_deref() else {
122            continue;
123        };
124        let Some(token) = tokens(client_id) else {
125            continue;
126        };
127        out.push(LogoutDelivery {
128            client_id: client_id.clone(),
129            uri: uri.to_string(),
130            logout_token: token,
131        });
132    }
133    out
134}

136/// POST one logout token, retrying a bounded number of times.
137///
138/// One guarded, non-redirecting hop per attempt (R101W2-04). `allow_private` is
139/// `false` in production; [`deliver_all`] passes the test seam's value.
140///
141/// Returns `true` if some attempt got a 2xx. Never panics and never propagates:
142/// the caller is a detached task whose failure must not affect the logout that
143/// triggered it.
144pub async fn deliver_one(delivery: &LogoutDelivery, allow_private: bool) -> bool {
145    for attempt in 0..MAX_ATTEMPTS {
146        if attempt > 0 {
147            tokio::time::sleep(BACKOFF[(attempt - 1).min(BACKOFF.len() - 1)]).await;
148        }
149        // The resolve and the connect are inside the guard, so the timeout
150        // wraps the whole attempt rather than only the request.
151        let result = tokio::time::timeout(
152            ATTEMPT_TIMEOUT,
153            ssrf::guarded_fetch_no_redirect(&delivery.uri, allow_private, |client, target| {
154                client
155                    .post(target)
156                    .timeout(ATTEMPT_TIMEOUT)
157                    // Back-Channel Logout 1.0 §2.5: form-encoded, one parameter.
158                    .form(&[("logout_token", &delivery.logout_token)])
159            }),
160        )
161        .await;
162
163        match result {
164            Ok(Ok(resp)) if resp.status().is_success() => {
165                debug!(
166                    client_id = %delivery.client_id,
167                    attempt = attempt + 1,
168                    "back-channel logout delivered"
169                );
170                return true;
171            }
172            Ok(Ok(resp)) if resp.status().is_redirection() => {
173                // Not followed: the token goes to the host the client
174                // registered or nowhere.
175                warn!(
176                    client_id = %delivery.client_id,
177                    status = resp.status().as_u16(),
178                    attempt = attempt + 1,
179                    "back-channel logout answered with a redirect, which is not followed"
180                );
181            }
182            Ok(Ok(resp)) => {
183                // Never log the token: it is a session-termination command,
184                // and a log line carrying one is a replayable instruction.
185                warn!(
186                    client_id = %delivery.client_id,
187                    status = resp.status().as_u16(),
188                    attempt = attempt + 1,
189                    "back-channel logout rejected by RP"
190                );
191            }
192            Ok(Err(e)) => {
193                warn!(
194                    client_id = %delivery.client_id,
195                    error = %e,
196                    attempt = attempt + 1,
197                    "back-channel logout delivery failed"
198                );
199            }
200            Err(_elapsed) => {
201                warn!(
202                    client_id = %delivery.client_id,
203                    attempt = attempt + 1,
204                    "back-channel logout delivery timed out"
205                );
206            }
207        }
208    }
209    false
210}

212/// Fan out to every target. Intended to be spawned, not awaited by a handler.
213pub async fn deliver_all(deliveries: Vec<LogoutDelivery>) {
214    let allow_private = ADMIT_PRIVATE_FOR_TESTS.load(Ordering::SeqCst);
215    for delivery in &deliveries {
216        // Sequential rather than concurrent: the list is one user's RPs, so it
217        // is small, and a slow RP delaying the next one costs nothing that
218        // matters — the user is already logged out.
219        deliver_one(delivery, allow_private).await;
220    }
221}

373    /// root: no file of `axiam-server` calls this module's setter. (The boot

379            for entry in std::fs::read_dir(dir).unwrap().flatten() {
380                let path = entry.path();
381                if path.is_dir() {
382                    walk(&path, hits);
383                } else if path.extension().is_some_and(|e| e == "rs")
384                    && std::fs::read_to_string(&path)
385                        .unwrap()
386                        .contains("backchannel_logout::admit_private_networks_for_tests")
387                {
388                    hits.push(path.display().to_string());
389                }
390            }
391        }
392        let server = std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../axiam-server/src");
393        let mut hits = Vec::new();
394        walk(&server, &mut hits);
395        assert!(hits.is_empty(), "axiam-server names the seam: {hits:?}");
396    }

406    /// A `307` is a failed attempt and its `Location` is never requested: the
407    /// second server, which the first one names, sees nothing.
408    #[tokio::test]
409    async fn a_redirect_from_the_rp_is_not_followed() {
410        use wiremock::matchers::method;
411        use wiremock::{Mock, MockServer, ResponseTemplate};
412
413        let elsewhere = MockServer::start().await;
414        Mock::given(method("POST"))
415            .respond_with(ResponseTemplate::new(200))
416            .mount(&elsewhere)
417            .await;
418        let rp = MockServer::start().await;
419        Mock::given(method("POST"))
420            .respond_with(
421                ResponseTemplate::new(307)
422                    .insert_header("Location", format!("{}/collect", elsewhere.uri())),
423            )
424            .mount(&rp)
425            .await;
426
427        let delivered = deliver_one(&delivery_to(format!("{}/logout", rp.uri())), true).await;
428
429        assert!(!delivered, "a 3xx is a failed delivery");
430        let attempts = rp.received_requests().await.unwrap().len();
431        assert_eq!(attempts, MAX_ATTEMPTS, "it is retried, not followed");
432        assert!(
433            elsewhere.received_requests().await.unwrap().is_empty(),
434            "the redirect target must see nothing"
435        );
436    }

462    /// Production posture (`allow_private = false`): a stored URI that points at
463    /// a private, loopback, link-local or metadata address, or speaks plain
464    /// `http`, reaches nothing. A client registered under 1.0.0 may hold such a
465    /// value (registration did not validate it); delivery refuses it.
466    #[tokio::test]
467    async fn a_stored_uri_at_a_non_public_address_reaches_nothing() {
468        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
469        let port = listener.local_addr().unwrap().port();
470        for uri in [
471            format!("https://127.0.0.1:{port}/logout"),
472            format!("http://127.0.0.1:{port}/logout"),
473            "https://10.0.0.8/logout".to_owned(),
474            "https://192.168.1.10/logout".to_owned(),
475            "https://169.254.169.254/latest/meta-data/".to_owned(),
476            "https://[::1]/logout".to_owned(),
477        ] {
478            // One attempt is enough to show the refusal; the retry loop is
479            // exercised by the redirect test above.
480            let result = tokio::time::timeout(
481                ATTEMPT_TIMEOUT,
482                ssrf::guarded_fetch_no_redirect(&uri, false, |c, u| c.post(u)),
483            )
484            .await
485            .expect("a refusal is immediate");
486            assert!(result.is_err(), "{uri} must be refused by the guard");
487        }
488        assert!(
489            tokio::time::timeout(Duration::from_millis(200), listener.accept())
490                .await
491                .is_err(),
492            "the loopback listener must see no connection"
493        );
494        // And through the delivery function itself.
495        let uri = format!("https://127.0.0.1:{port}/logout");
496        let attempt = tokio::time::timeout(
497            Duration::from_secs(10),
498            deliver_one(&delivery_to(uri), false),
499        )
500        .await
501        .expect("three refused attempts finish within the backoff schedule");
502        assert!(!attempt);
503        assert!(
504            tokio::time::timeout(Duration::from_millis(200), listener.accept())
505                .await
506                .is_err(),
507            "the loopback listener must still see no connection"
508        );
509    }
