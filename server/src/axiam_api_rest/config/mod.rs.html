11/// Configuration for the Actix-Web REST API server.
12#[derive(Debug, Clone, Deserialize)]
13#[serde(default)]
14pub struct ServerConfig {
15    /// Bind address (default: "127.0.0.1").
16    pub host: String,
17    /// Bind port (default: 8090).
18    pub port: u16,
19    /// Allowed CORS origins. Empty disables cross-origin requests (restrictive).
20    pub cors_allowed_origins: Vec<String>,
21    /// Optional direct-TLS termination. Disabled by default — the recommended
22    /// deployment terminates TLS at the proxy/load-balancer layer (ASVS V9.1.x,
23    /// D-06). When enabled, the server binds with rustls restricted to TLS 1.3
24    /// (see `axiam-server`).
25    pub tls: TlsConfig,
26    /// Set `TCP_NODELAY` (disable Nagle's algorithm) on accepted connections.
27    ///
28    /// **Default: `true`.** Override with
29    /// `AXIAM__SERVER__TCP_NODELAY=false`.
30    ///
31    /// # Why this exists and why the default is `true` (I5)
32    ///
33    /// actix-web 4.14 exposes `HttpServer::tcp_nodelay` but leaves it *unset*
34    /// by default, and an unset value means "never call `set_nodelay`"
35    /// (`actix-http-3.13.1 src/service.rs:28-35`, `src/h2/service.rs:29-35`:
36    /// `fn desired_nodelay(tcp_nodelay: Option<bool>) -> Option<bool> {
37    /// tcp_nodelay }`). The kernel default therefore applied and Nagle stayed
38    /// **on** for every REST connection — while the gRPC listener has had it
39    /// off all along, because tonic's `Server` defaults `tcp_nodelay` to `true`
40    /// (`tonic .../transport/server/mod.rs:132`).
41    ///
42    /// Nagle only costs anything when a response reaches the socket as more
43    /// than one write and the last write is a partial segment: the kernel holds
44    /// that final fragment until the peer acknowledges the previous one, and
45    /// Linux's delayed-ACK timer is **40 ms** (`TCP_DELACK_MIN`). A response
46    /// emitted as a single `sendmsg` is immune (`tcp_push` sets
47    /// `TCP_NAGLE_PUSH` on the tail), which is why an HTTP/1.1 plaintext bind
48    /// can look clean while the h2-over-TLS bind — where the `h2` crate emits
49    /// HEADERS and DATA frames through separate `poll_write`s into
50    /// `tokio-rustls`, each becoming its own TLS record — is exposed.
51    ///
52    /// That is the leading hypothesis for the run-4 TLS client-credentials
53    /// plateau: 1 180 req/s at a flat p50/p95/p99 of 42.6/43.9/44.8 ms with
54    /// nothing saturated, i.e. ~40 ms of pure timer plus ~2.6 ms of work, on
55    /// the one endpoint whose response happened to straddle a segment boundary.
56    /// It is a *hypothesis*, not a measured result: run 5 must A/B this knob
57    /// (see the run-5 runbook). Keeping it configurable is what makes that A/B
58    /// possible; `false` reproduces the pre-I5 behaviour exactly.
59    ///
60    /// Setting `TCP_NODELAY` is standard for request/response HTTP servers and
61    /// has no security implication.
62    pub tcp_nodelay: bool,
63}
64
65impl Default for ServerConfig {
66    fn default() -> Self {
67        Self {
68            host: "127.0.0.1".into(),
69            port: 8090,
70            cors_allowed_origins: Vec::new(),
71            tls: TlsConfig::default(),
72            tcp_nodelay: true,
73        }
74    }

77/// Native client-certificate (mTLS) authentication policy for the direct-TLS
78/// listener (D3).
79///
80/// When client auth is enabled, rustls verifies the presented client
81/// certificate against the CA bundle at [`TlsConfig::client_ca_path`] during
82/// the TLS handshake, and the *verified* leaf certificate is exposed to request
83/// handlers via the connection extensions — the certificate-auth flow then
84/// consumes that verified cert rather than a spoofable proxy header.
85#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Deserialize)]
86#[serde(rename_all = "lowercase")]
87pub enum ClientAuth {
88    /// No client certificate requested (server-auth only). Backward-compatible
89    /// default — the rustls config is built with `with_no_client_auth()`.
90    #[default]
91    Off,
92    /// Request a client certificate but allow anonymous clients. A presented
93    /// certificate is still verified against the CA bundle; connections without
94    /// one are accepted (`WebPkiClientVerifier::builder(..).allow_unauthenticated()`).
95    Optional,
96    /// Require a client certificate verified against the CA bundle. The TLS
97    /// handshake is rejected if the client presents no certificate or an
98    /// unverifiable one (`WebPkiClientVerifier::builder(..).build()`).
99    Required,
100}
101
102/// Default [`TlsConfig::reload_interval_secs`]: one hour.
103///
104/// Chosen against the renewal cadence it exists to catch, not against how fast
105/// a change could be noticed. Let's Encrypt issues for 90 days and ACME clients
106/// renew at 60, so the window between "a new certificate is on disk" and "the
107/// old one stops being accepted" is thirty days. An hourly poll closes it with
108/// four orders of magnitude to spare, and is rare enough that the `stat` never
109/// shows up anywhere.
110pub const DEFAULT_TLS_RELOAD_INTERVAL_SECS: u64 = 3600;
111
112/// Direct-TLS configuration for the REST API listener.
113///
114/// TLS is **opt-in**: the default (`enabled = false`) preserves the plaintext
115/// bind used behind a TLS-terminating proxy. When `enabled = true`, both
116/// `cert_path` and `key_path` must point at readable PEM files; the server
117/// negotiates TLS 1.3 only (ASVS V9.1.2), whose cipher suites are all
118/// ASVS-approved (V9.1.3).
119#[derive(Debug, Clone, Deserialize)]
120#[serde(default)]
121pub struct TlsConfig {
122    /// Enable direct TLS termination in the server process.
123    pub enabled: bool,
124    /// Path to the PEM-encoded certificate chain (leaf first).
125    pub cert_path: Option<PathBuf>,
126    /// Path to the PEM-encoded private key (PKCS#8, PKCS#1, or SEC1).
127    pub key_path: Option<PathBuf>,
128    /// Offer HTTP/2 (`h2`) over ALPN alongside `http/1.1`. Default `true`
129    /// (backward-compatible; matches what the actix-web rustls bind advertises).
130    ///
131    /// Set `false` to build the rustls listener advertising `http/1.1` only —
132    /// used to run the p2 benchmark apples-to-apples with the plaintext p0
133    /// listener (which is HTTP/1.1), isolating the h2-vs-h1.1 throughput effect
134    /// (B2). See `axiam-server`'s `tls` module for the important caveat that the
135    /// actix-web `HttpServer` bind re-adds `h2` to ALPN regardless.
136    pub http2: bool,
137    /// Native client-certificate (mTLS) policy (D3). Default `off` keeps the
138    /// server-auth-only behaviour. `optional`/`required` build the rustls config
139    /// with a `WebPkiClientVerifier` over [`Self::client_ca_path`].
140    pub client_auth: ClientAuth,
141    /// Path to the PEM CA bundle used to verify client certificates. Required
142    /// (and must be readable) when `client_auth` is `optional` or `required`;
143    /// ignored when `off`.
144    pub client_ca_path: Option<PathBuf>,
145    /// Where to write the client-CA bundle built from the organization CAs an
146    /// operator flagged as mTLS trust anchors.
147    ///
148    /// The flag lives on the CA
149    /// ([`axiam_core::models::certificate::CaCertificate::mtls_trust_anchor`]),
150    /// set through the admin UI or the API. At startup the server collects every
151    /// flagged CA's **public** certificate, writes them here as one PEM bundle,
152    /// and — only if the operator has not set [`Self::client_auth`] and
153    /// [`Self::client_ca_path`] themselves — points client-certificate
154    /// verification at it in `optional` mode.
155    ///
156    /// A deployment that flags no CA never writes this file and is unaffected.
157    /// An operator who configures `client_auth` and `client_ca_path` explicitly
158    /// keeps exactly what they configured: a value someone typed is a decision,
159    /// and quietly overriding it is how a trust store ends up containing
160    /// something nobody chose.
161    ///
162    /// Defaults to `client-ca-bundle.pem` beside the server's other TLS
163    /// material when `cert_path` is set, and is otherwise required for the
164    /// feature to do anything.
165    pub client_ca_bundle_path: Option<PathBuf>,
166    /// How often, in seconds, to re-read [`Self::cert_path`] and
167    /// [`Self::key_path`] and install the leaf certificate when it differs from
168    /// the one being served. `0` disables polling. Default `3600`.
169    ///
170    /// The comparison is on the parsed DER chain, not on file metadata: an
171    /// mtime can change without the certificate changing (a `touch`, a
172    /// re-render of an identical file) and — worse — can fail to change when
173    /// the content does. Comparing what will actually be served costs one file
174    /// read an hour and cannot be fooled either way.
175    ///
176    /// # Why this exists
177    ///
178    /// rustls resolves the server's certificate per handshake, but the pair is
179    /// read from disk exactly once unless something re-reads it. An ACME client
180    /// renewing a 90-day Let's Encrypt certificate at day 60 therefore has no
181    /// effect on a running server, and the listener serves an expired
182    /// certificate on day 90.
183    ///
184    /// `SIGHUP` is the primary, immediate trigger and is what an ACME deploy
185    /// hook should send. This poll is the safety net for the case that actually
186    /// happens in the field: the hook was never wired up, or the process runs
187    /// under something that does not forward signals, and nobody finds out
188    /// until the certificate expires. Reading two small files an hour costs
189    /// nothing.
190    ///
191    /// A poll can observe a half-updated pair — certbot writes the chain and
192    /// the key as two operations — so a reload that fails to produce a matching
193    /// pair leaves the previous certificate serving and is retried on the next
194    /// tick. See `axiam_server::tls::reload_leaf_certificate`.
195    pub reload_interval_secs: u64,
196}
197
198impl Default for TlsConfig {
199    fn default() -> Self {
200        Self {
201            enabled: false,
202            cert_path: None,
203            key_path: None,
204            http2: true,
205            client_auth: ClientAuth::Off,
206            client_ca_path: None,
207            client_ca_bundle_path: None,
208            reload_interval_secs: DEFAULT_TLS_RELOAD_INTERVAL_SECS,
209        }
210    }
211}

213impl ServerConfig {

220#[cfg(test)]
221mod tests {
222    use super::*;
223
224    /// I5: the whole point of the knob is that the socket option is now set
225    /// *explicitly*. If this default ever flips to `false`, the REST listener
226    /// silently goes back to running with Nagle enabled while gRPC does not.
227    #[test]
228    fn tcp_nodelay_defaults_to_enabled() {
229        assert!(
230            ServerConfig::default().tcp_nodelay,
231            "TCP_NODELAY must default to enabled — actix-web leaves the socket \
232             option untouched when unset, which means Nagle stays on"
233        );
234    }
235
236    /// The knob must be reachable from configuration, since the run-5
237    /// measurement plan depends on being able to turn it off.
238    #[test]
239    fn tcp_nodelay_can_be_disabled_from_config() {
240        let parsed: ServerConfig = serde_json::from_str(r#"{"tcp_nodelay": false}"#)
241            .expect("a partial server config must deserialize");
242        assert!(!parsed.tcp_nodelay);
243        // Everything else still falls back to the defaults.
244        assert_eq!(parsed.port, ServerConfig::default().port);
245    }
246
247    #[test]
248    fn client_auth_defaults_to_off() {
249        assert_eq!(ClientAuth::default(), ClientAuth::Off);
250        assert_eq!(TlsConfig::default().client_auth, ClientAuth::Off);
251        assert!(TlsConfig::default().client_ca_path.is_none());
252    }
253
254    #[test]
255    fn client_auth_parses_all_three_values() {
256        for (raw, expected) in [
257            ("\"off\"", ClientAuth::Off),
258            ("\"optional\"", ClientAuth::Optional),
259            ("\"required\"", ClientAuth::Required),
260        ] {
261            let parsed: ClientAuth =
262                serde_json::from_str(raw).unwrap_or_else(|e| panic!("{raw} must parse: {e}"));
263            assert_eq!(parsed, expected, "{raw}");
264        }
265    }
266
267    #[test]
268    fn client_auth_rejects_invalid_value() {
269        let err = serde_json::from_str::<ClientAuth>("\"enabled\"");
270        assert!(err.is_err(), "unknown client_auth value must be rejected");
271    }
272
273    #[test]
274    fn tls_config_deserializes_client_auth_fields() {
275        let cfg: TlsConfig = serde_json::from_value(serde_json::json!({
276            "enabled": true,
277            "cert_path": "/etc/axiam/server.crt",
278            "key_path": "/etc/axiam/server.key",
279            "client_auth": "required",
280            "client_ca_path": "/etc/axiam/ca.crt",
281        }))
282        .expect("TlsConfig with client-auth fields must deserialize");
283        assert_eq!(cfg.client_auth, ClientAuth::Required);
284        assert_eq!(
285            cfg.client_ca_path.as_deref(),
286            Some(std::path::Path::new("/etc/axiam/ca.crt"))
287        );
288    }
