5/// Rate-limit bucket-key derivation mode (D8).
6///
7/// Environment variable: `AXIAM__RATE_LIMIT__KEY` = `ip` | `client_id` |
8/// `ip_client_id` (default `ip`).
9///
10/// **Why this exists (NAT'd-fleet lesson):** the bucket key was
11/// unconditionally `"{endpoint}:{ip}"` (see
12/// `middleware::rate_limit_shared::RateLimitShared` and
13/// `extractors::rate_limit::XForwardedForKeyExtractor`). Behind a NAT/proxy
14/// fleet (many distinct OAuth2 clients — e.g. IoT devices or microservices —
15/// egressing through one shared IP), every client sharing that IP collided
16/// into a SINGLE bucket, so one noisy/misbehaving client could exhaust the
17/// `/oauth2/token`, `/oauth2/revoke`, or `/oauth2/introspect` quota for every
18/// other client behind the same NAT gateway. `client_id` and `ip_client_id`
19/// give each OAuth2 client its own bucket (optionally still scoped per-IP)
20/// on the endpoints where a client identity is actually known.
21///
22/// **Scope — where this setting applies:** ONLY the three endpoints where an
23/// OAuth2 client authenticates itself via a form-encoded `client_id`
24/// (`client_secret_post`, RFC 6749 §2.3.1): `/oauth2/token`,
25/// `/oauth2/revoke`, `/oauth2/introspect` (see `handlers::oauth2` and
26/// `server.rs`'s wiring of `RateLimitShared::new_client_identity_aware` /
27/// the client-aware governor for exactly those three resources).
28///
29/// **`/auth/login` (and every other rate-limited endpoint) ALWAYS keys
30/// per-IP, regardless of this setting.** Login authenticates a *user* via
31/// username/password — there is no OAuth2 client identity anywhere in that
32/// request, so there is nothing meaningful to key on besides the source IP.
33/// This is intentional and NOT a bug: switching this config value never
34/// changes login's (or MFA's, or password-reset's, etc.) rate-limit
35/// behavior. See `server.rs` — those resources are wired with the plain
36/// `build_governor`/`RateLimitShared::new` constructors, which never read
37/// this field.
38#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize, Default)]
39#[serde(rename_all = "snake_case")]
40pub enum RateLimitKeyMode {
41    /// Key on source IP only (current/default behavior, unchanged).
42    #[default]
43    Ip,
44    /// Key on the OAuth2 `client_id` alone — independent buckets per client,
45    /// regardless of which IP(s) it connects from.
46    ClientId,
47    /// Key on the `(ip, client_id)` pair — independent buckets per client
48    /// AND per IP, so a compromised/leaked client credential rate-limited
49    /// from one IP doesn't automatically throttle the same client_id
50    /// operating legitimately from a different IP.
51    IpClientId,
52}
53
54impl RateLimitKeyMode {
55    /// Stable, log-safe name — identical to the `AXIAM__RATE_LIMIT__KEY`
56    /// value that selects this mode.
57    pub const fn as_str(self) -> &'static str {
58        match self {
59            Self::Ip => "ip",
60            Self::ClientId => "client_id",
61            Self::IpClientId => "ip_client_id",
62        }
63    }
64}

66/// Deployment rate-limit **posture preset** (G7).
67///
68/// Environment variable: `AXIAM__RATE_LIMIT__PROFILE` = `internet` |
69/// `gateway` | `mesh` (default `internet`).
70///
71/// **Why a preset instead of new defaults.** Benchmark run 3 measured that
72/// the shipped defaults (`token 20/min`, `introspect 10/min`,
73/// `revoke 10/min`, `authz_check 300/min`, all keyed per-IP) flatten a
74/// 50-VU single-source-IP load to ~0–14 req/s at ~100% `429`, while the
75/// same 2-core server sustains ~1 800 token issuances/s and ~740–2 300
76/// authz checks/s. Those defaults are *correct* for a small
77/// internet-facing deployment and *wrong* for an M2M/NAT'd fleet where
78/// many OAuth2 clients share one egress IP. Rather than silently weaken
79/// the internet-facing abuse posture for everyone, the M2M sizing is an
80/// **opt-in preset**: one env var moves the whole machine-traffic family
81/// coherently (key mode + token/introspect/revoke/authz + the gRPC authz
82/// ceiling), and the shipped defaults are byte-for-byte unchanged.
83///
84/// **Human endpoints are never touched by any preset.** `login_per_min`,
85/// `register_per_min`, `password_reset_per_min` and `mfa_per_min` stay at
86/// their strict per-IP defaults in every profile — they gate
87/// password/OTP guessing and account-enumeration, which are human-scale
88/// attacks whose economics do not change because the deployment is a
89/// service mesh. Raise them deliberately, one env var at a time.
90///
91/// **Security caveat an operator MUST read before selecting a preset**
92/// (also in `docs/deployment/rate-limit-sizing.md`): both non-default
93/// presets set the key mode to `client_id`, and the `client_id` is parsed
94/// from the *unauthenticated* request body (`client_secret_post`,
95/// RFC 6749 §2.3.1) before any credential is verified. An attacker can
96/// therefore rotate `client_id` values to mint fresh buckets. Under those
97/// modes the token/introspect/revoke limits are a **fairness / noisy-
98/// neighbour control between cooperating clients**, not an anti-abuse
99/// control against a determined attacker; the anti-abuse job moves to the
100/// edge (WAF / API gateway / mTLS). This is a property of the D8 key
101/// modes themselves, not of the presets — the presets just make it the
102/// active posture, which is exactly why they are opt-in.
103#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize, Default)]
104#[serde(rename_all = "snake_case")]
105pub enum RateLimitProfile {
106    /// Shipped default: small internet-facing deployment, strict per-IP
107    /// limits on every endpoint. No field is preset — the defaults in
108    /// [`RateLimitConfig::default`] apply verbatim.
109    #[default]
110    Internet,
111    /// M2M fleet reaching an internet-facing AXIAM through a shared
112    /// NAT/egress gateway (the topology run 3 showed the defaults break).
113    /// Per-client buckets and machine-scale ceilings on the machine
114    /// endpoints; human endpoints unchanged.
115    Gateway,
116    /// AXIAM reachable only on a private network / service mesh (no
117    /// internet exposure on the machine endpoints). Ceilings sized as
118    /// runaway-loop guards rather than abuse controls; human endpoints
119    /// still unchanged, because "private network" is not "no humans".
120    Mesh,
121}
122
123/// The machine-traffic limits a non-default [`RateLimitProfile`] presets.
124///
125/// Human endpoints are deliberately absent from this struct — a preset
126/// cannot express them, so it cannot weaken them.
127#[derive(Debug, Clone, Copy, PartialEq, Eq)]
128pub struct MachineLimitPreset {
129    /// Bucket-key mode for `/oauth2/token`, `/oauth2/revoke`,
130    /// `/oauth2/introspect` (the only endpoints that honor it).
131    pub key: RateLimitKeyMode,
132    /// `/oauth2/token` per minute per bucket.
133    pub token_per_min: u32,
134    /// `/oauth2/introspect` per minute per bucket.
135    pub introspect_per_min: u32,
136    /// `/oauth2/revoke` per minute per bucket.
137    pub revoke_per_min: u32,
138    /// REST authz-check endpoints per minute per IP.
139    pub authz_check_per_min: u32,
140    /// gRPC `AuthorizationService` per second per IP — applied to
141    /// `axiam_api_grpc::GrpcConfig::grpc_authz_per_sec` by the composition
142    /// root (`axiam-server::main`). Kept here so the whole family has one
143    /// source of truth.
144    pub grpc_authz_per_sec: u32,
145}
146
147/// `AXIAM__RATE_LIMIT__KEY` — presence pins [`RateLimitConfig::key`].
148pub const ENV_KEY: &str = "AXIAM__RATE_LIMIT__KEY";
149/// `AXIAM__RATE_LIMIT__TOKEN_PER_MIN`.
150pub const ENV_TOKEN_PER_MIN: &str = "AXIAM__RATE_LIMIT__TOKEN_PER_MIN";
151/// `AXIAM__RATE_LIMIT__INTROSPECT_PER_MIN`.
152pub const ENV_INTROSPECT_PER_MIN: &str = "AXIAM__RATE_LIMIT__INTROSPECT_PER_MIN";
153/// `AXIAM__RATE_LIMIT__REVOKE_PER_MIN`.
154pub const ENV_REVOKE_PER_MIN: &str = "AXIAM__RATE_LIMIT__REVOKE_PER_MIN";
155/// `AXIAM__RATE_LIMIT__AUTHZ_CHECK_PER_MIN`.
156pub const ENV_AUTHZ_CHECK_PER_MIN: &str = "AXIAM__RATE_LIMIT__AUTHZ_CHECK_PER_MIN";
157
158impl RateLimitProfile {
159    /// Stable, log-safe name — identical to the
160    /// `AXIAM__RATE_LIMIT__PROFILE` value that selects this profile.
161    pub const fn as_str(self) -> &'static str {
162        match self {
163            Self::Internet => "internet",
164            Self::Gateway => "gateway",
165            Self::Mesh => "mesh",
166        }
167    }
168
169    /// The machine-traffic limits this profile presets, or `None` for
170    /// [`RateLimitProfile::Internet`] (which presets nothing at all).
171    ///
172    /// Every number below is anchored to the run-3 measured envelope on
173    /// 2-core server / 2-core DB containers (`benchmarks/PUBLIC_BENCH_ANALYSIS.md`
174    /// §6, laptop-class and explicitly temporary): token issuance
175    /// ~1 800/s (~108 000/min), introspection ~2 200/s (~132 000/min),
176    /// authz checks ~740/s cache-off and ~2 300/s cache-on
177    /// (~44 400 / ~138 000 per minute), gRPC single authz check ~603/s.
178    /// The presets are *per bucket*, so they are deliberately a small
179    /// fraction of the whole-server ceiling.
180    pub const fn preset(self) -> Option<MachineLimitPreset> {
181        match self {
182            Self::Internet => None,
183            // 600/min = 10/s per client ≈ 0.55% of the measured 1 800/s
184            // server ceiling: ~180 distinct clients running flat out are
185            // needed to saturate, so one compromised credential cannot
186            // become the DoS engine. Introspect is 10× token per the
187            // public §6 rule (resource servers introspect per request).
188            // Authz 6 000/min = 100/s per client, the low end of §6's
189            // 6 000–60 000 band.
190            Self::Gateway => Some(MachineLimitPreset {
191                key: RateLimitKeyMode::ClientId,
192                token_per_min: 600,
193                introspect_per_min: 6_000,
194                revoke_per_min: 600,
195                authz_check_per_min: 6_000,
196                grpc_authz_per_sec: 1_000,
197            }),
198            // Private-network sizing: 6 000/min token = 100/s per client
199            // (~5.6% of the measured ceiling) and 60 000/min authz =
200            // 1 000/s per client, ABOVE the measured 740/s cache-off
201            // ceiling — i.e. on this hardware the authz limit stops a
202            // runaway retry loop, not an attacker. Documented as such.
203            Self::Mesh => Some(MachineLimitPreset {
204                key: RateLimitKeyMode::ClientId,
205                token_per_min: 6_000,
206                introspect_per_min: 60_000,
207                revoke_per_min: 6_000,
208                authz_check_per_min: 60_000,
209                grpc_authz_per_sec: 5_000,
210            }),
211        }
212    }
213}

215/// What the composition root actually resolved — the input to the startup
216/// posture log line.
217#[derive(Debug, Clone, PartialEq, Eq)]
218pub struct RateLimitPosture {
219    /// The selected profile.
220    pub profile: RateLimitProfile,
221    /// `true` when a non-default preset was applied to at least the fields
222    /// the operator did not pin explicitly.
223    pub preset_applied: bool,
224    /// Env vars the operator set explicitly, which therefore beat the
225    /// preset. Names only — never values (some are not secret, but the
226    /// log line stays uniformly value-free for this list).
227    pub operator_overrides: Vec<&'static str>,
228    /// gRPC authz ceiling the preset wants, for the composition root to
229    /// hand to `axiam_api_grpc::GrpcConfig`. `None` under `internet`.
230    pub grpc_authz_per_sec_preset: Option<u32>,
231}
232
233impl RateLimitPosture {
234    /// Comma-joined override list for the startup log (`"none"` when empty).
235    pub fn overrides_display(&self) -> String {
236        if self.operator_overrides.is_empty() {
237            "none".to_owned()
238        } else {
239            self.operator_overrides.join(",")
240        }
241    }
242}

244/// Rate limit configuration.
245/// Environment variables: AXIAM__RATE_LIMIT__LOGIN_PER_MIN, etc.
246#[derive(Debug, Clone, Deserialize)]
247#[serde(default)]
248pub struct RateLimitConfig {
249    /// Max login requests per minute per IP (default: 10).
250    pub login_per_min: u32,
251    /// Max register requests per minute per IP (default: 5).
252    pub register_per_min: u32,
253    /// Max oauth2/token requests per minute per client (default: 20).
254    pub token_per_min: u32,
255    /// Max password-reset requests per minute per IP (default: 3).
256    pub password_reset_per_min: u32,
257    /// Max MFA requests per minute per IP (default: 5).
258    /// Covers /auth/mfa/enroll, /confirm, /verify, /setup/enroll, /setup/confirm (SEC-020).
259    pub mfa_per_min: u32,
260    /// Max oauth2/introspect requests per minute per IP (default: 10).
261    /// SEC-020: introspect endpoint rate-limited to prevent token probing.
262    pub introspect_per_min: u32,
263    /// Max oauth2/revoke requests per minute per IP (default: 10).
264    /// SEC-020: revoke endpoint rate-limited to prevent DoS via token flooding.
265    pub revoke_per_min: u32,
266    /// Max authz-check requests per minute per IP (default: 300).
267    /// Authz checks are read-only and high-frequency — used by UI permission gating.
268    /// Kept in a dedicated bucket so heavy UI use does not consume the login/token limit (D-07).
269    pub authz_check_per_min: u32,
270    /// Rate-limit bucket-key derivation mode (D8, default: `Ip` — current
271    /// behavior, unchanged). See [`RateLimitKeyMode`] for the full
272    /// rationale and scope (only `/oauth2/token`, `/oauth2/revoke`,
273    /// `/oauth2/introspect` honor this; `/auth/login` and every other
274    /// endpoint always stay per-IP).
275    pub key: RateLimitKeyMode,
276    /// Deployment posture preset (G7, default:
277    /// [`RateLimitProfile::Internet`] — presets nothing, so the shipped
278    /// defaults above apply verbatim). Configure via
279    /// `AXIAM__RATE_LIMIT__PROFILE`. Applied by
280    /// [`RateLimitConfig::apply_profile_from_env`] at startup; see
281    /// [`RateLimitProfile`] for the values and the security caveat.
282    pub profile: RateLimitProfile,
283}

302impl RateLimitConfig {
303    /// Applies [`RateLimitConfig::profile`]'s preset to every machine-traffic
304    /// limit the operator did **not** pin with an explicit
305    /// `AXIAM__RATE_LIMIT__*` env var, reading the process environment.
306    ///
307    /// Call once at startup, before [`RateLimitConfig::validate`] and before
308    /// the config is cloned into the App factory.
309    ///
310    /// **Precedence:** explicit env var > profile preset > shipped default.
311    /// **Known limitation:** "explicit" means *an env var is present*. A
312    /// value coming from `config/default.toml` (the optional, non-shipped
313    /// file source in `axiam-server::load_config`) is indistinguishable from
314    /// "unset" here and will be overwritten by a non-default profile. Set the
315    /// env var to pin it. This is documented in
316    /// `docs/deployment/rate-limit-sizing.md`.
317    pub fn apply_profile_from_env(&mut self) -> RateLimitPosture {
318        self.apply_profile(|name| std::env::var_os(name).is_some())
319    }
320
321    /// [`RateLimitConfig::apply_profile_from_env`] with the environment
322    /// lookup injected — `is_set(name)` answers "did the operator pin this
323    /// env var?". Pure and unit-testable; the env-reading wrapper is the
324    /// only impure part.
325    pub fn apply_profile<F>(&mut self, is_set: F) -> RateLimitPosture
326    where
327        F: Fn(&str) -> bool,
328    {
329        let Some(preset) = self.profile.preset() else {
330            // `internet`: nothing is preset — this is a no-op by
331            // construction, so the shipped posture cannot drift.
332            return RateLimitPosture {
333                profile: self.profile,
334                preset_applied: false,
335                operator_overrides: Vec::new(),
336                grpc_authz_per_sec_preset: None,
337            };
338        };
339
340        let mut overrides: Vec<&'static str> = Vec::new();
341        if is_set(ENV_KEY) {
342            overrides.push(ENV_KEY);
343        } else {
344            self.key = preset.key;
345        }
346        for (env, field, value) in [
347            (
348                ENV_TOKEN_PER_MIN,
349                &mut self.token_per_min,
350                preset.token_per_min,
351            ),
352            (
353                ENV_INTROSPECT_PER_MIN,
354                &mut self.introspect_per_min,
355                preset.introspect_per_min,
356            ),
357            (
358                ENV_REVOKE_PER_MIN,
359                &mut self.revoke_per_min,
360                preset.revoke_per_min,
361            ),
362            (
363                ENV_AUTHZ_CHECK_PER_MIN,
364                &mut self.authz_check_per_min,
365                preset.authz_check_per_min,
366            ),
367        ] {
368            if is_set(env) {
369                overrides.push(env);
370            } else {
371                *field = value;
372            }
373        }
374
375        // NOTE: login/register/password_reset/mfa are intentionally absent.
376        // No preset may touch a human endpoint — see `RateLimitProfile`.
377
378        RateLimitPosture {
379            profile: self.profile,
380            preset_applied: true,
381            operator_overrides: overrides,
382            grpc_authz_per_sec_preset: Some(preset.grpc_authz_per_sec),
383        }
384    }
385
386    /// Validates all rate limits are >= 1 (governor panics on zero).
387    pub fn validate(&self) {
388        assert!(self.login_per_min >= 1, "login_per_min must be >= 1");
389        assert!(self.register_per_min >= 1, "register_per_min must be >= 1");
390        assert!(self.token_per_min >= 1, "token_per_min must be >= 1");
391        assert!(
392            self.password_reset_per_min >= 1,
393            "password_reset_per_min must be >= 1"
394        );
395        assert!(self.mfa_per_min >= 1, "mfa_per_min must be >= 1");
396        assert!(
397            self.introspect_per_min >= 1,
398            "introspect_per_min must be >= 1"
399        );
400        assert!(self.revoke_per_min >= 1, "revoke_per_min must be >= 1");
401        assert!(
402            self.authz_check_per_min >= 1,
403            "authz_check_per_min must be >= 1"
404        );
405    }
406}

414    /// `docs/deployment/rate-limit-sizing.md` — the operator-facing sizing
415    /// page. Its posture table is the ONLY place the shipped values are
416    /// written down for humans, and these tests are what keep it honest.
417    const POSTURE_DOC: &str = "docs/deployment/rate-limit-sizing.md";
418    /// The published benchmark analysis carries the same "shipped default"
419    /// column in its §6 recommended-settings table. Checked opportunistically
420    /// (the file is owned by the benchmark harness, not this crate).
421    const PUBLIC_BENCH_DOC: &str = "benchmarks/PUBLIC_BENCH_ANALYSIS.md";
422
423    const TABLE_BEGIN: &str = "<!-- rate-limit-posture-table:begin -->";
424    const TABLE_END: &str = "<!-- rate-limit-posture-table:end -->";
425
426    /// `crates/axiam-api-rest` → repository root.
427    fn repo_root() -> PathBuf {
428        PathBuf::from(env!("CARGO_MANIFEST_DIR"))
429            .join("..")
430            .join("..")
431    }
432
433    fn cells(line: &str) -> Vec<String> {
434        line.trim()
435            .trim_matches('|')
436            .split('|')
437            .map(|c| c.trim().trim_matches('`').trim().to_owned())
438            .collect()
439    }
440
441    /// Parses the marker-delimited posture table into
442    /// `env var -> [internet, gateway, mesh]`.
443    fn posture_table() -> HashMap<String, Vec<String>> {
444        let path = repo_root().join(POSTURE_DOC);
445        let md = std::fs::read_to_string(&path)
446            .unwrap_or_else(|e| panic!("{} must be readable: {e}", path.display()));
447        let body = md
448            .split(TABLE_BEGIN)
449            .nth(1)
450            .unwrap_or_else(|| panic!("{POSTURE_DOC} must contain {TABLE_BEGIN}"))
451            .split(TABLE_END)
452            .next()
453            .unwrap_or_else(|| panic!("{POSTURE_DOC} must contain {TABLE_END}"));
454
455        let mut table = HashMap::new();
456        for line in body.lines() {
457            let line = line.trim();
458            if !line.starts_with("| `AXIAM__") {
459                continue;
460            }
461            let cells = cells(line);
462            assert_eq!(
463                cells.len(),
464                4,
465                "{POSTURE_DOC}: posture row must have env + 3 profile columns: {line}"
466            );
467            table.insert(cells[0].clone(), cells[1..].to_vec());
468        }
469        assert!(
470            !table.is_empty(),
471            "{POSTURE_DOC}: posture table parsed to zero rows — did the format change?"
472        );
473        table
474    }
475
476    /// Column index: 0 = `internet`, 1 = `gateway`, 2 = `mesh`.
477    fn documented(table: &HashMap<String, Vec<String>>, env: &str, column: usize) -> String {
478        table
479            .get(env)
480            .unwrap_or_else(|| panic!("{POSTURE_DOC} must document {env}"))[column]
481            .clone()
482    }
483
484    fn documented_u32(table: &HashMap<String, Vec<String>>, env: &str, column: usize) -> u32 {
485        let raw = documented(table, env, column);
486        raw.parse()
487            .unwrap_or_else(|e| panic!("{POSTURE_DOC}: {env} column {column} ({raw:?}): {e}"))
488    }
489
490    /// **Single source of truth guard (G7).** Every shipped default written
491    /// down in `docs/deployment/rate-limit-sizing.md` must equal
492    /// `RateLimitConfig::default()`. Change one, this test makes you change
493    /// the other.
494    #[test]
495    fn documented_defaults_match_shipped_config() {
496        let table = posture_table();
497        let d = RateLimitConfig::default();
498
499        assert_eq!(documented(&table, ENV_KEY, 0), d.key.as_str());
500        for (env, actual) in [
501            ("AXIAM__RATE_LIMIT__LOGIN_PER_MIN", d.login_per_min),
502            ("AXIAM__RATE_LIMIT__REGISTER_PER_MIN", d.register_per_min),
503            (
504                "AXIAM__RATE_LIMIT__PASSWORD_RESET_PER_MIN",
505                d.password_reset_per_min,
506            ),
507            ("AXIAM__RATE_LIMIT__MFA_PER_MIN", d.mfa_per_min),
508            (ENV_TOKEN_PER_MIN, d.token_per_min),
509            (ENV_INTROSPECT_PER_MIN, d.introspect_per_min),
510            (ENV_REVOKE_PER_MIN, d.revoke_per_min),
511            (ENV_AUTHZ_CHECK_PER_MIN, d.authz_check_per_min),
512        ] {
513            assert_eq!(
514                documented_u32(&table, env, 0),
515                actual,
516                "{POSTURE_DOC} documents a different shipped default for {env}"
517            );
518        }
519    }

521    /// The documented `gateway`/`mesh` columns must equal what
522    /// [`RateLimitConfig::apply_profile`] actually produces — including the
523    /// gRPC ceiling handed to `axiam-api-grpc` by the composition root.
524    #[test]
525    fn documented_presets_match_applied_profiles() {
526        let table = posture_table();
527        let shipped = RateLimitConfig::default();
528
529        for (column, profile) in [(1, RateLimitProfile::Gateway), (2, RateLimitProfile::Mesh)] {
530            let mut cfg = RateLimitConfig {
531                profile,
532                ..RateLimitConfig::default()
533            };
534            // Nothing pinned by the operator: the preset applies in full.
535            let posture = cfg.apply_profile(|_| false);
536            assert!(posture.preset_applied, "{profile:?}");
537            assert!(posture.operator_overrides.is_empty(), "{profile:?}");
538
539            assert_eq!(

584    /// The published benchmark analysis (§6 "Recommended settings by
585    /// deployment") repeats the shipped defaults in its own table; run 3's
586    /// publishing guidance requires those to stay in sync with code. The file
587    /// belongs to the benchmark harness, so a missing file is tolerated — a
588    /// *wrong* value is not.
589    #[test]
590    fn public_benchmark_doc_shipped_defaults_match_code() {
591        let path = repo_root().join(PUBLIC_BENCH_DOC);
592        let Ok(md) = std::fs::read_to_string(&path) else {
593            eprintln!("{PUBLIC_BENCH_DOC} not present — skipping cross-doc drift check");
594            return;
595        };
596        let d = RateLimitConfig::default();
597        let expected: HashMap<&str, String> = HashMap::from([
598            (ENV_KEY, d.key.as_str().to_owned()),
599            (ENV_TOKEN_PER_MIN, d.token_per_min.to_string()),
600            (ENV_INTROSPECT_PER_MIN, d.introspect_per_min.to_string()),
601            (ENV_AUTHZ_CHECK_PER_MIN, d.authz_check_per_min.to_string()),
602            (
603                "AXIAM__RATE_LIMIT__LOGIN_PER_MIN",
604                d.login_per_min.to_string(),
605            ),
606        ]);
607
608        let mut checked = 0usize;
609        for line in md.lines() {
610            let line = line.trim();
611            if !line.starts_with("| `AXIAM__RATE_LIMIT__") {
612                continue;
613            }
614            let cells = cells(line);
615            let Some(want) = expected.get(cells[0].as_str()) else {
616                continue;
617            };
618            // The §6 "Shipped default" column is prose-decorated
619            // (e.g. "10 (per IP)"); compare on the leading token only.
620            let got = cells
621                .get(1)
622                .map(|c| {
623                    c.split_whitespace()
624                        .next()
625                        .unwrap_or("")
626                        .trim_matches('`')
627                        .to_owned()
628                })
629                .unwrap_or_default();
630            assert_eq!(
631                &got, want,
632                "{PUBLIC_BENCH_DOC} §6 shipped default for {} disagrees with \
633                 RateLimitConfig::default()",
634                cells[0]
635            );
636            checked += 1;
637        }
638        assert!(
639            checked >= expected.len(),
640            "{PUBLIC_BENCH_DOC}: only matched {checked} of {} rate-limit rows — \
641             §6 table format changed?",
642            expected.len()
643        );
644    }

667    /// Precedence: an explicitly-set env var beats the preset, and is named
668    /// in the posture (so the startup log tells the operator why a value
669    /// isn't what the profile advertises).
670    #[test]
671    fn explicit_env_vars_win_over_the_preset() {
672        let mut cfg = RateLimitConfig {
673            profile: RateLimitProfile::Gateway,
674            token_per_min: 2_000,
675            key: RateLimitKeyMode::IpClientId,
676            ..RateLimitConfig::default()
677        };
678        let posture = cfg.apply_profile(|name| name == ENV_TOKEN_PER_MIN || name == ENV_KEY);
679
680        // Pinned by the operator — untouched.
681        assert_eq!(cfg.token_per_min, 2_000);
682        assert_eq!(cfg.key, RateLimitKeyMode::IpClientId);
683        // Not pinned — preset applies.
684        assert_eq!(cfg.introspect_per_min, 6_000);
685        assert_eq!(cfg.revoke_per_min, 600);
686        assert_eq!(cfg.authz_check_per_min, 6_000);
687
688        assert!(posture.preset_applied);
689        assert_eq!(posture.operator_overrides, vec![ENV_KEY, ENV_TOKEN_PER_MIN]);
690        assert_eq!(
691            posture.overrides_display(),
692            format!("{ENV_KEY},{ENV_TOKEN_PER_MIN}")
693        );
694    }
