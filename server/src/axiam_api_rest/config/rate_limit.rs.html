5/// Rate-limit bucket-key derivation mode (D8).
6///
7/// Environment variable: `AXIAM__RATE_LIMIT__KEY` = `ip` | `client_id` |
8/// `ip_client_id` (default `ip`).
9///
10/// **Why this exists (NAT'd-fleet lesson):** the bucket key was
11/// unconditionally `"{endpoint}:{ip}"` (see
12/// `middleware::rate_limit_shared::RateLimitShared` and
13/// `extractors::rate_limit::XForwardedForKeyExtractor`). Behind a NAT/proxy
14/// fleet (many distinct OAuth2 clients — e.g. IoT devices or microservices —
15/// egressing through one shared IP), every client sharing that IP collided
16/// into a SINGLE bucket, so one noisy/misbehaving client could exhaust the
17/// `/oauth2/token`, `/oauth2/revoke`, or `/oauth2/introspect` quota for every
18/// other client behind the same NAT gateway. `client_id` and `ip_client_id`
19/// give each OAuth2 client its own bucket (optionally still scoped per-IP)
20/// on the endpoints where a client identity is actually known.
21///
22/// **Scope — where this setting applies:** ONLY the three endpoints where an
23/// OAuth2 client authenticates itself via a form-encoded `client_id`
24/// (`client_secret_post`, RFC 6749 §2.3.1): `/oauth2/token`,
25/// `/oauth2/revoke`, `/oauth2/introspect` (see `handlers::oauth2` and
26/// `server.rs`'s wiring of `RateLimitShared::new_client_identity_aware` /
27/// the client-aware governor for exactly those three resources).
28///
29/// **`/auth/login` (and every other rate-limited endpoint) ALWAYS keys
30/// per-IP, regardless of this setting.** Login authenticates a *user* via
31/// username/password — there is no OAuth2 client identity anywhere in that
32/// request, so there is nothing meaningful to key on besides the source IP.
33/// This is intentional and NOT a bug: switching this config value never
34/// changes login's (or MFA's, or password-reset's, etc.) rate-limit
35/// behavior. See `server.rs` — those resources are wired with the plain
36/// `build_governor`/`RateLimitShared::new` constructors, which never read
37/// this field.
38#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize, Default)]
39#[serde(rename_all = "snake_case")]
40pub enum RateLimitKeyMode {
41    /// Key on source IP only (current/default behavior, unchanged).
42    #[default]
43    Ip,
44    /// Key on the OAuth2 `client_id` alone — independent buckets per client,
45    /// regardless of which IP(s) it connects from.
46    ClientId,
47    /// Key on the `(ip, client_id)` pair — independent buckets per client
48    /// AND per IP, so a compromised/leaked client credential rate-limited
49    /// from one IP doesn't automatically throttle the same client_id
50    /// operating legitimately from a different IP.
51    IpClientId,
52}
53
54impl RateLimitKeyMode {
55    /// Stable, log-safe name — identical to the `AXIAM__RATE_LIMIT__KEY`
56    /// value that selects this mode.
57    pub const fn as_str(self) -> &'static str {
58        match self {
59            Self::Ip => "ip",
60            Self::ClientId => "client_id",
61            Self::IpClientId => "ip_client_id",
62        }
63    }
64
65    /// How much of this mode's bucket key an **unauthenticated** caller
66    /// chooses (security-analysis-2026-08-02 §4 item 1).
67    pub const fn mintability(self) -> KeyMintability {
68        match self {
69            // Source IP, after `trusted_hops` normalisation. Not caller-chosen.
70            Self::Ip => KeyMintability::None,
71            // The whole key is the `client_id` read from the raw form body
72            // before any credential is verified.
73            Self::ClientId => KeyMintability::Full,
74            // Half the key (`client_id`) is caller-chosen; the IP half is not.
75            Self::IpClientId => KeyMintability::Partial,
76        }
77    }
78}

80/// How much of a rate-limit bucket key an unauthenticated caller controls.
81///
82/// The bucket key for `/oauth2/{token,introspect,revoke}` is derived
83/// **before** the credential check (`middleware/rate_limit_shared.rs`,
84/// `extractors/rate_limit.rs`) — RFC 6749 §2.3.1 puts `client_id` in the
85/// request body, so it cannot be authenticated before it is read. Whatever
86/// part of the key comes from that body is therefore attacker-chosen, and a
87/// caller rotating it mints a fresh bucket per value.
88#[derive(Debug, Clone, Copy, PartialEq, Eq)]
89pub enum KeyMintability {
90    /// No part of the key is caller-chosen (the shipped default, `ip`).
91    None,
92    /// Part of the key is caller-chosen, part is not (`ip_client_id`).
93    Partial,
94    /// The entire key is caller-chosen (`client_id`).
95    Full,
96}

98/// Deployment rate-limit **posture preset** (G7).
99///
100/// Environment variable: `AXIAM__RATE_LIMIT__PROFILE` = `internet` |
101/// `gateway` | `mesh` (default `internet`).
102///
103/// **Why a preset instead of new defaults.** Benchmark run 3 measured that
104/// the then-shipped defaults (`token 20/min`, `introspect 10/min`,
105/// `revoke 10/min`, `authz_check 300/min`, all keyed per-IP) flatten a
106/// 50-VU single-source-IP load to ~0–14 req/s at ~100% `429`, while the
107/// same 2-core server sustains ~1 800 token issuances/s and ~740–2 300
108/// authz checks/s. A *strict per-IP* posture is still correct for a small
109/// internet-facing deployment and still wrong for an M2M/NAT'd fleet where
110/// many OAuth2 clients share one egress IP — which is why the M2M sizing
111/// stays an **opt-in preset**: one env var moves the whole machine-traffic
112/// family coherently (key mode + token/introspect/revoke/authz + the gRPC
113/// authz ceiling).
114///
115/// Run 4 separately revised the shipped `internet` machine-endpoint numbers
116/// themselves (I3 — token 20→120, introspect 10→600, authz 300→1 800,
117/// revoke 10→60; see [`RateLimitConfig::default`]). That changed the
118/// starting point, not the argument above: the `internet` posture is still
119/// per-IP and still 2–3 orders of magnitude below measured capacity, and it
120/// still collapses a NAT'd fleet into one bucket. The presets remain the
121/// answer for machine fleets.
122///
123/// **Human endpoints are never touched by any preset.** `login_per_min`,
124/// `register_per_min`, `password_reset_per_min` and `mfa_per_min` stay at
125/// their strict per-IP defaults in every profile — they gate
126/// password/OTP guessing and account-enumeration, which are human-scale
127/// attacks whose economics do not change because the deployment is a
128/// service mesh. Raise them deliberately, one env var at a time.
129///
130/// **Security caveat an operator MUST read before selecting a preset**
131/// (also in `docs/deployment/rate-limit-sizing.md`): both non-default
132/// presets set the key mode to `client_id`, and the `client_id` is parsed
133/// from the *unauthenticated* request body (`client_secret_post`,
134/// RFC 6749 §2.3.1) before any credential is verified. An attacker can
135/// therefore rotate `client_id` values to mint fresh buckets. Under those
136/// modes the token/introspect/revoke limits are a **fairness / noisy-
137/// neighbour control between cooperating clients**, not an anti-abuse
138/// control against a determined attacker; the anti-abuse job moves to the
139/// edge (WAF / API gateway / mTLS). This is a property of the D8 key
140/// modes themselves, not of the presets — the presets just make it the
141/// active posture, which is exactly why they are opt-in.
142#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize, Default)]
143#[serde(rename_all = "snake_case")]
144pub enum RateLimitProfile {
145    /// Shipped default: small internet-facing deployment, strict per-IP
146    /// limits on every endpoint. No field is preset — the defaults in
147    /// [`RateLimitConfig::default`] apply verbatim.
148    #[default]
149    Internet,
150    /// M2M fleet reaching an internet-facing AXIAM through a shared
151    /// NAT/egress gateway (the topology run 3 showed the defaults break).
152    /// Per-client buckets and machine-scale ceilings on the machine
153    /// endpoints; human endpoints unchanged.
154    Gateway,
155    /// AXIAM reachable only on a private network / service mesh (no
156    /// internet exposure on the machine endpoints). Ceilings sized as
157    /// runaway-loop guards rather than abuse controls; human endpoints
158    /// still unchanged, because "private network" is not "no humans".
159    Mesh,
160}
161
162/// The machine-traffic limits a non-default [`RateLimitProfile`] presets.
163///
164/// Human endpoints are deliberately absent from this struct — a preset
165/// cannot express them, so it cannot weaken them.
166#[derive(Debug, Clone, Copy, PartialEq, Eq)]
167pub struct MachineLimitPreset {
168    /// Bucket-key mode for `/oauth2/token`, `/oauth2/revoke`,
169    /// `/oauth2/introspect` (the only endpoints that honor it).
170    pub key: RateLimitKeyMode,
171    /// `/oauth2/token` per minute per bucket.
172    pub token_per_min: u32,
173    /// `/oauth2/introspect` per minute per bucket.
174    pub introspect_per_min: u32,
175    /// `/oauth2/revoke` per minute per bucket.
176    pub revoke_per_min: u32,
177    /// REST authz-check endpoints per minute per IP.
178    pub authz_check_per_min: u32,
179    /// gRPC `AuthorizationService` per second per IP — applied to
180    /// `axiam_api_grpc::GrpcConfig::grpc_authz_per_sec` by the composition
181    /// root (`axiam-server::main`). Kept here so the whole family has one
182    /// source of truth.
183    pub grpc_authz_per_sec: u32,
184}

205impl RateLimitProfile {
206    /// Stable, log-safe name — identical to the
207    /// `AXIAM__RATE_LIMIT__PROFILE` value that selects this profile.
208    pub const fn as_str(self) -> &'static str {
209        match self {
210            Self::Internet => "internet",
211            Self::Gateway => "gateway",
212            Self::Mesh => "mesh",
213        }
214    }
215
216    /// The machine-traffic limits this profile presets, or `None` for
217    /// [`RateLimitProfile::Internet`] (which presets nothing at all).
218    ///
219    /// Every number below is anchored to the run-3 measured envelope on
220    /// 2-core server / 2-core DB containers (`benchmarks/PUBLIC_BENCH_ANALYSIS.md`
221    /// §6, laptop-class and explicitly temporary): token issuance
222    /// ~1 800/s (~108 000/min), introspection ~2 200/s (~132 000/min),
223    /// authz checks ~740/s cache-off and ~2 300/s cache-on
224    /// (~44 400 / ~138 000 per minute), gRPC single authz check ~603/s.
225    /// The presets are *per bucket*, so they are deliberately a small
226    /// fraction of the whole-server ceiling.
227    pub const fn preset(self) -> Option<MachineLimitPreset> {
228        match self {
229            Self::Internet => None,
230            // 600/min = 10/s per client ≈ 0.55% of the measured 1 800/s
231            // server ceiling: ~180 distinct clients running flat out are
232            // needed to saturate, so one compromised credential cannot
233            // become the DoS engine. Introspect is 10× token per the
234            // public §6 rule (resource servers introspect per request).
235            // Authz 6 000/min = 100/s per client, the low end of §6's
236            // 6 000–60 000 band.
237            Self::Gateway => Some(MachineLimitPreset {
238                key: RateLimitKeyMode::ClientId,
239                token_per_min: 600,
240                introspect_per_min: 6_000,
241                revoke_per_min: 600,
242                authz_check_per_min: 6_000,
243                grpc_authz_per_sec: 1_000,
244            }),
245            // Private-network sizing: 6 000/min token = 100/s per client
246            // (~5.6% of the measured ceiling) and 60 000/min authz =
247            // 1 000/s per client, ABOVE the measured 740/s cache-off
248            // ceiling — i.e. on this hardware the authz limit stops a
249            // runaway retry loop, not an attacker. Documented as such.
250            Self::Mesh => Some(MachineLimitPreset {
251                key: RateLimitKeyMode::ClientId,
252                token_per_min: 6_000,
253                introspect_per_min: 60_000,
254                revoke_per_min: 6_000,
255                authz_check_per_min: 60_000,
256                grpc_authz_per_sec: 5_000,
257            }),
258        }
259    }
260}

262/// What the composition root actually resolved — the input to the startup
263/// posture log line.
264#[derive(Debug, Clone, PartialEq, Eq)]
265pub struct RateLimitPosture {
266    /// The selected profile.
267    pub profile: RateLimitProfile,
268    /// `true` when a non-default preset was applied to at least the fields
269    /// the operator did not pin explicitly.
270    pub preset_applied: bool,
271    /// Env vars the operator set explicitly, which therefore beat the
272    /// preset. Names only — never values (some are not secret, but the
273    /// log line stays uniformly value-free for this list).
274    pub operator_overrides: Vec<&'static str>,
275    /// gRPC authz ceiling the preset wants, for the composition root to
276    /// hand to `axiam_api_grpc::GrpcConfig`. `None` under `internet`.
277    pub grpc_authz_per_sec_preset: Option<u32>,
278}

280impl RateLimitPosture {
281    /// Comma-joined override list for the startup log (`"none"` when empty).
282    pub fn overrides_display(&self) -> String {
283        if self.operator_overrides.is_empty() {
284            "none".to_owned()
285        } else {
286            self.operator_overrides.join(",")
287        }
288    }

291/// Rate limit configuration.
292/// Environment variables: AXIAM__RATE_LIMIT__LOGIN_PER_MIN, etc.
293#[derive(Debug, Clone, Deserialize)]
294#[serde(default)]
295pub struct RateLimitConfig {
296    /// Max login requests per minute per IP (default: 10).
297    pub login_per_min: u32,
298    /// Max register requests per minute per IP (default: 5).
299    pub register_per_min: u32,
300    /// Max oauth2/token requests per minute per client (default: 120 — I3).
301    pub token_per_min: u32,
302    /// Max password-reset requests per minute per IP (default: 3).
303    pub password_reset_per_min: u32,
304    /// Max MFA requests per minute per IP (default: 5).
305    /// Covers /auth/mfa/enroll, /confirm, /verify, /setup/enroll, /setup/confirm (SEC-020).
306    pub mfa_per_min: u32,
307    /// Max oauth2/introspect requests per minute per IP (default: 600 — I3).
308    /// SEC-020: introspect endpoint rate-limited to prevent token probing.
309    pub introspect_per_min: u32,
310    /// Max oauth2/revoke requests per minute per IP (default: 60 — I3).
311    /// SEC-020: revoke endpoint rate-limited to prevent DoS via token flooding.
312    pub revoke_per_min: u32,
313    /// Max authz-check requests per minute per IP (default: 1800 — I3).
314    /// Authz checks are read-only and high-frequency — used by UI permission gating.
315    /// Kept in a dedicated bucket so heavy UI use does not consume the login/token limit (D-07).
316    pub authz_check_per_min: u32,
317    /// Rate-limit bucket-key derivation mode (D8, default: `Ip` — current
318    /// behavior, unchanged). See [`RateLimitKeyMode`] for the full
319    /// rationale and scope (only `/oauth2/token`, `/oauth2/revoke`,
320    /// `/oauth2/introspect` honor this; `/auth/login` and every other
321    /// endpoint always stay per-IP).
322    pub key: RateLimitKeyMode,
323    /// Deployment posture preset (G7, default:
324    /// [`RateLimitProfile::Internet`] — presets nothing, so the shipped
325    /// defaults above apply verbatim). Configure via
326    /// `AXIAM__RATE_LIMIT__PROFILE`. Applied by
327    /// [`RateLimitConfig::apply_profile_from_env`] at startup; see
328    /// [`RateLimitProfile`] for the values and the security caveat.
329    pub profile: RateLimitProfile,
330}

332impl Default for RateLimitConfig {
333    fn default() -> Self {
334        Self {
335            // --- Human endpoints: NEVER sized from capacity (I3) ----------
336            // These gate password/OTP guessing and account enumeration.
337            // Benchmark capacity is irrelevant to their sizing and they are
338            // deliberately untouched by the I3 revision below.
339            login_per_min: 10,
340            register_per_min: 5,
341            password_reset_per_min: 3,
342            mfa_per_min: 5,
343            // --- Machine endpoints: I3 revision (run-4 sizing) ------------
344            // Run 4 measured the `internet` machine-endpoint defaults against
345            // the actual capacity behind them on a 2-core envelope
346            // (`benchmarks/PUBLIC_BENCH_ANALYSIS.md` §7): token 20/min vs
347            // ~163 000/min of capacity, introspect 10/min vs ~263 000/min,
348            // authz 300/min vs ~45 000/min. Those numbers did not protect
349            // anything the raised ones fail to protect — they broke the first
350            // healthy integration behind a NAT while sitting 4–5 orders of
351            // magnitude below the machine's ceiling. Each revised value still
352            // stays well below measured capacity — 25x at the tightest
353            // (authz_check, deliberately so: it is the endpoint a real service
354            // calls per request), 400-2 700x on the rest — so the abuse
355            // posture is intact. See `revised_defaults_keep_capacity_margin`
356            // for the pinned arithmetic. (Do NOT restate this as ">=500x":
357            // the run-4 planning docs made that claim and it is false for
358            // authz_check and introspect.)
359            //
360            // Previous shipped values, for the record: token 20,
361            // introspect 10, revoke 10, authz_check 300.
362            token_per_min: 120,
363            introspect_per_min: 600,
364            revoke_per_min: 60,
365            authz_check_per_min: 1_800,
366            key: RateLimitKeyMode::Ip,
367            profile: RateLimitProfile::Internet,
368        }
369    }
370}

372impl RateLimitConfig {
373    /// Applies [`RateLimitConfig::profile`]'s preset to every machine-traffic
374    /// limit the operator did **not** pin with an explicit
375    /// `AXIAM__RATE_LIMIT__*` env var, reading the process environment.
376    ///
377    /// Call once at startup, before [`RateLimitConfig::validate`] and before
378    /// the config is cloned into the App factory.
379    ///
380    /// **Precedence:** explicit env var > profile preset > shipped default.
381    /// **Known limitation:** "explicit" means *an env var is present*. A
382    /// value coming from `config/default.toml` (the optional, non-shipped
383    /// file source in `axiam-server::load_config`) is indistinguishable from
384    /// "unset" here and will be overwritten by a non-default profile. Set the
385    /// env var to pin it. This is documented in
386    /// `docs/deployment/rate-limit-sizing.md`.
387    pub fn apply_profile_from_env(&mut self) -> RateLimitPosture {
388        self.apply_profile(|name| std::env::var_os(name).is_some())
389    }
390
391    /// [`RateLimitConfig::apply_profile_from_env`] with the environment
392    /// lookup injected — `is_set(name)` answers "did the operator pin this
393    /// env var?". Pure and unit-testable; the env-reading wrapper is the
394    /// only impure part.
395    pub fn apply_profile<F>(&mut self, is_set: F) -> RateLimitPosture
396    where
397        F: Fn(&str) -> bool,
398    {
399        let Some(preset) = self.profile.preset() else {
400            // `internet`: nothing is preset — this is a no-op by
401            // construction, so the shipped posture cannot drift.
402            return RateLimitPosture {
403                profile: self.profile,
404                preset_applied: false,
405                operator_overrides: Vec::new(),
406                grpc_authz_per_sec_preset: None,
407            };
408        };
409
410        let mut overrides: Vec<&'static str> = Vec::new();
411        if is_set(ENV_KEY) {
412            overrides.push(ENV_KEY);
413        } else {
414            self.key = preset.key;
415        }
416        for (env, field, value) in [
417            (
418                ENV_TOKEN_PER_MIN,
419                &mut self.token_per_min,
420                preset.token_per_min,
421            ),
422            (
423                ENV_INTROSPECT_PER_MIN,
424                &mut self.introspect_per_min,
425                preset.introspect_per_min,
426            ),
427            (
428                ENV_REVOKE_PER_MIN,
429                &mut self.revoke_per_min,
430                preset.revoke_per_min,
431            ),
432            (
433                ENV_AUTHZ_CHECK_PER_MIN,
434                &mut self.authz_check_per_min,
435                preset.authz_check_per_min,
436            ),
437        ] {
438            if is_set(env) {
439                overrides.push(env);
440            } else {
441                *field = value;
442            }
443        }
444
445        // NOTE: login/register/password_reset/mfa are intentionally absent.
446        // No preset may touch a human endpoint — see `RateLimitProfile`.
447
448        RateLimitPosture {
449            profile: self.profile,
450            preset_applied: true,
451            operator_overrides: overrides,
452            grpc_authz_per_sec_preset: Some(preset.grpc_authz_per_sec),
453        }
454    }
455
456    /// Startup advisory for an **attacker-mintable** bucket-key mode
457    /// (security-analysis-2026-08-02 §4 item 1).
458    ///
459    /// Call once, at composition time, right after the "Rate-limit posture
460    /// active" line. Mirrors the two existing startup advisories: the I3
461    /// machine-traffic advisory (armed only when the shipped defaults are what
462    /// this process actually enforces) and the session-validation cache's
463    /// `warn!` (fires only for the opt-in, bounded-staleness mode). Like both,
464    /// this stays silent for the shipped default and speaks only when an
465    /// operator has selected the mode with the caveat.
466    ///
467    /// - **`client_id` → `warn!`.** The entire bucket key is read from the
468    ///   unauthenticated form body before the credential check, so rotating
469    ///   `client_id` values mints fresh buckets on
470    ///   `/oauth2/{token,introspect,revoke}`. The mode is intended only behind
471    ///   an edge (mTLS / API gateway / WAF) that already authenticates
472    ///   callers.
473    /// - **`ip_client_id` → `info!`, deliberately softer.** It is *partially*
474    ///   mintable: the same single-source evasion works, so this is not a
475    ///   "safe" mode. But the key still carries a component the caller cannot
476    ///   forge, which removes the one thing `client_id` mode uniquely allows —
477    ///   `client_id`s are not secret, so under `client_id` an attacker can
478    ///   exhaust a *known legitimate client's* bucket from anywhere and deny
479    ///   it service, while under `ip_client_id` that collateral is confined to
480    ///   the attacker's own source IP. Warning at the same level as
481    ///   `client_id` would flatten a real difference and train operators to
482    ///   ignore both; staying silent would hide a genuine caveat. `info!` is
483    ///   the honest middle.
484    /// - **`ip` (default) → silent.** Nothing about the key is caller-chosen.
485    pub fn warn_on_mintable_key(&self) {
486        match self.key.mintability() {
487            KeyMintability::None => {}
488            KeyMintability::Full => tracing::warn!(
489                key_mode = self.key.as_str(),
490                "rate-limit key mode `client_id` ACTIVE — the whole bucket key is the \
491                 client_id read from the unauthenticated OAuth2 form body BEFORE any \
492                 credential check, so a caller rotating client_id values mints fresh \
493                 buckets on /oauth2/{{token,introspect,revoke}}. Under this mode those \
494                 limits are a fairness control between cooperating clients, NOT an \
495                 anti-abuse control; the mode assumes an edge (mTLS / API gateway / WAF) \
496                 that already authenticates callers. See \
497                 docs/deployment/rate-limit-sizing.md section 5"
498            ),
499            KeyMintability::Partial => tracing::info!(
500                key_mode = self.key.as_str(),
501                "rate-limit key mode `ip_client_id` active — the client_id half of the \
502                 bucket key is caller-chosen before authentication, so a single source \

706    /// Canonical env-var name for a rate-limit row label found in the public
707    /// benchmark doc, or `None` when the row is not one of the knobs this
708    /// crate owns.
709    ///
710    /// §7 of the public doc keys its rows by **HTTP endpoint**
711    /// (`` `POST /oauth2/token` ``) while §7.2 keys them by **short knob
712    /// name** (`` `TOKEN_PER_MIN` ``); earlier drafts used the full
713    /// `AXIAM__RATE_LIMIT__…` env-var name. All three spellings are accepted
714    /// and normalized here so the check survives an editorial rewrite of the
715    /// tables without silently degrading to "matched nothing".
716    fn public_doc_row_knob(label: &str) -> Option<&'static str> {
717        let label = label.trim().trim_matches('`').trim();
718        let short = label
719            .strip_prefix("AXIAM__RATE_LIMIT__")
720            .unwrap_or(label)
721            .to_ascii_uppercase();
722        let by_knob = match short.as_str() {
723            "TOKEN_PER_MIN" => Some(ENV_TOKEN_PER_MIN),
724            "INTROSPECT_PER_MIN" => Some(ENV_INTROSPECT_PER_MIN),
725            "REVOKE_PER_MIN" => Some(ENV_REVOKE_PER_MIN),
726            "AUTHZ_CHECK_PER_MIN" => Some(ENV_AUTHZ_CHECK_PER_MIN),
727            "LOGIN_PER_MIN" => Some(ENV_LOGIN_PER_MIN),
728            _ => None,
729        };
730        by_knob.or_else(|| {
731            // §7's endpoint-keyed spelling.
732            match label.split_whitespace().next_back().unwrap_or(label) {
733                "/oauth2/token" => Some(ENV_TOKEN_PER_MIN),
734                "/oauth2/introspect" => Some(ENV_INTROSPECT_PER_MIN),
735                "/oauth2/revoke" => Some(ENV_REVOKE_PER_MIN),
736                "/api/v1/authz/check" => Some(ENV_AUTHZ_CHECK_PER_MIN),
737                "/api/v1/auth/login" => Some(ENV_LOGIN_PER_MIN),
738                _ => None,
739            }
740        })
741    }

743    /// `true` for a markdown table delimiter row (`|---|---:|`).
744    fn is_delimiter_row(line: &str) -> bool {
745        line.starts_with('|')
746            && line
747                .trim_matches('|')
748                .split('|')
749                .all(|c| !c.trim().is_empty() && c.trim().chars().all(|ch| matches!(ch, '-' | ':')))
750    }

752    /// Leading integer of a documented value cell, tolerating markdown
753    /// emphasis (`**120**`), a unit suffix (`120/min`, `100/s`), trailing
754    /// prose (`10 (per IP)`) and digit-group separators (`1 800`, `1 800`).
755    /// `None` when the cell does not start with a number at all.
756    fn documented_leading_number(cell: &str) -> Option<u32> {
757        let mut chars = cell.chars().peekable();
758        while matches!(chars.peek(), Some('*' | '`' | ' ')) {
759            chars.next();
760        }
761        let mut digits = String::new();
762        while let Some(&c) = chars.peek() {
763            if c.is_ascii_digit() {
764                digits.push(c);
765                chars.next();
766            } else if !digits.is_empty()
767                // A separator only counts as one when a digit follows it,
768                // so "10 / 5 / 3" stops at 10 instead of becoming 1053.
769                && matches!(c, ' ' | '\u{a0}' | '\u{202f}' | '_' | ',')
770                && chars.clone().nth(1).is_some_and(|n| n.is_ascii_digit())
771            {
772                chars.next();
773            } else {
774                break;
775            }
776        }
777        digits.parse().ok()
778    }

780    /// The published benchmark analysis (§7 "Production rate limits" and
781    /// §7.2 "the shipped defaults") repeats the shipped defaults in its own
782    /// tables; the publishing guidance requires those to stay in sync with
783    /// code. The file belongs to the benchmark harness, so a **missing file
784    /// is tolerated — a wrong value is not.**
785    ///
786    /// The scraper walks every markdown table in the doc, uses the table's
787    /// own header row to locate the "shipped default" column (§7 and §7.2
788    /// put it at different indices), and normalizes both the row label (§7
789    /// keys by endpoint, §7.2 by short knob name) and the value cell (which
790    /// is prose-decorated: `**120**`, `20/min`, `1 800`). Every knob in
791    /// `expected` must be found at least once, so an editorial rewrite that
792    /// drops or renames a row fails loudly instead of quietly checking
793    /// nothing — which is exactly how this test previously went blind.
794    #[test]
795    fn public_benchmark_doc_shipped_defaults_match_code() {
796        let path = repo_root().join(PUBLIC_BENCH_DOC);
797        let Ok(md) = std::fs::read_to_string(&path) else {
798            eprintln!("{PUBLIC_BENCH_DOC} not present — skipping cross-doc drift check");
799            return;
800        };
801        let d = RateLimitConfig::default();
802        let expected: HashMap<&str, u32> = HashMap::from([
803            (ENV_TOKEN_PER_MIN, d.token_per_min),
804            (ENV_INTROSPECT_PER_MIN, d.introspect_per_min),
805            (ENV_REVOKE_PER_MIN, d.revoke_per_min),
806            (ENV_AUTHZ_CHECK_PER_MIN, d.authz_check_per_min),
807            (ENV_LOGIN_PER_MIN, d.login_per_min),
808        ]);
809
810        let lines: Vec<&str> = md.lines().map(str::trim).collect();
811        // Index of the "shipped default" column in the table currently being
812        // walked; `None` outside a table, or in a table that has no such
813        // column (§7.1's posture table, §8's result matrix, …).
814        let mut shipped_default_col: Option<usize> = None;
815        let mut checked = 0usize;
816        let mut seen: HashMap<&str, u32> = HashMap::new();
817
818        for (i, line) in lines.iter().enumerate() {
819            if !line.starts_with('|') {
820                shipped_default_col = None;
821                continue;
822            }
823            if is_delimiter_row(line) {
824                continue;
825            }
826            // A row immediately followed by a delimiter row is this table's
827            // header — that is where the column index comes from.
828            if lines.get(i + 1).is_some_and(|next| is_delimiter_row(next)) {
829                shipped_default_col = cells(line)
830                    .iter()
831                    .position(|h| h.to_ascii_lowercase().contains("shipped default"));
832                continue;
833            }
834            let Some(column) = shipped_default_col else {
835                continue;
836            };
837            let cells = cells(line);
838            let Some(env) = cells.first().and_then(|l| public_doc_row_knob(l)) else {
839                continue;
840            };
841            let Some(want) = expected.get(env) else {
842                continue;
843            };
844            let got = cells
845                .get(column)
846                .and_then(|c| documented_leading_number(c))
847                .unwrap_or_else(|| {
848                    panic!(
849                        "{PUBLIC_BENCH_DOC}: row {:?} has no parseable number in the \
850                         'shipped default' column {column}",
851                        cells[0]
852                    )
853                });
854            assert_eq!(
855                got, *want,
856                "{PUBLIC_BENCH_DOC} §7 shipped default for {env} (row {:?}) disagrees \
857                 with RateLimitConfig::default()",
858                cells[0]
859            );
860            seen.insert(env, got);
861            checked += 1;
862        }
863
864        // Floor 1 — every knob must actually appear somewhere in §7/§7.2.
865        let mut missing: Vec<&str> = expected
866            .keys()
867            .filter(|env| !seen.contains_key(*env))
868            .copied()
869            .collect();
870        missing.sort_unstable();
871        assert!(
872            missing.is_empty(),
873            "{PUBLIC_BENCH_DOC}: no 'shipped default' row found for {missing:?} — \
874             the §7/§7.2 table shape changed and this drift check went blind. Fix the \
875             doc or teach `public_doc_row_knob`/`documented_leading_number` the new shape; \
876             do NOT relax this assertion."
877        );
878        // Floor 2 — the doc states these numbers in more than one table, so a
879        // silent collapse to a single surviving row is also a shape change.
880        assert!(
881            checked >= expected.len(),
882            "{PUBLIC_BENCH_DOC}: only matched {checked} rate-limit rows for {} knobs — \
883             §7/§7.2 table format changed?",
884            expected.len()
885        );
886    }

1086    /// In-memory `MakeWriter` so the tests below assert on what a real
1087    /// subscriber would print, not on a return value. Mirrors the capture
1088    /// helper in `tests/gdpr_audit_dlq_test.rs`.
1089    #[derive(Clone)]
1090    struct BufWriter(std::sync::Arc<std::sync::Mutex<Vec<u8>>>);
1091
1092    impl std::io::Write for BufWriter {
1093        fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
1094            self.0.lock().unwrap().extend_from_slice(buf);
1095            Ok(buf.len())
1096        }
1097        fn flush(&mut self) -> std::io::Result<()> {
1098            Ok(())
1099        }
1100    }
1101
1102    impl<'a> tracing_subscriber::fmt::MakeWriter<'a> for BufWriter {
1103        type Writer = BufWriter;
1104        fn make_writer(&'a self) -> Self::Writer {
1105            self.clone()
1106        }
1107    }
1108
1109    /// Runs `warn_on_mintable_key` under a capturing subscriber at TRACE and
1110    /// returns everything it emitted.
1111    fn captured_advisory(key: RateLimitKeyMode) -> String {
1112        let buf = std::sync::Arc::new(std::sync::Mutex::new(Vec::new()));
1113        let subscriber = tracing_subscriber::fmt()
1114            .with_writer(BufWriter(buf.clone()))
1115            .with_max_level(tracing::Level::TRACE)
1116            .with_ansi(false)
1117            .finish();
1118
1119        let cfg = RateLimitConfig {
1120            key,
1121            ..Default::default()
1122        };
1123        tracing::subscriber::with_default(subscriber, || cfg.warn_on_mintable_key());
1124
1125        let out = buf.lock().unwrap().clone();
1126        String::from_utf8(out).expect("tracing output is utf-8")
1127    }
