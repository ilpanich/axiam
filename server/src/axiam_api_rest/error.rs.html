1//! HTTP error response mapping for AXIAM API.
2//!
3//! Provides [`AxiamApiError`], a newtype around [`AxiamError`] that
4//! implements Actix-Web's [`ResponseError`] trait.
5
6use actix_web::HttpResponse;
7use actix_web::http::StatusCode;
8use axiam_core::error::AxiamError;
9use serde::Serialize;
10use tracing::error;

12/// Newtype wrapper so we can implement Actix-Web's `ResponseError`
13/// for the core `AxiamError` (orphan rule).
14#[derive(Debug)]
15pub struct AxiamApiError(pub AxiamError);

17impl std::fmt::Display for AxiamApiError {
18    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
19        write!(f, "{}", self.0)
20    }
21}

29#[derive(Serialize)]
30struct ErrorBody {
31    error: String,
32    message: String,
33    /// The action being checked (e.g. `"users:create"`), when known.
34    /// Populated only for `authorization_denied` (403) responses;
35    /// omitted from the JSON body otherwise (SDK-Q02).
36    #[serde(skip_serializing_if = "Option::is_none")]
37    action: Option<String>,
38    /// The resource id being checked, when known and not the "global"
39    /// nil-UUID sentinel. Populated only for `authorization_denied` (403)
40    /// responses; omitted from the JSON body otherwise (SDK-Q02).
41    #[serde(skip_serializing_if = "Option::is_none")]
42    resource_id: Option<String>,
43    /// A closed, machine-readable sub-code beside `error`, for the routes
44    /// that document one: today the network routes' `409`
45    /// (`sdks/CONTRACT.md` §36.4; `handlers::network::NetworkConflictReason`).
46    /// `message` stays for a human and is never the thing to branch on.
47    /// Omitted from the JSON body otherwise.
48    #[serde(skip_serializing_if = "Option::is_none")]
49    reason_code: Option<&'static str>,
50}
51
52impl AxiamApiError {
53    /// The response [`actix_web::ResponseError::error_response`] renders, with
54    /// `reason_code` set beside `error` when `Some`. The slug, the message and
55    /// the status are this type's, unchanged: a route that documents a
56    /// sub-code adds one member and changes nothing else.
57    #[must_use]
58    pub fn error_response_with_reason_code(
59        &self,
60        reason_code: Option<&'static str>,
61    ) -> HttpResponse {
62        self.render(reason_code)
63    }
64}
65
66/// A `422 Unprocessable Entity` in the crate's error shape (`{error, message}`).
67///
68/// [`AxiamError`] has no variant that maps to 422 for a well-formed request the
69/// server declines on a policy it can name (`Validation` is the `400` of a
70/// malformed one, and `PasswordPolicy` is specific to passwords), so a handler
71/// that needs the distinction builds the response here. `error` is the stable
72/// slug a client switches on; `message` is shown to the caller verbatim.
73pub fn unprocessable_entity(error: &str, message: String) -> HttpResponse {
74    HttpResponse::UnprocessableEntity().json(ErrorBody {
75        error: error.into(),
76        message,
77        action: None,
78        resource_id: None,
79        reason_code: None,
80    })
81}

83impl actix_web::ResponseError for AxiamApiError {
84    fn status_code(&self) -> StatusCode {
85        match &self.0 {
86            AxiamError::NotFound { .. } => StatusCode::NOT_FOUND,
87            AxiamError::AlreadyExists { .. } | AxiamError::Conflict { .. } => StatusCode::CONFLICT,
88            AxiamError::AuthenticationFailed { .. } | AxiamError::ReplayDetected => {
89                StatusCode::UNAUTHORIZED
90            }
91            AxiamError::AuthorizationDenied { .. }
92            | AxiamError::OpaqueRequired
93            | AxiamError::MfaEnforced => StatusCode::FORBIDDEN,
94            AxiamError::Validation { .. } | AxiamError::TenantContext => StatusCode::BAD_REQUEST,
95            AxiamError::PasswordPolicy { .. } => StatusCode::UNPROCESSABLE_ENTITY,
96            AxiamError::RateLimited => StatusCode::TOO_MANY_REQUESTS,
97            AxiamError::EmailConfig(_) => StatusCode::BAD_REQUEST,
98            AxiamError::ServiceUnavailable(_) | AxiamError::WriteContention => {
99                StatusCode::SERVICE_UNAVAILABLE
100            }
101            AxiamError::Database(_)
102            | AxiamError::Certificate(_)
103            | AxiamError::Crypto(_)
104            | AxiamError::EmailDelivery(_)
105            | AxiamError::WebhookDelivery(_)
106            | AxiamError::Internal(_) => StatusCode::INTERNAL_SERVER_ERROR,
107            // Note: ReplayDetected is handled above in the UNAUTHORIZED arm.
108        }
109    }
110
111    fn error_response(&self) -> HttpResponse {
112        self.render(None)
113    }
114}

116impl AxiamApiError {
117    fn render(&self, reason_code: Option<&'static str>) -> HttpResponse {
118        use actix_web::ResponseError;
119        // Client-facing error code slug.
120        let error_code = match &self.0 {
121            AxiamError::NotFound { .. } => "not_found",
122            AxiamError::AlreadyExists { .. } => "already_exists",
123            AxiamError::Conflict { .. } => "conflict",
124            AxiamError::AuthenticationFailed { .. } | AxiamError::ReplayDetected => {
125                "authentication_failed"
126            }
127            AxiamError::AuthorizationDenied { .. } => "authorization_denied",
128            // Distinct from `authentication_failed` on purpose: an SDK reads
129            // this to switch to `/auth/opaque/*` rather than to tell the user
130            // their password was wrong.
131            AxiamError::OpaqueRequired => "opaque_required",
132            // Distinct from `authorization_denied` on purpose, and for the
133            // same reason `opaque_required` is: the caller holds every
134            // permission the action needs — it is their own account — and the
135            // refusal is a tenant policy only an administrator can act
136            // against. A client switches on this to name the administrator
137            // rather than to say "forbidden" (T-267).
138            AxiamError::MfaEnforced => "mfa_enforced",
139            AxiamError::Validation { .. } => "validation_error",
140            AxiamError::PasswordPolicy { .. } => "password_policy_violation",
141            AxiamError::TenantContext => "tenant_context",
142            AxiamError::RateLimited => "rate_limited",
143            AxiamError::EmailConfig(_) => "email_config_error",
144            AxiamError::ServiceUnavailable(_) => "service_unavailable",
145            // T-262 / R-4. Its own slug rather than `service_unavailable`,
146            // because the two are different operational events and an operator
147            // reading logs needs to tell "the datastore is contended" from
148            // "the Argon2 gate is saturated".
149            AxiamError::WriteContention => "write_contention",
150            // Server-error variants: log detail, return generic message.
151            _ => "internal_error",
152        };
153
154        // Client-facing message: echo for known client errors; generic for 5xx.
155        // SEC-011/SEC-039/CQ-B33: internal detail (DB strings, crypto messages,
156        // stack traces) MUST NOT appear in the response body.
157        let message = match &self.0 {
158            AxiamError::NotFound { .. }
159            | AxiamError::AlreadyExists { .. }
160            | AxiamError::Conflict { .. }
161            | AxiamError::AuthenticationFailed { .. }
162            | AxiamError::ReplayDetected
163            | AxiamError::AuthorizationDenied { .. }
164            | AxiamError::OpaqueRequired
165            | AxiamError::MfaEnforced
166            | AxiamError::Validation { .. }
167            | AxiamError::PasswordPolicy { .. }
168            | AxiamError::TenantContext
169            | AxiamError::RateLimited
170            | AxiamError::EmailConfig(_)
171            | AxiamError::ServiceUnavailable(_)
172            // Safe to echo: the variant carries no payload, so its `Display`
173            // is the fixed sentence written on it and never the engine's own
174            // words, which stay on `DbError::Conflict` for the log.
175            | AxiamError::WriteContention => self.0.to_string(),
176            // 5xx variants: log the detail server-side, return only a generic message.
177            _ => {
178                error!(
179                    error = %self.0,
180                    "internal server error"
181                );
182                "An internal error occurred".to_string()
183            }
184        };
185
186        // SDK-Q02: surface the checked action/resource on authorization
187        // denials so SDKs can parse them from the response body (CONTRACT
188        // §2 "if available from the response body"). Every other error
189        // kind omits both fields (skip_serializing_if above).
190        let (action, resource_id) = match &self.0 {
191            AxiamError::AuthorizationDenied {
192                action,
193                resource_id,
194                ..
195            } => (action.clone(), resource_id.clone()),
196            _ => (None, None),
197        };
198
199        let mut builder = HttpResponse::build(self.status_code());
200        // T-262 / R-4: the one response that carries a header. A contended
201        // write is transient, and `Retry-After` is how an HTTP client is told
202        // so — CONTRACT §16.1 makes every SDK honour it as a **floor**, so a
203        // caller's own backoff still governs the wait and a `1` cannot shorten
204        // it. One second is a convention, not a measurement: the server does
205        // not know how long contention will last, and a fabricated number
206        // would be worse than a conventional one.
207        //
208        // Set here rather than inside the slug match above so the status and
209        // the header cannot drift apart.
210        if matches!(self.0, AxiamError::WriteContention) {
211            builder.insert_header((actix_web::http::header::RETRY_AFTER, "1"));
212        }
213        builder.json(ErrorBody {
214            error: error_code.into(),
215            message,
216            action,
217            resource_id,
218            reason_code,
219        })
220    }
221}

223#[cfg(test)]
224mod write_contention_tests {
225    use actix_web::ResponseError;
226    use actix_web::body::MessageBody;
227    use actix_web::http::header::RETRY_AFTER;
228
229    use super::*;
230
231    fn render(err: AxiamError) -> (u16, Option<String>, serde_json::Value) {
232        let response = AxiamApiError(err).error_response();
233        let status = response.status().as_u16();
234        let retry_after = response
235            .headers()
236            .get(RETRY_AFTER)
237            .map(|v| v.to_str().unwrap().to_owned());
238        let bytes = response.into_body().try_into_bytes().unwrap();
239        (status, retry_after, serde_json::from_slice(&bytes).unwrap())
240    }
241
242    /// T-262 / R-4. A write that lost a datastore race and stayed lost is a
243    /// statement about the server, so it answers `503` and says when to come
244    /// back — the answer an IdP driving SCIM provisioning retries, rather than
245    /// the `500` it reads as a failed sync and recovers from by re-sending the
246    /// whole record.
247    #[test]
248    fn a_contended_write_answers_503_with_retry_after() {
249        let (status, retry_after, body) = render(AxiamError::WriteContention);
250        assert_eq!(status, 503);
251        assert_eq!(retry_after.as_deref(), Some("1"));
252        assert_eq!(body["error"], "write_contention");
253    }
254
255    /// The response body must never carry the engine's own words. The variant
256    /// has no payload, so this cannot regress by accident — but it could
257    /// regress by somebody giving it one "for debugging", which is what this
258    /// pins.
259    #[test]
260    fn the_body_never_carries_the_engines_message() {
261        let (_, _, body) = render(AxiamError::WriteContention);
262        let rendered = body.to_string();
263        for leak in ["Transaction", "write conflict", "surreal", "SurrealDB"] {
264            assert!(
265                !rendered.contains(leak),
266                "{leak:?} must not appear in the response body: {rendered}"
267            );
268        }
269    }
270
271    /// **I4 twin.** The two neighbouring answers are unchanged, and they are
272    /// the ones it would be easy to fold this into. A uniqueness violation and
273    /// a state precondition are both statements about the *request*: they stay
274    /// `409`, carry no `Retry-After`, and mean "do something else", not "come
275    /// back".
276    #[test]
277    fn the_409_answers_are_untouched_and_carry_no_retry_after() {
278        for (err, slug) in [
279            (
280                AxiamError::AlreadyExists {
281                    entity: "user".into(),
282                },
283                "already_exists",
284            ),
285            (
286                AxiamError::Conflict {
287                    reason: "the export is stale".into(),
288                },
289                "conflict",
290            ),
291        ] {
292            let (status, retry_after, body) = render(err);
293            assert_eq!(status, 409);
294            assert_eq!(retry_after, None, "a 409 must not advertise a retry");
295            assert_eq!(body["error"], slug);
296        }
297    }
298
299    /// And `503 service_unavailable` — the Argon2-gate answer — keeps its own
300    /// slug and gains no header. The two are different operational events and
301    /// an operator reading logs has to be able to tell them apart.
302    #[test]
303    fn the_other_503_is_a_different_answer() {
304        let (status, retry_after, body) =
305            render(AxiamError::ServiceUnavailable("hash gate saturated".into()));
306        assert_eq!(status, 503);
307        assert_eq!(retry_after, None);
308        assert_eq!(body["error"], "service_unavailable");
309    }
310
311    /// `reason_code` is one member added beside `error` by a route that
312    /// documents one (§36.4); the slug, message and status are the type's own,
313    /// and every other response omits the member.
314    #[test]
315    fn a_reason_code_is_one_member_added_and_absent_otherwise() {
316        let err = AxiamApiError(AxiamError::Conflict {
317            reason: "the network segment changed since it was read".into(),
318        });
319        let response = err.error_response_with_reason_code(Some("stale_version"));
320        assert_eq!(response.status().as_u16(), 409);
321        let bytes = response.into_body().try_into_bytes().unwrap();
322        let body: serde_json::Value = serde_json::from_slice(&bytes).unwrap();
323        assert_eq!(body["error"], "conflict");
324        assert_eq!(body["reason_code"], "stale_version");
325        assert_eq!(
326            body["message"],
327            "the network segment changed since it was read"
328        );
329
330        let (_, _, plain) = render(AxiamError::Conflict { reason: "x".into() });
331        assert!(plain.get("reason_code").is_none(), "{plain}");
332    }
