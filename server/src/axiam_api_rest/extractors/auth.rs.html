26/// Object-safe per-request session-validity check (D-15 / REQ-7).
27///
28/// Access tokens are stateless JWTs, so revoking a session (deleting its row)
29/// has no effect unless every authenticated request re-checks that the session
30/// behind the token's `jti` (= `session.id`) is still active. This trait is the
31/// object-safe seam that lets the connection-agnostic [`AuthenticatedUser`]
32/// extractor perform that check without being generic over the DB `Connection`.
33///
34/// Mirrors the [`crate::authz::AuthzChecker`] boxed-future pattern because the
35/// underlying repository methods are native `async fn` (RPITIT, not dyn-safe).
36pub trait SessionValidator: Send + Sync {
37    /// Returns `true` if a non-expired session with `session_id` exists for
38    /// `tenant_id`. A revoked session (row deleted) or an expired one is
39    /// considered inactive.
40    fn is_session_active<'a>(
41        &'a self,
42        tenant_id: Uuid,
43        session_id: Uuid,
44    ) -> Pin<Box<dyn Future<Output = bool> + Send + 'a>>;
45}
46
47impl<C: Connection> SessionValidator for SurrealSessionRepository<C> {
48    fn is_session_active<'a>(
49        &'a self,
50        tenant_id: Uuid,
51        session_id: Uuid,
52    ) -> Pin<Box<dyn Future<Output = bool> + Send + 'a>> {
53        // I6: delegate to the repository, which answers from its optional
54        // short-TTL validity cache when one is attached and otherwise performs
55        // exactly the read this used to inline. Keeping the logic there is what
56        // guarantees the cache is invalidated by the same methods that delete
57        // the rows — see `axiam_db::session_validation_cache`.
58        Box::pin(self.is_session_active_checked(tenant_id, session_id))
59    }
60}

62/// The header a client uses to say which tenant it is acting on.
63///
64/// Only meaningful for an organization-level principal, which is the only kind
65/// that has more than one answer. Honoured for anyone else would be a request
66/// asking to be treated as belonging to a tenant it does not belong to.
67pub const ACTIVE_TENANT_HEADER: &str = "X-Axiam-Tenant";

69/// What the authorization layer needs to know about a tenant to decide whether
70/// a principal may act on it.
71#[derive(Debug, Clone, Copy)]
72pub struct TenantScope {
73    pub organization_id: Uuid,
74    /// Whether this is the organization's own reserved tenant.
75    pub is_organization: bool,
76}

78/// Object-safe tenant lookup for the authentication extractor.
79///
80/// The same boxed-future seam as [`SessionValidator`], and for the same reason:
81/// the extractor is not generic over the DB `Connection` and the repository
82/// methods are native `async fn`.
83///
84/// # Why this cannot be answered from the token
85///
86/// The token says which tenant the principal *lives in*. Whether some **other**
87/// tenant belongs to the same organization is a fact about that other tenant,
88/// and a client that could assert it could name a tenant in an organization it
89/// has nothing to do with. An organization-level principal's global grants
90/// carry across tenant boundaries by design; without this check they would
91/// carry across *organization* boundaries too, which is the one thing an
92/// organization is.
93pub trait TenantScopeResolver: Send + Sync {
94    fn resolve<'a>(
95        &'a self,
96        tenant_id: Uuid,
97    ) -> Pin<Box<dyn Future<Output = Option<TenantScope>> + Send + 'a>>;
98}
99
100impl<C: Connection> TenantScopeResolver for axiam_db::SurrealTenantRepository<C> {
101    fn resolve<'a>(
102        &'a self,
103        tenant_id: Uuid,
104    ) -> Pin<Box<dyn Future<Output = Option<TenantScope>> + Send + 'a>> {
105        Box::pin(async move {
106            use axiam_core::repository::TenantRepository as _;
107            self.get_by_id(tenant_id).await.ok().map(|t| TenantScope {
108                organization_id: t.organization_id,
109                is_organization: t.is_organization_scope(),
110            })
111        })
112    }
113}

115/// Authenticated user context extracted from a valid JWT.
116///
117/// Use this as a handler parameter to require authentication.
118/// If the audit middleware has already validated the token, the cached
119/// claims are reused to avoid double verification.
120///
121/// **Audience narrowing (D-19 / D-21):** only tokens with `aud = axiam:user`
122/// are accepted on user-facing routes. M2M tokens are rejected with 401.
123/// When `aud` is absent and `AuthConfig.allow_missing_aud_as_user` is `true`,
124/// the token is accepted with a `WARN` log on every request. No in-process
125/// rate limit is applied — operator-side log filtering (Loki dedup, fluent-bit
126/// rewrite_tag, etc.) handles burst noise. This trade-off keeps the extractor
127/// allocation-free and avoids shared mutable state.
128#[derive(Debug, Clone)]
129pub struct AuthenticatedUser {
130    pub user_id: Uuid,
131    /// The tenant being **acted upon**.
132    ///
133    /// Equal to [`Self::principal_tenant_id`] for every ordinary principal,
134    /// which is why the ~100 handlers that read this field did not change when
135    /// organization scope arrived. For an organization-level principal it is
136    /// whichever tenant the request named in [`ACTIVE_TENANT_HEADER`],
137    /// validated to be in the caller's own organization.
138    pub tenant_id: Uuid,
139    /// The tenant the caller's own record and grants live in.
140    ///
141    /// The organization tenant for an organization-level principal, and the
142    /// same as [`Self::tenant_id`] for everyone else.
143    pub principal_tenant_id: Uuid,
144    /// Whether [`Self::principal_tenant_id`] was **verified** to be the
145    /// organization's reserved scope.
146    ///
147    /// Set only by [`resolve_active_tenant`], which reads the tenant record and
148    /// checks its `kind`. `false` until then, so a request that never named
149    /// another tenant is evaluated as an ordinary tenant principal — which it
150    /// is, as far as this request is concerned.
151    ///
152    /// This is the flag that becomes
153    /// [`axiam_authz::types::SubjectScope::Organization`], and it is the whole
154    /// reason that claim is trustworthy: it cannot be reached without the
155    /// lookup that justifies it.
156    pub organization_level: bool,
157    pub org_id: Uuid,
158    /// The session ID — equals the JWT `jti` claim which is set to
159    /// `session.id` for user-facing tokens (D-15). Use this for
160    /// selective session invalidation on password change.
161    pub session_id: Uuid,
162    pub claims: ValidatedClaims,
163}
164
165impl AuthenticatedUser {
166    /// How the authorization engine should read this caller's grants.
167    ///
168    /// [`SubjectScope::Organization`] only when
169    /// [`Self::organization_level`] was set by the tenant lookup in
170    /// [`resolve_active_tenant`], so the claim can never be made on the
171    /// strength of request input alone.
172    pub fn subject_scope(&self) -> SubjectScope {
173        if self.organization_level {
174            SubjectScope::Organization {
175                tenant_id: self.principal_tenant_id,
176            }
177        } else {
178            SubjectScope::Tenant
179        }
180    }
181}

183impl actix_web::FromRequest for AuthenticatedUser {
184    type Error = AxiamApiError;
185    type Future = Pin<Box<dyn Future<Output = Result<Self, Self::Error>>>>;
186
187    fn from_request(req: &HttpRequest, _payload: &mut Payload) -> Self::Future {
188        // Synchronous JWT/aud/jti extraction first (ends the `req` borrow before
189        // the async block). Clone the optional session validator (an Arc) so the
190        // returned future is `'static`.
191        let user_result = extract_user(req);
192        let validator = req
193            .app_data::<web::Data<Arc<dyn SessionValidator>>>()
194            .map(|d| d.get_ref().clone());
195        let tenants = req
196            .app_data::<web::Data<Arc<dyn TenantScopeResolver>>>()
197            .map(|d| d.get_ref().clone());
198        // Read the header here, while `req` is still borrowed. A malformed
199        // value is dropped rather than refused: it names a tenant that cannot
200        // exist, so the request falls back to the caller's own tenant and is
201        // then denied by RBAC like any other over-reach.
202        let requested_tenant = req
203            .headers()
204            .get(ACTIVE_TENANT_HEADER)
205            .and_then(|v| v.to_str().ok())
206            .and_then(|v| Uuid::parse_str(v.trim()).ok());
207
208        Box::pin(async move {
209            let mut user = user_result?;
210
211            // REQ-7 / D-15: reject access tokens whose session has been revoked
212            // (row deleted on password change/reset/MFA reset) or expired. The
213            // validator is optional so non-session test harnesses are unaffected;
214            // the production server (and session-security tests) always register it.
215            //
216            // Checked against the *principal's* tenant, which is where the
217            // session row is. It is the same value as `tenant_id` at this point
218            // — the header is applied below — but ordering matters: resolving
219            // the active tenant first would look the session up in whichever
220            // tenant the request asked to act on, and find none.
221            if let Some(validator) = validator
222                && !validator
223                    .is_session_active(user.principal_tenant_id, user.session_id)
224                    .await
225            {
226                return Err(AxiamError::AuthenticationFailed {
227                    reason: "session revoked or expired".into(),
228                }
229                .into());
230            }
231
232            if let Some(target) = requested_tenant.filter(|t| *t != user.principal_tenant_id) {
233                user.tenant_id = resolve_active_tenant(&user, target, tenants.as_ref()).await?;
234                // Reached only when the resolution above confirmed the caller's
235                // own tenant is the organization scope — it refuses otherwise.
236                user.organization_level = true;
237            }
238
239            Ok(user)
240        })
241    }
242}

244/// Service-account context extracted from a valid M2M JWT.
245///
246/// Use this as a handler parameter to require M2M authentication.
247/// Accepts only tokens with `aud = axiam:m2m`; user tokens are rejected.
248#[derive(Debug, Clone)]
249pub struct AuthenticatedServiceAccount {
250    /// The token's `sub` claim — the **subject** of the machine token.
251    ///
252    /// **Its meaning depends on which principal obtained the token**, so do not
253    /// treat it as an OAuth2 `client_id` without checking `claims.sub_kind`:
254    ///
255    /// * `sub_kind = oauth2_client` → the OAuth2 `client_id` (`oa_…`), because
256    ///   `issue_client_credentials_token` stamps the client id as `sub`.
257    /// * `sub_kind = service_account` → the service-account **UUID**, because a
258    ///   service account's grants are resolved by subject id (§16.6). It is
259    ///   *not* the `sa_…` client id.
260    ///
261    /// Named `subject` rather than `client_id` for exactly that reason
262    /// (§17.2 residual 7): the old name was accurate for only one of the two
263    /// principals and invited the first consumer to treat a UUID as a client
264    /// id. No route consumes this extractor today, so the rename is free —
265    /// which is why it happens now rather than after something depends on it.
266    pub subject: String,
267    pub tenant_id: Uuid,
268    pub claims: ValidatedClaims,
269}

284/// Decode and validate the JWT from the request — shared between the
285/// extractors. Does not apply audience narrowing.
286pub(crate) fn parse_validated_claims(req: &HttpRequest) -> Result<ValidatedClaims, AxiamApiError> {
287    let config = req
288        .app_data::<web::Data<AuthConfig>>()
289        .ok_or(AxiamError::Internal("missing auth config".into()))?;
290
291    // Try cookie first (browser clients), then Authorization header (service clients).
292    let token = if let Some(cookie) = req.cookie("axiam_access") {
293        cookie.value().to_owned()
294    } else {
295        // Fall back to the Authorization header for non-browser clients.
296        let header = req
297            .headers()
298            .get("Authorization")
299            .and_then(|v| v.to_str().ok())
300            .ok_or(AxiamError::AuthenticationFailed {
301                reason: "missing authentication credentials".into(),
302            })?;
303
304        // Parse `Authorization` as a case-insensitive scheme with flexible
305        // whitespace. `DPoP` joins `Bearer` here (X5.1, RFC 9449 §7.1): a
306        // DPoP-bound token arrives under its own scheme, and a guard that only
307        // ever splits on `Bearer ` would not find the token at all.
308        let header = header.trim();
309        let mut parts = header.splitn(2, char::is_whitespace);
310        let scheme = parts.next().unwrap_or("");
311        let credentials = parts.next().unwrap_or("").trim();
312
313        let known_scheme =
314            scheme.eq_ignore_ascii_case("bearer") || axiam_oauth2::dpop::is_dpop_scheme(scheme);
315        if !known_scheme || credentials.is_empty() {
316            return Err(AxiamError::AuthenticationFailed {
317                reason: "invalid Authorization scheme, expected Bearer or DPoP".into(),
318            }
319            .into());
320        }
321        credentials.to_owned()
322    };
323
324    let validated = validate_access_token(&token, config).map_err(AxiamError::from)?;
325    enforce_sender_constraint(req, &token, &validated.0)?;
326    Ok(validated)
327}

329/// Enforce a token's `cnf` sender constraint against what this request actually
330/// proved (X5.1; RFC 8705 §3.2 and RFC 9449 §7.1).
331///
332/// AXIAM's own API is a resource server for its own tokens, so it owes the same
333/// obligation the SDK contract's §10.1 rule 9 places on every SDK: **a token
334/// carrying `cnf` is not a bearer token, and a `cnf` naming a method this
335/// validator cannot check is refused rather than read as unbound.**
336///
337/// # Ordinary callers are untouched
338///
339/// A token with no `cnf` — which is every token AXIAM issued before X5.1 and
340/// every token it issues to a client that has not asked for binding — returns
341/// `Ok` here without any evidence being gathered, and without either header
342/// being read. That is the property `an_unbound_token_is_never_asked_for_a_proof`
343/// pins, and it is the one this function is most likely to break.
344///
345/// # The replay caveat, stated rather than hidden
346///
347/// The proof's signature, `typ`, `alg`, `htm`, `htu`, `iat` freshness, `ath` and
348/// `jkt` are all checked. Its `jti` is **not** recorded here, because actix
349/// extractors are synchronous and the replay store is not. Within the 60-second
350/// freshness window a captured proof for *this exact method and URI*, presented
351/// with *the same token*, would therefore be accepted a second time on this
352/// path. The token endpoint — which is async and does hold the repository —
353/// does record it. This is exactly the limitation §21.7.2's row 8 requires an
354/// implementation to document rather than paper over, and closing it means
355/// moving this check into middleware that can await.
356fn enforce_sender_constraint(
357    req: &HttpRequest,
358    token: &str,
359    claims: &axiam_auth::token::AccessTokenClaims,
360) -> Result<(), AxiamApiError> {
361    use axiam_auth::token::{PresentedProofs, verify_token_binding};
362
363    // The fast path, and the common one: nothing to enforce, nothing read.
364    if claims.cnf.is_none() {
365        return Ok(());
366    }
367
368    let certificate_thumbprint = req
369        .conn_data::<crate::extractors::cert_auth::VerifiedClientCert>()
370        .map(|v| axiam_oauth2::mtls::thumbprint_s256(&v.der));
371
372    let dpop_thumbprint = verified_dpop_thumbprint(req, token);
373
374    verify_token_binding(
375        claims,
376        PresentedProofs {
377            certificate_thumbprint: certificate_thumbprint.as_deref(),
378            dpop_thumbprint: dpop_thumbprint.as_deref(),
379        },
380    )
381    .map_err(AxiamError::from)?;
382    Ok(())
383}

385/// The `jkt` of a DPoP proof on this request that **verified**, or `None`.
386///
387/// `None` for "no proof", "malformed proof" and "proof that failed
388/// verification" alike, which is the correct collapsing: every one of them means
389/// the caller has not proved possession, and
390/// [`axiam_auth::token::verify_token_binding`] refuses a `jkt`-bound token for
391/// all three. Returning a thumbprint from an unverified proof — the obvious
392/// shortcut, since the `jkt` is right there in the header — would turn DPoP
393/// into a self-signed permission slip.
394fn verified_dpop_thumbprint(req: &HttpRequest, token: &str) -> Option<String> {
395    use axiam_oauth2::dpop::{DpopExpectation, verify_dpop_proof};
396
397    let raw = req
398        .headers()
399        .get("DPoP")
400        .and_then(|v| v.to_str().ok())
401        .map(str::trim)
402        .filter(|v| !v.is_empty())?;
403    // RFC 9449 §4.3 step 2: more than one proof is a malformed request, not a
404    // choice of proofs.
405    if req.headers().get_all("DPoP").count() > 1 {
406        return None;
407    }
408
409    // SEC-102: the authority comes from the configured issuer, never from the
410    // request's `Forwarded` / `X-Forwarded-Host` / `Host` headers, which
411    // `HttpRequest::full_url()` would prefer in that order and none of which a
412    // trusted-proxy layer normalises here. Same reasoning as
413    // `handlers::oauth2::dpop_htu`, at the resource-server end of the same
414    // comparison: with a request-derived authority the caller chooses both
415    // sides of RFC 9449 §4.3 step 9.
416    //
417    // A deployment with no `AuthConfig` in scope cannot compute the server's
418    // own view of its URI, so no proof verifies — the fail-closed direction: a
419    // `jkt`-bound token is then refused by `verify_token_binding` rather than
420    // accepted against an attacker-chosen `htu`.
421    let config = req.app_data::<web::Data<AuthConfig>>()?;
422    let htu = format!("{}{}", config.effective_issuer(), req.path());
423    let expect = DpopExpectation::at_resource_server(
424        req.method().as_str(),
425        &htu,
426        token,
427        chrono::Utc::now().timestamp(),
428    );
429
430    match verify_dpop_proof(raw, &expect) {
431        Ok(proof) => Some(proof.jkt),
432        Err(e) => {
433            tracing::debug!(error = %e, "a DPoP proof on a resource request did not verify");
434            None
435        }
436    }
437}

443/// Decide whether `target` may be this request's active tenant.
444///
445/// Two conditions, both required, and both refusals rather than silent
446/// fallbacks — a caller that named a tenant it may not act on should be told
447/// so, not served a different tenant's data under a header it thinks was
448/// honoured.
449///
450/// 1. The caller must be **organization-level**: its own record lives in the
451///    organization's reserved tenant. A tenant user has exactly one tenant it
452///    can act on, and a request asking otherwise is asking to be somebody else.
453/// 2. `target` must belong to the **caller's own organization**. This is the
454///    check that keeps organization scope inside an organization: an
455///    organization-level principal's global grants deliberately cross tenant
456///    boundaries, so without it they would cross organization boundaries too.
457///
458/// With no resolver registered — the shape most test harnesses use — the header
459/// is refused rather than trusted. Failing closed is the only safe default for
460/// a check whose entire job is to say no.
461async fn resolve_active_tenant(
462    user: &AuthenticatedUser,
463    target: Uuid,
464    tenants: Option<&Arc<dyn TenantScopeResolver>>,
465) -> Result<Uuid, AxiamApiError> {
466    let deny = |reason: &str| -> AxiamApiError {
467        AxiamError::AuthorizationDenied {
468            reason: reason.to_string(),
469            action: None,
470            resource_id: None,
471        }
472        .into()
473    };
474
475    let Some(tenants) = tenants else {
476        return Err(deny(
477            "cannot verify the requested tenant: no tenant resolver is configured",
478        ));
479    };
480
481    let Some(home) = tenants.resolve(user.principal_tenant_id).await else {
482        return Err(deny("the caller's own tenant could not be resolved"));
483    };
484    if !home.is_organization {
485        return Err(deny(
486            "only organization-level principals may act on another tenant",
487        ));
488    }
489
490    let Some(scope) = tenants.resolve(target).await else {
491        return Err(deny("the requested tenant does not exist"));
492    };
493    // Compare against the *home tenant's* organization rather than the token's
494    // `org_id` claim: both should agree, and if they ever do not, the one
495    // backed by a row the caller cannot edit is the one to trust.
496    if scope.organization_id != home.organization_id {
497        return Err(deny(
498            "the requested tenant belongs to a different organization",
499        ));
500    }
501
502    Ok(target)
503}

505fn extract_user(req: &HttpRequest) -> Result<AuthenticatedUser, AxiamApiError> {
506    // Try to reuse claims cached by the audit middleware.
507    if let Some(cached) = req.extensions().get::<Arc<CachedUserIdentity>>() {
508        let config = req
509            .app_data::<web::Data<AuthConfig>>()
510            .ok_or(AxiamError::Internal("missing auth config".into()))?;
511
512        let session_id = check_user_aud_and_parse_jti(&cached.claims, config)?;
513        return Ok(AuthenticatedUser {
514            user_id: cached.user_id,
515            tenant_id: cached.tenant_id,
516            // The token's tenant is the home tenant until the header
517            // resolution in `from_request` says otherwise.
518            principal_tenant_id: cached.tenant_id,
519            organization_level: false,
520            org_id: cached.org_id,
521            session_id,
522            claims: cached.claims.clone(),
523        });
524    }
525
526    let config = req
527        .app_data::<web::Data<AuthConfig>>()
528        .ok_or(AxiamError::Internal("missing auth config".into()))?;
529
530    let validated = parse_validated_claims(req)?;
531
532    let session_id = check_user_aud_and_parse_jti(&validated, config)?;
533
534    let user_id =
535        Uuid::parse_str(&validated.0.sub).map_err(|_| AxiamError::AuthenticationFailed {
536            reason: "invalid sub claim".into(),
537        })?;
538
539    let tenant_id =
540        Uuid::parse_str(&validated.0.tenant_id).map_err(|_| AxiamError::AuthenticationFailed {
541            reason: "invalid tenant_id claim".into(),
542        })?;
543
544    let org_id =
545        Uuid::parse_str(&validated.0.org_id).map_err(|_| AxiamError::AuthenticationFailed {
546            reason: "invalid org_id claim".into(),
547        })?;
548
549    Ok(AuthenticatedUser {
550        user_id,
551        tenant_id,
552        principal_tenant_id: tenant_id,
553        organization_level: false,
554        org_id,
555        session_id,
556        claims: validated,
557    })
558}

560/// Enforce the audience narrowing policy for user-facing routes and
561/// parse the `jti` claim as a `Uuid` (= session ID).
562///
563/// Audience rules (D-19, D-21):
564/// - `aud = axiam:user` → accepted.
565/// - `aud` absent AND `allow_missing_aud_as_user = true` → accepted; a
566///   `WARN` is emitted on every such request. No in-process rate limiting is
567///   applied here — operator-side log deduplication (Loki, fluent-bit, etc.)
568///   handles burst noise, keeping this hot path allocation-free and
569///   lock-free.
570/// - `aud` absent AND `allow_missing_aud_as_user = false` → 401.
571/// - `aud = axiam:m2m` → 401 (audience mismatch — wrong token type for
572///   this route).
573/// - Any other `aud` value → 401 (unknown audience).
574fn check_user_aud_and_parse_jti(
575    validated: &ValidatedClaims,
576    config: &AuthConfig,
577) -> Result<Uuid, AxiamApiError> {
578    match validated.0.aud.as_deref() {
579        Some(AUD_USER) => {
580            // Expected audience — accept.
581        }
582        None => {
583            if config.allow_missing_aud_as_user {
584                // Back-compat window: treat as axiam:user, but warn unconditionally.
585                // TRADE-OFF: no in-process rate limit on this warn. Operator-side
586                // log filtering (e.g. Loki `rate()` alert, fluent-bit rewrite_tag
587                // dedup) is the intended mechanism for suppressing burst noise
588                // during the rollout window. Keeping this path lock-free is
589                // intentional — auth hot paths must not contend on shared atomics.
590                let jti = &validated.0.jti;
591                tracing::warn!(
592                    token_jti = %jti,
593                    "accepted access token without aud — backward-compat window active"
594                );
595            } else {
596                return Err(AxiamError::AuthenticationFailed {
597                    reason: "aud required".into(),
598                }
599                .into());
600            }
601        }
602        Some(AUD_M2M) => {
603            return Err(AxiamError::AuthenticationFailed {
604                reason: "audience mismatch — this route requires axiam:user audience".into(),
605            }
606            .into());
607        }
608        Some(_) => {
609            return Err(AxiamError::AuthenticationFailed {
610                reason: "unknown audience".into(),
611            }
612            .into());
613        }
614    }
615
616    Uuid::parse_str(&validated.0.jti).map_err(|_| {
617        AxiamError::AuthenticationFailed {
618            reason: "invalid jti".into(),
619        }
620        .into()
621    })
622}

663// ---------------------------------------------------------------------------
664// AuthenticatedPrincipal extractor — accepts either audience
665// ---------------------------------------------------------------------------
666
667/// An authenticated caller that may be **either** a human user
668/// (`aud = axiam:user`) or a machine (`aud = axiam:m2m`).
669///
670/// This exists so the mTLS device path can carry the machine audience
671/// (§17.2 residual 1) without losing the endpoints a device legitimately
672/// needs. Before it, `axiam:m2m` reached **no** REST route at all — every
673/// guarded handler takes [`AuthenticatedUser`], which rejects that audience —
674/// so moving device tokens to `axiam:m2m` on its own would have replaced a
675/// too-wide grant with no grant, which is an outage rather than a narrowing.
676///
677/// **What this deliberately is not:** a general-purpose relaxation. It is used
678/// only on the authorization-check endpoints, which are the machine-facing
679/// read-only surface. Every other route keeps [`AuthenticatedUser`] and so
680/// keeps rejecting machine tokens outright. Widening its use is a security
681/// decision, not a convenience one.
682///
683/// Two properties worth stating because they are easy to lose:
684///
685/// * **User tokens are not weakened.** A `axiam:user` token extracted this way
686///   still goes through the same session-revocation check
687///   [`AuthenticatedUser`] applies, so a revoked session cannot reach these
688///   endpoints through the wider extractor. That check is skipped only for
689///   machine tokens, which have no session row to check by construction.
690/// * **`aud` must be present.** The `allow_missing_aud_as_user` back-compat
691///   window is honoured for the user branch only, exactly as on the narrow
692///   extractor — an absent `aud` is never silently treated as a machine.
693#[derive(Debug, Clone)]
694pub struct AuthenticatedPrincipal {
695    /// The subject the authorization engine should evaluate: the user id for a
696    /// user token, the service-account id for a machine token. Both are the
697    /// token's `sub`, and both are what role assignments are keyed on.
698    pub subject_id: Uuid,
699    pub tenant_id: Uuid,
700    pub org_id: Uuid,
701    /// `true` when the caller authenticated as a machine (`aud = axiam:m2m`).
702    /// Handlers use this for audit attribution, not for authorization — RBAC
703    /// is applied identically to both kinds.
704    pub is_machine: bool,
705    pub claims: ValidatedClaims,
706}
707
708impl AuthenticatedPrincipal {
709    /// Audit actor type matching how this principal authenticated.
710    pub fn actor_type(&self) -> axiam_core::models::audit::ActorType {
711        if self.is_machine {
712            axiam_core::models::audit::ActorType::ServiceAccount
713        } else {
714            axiam_core::models::audit::ActorType::User
715        }
716    }
717}
718
719fn extract_principal(req: &HttpRequest) -> Result<AuthenticatedPrincipal, AxiamApiError> {
720    let config = req
721        .app_data::<web::Data<AuthConfig>>()
722        .ok_or(AxiamError::Internal("missing auth config".into()))?;
723
724    let validated = match req.extensions().get::<Arc<CachedUserIdentity>>() {
725        Some(cached) => cached.claims.clone(),
726        None => parse_validated_claims(req)?,
727    };
728
729    // Machine tokens take the m2m branch; everything else (including the
730    // absent-`aud` back-compat window) is decided by the *same* function the
731    // narrow user extractor uses, so the two cannot drift apart.
732    let is_machine = validated.0.aud.as_deref() == Some(AUD_M2M);
733    if !is_machine {
734        check_user_aud_and_parse_jti(&validated, config)?;
735    }
736
737    // `sub` must be a UUID, which quietly makes this extractor reject one of
738    // the two machine principal kinds — an asymmetry worth stating rather than
739    // leaving to be rediscovered.
740    //
741    // A *service account*'s `sub` is its UUID, so it parses. An *OAuth2
742    // client*'s `sub` is its `oa_…` client id, so it does not, and such a
743    // caller gets 401 here even though its token carries `axiam:m2m` and is
744    // accepted elsewhere.
745    //
746    // That is the correct outcome, not a gap to close: role assignments are
747    // keyed on a subject UUID, and an OAuth2 client has no row in that graph.
748    // Admitting one would produce a principal the authorization engine can
749    // only ever evaluate to "no grants" — a caller that authenticates and then
750    // fails every check, which is a worse experience than a clean 401 and
751    // invites someone to "fix" it later by inventing a subject mapping. If
752    // OAuth2 clients ever need to be RBAC subjects, that is a deliberate
753    // modelling change, and this parse is where it would start.
754    let subject_id =
755        Uuid::parse_str(&validated.0.sub).map_err(|_| AxiamError::AuthenticationFailed {
756            reason: "invalid sub claim".into(),
757        })?;
758    let tenant_id =
759        Uuid::parse_str(&validated.0.tenant_id).map_err(|_| AxiamError::AuthenticationFailed {
760            reason: "invalid tenant_id claim".into(),
761        })?;
762    let org_id =
763        Uuid::parse_str(&validated.0.org_id).map_err(|_| AxiamError::AuthenticationFailed {
764            reason: "invalid org_id claim".into(),
765        })?;
766
767    Ok(AuthenticatedPrincipal {
768        subject_id,
769        tenant_id,
770        org_id,
771        is_machine,
772        claims: validated,
773    })
774}
775
776impl actix_web::FromRequest for AuthenticatedPrincipal {
777    type Error = AxiamApiError;
778    type Future = Pin<Box<dyn Future<Output = Result<Self, Self::Error>>>>;
779
780    fn from_request(req: &HttpRequest, _payload: &mut Payload) -> Self::Future {
781        let principal_result = extract_principal(req);
782        let validator = req
783            .app_data::<web::Data<Arc<dyn SessionValidator>>>()
784            .map(|d| d.get_ref().clone());
785
786        Box::pin(async move {
787            let principal = principal_result?;
788
789            // A user token reaching these endpoints must satisfy exactly the
790            // same session-revocation rule it would on any other route
791            // (REQ-7 / D-15). Skipping it for machines is not a relaxation:
792            // a machine token has no session row, so there is nothing to
793            // revoke — revocation for machines is disabling the account,
794            // which the token-issuing path checks.
795            if !principal.is_machine
796                && let Some(validator) = validator
797            {
798                let session_id = Uuid::parse_str(&principal.claims.0.jti).map_err(|_| {
799                    AxiamError::AuthenticationFailed {
800                        reason: "invalid jti".into(),
801                    }
802                })?;
803                if !validator
804                    .is_session_active(principal.tenant_id, session_id)
805                    .await
806                {
807                    return Err(AxiamError::AuthenticationFailed {
808                        reason: "session revoked or expired".into(),
809                    }
810                    .into());
811                }
812            }
813            Ok(principal)
814        })
815    }
816}

1023    /// `axiam_auth::token::decode_access_token` deliberately accepts BOTH
1024    /// `axiam:user` and `axiam:m2m` (`token.rs`: `validation.set_audience(&[
1025    /// AUD_USER, AUD_M2M])`), deferring per-route audience separation to
1026    /// these extractors. That is only safe while the extractors really do
1027    /// narrow, so this test asserts the whole contract end to end in one
1028    /// place: **permissive at the token layer, strict at the route layer, in
1029    /// both directions.**
1030    ///
1031    /// If someone ever "simplifies" `check_user_aud_and_parse_jti` or
1032    /// `AuthenticatedServiceAccount`'s audience check away, the token layer's
1033    /// permissiveness silently becomes a cross-audience token-confusion bug
1034    /// (an M2M client-credentials token would be accepted as a user session,
1035    /// and vice-versa). This test is what stops that.
1036    #[test]
1037    fn token_layer_accepts_both_audiences_but_routes_narrow_in_both_directions() {
1038        let config = test_auth_config();
1039        let user_token = make_user_token(&config, None);
1040        let m2m_token = make_m2m_token(&config);
1041
1042        // 1. The token layer is deliberately permissive: both audiences
1043        //    decode successfully, with the audience preserved in the claims.
1044        let decoded_user = axiam_auth::token::decode_access_token(&user_token, &config)
1045            .expect("a user token must decode at the token layer");
1046        let decoded_m2m = axiam_auth::token::decode_access_token(&m2m_token, &config)
1047            .expect("an m2m token must ALSO decode at the token layer (D-20/SEC-006)");
1048        assert_eq!(decoded_user.aud.as_deref(), Some(AUD_USER));
1049        assert_eq!(decoded_m2m.aud.as_deref(), Some(AUD_M2M));
1050
1051        // 2. A user route accepts ONLY axiam:user.
1052        assert!(
1053            extract_user(&req_with_config_and_bearer(config.clone(), &user_token)).is_ok(),
1054            "axiam:user must be accepted on a user route"
1055        );
1056        assert!(
1057            extract_user(&req_with_config_and_bearer(config.clone(), &m2m_token)).is_err(),
1058            "axiam:m2m must be REJECTED on a user route — the token layer accepts it, so \
1059             this extractor is the only thing enforcing the separation"
1060        );

1062        // 3. An m2m route accepts ONLY axiam:m2m.
1063        assert!(
1064            extract_service_account(&req_with_config_and_bearer(config.clone(), &m2m_token))

1071        );
