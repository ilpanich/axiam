1//! Certificate-based authentication extractor for mTLS device auth.
2//!
3//! Two identity sources are supported, in order of trust:
4//!
5//! 1. **Native mTLS (D3):** when the server terminates TLS in-process with
6//!    client-cert auth enabled (`AXIAM__SERVER__TLS__CLIENT_AUTH`), rustls
7//!    cryptographically verifies the client certificate against the configured
8//!    CA bundle *during the handshake*. `axiam-server`'s `on_connect` hook then
9//!    stores the verified leaf certificate as a [`VerifiedClientCert`] in the
10//!    connection extensions. This extractor consumes that verified certificate —
11//!    it cannot be forged by a request header.
12//! 2. **Legacy proxy path:** the `X-Client-Certificate` header (URL-encoded PEM
13//!    forwarded by a TLS-terminating reverse proxy). Used only when no verified
14//!    certificate is present on the connection (i.e. TLS terminated upstream).
15//!
16//! Both paths validate the certificate via [`DeviceAuthService`].
17

31/// A client certificate that rustls **verified** against the configured client
32/// CA bundle during the TLS 1.3 handshake (D3).
33///
34/// `axiam-server`'s `HttpServer::on_connect` hook builds this from the rustls
35/// connection's `peer_certificates()` and inserts it into the per-connection
36/// [`actix_web::dev::Extensions`]; handlers read it back with
37/// `HttpRequest::conn_data::<VerifiedClientCert>()`. Because it originates from
38/// the verified peer chain — not a header — it is a trusted identity assertion.
39#[derive(Debug, Clone)]
40pub struct VerifiedClientCert {
41    /// What the TLS layer was able to say about this certificate: whether it
42    /// chained to a configured trust anchor, or was accepted self-asserted
43    /// under [`ClientAuth::OptionalSelfSigned`][cauth] (RFC 8705 §2.2).
44    ///
45    /// Every consumer must decide what it does with a
46    /// [`CertTrust::SelfAsserted`] certificate. Device authentication, in this
47    /// very module, refuses it.
48    ///
49    /// [cauth]: crate::config::ClientAuth::OptionalSelfSigned
50    pub trust: CertTrust,
51    /// DER encoding of the verified leaf certificate.
52    pub der: Vec<u8>,
53    /// Subject Alternative Names (DNS/URI/RFC822/IP) parsed from the leaf, in
54    /// certificate order. Empty if the leaf carries no SAN extension.
55    pub sans: Vec<String>,
56    /// Lowercase hex SHA-256 of the leaf's SubjectPublicKeyInfo (the SPKI
57    /// fingerprint) — a stable key-identity handle for cert-mapped identities.
58    pub spki_sha256: String,
59}
60
61impl VerifiedClientCert {
62    /// Parse SAN entries and the SPKI fingerprint from a DER-encoded leaf
63    /// certificate. Returns an error string only if the DER cannot be parsed as
64    /// an X.509 certificate (rustls has already accepted the same bytes by this
65    /// point, so this parse is expected to succeed).
66    ///
67    /// `trust` is not inferred, and there is no default: it is the caller's
68    /// statement of what the *handshake* established, and the only caller in a
69    /// position to know is `axiam-server`'s `on_connect` hook, which asks
70    /// `axiam_server::tls::peer_certificate_trust`. A defaulted parameter here
71    /// would have made [`CertTrust::ChainedToAnchor`] — the privileged value —
72    /// what a future call site gets by saying nothing.
73    pub fn from_der(der: &[u8], trust: CertTrust) -> Result<Self, String> {
74        let (_, cert) =
75            parse_x509_certificate(der).map_err(|e| format!("parse client cert DER: {e}"))?;
76
77        let mut sans = Vec::new();
78        if let Ok(Some(ext)) = cert.subject_alternative_name() {
79            for name in &ext.value.general_names {
80                match name {
81                    GeneralName::DNSName(s) => sans.push(format!("DNS:{s}")),
82                    GeneralName::RFC822Name(s) => sans.push(format!("email:{s}")),
83                    GeneralName::URI(s) => sans.push(format!("URI:{s}")),
84                    GeneralName::IPAddress(b) => sans.push(format!("IP:{}", fmt_ip(b))),
85                    _ => {}
86                }
87            }
88        }
89
90        // The `.raw` field of SubjectPublicKeyInfo is the DER of the full SPKI
91        // structure (RFC 5280 §4.1) — the standard input for an SPKI fingerprint.
92        let spki_sha256 = hex::encode(Sha256::digest(cert.public_key().raw));
93
94        Ok(Self {
95            trust,
96            der: der.to_vec(),
97            sans,
98            spki_sha256,
99        })
100    }
101
102    /// Refuse this certificate for device/IoT authentication unless it chained
103    /// to a configured mTLS trust anchor.
104    ///
105    /// # Why this exists at all (B-06)
106    ///
107    /// Device authentication's entire trust model is that the certificate was
108    /// issued by a CA an administrator flagged as an `mtls_trust_anchor`; the
109    /// lookup it performs resolves the certificate against the tenant's issued
110    /// certificates. A [`CertTrust::SelfAsserted`] certificate carries no such
111    /// issuance — it is admitted to the handshake only under
112    /// [`ClientAuth::OptionalSelfSigned`][cauth], and only so that an RFC 8705
113    /// §2.2 OAuth2 client can present a credential whose SHA-256 an
114    /// administrator registered. Letting one through here would be the
115    /// native-listener twin of **B-06**, where a certificate under a
116    /// never-flagged CA authenticated through the proxy header, reopened on the
117    /// listener that was supposed to be the trustworthy one.
118    ///
119    /// # Why it is a separate method
120    ///
121    /// [`CertificateAuthenticated::extract`] needs an `AppState<C>` and a live
122    /// `DeviceAuthService`, so it is reachable only from an integration test —
123    /// and the native-mTLS branch is not reachable even from there, because
124    /// `actix_web::test` performs no TLS handshake and offers no way to
125    /// populate the connection extensions this branch reads. Pulling the
126    /// decision out gives it somewhere to be tested at all.
127    ///
128    /// [cauth]: crate::config::ClientAuth::OptionalSelfSigned
129    ///
130    /// # Errors
131    ///
132    /// [`AxiamError::AuthenticationFailed`] for a self-asserted certificate.
133    /// The message names the actual problem — no trusted issuer — rather than
134    /// falling through to the proxy-header branch's message, which would tell
135    /// an operator to go and set `TRUST_FORWARDED_CLIENT_CERT`: advice that
136    /// would not help here and would, if taken, make the deployment worse.
137    pub fn check_usable_for_device_auth(&self) -> Result<(), AxiamError> {
138        if self.trust.is_chained_to_anchor() {
139            return Ok(());
140        }
141        tracing::warn!(
142            "refusing device certificate authentication: the peer certificate chains to \
143             no configured mTLS trust anchor. Self-asserted certificates are accepted by \
144             the listener only for RFC 8705 §2.2 OAuth2 client authentication \
145             (self_signed_tls_client_auth)"
146        );
147        Err(AxiamError::AuthenticationFailed {
148            reason: "client certificate authentication requires a certificate issued by a \
149                     trusted mTLS certificate authority"
150                .into(),
151        })
152    }
153}

155/// Best-effort textual rendering of a SAN IP address (4-byte v4 / 16-byte v6).
156fn fmt_ip(bytes: &[u8]) -> String {
157    match bytes.len() {
158        4 => bytes
159            .iter()
160            .map(|b| b.to_string())
161            .collect::<Vec<_>>()
162            .join("."),
163        16 => bytes
164            .chunks(2)
165            .map(|c| format!("{:02x}{:02x}", c[0], c[1]))
166            .collect::<Vec<_>>()
167            .join(":"),
168        _ => hex::encode(bytes),
169    }
170}

172/// Authenticated device context extracted from a client certificate.
173///
174/// Use this as a handler parameter to require certificate-based auth.
175#[derive(Debug, Clone)]
176pub struct CertificateAuthenticated {
177    pub service_account_id: Uuid,
178    pub tenant_id: Uuid,
179    pub certificate_id: Uuid,
180}
181
182impl CertificateAuthenticated {
183    /// Extract and validate the client certificate from the request.
184    ///
185    /// This is called manually from the handler rather than via
186    /// `FromRequest`, because the concrete `DeviceAuthService` type
187    /// depends on the DB connection generic `C`.
188    pub async fn extract<C: Connection + Clone>(req: &HttpRequest) -> Result<Self, AxiamApiError> {
189        let state = req
190            .app_data::<web::Data<AppState<C>>>()
191            .ok_or(AxiamError::Internal("missing AppState".into()))?;
192        let service = &state.pki.device_auth_service;
193
194        // Prefer the VERIFIED client certificate captured at TLS handshake time
195        // (D3 native mTLS): rustls has already checked it against the client-CA
196        // bundle, so it is authoritative and cannot be spoofed by a header. Only
197        // fall back to the `X-Client-Certificate` proxy header when TLS was
198        // terminated upstream (no verified cert on this connection) AND the
199        // operator has said that upstream is trusted to set the header.
200        let identity_result = if let Some(verified) = req.conn_data::<VerifiedClientCert>() {
201            // A certificate that chains to nothing must not authenticate a
202            // device, however well-formed it is and whatever it says about
203            // itself. See `check_usable_for_device_auth` for why (B-06).
204            verified.check_usable_for_device_auth()?;
205            service.authenticate_der(&verified.der).await
206        } else {
207            // A certificate is public data, and every check on the header path
208            // — fingerprint, status, expiry, chain to the CA — is satisfied by a
209            // *copy* of an enrolled device's certificate. Only the TLS handshake
210            // proves possession of the private key, and on this path there was
211            // no handshake to prove it in. So the header is worth exactly as
212            // much as the guarantee that a client could not have set it, and
213            // that guarantee is a property of the deployment, not of the
214            // request. Off by default; see
215            // `axiam_auth::AuthConfig::trust_forwarded_client_cert`.
216            if !state.auth_config.trust_forwarded_client_cert {
217                return Err(AxiamError::AuthenticationFailed {
218                    reason: "client certificate authentication requires a TLS-verified \
219                             peer certificate on this connection. The X-Client-Certificate \
220                             header is not trusted unless \
221                             AXIAM__AUTH__TRUST_FORWARDED_CLIENT_CERT is set, which is \
222                             only safe when a proxy terminates mTLS and overwrites that \
223                             header on every request"
224                        .into(),
225                }
226                .into());
227            }
228            let header = req
229                .headers()
230                .get("X-Client-Certificate")
231                .and_then(|v| v.to_str().ok())
232                .ok_or(AxiamError::AuthenticationFailed {
233                    reason: "missing client certificate (no native mTLS peer cert and no \
234                             X-Client-Certificate header)"
235                        .into(),
236                })?;
237
238            // URL-decode the PEM (reverse proxy URL-encodes it)
239            let pem = urldecode(header).map_err(|_| AxiamError::AuthenticationFailed {
240                reason: "invalid URL encoding in X-Client-Certificate".into(),
241            })?;
242
243            service.authenticate(&pem).await
244        };
245
246        let identity: DeviceIdentity = identity_result.map_err(|e| match &e {
247            // Map certificate/NotFound errors to proper 401/403 status codes
248            AxiamError::Certificate(msg) if msg.contains("not bound to a service account") => {
249                AxiamError::AuthorizationDenied {
250                    reason: msg.clone(),
251                    action: None,
252                    resource_id: None,
253                }
254            }
255            AxiamError::Certificate(msg) => AxiamError::AuthenticationFailed {
256                reason: msg.clone(),
257            },
258            AxiamError::NotFound { .. } => AxiamError::AuthenticationFailed {
259                reason: "unknown client certificate".into(),
260            },
261            _ => e,
262        })?;
263
264        Ok(CertificateAuthenticated {
265            service_account_id: identity.service_account_id,
266            tenant_id: identity.tenant_id,
267            certificate_id: identity.certificate_id,
268        })
269    }
270}

272/// Simple percent-decoding for the `X-Client-Certificate` header.
273fn urldecode(input: &str) -> Result<String, ()> {
274    let mut result = Vec::with_capacity(input.len());
275    let mut bytes = input.bytes();
276    while let Some(b) = bytes.next() {
277        if b == b'%' {
278            let hi = bytes.next().ok_or(())?;
279            let lo = bytes.next().ok_or(())?;
280            let hi = hex_val(hi)?;
281            let lo = hex_val(lo)?;
282            result.push((hi << 4) | lo);
283        } else {
284            result.push(b);
285        }
286    }
287    String::from_utf8(result).map_err(|_| ())
288}
289
290fn hex_val(b: u8) -> Result<u8, ()> {
291    match b {
292        b'0'..=b'9' => Ok(b - b'0'),
293        b'a'..=b'f' => Ok(b - b'a' + 10),
294        b'A'..=b'F' => Ok(b - b'A' + 10),
295        _ => Err(()),
296    }
297}

412    // -----------------------------------------------------------------------
413    // The B-06 guard: device authentication and self-asserted certificates.
414    //
415    // `client_auth = optional_self_signed` lets a certificate that chains to
416    // nothing complete a TLS handshake, so that an RFC 8705 §2.2 OAuth2 client
417    // can present the credential its registration is built around. Device
418    // authentication must be unaffected by that: its model is issuance by a
419    // flagged mTLS trust anchor, and a self-signed certificate has no issuer
420    // anybody flagged.
421    // -----------------------------------------------------------------------
422
423    fn device_cert(trust: CertTrust) -> VerifiedClientCert {
424        let cert = rcgen::generate_simple_self_signed(vec!["device.example.com".to_string()])
425            .expect("generate self-signed cert");
426        VerifiedClientCert::from_der(cert.cert.der().as_ref(), trust).expect("parse DER")
427    }

429    /// A self-asserted certificate cannot authenticate a device; a chained one
430    /// can.
431    ///
432    /// The certificates in both halves are generated the same way, so the only
433    /// difference between "refused" and "permitted" is what the TLS layer said
434    /// about the connection — which is the contract.
435    #[test]
436    fn device_auth_refuses_a_self_asserted_certificate() {
437        assert!(
438            device_cert(CertTrust::ChainedToAnchor)
439                .check_usable_for_device_auth()
440                .is_ok(),
441            "control: a certificate that chained to a configured anchor is device auth's \
442             normal case and must still pass"
443        );
444
445        let err = device_cert(CertTrust::SelfAsserted)
446            .check_usable_for_device_auth()
447            .expect_err("a certificate that chains to nothing must not authenticate a device");
448        assert!(
449            matches!(err, AxiamError::AuthenticationFailed { .. }),
450            "must fail authentication (401), not authorization or anything softer; got {err:?}"
451        );
452    }
453
454    /// The refusal must not send an operator to the wrong knob.
455    ///
456    /// Falling through to the proxy-header branch would have produced its error
457    /// text, which tells an operator to set
458    /// `AXIAM__AUTH__TRUST_FORWARDED_CLIENT_CERT`. That would not fix this —
459    /// there is no forwarded header on a native mTLS connection — and an
460    /// operator who took the advice would have widened a *different* trust
461    /// boundary while chasing this one. So the message is asserted, not just
462    /// the variant.
463    #[test]
464    fn the_device_auth_refusal_names_the_issuer_problem_not_the_header_knob() {
465        let AxiamError::AuthenticationFailed { reason } = device_cert(CertTrust::SelfAsserted)
466            .check_usable_for_device_auth()
467            .expect_err("must be refused")
468        else {
469            panic!("expected AuthenticationFailed");
470        };
471        assert!(
472            !reason.contains("TRUST_FORWARDED_CLIENT_CERT")
473                && !reason.contains("X-Client-Certificate"),
474            "the refusal must not point at the proxy-header configuration; got {reason:?}"
475        );
476        assert!(
477            reason.contains("certificate authority"),
478            "the refusal must say what is actually wrong — no trusted issuer; got {reason:?}"
479        );
480    }
