1//! Certificate-based authentication extractor for mTLS device auth.
2//!
3//! Two identity sources are supported, in order of trust:
4//!
5//! 1. **Native mTLS (D3):** when the server terminates TLS in-process with
6//!    client-cert auth enabled (`AXIAM__SERVER__TLS__CLIENT_AUTH`), rustls
7//!    cryptographically verifies the client certificate against the configured
8//!    CA bundle *during the handshake*. `axiam-server`'s `on_connect` hook then
9//!    stores the verified leaf certificate as a [`VerifiedClientCert`] in the
10//!    connection extensions. This extractor consumes that verified certificate —
11//!    it cannot be forged by a request header.
12//! 2. **Legacy proxy path:** the `X-Client-Certificate` header (URL-encoded PEM
13//!    forwarded by a TLS-terminating reverse proxy). Used only when no verified
14//!    certificate is present on the connection (i.e. TLS terminated upstream).
15//!
16//! Both paths validate the certificate via [`DeviceAuthService`].
17

31/// A client certificate that rustls **verified** against the configured client
32/// CA bundle during the TLS 1.3 handshake (D3).
33///
34/// `axiam-server`'s `HttpServer::on_connect` hook builds this from the rustls
35/// connection's `peer_certificates()` and inserts it into the per-connection
36/// [`actix_web::dev::Extensions`]; handlers read it back with
37/// `HttpRequest::conn_data::<VerifiedClientCert>()`. Because it originates from
38/// the verified peer chain — not a header — it is a trusted identity assertion.
39#[derive(Debug, Clone)]
40pub struct VerifiedClientCert {
41    /// DER encoding of the verified leaf certificate.
42    pub der: Vec<u8>,
43    /// Subject Alternative Names (DNS/URI/RFC822/IP) parsed from the leaf, in
44    /// certificate order. Empty if the leaf carries no SAN extension.
45    pub sans: Vec<String>,
46    /// Lowercase hex SHA-256 of the leaf's SubjectPublicKeyInfo (the SPKI
47    /// fingerprint) — a stable key-identity handle for cert-mapped identities.
48    pub spki_sha256: String,
49}
50
51impl VerifiedClientCert {
52    /// Parse SAN entries and the SPKI fingerprint from a DER-encoded leaf
53    /// certificate. Returns an error string only if the DER cannot be parsed as
54    /// an X.509 certificate (rustls has already verified the chain by this
55    /// point, so this parse is expected to succeed).
56    pub fn from_der(der: &[u8]) -> Result<Self, String> {
57        let (_, cert) =
58            parse_x509_certificate(der).map_err(|e| format!("parse client cert DER: {e}"))?;
59
60        let mut sans = Vec::new();
61        if let Ok(Some(ext)) = cert.subject_alternative_name() {
62            for name in &ext.value.general_names {
63                match name {
64                    GeneralName::DNSName(s) => sans.push(format!("DNS:{s}")),
65                    GeneralName::RFC822Name(s) => sans.push(format!("email:{s}")),
66                    GeneralName::URI(s) => sans.push(format!("URI:{s}")),
67                    GeneralName::IPAddress(b) => sans.push(format!("IP:{}", fmt_ip(b))),
68                    _ => {}
69                }
70            }
71        }
72
73        // The `.raw` field of SubjectPublicKeyInfo is the DER of the full SPKI
74        // structure (RFC 5280 §4.1) — the standard input for an SPKI fingerprint.
75        let spki_sha256 = hex::encode(Sha256::digest(cert.public_key().raw));
76
77        Ok(Self {
78            der: der.to_vec(),
79            sans,
80            spki_sha256,
81        })
82    }
83}

85/// Best-effort textual rendering of a SAN IP address (4-byte v4 / 16-byte v6).
86fn fmt_ip(bytes: &[u8]) -> String {
87    match bytes.len() {
88        4 => bytes
89            .iter()
90            .map(|b| b.to_string())
91            .collect::<Vec<_>>()
92            .join("."),
93        16 => bytes
94            .chunks(2)
95            .map(|c| format!("{:02x}{:02x}", c[0], c[1]))
96            .collect::<Vec<_>>()
97            .join(":"),
98        _ => hex::encode(bytes),
99    }
100}

102/// Authenticated device context extracted from a client certificate.
103///
104/// Use this as a handler parameter to require certificate-based auth.
105#[derive(Debug, Clone)]
106pub struct CertificateAuthenticated {
107    pub service_account_id: Uuid,
108    pub tenant_id: Uuid,
109    pub certificate_id: Uuid,
110}
111
112impl CertificateAuthenticated {
113    /// Extract and validate the client certificate from the request.
114    ///
115    /// This is called manually from the handler rather than via
116    /// `FromRequest`, because the concrete `DeviceAuthService` type
117    /// depends on the DB connection generic `C`.
118    pub async fn extract<C: Connection + Clone>(req: &HttpRequest) -> Result<Self, AxiamApiError> {
119        let state = req
120            .app_data::<web::Data<AppState<C>>>()
121            .ok_or(AxiamError::Internal("missing AppState".into()))?;
122        let service = &state.pki.device_auth_service;
123
124        // Prefer the VERIFIED client certificate captured at TLS handshake time
125        // (D3 native mTLS): rustls has already checked it against the client-CA
126        // bundle, so it is authoritative and cannot be spoofed by a header. Only
127        // fall back to the `X-Client-Certificate` proxy header when TLS was
128        // terminated upstream (no verified cert on this connection) AND the
129        // operator has said that upstream is trusted to set the header.
130        let identity_result = if let Some(verified) = req.conn_data::<VerifiedClientCert>() {
131            service.authenticate_der(&verified.der).await
132        } else {
133            // A certificate is public data, and every check on the header path
134            // — fingerprint, status, expiry, chain to the CA — is satisfied by a
135            // *copy* of an enrolled device's certificate. Only the TLS handshake
136            // proves possession of the private key, and on this path there was
137            // no handshake to prove it in. So the header is worth exactly as
138            // much as the guarantee that a client could not have set it, and
139            // that guarantee is a property of the deployment, not of the
140            // request. Off by default; see
141            // `axiam_auth::AuthConfig::trust_forwarded_client_cert`.
142            if !state.auth_config.trust_forwarded_client_cert {
143                return Err(AxiamError::AuthenticationFailed {
144                    reason: "client certificate authentication requires a TLS-verified \
145                             peer certificate on this connection. The X-Client-Certificate \
146                             header is not trusted unless \
147                             AXIAM__AUTH__TRUST_FORWARDED_CLIENT_CERT is set, which is \
148                             only safe when a proxy terminates mTLS and overwrites that \
149                             header on every request"
150                        .into(),
151                }
152                .into());
153            }
154            let header = req
155                .headers()
156                .get("X-Client-Certificate")
157                .and_then(|v| v.to_str().ok())
158                .ok_or(AxiamError::AuthenticationFailed {
159                    reason: "missing client certificate (no native mTLS peer cert and no \
160                             X-Client-Certificate header)"
161                        .into(),
162                })?;
163
164            // URL-decode the PEM (reverse proxy URL-encodes it)
165            let pem = urldecode(header).map_err(|_| AxiamError::AuthenticationFailed {
166                reason: "invalid URL encoding in X-Client-Certificate".into(),
167            })?;
168
169            service.authenticate(&pem).await
170        };
171
172        let identity: DeviceIdentity = identity_result.map_err(|e| match &e {
173            // Map certificate/NotFound errors to proper 401/403 status codes
174            AxiamError::Certificate(msg) if msg.contains("not bound to a service account") => {
175                AxiamError::AuthorizationDenied {
176                    reason: msg.clone(),
177                    action: None,
178                    resource_id: None,
179                }
180            }
181            AxiamError::Certificate(msg) => AxiamError::AuthenticationFailed {
182                reason: msg.clone(),
183            },
184            AxiamError::NotFound { .. } => AxiamError::AuthenticationFailed {
185                reason: "unknown client certificate".into(),
186            },
187            _ => e,
188        })?;
189
190        Ok(CertificateAuthenticated {
191            service_account_id: identity.service_account_id,
192            tenant_id: identity.tenant_id,
193            certificate_id: identity.certificate_id,
194        })
195    }
196}

198/// Simple percent-decoding for the `X-Client-Certificate` header.
199fn urldecode(input: &str) -> Result<String, ()> {
200    let mut result = Vec::with_capacity(input.len());
201    let mut bytes = input.bytes();
202    while let Some(b) = bytes.next() {
203        if b == b'%' {
204            let hi = bytes.next().ok_or(())?;
205            let lo = bytes.next().ok_or(())?;
206            let hi = hex_val(hi)?;
207            let lo = hex_val(lo)?;
208            result.push((hi << 4) | lo);
209        } else {
210            result.push(b);
211        }
212    }
213    String::from_utf8(result).map_err(|_| ())
214}
215
216fn hex_val(b: u8) -> Result<u8, ()> {
217    match b {
218        b'0'..=b'9' => Ok(b - b'0'),
219        b'a'..=b'f' => Ok(b - b'a' + 10),
220        b'A'..=b'F' => Ok(b - b'A' + 10),
221        _ => Err(()),
222    }
223}

229    // -----------------------------------------------------------------------
230    // urldecode / hex_val (R4) — these private helpers are only reachable
231    // indirectly through the HTTP `X-Client-Certificate` header path in
232    // integration tests, which only exercise the happy path plus one
233    // malformed-input case; test the pure logic directly here.

235
236    #[test]
237    fn urldecode_plain_string_round_trips() {
238        assert_eq!(urldecode("hello-world_1.2~3").unwrap(), "hello-world_1.2~3");
239    }
240
241    #[test]
242    fn urldecode_decodes_percent_sequences() {
243        // "%0A" -> newline, "%2B" -> '+'.
244        assert_eq!(urldecode("a%0Ab%2Bc").unwrap(), "a\nb+c");
245    }
246
247    #[test]
248    fn urldecode_rejects_truncated_percent_sequence() {
249        // Only one hex digit follows '%' before the string ends.
250        assert!(urldecode("abc%2").is_err());
251    }
252
253    #[test]
254    fn urldecode_rejects_invalid_hex_digit() {
255        assert!(urldecode("abc%zz").is_err());
256        assert!(urldecode("abc%2z").is_err());
257    }
258
259    #[test]
260    fn hex_val_accepts_all_case_variants() {
261        assert_eq!(hex_val(b'0').unwrap(), 0);
262        assert_eq!(hex_val(b'9').unwrap(), 9);
263        assert_eq!(hex_val(b'a').unwrap(), 10);
264        assert_eq!(hex_val(b'f').unwrap(), 15);
265        assert_eq!(hex_val(b'A').unwrap(), 10);
266        assert_eq!(hex_val(b'F').unwrap(), 15);
267    }
268
269    #[test]
270    fn hex_val_rejects_non_hex_byte() {
271        assert!(hex_val(b'g').is_err());
272        assert!(hex_val(b'Z').is_err());
273        assert!(hex_val(b' ').is_err());
274    }

306    #[test]
307    fn from_der_parses_dns_san_and_spki_fingerprint() {
308        let cert = rcgen::generate_simple_self_signed(vec!["device.example.com".to_string()])
309            .expect("generate self-signed cert");
310        let der = cert.cert.der().to_vec();
311
312        let parsed = VerifiedClientCert::from_der(&der).expect("parse DER");
313        assert!(
314            parsed.sans.iter().any(|s| s == "DNS:device.example.com"),
315            "expected a DNS SAN entry, got: {:?}",
316            parsed.sans
317        );
318        assert_eq!(
319            parsed.spki_sha256.len(),
320            64,
321            "SPKI fingerprint must be a 32-byte hex string (64 hex chars)"
322        );
323        assert!(
324            parsed.spki_sha256.chars().all(|c| c.is_ascii_hexdigit()),
325            "SPKI fingerprint must be lowercase hex"
326        );
327        assert_eq!(parsed.der, der);
328    }
