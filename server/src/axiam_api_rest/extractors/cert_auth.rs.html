1//! Certificate-based authentication extractor for mTLS device auth.
2//!
3//! Two identity sources are supported, in order of trust:
4//!
5//! 1. **Native mTLS (D3):** when the server terminates TLS in-process with
6//!    client-cert auth enabled (`AXIAM__SERVER__TLS__CLIENT_AUTH`), rustls
7//!    cryptographically verifies the client certificate against the configured
8//!    CA bundle *during the handshake*. `axiam-server`'s `on_connect` hook then
9//!    stores the verified leaf certificate as a [`VerifiedClientCert`] in the
10//!    connection extensions. This extractor consumes that verified certificate —
11//!    it cannot be forged by a request header.
12//! 2. **Legacy proxy path:** the `X-Client-Certificate` header (URL-encoded PEM
13//!    forwarded by a TLS-terminating reverse proxy). Used only when no verified
14//!    certificate is present on the connection (i.e. TLS terminated upstream).
15//!
16//! Both paths validate the certificate via [`DeviceAuthService`].
17

31/// A client certificate that rustls **verified** against the configured client
32/// CA bundle during the TLS 1.3 handshake (D3).
33///
34/// `axiam-server`'s `HttpServer::on_connect` hook builds this from the rustls
35/// connection's `peer_certificates()` and inserts it into the per-connection
36/// [`actix_web::dev::Extensions`]; handlers read it back with
37/// `HttpRequest::conn_data::<VerifiedClientCert>()`. Because it originates from
38/// the verified peer chain — not a header — it is a trusted identity assertion.
39#[derive(Debug, Clone)]
40pub struct VerifiedClientCert {
41    /// What the TLS layer was able to say about this certificate: whether it
42    /// chained to a configured trust anchor, or was accepted self-asserted
43    /// under [`ClientAuth::OptionalSelfSigned`][cauth] (RFC 8705 §2.2).
44    ///
45    /// Every consumer must decide what it does with a
46    /// [`CertTrust::SelfAsserted`] certificate. Device authentication, in this
47    /// very module, refuses it.
48    ///
49    /// [cauth]: crate::config::ClientAuth::OptionalSelfSigned
50    pub trust: CertTrust,
51    /// DER encoding of the verified leaf certificate.
52    pub der: Vec<u8>,
53    /// Subject Alternative Names (DNS/URI/RFC822/IP) parsed from the leaf, in
54    /// certificate order. Empty if the leaf carries no SAN extension.
55    pub sans: Vec<String>,
56    /// Lowercase hex SHA-256 of the leaf's SubjectPublicKeyInfo (the SPKI
57    /// fingerprint) — a stable key-identity handle for cert-mapped identities.
58    pub spki_sha256: String,
59}
60
61impl VerifiedClientCert {
62    /// Parse SAN entries and the SPKI fingerprint from a DER-encoded leaf
63    /// certificate. Returns an error string only if the DER cannot be parsed as
64    /// an X.509 certificate (rustls has already accepted the same bytes by this
65    /// point, so this parse is expected to succeed).
66    ///
67    /// `trust` is not inferred, and there is no default: it is the caller's
68    /// statement of what the *handshake* established, and the only caller in a
69    /// position to know is `axiam-server`'s `on_connect` hook, which asks
70    /// `axiam_server::tls::peer_certificate_trust`. A defaulted parameter here
71    /// would have made [`CertTrust::ChainedToAnchor`] — the privileged value —
72    /// what a future call site gets by saying nothing.
73    pub fn from_der(der: &[u8], trust: CertTrust) -> Result<Self, String> {
74        let (_, cert) =
75            parse_x509_certificate(der).map_err(|e| format!("parse client cert DER: {e}"))?;
76
77        let mut sans = Vec::new();
78        if let Ok(Some(ext)) = cert.subject_alternative_name() {
79            for name in &ext.value.general_names {
80                match name {
81                    GeneralName::DNSName(s) => sans.push(format!("DNS:{s}")),
82                    GeneralName::RFC822Name(s) => sans.push(format!("email:{s}")),
83                    GeneralName::URI(s) => sans.push(format!("URI:{s}")),
84                    GeneralName::IPAddress(b) => sans.push(format!("IP:{}", fmt_ip(b))),
85                    _ => {}
86                }
87            }
88        }
89
90        // The `.raw` field of SubjectPublicKeyInfo is the DER of the full SPKI
91        // structure (RFC 5280 §4.1) — the standard input for an SPKI fingerprint.
92        let spki_sha256 = hex::encode(Sha256::digest(cert.public_key().raw));
93
94        Ok(Self {
95            trust,
96            der: der.to_vec(),
97            sans,
98            spki_sha256,
99        })
100    }
101
102    /// Refuse this certificate for device/IoT authentication unless it chained
103    /// to a configured mTLS trust anchor.
104    ///
105    /// # Why this exists at all (B-06)
106    ///
107    /// Device authentication's entire trust model is that the certificate was
108    /// issued by a CA an administrator flagged as an `mtls_trust_anchor`; the
109    /// lookup it performs resolves the certificate against the tenant's issued
110    /// certificates. A [`CertTrust::SelfAsserted`] certificate carries no such
111    /// issuance — it is admitted to the handshake only under
112    /// [`ClientAuth::OptionalSelfSigned`][cauth], and only so that an RFC 8705
113    /// §2.2 OAuth2 client can present a credential whose SHA-256 an
114    /// administrator registered. Letting one through here would be the
115    /// native-listener twin of **B-06**, where a certificate under a
116    /// never-flagged CA authenticated through the proxy header, reopened on the
117    /// listener that was supposed to be the trustworthy one.
118    ///
119    /// # Why it is a separate method
120    ///
121    /// [`CertificateAuthenticated::extract`] needs an `AppState<C>` and a live
122    /// `DeviceAuthService`, so it is reachable only from an integration test —
123    /// and the native-mTLS branch is not reachable even from there, because
124    /// `actix_web::test` performs no TLS handshake and offers no way to
125    /// populate the connection extensions this branch reads. Pulling the
126    /// decision out gives it somewhere to be tested at all.
127    ///
128    /// [cauth]: crate::config::ClientAuth::OptionalSelfSigned
129    ///
130    /// # Errors
131    ///
132    /// [`AxiamError::AuthenticationFailed`] for a self-asserted certificate.
133    /// The message names the actual problem — no trusted issuer — rather than
134    /// falling through to the proxy-header branch's message, which would tell
135    /// an operator to go and set `TRUST_FORWARDED_CLIENT_CERT`: advice that
136    /// would not help here and would, if taken, make the deployment worse.
137    pub fn check_usable_for_device_auth(&self) -> Result<(), AxiamError> {
138        if self.trust.is_chained_to_anchor() {
139            return Ok(());
140        }
141        tracing::warn!(
142            "refusing device certificate authentication: the peer certificate chains to \
143             no configured mTLS trust anchor. Self-asserted certificates are accepted by \
144             the listener only for RFC 8705 §2.2 OAuth2 client authentication \
145             (self_signed_tls_client_auth)"
146        );
147        Err(AxiamError::AuthenticationFailed {
148            reason: "client certificate authentication requires a certificate issued by a \
149                     trusted mTLS certificate authority"
150                .into(),
151        })
152    }
153}

155/// Best-effort textual rendering of a SAN IP address (4-byte v4 / 16-byte v6).
156fn fmt_ip(bytes: &[u8]) -> String {
157    match bytes.len() {
158        4 => bytes
159            .iter()
160            .map(|b| b.to_string())
161            .collect::<Vec<_>>()
162            .join("."),
163        16 => bytes
164            .chunks(2)
165            .map(|c| format!("{:02x}{:02x}", c[0], c[1]))
166            .collect::<Vec<_>>()
167            .join(":"),
168        _ => hex::encode(bytes),
169    }
170}

172/// Authenticated device context extracted from a client certificate.
173///
174/// Use this as a handler parameter to require certificate-based auth.
175#[derive(Debug, Clone)]
176pub struct CertificateAuthenticated {
177    pub service_account_id: Uuid,
178    pub tenant_id: Uuid,
179    pub certificate_id: Uuid,
180    /// RFC 8705 §3.1 `x5t#S256` of the certificate this request presented —
181    /// **only when rustls verified it on this connection** (S-3).
182    ///
183    /// `Some` on the native mTLS path, where the thumbprint is taken over the
184    /// DER bytes of a chain the handshake already validated. `None` on the
185    /// trusted-proxy header path, and that asymmetry is deliberate rather than
186    /// an omission.
187    ///
188    /// A `cnf` claim is only worth minting where it can be *checked* again, and
189    /// the check is `enforce_sender_constraint`, which reads
190    /// [`VerifiedClientCert`] off the connection and nothing else — the
191    /// `X-Client-Certificate` header is refused there by construction, because
192    /// a value a client could set would make the whole mechanism decorative.
193    /// So on a deployment whose TLS terminates upstream, the certificate is
194    /// present at login and absent at every later request. Binding a token
195    /// there would mint a credential that AXIAM itself would then refuse on
196    /// its first use.
197    ///
198    /// The consequence is worth stating plainly: a deployment that terminates
199    /// mTLS at a proxy gets device tokens that are bearer credentials, exactly
200    /// as before this change. Moving that boundary is a deployment decision —
201    /// terminate mTLS at AXIAM — not something a claim can paper over.
202    pub certificate_thumbprint: Option<String>,
203}
204
205impl CertificateAuthenticated {
206    /// Extract and validate the client certificate from the request.
207    ///
208    /// This is called manually from the handler rather than via
209    /// `FromRequest`, because the concrete `DeviceAuthService` type
210    /// depends on the DB connection generic `C`.
211    pub async fn extract<C: Connection + Clone>(req: &HttpRequest) -> Result<Self, AxiamApiError> {
212        let state = req
213            .app_data::<web::Data<AppState<C>>>()
214            .ok_or(AxiamError::Internal("missing AppState".into()))?;
215        let service = &state.pki.device_auth_service;
216
217        // Prefer the VERIFIED client certificate captured at TLS handshake time
218        // (D3 native mTLS): rustls has already checked it against the client-CA
219        // bundle, so it is authoritative and cannot be spoofed by a header. Only
220        // fall back to the `X-Client-Certificate` proxy header when TLS was
221        // terminated upstream (no verified cert on this connection) AND the
222        // operator has said that upstream is trusted to set the header.
223        // Recorded here and nowhere else: the one place that knows the
224        // certificate came off a handshake rather than out of a header. See the
225        // field's own documentation for why the header path gets `None`.
226        let mut certificate_thumbprint = None;
227
228        let identity_result = if let Some(verified) = req.conn_data::<VerifiedClientCert>() {
229            // A certificate that chains to nothing must not authenticate a
230            // device, however well-formed it is and whatever it says about
231            // itself. See `check_usable_for_device_auth` for why (B-06).
232            verified.check_usable_for_device_auth()?;
233            certificate_thumbprint = Some(axiam_oauth2::mtls::thumbprint_s256(&verified.der));
234            service.authenticate_der(&verified.der).await
235        } else {
236            // A certificate is public data, and every check on the header path
237            // — fingerprint, status, expiry, chain to the CA — is satisfied by a
238            // *copy* of an enrolled device's certificate. Only the TLS handshake
239            // proves possession of the private key, and on this path there was
240            // no handshake to prove it in. So the header is worth exactly as
241            // much as the guarantee that a client could not have set it, and
242            // that guarantee is a property of the deployment, not of the
243            // request. Off by default; see
244            // `axiam_auth::AuthConfig::trust_forwarded_client_cert`.
245            if !state.auth_config.trust_forwarded_client_cert {
246                return Err(AxiamError::AuthenticationFailed {
247                    reason: "client certificate authentication requires a TLS-verified \
248                             peer certificate on this connection. The X-Client-Certificate \
249                             header is not trusted unless \
250                             AXIAM__AUTH__TRUST_FORWARDED_CLIENT_CERT is set, which is \
251                             only safe when a proxy terminates mTLS and overwrites that \
252                             header on every request"
253                        .into(),
254                }
255                .into());
256            }
257            let header = req
258                .headers()
259                .get("X-Client-Certificate")
260                .and_then(|v| v.to_str().ok())
261                .ok_or(AxiamError::AuthenticationFailed {
262                    reason: "missing client certificate (no native mTLS peer cert and no \
263                             X-Client-Certificate header)"
264                        .into(),
265                })?;
266
267            // URL-decode the PEM (reverse proxy URL-encodes it)
268            let pem = urldecode(header).map_err(|_| AxiamError::AuthenticationFailed {
269                reason: "invalid URL encoding in X-Client-Certificate".into(),
270            })?;
271
272            service.authenticate(&pem).await
273        };
274
275        // Every refusal on this path is a **401** (S-4 / DF-027).
276        //
277        // There used to be one exception: a certificate bound to no service
278        // account was matched by the text of its error message and turned into
279        // `AuthorizationDenied`, a 403. Two things were wrong with it.
280        //
281        // A 403 asserts an identity and then refuses what it may do. A
282        // certificate bound to no principal identifies **nobody** — there is no
283        // subject to have permissions, and the correct answer to "who are you"
284        // is "unauthenticated", which is what its three siblings here already
285        // said. (The counter-argument, that 403 hides "unknown certificate"
286        // from "known but unbound", does not survive the change: both are 401
287        // now and the bodies were always distinct messages, so nothing is newly
288        // disclosed.)
289        //
290        // And it matched on a string. `axiam-pki` owns that message and could
291        // reword it in a refactor that never mentions HTTP; the status code
292        // would have moved with it, silently, in another crate. A status that
293        // depends on the wording of a message in a lower layer is a status
294        // nobody can change safely.
295        let identity: DeviceIdentity = identity_result.map_err(|e| match &e {
296            AxiamError::Certificate(msg) => AxiamError::AuthenticationFailed {
297                reason: msg.clone(),
298            },
299            AxiamError::NotFound { .. } => AxiamError::AuthenticationFailed {
300                reason: "unknown client certificate".into(),
301            },
302            _ => e,
303        })?;
304
305        Ok(CertificateAuthenticated {
306            service_account_id: identity.service_account_id,
307            tenant_id: identity.tenant_id,
308            certificate_id: identity.certificate_id,
309            certificate_thumbprint,
310        })
311    }
312}

314/// Simple percent-decoding for the `X-Client-Certificate` header.
315fn urldecode(input: &str) -> Result<String, ()> {
316    let mut result = Vec::with_capacity(input.len());
317    let mut bytes = input.bytes();
318    while let Some(b) = bytes.next() {
319        if b == b'%' {
320            let hi = bytes.next().ok_or(())?;
321            let lo = bytes.next().ok_or(())?;
322            let hi = hex_val(hi)?;
323            let lo = hex_val(lo)?;
324            result.push((hi << 4) | lo);
325        } else {
326            result.push(b);
327        }
328    }
329    String::from_utf8(result).map_err(|_| ())
330}
331
332fn hex_val(b: u8) -> Result<u8, ()> {
333    match b {
334        b'0'..=b'9' => Ok(b - b'0'),
335        b'a'..=b'f' => Ok(b - b'a' + 10),
336        b'A'..=b'F' => Ok(b - b'A' + 10),
337        _ => Err(()),
338    }
339}

454    // -----------------------------------------------------------------------
455    // The B-06 guard: device authentication and self-asserted certificates.
456    //
457    // `client_auth = optional_self_signed` lets a certificate that chains to
458    // nothing complete a TLS handshake, so that an RFC 8705 §2.2 OAuth2 client
459    // can present the credential its registration is built around. Device
460    // authentication must be unaffected by that: its model is issuance by a
461    // flagged mTLS trust anchor, and a self-signed certificate has no issuer
462    // anybody flagged.
463    // -----------------------------------------------------------------------
464
465    fn device_cert(trust: CertTrust) -> VerifiedClientCert {
466        let cert = rcgen::generate_simple_self_signed(vec!["device.example.com".to_string()])
467            .expect("generate self-signed cert");
468        VerifiedClientCert::from_der(cert.cert.der().as_ref(), trust).expect("parse DER")
469    }

471    /// A self-asserted certificate cannot authenticate a device; a chained one
472    /// can.
473    ///
474    /// The certificates in both halves are generated the same way, so the only
475    /// difference between "refused" and "permitted" is what the TLS layer said
476    /// about the connection — which is the contract.
477    #[test]
478    fn device_auth_refuses_a_self_asserted_certificate() {
479        assert!(
480            device_cert(CertTrust::ChainedToAnchor)
481                .check_usable_for_device_auth()
482                .is_ok(),
483            "control: a certificate that chained to a configured anchor is device auth's \
484             normal case and must still pass"
485        );
486
487        let err = device_cert(CertTrust::SelfAsserted)
488            .check_usable_for_device_auth()
489            .expect_err("a certificate that chains to nothing must not authenticate a device");
490        assert!(
491            matches!(err, AxiamError::AuthenticationFailed { .. }),
492            "must fail authentication (401), not authorization or anything softer; got {err:?}"
493        );
494    }

496    /// The refusal must not send an operator to the wrong knob.
497    ///
498    /// Falling through to the proxy-header branch would have produced its error
499    /// text, which tells an operator to set
500    /// `AXIAM__AUTH__TRUST_FORWARDED_CLIENT_CERT`. That would not fix this —
501    /// there is no forwarded header on a native mTLS connection — and an
502    /// operator who took the advice would have widened a *different* trust
503    /// boundary while chasing this one. So the message is asserted, not just
504    /// the variant.
505    #[test]
506    fn the_device_auth_refusal_names_the_issuer_problem_not_the_header_knob() {
507        let AxiamError::AuthenticationFailed { reason } = device_cert(CertTrust::SelfAsserted)
508            .check_usable_for_device_auth()
509            .expect_err("must be refused")
510        else {
511            panic!("expected AuthenticationFailed");
512        };
513        assert!(
514            !reason.contains("TRUST_FORWARDED_CLIENT_CERT")
515                && !reason.contains("X-Client-Certificate"),
516            "the refusal must not point at the proxy-header configuration; got {reason:?}"
517        );
518        assert!(
519            reason.contains("certificate authority"),
520            "the refusal must say what is actually wrong — no trusted issuer; got {reason:?}"
521        );
522    }
