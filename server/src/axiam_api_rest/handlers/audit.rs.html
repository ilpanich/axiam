12/// `GET /api/v1/audit-logs`
13#[utoipa::path(
14    get,
15    path = "/api/v1/audit-logs",
16    tag = "audit",
17    params(Pagination, AuditLogFilter),
18    responses(
19        (status = 200, description = "Paginated audit log entries",
20         body = inline(PaginatedResult<AuditLogEntry>)),
21    ),
22    security(("bearer" = []))
23)]
24pub async fn list<C: Connection + Clone>(
25    user: AuthenticatedUser,
26    authz: AuthzData,
27    state: web::Data<AppState<C>>,
28    pagination: web::Query<Pagination>,
29    filter: web::Query<AuditLogFilter>,
30) -> Result<HttpResponse, AxiamApiError> {
31    // Self-service: always allowed. Backend already filters by tenant_id.
32    // Admins with audit_logs:list can see all entries; others see only their own.
33    let authz_result = RequirePermission::new("audit_logs:list", Uuid::nil())
34        .check(&user, authz.get_ref().as_ref())
35        .await;
36    let mut f = filter.into_inner();
37    if authz_result.is_err() {
38        // Not an admin — restrict to the caller's own entries via actor_id.
39        f.actor_id = Some(user.user_id);
40    }
41    let result = state
42        .audit_repo
43        .list(user.tenant_id, f, pagination.into_inner())
44        .await?;
45    Ok(HttpResponse::Ok().json(result))
46}

48/// `GET /api/v1/audit-logs/system`
49///
50/// Returns audit entries for unauthenticated/system requests (nil tenant_id).
51///
52/// Of the caller's organization only (SA-02, N1): the partition is
53/// deployment-wide, and this returns the entries whose recorded
54/// `organization_id` is the caller's own. The server records one on a new
55/// entry only where it resolved it with certainty — a deleted tenant's
56/// organization, a verified SCIM token's tenant's, an organization or tenant
57/// the request's path names and that exists. An entry with none, every entry
58/// written before the field existed among them, is returned to nobody.
59///
60/// Organization-level: requires `audit_logs:list_system` and a caller whose
61/// own record lives in the organization's scope tenant and whose role
62/// assignments are not restricted to particular tenants. A tenant
63/// administrator is refused `403` whatever its role holds, and so is an
64/// organization-level principal whose roles name particular tenants: the
65/// partition belongs to the organization as a whole.
66#[utoipa::path(
67    get,
68    path = "/api/v1/audit-logs/system",
69    tag = "audit",
70    params(Pagination, AuditLogFilter),
71    responses(
72        (status = 200, description = "Paginated system audit log entries of the caller's \
73                                      organization",
74         body = inline(PaginatedResult<AuditLogEntry>)),
75        (status = 403, description = "The caller is not an organization-level principal, is \
76                                      one whose role assignments are restricted to particular \
77                                      tenants, or lacks `audit_logs:list_system`"),
78    ),
79    security(("bearer" = []))
80)]
81pub async fn list_system<C: Connection + Clone>(
82    user: AuthenticatedUser,
83    authz: AuthzData,
84    state: web::Data<AppState<C>>,
85    pagination: web::Query<Pagination>,
86    filter: web::Query<AuditLogFilter>,
87) -> Result<HttpResponse, AxiamApiError> {
88    RequirePermission::new("audit_logs:list_system", Uuid::nil())
89        .check(&user, authz.get_ref().as_ref())
90        .await?;
91    // SA-02 (N1): the nil-tenant partition is deployment-wide — it holds the
92    // unattributed request rows (client IP, method and path) of every tenant,
93    // other organizations' included. Reading it is an organization-level act,
94    // never a tenant administrator's, whatever its role carries. See
95    // `handlers::org_scope`.
96    //
97    // And only the rows of the caller's own organization (review Q-3/S-2):
98    // the organization of the caller's own record, as the datastore has it —
99    // read from the scope tenant the guard has just loaded (QR2-10).
100    let organization_id =
101        crate::handlers::org_scope::require_organization_principal(&user, state.get_ref())
102            .await?
103            .organization_id;
104    let result = state
105        .audit_repo
106        .list_system(
107            organization_id,
108            filter.into_inner(),
109            pagination.into_inner(),
110        )
111        .await?;
112    Ok(HttpResponse::Ok().json(result))
113}
