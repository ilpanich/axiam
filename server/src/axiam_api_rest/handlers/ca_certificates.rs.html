21#[derive(Debug, Deserialize, utoipa::ToSchema)]
22pub struct CreateCaCertificateRequest {
23    pub subject: String,
24    pub key_algorithm: KeyAlgorithm,
25    /// Validity duration in days.
26    pub validity_days: u32,
27    /// Common name for the signing intermediate — `vault_pki` custody only.
28    ///
29    /// Under that custodian Vault generates a root and an intermediate beneath
30    /// it, and this names the second. Defaults to the root's subject with
31    /// `Intermediate Authority` appended. Ignored by every other custodian,
32    /// which produces one self-signed CA and has no second certificate to name.
33    #[serde(default)]
34    pub intermediate_subject: Option<String>,
35    /// Validity of the signing intermediate, in days. Defaults to the root's.
36    #[serde(default)]
37    pub intermediate_validity_days: Option<u32>,
38    /// Issue leaves straight from the generated root instead of creating an
39    /// intermediate. `vault_pki` custody only, and off by default.
40    ///
41    /// The default is the safer one: a root that signs only an intermediate can
42    /// have that intermediate revoked and replaced without redistributing the
43    /// trust anchor, and a root that signs leaves cannot.
44    #[serde(default)]
45    pub issue_from_root: bool,
46}

48/// `POST /api/v1/organizations/{org_id}/ca-certificates`
49///
50/// Generate a CA. What that means depends on the configured key custodian, and
51/// deliberately not on the request:
52///
53/// * Under `database` or `vault` custody, AXIAM generates a self-signed root
54///   and returns its private key once, in `private_key_pem`.
55/// * Under `vault_pki` custody, Vault generates a root and a signing
56///   intermediate beneath it, and the response has **no** `private_key_pem`
57///   because the key never existed outside Vault. The root's certificate comes
58///   back in `chain_pem`, which is the only copy anything outside Vault will
59///   see.
60#[utoipa::path(
61    post,
62    path = "/api/v1/organizations/{org_id}/ca-certificates",
63    tag = "ca-certificates",
64    params(("org_id" = Uuid, Path, description = "Organization ID")),
65    request_body = CreateCaCertificateRequest,
66    responses(
67        (status = 201, description = "CA certificate generated",
68         body = GeneratedCaCertificate),
69    ),
70    security(("bearer" = []))
71)]
72pub async fn generate<C: Connection + Clone>(
73    user: AuthenticatedUser,
74    authz: AuthzData,
75    path: web::Path<Uuid>,
76    state: web::Data<AppState<C>>,
77    body: web::Json<CreateCaCertificateRequest>,
78) -> Result<HttpResponse, AxiamApiError> {
79    RequirePermission::new("ca_certificates:generate", Uuid::nil())
80        .check(&user, authz.get_ref().as_ref())
81        .await?;
82    let org_id = path.into_inner();
83
84    // Authorization: only allow access to certificates in the caller's own org.
85    if org_id != user.org_id {
86        return Err(AxiamApiError(
87            axiam_core::error::AxiamError::AuthorizationDenied {
88                reason: "cannot access a different organization".into(),
89                action: None,
90                resource_id: None,
91            },
92        ));
93    }
94
95    let req = body.into_inner();
96    let input = CreateCaCertificate {
97        organization_id: org_id,
98        subject: req.subject,
99        key_algorithm: req.key_algorithm,
100        validity_days: req.validity_days,
101        // An empty string is not a subject; it would produce an intermediate
102        // with a blank CN rather than the documented default.
103        intermediate_subject: req.intermediate_subject.filter(|s| !s.trim().is_empty()),
104        intermediate_validity_days: req.intermediate_validity_days,
105        issue_from_root: req.issue_from_root,
106    };
107    let result = state.pki.ca_service.generate(input).await?;
108    Ok(HttpResponse::Created().json(result))
109}

111/// Body of `POST /api/v1/organizations/{org_id}/ca-certificates/import`.
112///
113/// Deliberately carries no subject, validity window or key algorithm: all
114/// three are read out of the certificate itself. A caller that could name them
115/// separately could name a subject the certificate does not have, and AXIAM
116/// would enforce the claim while every relying party read the certificate.
117#[derive(Deserialize, utoipa::ToSchema)]
118pub struct ImportCaCertificateRequest {
119    /// PEM-encoded CA certificate.
120    pub public_cert_pem: String,
121    /// PEM-encoded private key, if AXIAM is to take custody of it.
122    ///
123    /// Omit it to register the certificate as a trust anchor only. Write-only:
124    /// it goes to the configured custodian on the way in and no endpoint
125    /// returns it.
126    #[serde(default)]
127    pub private_key_pem: Option<String>,
128}

130/// Manual `Debug` (SECHRD-09 / D-06): actix's error and tracing paths format
131/// extractor payloads, and `#[serde(skip_serializing)]` governs `Serialize`
132/// only.
133impl std::fmt::Debug for ImportCaCertificateRequest {
134    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
135        f.debug_struct("ImportCaCertificateRequest")
136            .field("public_cert_pem", &self.public_cert_pem)
137            .field(
138                "private_key_pem",
139                &self.private_key_pem.as_ref().map(|_| "[REDACTED]"),
140            )
141            .finish()
142    }
143}

145/// `POST /api/v1/organizations/{org_id}/ca-certificates/import`
146///
147/// Register a CA an organization already has, instead of generating one.
148///
149/// The case this exists for: an organization whose root lives offline, in an
150/// HSM, or in an existing internal PKI, and which wants AXIAM in the chain
151/// rather than at the top of it. Without it, the only CA AXIAM could use was
152/// one it generated, which means every AXIAM-issued certificate chains to a
153/// root nothing else in the estate trusts.
154///
155/// With a private key, AXIAM takes custody of it and can issue against the CA.
156/// Where it goes is the configured custodian's business: sealed into the row,
157/// written to Vault's KV engine, or — under `vault_pki` — imported into Vault's
158/// PKI engine as an issuer, after which AXIAM signs through Vault and holds
159/// nothing. That last one is the BYOK counterpart to generating in Vault: the
160/// key does pass through AXIAM's memory on the way in, because AXIAM is what
161/// received the request, but it is never stored here and every later use of it
162/// is Vault's.
163///
164/// Without a private key, the certificate is a trust anchor and
165/// `/api/v1/certificates` says so rather than failing at the point of use.
166///
167/// Gated on `ca_certificates:generate`: importing a CA is the same act as
168/// generating one from every relying party's point of view — it decides what
169/// this organization's certificates chain to.
170#[utoipa::path(
171    post,
172    path = "/api/v1/organizations/{org_id}/ca-certificates/import",
173    tag = "ca-certificates",
174    params(("org_id" = Uuid, Path, description = "Organization ID")),
175    request_body = ImportCaCertificateRequest,
176    responses(
177        (status = 201, description = "CA certificate imported", body = CaCertificate),
178        (status = 400, description = "Not a CA certificate, already expired, or the \
179                                      supplied key does not match it"),
180    ),
181    security(("bearer" = []))
182)]
183pub async fn import<C: Connection + Clone>(
184    user: AuthenticatedUser,
185    authz: AuthzData,
186    path: web::Path<Uuid>,
187    state: web::Data<AppState<C>>,
188    body: web::Json<ImportCaCertificateRequest>,
189) -> Result<HttpResponse, AxiamApiError> {
190    RequirePermission::new("ca_certificates:generate", Uuid::nil())
191        .check(&user, authz.get_ref().as_ref())
192        .await?;
193    let org_id = path.into_inner();
194
195    if org_id != user.org_id {
196        return Err(AxiamApiError(
197            axiam_core::error::AxiamError::AuthorizationDenied {
198                reason: "cannot access a different organization".into(),
199                action: None,
200                resource_id: None,
201            },
202        ));
203    }
204
205    let req = body.into_inner();
206    let certificate = state
207        .pki
208        .ca_service
209        .import(ImportCaCertificate {
210            organization_id: org_id,
211            public_cert_pem: req.public_cert_pem,
212            // An empty string is not a key; treating it as one would send the
213            // custodian something to store and produce a CA that lists as
214            // issuable and fails at the first signature.
215            private_key_pem: req.private_key_pem.filter(|k| !k.trim().is_empty()),
216        })
217        .await?;
218
219    Ok(HttpResponse::Created().json(certificate))
220}
