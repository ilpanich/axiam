22#[derive(Debug, Deserialize, utoipa::ToSchema)]
23pub struct CreateCaCertificateRequest {
24    pub subject: String,
25    pub key_algorithm: KeyAlgorithm,
26    /// Validity duration in days.
27    pub validity_days: u32,
28    /// Common name for the signing intermediate — `vault_pki` custody only.
29    ///
30    /// Under that custodian Vault generates a root and an intermediate beneath
31    /// it, and this names the second. Defaults to the root's subject with
32    /// `Intermediate Authority` appended. Ignored by every other custodian,
33    /// which produces one self-signed CA and has no second certificate to name.
34    #[serde(default)]
35    pub intermediate_subject: Option<String>,
36    /// Validity of the signing intermediate, in days. Defaults to the root's.
37    #[serde(default)]
38    pub intermediate_validity_days: Option<u32>,
39    /// Issue leaves straight from the generated root instead of creating an
40    /// intermediate. `vault_pki` custody only, and off by default.
41    ///
42    /// The default is the safer one: a root that signs only an intermediate can
43    /// have that intermediate revoked and replaced without redistributing the
44    /// trust anchor, and a root that signs leaves cannot.
45    #[serde(default)]
46    pub issue_from_root: bool,
47}

49/// `POST /api/v1/organizations/{org_id}/ca-certificates`
50///
51/// Generate a CA. What that means depends on the configured key custodian, and
52/// deliberately not on the request:
53///
54/// * Under `database` or `vault` custody, AXIAM generates a self-signed root
55///   and returns its private key once, in `private_key_pem`.
56/// * Under `vault_pki` custody, Vault generates a root and a signing
57///   intermediate beneath it, and the response has **no** `private_key_pem`
58///   because the key never existed outside Vault. The root's certificate comes
59///   back in `chain_pem`, which is the only copy anything outside Vault will
60///   see.
61#[utoipa::path(
62    post,
63    path = "/api/v1/organizations/{org_id}/ca-certificates",
64    tag = "ca-certificates",
65    params(("org_id" = Uuid, Path, description = "Organization ID")),
66    request_body = CreateCaCertificateRequest,
67    responses(
68        (status = 201, description = "CA certificate generated",
69         body = GeneratedCaCertificate),
70    ),
71    security(("bearer" = []))
72)]
73pub async fn generate<C: Connection + Clone>(
74    user: AuthenticatedUser,
75    authz: AuthzData,
76    path: web::Path<Uuid>,
77    state: web::Data<AppState<C>>,
78    body: web::Json<CreateCaCertificateRequest>,
79) -> Result<HttpResponse, AxiamApiError> {
80    RequirePermission::new("ca_certificates:generate", Uuid::nil())
81        .check(&user, authz.get_ref().as_ref())
82        .await?;
83    let org_id = path.into_inner();
84
85    // Authorization: only allow access to certificates in the caller's own org.
86    if org_id != user.org_id {
87        return Err(AxiamApiError(
88            axiam_core::error::AxiamError::AuthorizationDenied {
89                reason: "cannot access a different organization".into(),
90                action: None,
91                resource_id: None,
92            },
93        ));
94    }
95
96    let req = body.into_inner();
97    let input = CreateCaCertificate {
98        organization_id: org_id,
99        subject: req.subject,
100        key_algorithm: req.key_algorithm,
101        validity_days: req.validity_days,
102        // An empty string is not a subject; it would produce an intermediate
103        // with a blank CN rather than the documented default.
104        intermediate_subject: req.intermediate_subject.filter(|s| !s.trim().is_empty()),
105        intermediate_validity_days: req.intermediate_validity_days,
106        issue_from_root: req.issue_from_root,
107    };
108    let result = state.pki.ca_service.generate(input).await?;
109    Ok(HttpResponse::Created().json(result))
110}

126    /// returns it.

129}

146/// `POST /api/v1/organizations/{org_id}/ca-certificates/import`
147///
148/// Register a CA an organization already has, instead of generating one.
149///
150/// The case this exists for: an organization whose root lives offline, in an
151/// HSM, or in an existing internal PKI, and which wants AXIAM in the chain
152/// rather than at the top of it. Without it, the only CA AXIAM could use was
153/// one it generated, which means every AXIAM-issued certificate chains to a
154/// root nothing else in the estate trusts.
155///
156/// With a private key, AXIAM takes custody of it and can issue against the CA.
157/// Where it goes is the configured custodian's business: sealed into the row,
158/// written to Vault's KV engine, or — under `vault_pki` — imported into Vault's
159/// PKI engine as an issuer, after which AXIAM signs through Vault and holds
160/// nothing. That last one is the BYOK counterpart to generating in Vault: the
161/// key does pass through AXIAM's memory on the way in, because AXIAM is what
162/// received the request, but it is never stored here and every later use of it
163/// is Vault's.
164///
165/// Without a private key, the certificate is a trust anchor and
166/// `/api/v1/certificates` says so rather than failing at the point of use.
167///
168/// Gated on `ca_certificates:generate`: importing a CA is the same act as
169/// generating one from every relying party's point of view — it decides what
170/// this organization's certificates chain to.
171#[utoipa::path(
172    post,
173    path = "/api/v1/organizations/{org_id}/ca-certificates/import",
174    tag = "ca-certificates",
175    params(("org_id" = Uuid, Path, description = "Organization ID")),
176    request_body = ImportCaCertificateRequest,
177    responses(
178        (status = 201, description = "CA certificate imported", body = CaCertificate),
179        (status = 400, description = "Not a CA certificate, already expired, or the \
180                                      supplied key does not match it"),
181    ),
182    security(("bearer" = []))
183)]
184pub async fn import<C: Connection + Clone>(
185    user: AuthenticatedUser,
186    authz: AuthzData,
187    path: web::Path<Uuid>,
188    state: web::Data<AppState<C>>,
189    body: web::Json<ImportCaCertificateRequest>,
190) -> Result<HttpResponse, AxiamApiError> {
191    RequirePermission::new("ca_certificates:generate", Uuid::nil())
192        .check(&user, authz.get_ref().as_ref())
193        .await?;
194    let org_id = path.into_inner();
195
196    if org_id != user.org_id {
197        return Err(AxiamApiError(
198            axiam_core::error::AxiamError::AuthorizationDenied {
199                reason: "cannot access a different organization".into(),
200                action: None,
201                resource_id: None,
202            },
203        ));
204    }
205
206    let req = body.into_inner();
207    let certificate = state
208        .pki
209        .ca_service
210        .import(ImportCaCertificate {
211            organization_id: org_id,
212            public_cert_pem: req.public_cert_pem,
213            // An empty string is not a key; treating it as one would send the
214            // custodian something to store and produce a CA that lists as
215            // issuable and fails at the first signature.
216            private_key_pem: req.private_key_pem.filter(|k| !k.trim().is_empty()),
217        })
218        .await?;
219
220    Ok(HttpResponse::Created().json(certificate))
221}

354#[derive(Debug, serde::Serialize, utoipa::ToSchema)]

356    /// The CA whose key moved.
357    pub ca_certificate_id: Uuid,
358    /// Where the key was.
359    #[schema(example = "database")]
360    pub previous_custody: String,
361    /// Where it is now.
362    #[schema(example = "vault")]
363    pub key_custody: String,
364    /// Where the new custodian filed it, when the custodian has a path.
365    pub key_locator: Option<String>,
366}

368/// `POST /api/v1/organizations/{org_id}/ca-certificates/{id}/migrate-custody`
369///
370/// Move this CA's signing key to the custodian the deployment is configured for.
371///
372/// Custody is recorded per CA rather than read from configuration, so adopting
373/// Vault does not move the CAs that already exist — their keys stay sealed in
374/// their rows until something moves them. This is that something, and without
375/// it the only route out of database custody is to generate a new CA and
376/// re-issue every leaf beneath it, which for a trust anchor means touching every
377/// relying party.
378///
379/// Takes effect immediately; no restart. The row is the authority on where a key
380/// lives, and the signing path reads it per request.
381#[utoipa::path(
382    post,
383    path = "/api/v1/organizations/{org_id}/ca-certificates/{id}/migrate-custody",
384    tag = "ca-certificates",
385    params(
386        ("org_id" = Uuid, Path, description = "Organization ID"),
387        ("id" = Uuid, Path, description = "CA certificate ID"),
388    ),
389    responses(
390        (status = 200, description = "Key moved to the configured custodian",
391         body = MigrateCustodyResponse),
392        (status = 400, description = "Already there, nothing to move, or a custodian \
393                                      that will not hand its key over"),
394        (status = 403, description = "Different organization"),
395        (status = 404, description = "CA certificate not found"),
396    ),
397    security(("bearer" = []))
398)]
399pub async fn migrate_custody<C: Connection + Clone>(
400    user: AuthenticatedUser,
401    authz: AuthzData,
402    path: web::Path<(Uuid, Uuid)>,
403    state: web::Data<AppState<C>>,
404) -> Result<HttpResponse, AxiamApiError> {
405    RequirePermission::new("ca_certificates:manage", Uuid::nil())
406        .check(&user, authz.get_ref().as_ref())
407        .await?;
408    let (org_id, id) = path.into_inner();
409
410    if org_id != user.org_id {
411        return Err(AxiamApiError(
412            axiam_core::error::AxiamError::AuthorizationDenied {
413                reason: "cannot access a different organization".into(),
414                action: None,
415                resource_id: None,
416            },
417        ));
418    }
419
420    // Read the previous custody before the move, so the response can say what
421    // actually changed rather than only where the key ended up.
422    let before = state.pki.ca_service.get(org_id, id).await?;
423    let updated = state.pki.ca_service.migrate_key_custody(org_id, id).await?;
424
425    Ok(HttpResponse::Ok().json(MigrateCustodyResponse {
426        ca_certificate_id: updated.id,
427        previous_custody: before.key_custody.to_string(),
428        key_custody: updated.key_custody.to_string(),
429        key_locator: updated.key_locator.clone(),
430    }))
431}

444/// The acknowledgement, which is mostly about the restart.

447    /// The CA this is about.
448    pub ca_certificate_id: Uuid,
449    /// The flag as now stored.
450    pub mtls_trust_anchor: bool,
451    /// Always `true`: rustls builds its client trust store once, when the
452    /// listener is constructed, so this takes effect at the next start.
453    pub restart_required: bool,
454    /// A sentence an operator can act on, rather than a bare boolean.
455    pub message: String,

458/// `PUT /api/v1/organizations/{org_id}/ca-certificates/{id}/mtls-trust-anchor`
459///
460/// Offer this CA — or stop offering it — as a trust anchor for mutual TLS.
461///
462/// When enabled, the next server start exports this CA's **public** certificate
463/// into the client-CA bundle and turns on client-certificate verification in
464/// `optional` mode, unless the operator has configured
465/// `AXIAM__SERVER__TLS__CLIENT_AUTH` / `CLIENT_CA_PATH` themselves, which is
466/// never overridden. The signing key is not copied and stays with its custodian.
467///
468/// `restart_required` is always true, and is the point of the response: there is
469/// no supported way to add a root to a rustls listener that is already serving.
470#[utoipa::path(
471    put,
472    path = "/api/v1/organizations/{org_id}/ca-certificates/{id}/mtls-trust-anchor",
473    tag = "ca-certificates",
474    request_body = SetMtlsTrustAnchor,
475    params(
476        ("org_id" = Uuid, Path, description = "Organization ID"),
477        ("id" = Uuid, Path, description = "CA certificate ID"),
478    ),
479    responses(
480        (status = 200, description = "Flag updated; takes effect at the next server start",
481         body = MtlsTrustAnchorResponse),
482        (status = 403, description = "Different organization"),
483        (status = 404, description = "CA certificate not found"),
484    ),
485    security(("bearer" = []))
486)]
487pub async fn set_mtls_trust_anchor<C: Connection + Clone>(
488    user: AuthenticatedUser,
489    authz: AuthzData,
490    path: web::Path<(Uuid, Uuid)>,
491    state: web::Data<AppState<C>>,
492    body: web::Json<SetMtlsTrustAnchor>,
493) -> Result<HttpResponse, AxiamApiError> {
494    // Gated on `ca_certificates:manage` rather than `:read`: this changes what
495    // the deployment's TLS listener will trust, which is a stronger act than
496    // reading a CA and a different one from issuing under it.
497    RequirePermission::new("ca_certificates:manage", Uuid::nil())
498        .check(&user, authz.get_ref().as_ref())
499        .await?;
500    let (org_id, id) = path.into_inner();
501
502    if org_id != user.org_id {
503        return Err(AxiamApiError(
504            axiam_core::error::AxiamError::AuthorizationDenied {
505                reason: "cannot access a different organization".into(),
506                action: None,
507                resource_id: None,
508            },
509        ));
510    }
511
512    let enabled = body.into_inner().enabled;
513    use axiam_core::repository::CaCertificateRepository as _;
514    let updated = state
515        .pki
516        .ca_cert_repo
517        .set_mtls_trust_anchor(org_id, id, enabled)
518        .await?;
519
520    let message = if updated.mtls_trust_anchor {
521        "This CA will be added to the mTLS client trust store when the server \
522         next starts. Clients presenting a certificate it issued will then be \
523         verified by TLS itself."
524            .to_string()
525    } else {
526        "This CA will be removed from the mTLS client trust store when the \
527         server next starts. Certificates it issued will no longer authenticate \
528         a client."
529            .to_string()
530    };
531
532    Ok(HttpResponse::Ok().json(MtlsTrustAnchorResponse {
533        ca_certificate_id: updated.id,
534        mtls_trust_anchor: updated.mtls_trust_anchor,
535        restart_required: true,
536        message,
537    }))
538}

540// ---------------------------------------------------------------------------
541// Tenant signing CAs
542// ---------------------------------------------------------------------------
543//
544// An organization CA is a trust anchor: long-lived, widely distributed, and
545// painful to replace. Issuing every tenant's leaves straight from it means one
546// tenant's compromised issuance is the whole estate's problem, and rotating it
547// is a co-ordinated change at every relying party.
548//
549// A tenant signing CA is the intermediate that fixes that. It is created
550// beneath an organization CA, constrained to a path length of zero, and is what
551// `POST /api/v1/certificates` names as its `issuer_ca_id` for that tenant's
552// users, services and devices. Revoking it revokes exactly one tenant's
553// issuance and touches nothing else.
554//
555// Two ways to get one, and they differ in where the private key is:
556//
557// * **Generate** — AXIAM (or, under `vault_pki`, Vault) makes the key. It goes
558//   to the configured custodian: Vault's PKI engine, Vault's KV engine, or
559//   encrypted into the row. The key is returned once, if there is one to
560//   return, and AXIAM can issue against the CA thereafter.
561// * **Sign a CSR** — the tenant makes the key elsewhere and sends only a
562//   PKCS#10 request. Nothing is returned once, because nothing was generated
563//   here, and the row records custody `External`: AXIAM has issued a
564//   certificate and holds no means of using it.
565
566/// Body of `POST .../tenants/{tenant_id}/signing-cas`.
567#[derive(Debug, Deserialize, utoipa::ToSchema)]
568pub struct CreateIntermediateCaRequest {
569    /// The organization CA that signs it.
570    pub parent_ca_id: Uuid,
571    /// Subject for the signing CA, e.g. `CN=ACME R&D Signing CA`.
572    pub subject: String,
573    pub key_algorithm: KeyAlgorithm,
574    /// Validity duration in days, capped to the parent's own expiry.
575    pub validity_days: u32,
576}
577
578/// Body of `POST .../tenants/{tenant_id}/signing-cas/sign-csr`.
579///
580/// Deliberately carries no key algorithm: it is the CSR's, read out of the
581/// request, because a caller who could state it separately could state one the
582/// key does not have.
583#[derive(Deserialize, utoipa::ToSchema)]
584pub struct SignIntermediateCsrRequest {
585    /// The organization CA that signs it.
586    pub parent_ca_id: Uuid,
587    /// PEM-encoded PKCS#10 certificate signing request.
588    pub csr_pem: String,
589    /// Validity duration in days, capped to the parent's own expiry.
590    pub validity_days: u32,
591}
592
593/// A CSR discloses nothing secret, but it is bulky enough to bury a tracing
594/// span; elided rather than printed.
595impl std::fmt::Debug for SignIntermediateCsrRequest {
596    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
597        f.debug_struct("SignIntermediateCsrRequest")
598            .field("parent_ca_id", &self.parent_ca_id)
599            .field("csr_pem", &format_args!("[{} bytes]", self.csr_pem.len()))
600            .field("validity_days", &self.validity_days)
601            .finish()
602    }
603}

605/// `POST /api/v1/organizations/{org_id}/tenants/{tenant_id}/signing-cas`
606///
607/// Create a tenant signing CA beneath one of the organization's CAs, with the
608/// key generated by whoever the deployment's custodian is. The private key is
609/// returned exactly once — and not at all under `vault_pki`, where it was born
610/// inside Vault and no API exports it.
611#[utoipa::path(
612    post,
613    path = "/api/v1/organizations/{org_id}/tenants/{tenant_id}/signing-cas",
614    tag = "ca-certificates",
615    params(
616        ("org_id" = Uuid, Path, description = "Organization ID"),
617        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
618    ),
619    request_body = CreateIntermediateCaRequest,
620    responses(
621        (status = 201, description = "Tenant signing CA created",
622         body = GeneratedCaCertificate),
623        (status = 400, description = "The parent CA is revoked, expired, holds no key, \
624                                      or is itself a tenant signing CA"),
625    ),
626    security(("bearer" = []))
627)]
628pub async fn generate_intermediate<C: Connection + Clone>(
629    user: AuthenticatedUser,
630    authz: AuthzData,
631    path: web::Path<(Uuid, Uuid)>,
632    state: web::Data<AppState<C>>,
633    body: web::Json<CreateIntermediateCaRequest>,
634) -> Result<HttpResponse, AxiamApiError> {
635    RequirePermission::new("ca_certificates:generate", Uuid::nil())
636        .check(&user, authz.get_ref().as_ref())
637        .await?;
638    let (org_id, tenant_id) = path.into_inner();
639    require_own_org(&user, org_id)?;
640
641    let req = body.into_inner();
642    let result = state
643        .pki
644        .ca_service
645        .generate_intermediate(CreateIntermediateCa {
646            organization_id: org_id,
647            tenant_id,
648            parent_ca_id: req.parent_ca_id,
649            subject: req.subject,
650            key_algorithm: req.key_algorithm,
651            validity_days: req.validity_days,
652        })
653        .await?;
654    Ok(HttpResponse::Created().json(result))
655}

744/// Refuse a path that names an organization other than the caller's own.
745///
746/// The same check the four older handlers in this module each spell out inline.
747/// Extracted here rather than restated a fifth, sixth and seventh time: this is
748/// the whole of the tenancy boundary on these routes, and a copy that drifted
749/// would be a copy that let one organization read another's CAs.
750fn require_own_org(user: &AuthenticatedUser, org_id: Uuid) -> Result<(), AxiamApiError> {
751    if org_id == user.org_id {
752        return Ok(());
753    }
754    Err(AxiamApiError(
755        axiam_core::error::AxiamError::AuthorizationDenied {
756            reason: "cannot access a different organization".into(),
757            action: None,
758            resource_id: None,
759        },
760    ))
