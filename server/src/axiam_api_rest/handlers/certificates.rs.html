27#[derive(Debug, Deserialize, utoipa::ToSchema)]
28pub struct CreateCertificateRequest {
29    pub issuer_ca_id: Uuid,
30    pub subject: String,
31    pub cert_type: CertificateType,
32    pub key_algorithm: KeyAlgorithm,
33    /// Validity duration in days.
34    pub validity_days: u32,
35    pub metadata: Option<serde_json::Value>,
36    /// The names a `Server` certificate is issued for, as
37    /// `[{"dns": "api.lakeside.internal"}, {"ip": "10.0.0.5"}]`. Required for
38    /// `cert_type: Server` and refused for every other type. Each name, and
39    /// the common name, must be admitted by the tenant's effective
40    /// `server_cert_allowed_names`, which is empty — refusing every `Server`
41    /// request — until an organization administrator lists names.
42    #[serde(default)]
43    pub subject_alt_names: Option<Vec<SubjectAltName>>,
44}
45
46/// The acting tenant's effective `server_cert_allowed_names` — read only for a
47/// `Server` request, so every other request makes exactly the calls it made
48/// before S-7 (I1). A failed read is an error, never an empty list read as
49/// "nothing is allowed" or a missing one read as "anything is": the fence has
50/// to be answered from the stored policy or not at all.
51async fn server_names_for<C: Connection + Clone>(
52    cert_type: &CertificateType,
53    principal: &AuthenticatedPrincipal,
54    state: &AppState<C>,
55) -> Result<Vec<String>, AxiamApiError> {
56    if *cert_type != CertificateType::Server {
57        return Ok(Vec::new());
58    }
59    let settings = axiam_core::repository::SettingsRepository::get_effective_settings(
60        &state.settings_repo,
61        principal.org_id,
62        principal.tenant_id,
63    )
64    .await?;
65    Ok(settings.certificate.server_cert_allowed_names)
66}
67
68/// The issuing scope this caller acts with on both leaf paths.
69///
70/// [`IssuingScope::Organization`] only for a human principal whose own record
71/// lives in the organization's reserved scope; everyone else — every service
72/// account included — is [`IssuingScope::Tenant`] and is confined to the
73/// signing CA of the tenant being acted on. The tenant being acted on is
74/// `principal.tenant_id`, which is the caller's own tenant unless it named
75/// another through `X-Axiam-Tenant` and was allowed to — which only an
76/// organization-level principal, person or machine, can be.
77async fn issuing_scope<C: Connection + Clone>(
78    principal: &AuthenticatedPrincipal,
79    state: &AppState<C>,
80) -> IssuingScope {
81    if is_organization_principal(principal, state).await {
82        IssuingScope::Organization
83    } else {
84        IssuingScope::Tenant
85    }
86}

88/// `POST /api/v1/certificates`
89#[utoipa::path(
90    post,
91    path = "/api/v1/certificates",
92    tag = "certificates",
93    request_body = CreateCertificateRequest,
94    responses(
95        (status = 201, description = "Certificate generated",
96         body = GeneratedCertificate),
97        (status = 400, description = "Invalid request: the subject is not a common name, the \
98                                      validity exceeds a cap, or the names are refused — \
99                                      `subject_alt_names` on a type other than `Server`, a \
100                                      `Server` request with none, or a SAN or common name not \
101                                      admitted by the tenant's `server_cert_allowed_names` \
102                                      (empty by default, which refuses every `Server` request)"),
103        (status = 404, description = "No such issuing CA within this caller's reach: it \
104                                      belongs to another organization, to another tenant, \
105                                      or is the organization CA and the caller is not an \
106                                      organization principal"),
107    ),
108    security(("bearer" = []), ("service_account" = []))
109)]
110pub async fn generate<C: Connection + Clone>(
111    principal: AuthenticatedPrincipal,
112    authz: AuthzData,
113    state: web::Data<AppState<C>>,
114    body: web::Json<CreateCertificateRequest>,
115) -> Result<HttpResponse, AxiamApiError> {
116    RequirePermission::new("certificates:generate", Uuid::nil())
117        .check(&principal, authz.get_ref().as_ref())
118        .await?;
119    let req = body.into_inner();
120    let input = CreateCertificate {
121        tenant_id: principal.tenant_id,
122        issuer_ca_id: req.issuer_ca_id,
123        subject: req.subject,
124        cert_type: req.cert_type,
125        key_algorithm: req.key_algorithm,
126        validity_days: req.validity_days,
127        metadata: req.metadata,
128        subject_alt_names: req.subject_alt_names.unwrap_or_default(),
129    };
130    let server_names = server_names_for(&input.cert_type, &principal, state.get_ref()).await?;
131
132    // Read tenant-level max_certificate_validity_days from metadata
133    let tenant = state.tenant_repo.get_by_id(principal.tenant_id).await?;
134    let max_validity = tenant
135        .metadata
136        .get("max_certificate_validity_days")
137        .and_then(|v| v.as_u64())
138        .map(|v| v as u32);
139
140    // Which CAs of the organization this call can reach: its own tenant's
141    // signing CA always, the organization CA only for a principal that lives in
142    // the organization scope. Resolved from the caller's own record, never from
143    // the body (DF-017).
144    let scope = issuing_scope(&principal, state.get_ref()).await;
145
146    let result = state
147        .pki
148        .cert_service
149        .generate(principal.org_id, scope, input, max_validity, &server_names)
150        .await?;
151    Ok(HttpResponse::Created().json(result))
152}

154/// Body of `POST /api/v1/certificates/sign-csr`.
155///
156/// No `subject` and no `key_algorithm`: both are read out of the CSR, which is
157/// the only place they can be stated without the row and the certificate being
158/// able to disagree. No key is returned, so there is no key field anywhere on
159/// this exchange.
160#[derive(Debug, Deserialize, utoipa::ToSchema)]
161pub struct SignCertificateCsrRequest {
162    pub issuer_ca_id: Uuid,
163    /// PEM-encoded PKCS#10 request — a `BEGIN CERTIFICATE REQUEST` block. The
164    /// legacy OpenSSL `BEGIN NEW CERTIFICATE REQUEST` header is not accepted.
165    pub csr_pem: String,
166    pub cert_type: CertificateType,
167    /// Validity duration in days.
168    pub validity_days: u32,
169    pub metadata: Option<serde_json::Value>,
170    /// See [`CreateCertificateRequest::subject_alt_names`]. Stated here and
171    /// never in the CSR, which is still refused if it requests a
172    /// `subjectAltName`. Under a CA whose key is held by `vault_pki` a `Server`
173    /// request on this path is refused; use `POST /api/v1/certificates`.
174    #[serde(default)]
175    pub subject_alt_names: Option<Vec<SubjectAltName>>,
176}

178/// `POST /api/v1/certificates/sign-csr`
179///
180/// Issue an end-entity certificate for a key AXIAM never sees. The subscriber's
181/// private key is generated by whoever made the CSR — an offline ceremony, an
182/// HSM, a device's secure element — and crosses the wire in neither direction.
183///
184/// `certificates:generate`, and not a permission of its own: a caller allowed
185/// to mint a certificate under a CA is allowed to mint one for a key they
186/// already hold, and this path is strictly the less powerful of the two
187/// because no key material is produced or returned. The signing-CA twin,
188/// `signing-cas/sign-csr`, sets the same precedent with
189/// `ca_certificates:generate`.
190///
191/// Responds `201` with a [`Certificate`] — not a `GeneratedCertificate`, whose
192/// `private_key_pem` would be a field that is always absent.
193#[utoipa::path(
194    post,
195    path = "/api/v1/certificates/sign-csr",
196    tag = "certificates",
197    request_body = SignCertificateCsrRequest,
198    responses(
199        (status = 201, description = "Certificate issued from the request. Carries no \
200                                      private key: there is none, which is the point.",
201         body = Certificate),
202        (status = 400, description = "The request is not a PEM PKCS#10 CSR, its signature \
203                                      does not verify against the public key it carries, its \
204                                      key is outside AXIAM's policy (Ed25519, or RSA with a \
205                                      modulus of at least 4096 bits), it asks for a \
206                                      `subjectAltName`, `keyUsage` or `extendedKeyUsage` \
207                                      extension, the validity exceeds the tenant cap, the \
208                                      825-day hard cap, or the issuer's own expiry, or the \
209                                      names are refused (see `POST /api/v1/certificates`; a \
210                                      `Server` request is also refused under a `vault_pki` CA)"),
211        (status = 401, description = "Unauthorized"),
212        (status = 403, description = "Forbidden — `certificates:generate` required"),
213        (status = 404, description = "No such issuing CA within this caller's reach: it \
214                                      belongs to another organization, to another tenant, \
215                                      or is the organization CA and the caller is not an \
216                                      organization principal"),
217    ),
218    security(("bearer" = []), ("service_account" = []))
219)]
220pub async fn sign_csr<C: Connection + Clone>(
221    principal: AuthenticatedPrincipal,
222    authz: AuthzData,
223    state: web::Data<AppState<C>>,
224    body: web::Json<SignCertificateCsrRequest>,
225) -> Result<HttpResponse, AxiamApiError> {
226    RequirePermission::new("certificates:generate", Uuid::nil())
227        .check(&principal, authz.get_ref().as_ref())
228        .await?;
229    let req = body.into_inner();
230    let input = SignCertificateCsr {
231        // From the authenticated context, never from the body (T-98).
232        tenant_id: principal.tenant_id,
233        issuer_ca_id: req.issuer_ca_id,
234        csr_pem: req.csr_pem,
235        cert_type: req.cert_type,
236        validity_days: req.validity_days,
237        metadata: req.metadata,
238        subject_alt_names: req.subject_alt_names.unwrap_or_default(),
239    };
240    let server_names = server_names_for(&input.cert_type, &principal, state.get_ref()).await?;
241
242    // The same tenant cap `generate` reads, from the same place.
243    let tenant = state.tenant_repo.get_by_id(principal.tenant_id).await?;
244    let max_validity = tenant
245        .metadata
246        .get("max_certificate_validity_days")
247        .and_then(|v| v.as_u64())
248        .map(|v| v as u32);
249
250    let scope = issuing_scope(&principal, state.get_ref()).await;
251
252    let certificate = state
253        .pki
254        .cert_service
255        .sign_csr(principal.org_id, scope, input, max_validity, &server_names)
256        .await?;
257    Ok(HttpResponse::Created().json(certificate))
258}

260/// `GET /api/v1/certificates`
261/// A certificate plus the service account it authenticates, if any.
262///
263/// The binding is a `cert_bound_to` graph edge, so it appeared on neither the
264/// certificate row nor the service account row and the admin UI had nothing to
265/// render — which is why it was "almost impossible to understand if a
266/// certificate was bound to the service account or not". You could perform the
267/// binding and then find no trace of it anywhere in the product.
268///
269/// A response wrapper rather than a field on
270/// [`axiam_core::models::certificate::Certificate`]: the domain type describes
271/// what a certificate *is*, and what it happens to be attached to is a fact
272/// about the graph. Flattened on the wire, so a client that already parses a
273/// certificate keeps working and simply gains a field.
274#[derive(Debug, serde::Serialize, utoipa::ToSchema)]
275pub struct CertificateWithBinding {
276    #[serde(flatten)]
277    pub certificate: Certificate,
278    /// The service account this certificate authenticates, or `null`.
279    pub bound_service_account_id: Option<Uuid>,
280}
281
282impl CertificateWithBinding {
283    fn attach(
284        certificates: Vec<Certificate>,
285        bindings: &std::collections::HashMap<Uuid, Uuid>,
286    ) -> Vec<Self> {
287        certificates
288            .into_iter()
289            .map(|certificate| Self {
290                bound_service_account_id: bindings.get(&certificate.id).copied(),
291                certificate,
292            })
293            .collect()
294    }
295}

297#[utoipa::path(
298    get,
299    path = "/api/v1/certificates",
300    tag = "certificates",
301    params(Pagination),
302    responses(
303        (status = 200, description = "List of certificates",
304         body = inline(PaginatedResult<CertificateWithBinding>)),
305    ),
306    security(("bearer" = []), ("service_account" = []))
307)]
308pub async fn list<C: Connection + Clone>(
309    principal: AuthenticatedPrincipal,
310    authz: AuthzData,
311    state: web::Data<AppState<C>>,
312    pagination: web::Query<Pagination>,
313) -> Result<HttpResponse, AxiamApiError> {
314    RequirePermission::new("certificates:list", Uuid::nil())
315        .check(&principal, authz.get_ref().as_ref())
316        .await?;
317    let result = state
318        .pki
319        .cert_service
320        .list(principal.tenant_id, pagination.into_inner())
321        .await?;
322
323    // One extra query for the whole page, not one per row — see
324    // `CertificateRepository::bound_service_accounts` for why the per-row form
325    // is what kept this off the page in the first place.
326    let ids: Vec<Uuid> = result.items.iter().map(|c| c.id).collect();
327    let bindings = {
328        use axiam_core::repository::CertificateRepository as _;
329        state.pki.cert_repo.bound_service_accounts(&ids).await?
330    };
331
332    Ok(HttpResponse::Ok().json(PaginatedResult {
333        items: CertificateWithBinding::attach(result.items, &bindings),
334        total: result.total,
335        offset: result.offset,
336        limit: result.limit,
337    }))
338}

394/// `POST /api/v1/service-accounts/{sa_id}/bind-certificate`
395#[utoipa::path(
396    post,
397    path = "/api/v1/service-accounts/{sa_id}/bind-certificate",
398    tag = "certificates",
399    request_body = BindCertificate,
400    params(("sa_id" = Uuid, Path, description = "Service account ID")),
401    responses(
402        (status = 200, description = "Certificate bound to service account"),
403        (status = 400, description = "The certificate cannot authenticate anything: it is not \
404                                      Active, it has expired, or it is a `Server` certificate"),
405    ),
406    security(("bearer" = []), ("service_account" = []))
407)]
408pub async fn bind<C: Connection + Clone>(
409    principal: AuthenticatedPrincipal,
410    authz: AuthzData,
411    path: web::Path<Uuid>,
412    state: web::Data<AppState<C>>,
413    body: web::Json<BindCertificate>,
414) -> Result<HttpResponse, AxiamApiError> {
415    RequirePermission::new("certificates:bind", Uuid::nil())
416        .check(&principal, authz.get_ref().as_ref())
417        .await?;
418    let sa_id = path.into_inner();
419    let input = body.into_inner();
420
421    // Verify the certificate belongs to the same tenant.
422    let cert = state
423        .pki
424        .cert_repo
425        .get_by_id(principal.tenant_id, input.certificate_id)
426        .await?;
427
428    // S-7 — a server certificate authenticates nobody, and binding one would
429    // make a certificate for a *host name* the credential of a principal. The
430    // device-login path refuses it as well; this is the refusal an operator
431    // sees, at the moment the mistake is made.
432    if cert.cert_type == CertificateType::Server {
433        return Err(AxiamApiError(AxiamError::Validation {
434            message: "a Server certificate cannot be bound to a service account: it identifies \
435                      a host, carries serverAuth only, and cannot authenticate a client"
436                .into(),
437        }));
438    }
439
440    // And that it can actually authenticate anything. A revoked or expired
441    // certificate binds happily and then fails every handshake, so the operator
442    // sees a service account that is configured for mTLS and cannot connect,
443    // with nothing on either record saying why. The admin UI already filters
444    // its picker to Active certificates; this is the same rule where it is
445    // enforceable, for the API clients that do not go through that picker.
446    if cert.status != CertificateStatus::Active {
447        return Err(AxiamApiError(AxiamError::Validation {
448            message: format!(
449                "certificate is {:?} and cannot authenticate a service account",
450                cert.status
451            ),
452        }));
453    }
454    if cert.not_after <= chrono::Utc::now() {
455        return Err(AxiamApiError(AxiamError::Validation {
456            message: format!(
457                "certificate expired on {} and cannot authenticate a service account",
458                cert.not_after.format("%Y-%m-%d")
459            ),
460        }));
461    }
462
463    // Verify the service account belongs to the same tenant.
464    use axiam_core::repository::ServiceAccountRepository;
465    state
466        .service_account_repo
467        .get_by_id(principal.tenant_id, sa_id)
468        .await?;
469
470    state
471        .pki
472        .cert_repo
473        .bind_to_service_account(principal.tenant_id, cert.id, sa_id)
474        .await?;
475
476    Ok(HttpResponse::Ok().json(serde_json::json!({
477        "certificate_id": cert.id,
478        "service_account_id": sa_id,
479        "status": "bound"
480    })))
