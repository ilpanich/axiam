1//! The CIBA approval API — the user's half of a backchannel authentication
2//! request (G-7, T23.7.2, D-68).
3//!
4//! `POST /oauth2/bc-authorize` (see [`super::ciba`]) stores a request for a user
5//! the client named; the user is told (a mail with a link, T23.7.2) and decides
6//! here, in the console's end-user area, **after a full sign-in**. Three routes,
7//! the device grant's user routes' shape (`/api/v1/device/verify` and
8//! `/decide`) with the same two properties of that placement:
9//!
10//! * `AuthzMiddleware` — the caller is an authenticated human with a session.
11//!   Approval records *that session* as the evidence behind the minted tokens
12//!   (`sid`, `auth_time`, `amr`, `acr`; D-67), so a caller without one is
13//!   refused, whatever token it holds — and so is a token AXIAM minted for an
14//!   OAuth2 client, which names a session too (W5 F4 review, T-447): only a
15//!   console sign-in's token decides.
16//! * `CsrfMiddleware` — another origin cannot silently POST an approval on a
17//!   victim's session; the `binding_message` is what lets the *user* tell a
18//!   request they started from one an attacker did.
19//!
20//! | Route | Does |
21//! |---|---|
22//! | `GET /api/v1/ciba/requests/{request_id}` | what the page shows, and the `version` it must send back |
23//! | `POST …/approve` | approve, conditional on that `version` |
24//! | `POST …/deny` | refuse, conditional on that `version` |
25//!
26//! # One answer for every reason (D-68)
27//!
28//! An unknown id, a request for **another user**, an expired one, one already
29//! decided, and one decided by someone else since the page read it are all
30//! `404 not_found`. The page must not become an oracle for which request ids
31//! exist or whose they are — the id is a handle, not a secret (D-68), so what
32//! protects a request is that only its own user, signed in, can see it.
33//!
34//! # Step-up
35//!
36//! A request may ask for an authentication class (`acr_values`) the session
37//! has not achieved. `GET` says so up front (`step_up_required`), and `approve`
38//! answers `403 step_up_required` naming the class if the page asked anyway. The
39//! page then sends the user through the **existing login hop**
40//! (`/login?return_to=…&reauth=1&acr=…`), which ends the weak session, demands
41//! the factor, and comes back to this page — no marker is consumed by the way
42//! back: the request is validated afresh by the next `GET`, and `approve` is
43//! conditional on the version that read returned (T-404's lesson).
44//!
45//! # What the audit says
46//!
47//! `ciba.approved` and `ciba.denied` record the user, the session that decided,
48//! the request id, the client and the delivery mode — and, for an approval, the
49//! `acr` achieved. **Never the
50//! `binding_message`**: a client chooses it, and a call-centre agent's text
51//! ("confirm the transfer to J. Doe") can be personal data that has no business
52//! in an append-only log. The reader who needs it has the request row for as
53//! long as the row lives.

75/// What the approval page shows. Never the `auth_req_id`, a token, or the
76/// client's notification endpoint.
77#[derive(Debug, Serialize, utoipa::ToSchema)]
78pub struct CibaApprovalPage {
79    /// The request's record id (the one in the URL).
80    pub request_id: Uuid,
81    /// Send this back with the decision: approval and refusal are conditional on
82    /// it (D-68), so a request changed since the page read it is not decided.
83    pub version: u64,
84    /// The client that asked, by `client_id`.
85    pub client_id: String,
86    /// The client's display name (its `client_id` if the client was deleted).
87    pub client_name: String,
88    /// The scopes requested, `openid` included.
89    pub scopes: Vec<String>,
90    /// The client's `binding_message`, to compare with what the client's own
91    /// device shows. Render it as **text**.
92    pub binding_message: Option<String>,
93    /// The authentication classes the request asked for that AXIAM implements
94    /// (`urn:axiam:acr:1fa`, `urn:axiam:acr:mfa`). Empty when it asked for none.
95    pub requested_acr: Vec<String>,
96    /// Set when this session has not achieved a requested class: the class to
97    /// step up to before approving. `None` when approval can go ahead.
98    pub step_up_required: Option<String>,
99    /// When the request expires.
100    pub expires_at: DateTime<Utc>,
101}

151/// The authentication behind the caller's session, or `None` for a caller with
152/// no session row (a machine token, a session ended since the token was
153/// validated) **or whose token AXIAM minted for an OAuth2 client**.
154///
155/// The last is the W5 F4 review's (T-447): an access token from the code,
156/// refresh or CIBA grant names the user and a live session, and the user
157/// extractor admits it like a console sign-in's. Approving is the person's act
158/// on this page (contract §33: "that page is the console's job and is not SDK
159/// surface"); a relying party holding one of its user's tokens — a CIBA client
160/// that redeemed an earlier request among them — must not be able to approve
161/// in the user's name. A console sign-in's token carries no `client_id`.
162async fn session_evidence<C: Connection + Clone>(
163    state: &AppState<C>,
164    user: &AuthenticatedUser,
165) -> Result<Option<(DateTime<Utc>, Vec<Amr>)>, ()> {
166    if user.claims.0.client_id.is_some() {
167        return Ok(None);
168    }
169    match state
170        .session_repo
171        .get_by_id(user.principal_tenant_id, user.session_id)
172        .await
173    {
174        Ok(session) if session.user_id == user.user_id => {
175            Ok(Some((session.authenticated_at, session.amr)))
176        }
177        Ok(_) => Ok(None),
178        Err(axiam_core::error::AxiamError::NotFound { .. }) => Ok(None),
179        Err(e) => {
180            tracing::error!(error = %e, "the CIBA approval could not read the caller's session");
181            Err(())
182        }
183    }
184}

186/// Read a pending CIBA sign-in request addressed to the signed-in user.
187#[utoipa::path(
188    get,
189    operation_id = "ciba_approval_get",
190    path = "/api/v1/ciba/requests/{request_id}",
191    tag = "ciba",
192    params(("request_id" = Uuid, Path, description = "The request's record id, from the notification link")),
193    responses(
194        (status = 200, description = "The pending request", body = CibaApprovalPage),
195        (status = 401, description = "Not authenticated"),
196        (status = 403, description = "The token is not a console sign-in's: no session row behind it, \
197                                      or AXIAM minted it for an OAuth2 client"),
198        (status = 404, description = "Unknown, another user's, expired or already decided — one answer"),
199    ),
200    security(("session" = [])),
201)]
202pub async fn get_request<C: Connection + Clone>(
203    user: AuthenticatedUser,
204    path: web::Path<Uuid>,
205    state: web::Data<AppState<C>>,
206) -> HttpResponse {
207    let request_id = path.into_inner();
208    let tenant_id = user.principal_tenant_id;
209    let Ok(evidence) = session_evidence(&state, &user).await else {
210        return server_error();
211    };
212    let Some((_, amr)) = evidence else {
213        return session_required();
214    };
215    let view = match state
216        .oauth2
217        .ciba_service
218        .lookup_for_approval(tenant_id, request_id, user.user_id)
219        .await
220    {
221        Ok(Some(view)) => view,
222        Ok(None) => return not_found(),
223        Err(e) => {
224            tracing::error!(error = %e, "the CIBA approval lookup failed");
225            return server_error();
226        }
227    };
228    let client_name = state
229        .oauth2_client_repo
230        .get_by_client_id(tenant_id, &view.client_id)
231        .await
232        .map_or_else(|_| view.client_id.clone(), |client| client.name);
233    let requested_acr = view
234        .acr_values
235        .iter()
236        .filter_map(|v| axiam_oauth2::acr::Acr::from_wire(v))
237        .map(|acr| acr.as_str().to_owned())
238        .collect();
239    HttpResponse::Ok()
240        .append_header(("Cache-Control", "no-store"))
241        .json(CibaApprovalPage {
242            request_id: view.request_id,
243            version: view.version,
244            client_id: view.client_id,
245            client_name,
246            scopes: view.scopes,
247            binding_message: view.binding_message,
248            requested_acr,
249            step_up_required: step_up_required(&view.acr_values, &amr)
250                .map(|acr| acr.as_str().to_owned()),
251            expires_at: view.expires_at,
252        })
253}

255/// Approve a pending CIBA sign-in request, conditional on the version read.
256#[utoipa::path(
257    post,
258    operation_id = "ciba_approval_approve",
259    path = "/api/v1/ciba/requests/{request_id}/approve",
260    tag = "ciba",
261    params(("request_id" = Uuid, Path, description = "The request's record id")),
262    request_body = CibaDecisionBody,
263    responses(
264        (status = 200, description = "Approved", body = CibaDecisionResponse),
265        (status = 401, description = "Not authenticated"),
266        (status = 403, description = "The request asked for an authentication class this session has not \
267                                      achieved (`step_up_required`, body below); or the token is not a \
268                                      console sign-in's (`authorization_denied`)", body = CibaStepUpRequired),
269        (status = 404, description = "Unknown, another user's, expired, already decided, or changed since read"),
270    ),
271    security(("session" = [])),
272)]
273pub async fn approve<C: Connection + Clone>(
274    http: HttpRequest,
275    user: AuthenticatedUser,
276    path: web::Path<Uuid>,
277    body: web::Json<CibaDecisionBody>,
278    state: web::Data<AppState<C>>,
279) -> HttpResponse {
280    let request_id = path.into_inner();
281    let tenant_id = user.principal_tenant_id;
282    let Ok(evidence) = session_evidence(&state, &user).await else {
283        return server_error();
284    };
285    let Some((auth_time, amr)) = evidence else {
286        return session_required();
287    };
288    let outcome = state
289        .oauth2
290        .ciba_service
291        .approve(
292            tenant_id,
293            request_id,
294            body.version,
295            CibaApproval {
296                user_id: user.user_id,
297                session_id: user.session_id,
298                auth_time,
299                amr,
300            },
301        )
302        .await;
303    match outcome {
304        Ok(CibaDecisionOutcome::Recorded(request)) => {
305            audit_decision(&state, &http, &user, &request, CIBA_APPROVED_AUDIT_ACTION).await;
306            decided("approved")
307        }
308        Ok(CibaDecisionOutcome::NotDecidable) => not_found(),
309        Ok(CibaDecisionOutcome::StepUpRequired { required }) => HttpResponse::Forbidden()
310            .append_header(("Cache-Control", "no-store"))
311            .json(CibaStepUpRequired {
312                error: "step_up_required",
313                message: "This request needs a stronger sign-in than your current session.",
314                required_acr: required.as_str().to_owned(),
315            }),
316        Err(e) => {
317            tracing::error!(error = %e, "a CIBA approval failed");
318            server_error()
319        }
320    }
321}

323/// Refuse a pending CIBA sign-in request, conditional on the version read.
324#[utoipa::path(
325    post,
326    operation_id = "ciba_approval_deny",
327    path = "/api/v1/ciba/requests/{request_id}/deny",
328    tag = "ciba",
329    params(("request_id" = Uuid, Path, description = "The request's record id")),
330    request_body = CibaDecisionBody,
331    responses(
332        (status = 200, description = "Refused", body = CibaDecisionResponse),
333        (status = 401, description = "Not authenticated"),
334        (status = 403, description = "The token is not a console sign-in's: no session row behind it, \
335                                      or AXIAM minted it for an OAuth2 client"),
336        (status = 404, description = "Unknown, another user's, expired, already decided, or changed since read"),
337    ),
338    security(("session" = [])),
339)]
340pub async fn deny<C: Connection + Clone>(
341    http: HttpRequest,
342    user: AuthenticatedUser,
343    path: web::Path<Uuid>,
344    body: web::Json<CibaDecisionBody>,
345    state: web::Data<AppState<C>>,
346) -> HttpResponse {
347    let request_id = path.into_inner();
348    let tenant_id = user.principal_tenant_id;
349    let Ok(evidence) = session_evidence(&state, &user).await else {
350        return server_error();
351    };
352    if evidence.is_none() {
353        return session_required();
354    }
355    match state
356        .oauth2
357        .ciba_service
358        .deny(tenant_id, request_id, body.version, user.user_id)
359        .await
360    {
361        Ok(CibaDecisionOutcome::Recorded(request)) => {
362            audit_decision(&state, &http, &user, &request, CIBA_DENIED_AUDIT_ACTION).await;
363            decided("denied")
364        }
365        // `deny` never asks for a step-up: refusing needs no stronger sign-in.
366        Ok(CibaDecisionOutcome::NotDecidable | CibaDecisionOutcome::StepUpRequired { .. }) => {
367            not_found()
368        }
369        Err(e) => {
370            tracing::error!(error = %e, "a CIBA refusal failed");
371            server_error()
372        }
373    }
374}

376fn decided(decision: &'static str) -> HttpResponse {
377    HttpResponse::Ok()
378        .append_header(("Cache-Control", "no-store"))
379        .json(CibaDecisionResponse { ok: true, decision })
380}

382/// The audit row of a recorded decision. Awaited, not detached: the decision
383/// is already durable, and a deployment that cannot write its audit log should
384/// say so in its own log rather than lose the row silently — but the user's
385/// answer is not undone by it.
386async fn audit_decision<C: Connection + Clone>(
387    state: &AppState<C>,
388    http: &HttpRequest,
389    user: &AuthenticatedUser,
390    request: &CibaRequest,
391    action: &str,
392) {
393    // The session that decided (W5 F4 review, T-435): the request row that
394    // also holds it for an approval is swept ten minutes after it expires, and
395    // a refusal records no session on the row at all.
396    let mut metadata = serde_json::json!({
397        "client_id": request.client_id,
398        "delivery_mode": request.delivery_mode.as_str(),
399        "session_id": user.session_id.to_string(),
400    });
401    if let Some(approval) = &request.approval {
402        metadata["acr"] = serde_json::Value::String(approval.acr.clone());
403    }
404    let entry = CreateAuditLogEntry {
405        tenant_id: request.tenant_id,
406        actor_id: user.user_id,
407        actor_type: ActorType::User,
408        action: action.into(),
409        resource_id: Some(request.id),
410        outcome: AuditOutcome::Success,
411        ip_address: client_ip(http),
412        metadata: Some(metadata),
413    };
414    if let Err(e) = state.audit_repo.append(entry).await {
415        tracing::error!(error = %e, action, "failed to write a CIBA decision audit row");
416    }
