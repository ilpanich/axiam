1//! `GET /pki/v1/{org_id}/ca/{ca_id}/crl` — an issuing CA's certificate
2//! revocation list (#565, T-102).
3//!
4//! Thin on purpose: what is published, which CAs publish one, how it is signed
5//! and cached are `axiam_pki::crl`'s. This turns a [`PublishedCrl`] into HTTP —
6//! the media type, the caching headers, and a `304` for a relying party that
7//! already holds the current list.
8

21/// The response body as the OpenAPI document names it: the DER bytes of an
22/// RFC 5280 `CertificateList`.
23#[derive(utoipa::ToSchema)]
24#[schema(value_type = String, format = Binary)]
25pub struct DerCrl(pub Vec<u8>);
26
27/// The route's path parameters.
28#[derive(Debug, Deserialize)]
29pub struct CrlPath {
30    org_id: Uuid,
31    ca_id: Uuid,
32}
33
34/// `GET`/`HEAD /pki/v1/{org_id}/ca/{ca_id}/crl`
35///
36/// The CA's current CRL (RFC 5280 §5), DER, as `application/pkix-crl`:
37/// signed with the CA's own key, `nextUpdate` a configured interval after
38/// `thisUpdate` (`AXIAM__PKI__CRL_NEXT_UPDATE_SECS`, one day by default), with
39/// a CRL number and an authority key identifier, and an entry for every
40/// certificate the CA signed and revoked that has not expired. Every
41/// certificate AXIAM signs names this URL in its CRL distribution points
42/// extension when the deployment has a public base URL.
43///
44/// Unauthenticated, because a relying party fetches the list before it can
45/// validate anything, and rate-limited per IP (`AXIAM__RATE_LIMIT__CRL_PER_MIN`).
46/// `ETag` is the list's digest and `Last-Modified` its `thisUpdate`; a matching
47/// `If-None-Match` answers `304`. `Cache-Control: public, max-age` runs to the
48/// origin's next scheduled re-sign — at most half the `nextUpdate` interval —
49/// and not to `nextUpdate` (#608): a shared cache revalidates with the `ETag`
50/// that often rather than serving a list from before a revocation for the
51/// whole interval. Every replica signs the same state in the same refresh
52/// epoch to the same bytes, CRL number and `ETag`, so revalidating behind a
53/// load balancer answers `304` wherever it lands.
54///
55/// `404` for a CA that does not exist in that organization and, alike, for one
56/// that publishes no list: a revoked or expired CA (its status is on its
57/// parent's list), an imported trust anchor AXIAM holds no key for, and a CA
58/// whose key Vault's PKI engine holds.
59///
60/// `503` when the list has to be re-signed and no permit of the crypto
61/// semaphore — shared with issuance and password hashing — came free within
62/// five seconds (#607), and also when the request waited that long (`permit_wait`)
63/// on another request's in-flight re-sign of the same CA, since waiters share
64/// that re-sign's outcome, failure included. A list that has not changed is served
65/// from the cache and needs no permit, so the route keeps answering relying
66/// parties that already hold the current list while the gate is saturated.
67#[utoipa::path(
68    get,
69    path = "/pki/v1/{org_id}/ca/{ca_id}/crl",
70    tag = "pki",
71    params(
72        ("org_id" = Uuid, Path, description = "Organization ID"),
73        ("ca_id" = Uuid, Path, description = "Issuing CA certificate ID"),
74    ),
75    responses(
76        (status = 200, description = "The CA's current certificate revocation list, DER \
77             (RFC 5280 §5). `Cache-Control: public, max-age` until the next scheduled \
78             re-sign (at most half the nextUpdate interval), `ETag`, `Last-Modified`",
79             content_type = "application/pkix-crl", body = DerCrl),
80        (status = 304, description = "The `If-None-Match` tag names the current list"),
81        (status = 404, description = "No such CA in this organization, or a CA that \
82             publishes no list (revoked, expired, keyless trust anchor, or a key held by \
83             Vault's PKI engine)"),
84        (status = 429, description = "Rate limit exceeded (`AXIAM__RATE_LIMIT__CRL_PER_MIN`)"),
85        (status = 503, description = "The list has to be re-signed and the crypto gate is \
86             saturated; retry shortly"),
87    ),
88)]
89pub async fn get_crl<C: Connection + Clone>(
90    state: web::Data<AppState<C>>,
91    req: HttpRequest,
92    path: web::Path<CrlPath>,
93) -> Result<HttpResponse, AxiamApiError> {
94    let crl = state
95        .pki
96        .crl_service
97        .current(path.org_id, path.ca_id)
98        .await?;
99    Ok(respond(&req, &crl))
100}
101
102/// The response for `crl`, or a `304` when the caller already holds it.
103fn respond(req: &HttpRequest, crl: &PublishedCrl) -> HttpResponse {
104    let revalidated = req
105        .headers()
106        .get(header::IF_NONE_MATCH)
107        .and_then(|v| v.to_str().ok())
108        .is_some_and(|v| crl.matches_if_none_match(v));
109    let mut response = if revalidated {
110        HttpResponse::NotModified()
111    } else {
112        HttpResponse::Ok()
113    };
114    response
115        .insert_header((
116            header::CACHE_CONTROL,
117            format!("public, max-age={}", crl.max_age_secs(Utc::now())),
118        ))
119        .insert_header((header::ETAG, crl.etag.clone()))
120        .insert_header((
121            header::LAST_MODIFIED,
122            crl.this_update
123                .format("%a, %d %b %Y %H:%M:%S GMT")
124                .to_string(),
125        ));
126    if revalidated {
127        return response.finish();
128    }
129    response.content_type(CRL_MEDIA_TYPE).body(crl.der.to_vec())
130}
