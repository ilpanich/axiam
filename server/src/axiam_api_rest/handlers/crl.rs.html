1//! `GET /pki/v1/{org_id}/ca/{ca_id}/crl` — an issuing CA's certificate
2//! revocation list (#565, T-102).
3//!
4//! Thin on purpose: what is published, which CAs publish one, how it is signed
5//! and cached are `axiam_pki::crl`'s. This turns a [`PublishedCrl`] into HTTP —
6//! the media type, the caching headers, and a `304` for a relying party that
7//! already holds the current list.
8
9use actix_web::http::header;
10use actix_web::{HttpRequest, HttpResponse, web};
11use axiam_pki::PublishedCrl;
12use axiam_pki::crl::CRL_MEDIA_TYPE;
13use chrono::Utc;
14use serde::Deserialize;
15use surrealdb::Connection;
16use uuid::Uuid;
17
18use crate::error::AxiamApiError;
19use crate::state::AppState;

21/// The response body as the OpenAPI document names it: the DER bytes of an
22/// RFC 5280 `CertificateList`.
23#[derive(utoipa::ToSchema)]
24#[schema(value_type = String, format = Binary)]
25pub struct DerCrl(pub Vec<u8>);
26
27/// The route's path parameters.
28#[derive(Debug, Deserialize)]
29pub struct CrlPath {
30    org_id: Uuid,
31    ca_id: Uuid,
32}
33
34/// `GET`/`HEAD /pki/v1/{org_id}/ca/{ca_id}/crl`
35///
36/// The CA's current CRL (RFC 5280 §5), DER, as `application/pkix-crl`:
37/// signed with the CA's own key, `nextUpdate` a configured interval after
38/// `thisUpdate` (`AXIAM__PKI__CRL_NEXT_UPDATE_SECS`, one day by default), with
39/// a CRL number and an authority key identifier, and an entry for every
40/// certificate the CA signed and revoked that has not expired. Every
41/// certificate AXIAM signs names this URL in its CRL distribution points
42/// extension when the deployment has a public base URL.
43///
44/// Unauthenticated, because a relying party fetches the list before it can
45/// validate anything, and rate-limited per IP (`AXIAM__RATE_LIMIT__CRL_PER_MIN`).
46/// `Cache-Control: public, max-age` runs to the list's `nextUpdate`, `ETag` is
47/// the list's digest and `Last-Modified` its `thisUpdate`; a matching
48/// `If-None-Match` answers `304`.
49///
50/// `404` for a CA that does not exist in that organization and, alike, for one
51/// that publishes no list: a revoked or expired CA (its status is on its
52/// parent's list), an imported trust anchor AXIAM holds no key for, and a CA
53/// whose key Vault's PKI engine holds.
54#[utoipa::path(
55    get,
56    path = "/pki/v1/{org_id}/ca/{ca_id}/crl",
57    tag = "pki",
58    params(
59        ("org_id" = Uuid, Path, description = "Organization ID"),
60        ("ca_id" = Uuid, Path, description = "Issuing CA certificate ID"),
61    ),
62    responses(
63        (status = 200, description = "The CA's current certificate revocation list, DER \
64             (RFC 5280 §5). `Cache-Control: public, max-age` until its nextUpdate, `ETag`, \
65             `Last-Modified`", content_type = "application/pkix-crl", body = DerCrl),
66        (status = 304, description = "The `If-None-Match` tag names the current list"),
67        (status = 404, description = "No such CA in this organization, or a CA that \
68             publishes no list (revoked, expired, keyless trust anchor, or a key held by \
69             Vault's PKI engine)"),
70        (status = 429, description = "Rate limit exceeded (`AXIAM__RATE_LIMIT__CRL_PER_MIN`)"),
71    ),
72)]
73pub async fn get_crl<C: Connection + Clone>(
74    state: web::Data<AppState<C>>,
75    req: HttpRequest,
76    path: web::Path<CrlPath>,
77) -> Result<HttpResponse, AxiamApiError> {
78    let crl = state
79        .pki
80        .crl_service
81        .current(path.org_id, path.ca_id)
82        .await?;
83    Ok(respond(&req, &crl))
84}
85
86/// The response for `crl`, or a `304` when the caller already holds it.
87fn respond(req: &HttpRequest, crl: &PublishedCrl) -> HttpResponse {
88    let revalidated = req
89        .headers()
90        .get(header::IF_NONE_MATCH)
91        .and_then(|v| v.to_str().ok())
92        .is_some_and(|v| crl.matches_if_none_match(v));
93    let mut response = if revalidated {
94        HttpResponse::NotModified()
95    } else {
96        HttpResponse::Ok()
97    };
98    response
99        .insert_header((
100            header::CACHE_CONTROL,
101            format!("public, max-age={}", crl.max_age_secs(Utc::now())),
102        ))
103        .insert_header((header::ETAG, crl.etag.clone()))
104        .insert_header((
105            header::LAST_MODIFIED,
106            crl.this_update
107                .format("%a, %d %b %Y %H:%M:%S GMT")
108                .to_string(),
109        ));
110    if revalidated {
111        return response.finish();
112    }
113    response.content_type(CRL_MEDIA_TYPE).body(crl.der.to_vec())
114}
