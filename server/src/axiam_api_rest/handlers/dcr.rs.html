1//! RFC 7591 dynamic client registration — the two endpoints (T21.4).
2//!
3//! * `POST /oauth2/register` — **unauthenticated**, in `PUBLIC_PATHS`, and the
4//!   first write endpoint in AXIAM that anybody can reach without a
5//!   credential.
6//! * `POST /api/v1/oauth2-clients/registration-tokens` — the admin endpoint
7//!   that mints the single-use initial access token the *protected*
8//!   registration profile (RFC 7591 §1.2) requires.
9//!
10//! The validation rules live in `axiam_oauth2::dcr`, which is a pure function
11//! of the request and the tenant's policy. What lives here is everything that
12//! needs state: the policy read, the quota, the single-use token, the audit
13//! event and the response shape.
14//!
15//! # The order the unauthenticated endpoint does things in
16//!
17//! Each step is chosen so that the cheapest refusal comes first and so that
18//! nothing a caller can see depends on state they should not be able to
19//! probe:
20//!
21//! 1. **Resolve the tenant's policy.** A tenant that has not enabled
22//!    registration is refused here, before any metadata is parsed, so the
23//!    endpoint cannot be used to discover what the metadata rules are.
24//! 2. **Apply the mode gate.** `initial_access_token` without a bearer is
25//!    refused with the same body shape as `disabled`, so the two modes are
26//!    indistinguishable to a caller holding nothing.
27//! 3. **Validate the metadata** (`axiam_oauth2::dcr::validate`) — pure, no
28//!    reads.
29//! 4. **Check the quota**, which is one indexed count.
30//! 5. **Spend the initial access token**, atomically, if the mode needs one.
31//!    Last of the checks, so a token is not burned by a request that was
32//!    going to be refused anyway.
33//! 6. **Write the client**, then audit.
34//!
35//! The per-IP rate limit is applied by the route wrap in `server.rs`, before
36//! any of this runs.
37

67/// An RFC 7591 §3.2.2 error body.
68///
69/// The same two members the OAuth2 error responses elsewhere in this crate
70/// carry, declared separately because RFC 7591 §3.2.2 defines its own code
71/// vocabulary (`invalid_redirect_uri`, `invalid_client_metadata`,
72/// `invalid_software_statement`) that has nothing to do with RFC 6749 §5.2's.
73#[derive(Debug, Serialize, utoipa::ToSchema)]
74pub struct DcrErrorResponse {
75    /// RFC 7591 §3.2.2 error code.
76    pub error: String,
77    /// Human-readable explanation.
78    pub error_description: String,

81fn error_response(err: &DcrError) -> HttpResponse {
82    let status = actix_web::http::StatusCode::from_u16(err.http_status())
83        .unwrap_or(actix_web::http::StatusCode::BAD_REQUEST);
84    HttpResponse::build(status)
85        // RFC 7591 §3.2 asks for the same no-store posture the token endpoint
86        // has: a registration response carries a client secret, and an error
87        // response carries the fact that a tenant runs registration at all.
88        .append_header(("Cache-Control", "no-store"))
89        .append_header(("Pragma", "no-cache"))
90        .json(DcrErrorResponse {
91            error: err.error_code().to_owned(),
92            error_description: err.description(),
93        })
94}

96/// Read the tenant's effective OIDC policy, or refuse.
97///
98/// An unknown tenant, an unreadable settings row and a tenant whose
99/// organization cannot be resolved all answer
100/// [`DcrError::RegistrationDisabled`] — the same answer a tenant that simply
101/// turned registration off gets. That is deliberate on two counts. It fails
102/// closed: a settings read that failed must never open an unauthenticated
103/// write endpoint. And it refuses to be an oracle: `404` for an unknown tenant
104/// would make this endpoint a tenant-enumeration probe, which is exactly the
105/// argument the discovery handler already makes for answering an unknown
106/// tenant with the deployment-wide document.
107async fn effective_policy<C: Connection + Clone>(
108    state: &AppState<C>,
109    tenant_id: Uuid,
110) -> Result<OidcPolicy, DcrError> {
111    let tenant = state
112        .tenant_repo
113        .get_by_id(tenant_id)
114        .await
115        .map_err(|_| DcrError::RegistrationDisabled)?;
116    let settings = SettingsRepository::get_effective_settings(
117        &state.settings_repo,
118        tenant.organization_id,
119        tenant_id,
120    )
121    .await
122    .map_err(|e| {
123        tracing::error!(
124            error = %e,
125            %tenant_id,
126            "could not read a tenant's effective settings while answering a registration \
127             request; treating dynamic registration as disabled"
128        );
129        DcrError::RegistrationDisabled
130    })?;
131    Ok(settings.oidc)
132}

134/// The bearer an `initial_access_token` registration presents, if any.
135///
136/// Only the `Bearer` scheme, and only a non-empty value. A malformed header is
137/// read as *absent* rather than as an error, so every way of failing to
138/// present a usable token produces the one refusal
139/// [`DcrError::InitialAccessTokenRequired`] describes.
140fn bearer_token(req: &HttpRequest) -> Option<String> {
141    req.headers()
142        .get(actix_web::http::header::AUTHORIZATION)?
143        .to_str()
144        .ok()?
145        .strip_prefix("Bearer ")
146        .map(str::trim)
147        .filter(|t| !t.is_empty())
148        .map(str::to_owned)
149}

151/// `POST /oauth2/register` — RFC 7591 §3.1 dynamic client registration.
152#[utoipa::path(
153    post,
154    path = "/oauth2/register",
155    tag = "oauth2",
156    params(TenantQuery),
157    request_body = RegistrationRequest,
158    responses(
159        (status = 201, description = "Client registered (secret, if any, shown once)",
160         body = RegistrationResponse),
161        (status = 400, description = "The metadata is not one this server will register",
162         body = DcrErrorResponse),
163        (status = 403, description = "Dynamic registration is not enabled for this tenant, \
164                                      the initial access token was missing or unusable, or \
165                                      the tenant's client limit is reached",
166         body = DcrErrorResponse),
167    ),
168)]
169pub async fn register<C: Connection + Clone>(
170    http_req: HttpRequest,
171    tenant_query: web::Query<TenantQuery>,
172    body: web::Json<RegistrationRequest>,
173    state: web::Data<AppState<C>>,
174) -> HttpResponse {
175    let tenant_id = tenant_query.into_inner().tenant_id;
176    let req = body.into_inner();
177
178    match register_inner(&http_req, tenant_id, req, &state).await {
179        Ok(response) => HttpResponse::Created()
180            .append_header(("Cache-Control", "no-store"))
181            .append_header(("Pragma", "no-cache"))
182            .json(response),
183        Err(err) => {
184            audit_registration(&state, &http_req, tenant_id, None, Some(&err)).await;
185            error_response(&err)
186        }
187    }
188}

190/// See [`register`]. Split so that every refusal path is one `?` and the
191/// audit event is written in one place.
192async fn register_inner<C: Connection + Clone>(
193    http_req: &HttpRequest,
194    tenant_id: Uuid,
195    req: RegistrationRequest,
196    state: &AppState<C>,
197) -> Result<RegistrationResponse, DcrError> {
198    // 1 & 2 — policy, then the mode gate, before any metadata is looked at.
199    let policy = effective_policy(state, tenant_id).await?;
200    let bearer = bearer_token(http_req);
201    dcr::gate(policy.dynamic_registration, bearer.is_some())?;
202
203    // 3 — the pure validation, plus the redirect-URI rules the admin API
204    // already applies. `validate_redirect_uris` is reused rather than
205    // reimplemented (the plan's item 3) so that "what is a usable redirect
206    // URI" has one answer in this server; its message is carried into the RFC
207    // 7591 error shape rather than answered as the admin API's `400`.
208    crate::handlers::oauth2_clients::validate_redirect_uris(&req.redirect_uris)
209        .map_err(|e| DcrError::InvalidRedirectUri(e.0.to_string()))?;
210    let validated = dcr::validate(tenant_id, &req, &policy)?;
211
212    // 4 — the per-tenant ceiling. After validation so that a malformed
213    // request is told what is wrong with it rather than being turned away at
214    // a quota it was never going to reach.
215    let existing = state
216        .oauth2_client_repo
217        .count_by_managed_by(tenant_id, ManagedBy::Dcr)
218        .await
219        .map_err(|e| {
220            tracing::error!(error = %e, %tenant_id, "could not count self-registered clients");
221            // Fail closed: an unreadable count must not be read as room.
222            DcrError::ClientQuotaExhausted {
223                limit: policy.dcr_max_clients,
224            }
225        })?;
226    if existing >= u64::from(policy.dcr_max_clients) {
227        return Err(DcrError::ClientQuotaExhausted {
228            limit: policy.dcr_max_clients,
229        });
230    }
231
232    // 5 — spend the initial access token, if the mode needs one. Last,
233    // because a single-use credential must not be burned by a request that
234    // was going to be refused for some other reason.
235    //
236    // Spent **before** the client is written, and that order is the
237    // single-use guarantee: a two-phase "create, then spend" would leave a
238    // window in which two concurrent registrations on one handle both create
239    // a client. The cost is that a handle is burned if the write then fails,
240    // which is recoverable by minting another; the alternative is not
241    // recoverable at all.
242    //
243    // The row records that it was spent and when, not which client it
244    // produced — the `client_id` does not exist yet. See the repository trait
245    // for why a placeholder is not written instead.
246    if policy.dynamic_registration == DynamicRegistrationMode::InitialAccessToken {
247        let presented = bearer.ok_or(DcrError::InitialAccessTokenRequired)?;
248        let hash = axiam_auth::token::hash_refresh_token(&presented);
249        let spent = state
250            .oauth2_registration_token_repo
251            .consume_by_token_hash(tenant_id, &hash, Utc::now())
252            .await
253            .map_err(|e| {
254                tracing::error!(error = %e, %tenant_id, "could not spend an initial access token");
255                DcrError::InitialAccessTokenRequired
256            })?;
257        if spent.is_none() {
258            return Err(DcrError::InitialAccessTokenRequired);
259        }
260    }
261
262    // 6 — write, then audit.
263    //
264    // `fapi::validate_registration` runs here for the same reason the admin
265    // handler runs it: it is where a registration that could never be served
266    // is refused. For a `dcr` row it also enforces I5 — a FAPI profile on an
267    // externally registered client — which `dcr::validate` has already made
268    // unreachable by forcing `standard`. Two gates, as everywhere else in this
269    // file, because the forcing is one line and the invariant should not
270    // depend on it staying written.
271    axiam_oauth2::fapi::validate_registration(&validated.create)
272        .map_err(|e| DcrError::InvalidClientMetadata(e.to_string()))?;
273
274    let (client, raw_secret) = state
275        .oauth2_client_repo
276        .create(validated.create)
277        .await
278        .map_err(|e| {
279            tracing::error!(error = %e, %tenant_id, "could not write a registered client");
280            DcrError::InvalidClientMetadata(
281                "the registration could not be stored; try again".into(),
282            )
283        })?;
284
285    audit_registration(state, http_req, tenant_id, Some(&client.client_id), None).await;
286
287    let is_public = client.token_endpoint_auth_method.is_public();
288    Ok(RegistrationResponse {
289        client_id: client.client_id,
290        // A public client has no secret to show — not one nobody is told
291        // about: none was minted. Omitted rather than `""`, which an MCP
292        // client would read as a secret that happens to be empty.
293        client_secret: (!is_public).then_some(raw_secret),
294        client_id_issued_at: client.created_at.timestamp(),
295        // RFC 7591 §3.2.1: REQUIRED if a secret was issued, and `0` means it
296        // does not expire. Absent when no secret was issued, because there is
297        // nothing for the member to describe.
298        client_secret_expires_at: (!is_public).then_some(0),
299        client_name: client.name,
300        redirect_uris: client.redirect_uris,
301        grant_types: client.grant_types,
302        response_types: vec!["code".to_owned()],
303        token_endpoint_auth_method: client.token_endpoint_auth_method.as_str().to_owned(),
304        scope: client.scopes.join(" "),
305    })
306}

308/// Record every registration attempt, successful or not.
309///
310/// Every one, and this is the plan's item 5: the endpoint is unauthenticated,
311/// so the audit log is the only record that a stranger reached it. The
312/// metadata deliberately carries no client-supplied string beyond the
313/// `client_id` AXIAM itself minted — a refused registration's `client_name`
314/// and `redirect_uris` are attacker-controlled, and an audit viewer is a place
315/// where attacker-controlled strings are read by people.
316async fn audit_registration<C: Connection + Clone>(
317    state: &AppState<C>,
318    http_req: &HttpRequest,
319    tenant_id: Uuid,
320    client_id: Option<&str>,
321    refusal: Option<&DcrError>,
322) {
323    let (action, outcome) = match refusal {
324        None => ("oauth2.client_registered", AuditOutcome::Success),
325        Some(_) => ("oauth2.client_registration_refused", AuditOutcome::Failure),
326    };
327    if let Err(e) = state
328        .audit_repo
329        .append(CreateAuditLogEntry {
330            tenant_id,
331            actor_id: Uuid::nil(),
332            actor_type: ActorType::System,
333            action: action.into(),
334            // The audit row's `resource_id` is a `Uuid` and a `client_id` is
335            // not one, so the identifier travels in the metadata below. The
336            // row is about an act, not about a record that may not exist: a
337            // refused registration created nothing to point at.
338            resource_id: None,
339            outcome,
340            ip_address: crate::extractors::client_info::client_ip(http_req),
341            metadata: Some(serde_json::json!({
342                "managed_by": ManagedBy::Dcr.as_str(),
343                "client_id": client_id,
344                "error": refusal.map(DcrError::error_code),
345            })),
346        })
347        .await
348    {
349        tracing::error!(
350            error = %e,
351            %tenant_id,
352            "could not record a dynamic client registration; the request is unaffected"
353        );
354    }
355}

424/// `POST /api/v1/oauth2-clients/registration-tokens`
425///
426/// Mints the single-use credential RFC 7591 §1.2's protected registration
427/// profile requires. Gated on `oauth2_clients:create` rather than a permission
428/// of its own: a token minted here authorises exactly one registration, and a
429/// registration is strictly less than what that permission already confers
430/// (an administrator holding it can create any client directly, with any
431/// scopes, any audiences and any profile). A separate permission would suggest
432/// this is the more dangerous of the two, which it is not.
433#[utoipa::path(
434    post,
435    path = "/api/v1/oauth2-clients/registration-tokens",
436    tag = "oauth2-clients",
437    request_body = CreateRegistrationTokenRequest,
438    responses(
439        (status = 201, description = "Token minted; the handle is returned once",
440         body = CreateRegistrationTokenResponse),
441        (status = 400, description = "Invalid lifetime or name, or this tenant is not in \
442                                      initial_access_token mode"),
443    ),
444    security(("bearer" = []))
445)]
446pub async fn create_registration_token<C: Connection + Clone>(
447    user: AuthenticatedUser,
448    authz: AuthzData,
449    state: web::Data<AppState<C>>,
450    body: web::Json<CreateRegistrationTokenRequest>,
451) -> Result<HttpResponse, AxiamApiError> {
452    RequirePermission::new("oauth2_clients:create", Uuid::nil())
453        .check(&user, authz.get_ref().as_ref())
454        .await?;
455
456    let input = body.into_inner();
457    let name = input.name.trim().to_owned();
458    if name.is_empty() {
459        return Err(AxiamApiError(AxiamError::Validation {
460            message: "name must not be empty".into(),
461        }));
462    }
463    let hours = input
464        .expires_in_hours
465        .unwrap_or(DEFAULT_REGISTRATION_TOKEN_TTL_HOURS);
466    if hours == 0 || hours > MAX_REGISTRATION_TOKEN_TTL_HOURS {
467        return Err(AxiamApiError(AxiamError::Validation {
468            message: format!(
469                "expires_in_hours must be between 1 and {MAX_REGISTRATION_TOKEN_TTL_HOURS}"
470            ),
471        }));
472    }
473
474    // Refuse to mint a credential for a tenant that will not honour it.
475    //
476    // The same argument `reject_unimplemented_dpop_nonce` makes next door: an
477    // operator who mints a token, hands it to somebody and watches them be
478    // refused has been told a lie by the API. `anonymous` is refused too,
479    // because there the token buys nothing — registration is already open, and
480    // a credential that authorises what was already permitted is one an
481    // operator will believe is doing something.
482    let tenant = state.tenant_repo.get_by_id(user.tenant_id).await?;
483    let settings = SettingsRepository::get_effective_settings(
484        &state.settings_repo,
485        tenant.organization_id,
486        user.tenant_id,
487    )
488    .await?;
489    if settings.oidc.dynamic_registration != DynamicRegistrationMode::InitialAccessToken {
490        return Err(AxiamApiError(AxiamError::Validation {
491            message: format!(
492                "this tenant's dynamic_registration is {}, so an initial access token would \
493                 authorise nothing: set it to initial_access_token first",
494                settings.oidc.dynamic_registration
495            ),
496        }));
497    }
498
499    let raw = format!(
500        "{REGISTRATION_TOKEN_PREFIX}{}",
501        axiam_auth::token::generate_refresh_token()
502    );
503    let token_hash = axiam_auth::token::hash_refresh_token(&raw);
504
505    let created = state
506        .oauth2_registration_token_repo
507        .create(CreateOAuth2RegistrationToken {
508            tenant_id: user.tenant_id,
509            name,
510            token_hash,
511            created_by: user.user_id,
512            expires_at: Utc::now() + Duration::hours(i64::from(hours)),
513        })
514        .await?;
515
516    tracing::info!(
517        target: "axiam::audit",
518        event = "oauth2.registration_token_created",
519        tenant_id = %user.tenant_id,
520        token_id = %created.id,
521        created_by = %user.user_id,
522        expires_at = %created.expires_at,
523        "RFC 7591 initial access token minted"
524    );
525
526    Ok(
527        HttpResponse::Created().json(CreateRegistrationTokenResponse {
528            token: RegistrationTokenResponse::from(created),
529            initial_access_token: raw,
530        }),
531    )
532}
