1//! Device-flow verification endpoints — the API behind the `/device` page (B2).
2//!
3//! RFC 8628 splits the ceremony across two devices. The input-constrained one
4//! (a television, a CLI, a sensor) talks to `/oauth2/device_authorization` and
5//! then polls `/oauth2/token`; the *user* walks to a second device with a
6//! keyboard, visits `/device`, types the short code, and approves or refuses.
7//! This module is that second half.
8//!
9//! # Why these two endpoints are authenticated and CSRF-protected
10//!
11//! They live under `/api/v1`, not under `/oauth2`, and that placement is the
12//! design rather than an accident of routing:
13//!
14//! * **Approval is an act of authorization by a specific human.** The grant
15//!   records `user_id` as the subject the eventual token is minted for, so the
16//!   caller must be authenticated — an anonymous approve endpoint would let
17//!   anyone who guesses a user code mint a token for *themselves* on a device
18//!   someone else is holding, or worse, be tricked into approving one.
19//! * **Approval is state-changing and cross-site reachable.** A user code is
20//!   short and typed by a human, which means a page on another origin can
21//!   plausibly know or guess one. `/api/v1`'s CSRF double-submit is what stops
22//!   an attacker's page from silently POSTing an approval using the victim's
23//!   session — the exact "device code phishing" shape RFC 8628 §5.4 warns
24//!   about, from the other direction.
25//!
26//! # Why both endpoints answer the same way for three different failures
27//!
28//! Unknown code, expired code, already-decided code: all three come back as
29//! `found: false` / `ok: false` with one message. Distinguishing them would
30//! turn this page into an oracle telling an attacker which of the codes they
31//! are guessing exist and are still live — and the user-code space is small by
32//! construction (8 characters from a 20-letter alphabet), because a human has
33//! to read it off a screen and type it.
34//!
35//! Guessing is additionally bounded by this endpoint's own rate-limit bucket
36//! (see `server.rs`), which is what keeps the brute-force window above the
37//! OWASP bar rather than the code length alone.
38
39use actix_web::{HttpResponse, web};
40use serde::{Deserialize, Serialize};
41use surrealdb::Connection;
42
43use crate::extractors::auth::AuthenticatedUser;
44use crate::state::AppState;

46/// `GET /api/v1/device/verify?user_code=…` — what is this code asking for?
47#[derive(Debug, Deserialize, utoipa::IntoParams)]
48pub struct VerifyQuery {
49    /// The code as the user typed it. Case, spaces and dashes are normalized
50    /// server-side, so `WXYZ-1234`, `wxyz 1234` and `WXYZ1234` are one code.
51    pub user_code: String,
52}
53
54/// The consent screen's data. Deliberately minimal: a client id and the
55/// scopes being asked for. Nothing here identifies the device or its network.
56#[derive(Debug, Serialize, utoipa::ToSchema)]
57pub struct VerifyResponse {
58    /// `false` for unknown, expired, or already-decided codes alike.
59    pub found: bool,
60    #[serde(skip_serializing_if = "Option::is_none")]
61    pub client_id: Option<String>,
62    #[serde(skip_serializing_if = "Option::is_none")]
63    pub scopes: Option<Vec<String>>,
64}
65
66/// `POST /api/v1/device/decide` — the user approves or refuses.
67#[derive(Debug, Deserialize, utoipa::ToSchema)]
68pub struct DecideRequest {
69    pub user_code: String,
70    /// `true` approves, `false` refuses. A refusal is recorded rather than
71    /// ignored: it is what lets the polling device stop immediately instead
72    /// of waiting out the full grant lifetime after the user has already said
73    /// no (see `device_service`'s polling table).
74    pub approved: bool,
75}
76
77#[derive(Debug, Serialize, utoipa::ToSchema)]
78pub struct DecideResponse {
79    /// `false` for unknown, expired, or already-decided codes alike.
80    pub ok: bool,
81}
82
83/// Look up a pending grant by the code the user typed.
84#[utoipa::path(
85    get,
86    path = "/api/v1/device/verify",
87    tag = "device",
88    params(VerifyQuery),
89    responses(
90        (status = 200, description = "Grant details, or found=false", body = VerifyResponse),
91        (status = 401, description = "Not authenticated"),
92    ),
93    security(("session" = [])),
94)]
95pub async fn verify<C: Connection + Clone>(
96    user: AuthenticatedUser,
97    query: web::Query<VerifyQuery>,
98    state: web::Data<AppState<C>>,
99) -> HttpResponse {
100    match state
101        .device_authorization_service
102        .lookup_for_verification(user.tenant_id, &query.user_code)
103        .await
104    {
105        Ok(Some((client_id, scopes))) => HttpResponse::Ok().json(VerifyResponse {
106            found: true,
107            client_id: Some(client_id),
108            scopes: Some(scopes),
109        }),
110        Ok(None) => HttpResponse::Ok().json(VerifyResponse {
111            found: false,
112            client_id: None,
113            scopes: None,
114        }),
115        Err(e) => {
116            tracing::error!(error = %e, "device verification lookup failed");
117            HttpResponse::InternalServerError().finish()
118        }
119    }
120}
121
122/// Record the user's approval or refusal.
123#[utoipa::path(
124    post,
125    path = "/api/v1/device/decide",
126    tag = "device",
127    request_body = DecideRequest,
128    responses(
129        (status = 200, description = "Decision recorded, or ok=false", body = DecideResponse),
130        (status = 401, description = "Not authenticated"),
131    ),
132    security(("session" = [])),
133)]
134pub async fn decide<C: Connection + Clone>(
135    user: AuthenticatedUser,
136    body: web::Json<DecideRequest>,
137    state: web::Data<AppState<C>>,
138) -> HttpResponse {
139    let body = body.into_inner();
140    match state
141        .device_authorization_service
142        .decide(user.tenant_id, &body.user_code, body.approved, user.user_id)
143        .await
144    {
145        Ok(ok) => HttpResponse::Ok().json(DecideResponse { ok }),
146        Err(e) => {
147            tracing::error!(error = %e, "device decision failed");
148            HttpResponse::InternalServerError().finish()
149        }
150    }
151}
