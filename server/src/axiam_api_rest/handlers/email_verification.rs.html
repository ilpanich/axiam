42/// Mint a verification token for `user_id` and enqueue the activation email.
43///
44/// Two callers, and they must produce the same email: `POST /api/v1/users`
45/// (an administrator creating an account, which is the account's *first*
46/// activation link) and `POST /api/v1/auth/resend-verification` (the user
47/// asking for another). Before this existed only the second one sent anything,
48/// so an administrator created a user, the user was written as
49/// `PendingVerification`, and nobody was ever told — the account simply could
50/// not be activated unless someone knew to call the resend endpoint by hand.
51///
52/// Best-effort by construction: it returns `()` and logs its failures. A user
53/// row that has already been committed must not be reported as a failed
54/// creation because the mail queue was unreachable, and the resend endpoint
55/// must answer identically whatever happened (D-15). The account stays
56/// activatable either way — `resend-verification` mints a fresh token.
57///
58/// The `action_url` is a relative frontend route carrying the raw token and the
59/// tenant id, matching the `VerifyEmailPage` route (`/auth/verify-email?token=…
60/// &tenant_id=…`) and the shape `gdpr.rs` uses for its cancel link.
61pub(crate) async fn enqueue_verification_email<C: Connection + Clone>(
62    state: &AppState<C>,
63    tenant_id: Uuid,
64    user_id: Uuid,
65    email: &str,
66) {
67    let (raw_token, expires_at) = match state
68        .mail
69        .email_verification_service
70        .initiate_verification(tenant_id, user_id)
71        .await
72    {
73        Ok(pair) => pair,
74        Err(e) => {
75            tracing::warn!(
76                error = %e,
77                %tenant_id,
78                %user_id,
79                "could not mint an email-verification token; no activation mail enqueued"
80            );
81            return;
82        }
83    };
84
85    let org_id = match state.tenant_repo.get_by_id(tenant_id).await {
86        Ok(tenant) => tenant.organization_id,
87        Err(e) => {
88            // Nil rather than abandoning the send: the mail consumer resolves
89            // the effective email config by (org, tenant) and a tenant-level
90            // config still matches. Losing the org fallback is better than
91            // losing the only activation link the account will be offered.
92            tracing::warn!(
93                error = %e,
94                %tenant_id,
95                "failed to resolve org_id for activation mail; using nil"
96            );
97            Uuid::nil()
98        }
99    };
100
101    let verify_url = format!("/auth/verify-email?token={raw_token}&tenant_id={tenant_id}");
102
103    let msg = OutboundMailMessage {
104        mail_type: MailType::EmailVerification,
105        tenant_id,
106        org_id,
107        user_id,
108        to_address: email.to_owned(),
109        template_context: serde_json::json!({
110            "token": raw_token,
111            "action_url": verify_url,
112            "expiry_time": expires_at.to_rfc3339(),
113        }),
114        attempt_count: 0,
115        enqueued_at: Utc::now(),
116    };
117
118    if let Err(e) = state.mail.mail_outbound_publisher.publish(msg).await {
119        tracing::warn!(error = %e, "failed to enqueue activation mail; continuing");
120    } else {
121        tracing::debug!(%tenant_id, %user_id, "activation email enqueued");
122    }
123}

129/// `POST /api/v1/auth/verify-email`
130///
131/// Verifies a user's email using a one-time token. The token is
132/// consumed atomically — replaying the same token returns an error.
133#[utoipa::path(
134    post,
135    path = "/api/v1/auth/verify-email",
136    tag = "auth",
137    request_body = VerifyEmailRequest,
138    responses(
139        (status = 200, description = "Email verified successfully"),
140        (status = 400, description = "Invalid or expired token"),
141    )
142)]
143pub async fn verify_email<C: Connection + Clone>(
144    state: web::Data<AppState<C>>,
145    body: web::Json<VerifyEmailRequest>,
146) -> Result<HttpResponse, AxiamApiError> {
147    let req = body.into_inner();
148
149    // QUAL-07: EmailVerificationService is now a hoisted AppState singleton.
150    state
151        .mail
152        .email_verification_service
153        .verify_email(req.tenant_id, &req.token)
154        .await?;
155
156    Ok(HttpResponse::Ok().json(serde_json::json!({ "verified": true })))
157}

159/// `POST /api/v1/auth/resend-verification`
160///
161/// Creates a new email verification token and enqueues it for async
162/// delivery via the mail queue (D-14).  Always returns `{"sent": true}`
163/// to prevent email enumeration (D-15) — regardless of whether the email
164/// exists, is already verified, or has hit a rate limit.
165#[utoipa::path(
166    post,
167    path = "/api/v1/auth/resend-verification",
168    tag = "auth",
169    request_body = ResendVerificationRequest,
170    responses(
171        (status = 200, description = "Verification email enqueued"),
172    )
173)]
174pub async fn resend_verification<C: Connection + Clone>(
175    state: web::Data<AppState<C>>,
176    body: web::Json<ResendVerificationRequest>,
177) -> Result<HttpResponse, AxiamApiError> {
178    let req = body.into_inner();
179
180    // QUAL-07: EmailVerificationService is now a hoisted AppState singleton.
181    match state
182        .mail
183        .email_verification_service
184        .resend_verification(req.tenant_id, &req.email)
185        .await
186    {
187        Ok(Some((_raw_token, user_id, _expires_at))) => {
188            // The token minted above is discarded and a fresh one minted inside
189            // the shared helper. That costs one extra row and buys the guarantee
190            // that a resent activation email is byte-for-byte the one account
191            // creation sends — same URL shape, same context keys, same template.
192            // Two hand-rolled copies of this block is how they drift.
193            enqueue_verification_email(&state, req.tenant_id, user_id, &req.email).await;
194        }
195        Ok(None) => {
196            // User not found or already verified — silently ignore (D-15).
197            tracing::debug!(
198                email = %req.email,
199                "resend-verification: no action (unknown or verified)"
200            );
201        }
202        Err(AxiamError::RateLimited) => {
203            // Swallow rate-limit to prevent user enumeration via
204            // differential 429 responses (D-15).
205            tracing::debug!(
206                email = %req.email,
207                "resend-verification: rate-limited (suppressed)"
208            );
209        }
210        Err(e) => return Err(e.into()),
211    }
212
213    // Always return identical 200 regardless of outcome (D-15).
214    Ok(HttpResponse::Ok().json(serde_json::json!({ "sent": true })))
215}

217/// `POST /api/v1/users/me/resend-verification`
218///
219/// Resend the caller's **own** verification email, and say what happened.
220///
221/// # Why this exists beside the public endpoint
222///
223/// `POST /auth/resend-verification` answers a constant `200 {"sent": true}`
224/// whatever the outcome, and it must: it takes an address from an
225/// unauthenticated caller, so a 404 for "no such user" or a 429 for "rate
226/// limited" would be an oracle for which addresses have accounts.
227///
228/// That reasoning does not apply here. The caller is authenticated and asking
229/// about the address on its own record — it already knows the account exists,
230/// because it is signed in to it. Reusing the enumeration-safe endpoint for
231/// this is what made the profile page's button report success while doing
232/// nothing: the address was already verified, or the account was locked, or the
233/// daily limit was reached, and the response looked identical in every case.
234///
235/// So this one tells the truth:
236///
237/// * `200 {"sent": true}` — a token was minted and the mail enqueued.
238/// * `409` — the address is already verified, or the account is in a state that
239///   must not be sent a live token.
240/// * `429` — the daily resend limit is reached.
241///
242/// None of those disclose anything the caller did not bring with it.
243///
244/// Note what "sent" means: the mail is *enqueued*. Delivery is asynchronous and
245/// can still fail at the provider — which is worth knowing, because a mail
246/// queue that accepts everything and a provider that rejects it looks exactly
247/// like this endpoint working.
248#[utoipa::path(
249    post,
250    path = "/api/v1/users/me/resend-verification",
251    tag = "users",
252    responses(
253        (status = 200, description = "Verification email enqueued"),
254        (status = 409, description = "Already verified, or the account may not be sent one"),
255        (status = 429, description = "Daily resend limit reached"),
256    ),
257    security(("bearer" = []))
258)]
259pub async fn resend_own_verification<C: Connection + Clone>(
260    user: AuthenticatedUser,
261    state: web::Data<AppState<C>>,
262) -> Result<HttpResponse, AxiamApiError> {
263    // No permission check: this is self-service on the caller's own record, the
264    // same shape as the rest of `/users/me`. The address is read from the
265    // record rather than taken from the request — a caller that could name the
266    // address would be able to mail an arbitrary one from an authenticated
267    // session.
268    let me = state
269        .user_repo
270        .get_by_id(user.principal_tenant_id, user.user_id)
271        .await?;
272
273    match state
274        .mail
275        .email_verification_service
276        .resend_verification(user.principal_tenant_id, &me.email)
277        .await
278    {
279        Ok(Some((_raw_token, user_id, _expires_at))) => {
280            enqueue_verification_email(&state, user.principal_tenant_id, user_id, &me.email).await;
281            Ok(HttpResponse::Ok().json(serde_json::json!({ "sent": true })))
282        }
283        // The service returns `None` for "nothing to do", which here can only
284        // mean the address is already verified or the account's status forbids
285        // it — the "unknown user" case is impossible for a caller reading its
286        // own record.
287        Ok(None) => Err(AxiamError::AlreadyExists {
288            entity: "a verified email address for this account, or an account \
289                     state that permits a verification email"
290                .into(),
291        }
292        .into()),
293        Err(AxiamError::RateLimited) => Err(AxiamError::RateLimited.into()),
294        Err(e) => Err(e.into()),
295    }
296}

310    /// Fake mail publisher for test assertions.
311    #[derive(Clone, Default)]
312    struct RecordingPublisher {
313        sent: Arc<Mutex<Vec<OutboundMailMessage>>>,
314    }
315
316    impl RecordingPublisher {
317        fn new() -> Self {
318            Self {
319                sent: Arc::new(Mutex::new(Vec::new())),
320            }
321        }
322
323        fn count(&self) -> usize {
324            self.sent.lock().unwrap().len()
325        }
326    }
327
328    impl MailPublisher for RecordingPublisher {
329        async fn publish(&self, msg: OutboundMailMessage) -> AxiamResult<()> {
330            self.sent.lock().unwrap().push(msg);
331            Ok(())
332        }
333    }

342        // Simulate the unknown-address / already-verified branch:
343        // `svc.resend_verification` returns `Ok(None)` → no enqueue, but

346        assert!(

348            "unknown-email response MUST NOT contain a token field (D-15)"

352            Some(true),
353            "unknown-email response must be {{\"sent\": true}}"

395    // -----------------------------------------------------------------------
396    // PHASE-DEFINING (23-RESEARCH Pattern 6 / Pitfall 3): action_url
397    // substitution — a runtime assertion on the RENDERED email string, not

401    /// The action_url built by `resend_verification` (mirroring gdpr.rs's
402    /// cancel_url) fully substitutes into the rendered verification email
403    /// (`MailType::EmailVerification` -> `TemplateKind::Activation`): the
404    /// rendered link contains the token + tenant_id, and the literal
405    /// `{{action_url}}` mustache placeholder is gone.
406    #[tokio::test]
407    async fn action_url_is_substituted_in_rendered_verification_email() {
408        use axiam_core::models::email_template::TemplateKind;
409        use axiam_email::template::{TemplateContext, render_email, resolve_template};
410
411        let raw_token = "verify-token-substitution-check";
412        let tenant_id = Uuid::new_v4();
413        let expires_at = Utc::now() + chrono::Duration::hours(24);
414
415        // Exactly what `resend_verification` now builds into template_context.
416        let verify_url = format!("/auth/verify-email?token={raw_token}&tenant_id={tenant_id}");
417        let template_context = serde_json::json!({
418            "token": raw_token,
419            "action_url": verify_url,
420            "expiry_time": expires_at.to_rfc3339(),
421        });
422
423        // Mirror `axiam-amqp::mail_consumer::build_template_context`'s
424        // JSON-object -> TemplateContext conversion (string values as-is).
425        let mut ctx = TemplateContext::new();
426        if let serde_json::Value::Object(obj) = &template_context {
427            for (k, v) in obj {
428                if let serde_json::Value::String(s) = v {
429                    ctx.insert(k.clone(), s.clone());
430                }
431            }
432        }
433
434        // MailType::EmailVerification -> TemplateKind::Activation
435        // (crates/axiam-amqp/src/mail_consumer.rs::template_kind_for).
436        let template = resolve_template(TemplateKind::Activation, None, None);
437        let rendered = render_email(&template, "user@example.com", &ctx);
438        let html = rendered.html_body.expect("html body must be rendered");
439        let text = rendered.text_body.expect("text body must be rendered");
440
441        for body in [&html, &text] {
442            assert!(
443                body.contains(raw_token),
444                "rendered email must contain the raw verification token: {body}"
445            );
446            assert!(
447                body.contains(&tenant_id.to_string()),
448                "rendered email must contain the tenant_id: {body}"
449            );
450            assert!(
451                !body.contains("{{action_url}}"),
452                "rendered email MUST NOT contain the unsubstituted action_url \
453                 placeholder (23-RESEARCH Pitfall 3): {body}"
454            );
455        }
456    }
