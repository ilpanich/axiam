86/// SHA-256 hex of `raw_token`.
87pub fn sha256_hex(raw: &str) -> String {
88    let mut h = Sha256::new();
89    h.update(raw.as_bytes());
90    hex::encode(h.finalize())
91}

93/// Generate a cryptographically-random 256-bit (32-byte) cancel token,
94/// hex-encoded as a 64-character string (CQ-B39).
95///
96/// Replaces the previous `Uuid::new_v4().to_string()` (128-bit) token.
97fn generate_cancel_token() -> String {
98    use rand::RngExt;
99    let bytes: [u8; 32] = rand::rng().random();
100    hex::encode(bytes)
101}

103/// Append a GDPR audit log entry, logging any failure without propagating it.
104///
105/// Factored out of the individual GDPR handlers to eliminate the repeated
106/// audit-append block pattern (CQ-B39).  The fire-and-forget `let _ = …`
107/// pattern is intentional: an audit failure must not block the user response,
108/// but failures are logged at `error!` level since GDPR audit trails are legally
109/// significant (CQ-B31 / T-12-01).
110async fn append_gdpr_audit<C: Connection + Clone>(
111    audit_repo: &SurrealAuditLogRepository<C>,
112    tenant_id: Uuid,
113    actor_id: Uuid,
114    action: &str,
115    resource_id: Option<Uuid>,
116    metadata: Option<serde_json::Value>,
117) {
118    if let Err(e) = audit_repo
119        .append(axiam_core::models::audit::CreateAuditLogEntry {
120            tenant_id,
121            actor_id,
122            actor_type: ActorType::User,
123            action: action.into(),
124            resource_id,
125            outcome: AuditOutcome::Success,
126            ip_address: None,
127            metadata,
128        })
129        .await
130    {
131        tracing::error!(
132            error = %e,
133            %tenant_id,
134            "gdpr: failed to write audit log for GDPR request (legally significant)"
135        );
136    }
137}

143/// Environment variable naming the append-only dead-letter file for the GDPR
144/// erasure audit event. Intended to point at a mounted volume so the file
145/// survives container restarts. Absent = the file sink is skipped for this
146/// event (the structured tracing event sink below still fires
147/// unconditionally on failure).
148pub const GDPR_AUDIT_DLQ_FILE_ENV: &str = "AXIAM__GDPR_AUDIT_DLQ_FILE";

150/// Injectable seam for the erasure audit DB-write (SECHRD-12).
151///
152/// `SurrealAuditLogRepository<C>` is the production implementation (forwards
153/// to [`AuditLogRepository::append`]); tests inject a failing double to drive
154/// the dead-letter path in [`write_erasure_audit_with_dlq`] without a live
155/// broken database. Defined here (not in `axiam-core::repository`) so it
156/// stays out of the generic repository trait surface owned by another plan.
157pub trait AuditWriteSink: Send + Sync {
158    fn write(
159        &self,
160        entry: CreateAuditLogEntry,
161    ) -> impl Future<Output = AxiamResult<AuditLogEntry>> + Send;
162}
163
164impl<C: Connection> AuditWriteSink for SurrealAuditLogRepository<C> {
165    fn write(
166        &self,
167        entry: CreateAuditLogEntry,
168    ) -> impl Future<Output = AxiamResult<AuditLogEntry>> + Send {
169        self.append(entry)
170    }
171}

173/// Write the erasure audit record via `sink`. If the DB write fails, the
174/// record is dead-lettered to BOTH an append-only local file AND a
175/// structured `tracing` audit event (D-02), so a legally-significant
176/// erasure event is never silently lost to a transient DB failure (T-24-61).
177///
178/// Never propagates the DB error — the caller (the cleanup ticker, T-04-36)
179/// must not panic or abort the sweep on this failure path; the two
180/// dead-letter sinks ARE the durability guarantee for this branch.
181pub async fn write_erasure_audit_with_dlq<S: AuditWriteSink>(sink: &S, entry: CreateAuditLogEntry) {
182    let dlq_entry = entry.clone();
183    if let Err(e) = sink.write(entry).await {
184        tracing::error!(
185            error = %e,
186            tenant_id = %dlq_entry.tenant_id,
187            action = %dlq_entry.action,
188            "gdpr: erasure audit DB-write failed — dead-lettering to append-only file + \
189             structured event (legally significant, SECHRD-12)"
190        );
191        dead_letter_erasure_audit(&dlq_entry, &e);
192    }
193}
194
195/// Dead-letter a failed erasure audit record to the two durable sinks (D-02).
196fn dead_letter_erasure_audit(entry: &CreateAuditLogEntry, db_error: &AxiamError) {
197    // Sink 1: append-only local file on a mounted volume (T-24-62). Opened
198    // with `.append(true)` — an existing file is never truncated/rewritten,
199    // matching AXIAM's append-only audit posture.
200    match std::env::var(GDPR_AUDIT_DLQ_FILE_ENV) {
201        Ok(path) => match serde_json::to_string(entry) {
202            Ok(line) => match OpenOptions::new().create(true).append(true).open(&path) {
203                Ok(mut file) => {
204                    if let Err(e) = writeln!(file, "{line}") {
205                        tracing::error!(
206                            error = %e,
207                            path = %path,
208                            "gdpr: failed to append erasure audit dead-letter file"
209                        );
210                    }
211                }
212                Err(e) => {
213                    tracing::error!(
214                        error = %e,
215                        path = %path,
216                        "gdpr: failed to open erasure audit dead-letter file"
217                    );
218                }
219            },
220            Err(e) => {
221                tracing::error!(
222                    error = %e,
223                    "gdpr: failed to serialize erasure audit dead-letter record"
224                );
225            }
226        },
227        Err(_) => {
228            tracing::warn!(
229                env_var = GDPR_AUDIT_DLQ_FILE_ENV,
230                "gdpr: erasure audit dead-letter FILE sink skipped (env var not set); \
231                 structured event sink below still fires"
232            );
233        }
234    }
235
236    // Sink 2: structured `tracing` audit event — the "audit syslog" sink
237    // (RESEARCH Assumption A4: satisfied by a structured tracing JSON event
238    // captured by the container log driver, NOT a literal syslog(3) socket;
239    // the distroless deployment has no local syslogd, per 06-01 decisions).
240    tracing::error!(
241        target: "axiam.audit.dlq",
242        tenant_id = %entry.tenant_id,
243        actor_id = %entry.actor_id,
244        action = %entry.action,
245        resource_id = ?entry.resource_id,
246        db_error = %db_error,
247        "gdpr_audit_dlq"
248    );
249}

255/// `POST /api/v1/account/export`
256///
257/// Enqueues an async GDPR Art. 15 data-export job.  Returns immediately with
258/// `{"queued": true}`; the cleanup sweep generates the encrypted JSON, stores
259/// it, and emails a single-use 24 h download link (D-12).
260///
261/// Self-service (own account) or admin with `gdpr:export` permission.
262#[utoipa::path(
263    post,
264    path = "/api/v1/account/export",
265    tag = "gdpr",
266    request_body = inline(serde_json::Value),
267    responses(
268        (status = 200, description = "Export job enqueued"),
269        (status = 403, description = "Forbidden"),
270    ),
271    security(("bearer" = []))
272)]
273#[allow(clippy::too_many_arguments)]
274pub async fn request_account_export<C: Connection + Clone>(
275    auth_user: AuthenticatedUser,
276    authz: AuthzData,
277    state: web::Data<AppState<C>>,
278    body: web::Json<ExportRequest>,
279) -> Result<HttpResponse, AxiamApiError> {
280    let req = body.into_inner();
281    let target_id = req.user_id.unwrap_or(auth_user.user_id);
282
283    // Ownership check: self OR gdpr:export permission.
284    if !is_own_resource(&auth_user, target_id) {
285        RequirePermission::new("gdpr:export", Uuid::nil())
286            .check(&auth_user, authz.get_ref().as_ref())
287            .await?;
288    }
289
290    // The subject's data lives in the subject's own tenant. For a self-service
291    // request that is the caller's tenant, which is not the one an
292    // organization-level principal has selected — see [`user_scope_tenant`].
293    // An export job written into the selected tenant would be queued against a
294    // subject that tenant does not have.
295    let scope = user_scope_tenant(&auth_user, target_id);
296
297    // CQ-B39: Deduplicate concurrent export requests — reject if a queued
298    // export already exists for this user to avoid duplicate processing.
299    if state
300        .gdpr
301        .export_job_repo
302        .has_pending_for_user(scope, target_id)
303        .await?
304    {
305        return Err(AxiamError::AlreadyExists {
306            entity: "export_job".into(),
307        }
308        .into());
309    }
310
311    // Create a queued export job.
312    state
313        .gdpr
314        .export_job_repo
315        .create(CreateExportJob {
316            tenant_id: scope,
317            user_id: target_id,
318        })
319        .await?;
320
321    // Audit: gdpr.data_export_requested.
322    append_gdpr_audit(
323        &state.audit_repo,
324        scope,
325        auth_user.user_id,
326        "gdpr.data_export_requested",
327        Some(target_id),
328        Some(serde_json::json!({ "subject_id": target_id.to_string() })),
329    )
330    .await;
331
332    Ok(HttpResponse::Ok().json(QueuedResponse { queued: true }))
333}

335/// `GET /api/v1/account/export/{token}`
336///
337/// Single-use download of the GDPR export blob (D-13).  Looks up the job by
338/// SHA-256(token), verifies it is ready and not expired/used, decrypts the
339/// blob, marks it as downloaded, and deletes the record.
340///
341/// Requires authentication (self or admin with `gdpr:export`).
342#[utoipa::path(
343    get,
344    path = "/api/v1/account/export/{token}",
345    tag = "gdpr",
346    params(("token" = String, Path, description = "Single-use download token")),
347    responses(
348        (status = 200, description = "Export JSON"),
349        (status = 404, description = "Token not found or expired"),
350    ),
351    security(("bearer" = []))
352)]
353#[allow(clippy::too_many_arguments)]
354pub async fn download_account_export<C: Connection + Clone>(
355    auth_user: AuthenticatedUser,
356    authz: AuthzData,
357    state: web::Data<AppState<C>>,
358    path: web::Path<DownloadPath>,
359) -> Result<HttpResponse, AxiamApiError> {
360    let token = path.into_inner().token;
361    let token_hash = sha256_hex(&token);
362
363    // A download link mailed to the caller points at a job in the caller's own
364    // tenant. Looking it up in the tenant an organization-level principal has
365    // selected found nothing and answered 404 for a link that was perfectly
366    // valid — so the search is scoped to the principal's tenant first, and only
367    // falls back to the acting tenant for an administrator downloading somebody
368    // else's export. `find_by_download_token_hash` matches on a SHA-256 of a
369    // 256-bit token, so neither lookup can be steered by the caller.
370    let job = match state
371        .gdpr
372        .export_job_repo
373        .find_by_download_token_hash(auth_user.principal_tenant_id, &token_hash)
374        .await?
375    {
376        Some(job) => Some(job),
377        None if auth_user.tenant_id != auth_user.principal_tenant_id => {
378            state
379                .gdpr
380                .export_job_repo
381                .find_by_download_token_hash(auth_user.tenant_id, &token_hash)
382                .await?
383        }
384        None => None,
385    };
386    let job = job.ok_or_else(|| AxiamError::NotFound {
387        entity: "export_job".into(),
388        id: "by-token".into(),
389    })?;
390
391    // Ownership check.
392    if !is_own_resource(&auth_user, job.user_id) {
393        RequirePermission::new("gdpr:export", Uuid::nil())
394            .check(&auth_user, authz.get_ref().as_ref())
395            .await?;
396    }
397
398    // Validate status and expiry (D-13: single-use, 24h TTL).
399    use axiam_core::models::gdpr::ExportJobStatus;
400    if job.status != ExportJobStatus::Ready {
401        return Err(AxiamError::AuthorizationDenied {
402            reason: "export token already used or not ready".into(),
403            action: None,
404            resource_id: None,
405        }
406        .into());
407    }
408    if let Some(expires_at) = job.expires_at
409        && expires_at < Utc::now()
410    {
411        return Err(AxiamError::AuthorizationDenied {
412            reason: "export token expired".into(),
413            action: None,
414            resource_id: None,
415        }
416        .into());
417    }
418
419    // Decrypt the blob.
420    let (nonce, ciphertext) = match (&job.blob_nonce, &job.encrypted_blob) {
421        (Some(n), Some(ct)) => (n.clone(), ct.clone()),
422        _ => {
423            return Err(AxiamError::Internal("export blob missing on ready job".into()).into());
424        }
425    };
426    let key = state
427        .mail
428        .email_encryption_key
429        .as_ref()
430        .ok_or_else(|| AxiamError::Internal("export encryption key not configured".into()))?;
431
432    let plaintext_bytes = decrypt_separate(key, &nonce, &ciphertext)
433        .map_err(|e| AxiamError::Internal(format!("export decrypt failed: {e}")))?;
434
435    // Atomic single-use consume: UPDATE WHERE status = 'ready' + DELETE.
436    // If 0 rows were updated the token was already consumed (TOCTTOU-safe,
437    // D-13 / CQ-B38 / REQ-14 AC-5).
438    let consumed = state
439        .gdpr
440        .export_job_repo
441        .consume_ready_and_delete(job.id)
442        .await?;
443    if !consumed {
444        return Err(AxiamError::AuthorizationDenied {
445            reason: "export token already consumed".into(),
446            action: None,
447            resource_id: None,
448        }
449        .into());
450    }
451
452    let plaintext = String::from_utf8(plaintext_bytes)
453        .map_err(|e| AxiamError::Internal(format!("export UTF-8 decode: {e}")))?;
454
455    Ok(HttpResponse::Ok()
456        .content_type("application/json")
457        .body(plaintext))
458}

460/// `POST /api/v1/account/delete`
461///
462/// Initiates Art. 17 erasure: immediately disables the account, revokes all
463/// sessions, emails a single-use cancel link, and schedules purge at +30 d
464/// (D-07/D-08/D-09).
465///
466/// Self-service (own account) or admin with `users:erase` permission.
467#[utoipa::path(
468    post,
469    path = "/api/v1/account/delete",
470    tag = "gdpr",
471    request_body = inline(serde_json::Value),
472    responses(
473        (status = 200, description = "Deletion scheduled"),
474        (status = 403, description = "Forbidden"),
475    ),
476    security(("bearer" = []))
477)]
478#[allow(clippy::too_many_arguments)]
479pub async fn request_account_delete<C: Connection + Clone>(
480    auth_user: AuthenticatedUser,
481    authz: AuthzData,
482    state: web::Data<AppState<C>>,
483    body: web::Json<DeleteRequest>,
484) -> Result<HttpResponse, AxiamApiError> {
485    let req = body.into_inner();
486    let target_id = req.user_id.unwrap_or(auth_user.user_id);
487
488    // Ownership check: self OR users:erase permission.
489    if !is_own_resource(&auth_user, target_id) {
490        RequirePermission::new("users:erase", Uuid::nil())
491            .check(&auth_user, authz.get_ref().as_ref())
492            .await?;
493    }
494
495    // The subject's record, the deletion row, the sessions and the audit entry
496    // all belong to the subject's own tenant — for a self-service erasure the
497    // caller's, not the one an organization-level principal has selected. See
498    // [`user_scope_tenant`].
499    let scope = user_scope_tenant(&auth_user, target_id);
500
501    // Load target user to get email for the cancel mail.
502    let target_user = state.user_repo.get_by_id(scope, target_id).await?;
503
504    // Schedule the purge at the end of the tenant's grace window (D-08).
505    //
506    // This was a hard-coded 30 days, which left the "cancel a pending
507    // deletion" control in the admin UI referring to a duration no operator
508    // could see or change. It is now `privacy.deletion_grace_period_days` —
509    // an org baseline a tenant may only lower — and still 30 by default, so
510    // an upgrade changes nothing.
511    //
512    // A read failure falls back to that default rather than refusing the
513    // erasure: an unavailable settings row must not stop a data subject
514    // exercising Art. 17, and the default is the value every deployment had
515    // until now.
516    let grace_days = {
517        use axiam_core::repository::{SettingsRepository as _, TenantRepository as _};
518        let resolved = async {
519            let tenant = state.tenant_repo.get_by_id(scope).await.ok()?;
520            let settings = state
521                .settings_repo
522                .get_effective_settings(tenant.organization_id, scope)
523                .await
524                .ok()?;
525            Some(settings.privacy.deletion_grace_period_days)
526        }
527        .await;
528        resolved.unwrap_or_else(|| {
529            tracing::warn!(
530                tenant_id = %scope,
531                "could not resolve the erasure grace window; using the 30-day default"
532            );
533            30
534        })
535    };
536    let scheduled_purge_at = Utc::now() + chrono::Duration::days(i64::from(grace_days));
537
538    // Generate cancel token: 256-bit random token, hex-encoded (CQ-B39).
539    // Only the SHA-256 hash is stored in the DB; the raw token is emailed.
540    let raw_cancel_token = generate_cancel_token();
541    let cancel_token_hash = sha256_hex(&raw_cancel_token);
542
543    // D-14: mark the user deletion-pending AND create the account_deletion
544    // row (holding the cancel_token_hash) in ONE transaction — a create
545    // failure (e.g. a duplicate pending-deletion conflict) must never
546    // strand the user in deletion_pending=true with no account_deletion row
547    // to hold a cancellable token (CQ-B39 residual / uncancellable purge).
548    state
549        .gdpr
550        .account_deletion_repo
551        .create_with_pending_flag(scope, target_id, scheduled_purge_at, cancel_token_hash)
552        .await?;
553
554    // Revoke all sessions immediately (D-08: account disabled). Kept as a
555    // separate, subsequent call — not part of the D-14 strand-risk
556    // transaction: a failure here leaves an unrevoked session, a lesser and
557    // pre-existing concern, not an uncancellable purge.
558    state
559        .auth_service
560        .revoke_all_sessions(scope, target_id)
561        .await?;
562
563    // Resolve org_id for the mail message.
564    let org_id = match state.tenant_repo.get_by_id(scope).await {
565        Ok(tenant) => tenant.organization_id,
566        Err(e) => {
567            tracing::warn!(error = %e, "failed to resolve org_id for delete-cancel mail");
568            Uuid::nil()
569        }
570    };
571
572    // Enqueue the cancel-link email (D-09).
573    let cancel_url = format!(
574        "/api/v1/auth/account/delete/cancel?token={}",
575        raw_cancel_token
576    );
577    let msg = OutboundMailMessage {
578        mail_type: MailType::DeletionCancel,
579        tenant_id: scope,
580        org_id,
581        user_id: target_id,
582        to_address: target_user.email.clone(),
583        template_context: serde_json::json!({
584            "action_url": cancel_url,
585            "expiry_time": scheduled_purge_at.to_rfc3339(),
586        }),
587        attempt_count: 0,
588        enqueued_at: Utc::now(),
589    };
590    if let Err(e) = state.mail.mail_outbound_publisher.publish(msg).await {
591        tracing::warn!(error = %e, "failed to enqueue delete-cancel email; continuing");
592    }
593
594    // Audit: gdpr.erasure_requested.
595    append_gdpr_audit(
596        &state.audit_repo,
597        scope,
598        auth_user.user_id,
599        "gdpr.erasure_requested",
600        Some(target_id),
601        Some(serde_json::json!({
602            "subject_id": target_id.to_string(),
603            "scheduled_purge_at": scheduled_purge_at.to_rfc3339(),
604        })),
605    )
606    .await;
607
608    Ok(HttpResponse::Ok().json(ScheduledResponse { scheduled: true }))
609}

626    )
627)]
628pub async fn cancel_account_delete<C: Connection + Clone>(
629    state: web::Data<AppState<C>>,
630    query: web::Query<CancelQuery>,
631) -> Result<HttpResponse, AxiamApiError> {
632    let token_hash = sha256_hex(&query.token);
633
634    // Global lookup by token hash (no auth context — public endpoint, D-09).
635    let deletion = state
636        .gdpr
637        .account_deletion_repo
638        .find_by_token_hash_global(&token_hash)
639        .await?
640        .ok_or_else(|| AxiamError::NotFound {
641            entity: "account_deletion".into(),
642            id: "by-token".into(),
643        })?;
644
645    // Validate: must be pending and within grace window.
646    if deletion.status != AccountDeletionStatus::Pending {
647        return Err(AxiamError::AuthorizationDenied {
648            reason: "cancel token already used".into(),
649            action: None,
650            resource_id: None,
651        }
652        .into());
653    }
654    if deletion.scheduled_purge_at < Utc::now() {
655        return Err(AxiamError::AuthorizationDenied {
656            reason: "grace window expired — cannot cancel".into(),
657            action: None,
658            resource_id: None,
659        }
660        .into());
661    }
662
663    // Abort deletion: mark cancelled (single-use) + re-enable account.
664    state
665        .gdpr
666        .account_deletion_repo
667        .mark_cancelled(deletion.tenant_id, deletion.id)
668        .await?;
669
670    state
671        .user_repo
672        .clear_deletion_pending(deletion.tenant_id, deletion.user_id)
673        .await?;
674
675    Ok(HttpResponse::Ok().json(CancelledResponse { cancelled: true }))
676}
