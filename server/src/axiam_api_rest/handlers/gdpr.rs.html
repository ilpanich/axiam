85/// SHA-256 hex of `raw_token`.
86pub fn sha256_hex(raw: &str) -> String {
87    let mut h = Sha256::new();
88    h.update(raw.as_bytes());
89    hex::encode(h.finalize())
90}

92/// Generate a cryptographically-random 256-bit (32-byte) cancel token,
93/// hex-encoded as a 64-character string (CQ-B39).
94///
95/// Replaces the previous `Uuid::new_v4().to_string()` (128-bit) token.
96fn generate_cancel_token() -> String {
97    use rand::RngExt;
98    let bytes: [u8; 32] = rand::rng().random();
99    hex::encode(bytes)
100}

102/// Record a GDPR request in the audit log. A refused append is dead-lettered,
103/// never propagated.
104///
105/// Factored out of the individual GDPR handlers to eliminate the repeated
106/// audit-append block pattern (CQ-B39).  An audit failure must not block the
107/// user response, but a GDPR audit trail is legally significant (CQ-B31 /
108/// T-12-01), so a record the datastore refuses goes to the dead-letter file and
109/// the `axiam.audit.dlq` event through [`write_audit_with_dead_letter`], the
110/// route the erasure records already take (P23W5-A8, T19.27).
111async fn append_gdpr_audit<S: AuditWriteSink>(
112    audit_sink: &S,
113    tenant_id: Uuid,
114    actor_id: Uuid,
115    action: &str,
116    resource_id: Option<Uuid>,
117    metadata: Option<serde_json::Value>,
118) {
119    write_audit_with_dead_letter(
120        audit_sink,
121        CreateAuditLogEntry {
122            tenant_id,
123            actor_id,
124            actor_type: ActorType::User,
125            action: action.into(),
126            resource_id,
127            outcome: AuditOutcome::Success,
128            ip_address: None,
129            metadata,
130        },
131    )
132    .await;
133}

139/// Environment variable naming the append-only dead-letter file for the GDPR
140/// audit records (erasure, export and erasure requests, tenant deletion).
141/// Intended to point at a mounted volume so the file survives container
142/// restarts. The request-audit middleware's lost rows go to the same file
143/// (T-108). Absent = the file sink is skipped for this event (the structured
144/// tracing event sink below still fires unconditionally on failure).
145pub const GDPR_AUDIT_DLQ_FILE_ENV: &str = axiam_audit::dead_letter::DEAD_LETTER_FILE_ENV;

147/// Injectable seam for the GDPR audit DB-write (SECHRD-12).
148///
149/// `SurrealAuditLogRepository<C>` is the production implementation (forwards
150/// to [`AuditLogRepository::append`]); tests inject a failing double to drive
151/// the dead-letter path in [`write_audit_with_dead_letter`] without a live
152/// broken database. Defined here (not in `axiam-core::repository`) so it
153/// stays out of the generic repository trait surface owned by another plan.
154pub trait AuditWriteSink: Send + Sync {
155    fn write(
156        &self,
157        entry: CreateAuditLogEntry,
158    ) -> impl Future<Output = AxiamResult<AuditLogEntry>> + Send;
159}
160
161impl<C: Connection> AuditWriteSink for SurrealAuditLogRepository<C> {
162    fn write(
163        &self,
164        entry: CreateAuditLogEntry,
165    ) -> impl Future<Output = AxiamResult<AuditLogEntry>> + Send {
166        self.append(entry)
167    }
168}

170/// Write a GDPR audit record via `sink`. If the DB write fails, the record is
171/// dead-lettered to BOTH an append-only local file AND a structured `tracing`
172/// audit event (D-02), so a legally-significant event is never silently lost
173/// to a transient DB failure (T-24-61).
174///
175/// The records that take this route: `gdpr.user_pseudonymized` (the cleanup
176/// sweep), `tenants.deleted`, and the two request records
177/// `gdpr.data_export_requested` and `gdpr.erasure_requested` (P23W5-A8).
178///
179/// Never propagates the DB error — the callers (the cleanup ticker, T-04-36,
180/// and the request handlers, whose user-visible work is already done) must not
181/// panic or abort on this failure path; the two dead-letter sinks ARE the
182/// durability guarantee for this branch.
183pub async fn write_audit_with_dead_letter<S: AuditWriteSink>(sink: &S, entry: CreateAuditLogEntry) {
184    let dlq_entry = entry.clone();
185    if let Err(e) = sink.write(entry).await {
186        tracing::error!(
187            error = %e,
188            tenant_id = %dlq_entry.tenant_id,
189            action = %dlq_entry.action,
190            "gdpr: audit DB-write failed — dead-lettering to append-only file + \
191             structured event (legally significant, SECHRD-12)"
192        );
193        dead_letter_audit(&dlq_entry, &e);
194    }
195}
196
197/// Dead-letter a failed GDPR audit record to the two durable sinks (D-02).
198fn dead_letter_audit(entry: &CreateAuditLogEntry, db_error: &AxiamError) {
199    // Sink 1: append-only local file on a mounted volume (T-24-62). Opened
200    // with `.append(true)` — an existing file is never truncated/rewritten,
201    // matching AXIAM's append-only audit posture. The line is the one the
202    // request-audit worker's dead letters use (`axiam_audit::dead_letter`).
203    match std::env::var(GDPR_AUDIT_DLQ_FILE_ENV) {
204        Ok(path) => {
205            // The whole budget, the reserve above the request rows' share
206            // included (R1W2-02). An invalid value failed the boot already.
207            let max_bytes = axiam_audit::dead_letter::max_bytes_from_env()
208                .unwrap_or(axiam_audit::dead_letter::DEFAULT_MAX_BYTES);
209            if let Err(e) =
210                axiam_audit::dead_letter::append_blocking(Path::new(&path), entry, max_bytes)
211            {
212                tracing::error!(
213                    error = %e,
214                    path = %path,
215                    "gdpr: failed to append audit dead-letter file"
216                );
217            }
218        }
219        Err(_) => {
220            tracing::warn!(
221                env_var = GDPR_AUDIT_DLQ_FILE_ENV,
222                "gdpr: audit dead-letter FILE sink skipped (env var not set); \
223                 structured event sink below still fires"
224            );
225        }
226    }
227
228    // Sink 2: structured `tracing` audit event — the "audit syslog" sink
229    // (RESEARCH Assumption A4: satisfied by a structured tracing JSON event
230    // captured by the container log driver, NOT a literal syslog(3) socket;
231    // the distroless deployment has no local syslogd, per 06-01 decisions).
232    tracing::error!(
233        target: "axiam.audit.dlq",
234        tenant_id = %entry.tenant_id,
235        actor_id = %entry.actor_id,
236        action = %entry.action,
237        resource_id = ?entry.resource_id,
238        db_error = %db_error,
239        "gdpr_audit_dlq"
240    );
241}

247/// `POST /api/v1/account/export`
248///
249/// Enqueues an async GDPR Art. 15 data-export job.  Returns immediately with
250/// `{"queued": true}`; the cleanup sweep generates the encrypted JSON, stores
251/// it, and emails a single-use 24 h download link (D-12).
252///
253/// Self-service (own account) or admin with `gdpr:export` permission.
254#[utoipa::path(
255    post,
256    path = "/api/v1/account/export",
257    tag = "gdpr",
258    request_body = inline(serde_json::Value),
259    responses(
260        (status = 200, description = "Export job enqueued"),
261        (status = 403, description = "Forbidden"),
262    ),
263    security(("bearer" = []))
264)]
265#[allow(clippy::too_many_arguments)]
266pub async fn request_account_export<C: Connection + Clone>(
267    auth_user: AuthenticatedUser,
268    authz: AuthzData,
269    state: web::Data<AppState<C>>,
270    body: web::Json<ExportRequest>,
271) -> Result<HttpResponse, AxiamApiError> {
272    let req = body.into_inner();
273    let target_id = req.user_id.unwrap_or(auth_user.user_id);
274
275    // Ownership check: self OR gdpr:export permission.
276    if !is_own_resource(&auth_user, target_id) {
277        RequirePermission::new("gdpr:export", Uuid::nil())
278            .check(&auth_user, authz.get_ref().as_ref())
279            .await?;
280    }
281
282    // The subject's data lives in the subject's own tenant. For a self-service
283    // request that is the caller's tenant, which is not the one an
284    // organization-level principal has selected — see [`user_scope_tenant`].
285    // An export job written into the selected tenant would be queued against a
286    // subject that tenant does not have.
287    let scope = user_scope_tenant(&auth_user, target_id);
288
289    // CQ-B39: Deduplicate concurrent export requests — reject if a queued
290    // export already exists for this user to avoid duplicate processing.
291    if state
292        .gdpr
293        .export_job_repo
294        .has_pending_for_user(scope, target_id)
295        .await?
296    {
297        return Err(AxiamError::AlreadyExists {
298            entity: "export_job".into(),
299        }
300        .into());
301    }
302
303    // Create a queued export job.
304    state
305        .gdpr
306        .export_job_repo
307        .create(CreateExportJob {
308            tenant_id: scope,
309            user_id: target_id,
310        })
311        .await?;
312
313    // Audit: gdpr.data_export_requested.
314    append_gdpr_audit(
315        &state.audit_repo,
316        scope,
317        auth_user.user_id,
318        "gdpr.data_export_requested",
319        Some(target_id),
320        Some(serde_json::json!({ "subject_id": target_id.to_string() })),
321    )
322    .await;
323
324    Ok(HttpResponse::Ok().json(QueuedResponse { queued: true }))
325}

327/// `GET /api/v1/account/export/{token}`
328///
329/// Single-use download of the GDPR export blob (D-13).  Looks up the job by
330/// SHA-256(token), verifies it is ready and not expired/used, decrypts the
331/// blob, marks it as downloaded, and deletes the record.
332///
333/// Requires authentication (self or admin with `gdpr:export`).
334#[utoipa::path(
335    get,
336    path = "/api/v1/account/export/{token}",
337    tag = "gdpr",
338    params(("token" = String, Path, description = "Single-use download token")),
339    responses(
340        (status = 200, description = "Export JSON"),
341        (status = 404, description = "Token not found or expired"),
342    ),
343    security(("bearer" = []))
344)]
345#[allow(clippy::too_many_arguments)]
346pub async fn download_account_export<C: Connection + Clone>(
347    auth_user: AuthenticatedUser,
348    authz: AuthzData,
349    state: web::Data<AppState<C>>,
350    path: web::Path<DownloadPath>,
351) -> Result<HttpResponse, AxiamApiError> {
352    let token = path.into_inner().token;
353    let token_hash = sha256_hex(&token);
354
355    // A download link mailed to the caller points at a job in the caller's own
356    // tenant. Looking it up in the tenant an organization-level principal has
357    // selected found nothing and answered 404 for a link that was perfectly
358    // valid — so the search is scoped to the principal's tenant first, and only
359    // falls back to the acting tenant for an administrator downloading somebody
360    // else's export. `find_by_download_token_hash` matches on a SHA-256 of a
361    // 256-bit token, so neither lookup can be steered by the caller.
362    let job = match state
363        .gdpr
364        .export_job_repo
365        .find_by_download_token_hash(auth_user.principal_tenant_id, &token_hash)
366        .await?
367    {
368        Some(job) => Some(job),
369        None if auth_user.tenant_id != auth_user.principal_tenant_id => {
370            state
371                .gdpr
372                .export_job_repo
373                .find_by_download_token_hash(auth_user.tenant_id, &token_hash)
374                .await?
375        }
376        None => None,
377    };
378    let job = job.ok_or_else(|| AxiamError::NotFound {
379        entity: "export_job".into(),
380        id: "by-token".into(),
381    })?;
382
383    // Ownership check.
384    if !is_own_resource(&auth_user, job.user_id) {
385        RequirePermission::new("gdpr:export", Uuid::nil())
386            .check(&auth_user, authz.get_ref().as_ref())
387            .await?;
388    }
389
390    // Validate status and expiry (D-13: single-use, 24h TTL).
391    use axiam_core::models::gdpr::ExportJobStatus;
392    if job.status != ExportJobStatus::Ready {
393        return Err(AxiamError::AuthorizationDenied {
394            reason: "export token already used or not ready".into(),
395            action: None,
396            resource_id: None,
397        }
398        .into());
399    }
400    if let Some(expires_at) = job.expires_at
401        && expires_at < Utc::now()
402    {
403        return Err(AxiamError::AuthorizationDenied {
404            reason: "export token expired".into(),
405            action: None,
406            resource_id: None,
407        }
408        .into());
409    }
410
411    // Decrypt the blob.
412    let (nonce, ciphertext) = match (&job.blob_nonce, &job.encrypted_blob) {
413        (Some(n), Some(ct)) => (n.clone(), ct.clone()),
414        _ => {
415            return Err(AxiamError::Internal("export blob missing on ready job".into()).into());
416        }
417    };
418    let key = state
419        .mail
420        .email_encryption_key
421        .as_ref()
422        .ok_or_else(|| AxiamError::Internal("export encryption key not configured".into()))?;
423
424    let plaintext_bytes = decrypt_separate(key, &nonce, &ciphertext)
425        .map_err(|e| AxiamError::Internal(format!("export decrypt failed: {e}")))?;
426
427    // Atomic single-use consume: UPDATE WHERE status = 'ready' + DELETE.
428    // If 0 rows were updated the token was already consumed (TOCTTOU-safe,
429    // D-13 / CQ-B38 / REQ-14 AC-5).
430    let consumed = state
431        .gdpr
432        .export_job_repo
433        .consume_ready_and_delete(job.id)
434        .await?;
435    if !consumed {
436        return Err(AxiamError::AuthorizationDenied {
437            reason: "export token already consumed".into(),
438            action: None,
439            resource_id: None,
440        }
441        .into());
442    }
443
444    let plaintext = String::from_utf8(plaintext_bytes)
445        .map_err(|e| AxiamError::Internal(format!("export UTF-8 decode: {e}")))?;
446
447    Ok(HttpResponse::Ok()
448        .content_type("application/json")
449        .body(plaintext))
450}

452/// `POST /api/v1/account/delete`
453///
454/// Initiates Art. 17 erasure: immediately disables the account, revokes all
455/// sessions, emails a single-use cancel link, and schedules purge at +30 d
456/// (D-07/D-08/D-09).
457///
458/// Self-service (own account) or admin with `users:erase` permission.
459#[utoipa::path(
460    post,
461    path = "/api/v1/account/delete",
462    tag = "gdpr",
463    request_body = inline(serde_json::Value),
464    responses(
465        (status = 200, description = "Deletion scheduled"),
466        (status = 403, description = "Forbidden"),
467    ),
468    security(("bearer" = []))
469)]
470#[allow(clippy::too_many_arguments)]
471pub async fn request_account_delete<C: Connection + Clone>(
472    auth_user: AuthenticatedUser,
473    authz: AuthzData,
474    state: web::Data<AppState<C>>,
475    body: web::Json<DeleteRequest>,
476) -> Result<HttpResponse, AxiamApiError> {
477    let req = body.into_inner();
478    let target_id = req.user_id.unwrap_or(auth_user.user_id);
479
480    // Ownership check: self OR users:erase permission.
481    if !is_own_resource(&auth_user, target_id) {
482        RequirePermission::new("users:erase", Uuid::nil())
483            .check(&auth_user, authz.get_ref().as_ref())
484            .await?;
485    }
486
487    // The subject's record, the deletion row, the sessions and the audit entry
488    // all belong to the subject's own tenant — for a self-service erasure the
489    // caller's, not the one an organization-level principal has selected. See
490    // [`user_scope_tenant`].
491    let scope = user_scope_tenant(&auth_user, target_id);
492
493    // Load target user to get email for the cancel mail.
494    let target_user = state.user_repo.get_by_id(scope, target_id).await?;
495
496    // Schedule the purge at the end of the tenant's grace window (D-08).
497    //
498    // This was a hard-coded 30 days, which left the "cancel a pending
499    // deletion" control in the admin UI referring to a duration no operator
500    // could see or change. It is now `privacy.deletion_grace_period_days` —
501    // an org baseline a tenant may only lower — and still 30 by default, so
502    // an upgrade changes nothing.
503    //
504    // A read failure falls back to that default rather than refusing the
505    // erasure: an unavailable settings row must not stop a data subject
506    // exercising Art. 17, and the default is the value every deployment had
507    // until now.
508    let grace_days = {
509        use axiam_core::repository::{SettingsRepository as _, TenantRepository as _};
510        let resolved = async {
511            let tenant = state.tenant_repo.get_by_id(scope).await.ok()?;
512            let settings = state
513                .settings_repo
514                .get_effective_settings(tenant.organization_id, scope)
515                .await
516                .ok()?;
517            Some(settings.privacy.deletion_grace_period_days)
518        }
519        .await;
520        resolved.unwrap_or_else(|| {
521            tracing::warn!(
522                tenant_id = %scope,
523                "could not resolve the erasure grace window; using the 30-day default"
524            );
525            30
526        })
527    };
528    let scheduled_purge_at = Utc::now() + chrono::Duration::days(i64::from(grace_days));
529
530    // Generate cancel token: 256-bit random token, hex-encoded (CQ-B39).
531    // Only the SHA-256 hash is stored in the DB; the raw token is emailed.
532    let raw_cancel_token = generate_cancel_token();
533    let cancel_token_hash = sha256_hex(&raw_cancel_token);
534
535    // D-14: mark the user deletion-pending AND create the account_deletion
536    // row (holding the cancel_token_hash) in ONE transaction — a create
537    // failure (e.g. a duplicate pending-deletion conflict) must never
538    // strand the user in deletion_pending=true with no account_deletion row
539    // to hold a cancellable token (CQ-B39 residual / uncancellable purge).
540    state
541        .gdpr
542        .account_deletion_repo
543        .create_with_pending_flag(scope, target_id, scheduled_purge_at, cancel_token_hash)
544        .await?;
545
546    // Revoke all sessions immediately (D-08: account disabled). Kept as a
547    // separate, subsequent call — not part of the D-14 strand-risk
548    // transaction: a failure here leaves an unrevoked session, a lesser and
549    // pre-existing concern, not an uncancellable purge.
550    state
551        .auth_service
552        .revoke_all_sessions(scope, target_id)
553        .await?;
554
555    // Resolve org_id for the mail message.
556    let org_id = match state.tenant_repo.get_by_id(scope).await {
557        Ok(tenant) => tenant.organization_id,
558        Err(e) => {
559            tracing::warn!(error = %e, "failed to resolve org_id for delete-cancel mail");
560            Uuid::nil()
561        }
562    };
563
564    // Enqueue the cancel-link email (D-09).
565    let cancel_url = format!(
566        "/api/v1/auth/account/delete/cancel?token={}",
567        raw_cancel_token
568    );
569    let msg = OutboundMailMessage {
570        mail_type: MailType::DeletionCancel,
571        tenant_id: scope,
572        org_id,
573        user_id: target_id,
574        to_address: target_user.email.clone(),
575        template_context: serde_json::json!({
576            "action_url": cancel_url,
577            "expiry_time": scheduled_purge_at.to_rfc3339(),
578        }),
579        attempt_count: 0,
580        enqueued_at: Utc::now(),
581    };
582    if let Err(e) = state.mail.mail_outbound_publisher.publish(msg).await {
583        tracing::warn!(error = %e, "failed to enqueue delete-cancel email; continuing");
584    }
585
586    // Audit: gdpr.erasure_requested.
587    append_gdpr_audit(
588        &state.audit_repo,
589        scope,
590        auth_user.user_id,
591        "gdpr.erasure_requested",
592        Some(target_id),
593        Some(serde_json::json!({
594            "subject_id": target_id.to_string(),
595            "scheduled_purge_at": scheduled_purge_at.to_rfc3339(),
596        })),
597    )
598    .await;
599
600    Ok(HttpResponse::Ok().json(ScheduledResponse { scheduled: true }))
601}

613    tag = "gdpr",
614    params(("token" = String, Query, description = "Single-use cancel token")),
615    responses(
616        (status = 200, description = "Deletion cancelled"),
617        (status = 400, description = "Token invalid or expired"),
618    )
619)]
620pub async fn cancel_account_delete<C: Connection + Clone>(
621    state: web::Data<AppState<C>>,
622    query: web::Query<CancelQuery>,
623) -> Result<HttpResponse, AxiamApiError> {
624    let token_hash = sha256_hex(&query.token);
625
626    // Global lookup by token hash (no auth context — public endpoint, D-09).
627    let deletion = state
628        .gdpr
629        .account_deletion_repo
630        .find_by_token_hash_global(&token_hash)
631        .await?
632        .ok_or_else(|| AxiamError::NotFound {
633            entity: "account_deletion".into(),
634            id: "by-token".into(),
635        })?;
636
637    // Validate: must be pending and within grace window.
638    if deletion.status != AccountDeletionStatus::Pending {
639        return Err(AxiamError::AuthorizationDenied {
640            reason: "cancel token already used".into(),
641            action: None,
642            resource_id: None,
643        }
644        .into());
645    }
646    if deletion.scheduled_purge_at < Utc::now() {
647        return Err(AxiamError::AuthorizationDenied {
648            reason: "grace window expired — cannot cancel".into(),
649            action: None,
650            resource_id: None,
651        }
652        .into());
653    }
654
655    // Abort deletion: mark cancelled (single-use) + re-enable account.
656    state
657        .gdpr
658        .account_deletion_repo
659        .mark_cancelled(deletion.tenant_id, deletion.id)
660        .await?;
661
662    state
663        .user_repo
664        .clear_deletion_pending(deletion.tenant_id, deletion.user_id)
665        .await?;
666
667    Ok(HttpResponse::Ok().json(CancelledResponse { cancelled: true }))
668}

670// ---------------------------------------------------------------------------
671// GDPR Art. 7 — OIDC scope-release consent (X7 G8 / W7)
672// ---------------------------------------------------------------------------
673//
674// The plan's §4.8 says withdrawal is "an entry on the user's GDPR self-service
675// page (the existing consent list)". There was no such list, and no consent
676// endpoint of any kind: `consent` rows were written by registration and read
677// only by the Art. 15 export. So the three endpoints below are new, and the
678// first of them is the list the plan assumed.
679//
680// All three are strictly self-service. An administrator cannot consent on a
681// subject's behalf — that is what Art. 4(11)'s "freely given, specific,
682// informed and unambiguous indication of the data subject's wishes" rules
683// out — and there is deliberately no `user_id` parameter to try it with,
684// unlike the export and erasure endpoints above, where acting for a subject on
685// their request is exactly the point.
686
687/// One consent record, as the subject sees it.
688#[derive(Debug, Serialize, utoipa::ToSchema)]
689pub struct ConsentView {
690    /// What was consented to, e.g. `terms_of_service` or
691    /// `oidc_scope_release:<client_id>`.
692    pub consent_type: String,
693    /// The document version or, for a scope release, the consented scopes.
694    pub version: String,
695    pub accepted_at: chrono::DateTime<Utc>,
696    /// Whether this record can be withdrawn here.
697    ///
698    /// `false` for `terms_of_service`: withdrawing it is not a consent
699    /// operation but an erasure, and it has its own endpoint with its own
700    /// grace period. Reported rather than silently absent so the self-service
701    /// page can show the record and explain it.
702    pub withdrawable: bool,
703}
704
705/// Body for recording an OIDC scope-release consent.
706#[derive(Debug, Deserialize, utoipa::ToSchema)]
707pub struct GrantScopeConsent {
708    /// The relying party the claims would be released to.
709    pub client_id: String,
710    /// The sensitive scopes being consented to. Order does not matter; the
711    /// record is written in the canonical order so that the same consent has
712    /// one name.
713    pub scopes: Vec<String>,
714}
715
716/// `GET /api/v1/account/consents` — the caller's own consent records.
717///
718/// Art. 7(1) says a controller must be able to demonstrate that consent was
719/// given; Art. 15(1)(a) says the subject may see what is held about them. This
720/// endpoint is the second, and it is also the page the withdrawal control
721/// lives on.
722#[utoipa::path(
723    get,
724    path = "/api/v1/account/consents",
725    tag = "gdpr",
726    responses(
727        (status = 200, description = "The caller's consent records",
728         body = Vec<ConsentView>),
729        (status = 401, description = "Not authenticated"),
730    ),
731    security(("bearer" = []))
732)]
733pub async fn list_own_consents<C: Connection + Clone>(
734    user: AuthenticatedUser,
735    state: web::Data<AppState<C>>,
736) -> Result<HttpResponse, AxiamApiError> {
737    let records = axiam_core::repository::ConsentRepository::list_by_user(
738        &state.gdpr.consent_repo,
739        user.principal_tenant_id,
740        user.user_id,
741    )
742    .await?;
743
744    let view: Vec<ConsentView> = records
745        .into_iter()
746        .map(|c| ConsentView {
747            withdrawable: c
748                .consent_type
749                .starts_with(axiam_core::repository::OIDC_SCOPE_RELEASE_CONSENT_PREFIX),
750            consent_type: c.consent_type,
751            version: c.version,
752            accepted_at: c.accepted_at,
753        })
754        .collect();
755    Ok(HttpResponse::Ok().json(view))
756}

758/// `POST /api/v1/account/consents/oidc-scopes` — record a scope-release consent.
759///
760/// What the SPA's consent screen calls when the end user says yes. Everything
761/// it accepts is checked against the registration rather than taken on trust,
762/// because a consent record is the thing UserInfo releases personal data on
763/// the strength of:
764///
765/// * the scopes must be sensitive ones — nothing else belongs in this
766///   namespace, and a record naming `openid` would be a record that never
767///   matches and never expires;
768/// * the client must exist in the caller's tenant;
769/// * the client must have every named scope **registered**, so a consent
770///   cannot be recorded for a release the client could never have been
771///   authorised for;
772/// * the tenant switch must be on, so consent collected while the capability
773///   is off cannot sit waiting for somebody to turn it on.
774///
775/// Idempotent: the `(tenant, user, type, version)` index makes a repeated
776/// grant the same grant, and a second call is answered `200` rather than a
777/// conflict. A consent screen the user double-submits has consented once.
778#[utoipa::path(
779    post,
780    path = "/api/v1/account/consents/oidc-scopes",
781    tag = "gdpr",
782    request_body = GrantScopeConsent,
783    responses(
784        (status = 200, description = "Consent recorded"),
785        (status = 400, description = "Unknown client, unregistered scope, \
786                                      non-sensitive scope, or the capability is off"),
787        (status = 401, description = "Not authenticated"),
788    ),
789    security(("bearer" = []))
790)]
791pub async fn grant_oidc_scope_consent<C: Connection + Clone>(
792    http_req: actix_web::HttpRequest,
793    user: AuthenticatedUser,
794    state: web::Data<AppState<C>>,
795    body: web::Json<GrantScopeConsent>,
796) -> Result<HttpResponse, AxiamApiError> {
797    let body = body.into_inner();
798
799    // The client must exist in the caller's tenant whichever gate this
800    // consent is for. Read first, because it is what decides which gate it is.
801    let client = axiam_core::repository::OAuth2ClientRepository::get_by_client_id(
802        &state.oauth2_client_repo,
803        user.principal_tenant_id,
804        &body.client_id,
805    )
806    .await?;
807
808    // T21.4 / D4 — the external-client lane.
809    //
810    // For a client an administrator did not create, the consent being recorded
811    // is not "you may release my address", it is "this application may act as
812    // me". It therefore covers the **whole** requested scope set rather than
813    // the sensitive subset, and its version carries a prefix so the two kinds
814    // of record can share one `consent_type` — which is what makes
815    // `DELETE /account/consents/oidc-scopes/{client_id}` withdraw both with no
816    // new control. See `axiam_oauth2::external_consent`.
817    //
818    // The two lanes never overlap: the settings layer refuses `address` and
819    // `phone` in `dcr_allowed_scopes`, so a self-registered client cannot hold
820    // a scope the W7 lane governs.
821    let (consent_type, version) = if axiam_oauth2::external_consent::applies_to(client.managed_by) {
822        // Every scope must be registered on the client, for the reason the
823        // W7 lane checks the same thing: a consent recorded for a scope
824        // the client could never be authorised for authorises nothing and
825        // will never match at the authorization endpoint.
826        if let Some(missing) = body
827            .scopes
828            .iter()
829            .find(|s| !client.scopes.iter().any(|r| r == *s))
830        {
831            return Err(AxiamApiError::from(AxiamError::Validation {
832                message: format!(
833                    "this client has no {missing} scope registered, so consenting to it \
834                         would authorise nothing"
835                ),
836            }));
837        }
838        (
839            axiam_oauth2::external_consent::consent_type(&body.client_id),
840            axiam_oauth2::external_consent::version(&body.scopes),
841        )
842    } else {
843        // W7's lane, unchanged. Only the sensitive scopes, and in
844        // canonical order: `requested` filters and orders in one step, so
845        // a body naming `openid` yields an empty set and is refused here
846        // rather than silently recorded.
847        let wanted = axiam_oauth2::sensitive::requested(&body.scopes);
848        if wanted.is_empty() || wanted.len() != body.scopes.len() {
849            return Err(AxiamApiError::from(AxiamError::Validation {
850                message: format!(
851                    "consent may be recorded only for the scopes {}, and for nothing else",
852                    axiam_oauth2::sensitive::SENSITIVE_SCOPES.join(", ")
853                ),
854            }));
855        }
856        if let Some(missing) = wanted
857            .iter()
858            .find(|s| !client.scopes.iter().any(|r| r == *s))
859        {
860            return Err(AxiamApiError::from(AxiamError::Validation {
861                message: format!(
862                    "this client has no {missing} scope registered, so consenting to \
863                         release it would authorise nothing"
864                ),
865            }));
866        }
867
868        let tenant = state
869            .tenant_repo
870            .get_by_id(user.principal_tenant_id)
871            .await?;
872        let settings = axiam_core::repository::SettingsRepository::get_effective_settings(
873            &state.settings_repo,
874            tenant.organization_id,
875            user.principal_tenant_id,
876        )
877        .await?;
878        if !settings.oidc.sensitive_scopes_enabled {
879            return Err(AxiamApiError::from(AxiamError::Validation {
880                message: "the address and phone scopes are not enabled for this tenant".into(),
881            }));
882        }
883
884        (
885            axiam_oauth2::sensitive::consent_type(&body.client_id),
886            axiam_oauth2::sensitive::consent_version(&wanted),
887        )
888    };
889
890    // Idempotence, checked rather than relying on the unique index's error:
891    // a second grant is the same grant, and answering it with a conflict
892    // would make a double-submitted consent screen look broken.
893    let already = axiam_core::repository::ConsentRepository::list_by_user(
894        &state.gdpr.consent_repo,
895        user.principal_tenant_id,
896        user.user_id,
897    )
898    .await?
899    .into_iter()
900    .any(|c| c.consent_type == consent_type && c.version == version);
901
902    if !already {
903        axiam_core::repository::ConsentRepository::create(
904            &state.gdpr.consent_repo,
905            axiam_core::models::gdpr::CreateConsent {
906                tenant_id: user.principal_tenant_id,
907                user_id: user.user_id,
908                consent_type: consent_type.clone(),
909                version: version.clone(),
910                ip_address: crate::extractors::client_info::client_ip(&http_req),
911                user_agent: crate::extractors::client_info::user_agent(&http_req),
912            },
913        )
914        .await?;
915
916        // Art. 7(1) proof lives in two places on purpose: the record, which is
917        // live state and can be withdrawn, and the audit log, which is
918        // append-only and cannot. Scope *names*, never the values they would
919        // release.
920        audit_consent(
921            &state,
922            &http_req,
923            &user,
924            "gdpr.oidc_scope_consent_granted",
925            &body.client_id,
926            &version,
927        )
928        .await;
929    }
930
931    Ok(HttpResponse::Ok().json(serde_json::json!({
932        "consent_type": consent_type,
933        "version": version,
934    })))
935}

937/// `DELETE /api/v1/account/consents/oidc-scopes/{client_id}` — withdraw.
938///
939/// Art. 7(3): as easy to withdraw as to give. One call, no grace period, no
940/// confirmation step, and it takes effect on the **next UserInfo call with the
941/// token the relying party already holds** — not on the next token. That is
942/// the property T8.4 asserts, and it is why the release gate re-reads the
943/// record on every call rather than trusting the one taken at authorization.
944///
945/// Withdraws every scope set consented to for this relying party, not one of
946/// them: a subject saying "stop giving my address to this app" does not mean
947/// "stop giving it under the two-scope record but carry on under the
948/// one-scope one".
949///
950/// Answers `200` whether or not anything was there, and says how many records
951/// went. A subject who withdraws twice is not told off, and an attacker who
952/// guesses `client_id`s learns nothing from the status code — though they
953/// would have to be the subject to ask at all.
954#[utoipa::path(
955    delete,
956    path = "/api/v1/account/consents/oidc-scopes/{client_id}",
957    tag = "gdpr",
958    params(("client_id" = String, Path, description = "The relying party to stop releasing to")),
959    responses(
960        (status = 200, description = "Consent withdrawn (or there was none)"),
961        (status = 401, description = "Not authenticated"),
962    ),
963    security(("bearer" = []))
964)]
965pub async fn withdraw_oidc_scope_consent<C: Connection + Clone>(
966    http_req: actix_web::HttpRequest,
967    user: AuthenticatedUser,
968    state: web::Data<AppState<C>>,
969    path: web::Path<String>,
970) -> Result<HttpResponse, AxiamApiError> {
971    let client_id = path.into_inner();
972    let consent_type = axiam_oauth2::sensitive::consent_type(&client_id);
973
974    let removed = axiam_core::repository::ConsentRepository::withdraw(
975        &state.gdpr.consent_repo,
976        user.principal_tenant_id,
977        user.user_id,
978        &consent_type,
979    )
980    .await?;
981
982    if removed > 0 {
983        audit_consent(
984            &state,
985            &http_req,
986            &user,
987            "gdpr.oidc_scope_consent_withdrawn",
988            &client_id,
989            "",
990        )
991        .await;
992    }
993
994    Ok(HttpResponse::Ok().json(serde_json::json!({ "withdrawn": removed })))
995}

997/// Record a consent grant or withdrawal in the append-only log.
998///
999/// Carries the relying party and the scope *names*, and no claim values —
1000/// there are none to carry at this point, and there will not be any at this
1001/// point in a future version either. Failure is logged and swallowed: refusing
1002/// a withdrawal because the audit store is unavailable would make Art. 7(3)
1003/// conditional on infrastructure.
1004async fn audit_consent<C: Connection + Clone>(
1005    state: &AppState<C>,
1006    http_req: &actix_web::HttpRequest,
1007    user: &AuthenticatedUser,
1008    action: &str,
1009    client_id: &str,
1010    scopes: &str,
1011) {
1012    if let Err(e) = state
1013        .audit_repo
1014        .append(CreateAuditLogEntry {
1015            tenant_id: user.principal_tenant_id,
1016            actor_id: user.user_id,
1017            actor_type: ActorType::User,
1018            action: action.to_string(),
1019            resource_id: None,
1020            outcome: AuditOutcome::Success,
1021            ip_address: crate::extractors::client_info::client_ip(http_req),
1022            metadata: Some(serde_json::json!({
1023                "client_id": client_id,
1024                "scopes": scopes,
1025            })),
1026        })
1027        .await
1028    {
1029        tracing::error!(
1030            error = %e,
1031            user_id = %user.user_id,
1032            %action,
1033            "could not record a consent change; the change itself stands"
1034        );
1035    }
