84/// SHA-256 hex of `raw_token`.
85pub fn sha256_hex(raw: &str) -> String {
86    let mut h = Sha256::new();
87    h.update(raw.as_bytes());
88    hex::encode(h.finalize())
89}

91/// Generate a cryptographically-random 256-bit (32-byte) cancel token,
92/// hex-encoded as a 64-character string (CQ-B39).
93///
94/// Replaces the previous `Uuid::new_v4().to_string()` (128-bit) token.
95fn generate_cancel_token() -> String {
96    use rand::RngExt;
97    let bytes: [u8; 32] = rand::rng().random();
98    hex::encode(bytes)
99}

101/// Record a GDPR request in the audit log. A refused append is dead-lettered,
102/// never propagated.
103///
104/// Factored out of the individual GDPR handlers to eliminate the repeated
105/// audit-append block pattern (CQ-B39).  An audit failure must not block the
106/// user response, but a GDPR audit trail is legally significant (CQ-B31 /
107/// T-12-01), so a record the datastore refuses goes to the dead-letter file and
108/// the `axiam.audit.dlq` event through [`write_audit_with_dead_letter`], the
109/// route the erasure records already take (P23W5-A8, T19.27).
110async fn append_gdpr_audit<S: AuditWriteSink>(
111    audit_sink: &S,
112    tenant_id: Uuid,
113    actor_id: Uuid,
114    action: &str,
115    resource_id: Option<Uuid>,
116    metadata: Option<serde_json::Value>,
117) {
118    write_audit_with_dead_letter(
119        audit_sink,
120        CreateAuditLogEntry {
121            tenant_id,
122            actor_id,
123            actor_type: ActorType::User,
124            action: action.into(),
125            resource_id,
126            outcome: AuditOutcome::Success,
127            ip_address: None,
128            metadata,
129        },
130    )
131    .await;
132}

146/// Injectable seam for the GDPR audit DB-write (SECHRD-12).
147///
148/// `SurrealAuditLogRepository<C>` is the production implementation (forwards
149/// to [`AuditLogRepository::append`]); tests inject a failing double to drive
150/// the dead-letter path in [`write_audit_with_dead_letter`] without a live
151/// broken database. Defined here (not in `axiam-core::repository`) so it
152/// stays out of the generic repository trait surface owned by another plan.
153pub trait AuditWriteSink: Send + Sync {
154    fn write(
155        &self,
156        entry: CreateAuditLogEntry,
157    ) -> impl Future<Output = AxiamResult<AuditLogEntry>> + Send;
158}
159
160impl<C: Connection> AuditWriteSink for SurrealAuditLogRepository<C> {
161    fn write(
162        &self,
163        entry: CreateAuditLogEntry,
164    ) -> impl Future<Output = AxiamResult<AuditLogEntry>> + Send {
165        self.append(entry)
166    }
167}

169/// Write a GDPR audit record via `sink`. If the DB write fails, the record is
170/// dead-lettered to BOTH an append-only local file AND a structured `tracing`
171/// audit event (D-02), so a legally-significant event is never silently lost
172/// to a transient DB failure (T-24-61).
173///
174/// The records that take this route: `gdpr.user_pseudonymized` (the cleanup
175/// sweep), `tenants.deleted`, and the two request records
176/// `gdpr.data_export_requested` and `gdpr.erasure_requested` (P23W5-A8).
177///
178/// Never propagates the DB error — the callers (the cleanup ticker, T-04-36,
179/// and the request handlers, whose user-visible work is already done) must not
180/// panic or abort on this failure path; the two dead-letter sinks ARE the
181/// durability guarantee for this branch.
182pub async fn write_audit_with_dead_letter<S: AuditWriteSink>(sink: &S, entry: CreateAuditLogEntry) {
183    // When the event happened, for the dead-letter line (#595): the moment the
184    // record was handed here, right after the action it records and before the
185    // datastore write that may be refused — not when the file is appended.
186    let occurred_at = Utc::now();
187    let dlq_entry = entry.clone();
188    if let Err(e) = sink.write(entry).await {
189        tracing::error!(
190            error = %e,
191            tenant_id = %dlq_entry.tenant_id,
192            action = %dlq_entry.action,
193            "gdpr: audit DB-write failed — dead-lettering to append-only file + \
194             structured event (legally significant, SECHRD-12)"
195        );
196        dead_letter_audit(dlq_entry, occurred_at, &e).await;
197    }
198}

200/// Dead-letter a failed GDPR audit record to the two durable sinks (D-02).
201///
202/// The structured event is emitted FIRST, before the first `await` (R101W4-04,
203/// #625). The file append is awaited, and a future dropped there — the cleanup
204/// task's bounded join aborting the erasure's `gdpr.user_pseudonymized` write
205/// (#598), the lease-loss cut, a handler dropped when its client goes — would
206/// otherwise skip an event that does not depend on the file's result. Emitted
207/// ahead of any suspension point, it cannot be skipped by a cancellation.
208///
209/// The file append runs on the runtime's blocking pool and is synced to the
210/// device before it returns (#594): the handler awaits it, so a record that was
211/// reported as dead-lettered is on disk, and no actix worker blocks on the file.
212/// The blocking task runs to completion if its awaiter is dropped, so the file
213/// sink needs no protection of its own from a cancellation.
214///
215/// Why the awaited write is not also spawned (#625): the callers are the cleanup
216/// ticker and two request handlers, and the first is *meant* to be cut — its
217/// bounded stop-join (#598) is what keeps a stop from waiting on a tenant purge
218/// or an erasure. A detached `dead_letter_audit` would escape that join: no one
219/// awaits it at shutdown, and a task still suspended on the blocking write when
220/// the runtime is dropped is dropped with it, losing the one thing spawning would
221/// add — the file-failure log. The event is what operators alert on and it needs
222/// no task to survive; a write that fails after its awaiter was cut is the
223/// residual, recorded in the threat model beside the rest of T-108.
224async fn dead_letter_audit(
225    entry: CreateAuditLogEntry,
226    occurred_at: DateTime<Utc>,
227    db_error: &AxiamError,
228) {
229    // Sink 2 of D-02, emitted before sink 1: the structured `tracing` audit
230    // event — the "audit syslog" sink (RESEARCH Assumption A4: satisfied by a
231    // structured tracing JSON event captured by the container log driver, NOT a
232    // literal syslog(3) socket; the distroless deployment has no local syslogd,
233    // per 06-01 decisions). The fields are the ones operators' queries match.
234    tracing::error!(
235        target: "axiam.audit.dlq",
236        tenant_id = %entry.tenant_id,
237        actor_id = %entry.actor_id,
238        action = %entry.action,
239        resource_id = ?entry.resource_id,
240        db_error = %db_error,
241        "gdpr_audit_dlq"
242    );
243
244    // Sink 1: append-only local file on a mounted volume (T-24-62). Opened
245    // with `.append(true)` — an existing file is never truncated/rewritten,
246    // matching AXIAM's append-only audit posture. The line is the one the
247    // request-audit worker's dead letters use (`axiam_audit::dead_letter`).
248    // The path and the budget (the whole of it, the reserve above the request
249    // rows' share included, R1W2-02) are read once, here; an invalid budget
250    // failed the boot already.
251    match axiam_audit::dead_letter::DeadLetterTarget::from_env() {
252        Some(target) => {
253            let path = target.path.display().to_string();
254            // A join error (the write task panicked) is an I/O error like any
255            // other: logged on its own event, the structured event above having
256            // fired already.
257            // The reason is the datastore refusing the write; its error text is
258            // logged above and on the event, never written to the file.
259            let record = axiam_audit::dead_letter::DeadLetterRecord::new(
260                entry,
261                occurred_at,
262                axiam_audit::dead_letter::DeadLetterReason::DatastoreError,
263            );
264            if let Err(e) = axiam_audit::dead_letter::append_off_thread(target, record).await {
265                tracing::error!(
266                    error = %e,
267                    path = %path,
268                    "gdpr: failed to append audit dead-letter file"
269                );
270            }
271        }
272        None => {
273            tracing::warn!(
274                env_var = GDPR_AUDIT_DLQ_FILE_ENV,
275                "gdpr: audit dead-letter FILE sink skipped (env var not set); \
276                 structured event above already fired"
277            );
278        }
279    }
280}

286/// `POST /api/v1/account/export`
287///
288/// Enqueues an async GDPR Art. 15 data-export job.  Returns immediately with
289/// `{"queued": true}`; the cleanup sweep generates the encrypted JSON, stores
290/// it, and emails a single-use 24 h download link (D-12).
291///
292/// Self-service (own account) or admin with `gdpr:export` permission.
293#[utoipa::path(
294    post,
295    path = "/api/v1/account/export",
296    tag = "gdpr",
297    request_body = inline(serde_json::Value),
298    responses(
299        (status = 200, description = "Export job enqueued"),
300        (status = 403, description = "Forbidden"),
301    ),
302    security(("bearer" = []))
303)]
304#[allow(clippy::too_many_arguments)]
305pub async fn request_account_export<C: Connection + Clone>(
306    auth_user: AuthenticatedUser,
307    authz: AuthzData,
308    state: web::Data<AppState<C>>,
309    body: web::Json<ExportRequest>,
310) -> Result<HttpResponse, AxiamApiError> {
311    let req = body.into_inner();
312    let target_id = req.user_id.unwrap_or(auth_user.user_id);
313
314    // Ownership check: self OR gdpr:export permission.
315    if !is_own_resource(&auth_user, target_id) {
316        RequirePermission::new("gdpr:export", Uuid::nil())
317            .check(&auth_user, authz.get_ref().as_ref())
318            .await?;
319    }
320
321    // The subject's data lives in the subject's own tenant. For a self-service
322    // request that is the caller's tenant, which is not the one an
323    // organization-level principal has selected — see [`user_scope_tenant`].
324    // An export job written into the selected tenant would be queued against a
325    // subject that tenant does not have.
326    let scope = user_scope_tenant(&auth_user, target_id);
327
328    // CQ-B39: Deduplicate concurrent export requests — reject if a queued
329    // export already exists for this user to avoid duplicate processing.
330    if state
331        .gdpr
332        .export_job_repo
333        .has_pending_for_user(scope, target_id)
334        .await?
335    {
336        return Err(AxiamError::AlreadyExists {
337            entity: "export_job".into(),
338        }
339        .into());
340    }
341
342    // Create a queued export job.
343    state
344        .gdpr
345        .export_job_repo
346        .create(CreateExportJob {
347            tenant_id: scope,
348            user_id: target_id,
349        })
350        .await?;
351
352    // Audit: gdpr.data_export_requested.
353    append_gdpr_audit(
354        &state.audit_repo,
355        scope,
356        auth_user.user_id,
357        "gdpr.data_export_requested",
358        Some(target_id),
359        Some(serde_json::json!({ "subject_id": target_id.to_string() })),
360    )
361    .await;
362
363    Ok(HttpResponse::Ok().json(QueuedResponse { queued: true }))
364}

366/// `GET /api/v1/account/export/{token}`
367///
368/// Single-use download of the GDPR export blob (D-13).  Looks up the job by
369/// SHA-256(token), verifies it is ready and not expired/used, decrypts the
370/// blob, marks it as downloaded, and deletes the record.
371///
372/// Requires authentication (self or admin with `gdpr:export`).
373#[utoipa::path(
374    get,
375    path = "/api/v1/account/export/{token}",
376    tag = "gdpr",
377    params(("token" = String, Path, description = "Single-use download token")),
378    responses(
379        (status = 200, description = "Export JSON"),
380        (status = 404, description = "Token not found or expired"),
381    ),
382    security(("bearer" = []))
383)]
384#[allow(clippy::too_many_arguments)]
385pub async fn download_account_export<C: Connection + Clone>(
386    auth_user: AuthenticatedUser,
387    authz: AuthzData,
388    state: web::Data<AppState<C>>,
389    path: web::Path<DownloadPath>,
390) -> Result<HttpResponse, AxiamApiError> {
391    let token = path.into_inner().token;
392    let token_hash = sha256_hex(&token);
393
394    // A download link mailed to the caller points at a job in the caller's own
395    // tenant. Looking it up in the tenant an organization-level principal has
396    // selected found nothing and answered 404 for a link that was perfectly
397    // valid — so the search is scoped to the principal's tenant first, and only
398    // falls back to the acting tenant for an administrator downloading somebody
399    // else's export. `find_by_download_token_hash` matches on a SHA-256 of a
400    // 256-bit token, so neither lookup can be steered by the caller.
401    let job = match state
402        .gdpr
403        .export_job_repo
404        .find_by_download_token_hash(auth_user.principal_tenant_id, &token_hash)
405        .await?
406    {
407        Some(job) => Some(job),
408        None if auth_user.tenant_id != auth_user.principal_tenant_id => {
409            state
410                .gdpr
411                .export_job_repo
412                .find_by_download_token_hash(auth_user.tenant_id, &token_hash)
413                .await?
414        }
415        None => None,
416    };
417    let job = job.ok_or_else(|| AxiamError::NotFound {
418        entity: "export_job".into(),
419        id: "by-token".into(),
420    })?;
421
422    // Ownership check.
423    if !is_own_resource(&auth_user, job.user_id) {
424        RequirePermission::new("gdpr:export", Uuid::nil())
425            .check(&auth_user, authz.get_ref().as_ref())
426            .await?;
427    }
428
429    // Validate status and expiry (D-13: single-use, 24h TTL).
430    use axiam_core::models::gdpr::ExportJobStatus;
431    if job.status != ExportJobStatus::Ready {
432        return Err(AxiamError::AuthorizationDenied {
433            reason: "export token already used or not ready".into(),
434            action: None,
435            resource_id: None,
436        }
437        .into());
438    }
439    if let Some(expires_at) = job.expires_at
440        && expires_at < Utc::now()
441    {
442        return Err(AxiamError::AuthorizationDenied {
443            reason: "export token expired".into(),
444            action: None,
445            resource_id: None,
446        }
447        .into());
448    }
449
450    // Decrypt the blob.
451    let (nonce, ciphertext) = match (&job.blob_nonce, &job.encrypted_blob) {
452        (Some(n), Some(ct)) => (n.clone(), ct.clone()),
453        _ => {
454            return Err(AxiamError::Internal("export blob missing on ready job".into()).into());
455        }
456    };
457    let key = state
458        .mail
459        .email_encryption_key
460        .as_ref()
461        .ok_or_else(|| AxiamError::Internal("export encryption key not configured".into()))?;
462
463    let plaintext_bytes = decrypt_separate(key, &nonce, &ciphertext)
464        .map_err(|e| AxiamError::Internal(format!("export decrypt failed: {e}")))?;
465
466    // Atomic single-use consume: UPDATE WHERE status = 'ready' + DELETE.
467    // If 0 rows were updated the token was already consumed (TOCTTOU-safe,
468    // D-13 / CQ-B38 / REQ-14 AC-5).
469    let consumed = state
470        .gdpr
471        .export_job_repo
472        .consume_ready_and_delete(job.id)
473        .await?;
474    if !consumed {
475        return Err(AxiamError::AuthorizationDenied {
476            reason: "export token already consumed".into(),
477            action: None,
478            resource_id: None,
479        }
480        .into());
481    }
482
483    let plaintext = String::from_utf8(plaintext_bytes)
484        .map_err(|e| AxiamError::Internal(format!("export UTF-8 decode: {e}")))?;
485
486    Ok(HttpResponse::Ok()
487        .content_type("application/json")
488        .body(plaintext))
489}

491/// `POST /api/v1/account/delete`
492///
493/// Initiates Art. 17 erasure: immediately disables the account, revokes all
494/// sessions, emails a single-use cancel link, and schedules purge at +30 d
495/// (D-07/D-08/D-09).
496///
497/// Self-service (own account) or admin with `users:erase` permission.
498#[utoipa::path(
499    post,
500    path = "/api/v1/account/delete",
501    tag = "gdpr",
502    request_body = inline(serde_json::Value),
503    responses(
504        (status = 200, description = "Deletion scheduled"),
505        (status = 403, description = "Forbidden"),
506    ),
507    security(("bearer" = []))
508)]
509#[allow(clippy::too_many_arguments)]
510pub async fn request_account_delete<C: Connection + Clone>(
511    auth_user: AuthenticatedUser,
512    authz: AuthzData,
513    state: web::Data<AppState<C>>,
514    body: web::Json<DeleteRequest>,
515) -> Result<HttpResponse, AxiamApiError> {
516    let req = body.into_inner();
517    let target_id = req.user_id.unwrap_or(auth_user.user_id);
518
519    // Ownership check: self OR users:erase permission.
520    if !is_own_resource(&auth_user, target_id) {
521        RequirePermission::new("users:erase", Uuid::nil())
522            .check(&auth_user, authz.get_ref().as_ref())
523            .await?;
524    }
525
526    // The subject's record, the deletion row, the sessions and the audit entry
527    // all belong to the subject's own tenant — for a self-service erasure the
528    // caller's, not the one an organization-level principal has selected. See
529    // [`user_scope_tenant`].
530    let scope = user_scope_tenant(&auth_user, target_id);
531
532    // Load target user to get email for the cancel mail.
533    let target_user = state.user_repo.get_by_id(scope, target_id).await?;
534
535    // Schedule the purge at the end of the tenant's grace window (D-08).
536    //
537    // This was a hard-coded 30 days, which left the "cancel a pending
538    // deletion" control in the admin UI referring to a duration no operator
539    // could see or change. It is now `privacy.deletion_grace_period_days` —
540    // an org baseline a tenant may only lower — and still 30 by default, so
541    // an upgrade changes nothing.
542    //
543    // A read failure falls back to that default rather than refusing the
544    // erasure: an unavailable settings row must not stop a data subject
545    // exercising Art. 17, and the default is the value every deployment had
546    // until now.
547    let grace_days = {
548        use axiam_core::repository::{SettingsRepository as _, TenantRepository as _};
549        let resolved = async {
550            let tenant = state.tenant_repo.get_by_id(scope).await.ok()?;
551            let settings = state
552                .settings_repo
553                .get_effective_settings(tenant.organization_id, scope)
554                .await
555                .ok()?;
556            Some(settings.privacy.deletion_grace_period_days)
557        }
558        .await;
559        resolved.unwrap_or_else(|| {
560            tracing::warn!(
561                tenant_id = %scope,
562                "could not resolve the erasure grace window; using the 30-day default"
563            );
564            30
565        })
566    };
567    let scheduled_purge_at = Utc::now() + chrono::Duration::days(i64::from(grace_days));
568
569    // Generate cancel token: 256-bit random token, hex-encoded (CQ-B39).
570    // Only the SHA-256 hash is stored in the DB; the raw token is emailed.
571    let raw_cancel_token = generate_cancel_token();
572    let cancel_token_hash = sha256_hex(&raw_cancel_token);
573
574    // D-14: mark the user deletion-pending AND create the account_deletion
575    // row (holding the cancel_token_hash) in ONE transaction — a create
576    // failure (e.g. a duplicate pending-deletion conflict) must never
577    // strand the user in deletion_pending=true with no account_deletion row
578    // to hold a cancellable token (CQ-B39 residual / uncancellable purge).
579    state
580        .gdpr
581        .account_deletion_repo
582        .create_with_pending_flag(scope, target_id, scheduled_purge_at, cancel_token_hash)
583        .await?;
584
585    // Revoke all sessions immediately (D-08: account disabled). Kept as a
586    // separate, subsequent call — not part of the D-14 strand-risk
587    // transaction: a failure here leaves an unrevoked session, a lesser and
588    // pre-existing concern, not an uncancellable purge.
589    state
590        .auth_service
591        .revoke_all_sessions(scope, target_id)
592        .await?;
593
594    // Resolve org_id for the mail message.
595    let org_id = match state.tenant_repo.get_by_id(scope).await {
596        Ok(tenant) => tenant.organization_id,
597        Err(e) => {
598            tracing::warn!(error = %e, "failed to resolve org_id for delete-cancel mail");
599            Uuid::nil()
600        }
601    };
602
603    // Enqueue the cancel-link email (D-09).
604    let cancel_url = format!(
605        "/api/v1/auth/account/delete/cancel?token={}",
606        raw_cancel_token
607    );
608    let msg = OutboundMailMessage {
609        mail_type: MailType::DeletionCancel,
610        tenant_id: scope,
611        org_id,
612        user_id: target_id,
613        to_address: target_user.email.clone(),
614        template_context: serde_json::json!({
615            "action_url": cancel_url,
616            "expiry_time": scheduled_purge_at.to_rfc3339(),
617        }),
618        attempt_count: 0,
619        enqueued_at: Utc::now(),
620    };
621    if let Err(e) = state.mail.mail_outbound_publisher.publish(msg).await {
622        tracing::warn!(error = %e, "failed to enqueue delete-cancel email; continuing");
623    }
624
625    // Audit: gdpr.erasure_requested.
626    append_gdpr_audit(
627        &state.audit_repo,
628        scope,
629        auth_user.user_id,
630        "gdpr.erasure_requested",
631        Some(target_id),
632        Some(serde_json::json!({
633            "subject_id": target_id.to_string(),
634            "scheduled_purge_at": scheduled_purge_at.to_rfc3339(),
635        })),
636    )
637    .await;
638
639    Ok(HttpResponse::Ok().json(ScheduledResponse { scheduled: true }))
640}

642/// `GET /api/v1/auth/account/delete/cancel?token=<opaque>`
643///
644/// **Public endpoint** (token-authenticated; listed in `PUBLIC_PATHS`).
645///
646/// Validates the single-use cancel token; if the deletion is still pending
647/// and within the grace window, aborts it and re-enables the account (D-09).
648/// A second call with the same token is rejected (single-use).
649#[utoipa::path(
650    get,
651    path = "/api/v1/auth/account/delete/cancel",
652    tag = "gdpr",
653    params(("token" = String, Query, description = "Single-use cancel token")),
654    responses(
655        (status = 200, description = "Deletion cancelled; the account is active again"),
656        (status = 400, description = "Token invalid or expired"),
657        (status = 403, description = "The token was already used, or the grace window has ended"),
658        (status = 404, description = "No deletion request carries this token"),
659        (status = 409, description = "The erasure has already started (the purge sweep claimed the request, or completed it): the deletion can no longer be cancelled, and nothing was changed"),
660    )
661)]
662pub async fn cancel_account_delete<C: Connection + Clone>(
663    state: web::Data<AppState<C>>,
664    query: web::Query<CancelQuery>,
665) -> Result<HttpResponse, AxiamApiError> {
666    let token_hash = sha256_hex(&query.token);
667
668    // Global lookup by token hash (no auth context — public endpoint, D-09).
669    let deletion = state
670        .gdpr
671        .account_deletion_repo
672        .find_by_token_hash_global(&token_hash)
673        .await?
674        .ok_or_else(|| AxiamError::NotFound {
675            entity: "account_deletion".into(),
676            id: "by-token".into(),
677        })?;
678
679    // Item 11 of the 1.1.0 atomicity audit: the checks and the two writes —
680    // the request cancelled, the account re-enabled — are one transaction, and
681    // the purge sweep claims a request before its first destructive step. A
682    // cancel that loses to the sweep is told so; it is never answered
683    // "cancelled" for an account that is being erased.
684    match state
685        .gdpr
686        .account_deletion_repo
687        .cancel_pending(deletion.tenant_id, deletion.id)
688        .await?
689    {
690        CancelOutcome::Cancelled => {}
691        CancelOutcome::AlreadyCancelled => {
692            return Err(AxiamError::AuthorizationDenied {
693                reason: "cancel token already used".into(),
694                action: None,
695                resource_id: None,
696            }
697            .into());
698        }
699        CancelOutcome::Expired => {
700            return Err(AxiamError::AuthorizationDenied {
701                reason: "grace window expired — cannot cancel".into(),
702                action: None,
703                resource_id: None,
704            }
705            .into());
706        }
707        CancelOutcome::Erasing => {
708            return Err(AxiamError::Conflict {
709                reason: "the account's erasure has already started; the deletion can no longer \
710                         be cancelled"
711                    .into(),
712            }
713            .into());
714        }
715    }
716
717    Ok(HttpResponse::Ok().json(CancelledResponse { cancelled: true }))
718}

720// ---------------------------------------------------------------------------
721// GDPR Art. 7 — OIDC scope-release consent (X7 G8 / W7)
722// ---------------------------------------------------------------------------
723//
724// The plan's §4.8 says withdrawal is "an entry on the user's GDPR self-service
725// page (the existing consent list)". There was no such list, and no consent
726// endpoint of any kind: `consent` rows were written by registration and read
727// only by the Art. 15 export. So the three endpoints below are new, and the
728// first of them is the list the plan assumed.
729//
730// All three are strictly self-service. An administrator cannot consent on a
731// subject's behalf — that is what Art. 4(11)'s "freely given, specific,
732// informed and unambiguous indication of the data subject's wishes" rules
733// out — and there is deliberately no `user_id` parameter to try it with,
734// unlike the export and erasure endpoints above, where acting for a subject on
735// their request is exactly the point.
736
737/// One consent record, as the subject sees it.
738#[derive(Debug, Serialize, utoipa::ToSchema)]
739pub struct ConsentView {
740    /// What was consented to, e.g. `terms_of_service` or
741    /// `oidc_scope_release:<client_id>`.
742    pub consent_type: String,
743    /// The document version or, for a scope release, the consented scopes.
744    pub version: String,
745    pub accepted_at: chrono::DateTime<Utc>,
746    /// Whether this record can be withdrawn here.
747    ///
748    /// `false` for `terms_of_service`: withdrawing it is not a consent
749    /// operation but an erasure, and it has its own endpoint with its own
750    /// grace period. Reported rather than silently absent so the self-service
751    /// page can show the record and explain it.
752    pub withdrawable: bool,
753}
754
755/// Body for recording an OIDC scope-release consent.
756#[derive(Debug, Deserialize, utoipa::ToSchema)]
757pub struct GrantScopeConsent {
758    /// The relying party the claims would be released to.
759    pub client_id: String,
760    /// The sensitive scopes being consented to. Order does not matter; the
761    /// record is written in the canonical order so that the same consent has
762    /// one name.
763    pub scopes: Vec<String>,
764}
765
766/// `GET /api/v1/account/consents` — the caller's own consent records.
767///
768/// Art. 7(1) says a controller must be able to demonstrate that consent was
769/// given; Art. 15(1)(a) says the subject may see what is held about them. This
770/// endpoint is the second, and it is also the page the withdrawal control
771/// lives on.
772#[utoipa::path(
773    get,
774    path = "/api/v1/account/consents",
775    tag = "gdpr",
776    responses(
777        (status = 200, description = "The caller's consent records",
778         body = Vec<ConsentView>),
779        (status = 401, description = "Not authenticated"),
780    ),
781    security(("bearer" = []))
782)]
783pub async fn list_own_consents<C: Connection + Clone>(
784    user: AuthenticatedUser,
785    state: web::Data<AppState<C>>,
786) -> Result<HttpResponse, AxiamApiError> {
787    let records = axiam_core::repository::ConsentRepository::list_by_user(
788        &state.gdpr.consent_repo,
789        user.principal_tenant_id,
790        user.user_id,
791    )
792    .await?;
793
794    let view: Vec<ConsentView> = records
795        .into_iter()
796        .map(|c| ConsentView {
797            withdrawable: c
798                .consent_type
799                .starts_with(axiam_core::repository::OIDC_SCOPE_RELEASE_CONSENT_PREFIX),
800            consent_type: c.consent_type,
801            version: c.version,
802            accepted_at: c.accepted_at,
803        })
804        .collect();
805    Ok(HttpResponse::Ok().json(view))
806}

808/// `POST /api/v1/account/consents/oidc-scopes` — record a scope-release consent.
809///
810/// What the SPA's consent screen calls when the end user says yes. Everything
811/// it accepts is checked against the registration rather than taken on trust,
812/// because a consent record is the thing UserInfo releases personal data on
813/// the strength of:
814///
815/// * the scopes must be sensitive ones — nothing else belongs in this
816///   namespace, and a record naming `openid` would be a record that never
817///   matches and never expires;
818/// * the client must exist in the caller's tenant;
819/// * the client must have every named scope **registered**, so a consent
820///   cannot be recorded for a release the client could never have been
821///   authorised for;
822/// * the tenant switch must be on, so consent collected while the capability
823///   is off cannot sit waiting for somebody to turn it on.
824///
825/// Idempotent: the `(tenant, user, type, version)` index makes a repeated
826/// grant the same grant, and a second call is answered `200` rather than a
827/// conflict. A consent screen the user double-submits has consented once.
828#[utoipa::path(
829    post,
830    path = "/api/v1/account/consents/oidc-scopes",
831    tag = "gdpr",
832    request_body = GrantScopeConsent,
833    responses(
834        (status = 200, description = "Consent recorded"),
835        (status = 400, description = "Unknown client, unregistered scope, \
836                                      non-sensitive scope, or the capability is off"),
837        (status = 401, description = "Not authenticated"),
838    ),
839    security(("bearer" = []))
840)]
841pub async fn grant_oidc_scope_consent<C: Connection + Clone>(
842    http_req: actix_web::HttpRequest,
843    user: AuthenticatedUser,
844    state: web::Data<AppState<C>>,
845    body: web::Json<GrantScopeConsent>,
846) -> Result<HttpResponse, AxiamApiError> {
847    let body = body.into_inner();
848
849    // The client must exist in the caller's tenant whichever gate this
850    // consent is for. Read first, because it is what decides which gate it is.
851    let client = axiam_core::repository::OAuth2ClientRepository::get_by_client_id(
852        &state.oauth2_client_repo,
853        user.principal_tenant_id,
854        &body.client_id,
855    )
856    .await?;
857
858    // T21.4 / D4 — the external-client lane.
859    //
860    // For a client an administrator did not create, the consent being recorded
861    // is not "you may release my address", it is "this application may act as
862    // me". It therefore covers the **whole** requested scope set rather than
863    // the sensitive subset, and its version carries a prefix so the two kinds
864    // of record can share one `consent_type` — which is what makes
865    // `DELETE /account/consents/oidc-scopes/{client_id}` withdraw both with no
866    // new control. See `axiam_oauth2::external_consent`.
867    //
868    // The two lanes never overlap: the settings layer refuses `address` and
869    // `phone` in `dcr_allowed_scopes`, so a self-registered client cannot hold
870    // a scope the W7 lane governs.
871    let (consent_type, version) = if axiam_oauth2::external_consent::applies_to(client.managed_by) {
872        // Every scope must be registered on the client, for the reason the
873        // W7 lane checks the same thing: a consent recorded for a scope
874        // the client could never be authorised for authorises nothing and
875        // will never match at the authorization endpoint.
876        if let Some(missing) = body
877            .scopes
878            .iter()
879            .find(|s| !client.scopes.iter().any(|r| r == *s))
880        {
881            return Err(AxiamApiError::from(AxiamError::Validation {
882                message: format!(
883                    "this client has no {missing} scope registered, so consenting to it \
884                         would authorise nothing"
885                ),
886            }));
887        }
888        (
889            axiam_oauth2::external_consent::consent_type(&body.client_id),
890            axiam_oauth2::external_consent::version(&body.scopes),
891        )
892    } else {
893        // W7's lane, unchanged. Only the sensitive scopes, and in
894        // canonical order: `requested` filters and orders in one step, so
895        // a body naming `openid` yields an empty set and is refused here
896        // rather than silently recorded.
897        let wanted = axiam_oauth2::sensitive::requested(&body.scopes);
898        if wanted.is_empty() || wanted.len() != body.scopes.len() {
899            return Err(AxiamApiError::from(AxiamError::Validation {
900                message: format!(
901                    "consent may be recorded only for the scopes {}, and for nothing else",
902                    axiam_oauth2::sensitive::SENSITIVE_SCOPES.join(", ")
903                ),
904            }));
905        }
906        if let Some(missing) = wanted
907            .iter()
908            .find(|s| !client.scopes.iter().any(|r| r == *s))
909        {
910            return Err(AxiamApiError::from(AxiamError::Validation {
911                message: format!(
912                    "this client has no {missing} scope registered, so consenting to \
913                         release it would authorise nothing"
914                ),
915            }));
916        }
917
918        let tenant = state
919            .tenant_repo
920            .get_by_id(user.principal_tenant_id)
921            .await?;
922        let settings = axiam_core::repository::SettingsRepository::get_effective_settings(
923            &state.settings_repo,
924            tenant.organization_id,
925            user.principal_tenant_id,
926        )
927        .await?;
928        if !settings.oidc.sensitive_scopes_enabled {
929            return Err(AxiamApiError::from(AxiamError::Validation {
930                message: "the address and phone scopes are not enabled for this tenant".into(),
931            }));
932        }
933
934        (
935            axiam_oauth2::sensitive::consent_type(&body.client_id),
936            axiam_oauth2::sensitive::consent_version(&wanted),
937        )
938    };
939
940    // Idempotence, checked rather than relying on the unique index's error:
941    // a second grant is the same grant, and answering it with a conflict
942    // would make a double-submitted consent screen look broken.
943    let already = axiam_core::repository::ConsentRepository::list_by_user(
944        &state.gdpr.consent_repo,
945        user.principal_tenant_id,
946        user.user_id,
947    )
948    .await?
949    .into_iter()
950    .any(|c| c.consent_type == consent_type && c.version == version);
951
952    if !already {
953        axiam_core::repository::ConsentRepository::create(
954            &state.gdpr.consent_repo,
955            axiam_core::models::gdpr::CreateConsent {
956                tenant_id: user.principal_tenant_id,
957                user_id: user.user_id,
958                consent_type: consent_type.clone(),
959                version: version.clone(),
960                ip_address: crate::extractors::client_info::client_ip(&http_req),
961                user_agent: crate::extractors::client_info::user_agent(&http_req),
962            },
963        )
964        .await?;
965
966        // Art. 7(1) proof lives in two places on purpose: the record, which is
967        // live state and can be withdrawn, and the audit log, which is
968        // append-only and cannot. Scope *names*, never the values they would
969        // release.
970        audit_consent(
971            &state,
972            &http_req,
973            &user,
974            "gdpr.oidc_scope_consent_granted",
975            &body.client_id,
976            &version,
977        )
978        .await;
979    }
980
981    Ok(HttpResponse::Ok().json(serde_json::json!({
982        "consent_type": consent_type,
983        "version": version,
984    })))
985}

987/// `DELETE /api/v1/account/consents/oidc-scopes/{client_id}` — withdraw.
988///
989/// Art. 7(3): as easy to withdraw as to give. One call, no grace period, no
990/// confirmation step, and it takes effect on the **next UserInfo call with the
991/// token the relying party already holds** — not on the next token. That is
992/// the property T8.4 asserts, and it is why the release gate re-reads the
993/// record on every call rather than trusting the one taken at authorization.
994///
995/// Withdraws every scope set consented to for this relying party, not one of
996/// them: a subject saying "stop giving my address to this app" does not mean
997/// "stop giving it under the two-scope record but carry on under the
998/// one-scope one".
999///
1000/// Answers `200` whether or not anything was there, and says how many records
1001/// went. A subject who withdraws twice is not told off, and an attacker who
1002/// guesses `client_id`s learns nothing from the status code — though they
1003/// would have to be the subject to ask at all.
1004#[utoipa::path(
1005    delete,
1006    path = "/api/v1/account/consents/oidc-scopes/{client_id}",
1007    tag = "gdpr",
1008    params(("client_id" = String, Path, description = "The relying party to stop releasing to")),
1009    responses(
1010        (status = 200, description = "Consent withdrawn (or there was none)"),
1011        (status = 401, description = "Not authenticated"),
1012    ),
1013    security(("bearer" = []))
1014)]
1015pub async fn withdraw_oidc_scope_consent<C: Connection + Clone>(
1016    http_req: actix_web::HttpRequest,
1017    user: AuthenticatedUser,
1018    state: web::Data<AppState<C>>,
1019    path: web::Path<String>,
1020) -> Result<HttpResponse, AxiamApiError> {
1021    let client_id = path.into_inner();
1022    let consent_type = axiam_oauth2::sensitive::consent_type(&client_id);
1023
1024    let removed = axiam_core::repository::ConsentRepository::withdraw(
1025        &state.gdpr.consent_repo,
1026        user.principal_tenant_id,
1027        user.user_id,
1028        &consent_type,
1029    )
1030    .await?;
1031
1032    if removed > 0 {
1033        audit_consent(
1034            &state,
1035            &http_req,
1036            &user,
1037            "gdpr.oidc_scope_consent_withdrawn",
1038            &client_id,
1039            "",
1040        )
1041        .await;
1042    }
1043
1044    Ok(HttpResponse::Ok().json(serde_json::json!({ "withdrawn": removed })))
1045}

1050/// there are none to carry at this point, and there will not be any at this
1051/// point in a future version either. Failure is logged and swallowed: refusing
1052/// a withdrawal because the audit store is unavailable would make Art. 7(3)
1053/// conditional on infrastructure.
1054async fn audit_consent<C: Connection + Clone>(
1055    state: &AppState<C>,
1056    http_req: &actix_web::HttpRequest,
1057    user: &AuthenticatedUser,
1058    action: &str,
1059    client_id: &str,
1060    scopes: &str,
1061) {

1063        .audit_repo
1064        .append(CreateAuditLogEntry {
1065            tenant_id: user.principal_tenant_id,
1066            actor_id: user.user_id,
1067            actor_type: ActorType::User,
1068            action: action.to_string(),
1069            resource_id: None,
