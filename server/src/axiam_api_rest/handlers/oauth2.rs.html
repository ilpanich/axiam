41/// Cap for the caller-supplied `client_id` recorded in a failed-client-auth
42/// audit row (SEC-087). Server-generated ids are `oa_`/`sa_` + a 32-char
43/// encoding, so this leaves generous headroom for a legitimate value while
44/// bounding what an anonymous caller can push into the log.
45const MAX_CLIENT_ID_LEN: usize = 128;
46
47/// Truncate to at most [`MAX_CLIENT_ID_LEN`] **bytes**, cutting on a character
48/// boundary so the result is still valid UTF-8.
49///
50/// §22.3 residual 3: the previous `chars().take(N)` capped code points, not
51/// bytes, so a multibyte id could still store 4× the intended size. The cap
52/// exists to bound what an anonymous caller can push into an append-only log,
53/// and a bound expressed in the wrong unit is not the bound that was intended.
54fn truncate_bytes_on_char_boundary(s: &str) -> String {
55    if s.len() <= MAX_CLIENT_ID_LEN {
56        return s.to_owned();
57    }
58    let mut end = MAX_CLIENT_ID_LEN;
59    while end > 0 && !s.is_char_boundary(end) {
60        end -= 1;
61    }
62    s[..end].to_owned()
63}

177/// Read the authentication behind this request's session (X7.2, plan §4.3).
178///
179/// The evidence is snapshotted onto the authorization code, so it has to be
180/// read here, while the session still exists: refresh rotation replaces the
181/// row, and by the time the code is redeemed the session it was minted from
182/// may be a different row with a different id.
183///
184/// # It never fails the request
185///
186/// A session that cannot be read yields empty evidence and a `debug!` line,
187/// not an error. Two reasons, and both are invariant 4: an authorization
188/// request that works today must not start failing because of a column added
189/// this wave, and the principal on this path is not always a browser session
190/// at all (a bearer token minted for a service, a session already rotated
191/// away). Empty evidence is also the strict reading — a code with no
192/// `auth_time` can never present as fresh.
193async fn resolve_session_evidence<C: Connection + Clone>(
194    state: &AppState<C>,
195    tenant_id: Uuid,
196    session_id: Uuid,
197) -> SessionEvidence {
198    match state.session_repo.get_by_id(tenant_id, session_id).await {
199        Ok(session) => SessionEvidence {
200            auth_time: Some(session.authenticated_at),
201            amr: session.amr,
202        },
203        Err(e) => {
204            tracing::debug!(
205                error = %e,
206                %session_id,
207                "no session row behind this authorization request; the code is issued \
208                 without authentication evidence"
209            );
210            SessionEvidence::default()
211        }
212    }
213}

215/// Record the outcome of a `prompt=none` authorization request (W4, plan §4.2).
216///
217/// `prompt=none` is a **silent-authentication oracle**: a registered relying
218/// party learns, without any interaction, whether this browser is signed in.
219/// That is inherent to the parameter and it is accepted — it is bounded to
220/// clients registered in this tenant with exact `redirect_uri` matching, and
221/// nothing but one bit crosses — but "accepted" and "invisible" are different
222/// things. One audit row per outcome is what makes abuse of it something an
223/// operator can see and count rather than something they have to be told about
224/// by the party doing it.
225///
226/// Two actions are emitted, `oauth2.prompt_none.code` and
227/// `oauth2.prompt_none.login_required`. The plan names a third,
228/// `oauth2.prompt_none.consent_required`, which **cannot** be emitted in W4:
229/// it needs a consent-gated scope and there are none until W7 (plan §4.8). It
230/// is not written here rather than written into a branch that can never run.
231///
232/// Never fails the request. An audit sink that is down costs a row, not a
233/// login (T-15-04).
234async fn audit_prompt_none<C: Connection + Clone>(
235    state: &AppState<C>,
236    http_req: &HttpRequest,
237    tenant_id: Uuid,
238    client_id: &str,
239    refusal: Option<&OAuth2Error>,
240) {
241    let (action, result) = match refusal {
242        None => ("oauth2.prompt_none.code", AuditOutcome::Success),
243        Some(e) => (
244            match e.error_code() {
245                "login_required" => "oauth2.prompt_none.login_required",
246                _ => "oauth2.prompt_none.refused",
247            },
248            AuditOutcome::Failure,
249        ),
250    };
251    if let Err(e) = state
252        .audit_repo
253        .append(CreateAuditLogEntry {
254            tenant_id,
255            actor_id: Uuid::nil(),
256            actor_type: ActorType::System,
257            action: action.into(),
258            resource_id: None,
259            outcome: result,
260            ip_address: peer_ip(http_req),
261            metadata: Some(serde_json::json!({
262                // The `client_id` is the point of the row: the oracle is
263                // per-client, so counting it per client is what makes a
264                // relying party polling every few seconds visible.
265                "client_id": truncate_bytes_on_char_boundary(client_id),
266                "error": refusal.map(OAuth2Error::error_code),
267            })),
268        })
269        .await
270    {
271        tracing::error!(
272            error = %e,
273            %tenant_id,
274            "could not record a prompt=none outcome; the authorization request is unaffected"
275        );
276    }
277}

279/// Decode an `id_token_hint` presented at the authorization endpoint (W4,
280/// plan §4.2).
281///
282/// One decoder, shared with `/oauth2/end_session`
283/// ([`axiam_oauth2::logout::decode_id_token_hint`]): the signature is checked,
284/// the expiry deliberately is not — an `id_token_hint` has routinely expired,
285/// which is exactly why an RP is sending one rather than a live token — and
286/// `aud` is read out rather than validated against, because who the token was
287/// for is part of what the caller wants to learn. A second decoder here would
288/// be a second opinion about what a hint proves.
289///
290/// Returns `None` when no hint arrived and when one arrived and did not
291/// verify. The caller distinguishes them by
292/// [`AuthnRequestParams::id_token_hint`], and treats the second as a hint that
293/// names somebody else rather than as an absent one — dropping an unverifiable
294/// hint would let any string turn a `prompt=none` refusal into a code.
295///
296/// Run for every request carrying the parameter, not only for the honour lane.
297/// It costs one Ed25519 verification of a value the request already carried,
298/// it cannot change any response on the `ignore` lane (nothing reads the
299/// result there), and making it conditional would put the lane decision in a
300/// second place.
301fn decode_authorize_id_token_hint<C: Connection + Clone>(
302    state: &AppState<C>,
303    params: &AuthnRequestParams,
304) -> Option<axiam_oauth2::logout::IdTokenHint> {
305    let raw = params.id_token_hint.as_deref()?;
306    axiam_oauth2::logout::decode_id_token_hint(raw, &state.auth_config.jwt_public_key_pem)
307}

309/// Classify a request object on an authorization request (X7 G12, plan §4.10).
310///
311/// AXIAM accepts neither form. `request` is RFC 9101's request object by
312/// value; a `request_uri` that is not a `urn:ietf:params:oauth:request_uri:`
313/// handle is a request object by reference, which would have the authorization
314/// server fetch an attacker-chosen URL — an SSRF primitive that PAR (RFC 9126
315/// §1) made unnecessary and that FAPI 2.0 does not ask for.
316///
317/// `request` is checked first: when both arrive, the by-value object is the
318/// one the server would have had to *read*, so it is the one the refusal
319/// should name.
320///
321/// Returns `None` for a request carrying no request object at all, including
322/// one whose `request_uri` is a genuine PAR handle — that path is untouched.
323fn classify_request_object(
324    request: Option<&str>,
325    request_uri: Option<&str>,
326) -> Option<RequestObject> {
327    // A blank value is a client library filling in a template, not an object.
328    // Refusing it would break a request that works today and carries nothing.
329    fn present(v: Option<&str>) -> Option<&str> {
330        v.map(str::trim).filter(|s| !s.is_empty())
331    }
332
333    if present(request).is_some() {
334        return Some(RequestObject::ByValue);
335    }
336    match present(request_uri) {
337        Some(uri) if !uri.starts_with(axiam_oauth2::par::REQUEST_URI_PREFIX) => {
338            Some(RequestObject::ByReference)
339        }
340        _ => None,
341    }
342}

353/// The principal an authorization request is acting for (W3, plan §4.0).

367/// Is this `return_to` a path on an origin this deployment owns?
368///
369/// The second of the two server-side checks the plan asks for. The first is
370/// [`axiam_oauth2::login_hop::validate_return_to`], which is syntactic and
371/// demands the exact authorize path; this one *resolves* the candidate against
372/// the issuer and hands the result to
373/// [`require_deployment_spa_origin`](crate::handlers::federation_login::require_deployment_spa_origin)
374/// — the rule that already decides where a federation SSO handoff code may be
375/// sent, rather than a second rule that could come to disagree with it.
376///
377/// While the syntactic check demands `/oauth2/authorize?…` this is redundant,
378/// and it is kept for the case that stops being true: `Url::join` is what turns
379/// `//evil.example` into a different host and `/a/../../b` into `/b`, so it is
380/// an independent opinion on the two attacks that are about *resolution*
381/// rather than about spelling.
382fn return_to_is_on_this_deployment<C: Connection + Clone>(
383    state: &AppState<C>,
384    return_to: &str,
385) -> bool {
386    let Ok(base) = url::Url::parse(state.auth_config.effective_issuer()) else {
387        return false;
388    };
389    let Ok(resolved) = base.join(return_to) else {
390        return false;
391    };
392    if resolved.path() != axiam_oauth2::login_hop::AUTHORIZE_PATH {
393        return false;
394    }
395    crate::handlers::federation_login::require_deployment_spa_origin(state, resolved.as_str())
396        .is_ok()
397}

399/// Resolve who this authorization request is acting for (W3, plan §4.0).
400///
401/// Two ways in, in this order:
402///
403/// 1. **An access token** — the `axiam_access` cookie or a `Bearer`/`DPoP`
404///    header, via [`AuthenticatedUser`]. Unchanged, and it wins whenever it
405///    succeeds. Every request that works today takes this path and cannot tell
406///    that the other exists.
407/// 2. **The `axiam_op_session` cookie**, and only for a client registered
408///    `browser_sso`. This is the path that makes AXIAM usable as an OP for a
409///    third-party relying party at all: `axiam_access` is `SameSite=Strict`, so
410///    the cross-site top-level navigation an RP redirect *is* never carries it,
411///    and until this wave a signed-in user arrived here anonymous.
412///
413/// # Order of operations, and why the client is loaded first
414///
415/// The client must be read before the cookie is, because the client's
416/// registration is what decides whether the cookie is consulted. That is the
417/// reordering plan §4.0 describes, and it is not a new trust decision — the
418/// authorize service already loads this row before any redirectable error, so
419/// the lookup moves one step earlier rather than appearing.
420///
421/// # Every anonymous refusal is today's refusal
422///
423/// No tenant, no such client, a client that did not opt in, a repository that
424/// failed: all four return the **same** 401 the extractor would have produced,
425/// byte for byte, by returning the extractor's own error object. That is
426/// invariant 4 and it is also what keeps the endpoint from becoming an oracle:
427/// an anonymous caller cannot learn from the response whether a `client_id`
428/// exists in a tenant, only whether it opted into the hop.
429///
430/// # Errors
431///
432/// Returns a complete [`HttpResponse`] — the 401, the 302 into the login page,
433/// or the loop guard's terminal `login_required` — because each of the three
434/// is finished at the point it is decided and none is expressible as an
435/// [`OAuth2Error`] the caller could usefully re-render. Boxed: see
436/// [`AuthorizeRefusal`].
437async fn resolve_authorize_principal<C: Connection + Clone>(
438    state: &AppState<C>,
439    http_req: &HttpRequest,
440    q: &AuthorizeQuery,
441    maybe_user: MaybeAuthenticatedUser,
442) -> Result<AuthorizePrincipal, AuthorizeRefusal> {
443    use actix_web::ResponseError;
444
445    let auth_error = match maybe_user.into_result() {
446        Ok(user) => {
447            // Path 1. `tenant_id` on the query is ignored here, deliberately:
448            // the tenant a token was minted for is the tenant it acts in, and
449            // letting a query parameter move that would be a tenant-crossing
450            // primitive handed to whoever holds the browser.
451            return Ok(AuthorizePrincipal {
452                tenant_id: user.tenant_id,
453                user_id: user.user_id,
454                session_id: user.session_id,
455            });
456        }
457        Err(e) => e,
458    };
459
460    // ---- Path 2: anonymous ------------------------------------------------
461    let Some(tenant_id) = q.tenant_id else {
462        return Err(Box::new(auth_error.error_response()));
463    };
464    let Ok(client) = state
465        .oauth2_client_repo
466        .get_by_client_id(tenant_id, &q.client_id)
467        .await
468    else {
469        return Err(Box::new(auth_error.error_response()));
470    };
471    if !client.browser_sso {
472        // **T0.1 / T0.5 / M7's I4 twin.** Every client registered today lands
473        // here, including one whose browser happens to be carrying an
474        // `axiam_op_session` cookie: the cookie is not read, because reading it
475        // is what `browser_sso` opts into.
476        return Err(Box::new(auth_error.error_response()));
477    }
478
479    // The person said no.
480    //
481    // Answered here, before the session cookie is consulted, because the answer
482    // does not depend on it: somebody who cancels the sign-in page has by
483    // definition not signed in, and asking whether they might have a session
484    // from some earlier request would let a stale cookie turn a refusal into a
485    // grant. The two conditions above still apply first — an unknown client and
486    // a client that never opted into the hop are answered exactly as they are
487    // today, because this endpoint must not become an oracle for either.
488    //
489    // Delivered by redirect on the same terms as every other authorization
490    // error (RFC 6749 §4.1.2.1): only to a `redirect_uri` this client
491    // registered, compared exactly, with the request's own `state`. An
492    // unregistered or absent one is answered directly, because a refusal is
493    // still not a licence to send a browser somewhere the client never named.
494    if axiam_oauth2::login_hop::user_declined(q.user_declined.as_deref()) {
495        let refusal = OAuth2Error::AccessDenied(
496            "the end user declined the authorization request at the sign-in page".into(),
497        );
498
499        // Where to answer, and with whose `state` — and for a pushed request
500        // neither comes from the query string.
501        //
502        // RFC 9126 §1: the pushed copy is the request. A FAPI client sends
503        // `client_id` and `request_uri` to the authorization endpoint and may
504        // send nothing else, so reading `state` off the query answered a
505        // refusal with no `state` at all — which the suite reports as
506        // `CheckStateInAuthorizationResponse: State was passed in request, but
507        // is missing from response`, and which a relying party would be right
508        // to discard as an unsolicited response (RFC 6749 §10.12).
509        //
510        // Consuming the `request_uri` here is correct rather than merely
511        // convenient: it is single-use, and this request has just been answered
512        // terminally. Leaving it spendable would let the same pushed request be
513        // presented again after its user said no.
514        let pushed = match q.request_uri.as_deref() {
515            Some(uri) => state
516                .oauth2
517                .par_service
518                .consume(tenant_id, &q.client_id, uri)
519                .await
520                .ok(),
521            None => None,
522        };
523        let (redirect_uri, echo_state) = match pushed.as_ref() {
524            Some(p) => (Some(p.redirect_uri.as_str()), p.state.as_deref()),
525            None => (q.redirect_uri.as_deref(), q.state.as_deref()),
526        };
527
528        return Err(Box::new(match redirect_uri {
529            Some(uri) if client.redirect_uris.iter().any(|r| r == uri) => {
530                build_error_redirect(uri, &refusal, echo_state, &state.auth_config)
531            }
532            _ => authorize_error_response(http_req, &refusal),
533        }));
534    }
535
536    // The cookie, and whether it still names a live session in this tenant.
537    let presented = http_req.cookie(crate::middleware::csrf::COOKIE_OP_SESSION);
538    let resolved = match presented.as_ref() {
539        Some(cookie) => {
540            let digest = axiam_auth::token::hash_browser_session_token(cookie.value());
541            match state
542                .session_repo
543                .get_by_browser_token_hash(tenant_id, &digest)
544                .await
545            {
546                Ok(found) => found,
547                Err(e) => {
548                    // A read failure is not a principal. Logged without the
549                    // cookie value or its digest — the first is a credential and
550                    // the second is enough to recognise one.
551                    tracing::warn!(
552                        error = %e,
553                        %tenant_id,
554                        "could not resolve the OP browser session; treating the \
555                         request as anonymous"
556                    );
557                    None
558                }
559            }
560        }
561        None => None,
562    };
563
564    if let Some(session) = resolved {
565        return Ok(AuthorizePrincipal {
566            tenant_id,
567            user_id: session.user_id,
568            session_id: session.id,
569        });
570    }
571
572    // ---- No principal: refuse silently, hop, or stop ---------------------
573    //
574    // W4 (plan §4.2, T1.1/T1.7). `prompt=none` says: answer without showing
575    // the end user anything. There is nothing to answer with — no session
576    // resolved — so the answer is `login_required`, **redirected to the
577    // relying party** rather than to a sign-in page it forbade.
578    //
579    // Read from the query string only. A pushed request's `prompt` cannot be
580    // seen here at all (the handle is consumed later, inside the handler), so
581    // a PAR client's `prompt=none` takes the hop first and is refused on the
582    // return leg instead, by the marker rule in `axiam_oauth2::honour` — the
583    // interaction is never converted into a code either way.
584    //
585    // Honour lane only, and that is what keeps T0.1 exact: a `browser_sso`
586    // client registered `ignore` — which is every client that exists — sending
587    // `prompt=none` gets today's answer, byte for byte.
588    //
589    // W5 (plan §4.5/§4.6) reads the same bundle for the cosmetic four, which is
590    // why it is parsed once here with every field rather than only `prompt`.
591    // Query string only, as before: a pushed request's parameters cannot be
592    // seen at this point (the handle is consumed inside the handler, after a
593    // principal exists), so a PAR client's presentation reaches the page on the
594    // *interaction* arm below instead. Both arms build their URL through the
595    // one builder, so the two carriers cannot come to disagree about what a
596    // `/login` URL looks like.
597    let honour = client.authn_request_params.is_honour();
598    let authn_params = honour.then(|| {
599        AuthnRequestParams::parse(&RawAuthnParams {
600            prompt: q.prompt.as_deref(),
601            login_hint: q.login_hint.as_deref(),
602            display: q.display.as_deref(),
603            ui_locales: q.ui_locales.as_deref(),
604            // `claims_locales` is deliberately not read: it selects the
605            // language of *claim values*, which AXIAM does not localise, and
606            // it must not reach the page's language. See
607            // `axiam_oauth2::login_hop::Cosmetic::from_params`.
608            ..Default::default()
609        })
610    });
611    if let Some(params) = authn_params.as_ref()
612        && params
613            .prompt
614            .contains(&axiam_oauth2::authn_params::Prompt::None)
615    {
616        let refusal = OAuth2Error::LoginRequired(
617            "no end user is authenticated at this authorization server, and prompt=none \
618                 forbids asking them to sign in"
619                .into(),
620        );
621        audit_prompt_none(state, http_req, tenant_id, &q.client_id, Some(&refusal)).await;
622        // RFC 6749 §4.1.2.1: an error is redirected only to a
623        // `redirect_uri` this client registered. Exact match, the same
624        // comparison `AuthorizeService::authorize` makes — an unregistered
625        // or absent one is answered directly instead.
626        return Err(Box::new(match q.redirect_uri.as_deref() {
627            Some(uri) if client.redirect_uris.iter().any(|r| r == uri) => {
628                build_error_redirect(uri, &refusal, q.state.as_deref(), &state.auth_config)
629            }
630            _ => authorize_error_response(http_req, &refusal),
631        }));
632    }
633
634    // The loop guard. A request that already carries the marker is one this
635    // server sent to the login page and got back; sending it there a second
636    // time is the non-terminating case, so it is answered instead. See
637    // `axiam_oauth2::login_hop` for the full argument.
638    if axiam_oauth2::login_hop::is_return_leg(q.login_hop.as_deref()) {
639        tracing::warn!(
640            %tenant_id,
641            client_id = %q.client_id,
642            "an authorization request returned from the login hop still \
643             carrying no OP session; refusing to redirect again"
644        );
645        return Err(Box::new(authorize_error_response(
646            http_req,
647            &OAuth2Error::LoginRequired(
648                "the sign-in did not establish a session for this tenant at this \
649                 origin; sign in again from the relying party, and check that the \
650                 browser accepts the axiam_op_session cookie"
651                    .into(),
652            ),
653        )));
654    }
655
656    let Some(return_to) = axiam_oauth2::login_hop::build_return_to(http_req.query_string()) else {
657        // Nothing safe to come back to. Answer as if the request had been
658        // anonymous with no `browser_sso` at all rather than send a browser
659        // somewhere on a value that did not validate.
660        return Err(Box::new(auth_error.error_response()));
661    };
662    if !return_to_is_on_this_deployment(state, &return_to) {
663        return Err(Box::new(auth_error.error_response()));
664    }
665
666    // A cookie that was presented and did not resolve is stale: this browser
667    // believes it is signed in and is not. `reauth` tells the SPA to say so and
668    // to authenticate afresh instead of trusting what it already holds, and the
669    // dead cookie is removed on the way out so the next attempt starts clean.
670    let stale = presented.is_some();
671    // W5. The presentation the relying party asked for, on the honour lane and
672    // nowhere else: `authn_params` is `None` for every client registered
673    // `ignore`, so `Cosmetic::NONE` is what they get and their `/login` URL is
674    // byte-identical to the one W3 built. Invariant 4, as a type rather than as
675    // a branch somebody has to remember to write.
676    //
677    // W7 — the tenant default, which W5 deferred for want of a place to
678    // configure it. Read only when a bundle arrived at all: a client on the
679    // `ignore` lane gets `Cosmetic::NONE` and no settings read, so its
680    // `/login` URL is byte-identical to W3's and costs the same.
681    let tenant_default = match authn_params.as_ref() {
682        Some(_) => tenant_default_locale(state, tenant_id).await,
683        None => None,
684    };
685    let cosmetic = authn_params
686        .as_ref()
687        .map_or(axiam_oauth2::login_hop::Cosmetic::NONE, |params| {
688            axiam_oauth2::login_hop::Cosmetic::from_params(params, tenant_default)
689        });
690    let location =
691        axiam_oauth2::login_hop::build_login_redirect_for(&return_to, stale, None, &cosmetic);
692
693    let mut builder = HttpResponse::Found();
694    builder
695        .append_header((actix_web::http::header::LOCATION, location))
696        // The URL carries the whole authorization request; it must not be
697        // cached, and it must not travel onward as a referrer.
698        .append_header((actix_web::http::header::CACHE_CONTROL, "no-store"))
699        .append_header(("Referrer-Policy", "no-referrer"));
700    if stale {
701        builder.cookie(crate::middleware::csrf::clear_op_session_cookie());
702    }
703    Err(Box::new(builder.finish()))
704}

707/// `ui_locales` chain (plan §4.6's deferral, now picked up).
708///
709/// W5 shipped `select_ui_locale(requested, tenant_default)` and unit-tested it
710/// through all three steps, then passed `None` at both call sites because there
711/// was nowhere for an operator to set the value. There is now:
712/// `TenantSettingsOverride::default_locale`, on the settings surface every
713/// other per-tenant control lives on. This is the one argument those two call
714/// sites were waiting for.
715///
716/// Three ways to get `None`, and all three land on the deployment default
717/// (`en`), which is what every deployment did before this wave:
718///
719/// * the tenant has no preference;
720/// * the settings could not be read (a login page is not worth failing over a
721///   language);
722/// * the stored tag is one this build does not ship. `Locale::from_tag` is an
723///   **exact** match rather than an RFC 4647 lookup, deliberately: a stored

726async fn tenant_default_locale<C: Connection + Clone>(
727    state: &AppState<C>,
728    tenant_id: Uuid,
729) -> Option<axiam_oauth2::locale::Locale> {
730    let tenant = state.tenant_repo.get_by_id(tenant_id).await.ok()?;
731    let settings = axiam_core::repository::SettingsRepository::get_effective_settings(
732        &state.settings_repo,
733        tenant.organization_id,
734        tenant_id,
735    )
736    .await
737    .ok()?;
738    let tag = settings.oidc.default_locale?;
739    let locale = axiam_oauth2::locale::Locale::from_tag(&tag);
740    if locale.is_none() {
741        tracing::warn!(
742            %tenant_id,
743            configured = %tag,
744            "this tenant's default_locale is not a locale this build ships; the sign-in page \
745             falls back to the deployment default"
746        );
747    }
748    locale
749}

751/// W7 / X7 G8 — resolve what this authorization request's GDPR-sensitive
752/// scopes have earned, so `AuthorizeService` can decide without a repository.
753///
754/// Returns `(state, switch_is_off)`. Two values rather than one because they
755/// come from two reads and `axiam_oauth2::sensitive::decide` checks them
756/// against each other; collapsing them here would be collapsing the check.
757///
758/// # Costs nothing for a request that asks for nothing
759///
760/// The first line answers "did this request name `address` or `phone`", which
761/// is a scan of a short vector, and returns before either database read. Every
762/// client registered today asks for neither — they could not, the scopes were
763/// unregistrable — so this function adds one string comparison per scope to
764/// every authorization request in every existing deployment and nothing else.
765/// That is invariant 4 measured rather than asserted.
766///
767/// # A read that fails is not a release
768///
769/// Both reads fail closed and in different directions, which is deliberate:
770/// a settings read that fails reports the switch as **off** (refuse), and a
771/// consent read that fails reports the consent as **missing** (ask again).
772/// Neither answer can release data, and the second is recoverable by the end
773/// user rather than terminal.
774async fn resolve_sensitive_scopes<C: Connection + Clone>(
775    state: &AppState<C>,
776    tenant_id: Uuid,
777    user_id: Uuid,
778    client_id: &str,
779    scope: Option<&str>,
780) -> (axiam_oauth2::sensitive::Requested, bool) {
781    let scopes: Vec<String> = scope
782        .unwrap_or("")
783        .split_whitespace()
784        .map(String::from)
785        .collect();
786    let wanted = axiam_oauth2::sensitive::requested(&scopes);
787    if wanted.is_empty() {
788        return (axiam_oauth2::sensitive::Requested::None, false);
789    }
790
791    // The organization the settings baseline belongs to. Resolved from the
792    // tenant rather than taken from the caller's principal, the same way
793    // `axiam_auth::lockout` resolves it: the authorization endpoint's
794    // principal type carries no `org_id`, and adding one to it would put a
795    // field on the hot path for a branch almost no request takes.
796    let enabled = match state.tenant_repo.get_by_id(tenant_id).await {
797        Ok(tenant) => match axiam_core::repository::SettingsRepository::get_effective_settings(
798            &state.settings_repo,
799            tenant.organization_id,
800            tenant_id,
801        )
802        .await
803        {
804            Ok(settings) => settings.oidc.sensitive_scopes_enabled,
805            Err(e) => {
806                tracing::error!(
807                    error = %e,
808                    %tenant_id,
809                    "could not read the tenant's effective settings while deciding whether the \
810                     address/phone scopes are enabled; treating them as disabled"
811                );
812                false
813            }
814        },
815        Err(e) => {
816            tracing::error!(
817                error = %e,
818                %tenant_id,
819                "could not resolve the organization behind this tenant; treating the \
820                 address/phone scopes as disabled"
821            );
822            false
823        }
824    };
825    if !enabled {
826        return (axiam_oauth2::sensitive::Requested::Disabled, true);
827    }
828
829    let consent_type = axiam_oauth2::sensitive::consent_type(client_id);
830    let version = axiam_oauth2::sensitive::consent_version(&wanted);
831    let consented = match axiam_core::repository::ConsentRepository::list_by_user(
832        &state.gdpr.consent_repo,
833        tenant_id,
834        user_id,
835    )
836    .await
837    {
838        Ok(records) => records
839            .iter()
840            .any(|c| c.consent_type == consent_type && c.version == version),
841        Err(e) => {
842            tracing::error!(
843                error = %e,
844                %tenant_id,
845                "could not read the end user's consent records; treating the release as \
846                 unconsented, which asks again rather than releasing"
847            );
848            false
849        }
850    };
851
852    (
853        if consented {
854            axiam_oauth2::sensitive::Requested::Consented
855        } else {
856            axiam_oauth2::sensitive::Requested::ConsentMissing
857        },
858        false,
859    )
860}

862/// `GET /oauth2/authorize` -- OAuth2 authorization endpoint.
863///
864/// A request carrying an access token authorizes as its subject. A request
865/// carrying none is answered with a 401 JSON body — unless the client it names
866/// is registered `browser_sso`, in which case it is either recognised by its
867/// `axiam_op_session` cookie or redirected to the sign-in page and brought back
868/// (W3, plan §4.0). See [`resolve_authorize_principal`].
869///
870/// On success, redirects to `redirect_uri?code=...&state=...`.
871/// On error, redirects with `?error=...&error_description=...&state=...`.
872#[utoipa::path(
873    get,
874    path = "/oauth2/authorize",
875    tag = "oauth2",
876    params(AuthorizeQuery),
877    responses(
878        (status = 302, description = "Redirect with authorization code, or to \
879                                      the sign-in page for a browser_sso client"),
880        (status = 400, description = "OAuth2 error", body = OAuth2ErrorResponse),
881        (status = 401, description = "No authenticated principal, and the \
882                                      client did not opt into the login hop"),
883    ),
884    security(("bearer" = []))
885)]
886pub async fn authorize<C: Connection + Clone>(
887    http_req: HttpRequest,
888    maybe_user: MaybeAuthenticatedUser,
889    // W3: a `Result` rather than a plain extractor so that the **order** of
890    // today's refusals is preserved. `AuthenticatedUser` used to be the first
891    // extractor on this handler, so an unauthenticated request with an
892    // unparseable query was answered 401 and never reached the body.
893    // `MaybeAuthenticatedUser` cannot fail, which would have promoted the query
894    // error to first place and turned that 401 into a 400 — a behaviour change
895    // for nobody's benefit. Resolving both here keeps the precedence explicit.
896    query: Result<web::Query<AuthorizeQuery>, actix_web::Error>,
897    state: web::Data<AppState<C>>,
898) -> HttpResponse {
899    use actix_web::ResponseError;
900
901    let q = match query {
902        Ok(q) => q.into_inner(),
903        Err(query_error) => {
904            return match maybe_user.into_result() {
905                Err(auth_error) => auth_error.error_response(),
906                Ok(_) => query_error.error_response(),
907            };
908        }
909    };
910
911    let user = match resolve_authorize_principal(&state, &http_req, &q, maybe_user).await {
912        Ok(principal) => principal,
913        Err(response) => return *response,
914    };
915
916    // X7 G12 (plan §4.10). Classify request objects FIRST, before the PAR
917    // branch below: a `request_uri` that is not a PAR handle would otherwise
918    // fall into `par_service.consume` and come back as a generic
919    // `invalid_request`, losing the code OIDC Core §3.1.2.6 defines and a
920    // conformance suite matches on.
921    //
922    // The marker travels on the `AuthorizeRequest` rather than being answered
923    // here so that the client and its `redirect_uri` are validated first: a
924    // refusal is redirected only to a URI the client actually registered, and
925    // otherwise answered directly.
926    let request_object = classify_request_object(q.request.as_deref(), q.request_uri.as_deref());
927
928    // X7.2 — resolved once, for both carriers: the evidence describes the
929    // session this request arrives in, which is the same session whether the
930    // parameters came inline or through PAR.
931    let session_evidence = resolve_session_evidence(&state, user.tenant_id, user.session_id).await;
932
933    // B5. The pushed copy wins; the query string's copies are IGNORED, not
934    // merged and not refused.
935    //
936    // The distinction is the whole of RFC 9101 §6.3, which RFC 9126 §4 adopts
937    // by reference ("build an authorization request as defined in [RFC9101]"):
938    //
939    //   The authorization server MUST extract the set of authorization request
940    //   parameters from the Request Object value. The authorization server MUST
941    //   only use the parameters in the Request Object, even if the same
942    //   parameter is provided in the query parameter.
943    //
944    // and §5 says in terms that a client MAY send them duplicated. So a
945    // duplicated `response_type`/`redirect_uri`/`scope`/`code_challenge` is a
946    // conformant request, and an earlier revision of this handler refused it —
947    // which failed every FAPI 2.0 authorization module, since the OIDF suite
948    // sends exactly that shape.
949    //
950    // The security argument the refusal was built on is sound and is satisfied
951    // by ignoring rather than refusing: parameter confusion needs the inline
952    // value to be *read* by something, and nothing below reads it. Every field
953    // of the request comes from `params`, the pushed copy — as `state` and
954    // `nonce` already did, for exactly this reason. `client_id` is the one
955    // parameter that is still compared rather than ignored, because §6.3
956    // requires the two to be identical; `par_service.consume` does that by
957    // scoping the handle to the client that pushed it.
958    //
959    // A refused request object takes the inline branch whatever it carried:
960    // there is nothing to consume, and the branch exists only to reach the
961    // validation that decides how the refusal is reported.
962    let request_uri = match request_object {
963        Some(_) => None,
964        None => q.request_uri,
965    };
966    let req = match request_uri {
967        Some(request_uri) => {
968            let params = match state
969                .oauth2
970                .par_service
971                .consume(user.tenant_id, &q.client_id, &request_uri)
972                .await
973            {
974                Ok(params) => params,
975                // W3 (plan §4.0, F10). A pushed request lives 60 seconds. A
976                // user who took longer than that to type a password comes back
977                // here to a handle that no longer exists — and telling them
978                // `invalid_request: request_uri is unknown, expired, or used`
979                // describes a client bug that did not happen. On a return leg,
980                // and only there, the recoverable code is used instead: the
981                // relying party pushes again and restarts, which is RFC 9126
982                // §2.2's own design rather than a workaround for it.
983                //
984                // Narrow on purpose. Only the "gone" refusal is remapped, only
985                // when the marker says this request came back from the login
986                // page, so an ordinary authorization request with a dead handle
987                // gets exactly the answer it got before this wave.
988                Err(e)
989                    if axiam_oauth2::login_hop::is_return_leg(q.login_hop.as_deref())
990                        && axiam_oauth2::par::is_request_uri_gone(&e) =>
991                {
992                    return authorize_error_response(
993                        &http_req,
994                        &OAuth2Error::InvalidRequestUri(
995                            "the pushed authorization request expired while signing in \
996                         (a request_uri lives 60 seconds); push it again and restart \
997                         the authorization request"
998                                .into(),
999                        ),
1000                    );
1001                }
1002                Err(e) => return authorize_error_response(&http_req, &e),
1003            };
1004
1005            // X7.1 — parsed from the *pushed* copy, never from the query
1006            // string. `state` and `nonce` already work this way and for the
1007            // same reason: a parameter the client chose at push time must not
1008            // be substitutable by the browser that carries the handle.
1009            let authn_params = AuthnRequestParams::parse(&RawAuthnParams::from(&params));
1010
1011            // W4 (T1.3) — and this is the other half of that sentence: the
1012            // query string's copies are not merely ignored, they are grounds
1013            // to refuse the request on the honour lane. Parsed rather than
1014            // tested field by field so "an authentication-request parameter
1015            // arrived" means the same thing here as everywhere else, blank
1016            // template values included.
1017            let inline_authn_params_beside_request_uri =
1018                !AuthnRequestParams::parse(&RawAuthnParams {
1019                    prompt: q.prompt.as_deref(),
1020                    max_age: q.max_age.as_deref(),
1021                    acr_values: q.acr_values.as_deref(),
1022                    claims: q.claims.as_deref(),
1023                    id_token_hint: q.id_token_hint.as_deref(),
1024                    login_hint: q.login_hint.as_deref(),
1025                    display: q.display.as_deref(),
1026                    ui_locales: q.ui_locales.as_deref(),
1027                    claims_locales: q.claims_locales.as_deref(),
1028                })
1029                .is_empty();
1030
1031            AuthorizeRequest {
1032                tenant_id: user.tenant_id,
1033                user_id: user.user_id,
1034                response_type: params.response_type,
1035                client_id: q.client_id,
1036                redirect_uri: params.redirect_uri,
1037                scope: params.scope,
1038                // `state` and `nonce` come from the pushed request too: they
1039                // were the client's to choose at push time, and honouring a
1040                // query-string copy would let the browser substitute its own.
1041                state: params.state,
1042                code_challenge: params.code_challenge,
1043                code_challenge_method: params.code_challenge_method,
1044                nonce: params.nonce,
1045                // B5: the browser session this authorization happens within.
1046                // It reaches the ID token as `sid`, which is what lets a
1047                // back-channel logout token name one session rather than the
1048                // subject's every session.
1049                session_id: Some(user.session_id),
1050                via_par: true,
1051                id_token_hint: decode_authorize_id_token_hint(&state, &authn_params),
1052                inline_authn_params_beside_request_uri,
1053                login_hop_return_leg: axiam_oauth2::login_hop::is_return_leg(
1054                    q.login_hop.as_deref(),
1055                ),
1056                authn_params,
1057                request_object,
1058                session_evidence,
1059                consent_hop_return_leg: q.consent_hop.is_some(),
1060                // RFC 9449 §10 — from the PUSHED copy, never the query string,
1061                // for exactly the reason `state` and `nonce` are just above.
1062                // The PAR endpoint already resolved §10.1's two carriers into
1063                // one value under client authentication; a query-string
1064                // `dpop_jkt` here is a browser proposing a key, which is the
1065                // substitution the binding exists to prevent.
1066                dpop_jkt: params.dpop_jkt,
1067                // OIDC Core §5.5 — from the PUSHED copy, like every other
1068                // parameter beside a `request_uri`.
1069                requested_userinfo_claims: params
1070                    .claims
1071                    .as_deref()
1072                    .map(axiam_oauth2::claims_request::userinfo_claims)
1073                    .unwrap_or_default(),
1074                // Resolved below, once, for both carriers.
1075                sensitive_scopes: axiam_oauth2::sensitive::Requested::None,
1076                sensitive_scopes_switch_is_off: false,
1077            }
1078        }
1079        None => {
1080            let authn_params = AuthnRequestParams::parse(&RawAuthnParams {
1081                prompt: q.prompt.as_deref(),
1082                max_age: q.max_age.as_deref(),
1083                acr_values: q.acr_values.as_deref(),
1084                claims: q.claims.as_deref(),
1085                id_token_hint: q.id_token_hint.as_deref(),
1086                login_hint: q.login_hint.as_deref(),
1087                display: q.display.as_deref(),
1088                ui_locales: q.ui_locales.as_deref(),
1089                claims_locales: q.claims_locales.as_deref(),
1090            });
1091            // Cloned into the scrutinee so the else-arm can still read
1092            // `q.redirect_uri`: a `let`-else MOVES what it destructures, and
1093            // the arm's whole job is to decide whether that URI is one this
1094            // server may redirect an error to. Two small `Option<String>`
1095            // clones on a request that is about to be refused.
1096            let (Some(response_type), Some(redirect_uri)) =
1097                (q.response_type.clone(), q.redirect_uri.clone())
1098            else {
1099                // Which of the two is missing decides how the error travels,
1100                // and until this wave it did not: both were answered with a
1101                // JSON body in the browser's window.
1102                //
1103                // RFC 6749 §4.1.2.1 draws the line at whether the server can
1104                // trust where it would be sending the browser. It MUST NOT
1105                // redirect when the `redirect_uri` is missing or does not match
1106                // a registered one — that is the open-redirect case, and a JSON
1107                // body is the right answer. But when `client_id` names a real
1108                // client and the `redirect_uri` is one it registered, the error
1109                // MUST be delivered by redirecting, with `state` echoed, so the
1110                // relying party learns what happened instead of the end user
1111                // reading a machine-readable body they cannot act on.
1112                //
1113                // Found by the OpenID Foundation suite:
1114                // `oidcc-response-type-missing` sends a valid client and a
1115                // registered `redirect_uri` with no `response_type`, waited for
1116                // a redirect that never came, and stalled the whole plan.
1117                // `user.tenant_id`, NOT `q.tenant_id`. The query parameter is
1118                // ignored on this arm for the reason stated where the principal
1119                // was resolved: the tenant a token was minted for is the tenant
1120                // it acts in, and letting a query parameter move it would be a
1121                // tenant-crossing primitive handed to whoever holds the
1122                // browser. Reintroducing it here — on a lookup that decides
1123                // where a browser gets redirected — would be the worst place to
1124                // reintroduce it.
1125                let redirect_target = match &q.redirect_uri {
1126                    Some(candidate) => state
1127                        .oauth2_client_repo
1128                        .get_by_client_id(user.tenant_id, &q.client_id)
1129                        .await
1130                        .ok()
1131                        .filter(|client| client.redirect_uris.contains(candidate))
1132                        .map(|_| candidate.clone()),
1133                    None => None,
1134                };
1135
1136                // A refused request object with no inline redirect_uri cannot
1137                // be reported by redirecting — there is no validated URI to
1138                // redirect to — so it is answered directly, with its own code
1139                // rather than the generic complaint about missing parameters.
1140                let error = match request_object {
1141                    Some(object) => object.into_error(),
1142                    None => OAuth2Error::InvalidRequest(
1143                        "response_type is required unless request_uri is used".into(),
1144                    ),
1145                };
1146
1147                return match redirect_target {
1148                    Some(uri) => {
1149                        build_error_redirect(&uri, &error, q.state.as_deref(), &state.auth_config)
1150                    }
1151                    None => authorize_error_response(&http_req, &error),
1152                };
1153            };
1154            AuthorizeRequest {
1155                tenant_id: user.tenant_id,
1156                user_id: user.user_id,
1157                response_type,
1158                client_id: q.client_id,
1159                redirect_uri,
1160                scope: q.scope,
1161                state: q.state,
1162                code_challenge: q.code_challenge,
1163                code_challenge_method: q.code_challenge_method,
1164                nonce: q.nonce,
1165                session_id: Some(user.session_id),
1166                via_par: false,
1167                id_token_hint: decode_authorize_id_token_hint(&state, &authn_params),
1168                // Not a pushed request: there is no second copy to conflict
1169                // with.
1170                inline_authn_params_beside_request_uri: false,
1171                login_hop_return_leg: axiam_oauth2::login_hop::is_return_leg(
1172                    q.login_hop.as_deref(),
1173                ),
1174                authn_params,
1175                request_object,
1176                session_evidence,
1177                consent_hop_return_leg: q.consent_hop.is_some(),
1178                // RFC 9449 §10, Figure 25 — the parameter on a plain
1179                // authorization request. Taken at face value: it is only a
1180                // *commitment*, and the client makes the commitment harder to
1181                // meet, never easier. A caller who pins a key they do not hold
1182                // has locked themselves out of their own code and nobody else.
1183                dpop_jkt: q.dpop_jkt,
1184                // OIDC Core §5.5, from the query string. Parsed rather than
1185                // trusted: `userinfo_claims` filters to the claims AXIAM will
1186                // release on a request alone, so a client naming
1187                // `phone_number` here gets what it would have got before —
1188                // nothing, unless the consent ceremony ran.
1189                requested_userinfo_claims: q
1190                    .claims
1191                    .as_deref()
1192                    .map(axiam_oauth2::claims_request::userinfo_claims)
1193                    .unwrap_or_default(),
1194                // Resolved below, once, for both carriers.
1195                sensitive_scopes: axiam_oauth2::sensitive::Requested::None,
1196                sensitive_scopes_switch_is_off: false,
1197            }
1198        }
1199    };
1200
1201    // W7 — resolved here rather than inside either arm above because both
1202    // arms produce the same three inputs (client, scope, subject) and a copy
1203    // of this in each is a copy that can drift. It is also the first point at
1204    // which the *resolved* scope is known: a pushed request's scope comes from
1205    // the pushed copy, never from the query string.
1206    let mut req = req;
1207    (req.sensitive_scopes, req.sensitive_scopes_switch_is_off) = resolve_sensitive_scopes(
1208        &state,
1209        user.tenant_id,
1210        user.user_id,
1211        &req.client_id,
1212        req.scope.as_deref(),
1213    )
1214    .await;
1215    let req = req;
1216
1217    // Captured before `req` moves: the error path needs the *resolved*
1218    // redirect_uri and state, which for a PAR request came from the pushed
1219    // parameters rather than the query string.
1220    let resolved_redirect_uri = req.redirect_uri.clone();
1221    let resolved_state = req.state.clone();
1222    let authorized_client_id = req.client_id.clone();
1223
1224    // W4 — was this a `prompt=none` request? Captured before `req` moves so
1225    // the audit rows below can name the outcome of a silent authorization
1226    // (plan §4.2, "Threats and what remains"): the one-bit login-status oracle
1227    // `prompt=none` inherently gives a registered relying party is accepted,
1228    // and made visible.
1229    let silent = req
1230        .authn_params
1231        .prompt
1232        .contains(&axiam_oauth2::authn_params::Prompt::None);
1233
1234    // W5 — the presentation the sign-in page will be asked for, if it is
1235    // shown. Captured before `req` moves, from the bundle whichever carrier
1236    // delivered it, and used **only** inside the interaction arm below.
1237    //
1238    // There is no honour-lane condition here, and W7 is why it now needs
1239    // saying rather than assuming. W4 produced `AuthorizeOutcome::Interact` at
1240    // exactly one place — step 6b, inside `if fapi::honours_authn_params` — so
1241    // a client on the `ignore` lane could not reach the arm that reads this.
1242    // W7 added a second producer, step 6c's consent gate, which is *not*
1243    // lane-conditional: the tenant switch and the consent record decide it,
1244    // not the client's `authn_request_params`.
1245    //
1246    // That is still invariant 4, and for a reason stronger than a lane check:
1247    // reaching step 6c at all requires the request to carry `address` or
1248    // `phone`, and requires the client to have them registered. No client in
1249    // any existing deployment does — the scopes were unregistrable before this
1250    // wave, which is the whole of I4 for G8. A client that has them was
1251    // registered after W7 shipped, by an operator who turned the tenant switch
1252    // on to do it.
1253    //
1254    // `build_interaction_redirect` then decides which page and which of these
1255    // values it may carry; `login_hint` reaches the consent page from neither
1256    // lane.
1257    //
1258    // Assembled here rather than inside `honour::evaluate` because the cosmetic
1259    // four decide no `Outcome` — see that module's "W5's cosmetic four are not
1260    // here".
1261    //
1262    // W7 supplies the tenant default W5 left as `None`, and reads it only when
1263    // the request carried a bundle: an empty bundle selects nothing whatever
1264    // the tenant prefers, so the read would be a database round trip on every
1265    // authorization request in the deployment to reach the same answer.
1266    let cosmetic_owned = {
1267        let tenant_default = if req.authn_params.is_empty() {
1268            None
1269        } else {
1270            tenant_default_locale(&state, user.tenant_id).await
1271        };
1272        let c = axiam_oauth2::login_hop::Cosmetic::from_params(&req.authn_params, tenant_default);
1273        (c.login_hint.map(str::to_owned), c.display, c.ui_locale)
1274    };
1275
1276    let outcome = match state.oauth2.authorize_service.authorize(req).await {
1277        Ok(axiam_oauth2::authorize::AuthorizeOutcome::Interact(interaction)) => {
1278            // W4 — the honour lane asked for an interaction. It rides W3's
1279            // login hop: same `return_to`, same validation on both sides, same
1280            // marker, so the chain is still bounded at one redirect.
1281            //
1282            // `reauth` is set for every reason the *honour lane* interacts —
1283            // `prompt=login`, a `max_age` this session cannot satisfy, a
1284            // step-up, an `id_token_hint` naming somebody else — because each
1285            // is a reason not to trust what the browser already holds; a
1286            // sign-in page that silently reused the current session would
1287            // return the same unsatisfying session and the loop guard would
1288            // have to catch it.
1289            //
1290            // W7 adds the one reason that is not: a consent question. The end
1291            // user is signed in, and what is missing is their answer, not
1292            // their password. `Reason::requires_reauthentication` is where
1293            // that distinction lives, so the two callers of this redirect
1294            // cannot come to disagree about it.
1295            tracing::debug!(
1296                client_id = %authorized_client_id,
1297                reason = ?interaction.reason,
1298                required_acr = ?interaction.required_acr,
1299                "an authorization request on the honour lane needs an interaction"
1300            );
1301            // W7 — a consent hop marks its own return leg. Same `return_to`
1302            // machinery, same validation on both sides, one more marker; see
1303            // `login_hop::CONSENT_HOP_MARKER` for why sharing the login one
1304            // would answer the consent question with the login hop's evidence.
1305            let built = if interaction.reason.requires_reauthentication() {
1306                axiam_oauth2::login_hop::build_return_to(http_req.query_string())
1307            } else {
1308                axiam_oauth2::login_hop::build_consent_return_to(http_req.query_string())
1309            };
1310            let Some(return_to) = built else {
1311                // Nothing safe to come back to, so there is nothing to send
1312                // the browser away for. The relying party is told what is
1313                // missing instead.
1314                return build_error_redirect(
1315                    &resolved_redirect_uri,
1316                    &OAuth2Error::LoginRequired(
1317                        "this authorization request needs the end user to authenticate, and \
1318                         could not be resumed after a sign-in"
1319                            .into(),
1320                    ),
1321                    resolved_state.as_deref(),
1322                    &state.auth_config,
1323                );
1324            };
1325            if !return_to_is_on_this_deployment(&state, &return_to) {
1326                return build_error_redirect(
1327                    &resolved_redirect_uri,
1328                    &OAuth2Error::LoginRequired(
1329                        "this authorization request needs the end user to authenticate, and \
1330                         could not be resumed after a sign-in"
1331                            .into(),
1332                    ),
1333                    resolved_state.as_deref(),
1334                    &state.auth_config,
1335                );
1336            }
1337            let (login_hint, display, ui_locale) = &cosmetic_owned;
1338            let cosmetic = axiam_oauth2::login_hop::Cosmetic {
1339                login_hint: login_hint.as_deref(),
1340                display: *display,
1341                ui_locale: *ui_locale,
1342            };
1343            let location = axiam_oauth2::login_hop::build_interaction_redirect(
1344                &return_to,
1345                interaction.reason,
1346                interaction.required_acr,
1347                &cosmetic,
1348            );
1349            return HttpResponse::Found()
1350                .append_header((actix_web::http::header::LOCATION, location))
1351                .append_header((actix_web::http::header::CACHE_CONTROL, "no-store"))
1352                .append_header(("Referrer-Policy", "no-referrer"))
1353                .finish();
1354        }
1355        other => other.map(|o| match o {
1356            axiam_oauth2::authorize::AuthorizeOutcome::Code(resp) => resp,
1357            axiam_oauth2::authorize::AuthorizeOutcome::Interact(_) => {
1358                unreachable!("the interaction arm returned above")
1359            }
1360        }),
1361    };
1362
1363    if silent {
1364        audit_prompt_none(
1365            &state,
1366            &http_req,
1367            user.tenant_id,
1368            &authorized_client_id,
1369            outcome.as_ref().err(),
1370        )
1371        .await;
1372    }
1373
1374    match outcome {
1375        Ok(resp) => {
1376            // B5: the client has just joined this session. Recorded here —
1377            // the moment a code is issued — because that is when
1378            // participation becomes true, and back-channel logout iterates
1379            // exactly these rows. A failure is logged and swallowed: losing a
1380            // logout notification is bad, but failing the login that would
1381            // have created the session is worse.
1382            if let Err(e) = state
1383                .oauth2
1384                .session_client_repo
1385                .record(axiam_core::models::oauth2_client::CreateSessionClient {
1386                    tenant_id: user.tenant_id,
1387                    session_id: user.session_id,
1388                    client_id: authorized_client_id.clone(),
1389                    user_id: user.user_id,
1390                })
1391                .await
1392            {
1393                tracing::warn!(
1394                    error = %e,
1395                    "could not record session participation; back-channel \
1396                     logout will not reach this client"
1397                );
1398            }
1399
1400            match url::Url::parse(&resp.redirect_uri) {
1401                Ok(mut url) => {
1402                    url.query_pairs_mut().append_pair("code", &resp.code);
1403                    if let Some(ref state) = resp.state {
1404                        url.query_pairs_mut().append_pair("state", state);
1405                    }
1406                    append_issuer(&mut url, &state.auth_config);
1407                    HttpResponse::Found()
1408                        .append_header(("Location", url.to_string()))
1409                        .finish()
1410                }
1411                Err(_) => {
1412                    // Never leak the authorization code to an
1413                    // unknown host.
1414                    HttpResponse::InternalServerError().json(OAuth2ErrorResponse {
1415                        error: "server_error".into(),
1416                        error_description: "invalid redirect_uri in \
1417                                 authorization response"
1418                            .into(),
1419                    })
1420                }
1421            }
1422        }
1423        Err(e) => {
1424            // Per RFC 6749: only redirect when the redirect_uri has
1425            // been validated. For InvalidClient / redirect_uri
1426            // errors, return a direct HTTP error instead.
1427            match &e {
1428                // B5: `ParRequired` joins these two because it is raised
1429                // BEFORE redirect_uri validation — redirecting would bounce
1430                // the user agent to an unvalidated URI that arrived by the
1431                // very channel the setting forbids.
1432                OAuth2Error::InvalidClient(_)
1433                | OAuth2Error::InvalidRedirectUri(_)
1434                | OAuth2Error::ParRequired(_) => authorize_error_response(&http_req, &e),
1435                _ => {
1436                    // These errors occur after client+redirect_uri
1437                    // were validated — safe to redirect.
1438                    build_error_redirect(
1439                        &resolved_redirect_uri,
1440                        &e,
1441                        resolved_state.as_deref(),
1442                        &state.auth_config,
1443                    )
1444                }
1445            }
1446        }
1447    }
1448}

1450/// `POST /oauth2/token` -- OAuth2 token endpoint.
1451///
1452/// Accepts form-encoded body per RFC 6749. Supports authorization_code,
1453/// client_credentials, and refresh_token grant types. The `tenant_id` is
1454/// passed as a query parameter since the token endpoint is unauthenticated
1455/// (the client is authenticating itself here).
1456#[utoipa::path(
1457    post,
1458    path = "/oauth2/token",
1459    tag = "oauth2",
1460    params(TenantQuery),
1461    request_body(
1462        content_type = "application/x-www-form-urlencoded",
1463        content = TokenRequest,
1464    ),
1465    responses(
1466        (status = 200, description = "Token response",
1467         body = TokenResponse),
1468        (status = 400, description = "OAuth2 error",
1469         body = OAuth2ErrorResponse),
1470    ),
1471)]
1472pub async fn token<C: Connection + Clone>(
1473    req: HttpRequest,
1474    tenant_query: web::Query<TenantQuery>,
1475    form: web::Form<TokenRequest>,
1476    state: web::Data<AppState<C>>,
1477) -> HttpResponse {
1478    // W8 / RFC 6749 §5.2 — the challenge is decided from the request,
1479    // before the handler runs, and applied to whatever 401 it produces.
1480    let challenge = client_auth_challenge(&req);
1481    with_client_auth_challenge(token_inner(req, tenant_query, form, state).await, challenge)
1482}

1484/// The token endpoint proper. Split from [`token`] only so that the RFC 6749
1485/// §5.2 challenge is applied to every one of its exits.
1486async fn token_inner<C: Connection + Clone>(
1487    req: HttpRequest,
1488    tenant_query: web::Query<TenantQuery>,
1489    form: web::Form<TokenRequest>,
1490    state: web::Data<AppState<C>>,
1491) -> HttpResponse {
1492    let tenant_id = tenant_query.into_inner().tenant_id;
1493
1494    let form = form.into_inner();
1495    // I5: keep the grant type for the stage-timing event below. The token
1496    // service moves `form`, so copy the (short) discriminator first.
1497    let grant_type = form.grant_type.clone();
1498    // §17.2 residual 3: same reason, for the failed-client-auth audit event
1499    // below. Never the secret — only the client id, which is the caller's own
1500    // input and is what an operator needs to correlate an attack.
1501    let attempted_client_id = form.client_id.clone();
1502    let started = std::time::Instant::now();
1503
1504    // B2 / RFC 8628 §3.4: the device-code grant is served by its own service.
1505    // Dispatching here, before `TokenService::exchange`, is the single `match`
1506    // arm the device flow costs the rest of the codebase — `TokenService`
1507    // never learns that this grant exists.
1508    //
1509    // Note what is deliberately NOT done: no client authentication. RFC 8628
1510    // targets public clients (a television cannot keep a secret), so the
1511    // device presents only its `device_code`, which is a 256-bit CSPRNG value
1512    // stored as a hash. The device code IS the credential.
1513    if grant_type == DEVICE_CODE_GRANT_TYPE {
1514        let device_code = match form.device_code.as_deref() {
1515            Some(code) if !code.is_empty() => code.to_owned(),
1516            _ => {
1517                return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
1518                    "device_code is required for the device_code grant".into(),
1519                ));
1520            }
1521        };
1522        return match state
1523            .oauth2
1524            .device_authorization_service
1525            .poll(tenant_id, &device_code)
1526            .await
1527        {
1528            Ok(resp) => HttpResponse::Ok()
1529                .append_header(("Cache-Control", "no-store"))
1530                .append_header(("Pragma", "no-cache"))
1531                .json(resp),
1532            // `authorization_pending` and `slow_down` are the NORMAL answers
1533            // here — a device polls for as long as the user takes to walk to
1534            // another screen — so they are not logged as failures. Treating
1535            // the expected path as an error condition would bury the real
1536            // ones under it.
1537            Err(e) => build_oauth2_error_response(&e),
1538        };
1539    }
1540
1541    // SEC-096: ONE context construction and ONE DPoP verification, ahead of
1542    // every grant that authenticates a client. Before this, the
1543    // token-exchange and uma-ticket grants each `return`ed above this point,
1544    // so a DPoP proof on those requests was never verified, never recorded as
1545    // single-use, and never available to the binding decision — which meant a
1546    // client holding a `cnf`-bound token could exchange it for a plain bearer
1547    // one, and a `fapi2` client could obtain an unconstrained token from a
1548    // grant `fapi::enforce_token_request` never saw.
1549    //
1550    // The device-code grant stays above this line deliberately: RFC 8628
1551    // performs no client authentication at all, so there is no registration
1552    // in hand for the profile gate to enforce and no party for a proof to
1553    // bind to.
1554    let ctx = match token_request_context(&req) {
1555        Ok(ctx) => ctx.with_assertion_from(&form),
1556        Err(response) => return *response,
1557    };
1558    let ctx = match dpop_from_request(&req, &state, tenant_id, ctx).await {
1559        Ok(ctx) => ctx,
1560        Err(response) => return *response,
1561    };
1562
1563    // B3 / RFC 8693. Like the device grant, its own service behind one match
1564    // arm. Unlike the device grant, the exchanging client DOES authenticate —
1565    // it is a confidential service, not a television, and an exchange is
1566    // precisely the operation that should be attributable.
1567    if grant_type == TOKEN_EXCHANGE_GRANT_TYPE {
1568        // SEC-093: the transport context travels with the grant, so a client
1569        // registered for `tls_client_auth` / `private_key_jwt` is
1570        // authenticated by the credential its registration names rather than
1571        // by the secret it was also, unavoidably, issued.
1572        return handle_token_exchange(tenant_id, form, &state, &ctx).await;
1573    }
1574
1575    // X2 / UMA 2.0 §3.3.1. Third grant dispatched here for the same structural
1576    // reason as the other two: the body is deserialized once, so the grant is
1577    // only knowable after `grant_type` has been read out of it.
1578    if grant_type == UMA_TICKET_GRANT_TYPE {
1579        // The authz checker is pulled from the request here rather than taken
1580        // as a handler parameter on purpose. `/oauth2/token` serves five
1581        // grants, and only this one evaluates anything against the RBAC
1582        // engine; making it an extractor would make the checker a hard
1583        // requirement of the whole endpoint, so a deployment (or a test
1584        // harness) that never uses UMA would start answering 500 to ordinary
1585        // token requests because of a dependency none of them use.
1586        let Some(authz) = req.app_data::<crate::authz::AuthzData>().cloned() else {
1587            return build_oauth2_error_response(&OAuth2Error::ServerError(
1588                "the uma-ticket grant requires an authorization checker".into(),
1589            ));
1590        };
1591        // SEC-093, as for the token-exchange grant above.
1592        return handle_uma_ticket(tenant_id, form, &state, &authz, &ctx).await;
1593    }
1594
1595    match state
1596        .oauth2
1597        .token_service
1598        .exchange(tenant_id, form, &ctx)
1599        .await
1600    {
1601        Ok(resp) => {
1602            let exchange_us = started.elapsed().as_micros() as u64;
1603
1604            // I5: serialize explicitly rather than via `.json()` so the
1605            // serialization cost and the exact response body size are
1606            // observable. Body size is the load-bearing number for the
1607            // Nagle/delayed-ACK hypothesis behind the TLS client-credentials
1608            // plateau: the pathology only fires when a response reaches the
1609            // socket as a full segment plus a small trailing fragment, so run 5
1610            // needs to know where this body sits relative to the path MSS
1611            // (1448 bytes on a standard 1500-byte-MTU docker veth) once TLS
1612            // record and HTTP/2 frame overhead are added.
1613            let t_serialize = std::time::Instant::now();
1614            let body = match serde_json::to_vec(&resp) {
1615                Ok(body) => body,
1616                Err(e) => {
1617                    tracing::error!(error = %e, "failed to serialize token response");
1618                    return HttpResponse::InternalServerError().finish();
1619                }
1620            };
1621            let serialize_us = t_serialize.elapsed().as_micros() as u64;
1622
1623            tracing::debug!(
1624                target: "axiam::perf",
1625                stage = "oauth2.token_endpoint",
1626                %grant_type,
1627                exchange_us,
1628                serialize_us,
1629                response_body_bytes = body.len(),
1630                handler_total_us = started.elapsed().as_micros() as u64,
1631                "token endpoint stage timings (I5)"
1632            );
1633
1634            HttpResponse::Ok()
1635                .append_header(("Cache-Control", "no-store"))
1636                .append_header(("Pragma", "no-cache"))
1637                .content_type("application/json")
1638                .body(body)
1639        }
1640        Err(e) => {
1641            if matches!(e, OAuth2Error::InvalidClient(_)) {
1642                // §22.3 residual 1: this used to be awaited, which put a tenant
1643                // read (and, when it succeeded, an insert) on the response path
1644                // — so a failed client auth returned measurably faster for a
1645                // NONEXISTENT tenant than a real one, on an unauthenticated
1646                // endpoint. Detaching it removes the differential at its source
1647                // rather than trying to balance the two branches, and makes the
1648                // "fire-and-forget" this function has always claimed actually
1649                // true. The response below no longer waits on the audit sink at
1650                // all.
1651                let tenant_repo = state.tenant_repo.clone();
1652                let audit_repo = state.audit_repo.clone();
1653                let attempted = attempted_client_id.clone();
1654                let grant = grant_type.clone();
1655                let peer = peer_ip(&req);
1656                let forwarded = client_ip(&req);
1657                actix_web::rt::spawn(async move {
1658                    append_client_auth_failure_audit(
1659                        &tenant_repo,
1660                        &audit_repo,
1661                        tenant_id,
1662                        attempted.as_deref(),
1663                        &grant,
1664                        peer,
1665                        forwarded,
1666                    )
1667                    .await;
1668                });
1669            }
1670            build_oauth2_error_response(&e)
1671        }
1672    }
1673}

1675/// Build the token endpoint's transport context from the connection (X5.1).
1676///
1677/// The certificate comes from [`VerifiedClientCert`], which `axiam-server`'s
1678/// `on_connect` hook populates from the peer chain **rustls verified during
1679/// the TLS 1.3 handshake**. Nothing else is consulted: in particular the
1680/// `X-Client-Certificate` header that `CertificateAuthenticated` accepts as a
1681/// proxy fallback is deliberately not read here, because an OAuth2 client
1682/// credential must not be assertable by anything that can set a header. See
1683/// `axiam_oauth2::mtls`'s module docs for the full argument.
1684///
1685/// Cost on the ordinary path — a connection with no client certificate, which
1686/// is every non-mTLS deployment — is one `conn_data` lookup that misses and no
1687/// allocation at all. On an mTLS connection it is one copy of the DER plus one
1688/// SHA-256 over it. The X.509 *parse* is deliberately not done here: it is
1689/// deferred to `PresentedCertificate::identity`, which only the
1690/// `tls_client_auth` branch calls, so a deployment running mTLS with ordinary
1691/// secret-authenticating clients does not pay for a DN nobody reads.
1692fn token_request_context(req: &HttpRequest) -> Result<TokenRequestContext, Box<HttpResponse>> {
1693    // W8 — RFC 6749 §2.3.1. The header is read here and decoded by
1694    // `axiam_oauth2::client_secret_basic`, so this crate owns "where the bytes
1695    // came from" and that crate owns "what they mean". A malformed header is
1696    // an error rather than an absence, for the same reason a bad DPoP proof is:
1697    // treating it as "no credentials presented" would make a corrupt Basic
1698    // header silently equivalent to sending none.
1699    let basic = basic_credentials_from_request(req)?;
1700
1701    // The trust level travels with the certificate rather than being inferred
1702    // here: `authenticate_mtls_client` refuses a self-asserted certificate for
1703    // `tls_client_auth` (§2.1) and accepts one for `self_signed_tls_client_auth`
1704    // (§2.2), and this is the seam that lets it tell them apart.
1705    let certificate = req
1706        .conn_data::<VerifiedClientCert>()
1707        .map(|verified| PresentedCertificate::from_der(&verified.der, verified.trust));
1708
1709    Ok(TokenRequestContext {
1710        client_certificate: certificate,
1711        ..TokenRequestContext::default()
1712    }
1713    .with_basic_credentials(basic))
1714}

1716/// Decode an `Authorization: Basic` header, if one is present (W8, RFC 6749
1717/// §2.3.1).
1718///
1719/// `Ok(None)` — no `Authorization` header, or one naming a different scheme.
1720/// Not an error: four of AXIAM's five client-authentication methods put
1721/// nothing in this header.
1722///
1723/// `Err(_)` — the client *did* attempt Basic authentication and got the
1724/// encoding wrong. Answered with `invalid_client`, uniform with every other
1725/// client-authentication failure (SEC-086): the four distinguishable causes in
1726/// [`BasicAuthError`] are a `debug!` for the operator and nothing at all for
1727/// the caller. RFC 6749 §5.2's `WWW-Authenticate: Basic` challenge is attached
1728/// by [`with_client_auth_challenge`] on the way out, so it is impossible for
1729/// one of these endpoints to answer a Basic attempt with a `Bearer` challenge.
1730///
1731/// A second `Authorization` header is refused rather than resolved. Taking the
1732/// first would let an intermediary that appends a header decide which
1733/// credential authenticates — the request-smuggling shape `dpop_from_request`
1734/// refuses for the same reason.
1735fn basic_credentials_from_request(
1736    req: &HttpRequest,
1737) -> Result<Option<BasicCredentials>, Box<HttpResponse>> {
1738    use actix_web::http::header::AUTHORIZATION;
1739
1740    let Some(raw) = req
1741        .headers()
1742        .get(AUTHORIZATION)
1743        .and_then(|v| v.to_str().ok())
1744    else {
1745        return Ok(None);
1746    };
1747
1748    match axiam_oauth2::client_secret_basic::parse_authorization_header(raw) {
1749        // Not a Basic attempt. Nothing is refused here, including a request
1750        // carrying several `Authorization` headers of some other scheme: the
1751        // duplicate-header rule below exists to stop an intermediary choosing
1752        // between two *credentials*, and this endpoint reads no credential
1753        // from any other scheme. Refusing more widely would change the
1754        // behaviour of requests no client sends today, which is I4's whole
1755        // point.
1756        None => Ok(None),
1757        Some(Ok(credentials)) => {
1758            // RFC 9110 §5.2 lets a recipient combine repeated field lines, and
1759            // `Authorization` is not a list-valued field — so two of them is a
1760            // malformed request, not a choice of credentials. Taking the first
1761            // would let an intermediary that appends a header decide which
1762            // credential authenticates, the request-smuggling shape
1763            // `dpop_from_request` refuses for the same reason.
1764            if req.headers().get_all(AUTHORIZATION).count() > 1 {
1765                tracing::debug!(
1766                    "more than one Authorization header on a request attempting Basic client \
1767                     authentication; refusing rather than choosing which credential counts"
1768                );
1769                return Err(Box::new(build_oauth2_error_response(
1770                    &OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()),
1771                )));
1772            }
1773            Ok(Some(credentials))
1774        }
1775        Some(Err(cause)) => {
1776            // The cause, never the header. `BasicAuthError` is four unit
1777            // variants and carries no part of the credential, which is why it
1778            // is safe to log at all.
1779            tracing::debug!(
1780                ?cause,
1781                "an Authorization header using the Basic scheme could not be decoded (RFC 6749 \
1782                 §2.3.1)"
1783            );
1784            Err(Box::new(build_oauth2_error_response(
1785                &OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()),
1786            )))
1787        }
1788    }
1789}

1791/// The `WWW-Authenticate` challenge this request must be answered with if it
1792/// ends in a 401 (RFC 6749 §5.2).
1793///
1794/// The RFC's rule is that the challenge names *the scheme the client used*, so
1795/// it is a property of the request and is computed before the handler runs —
1796/// which is also why it survives every early return inside one.
1797/// `None` means "the client did not authenticate through the `Authorization`
1798/// header", and the endpoint's existing `Bearer realm="axiam"` stands.
1799fn client_auth_challenge(req: &HttpRequest) -> Option<&'static str> {
1800    use actix_web::http::header::AUTHORIZATION;
1801    let raw = req.headers().get(AUTHORIZATION)?.to_str().ok()?;
1802    // Deliberately keyed on the *scheme*, not on whether the credentials
1803    // decoded: a client that sent a corrupt Basic header still used Basic, and
1804    // RFC 6749 §5.2 says the challenge must tell it so.
1805    axiam_oauth2::client_secret_basic::names_basic_scheme(raw)
1806        .then_some(axiam_oauth2::client_secret_basic::BASIC_CHALLENGE)
1807}

1809/// Replace the `WWW-Authenticate` challenge on a 401 with the scheme the
1810/// client actually used (RFC 6749 §5.2).
1811///
1812/// Applied at the four client-authenticating endpoints' single exit rather
1813/// than at each of the thirty-odd `build_oauth2_error_response` call sites
1814/// inside them. That placement is the point: a 401 raised by a path nobody
1815/// thought about while writing this still gets the right challenge.
1816///
1817/// Nothing else about the response is touched, and a non-401 is returned
1818/// unchanged — a challenge on a 200 or a 400 would be a protocol error of its
1819/// own.
1820fn with_client_auth_challenge(
1821    mut resp: HttpResponse,
1822    challenge: Option<&'static str>,
1823) -> HttpResponse {
1824    let Some(challenge) = challenge else {
1825        return resp;
1826    };
1827    if resp.status() != actix_web::http::StatusCode::UNAUTHORIZED {
1828        return resp;
1829    }
1830    resp.headers_mut().insert(
1831        actix_web::http::header::WWW_AUTHENTICATE,
1832        actix_web::http::header::HeaderValue::from_static(challenge),
1833    );
1834    resp
1835}

1837/// The `htu` a DPoP proof must name for this request (SEC-102, RFC 9449 §4.3
1838/// step 9).
1839///
1840/// Built from the **configured** issuer plus the request's path. RFC 9449 says
1841/// `htu` is compared against "the HTTP URI used for the request, without query
1842/// and fragment parts" — meaning the server's own view of its own endpoint.
1843/// `HttpRequest::full_url()` does not give that: it takes the authority from
1844/// `ConnectionInfo`, which prefers `Forwarded`, then `X-Forwarded-Host`, then
1845/// `Host`, and this deployment has no trusted-proxy layer that normalises any
1846/// of the three. With the authority sourced from the request, a caller chooses
1847/// *both* sides of the comparison and the check degrades to "the proof names
1848/// some authority consistently" — so a client phished into signing a proof for
1849/// `https://attacker.example/oauth2/token` could have it accepted here by
1850/// sending a matching `Host`.
1851///
1852/// The path still comes from the request, and must: it is what keeps a
1853/// `/oauth2/par` proof from being replayed at `/oauth2/token`. `req.path()` is
1854/// actix's already-normalised path and carries neither query nor fragment,
1855/// which is exactly the string the RFC compares.
1856///
1857/// `effective_issuer()` is trimmed of any trailing slash by `AuthConfig`, and
1858/// `req.path()` always begins with one, so the join never doubles it.
1859fn dpop_htu<C: Connection + Clone>(state: &AppState<C>, req: &HttpRequest) -> String {
1860    format!("{}{}", state.auth_config.effective_issuer(), req.path())
1861}

1869/// Fold a verified `DPoP` proof into the token-endpoint context (X5.1).
1870///
1871/// All of the verification is [`verify_dpop_header`]; this is the one line of
1872/// token-endpoint-specific work, kept separate so that the PAR endpoint —
1873/// which needs the proof's thumbprint but has no `TokenRequestContext` to put
1874/// it in — reaches the same verification rather than a second copy of it.
1875async fn dpop_from_request<C: Connection + Clone>(
1876    req: &HttpRequest,
1877    state: &AppState<C>,
1878    tenant_id: Uuid,
1879    mut ctx: TokenRequestContext,
1880) -> Result<TokenRequestContext, Box<HttpResponse>> {
1881    let Some(verified) = verify_dpop_header(req, state, tenant_id).await? else {
1882        return Ok(ctx);
1883    };
1884    ctx.dpop_proof = Some(verified);
1885    Ok(ctx)
1886}

1888/// Verify the `DPoP` header, if one is present (X5.1, RFC 9449 §4.3).
1889///
1890/// `Ok(None)` means no header arrived. `Err` means one arrived and did not
1891/// verify, and carries the response to send.
1892///
1893/// Two properties this function exists to hold, both of which are easy to lose
1894/// by writing the obvious thing instead:
1895///
1896/// 1. **A proof that fails verification is an error, not an absence.** Returning
1897///    `Ok(None)` for a *bad* proof would be silently equivalent to not sending
1898///    one — so a client that sent a forged proof would get whatever an unbound
1899///    client gets. The `Err` arm is what stops that.
1900/// 2. **An absent header is not an error here.** Whether this particular client
1901///    needed a proof is `fapi::enforce_token_request`'s question and
1902///    `certificate_binding_for`'s, both of which read the *registration*. Making
1903///    the decision here would require loading the client twice, and would put
1904///    the "does this client need a proof" rule in two places.
1905///
1906/// The `jti` is recorded on success, through the same `UNIQUE`-index guard the
1907/// client-assertion path uses. A proof whose `jti` cannot be recorded is
1908/// refused rather than accepted: failing open would turn a database blip into
1909/// an unlimited replay window.
1910///
1911/// Shared with `/oauth2/par` since RFC 9449 §10.1, which requires the PAR
1912/// endpoint to check a proof "as defined in Section 4.3" — the same check,
1913/// with `htu` naming the PAR path because [`dpop_htu`] reads it from the
1914/// request.
1915async fn verify_dpop_header<C: Connection + Clone>(
1916    req: &HttpRequest,
1917    state: &AppState<C>,
1918    tenant_id: Uuid,
1919) -> Result<Option<axiam_oauth2::dpop::VerifiedDpopProof>, Box<HttpResponse>> {
1920    use axiam_core::repository::{ProofKind, ProofReplayRepository};
1921    use axiam_oauth2::dpop::{self, DpopExpectation};
1922
1923    let Some(raw) = req
1924        .headers()
1925        .get(DPOP_HEADER)
1926        .and_then(|v| v.to_str().ok())
1927        .map(str::trim)
1928        .filter(|v| !v.is_empty())
1929    else {
1930        return Ok(None);
1931    };
1932
1933    // RFC 9449 §4.3 step 2: more than one DPoP header is a malformed request,
1934    // not a choice of proofs. Refusing rather than taking the first is what
1935    // stops a request-smuggling intermediary from deciding which proof counts.
1936    if req.headers().get_all(DPOP_HEADER).count() > 1 {
1937        return Err(dpop_error_response(
1938            "invalid_dpop_proof",
1939            "a request may carry at most one DPoP header",
1940            None,
1941        ));
1942    }
1943
1944    // SEC-102: the authority comes from the CONFIGURED issuer, never from the
1945    // request. `HttpRequest::full_url()` builds it from `ConnectionInfo`,
1946    // which prefers `Forwarded`, then `X-Forwarded-Host`, then `Host` — all
1947    // three request-controlled. Sourcing it there let the caller choose both
1948    // sides of RFC 9449 §4.3 step 9's comparison, reducing the `htu` check to
1949    // "the proof names *some* authority consistently". Only the path is taken
1950    // from the request, and only `req.path()`, which actix has already
1951    // normalised and which carries no query or fragment — exactly what the
1952    // RFC compares against.
1953    let htu = dpop_htu(state, req);
1954    let expect = DpopExpectation {
1955        htm: req.method().as_str(),
1956        htu: &htu,
1957        // SEC-097: `expected_nonce` stays `None` because this deployment
1958        // stores no per-client nonce to compare against — see
1959        // `dpop_nonce_required_for`, which is what decides whether a nonce is
1960        // demanded at all, and `docs/security-profiles.md` for what v1's
1961        // nonce challenge does and does not prove.
1962        expected_nonce: None,
1963        require_nonce: false,
1964        access_token: None,
1965        now: chrono::Utc::now().timestamp(),
1966        max_age_secs: dpop::DEFAULT_PROOF_MAX_AGE_SECS,
1967    };
1968
1969    let verified = match dpop::verify_dpop_proof(raw, &expect) {
1970        Ok(v) => v,
1971        Err(e) if e.is_nonce_challenge() => {
1972            return Err(dpop_error_response(
1973                e.error_code(),
1974                "a DPoP nonce is required",
1975                Some(dpop::new_nonce()),
1976            ));
1977        }
1978        Err(e) => {
1979            tracing::debug!(error = %e, "DPoP proof verification failed at the token endpoint");
1980            return Err(dpop_error_response(
1981                e.error_code(),
1982                "the DPoP proof could not be verified",
1983                None,
1984            ));
1985        }
1986    };
1987
1988    match state
1989        .oauth2
1990        .proof_replay_repo
1991        .insert_proof_jti(
1992            tenant_id,
1993            ProofKind::DpopProof,
1994            &verified.jkt,
1995            &verified.jti,
1996            verified.replay_expiry(dpop::DEFAULT_PROOF_MAX_AGE_SECS),
1997        )
1998        .await
1999    {
2000        Ok(()) => {}
2001        Err(axiam_core::error::AxiamError::ReplayDetected) => {
2002            tracing::warn!(jkt = %verified.jkt, "a DPoP proof was replayed; refusing");
2003            return Err(dpop_error_response(
2004                "invalid_dpop_proof",
2005                "this DPoP proof has already been used",
2006                None,
2007            ));
2008        }
2009        Err(e) => {
2010            tracing::error!(
2011                error = %e,
2012                "could not record a DPoP proof's jti; refusing rather than accepting a proof \
2013                 that cannot be made single-use"
2014            );
2015            return Err(dpop_error_response(
2016                "invalid_dpop_proof",
2017                "the DPoP proof could not be verified",
2018                None,
2019            ));
2020        }
2021    }
2022
2023    Ok(Some(verified))
2024}

2027///
2028/// Boxed because it is only ever the `Err` of `dpop_from_request`, whose `Ok`

2037        builder.append_header((DPOP_NONCE_HEADER, nonce));

2050/// Append an `oauth2.client_auth_failed` audit entry, fire-and-forget.
2051///
2052/// §17.2 residual 3: a failed client authentication at the token endpoint used
2053/// to accrue **nothing** — no counter, no lockout, no audit row, no metric —
2054/// and `/oauth2` is not covered by the audit middleware, so nothing else
2055/// filled the gap either. Contrast the interactive login path, which records
2056/// every failure and applies exponential backoff.
2057///
2058/// This closes the **detection** half only, which is the half that was
2059/// genuinely missing. Lockout is deliberately *not* added here: client
2060/// credentials are 256-bit CSPRNG values, so online guessing is not the threat
2061/// — and a lockout keyed on a caller-supplied `client_id` would hand any
2062/// unauthenticated party a denial-of-service against a known client. The
2063/// endpoint's existing rate limit remains the volumetric control.
2064///
2065/// Recorded with `actor_id = Uuid::nil()` and [`ActorType::System`] because
2066/// the caller is by definition unauthenticated: the presented `client_id`
2067/// either does not exist or did not prove possession of its secret, so it must
2068/// not be promoted into the actor field as though it had. It goes in the
2069/// metadata instead, where it is evidence rather than identity.
2070///
2071/// The secret is never recorded, in any form.
2072///
2073/// # SEC-087 — this endpoint is unauthenticated, so nothing here may be trusted
2074///
2075/// `/oauth2/token` takes its `tenant_id` from a **query parameter** and requires
2076/// no credential to reach. The first version of this function wrote the row
2077/// unconditionally, which let any anonymous caller append rows into an arbitrary
2078/// tenant's append-only log — or into a tenant id that never existed — at
2079/// request rate. Three separate controls apply here as a result:
2080///
2081/// 1. **The tenant is resolved before the write**, and a row that cannot be
2082///    attributed to a real tenant goes to the **system partition**
2083///    (`tenant_id = nil`, the one [`AuditLogRepository::list_system`] serves)
2084///    rather than to a caller-chosen id. This bounds what an anonymous caller
2085///    can steer: they may cause a row, but never choose which real tenant's
2086///    append-only log it lands in. It does *not* stop a caller from generating
2087///    failures against a real tenant they do not belong to — and deliberately
2088///    doesn't try to, because that event is **true**: a failed client
2089///    authentication did occur against that tenant, and suppressing it would
2090///    blind the operator to a real attack. The volumetric control is the
2091///    endpoint's rate limit, not the audit layer.
2092///
2093///    §22.3 residual 2: an earlier version *dropped* the row whenever the
2094///    tenant read failed for any reason other than `NotFound`. The reasoning
2095///    (a DB outage must not turn a 401 into a 500) was right, but the effect
2096///    was that a database fault cost exactly the telemetry this row exists to
2097///    provide — and a database fault is plausibly correlated with an incident.
2098///    Routing to the system partition keeps the signal in precisely the case
2099///    where losing it hurts most, while still refusing to guess a tenant.
2100/// 2. **The client id is truncated on a character boundary, by bytes.**
2101///    §22.3 residual 3: the cap was applied with `chars().take(N)`, so a
2102///    multibyte id still stored up to 4N bytes. It is caller-supplied and
2103///    otherwise unbounded, unlike the `client_ip`/`user_agent` helpers which
2104///    have always capped their inputs (and which share the same char-vs-byte
2105///    looseness, bounded there by much smaller limits).
2106/// 3. **The IP is recorded at two trust levels.** `ip_address` carries the
2107///    transport peer address, which a caller cannot forge. The
2108///    `X-Forwarded-For`-derived value goes to metadata under a name that says
2109///    it is untrusted: on an authenticated path behind a trusted proxy the
2110///    forwarded value is the useful one, but here anyone can assert it, and an
2111///    operator who blocks a forged IP has been made to act against the wrong
2112///    host.
2113async fn append_client_auth_failure_audit<C: Connection + Clone>(
2114    tenant_repo: &SurrealTenantRepository<C>,
2115    audit_repo: &SurrealAuditLogRepository<C>,
2116    tenant_id: Uuid,
2117    attempted_client_id: Option<&str>,
2118    grant_type: &str,
2119    peer_ip: Option<String>,
2120    forwarded_ip_untrusted: Option<String>,
2121) {
2122    // Control 1 — attribute the row to the request's tenant only when that
2123    // tenant demonstrably exists; otherwise fall back to the system partition.
2124    // `attributed_tenant` is nil in both the unknown and the indeterminate
2125    // case, so a caller can never place a row in a real tenant's log by naming
2126    // one.
2127    let (attributed_tenant, tenant_note) = match tenant_repo.get_by_id(tenant_id).await {
2128        Ok(_) => (tenant_id, None),
2129        Err(AxiamError::NotFound { .. }) => {
2130            tracing::warn!(
2131                claimed_tenant_id = %tenant_id,
2132                %grant_type,
2133                "oauth2: client-auth-failure audit routed to the system partition — unknown tenant (SEC-087)"
2134            );
2135            (Uuid::nil(), Some("unknown_tenant"))
2136        }
2137        Err(e) => {
2138            // A DB fault must not turn a 401 into a 500 (T-15-04), and must not
2139            // cost the telemetry either (§22.3 residual 2). Record it as
2140            // unattributed rather than dropping it.
2141            tracing::error!(
2142                error = %e,

2360    state: web::Data<AppState<C>>,
2361) -> HttpResponse {

2542/// address. It is used for **nothing else**: the response stays a function of
2543/// the token's subject and its scopes, which is what keeps `GET` and `POST`

2553        .0

2702        // Scope-only: `updated_at` describes when the profile last changed, so
2703        // releasing it to a client that was granted no profile claim would
2704        // answer a question it did not ask and could not interpret.
2705        updated_at: profile
2706            .as_ref()
2707            .filter(|_| has_scope("profile"))
2708            .map(|&(_, _, at)| at.timestamp()),
2709        phone_number: released.phone_number,
2710        phone_number_verified: released.phone_number_verified,
2711        address: released.address,
2712        tenant_id: user.tenant_id.to_string(),
2713        org_id: user.org_id.to_string(),
2714    })
2715}

2718///
2719/// A struct rather than a tuple so the two telephone members cannot be
2720/// separated: `phone_number_verified` is a statement *about* `phone_number`
2721/// and emitting it alone would assert something about a value the relying
2722/// party was not given.
2723#[derive(Default)]
2724struct ReleasedSensitiveClaims {
2725    phone_number: Option<String>,
2726    phone_number_verified: Option<bool>,
2727    address: Option<axiam_core::models::user::Address>,
2728}

2730impl ReleasedSensitiveClaims {
2731    /// The claim **names** released, for the audit row. Never the values.
2732    fn names(&self) -> Vec<&'static str> {
2733        let mut names = Vec::new();
2734        if self.phone_number.is_some() {
2735            names.push("phone_number");
2736        }
2737        if self.phone_number_verified.is_some() {
2738            names.push("phone_number_verified");
2739        }
2740        if self.address.is_some() {
2741            names.push("address");
2742        }
2743        names
2744    }

2747/// Decide, and record, what `address` and `phone` release for this call
2748/// (X7 G8, plan §4.8).
2749///
2750/// # Every gate is re-asked here, at the moment of release
2751///
2752/// The authorization endpoint asked all of them too, and none of its answers
2753/// is reused. That is the whole design: an access token lives fifteen minutes
2754/// and a refresh of it lives thirty days, so a decision taken when the token
2755/// was minted is a decision that outlives the facts it rested on. Asking again
2756/// is what makes withdrawal immediate (T8.4) rather than effective on the next
2757/// token.
2758///
2759/// In order, cheapest and most decisive first:
2760///
2761/// 1. **The token's scopes.** No `phone`, no telephone number. A scope the
2762///    relying party never asked for cannot be released by any later gate
2763///    saying yes.
2764/// 2. **The relying party.** Taken from the token's `client_id`
2765///    (RFC 9068 §2.2). A token that names none — every token issued before W7,
2766///    and every token minted by a login, a device flow or an exchange — cannot
2767///    have a consent record looked up for it, and releases nothing. Fail
2768///    closed, and the direction matters: the alternative is "any consent this
2769///    subject ever gave", which would hand a postal address to a client the
2770///    subject consented to a *different* client receiving.
2771/// 3. **The profile (M10).** A `fapi2` client is refused here regardless of
2772///    scope, switch or record — the third of the three places the FAPI gate
2773///    runs, and the only one where no authorization request is in hand.
2774/// 4. **The tenant switch.** The operator's decision outranks the subject's.
2775/// 5. **The consent record**, matched on client *and* on the exact scope set.
2776///
2777/// # A read that fails releases nothing
2778///
2779/// Every fallible step below resolves to "release nothing" and logs. A
2780/// UserInfo response missing an optional claim is what OIDC Core §5.3.2 calls
2781/// an omitted claim and what the conformance suite treats as a WARNING; a
2782/// UserInfo response carrying a postal address the subject withdrew is a
2783/// personal-data breach. Those are not comparable failures and the code does
2784/// not treat them as if they were.
2785async fn release_sensitive_claims<C: Connection + Clone>(
2786    user: &AuthenticatedUser,
2787    state: &AppState<C>,
2788    scopes: &[String],
2789    http_req: Option<&HttpRequest>,
2790) -> ReleasedSensitiveClaims {
2791    let none = ReleasedSensitiveClaims::default();
2792
2793    // 1. Scope.
2794    let wanted = axiam_oauth2::sensitive::requested(scopes);
2795    if wanted.is_empty() {
2796        return none;
2797    }
2798
2799    // 2. The relying party.
2800    let Some(client_id) = user.claims.0.client_id.as_deref() else {
2801        tracing::debug!(
2802            user_id = %user.user_id,
2803            "an access token carrying a sensitive scope names no client_id, so no consent \
2804             record can be found for it; releasing nothing"
2805        );
2806        return none;
2807    };
2808
2809    // 3. The profile — M10.
2810    match state
2811        .oauth2_client_repo
2812        .get_by_client_id(user.principal_tenant_id, client_id)
2813        .await
2814    {
2815        Ok(client) if client.profile.is_fapi2() => {
2816            tracing::warn!(
2817                %client_id,
2818                "a fapi2-issued access token carries a GDPR-sensitive scope; the fapi2 lane \
2819                 collects no consent record and nothing is released"
2820            );
2821            return none;
2822        }
2823        Ok(_) => {}
2824        Err(e) => {
2825            tracing::error!(
2826                error = %e,
2827                %client_id,
2828                "could not read the client behind this access token; releasing no sensitive \
2829                 claims rather than releasing them unprofiled"
2830            );
2831            return none;
2832        }
2833    }
2834
2835    // 4. The tenant switch.
2836    let enabled = match state.tenant_repo.get_by_id(user.principal_tenant_id).await {
2837        Ok(tenant) => axiam_core::repository::SettingsRepository::get_effective_settings(
2838            &state.settings_repo,
2839            tenant.organization_id,
2840            user.principal_tenant_id,
2841        )
2842        .await
2843        .map(|s| s.oidc.sensitive_scopes_enabled)
2844        .unwrap_or(false),
2845        Err(_) => false,
2846    };
2847    if !enabled {
2848        return none;
2849    }
2850
2851    // 5. The consent record, on this client and this exact scope set.
2852    let consent_type = axiam_oauth2::sensitive::consent_type(client_id);
2853    let version = axiam_oauth2::sensitive::consent_version(&wanted);
2854    let consented = axiam_core::repository::ConsentRepository::list_by_user(
2855        &state.gdpr.consent_repo,
2856        user.principal_tenant_id,
2857        user.user_id,
2858    )
2859    .await
2860    .map(|records| {
2861        records
2862            .iter()
2863            .any(|c| c.consent_type == consent_type && c.version == version)
2864    })
2865    .unwrap_or(false);
2866    if !consented {
2867        return none;
2868    }
2869
2870    // Only now is the row read. Reaching for the data before the gates had
2871    // finished would mean a telephone number in memory on a path that was
2872    // never going to release it.
2873    let Ok(subject) = state
2874        .user_repo
2875        .get_by_id(user.principal_tenant_id, user.user_id)
2876        .await
2877    else {
2878        tracing::error!(
2879            user_id = %user.user_id,
2880            "could not read the subject for its consented claims; releasing nothing"
2881        );
2882        return none;
2883    };
2884
2885    let mut released = ReleasedSensitiveClaims::default();
2886    if wanted.contains(&"phone") {
2887        // The verified flag rides with the number and never travels alone.
2888        if let Some(number) = subject.phone_number {
2889            released.phone_number_verified = Some(subject.phone_number_verified_at.is_some());
2890            released.phone_number = Some(number);
2891        }
2892    }
2893    if wanted.contains(&"address") {
2894        released.address = subject.address;
2895    }
2896
2897    let names = released.names();
2898    if !names.is_empty() {
2899        audit_sensitive_claims_released(state, http_req, user, client_id, &names).await;
2900    }
2901    released
2902}

2980/// `POST /oauth2/userinfo` — the same endpoint, the other method (W6, plan
2981/// §4.9 G10; OIDC Core §5.3, which says an OP MUST support both).
2982///
2983/// # Why a second handler rather than a second extractor
2984///
2985/// The access token may arrive in the `Authorization` header (RFC 6750 §2.1) or
2986/// — on POST only — in an `access_token` form field (§2.2). The second carrier
2987/// lives in the body, which [`AuthenticatedUser`]'s `FromRequest` impl does not
2988/// read and must not start reading: that impl is on the path `GET` takes, and
2989/// on the path every other authenticated route in the tree takes. Resolving the
2990/// token here instead leaves all of them untouched, which is what makes "GET
2991/// behaviour is unchanged" a property of the routing table rather than a claim
2992/// about a shared code path. Both arms then hand the same
2993/// [`AuthenticatedUser`] to the same [`userinfo_claims`].
2994///
2995/// # The refusals, and where each is decided
2996///
2997/// * **Two carriers, one request.** RFC 6750 §2: *"Clients MUST NOT use more
2998///   than one method to transmit the token in each request."* A form field
2999///   presented together with an `Authorization` header is refused here with
3000///   `400 invalid_request`, and the refusal names neither token — it does not
3001///   read them at all, so there is nothing to leak into the body or the log.
3002///   The `axiam_access` cookie is refused alongside the form field for the same
3003///   reason even though it is not one of RFC 6750's methods: it is a second
3004///   credential naming a possibly different subject, and the alternative to
3005///   refusing is choosing silently between two identities.
3006/// * **The cookie alone still authenticates, exactly as on GET.** That is
3007///   today's behaviour and invariant 4 keeps it. It is a CSRF surface in
3008///   principle — `/oauth2` is not wrapped in `CsrfMiddleware`, only `/api/v1`
3009///   is — and it fails closed in practice because `axiam_access` is
3010///   `SameSite=Strict` (`middleware::csrf::access_cookie`), so a cross-site
3011///   form POST carries no cookie and lands here unauthenticated. That is a
3012///   property of the cookie, not of this handler, so it is pinned by a test
3013///   rather than asserted by this comment
3014///   (`oauth2_userinfo_post_test.rs::the_access_cookie_is_strict_so_a_cross_site_post_cannot_be_authenticated_by_it`).
3015/// * **`access_token` in the query string is never read** — on either method.
3016///   RFC 6750 §2.3 deprecates that form because it puts a credential in a URL,
3017///   a `Referer` and every access log in the path. Refusing it would mean
3018///   reading it first, and the parameter's problem is that it exists at all,
3019///   not that a server honours it. So there is no code here that looks at the
3020///   query string, and a `GET /oauth2/userinfo?access_token=…` is simply an
3021///   unauthenticated request with an odd URL.
3022///
3023/// # An empty field is not a carrier
3024///
3025/// `access_token=` transmits no token, so it is treated as absent rather than
3026/// as a second method or as an invalid credential. A request carrying an empty
3027/// field and a real `Authorization` header is answered, not refused.
3028#[utoipa::path(
3029    post,
3030    path = "/oauth2/userinfo",
3031    tag = "oidc",
3032    request_body(
3033        content = UserInfoPostForm,
3034        content_type = "application/x-www-form-urlencoded",
3035        description = "RFC 6750 §2.2 form-encoded access token. Omit the body \
3036                       entirely to present the token in the Authorization header \
3037                       instead; presenting it both ways is refused."
3038    ),
3039    responses(
3040        (status = 200, description = "UserInfo response — byte-identical to what \
3041                                      GET answers for the same token",
3042         body = UserInfoResponse),
3043        (status = 400, description = "The access token was presented by more than \
3044                                      one method (RFC 6750 §2)"),
3045        (status = 401, description = "Invalid or missing access token"),
3046    ),
3047    security(("bearer" = []))
3048)]
3049pub async fn userinfo_post<C: Connection + Clone>(
3050    req: HttpRequest,
3051    form: Option<web::Form<UserInfoPostForm>>,
3052    maybe_user: MaybeAuthenticatedUser,
3053    state: web::Data<AppState<C>>,
3054) -> HttpResponse {
3055    use actix_web::ResponseError as _;
3056
3057    let body_token = form
3058        .as_ref()
3059        .and_then(|f| f.access_token.as_deref())
3060        .map(str::trim)
3061        .filter(|t| !t.is_empty());
3062
3063    let Some(body_token) = body_token else {
3064        // No §2.2 carrier: this is a header- or cookie-authenticated request,
3065        // and `MaybeAuthenticatedUser` holds precisely what actix would have
3066        // handed a `AuthenticatedUser` handler — the same principal, or the
3067        // same 401.
3068        return match maybe_user.into_result() {
3069            Ok(user) => userinfo_claims_for(&user, &state, Some(&req)).await,
3070            Err(e) => e.error_response(),
3071        };
3072    };
3073
3074    let header_present = req
3075        .headers()
3076        .contains_key(actix_web::http::header::AUTHORIZATION);

3080            error: "invalid_request".into(),

3084        });
3085    }
3086
3087    match crate::extractors::auth::authenticate_presented_token(&req, body_token).await {
3088        Ok(user) => userinfo_claims_for(&user, &state, Some(&req)).await,
3089        Err(e) => e.error_response(),
3090    }
3091}

3097/// Build a redirect response with error parameters per RFC 6749
3098/// section 4.1.2.1.
3099/// Append the RFC 9207 `iss` authorization-response parameter (X5.1).
3100///
3101/// # Why this matters, and why it is unconditional
3102///
3103/// RFC 9207 exists because of the **mix-up attack**: a client that talks to
3104/// more than one authorization server, and receives an authorization response
3105/// on a shared redirect URI, cannot otherwise tell which server sent it. An
3106/// attacker who controls one of those servers can therefore have a code minted
3107/// by an honest server delivered to the attacker's own token endpoint — or the
3108/// reverse. `iss` names the sender, so the client can check that the response
3109/// came from the server it started the flow with.
3110///
3111/// Emitted for **every** client, not just FAPI ones. RFC 9207 §2.4 says an AS
3112/// that supports the parameter SHOULD include it in every authorization
3113/// response; a client that does not understand it ignores an unknown query
3114/// parameter, which is the behaviour RFC 6749 §4.1.2 already requires of it.
3115/// Making it conditional would mean a client only gets mix-up protection when
3116/// somebody remembered to switch it on — and mix-up is precisely the attack a
3117/// client does not know it is under.
3118///
3119/// It goes on the **error** redirect too. RFC 9207 §2.4 is explicit about
3120/// this, and it is not a formality: one variant of the mix-up attack works by
3121/// injecting an error response, so a client that validates `iss` on success
3122/// and skips it on failure has left the door it just closed ajar.
3123fn append_issuer(url: &mut url::Url, auth_config: &AuthConfig) {
3124    let issuer = auth_config.effective_issuer();
3125    if issuer.is_empty() {
3126        // Nothing sensible to assert. Emitting an empty `iss` would be worse
3127        // than omitting it: a client comparing it against its configured
3128        // issuer would fail every flow, and one comparing loosely might pass
3129        // anything.
3130        tracing::warn!(
3131            "no issuer is configured, so the RFC 9207 `iss` authorization-response              parameter cannot be emitted; clients cannot detect a mix-up attack"
3132        );
3133        return;
3134    }
3135    url.query_pairs_mut().append_pair("iss", issuer);
3136}

3138fn build_error_redirect(
3139    redirect_uri: &str,
3140    error: &OAuth2Error,
3141    state: Option<&str>,
3142    auth_config: &AuthConfig,
3143) -> HttpResponse {
3144    let mut url = match url::Url::parse(redirect_uri) {
3145        Ok(u) => u,
3146        Err(_) => {
3147            // Unparseable redirect_uri — return direct error
3148            return HttpResponse::BadRequest().json(OAuth2ErrorResponse {
3149                error: error.error_code().to_string(),
3150                error_description: error.error_description(),
3151            });
3152        }
3153    };
3154    url.query_pairs_mut()
3155        .append_pair("error", error.error_code())
3156        .append_pair("error_description", &error.error_description());
3157    if let Some(state) = state {
3158        url.query_pairs_mut().append_pair("state", state);
3159    }
3160    // RFC 9207 §2.4 — see `append_issuer`. Error responses carry `iss` too,
3161    // because one form of the mix-up attack is an injected error response.
3162    append_issuer(&mut url, auth_config);
3163    HttpResponse::Found()
3164        .append_header(("Location", url.to_string()))
3165        .finish()
3166}

3223/// The `urn:ietf:params:oauth:grant-type:token-exchange` arm of the token
3224/// endpoint (B3).
3225///
3226/// The form is re-read rather than typed at the route because RFC 8693's
3227/// parameters are disjoint from RFC 6749's and actix deserializes one body
3228/// once — so `TokenRequest` carries the shared fields and the
3229/// exchange-specific ones are pulled from the raw form here.
3230/// X2 — `grant_type=urn:ietf:params:oauth:grant-type:uma-ticket`.
3231///
3232/// Redeems a permission ticket for an RPT. Four things are worth naming.
3233///
3234/// **The client authenticates, and the ticket is bound to it.** A ticket is
3235/// minted by a resource server through the Protection API and redeemed by that
3236/// same resource server; `consume` matches on `client_id`, so a ticket leaked
3237/// to another client is not a usable credential — and a wrong-client attempt
3238/// changes nothing rather than burning the rightful holder's ticket.
3239///
3240/// **`claim_token` is required, though UMA 2.0 calls it optional.** The spec's
3241/// other two ways to name a requesting party — an RPT presented for
3242/// incremental authorization, and interactive claims-gathering — are both
3243/// deferred to v2, so this is the only channel that exists. Requiring it and
3244/// saying so beats resolving to some default subject, which would mint an RPT
3245/// for a party nobody named.
3246///
3247/// **The subject token is validated, not trusted.** It is a signed AXIAM access
3248/// token; its remaining life bounds the RPT's, which is what stops an RPT from
3249/// outliving the authorization it rests on.
3250///
3251/// **Refusals are uniform.** Every ticket rejection — unknown, consumed,
3252/// expired, wrong client — answers `invalid_grant` with one message, because a
3253/// caller that could tell them apart could probe for live ticket handles.
3254async fn handle_uma_ticket<C: Connection + Clone>(
3255    tenant_id: Uuid,
3256    form: TokenRequest,
3257    state: &web::Data<AppState<C>>,
3258    authz: &crate::authz::AuthzData,
3259    ctx: &TokenRequestContext,
3260) -> HttpResponse {
3261    // SEC-093: only `client_id` is required up front. Whether the credential
3262    // that must accompany it is a secret, a certificate or a signed assertion
3263    // is a property of the REGISTRATION, so it cannot be decided before the
3264    // lookup — `authenticate_client` decides it, and answers the same
3265    // `invalid_client` for "no credential" as for "wrong credential".
3266    let Some(client_id) = form.client_id.as_deref() else {
3267        return build_oauth2_error_response(&OAuth2Error::InvalidClient(
3268            "the uma-ticket grant requires client authentication".into(),
3269        ));
3270    };
3271
3272    let client = match state
3273        .oauth2
3274        .token_service
3275        .authenticate_client(tenant_id, client_id, form.client_secret.as_deref(), ctx)
3276        .await
3277    {
3278        Ok(client) => client,
3279        Err(e) => return build_oauth2_error_response(&e),
3280    };
3281
3282    // Counted after authentication, keyed by the authenticated client — the
3283    // same reasoning as the token-exchange bucket: an unauthenticated caller
3284    // must not be able to consume a real client's allowance, and per-IP would
3285    // collapse a whole mesh behind one NAT into a single bucket.
3286    if !state.shared_rate_limit.check_at(
3287        &format!("oauth2_uma_ticket:{}:{}", tenant_id, client.client_id),
3288        chrono::Utc::now(),
3289        state.rate_limit_cfg.uma_ticket_per_min,
3290    ) {
3291        return HttpResponse::TooManyRequests()
3292            .append_header(("Cache-Control", "no-store"))
3293            .json(OAuth2ErrorResponse {
3294                error: "slow_down".into(),
3295                error_description: "uma-ticket rate limit exceeded for this client".into(),
3296            });
3297    }
3298
3299    // SEC-096: the same two gates the token-exchange grant now runs, for the
3300    // same reason — an RPT is an access token, and a `fapi2` client must not
3301    // be able to obtain an unconstrained one through the grant nobody wired
3302    // the profile gate into.
3303    if let Err(e) = axiam_oauth2::fapi::enforce_token_request(&client, ctx.evidence()) {
3304        return build_oauth2_error_response(&e);
3305    }
3306    let cnf = match state
3307        .oauth2
3308        .token_service
3309        .certificate_binding_for(&client, ctx)
3310    {
3311        Ok(cnf) => cnf,
3312        Err(e) => return build_oauth2_error_response(&e),
3313    };
3314
3315    let Some(ticket) = form.ticket.as_deref().filter(|t| !t.is_empty()) else {
3316        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
3317            "ticket is required for the uma-ticket grant".into(),
3318        ));
3319    };
3320
3321    let Some(claim_token) = form.claim_token.as_deref().filter(|t| !t.is_empty()) else {
3322        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
3323            "claim_token is required for the uma-ticket grant: it names the \
3324             requesting party, and v1 implements neither incremental \
3325             authorization nor claims-gathering"
3326                .into(),
3327        ));
3328    };
3329
3330    // An unrecognised format is refused rather than assumed. Guessing would
3331    // mean validating a foreign token with the wrong verifier.
3332    if let Some(format) = form.claim_token_format.as_deref().filter(|f| !f.is_empty())
3333        && format != UMA_CLAIM_TOKEN_FORMAT
3334    {
3335        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(format!(
3336            "unsupported claim_token_format '{format}'; v1 accepts only \
3337             '{UMA_CLAIM_TOKEN_FORMAT}'"
3338        )));
3339    }
3340
3341    let claims = match axiam_auth::token::validate_access_token(claim_token, &state.auth_config) {
3342        Ok(validated) => validated.0,
3343        Err(_) => {
3344            return build_oauth2_error_response(&OAuth2Error::InvalidGrant(
3345                "claim_token is not a valid access token".into(),
3346            ));
3347        }
3348    };
3349
3350    let Ok(subject_id) = claims.sub.parse::<Uuid>() else {
3351        return build_oauth2_error_response(&OAuth2Error::InvalidGrant(
3352            "claim_token does not name a user as its subject".into(),
3353        ));
3354    };
3355
3356    // The requesting party must belong to the tenant the ticket was minted in.
3357    // Without this a token from tenant A could redeem a ticket in tenant B and
3358    // be evaluated against B's grants under A's subject id.
3359    if claims.tenant_id != tenant_id.to_string() {
3360        return build_oauth2_error_response(&OAuth2Error::InvalidGrant(
3361            "claim_token belongs to a different tenant".into(),
3362        ));
3363    }
3364
3365    let subject_remaining_secs = claims.exp - chrono::Utc::now().timestamp();
3366
3367    let granted = match state
3368        .uma_service(authz.get_ref().clone())
3369        .exchange_ticket(
3370            tenant_id,
3371            &client.client_id,
3372            ticket,
3373            subject_id,
3374            subject_remaining_secs,
3375        )
3376        .await
3377    {
3378        Ok(granted) => granted,
3379        Err(e) => {
3380            tracing::info!(
3381                target: "axiam::audit",
3382                event = "uma.rpt_refused",
3383                tenant_id = %tenant_id,
3384                client_id = %client.client_id,
3385                subject = %subject_id,
3386                error = e.wire_error(),
3387                "uma-ticket grant refused"
3388            );
3389            return uma_refusal_response(e);
3390        }
3391    };
3392
3393    let org_id = match claims.org_id.parse::<Uuid>() {
3394        Ok(id) => id,
3395        Err(_) => {
3396            return build_oauth2_error_response(&OAuth2Error::ServerError(
3397                "claim_token carries an unparseable org_id".into(),
3398            ));
3399        }
3400    };
3401
3402    let rpt = match axiam_auth::token::issue_rpt(
3403        subject_id,
3404        tenant_id,
3405        org_id,
3406        granted.permissions.clone(),
3407        granted.lifetime_secs,
3408        &state.auth_config,
3409        cnf.clone(),
3410    ) {
3411        Ok(token) => token,
3412        Err(e) => return build_oauth2_error_response(&OAuth2Error::ServerError(e.to_string())),
3413    };
3414
3415    // Audited before the token is returned: the ticket is already consumed at
3416    // this point, so a crash between issuing and logging would leave a spent
3417    // ticket with no record of what it bought.
3418    tracing::info!(
3419        target: "axiam::audit",
3420        event = "uma.rpt_issued",
3421        tenant_id = %tenant_id,
3422        client_id = %client.client_id,
3423        subject = %subject_id,
3424        pairs = granted.permissions.len(),
3425        lifetime_secs = granted.lifetime_secs,
3426        "RPT issued"
3427    );
3428
3429    HttpResponse::Ok()
3430        .append_header(("Cache-Control", "no-store"))
3431        .append_header(("Pragma", "no-cache"))
3432        .json(TokenResponse {
3433            access_token: rpt,
3434            // SEC-096 / RFC 9449 §5: `DPoP` when the RPT is DPoP-bound,
3435            // `Bearer` for everything else — which is every client that did
3436            // not register DPoP binding.
3437            token_type: axiam_oauth2::dpop::token_type_for(cnf.as_ref()),
3438            expires_in: granted.lifetime_secs as u64,
3439            refresh_token: None,
3440            scope: None,
3441            id_token: None,
3442        })
3443}

3445/// Render a [`UmaError`] as the token endpoint's error response.
3446///
3447/// # Why `access_denied` does not go through `build_oauth2_error_response`
3448///
3449/// That helper maps everything except `invalid_client` and `server_error` to
3450/// **400**, which is right for RFC 6749 and right for RFC 8628 — the device
3451/// grant's `access_denied` (the user said no) is a 400 and must stay one.
3452///
3453/// UMA 2.0 §3.3.6 specifies **403** for a request the authorization server
3454/// refuses on the merits, and a UMA-conforming resource server distinguishes
3455/// "you may not have this" from "your request was malformed" by exactly that
3456/// status. Routing the UMA refusal through the shared helper would answer 400
3457/// and quietly fail conformance, so this one status is built here. Every other
3458/// UMA error keeps the shared mapping.
3459fn uma_refusal_response(e: UmaError) -> HttpResponse {
3460    if matches!(e, UmaError::AccessDenied) {
3461        return HttpResponse::Forbidden()
3462            .append_header(("Cache-Control", "no-store"))
3463            .append_header(("Pragma", "no-cache"))
3464            .json(OAuth2ErrorResponse {
3465                // v1 sends no `need_info` alongside it: that field exists to
3466                // steer a client into claims-gathering, which is deferred, so
3467                // sending it would point at a flow that does not run.
3468                error: "access_denied".into(),
3469                error_description:
3470                    "the requesting party is not authorized for every requested permission".into(),
3471            });
3472    }
3473    build_oauth2_error_response(&uma_error_to_oauth2(e))
3474}
3475
3476/// Map a [`UmaError`] onto the OAuth2 error shape the token endpoint answers.
3477fn uma_error_to_oauth2(e: UmaError) -> OAuth2Error {
3478    match e {
3479        UmaError::InvalidRequest(m) => OAuth2Error::InvalidRequest(m),
3480        UmaError::InvalidGrant(m) => OAuth2Error::InvalidGrant(m),
3481        UmaError::AccessDenied => OAuth2Error::AccessDenied(
3482            "the requesting party is not authorized for every requested permission".into(),
3483        ),
3484        UmaError::ExpiredSubjectToken => {
3485            OAuth2Error::InvalidGrant("the presented claim_token has expired".into())
3486        }
3487        UmaError::Server(m) => OAuth2Error::ServerError(m),
3488    }
3489}

3491async fn handle_token_exchange<C: Connection + Clone>(
3492    tenant_id: Uuid,
3493    form: TokenRequest,
3494    state: &web::Data<AppState<C>>,
3495    ctx: &TokenRequestContext,
3496) -> HttpResponse {
3497    // SEC-093 — see `handle_uma_ticket` for why only `client_id` is required
3498    // before the lookup.
3499    let Some(client_id) = form.client_id.as_deref() else {
3500        return build_oauth2_error_response(&OAuth2Error::InvalidClient(
3501            "the token-exchange grant requires client authentication".into(),
3502        ));
3503    };
3504
3505    let client = match state
3506        .oauth2
3507        .token_service
3508        .authenticate_client(tenant_id, client_id, form.client_secret.as_deref(), ctx)
3509        .await
3510    {
3511        Ok(client) => client,
3512        Err(e) => return build_oauth2_error_response(&e),
3513    };
3514
3515    // B3: the exchange's own bucket, applied here rather than as route
3516    // middleware because `/oauth2/token` serves every grant and actix
3517    // deserializes the body once — the grant is not knowable until the form
3518    // has been read, which is after the middleware chain has run.
3519    //
3520    // Keyed by the authenticated client, deliberately: an exchange always
3521    // carries client credentials, so unlike the device endpoints there is a
3522    // real identity to key on, and per-IP would collapse a whole mesh behind
3523    // one NAT into a single bucket. Counted AFTER authentication so an
3524    // unauthenticated caller cannot consume a real client's allowance.
3525    //
3526    // `check_at` rather than `check`: it derives the window itself so it can
3527    // also see the OFFSET into it, which is the sliding-window bound the J1
3528    // fix added after run 5 measured boundary over-admission. Re-deriving a
3529    // truncated window here would quietly opt this endpoint out of that fix.
3530    if !state.shared_rate_limit.check_at(
3531        &format!("oauth2_token_exchange:{}:{}", tenant_id, client.client_id),
3532        chrono::Utc::now(),
3533        state.rate_limit_cfg.token_exchange_per_min,
3534    ) {
3535        return HttpResponse::TooManyRequests()
3536            .append_header(("Cache-Control", "no-store"))
3537            .json(OAuth2ErrorResponse {
3538                error: "slow_down".into(),
3539                error_description: "token-exchange rate limit exceeded for this client".into(),
3540            });
3541    }
3542
3543    // SEC-096, gate 1: the profile the REGISTRATION declares, re-checked at
3544    // the moment it matters — the same call the authorization_code,
3545    // client_credentials and refresh_token grants make. Without it a `fapi2`
3546    // client could obtain an unconstrained token through this grant while
3547    // being refused one through the other three, which is a conformance
3548    // failure and an authentication downgrade in the same request.
3549    if let Err(e) = axiam_oauth2::fapi::enforce_token_request(&client, ctx.evidence()) {
3550        return build_oauth2_error_response(&e);
3551    }
3552
3553    // SEC-096, gate 2: the confirmation claim the EXCHANGING client earned on
3554    // this request. `None` for every client that registered no binding, so
3555    // this is byte-identical to the pre-SEC-096 response for them; for a
3556    // client that registered binding and presented nothing it refuses rather
3557    // than laundering a bound token into an unbound one.
3558    let cnf = match state
3559        .oauth2
3560        .token_service
3561        .certificate_binding_for(&client, ctx)
3562    {
3563        Ok(cnf) => cnf,
3564        Err(e) => return build_oauth2_error_response(&e),
3565    };
3566
3567    let Some(exchange_req) = form.exchange_request() else {
3568        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
3569            "subject_token and subject_token_type are required for the \
3570             token-exchange grant"
3571                .into(),
3572        ));
3573    };
3574
3575    match state
3576        .oauth2
3577        .token_exchange_service
3578        .exchange(tenant_id, &client, exchange_req, cnf)
3579        .await
3580    {
3581        Ok(outcome) => {
3582            // Audited BEFORE the token is returned, not after. For an
3583            // impersonation this record is the only surviving evidence that
3584            // the acting party was not the subject — the token deliberately
3585            // carries no `act` claim — so it must not be lost to a crash
3586            // between issuing and logging.
3587            // X4: an external exchange's provenance is recorded in the same
3588            // entry, not a second one. A reader correlating two lines by
3589            // timestamp is a reader who can be made to correlate the wrong
3590            // two, and for a JIT provision this is the only record that a
3591            // partner's IdP created an AXIAM user.
3592            let ext = outcome.external.as_ref();
3593            tracing::info!(
3594                target: "axiam::audit",
3595                event = "oauth2.token_exchange",
3596                tenant_id = %tenant_id,
3597                client_id = %client.client_id,
3598                kind = outcome.kind.as_str(),
3599                subject = %outcome.subject,
3600                actor = outcome.actor.as_deref().unwrap_or("-"),
3601                granted_scopes = %outcome.granted_scopes.join(" "),
3602                audience = %outcome.audience,
3603                external_issuer = ext.map(|e| e.issuer.as_str()).unwrap_or("-"),
3604                external_subject = ext.map(|e| e.external_subject.as_str()).unwrap_or("-"),
3605                federation_provider = ext.map(|e| e.provider_name.as_str()).unwrap_or("-"),
3606                federation_config_id = ext
3607                    .map(|e| e.provider_id.to_string())
3608                    .unwrap_or_else(|| "-".into()),
3609                jit_provisioned = ext.is_some_and(|e| e.newly_provisioned),
3610                "token exchange"
3611            );
3612            HttpResponse::Ok()
3613                .append_header(("Cache-Control", "no-store"))
3614                .append_header(("Pragma", "no-cache"))
3615                .json(outcome.response)
3616        }
3617        Err(e) => {
3618            tracing::info!(
3619                target: "axiam::audit",
3620                event = "oauth2.token_exchange",
3621                tenant_id = %tenant_id,
3622                client_id = %client.client_id,
3623                outcome = "refused",
3624                error = e.error_code(),
3625                "token exchange refused"
3626            );
3627            build_oauth2_error_response(&e)
3628        }
3629    }
3630}

3632/// `invalid_client` returns 401 with a `WWW-Authenticate` header per
3633/// RFC 6749 §5.2.  Although the token endpoint uses `client_secret_post`,
3634/// RFC 6749 §5.2 still requires the 401 response to include
3635/// `WWW-Authenticate` indicating the authentication scheme.
3636/// Whether this request came from something that will *render* the answer.
3637///
3638/// Only an explicit `text/html` in `Accept` counts. A missing header, `*/*`
3639/// (curl's default) or an API client's `application/json` all keep the JSON
3640/// body byte-for-byte, so nothing that integrates against AXIAM today can
3641/// observe this.
3642fn prefers_html(req: &HttpRequest) -> bool {
3643    req.headers()
3644        .get(actix_web::http::header::ACCEPT)
3645        .and_then(|v| v.to_str().ok())
3646        .is_some_and(|accept| {
3647            accept
3648                .split(',')
3649                .any(|part| part.trim().starts_with("text/html"))
3650        })
3651}
3652
3653/// The five characters that turn text into markup.
3654///
3655/// Written out rather than pulled from a crate because the set is closed and
3656/// the reason it exists should be visible at the point of use: an
3657/// `error_description` is not always AXIAM's own prose — `par_form_error`
3658/// forwards actix's message, which quotes a field name the caller chose.
3659fn escape_html(raw: &str) -> String {
3660    let mut out = String::with_capacity(raw.len());
3661    for c in raw.chars() {
3662        match c {
3663            '&' => out.push_str("&amp;"),
3664            '<' => out.push_str("&lt;"),
3665            '>' => out.push_str("&gt;"),
3666            '"' => out.push_str("&quot;"),
3667            '\'' => out.push_str("&#39;"),
3668            _ => out.push(c),
3669        }
3670    }
3671    out
3672}

3674/// An authorization-endpoint refusal, rendered for whoever asked.
3675///
3676/// # Why the authorization endpoint gets its own
3677///
3678/// `/oauth2/authorize` is the one OAuth2 endpoint a **person** arrives at.
3679/// When a refusal cannot be redirected to the client — RFC 6749 §4.1.2.1 for a
3680/// bad `redirect_uri` or `client_id`, and every `request_uri` failure, where
3681/// the redirect target is precisely what could not be resolved — the
3682/// specification says the server "SHOULD inform the resource owner of the
3683/// error". A `Content-Type: application/json` body informs a developer reading
3684/// a browser window; it does not inform a resource owner.
3685///
3686/// It is also what a certification reviewer is looking at. Five OIDF modules
3687/// end in REVIEW with instructions of the form "it must show an error page
3688/// saying the request_uri is invalid - upload a screenshot of the error page",
3689/// and the screenshot they were being handed was a raw JSON object.
3690///
3691/// # What it does not do
3692///
3693/// It does not change a single byte for a non-browser caller, and it does not
3694/// invent content: the page renders the same `error` and `error_description`
3695/// the JSON carries, escaped, and nothing else. In particular it echoes no
3696/// `state`, no `redirect_uri` and no `request_uri` — the rule
3697/// [`logged_out_page`] states and for the same reason, since those are
3698/// attacker-supplied strings and this page is served from AXIAM's own origin.
3699fn authorize_error_response(req: &HttpRequest, e: &OAuth2Error) -> HttpResponse {
3700    if !prefers_html(req) {
3701        return build_oauth2_error_response(e);
3702    }
3703    let status = match e {
3704        OAuth2Error::InvalidClient(_) => actix_web::http::StatusCode::UNAUTHORIZED,
3705        OAuth2Error::ServerError(_) => actix_web::http::StatusCode::INTERNAL_SERVER_ERROR,
3706        _ => actix_web::http::StatusCode::BAD_REQUEST,
3707    };
3708    let code = escape_html(e.error_code());
3709    let description = escape_html(&e.error_description());
3710    HttpResponse::build(status)
3711        .append_header(("Cache-Control", "no-store"))
3712        .append_header(("Pragma", "no-cache"))
3713        .content_type("text/html; charset=utf-8")
3714        .body(format!(
3715            "<!doctype html><html lang=\"en\"><head><meta charset=\"utf-8\">\
3716             <meta name=\"viewport\" content=\"width=device-width,initial-scale=1\">\
3717             <title>This authorization request cannot be completed</title></head>\
3718             <body><h1>This authorization request cannot be completed</h1>\
3719             <p>{description}</p><p><small>Error code: <code>{code}</code></small></p>\
3720             <p>Nothing has been shared with the application that sent you here. \
3721             Return to it and start again.</p></body></html>"
3722        ))
3723}

3725fn build_oauth2_error_response(e: &OAuth2Error) -> HttpResponse {
3726    let status = match e {
3727        OAuth2Error::InvalidClient(_) => actix_web::http::StatusCode::UNAUTHORIZED,
3728        OAuth2Error::ServerError(_) => actix_web::http::StatusCode::INTERNAL_SERVER_ERROR,
3729        _ => actix_web::http::StatusCode::BAD_REQUEST,
3730    };
3731    let mut builder = HttpResponse::build(status);
3732    builder
3733        .append_header(("Cache-Control", "no-store"))
3734        .append_header(("Pragma", "no-cache"));
3735    // RFC 6749 §5.2: 401 responses MUST include WWW-Authenticate
3736    if status == actix_web::http::StatusCode::UNAUTHORIZED {
3737        builder.append_header(("WWW-Authenticate", "Bearer realm=\"axiam\""));
3738    }
3739    builder.json(OAuth2ErrorResponse {
3740        error: e.error_code().to_string(),
3741        error_description: e.error_description(),
3742    })
3743}

3917    /// Optional since SEC-093: a client registered for `tls_client_auth` or
3918    /// `private_key_jwt` has no secret to present here, and presenting one no
3919    /// longer authenticates it.
3920    pub client_secret: Option<String>,
3921    /// RFC 7521 §4.2 — `private_key_jwt` client authentication at PAR.
3922    pub client_assertion: Option<String>,
3923    pub client_assertion_type: Option<String>,
3924    pub response_type: String,
3925    pub redirect_uri: String,
3926    pub scope: Option<String>,
3927    pub state: Option<String>,
3928    pub code_challenge: Option<String>,
3929    pub code_challenge_method: Option<String>,
3930    pub nonce: Option<String>,

3932    // carrier a `require_par` client has.

3934    pub max_age: Option<String>,
3935    pub acr_values: Option<String>,
3936    pub claims: Option<String>,
3937    pub id_token_hint: Option<String>,
3938    pub login_hint: Option<String>,
3939    pub display: Option<String>,
3940    pub ui_locales: Option<String>,
3941    pub claims_locales: Option<String>,

3943    ///
3944    /// Modelled so that it can be refused. Leaving it off the struct made
3945    /// `serde` drop it silently and the endpoint answer `201` to a request the
3946    /// specification says must be rejected:
3947    ///

3949    /// > and it MUST NOT be provided.
3950    ///
3951    /// Refusing matters beyond conformance. A `request_uri` accepted here would
3952    /// be a pushed request that names another pushed request, and the second

3954    /// authenticated origin is a different request from the one finally
3955    /// presented at the authorization endpoint.

3958    /// possession of at the token endpoint.
3959    ///
3960    /// One of the two carriers §10.1 requires an AS supporting both PAR and
3961    /// DPoP to accept; the other is a `DPoP` header on this same request. When
3962    /// both arrive they must agree, and the handler refuses the request when
3963    /// they do not.

4053/// See [`token_inner`] for why this split exists.
4054async fn pushed_authorization_request_inner<C: Connection + Clone>(
4055    http_req: HttpRequest,
4056    tenant_query: web::Query<TenantQuery>,
4057    form: web::Form<PushedAuthorizationRequest>,
4058    state: web::Data<AppState<C>>,
4059) -> HttpResponse {
4060    let tenant_id = tenant_query.into_inner().tenant_id;
4061    let req = form.into_inner();
4062
4063    // RFC 9126 §2.1-2, answered before anything else is looked at.
4064    //
4065    // Ahead of client authentication deliberately, and it costs nothing: the
4066    // refusal names a parameter the caller sent rather than anything about the
4067    // client, so it is not an oracle, and a request that cannot be honoured
4068    // whoever sent it should not first consume an authentication.
4069    if req.request_uri.is_some() {
4070        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
4071            "request_uri must not be provided to the pushed authorization request endpoint \
4072             (RFC 9126 §2.1)"
4073                .into(),
4074        ));
4075    }
4076
4077    // One client-authentication path in the codebase, shared with the token
4078    // endpoint, rather than a second one to keep correct — and since SEC-093
4079    // that path honours the registered `token_endpoint_auth_method`. PAR is
4080    // the sharpest instance of the old bug: FAPI 2.0 §5.3.1.1 requires strong
4081    // client authentication AND requires PAR, so a FAPI deployment whose PAR
4082    // endpoint accepted a shared secret was both an authentication downgrade
4083    // and a conformance failure.
4084    let ctx = match token_request_context(&http_req) {
4085        Ok(ctx) => ctx.with_assertion(
4086            req.client_assertion.as_deref(),
4087            req.client_assertion_type.as_deref(),
4088        ),
4089        Err(response) => return *response,
4090    };
4091    let client = match state
4092        .oauth2
4093        .token_service
4094        .authenticate_client(
4095            tenant_id,
4096            &req.client_id,
4097            req.client_secret.as_deref(),
4098            &ctx,
4099        )
4100        .await
4101    {
4102        Ok(client) => client,
4103        Err(e) => return build_oauth2_error_response(&e),
4104    };
4105    // Cloned because `client.client_id` is moved into the pushed request
4106    // below, and the §10.1 refusal between here and there wants to name the
4107    // client it is refusing.
4108    let client_id_for_log = client.client_id.clone();
4109
4110    // Keyed by the authenticated client and counted AFTER authentication, for
4111    // the same reasons as the token exchange: PAR always carries credentials
4112    // so there is a real identity to key on, per-IP would collapse a whole
4113    // deployment behind one NAT into a single bucket, and counting before
4114    // authentication would let an unauthenticated caller burn a real client's
4115    // allowance.
4116    if !state.shared_rate_limit.check_at(
4117        &format!("oauth2_par:{}:{}", tenant_id, client.client_id),
4118        chrono::Utc::now(),
4119        state.rate_limit_cfg.par_per_min,
4120    ) {
4121        return HttpResponse::TooManyRequests()
4122            .append_header(("Cache-Control", "no-store"))
4123            .json(OAuth2ErrorResponse {
4124                error: "slow_down".into(),
4125                error_description: "PAR rate limit exceeded for this client".into(),
4126            });
4127    }
4128
4129    // RFC 9449 §10.1 — resolve the DPoP key this authorization binds to.
4130    //
4131    //   Both mechanisms MUST be supported by an authorization server that
4132    //   supports PAR and DPoP.  If both mechanisms are used at the same time,
4133    //   the authorization server MUST reject the request if the JWK Thumbprint
4134    //   in dpop_jkt does not match the public key in the DPoP header.
4135    //
4136    // Deliberately after client authentication and after the rate limit.
4137    // Verifying a proof *records its `jti`*, so doing it earlier would let an
4138    // unauthenticated caller write to the replay table at will — and the
4139    // refusals below describe a key, which is a fact about the caller and not
4140    // about the client registration, so nothing is leaked by the wait.
4141    //
4142    // The header alone binds just as firmly as the parameter alone: §10.1 says
4143    // the server "MUST further behave as if the contained public key's
4144    // thumbprint was provided using dpop_jkt". That is the whole reason this
4145    // resolves to a single value rather than carrying both onward — by the
4146    // time the token endpoint asks, there is only one right answer and no
4147    // benefit in it having to reconstruct which carrier produced it.
4148    let proof_jkt = match verify_dpop_header(&http_req, &state, tenant_id).await {
4149        Ok(verified) => verified.map(|v| v.jkt),
4150        Err(response) => return *response,
4151    };
4152    let dpop_jkt = match (req.dpop_jkt.as_deref(), proof_jkt.as_deref()) {
4153        (Some(param), Some(proof)) if param != proof => {
4154            tracing::debug!(
4155                client_id = %client_id_for_log,
4156                "PAR carried a dpop_jkt naming a different key from the DPoP proof; refusing"
4157            );
4158            return *dpop_error_response(
4159                "invalid_dpop_proof",
4160                "dpop_jkt does not match the key of the DPoP proof on this request \
4161                 (RFC 9449 §10.1)",
4162                None,
4163            );
4164        }
4165        // Either they agree, or exactly one arrived, or neither did. In all
4166        // three the parameter wins where it exists and the proof supplies the
4167        // binding where it does not — which is the same value in the agreeing
4168        // case, so the arms need not be distinguished.
4169        (Some(param), _) => Some(param.to_owned()),
4170        (None, proof) => proof.map(str::to_owned),
4171    };
4172
4173    match state
4174        .oauth2
4175        .par_service
4176        .push(axiam_oauth2::par::PushedRequest {
4177            tenant_id,
4178            client_id: client.client_id,
4179            response_type: req.response_type,
4180            redirect_uri: req.redirect_uri,
4181            scope: req.scope,
4182            state: req.state,
4183            code_challenge: req.code_challenge,
4184            code_challenge_method: req.code_challenge_method,
4185            nonce: req.nonce,
4186            prompt: req.prompt,
4187            max_age: req.max_age,
4188            acr_values: req.acr_values,
4189            claims: req.claims,
4190            id_token_hint: req.id_token_hint,
4191            login_hint: req.login_hint,
4192            display: req.display,
4193            ui_locales: req.ui_locales,
4194            claims_locales: req.claims_locales,
4195            dpop_jkt,
4196        })
4197        .await
4198    {
4199        Ok(resp) => HttpResponse::Created()
4200            .append_header(("Cache-Control", "no-store"))
4201            .append_header(("Pragma", "no-cache"))
4202            .json(PushedAuthorizationResponse {
4203                request_uri: resp.request_uri,
4204                expires_in: resp.expires_in,
4205            }),
4206        Err(e) => build_oauth2_error_response(&e),
4207    }
4208}

4219    /// *authenticated* statement of which session and client this is about.
4220    pub id_token_hint: Option<String>,

4222    /// client's allow-list, by exact match.

4229
4230/// `GET`/`POST /oauth2/end_session` — OIDC RP-Initiated Logout 1.0 (B5).
4231///
4232/// Ends the session named by `id_token_hint` and notifies every client that
4233/// participated in it (Back-Channel Logout 1.0), then either redirects to an
4234/// allow-listed `post_logout_redirect_uri` or renders AXIAM's own logged-out
4235/// page.
4236///
4237/// # The redirect is the whole security surface
4238///
4239/// This endpoint is reachable without authentication by design — a user whose
4240/// session has already expired must still be able to complete a logout — so an
4241/// unvalidated `post_logout_redirect_uri` would be an open redirect on an
4242/// unauthenticated endpoint. It is honoured **only** on exact match against
4243/// the identified client's `post_logout_redirect_uris`.
4244///
4245/// A non-matching URI does not fail the request: the session is still ended
4246/// and AXIAM renders its own page. Refusing to log a user out because their RP
4247/// sent a bad parameter would be the wrong failure — they asked to log out.
4248///
4249/// # Why an unverifiable hint cannot end a session
4250///
4251/// Without a verifiable `id_token_hint` there is nothing to identify but the
4252/// browser's own AXIAM cookie, so the endpoint ends the cookie session if
4253/// there is one and does nothing otherwise. It deliberately does **not** fall
4254/// back to "end every session for the subject named in an unverified
4255/// parameter": that is a denial-of-service primitive handed to anyone who
4256/// knows a user id.
4257///
4258/// # No confirmation prompt
4259///
4260/// RP-Initiated Logout 1.0 §2 permits one, and a prompt is the mitigation for
4261/// logout CSRF. We take the other side deliberately: a forced logout is a
4262/// nuisance, not a privilege escalation, and a prompt shown on every logout is
4263/// trained away within a week.
4264#[utoipa::path(
4265    get,
4266    path = "/oauth2/end_session",
4267    tag = "oauth2",
4268    params(EndSessionQuery),
4269    responses(
4270        (status = 302, description = "Session ended; redirected to an allow-listed URI"),
4271        (status = 200, description = "Session ended; AXIAM's logged-out page"),
4272        (status = 400, description = "OAuth2 error", body = OAuth2ErrorResponse),
4273    ),
4274)]
4275pub async fn end_session<C: Connection + Clone>(
4276    query: web::Query<EndSessionQuery>,
4277    state: web::Data<AppState<C>>,
4278) -> HttpResponse {
4279    let q = query.into_inner();
4280    let tenant_id = q.tenant_id;
4281
4282    // 1. Identify the session and client. The hint is a *signed* statement of
4283    //    both; `client_id` is an unauthenticated parameter and is only a
4284    //    fallback for choosing the redirect allow-list.
4285    let hinted = q.id_token_hint.as_deref().and_then(|t| {
4286        axiam_oauth2::logout::decode_id_token_hint(t, &state.auth_config.jwt_public_key_pem)
4287    });
4288
4289    let (session_id, hint_client_id, subject_id) = match hinted {
4290        Some(h) => (h.session_id, Some(h.client_id), h.subject_id),
4291        None => (None, None, None),
4292    };
4293
4294    // 2. When both are present and disagree, refuse rather than resolve in
4295    //    favour of either. Picking the signed one would silently ignore what
4296    //    the caller asked; picking the unsigned one would let a parameter
4297    //    override a signature.
4298    if let (Some(hint_client), Some(param_client)) =
4299        (hint_client_id.as_deref(), q.client_id.as_deref())
4300        && hint_client != param_client
4301    {
4302        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
4303            "id_token_hint and client_id identify different clients".into(),
4304        ));
4305    }
4306
4307    let effective_client_id = hint_client_id.or_else(|| q.client_id.clone());
4308
4309    // 3. End the session, and notify the clients that were in it.
4310    if let (Some(session_id), Some(subject_id)) = (session_id, subject_id) {
4311        dispatch_backchannel_logout(&state, tenant_id, session_id, subject_id).await;
4312        // Best-effort: a session that has already expired is not an error —
4313        // the user asked to be logged out and they are.
4314        let _ = state.auth_service.logout(tenant_id, session_id).await;
4315    }
4316
4317    // 4. Resolve the redirect against the identified client's allow-list.
4318    let allow_list = match effective_client_id.as_deref() {
4319        Some(client_id) => state
4320            .oauth2_client_repo
4321            .get_by_client_id(tenant_id, client_id)
4322            .await
4323            .map(|c| c.post_logout_redirect_uris)
4324            .unwrap_or_default(),
4325        None => Vec::new(),
4326    };
4327
4328    let cookie_secure = state.auth_config.cookie_secure;
4329
4330    match axiam_oauth2::logout::resolve_post_logout_redirect(
4331        q.post_logout_redirect_uri.as_deref(),
4332        &allow_list,
4333        q.state.as_deref(),
4334    ) {
4335        axiam_oauth2::logout::LogoutOutcome::Redirect { uri, state: st } => {
4336            let location = match (url::Url::parse(&uri), st) {
4337                (Ok(mut url), Some(st)) => {
4338                    url.query_pairs_mut().append_pair("state", &st);
4339                    url.to_string()
4340                }
4341                (Ok(url), None) => url.to_string(),
4342                // An allow-listed entry that will not parse is an operator
4343                // error, not a user-facing one; render rather than emit a
4344                // Location header we could not construct.
4345                (Err(_), _) => {
4346                    return logged_out_page(cookie_secure);
4347                }
4348            };
4349            HttpResponse::Found()
4350                .append_header(("Location", location))
4351                .append_header(("Cache-Control", "no-store"))
4352                .cookie(crate::middleware::csrf::clear_access_cookie(cookie_secure))
4353                .cookie(crate::middleware::csrf::clear_refresh_cookie(cookie_secure))
4354                .cookie(crate::middleware::csrf::clear_csrf_cookie(cookie_secure))
4355                // W3: RP-initiated logout clears the OP browser session too.
4356                // Leaving it would mean a user who logged out through one
4357                // relying party is still recognised, silently, by the next.
4358                .cookie(crate::middleware::csrf::clear_op_session_cookie())
4359                .finish()
4360        }
4361        axiam_oauth2::logout::LogoutOutcome::Rendered => logged_out_page(cookie_secure),
4362    }
4363}

4365/// AXIAM's own logged-out page.
4366///
4367/// Deliberately carries no RP-supplied content — not the `state`, not the
4368/// rejected URI. Echoing either would put an attacker-controlled string into
4369/// a page served from AXIAM's own origin.
4370///
4371/// `cookie_secure` is `AuthConfig::cookie_secure` (D-18): the removal cookies
4372/// must carry the same attributes as the ones they clear.
4373fn logged_out_page(cookie_secure: bool) -> HttpResponse {
4374    HttpResponse::Ok()
4375        .append_header(("Cache-Control", "no-store"))
4376        .content_type("text/html; charset=utf-8")
4377        .cookie(crate::middleware::csrf::clear_access_cookie(cookie_secure))
4378        .cookie(crate::middleware::csrf::clear_refresh_cookie(cookie_secure))
4379        .cookie(crate::middleware::csrf::clear_csrf_cookie(cookie_secure))
4380        .cookie(crate::middleware::csrf::clear_op_session_cookie())
4381        .body(
4382            "<!doctype html><html lang=\"en\"><head><meta charset=\"utf-8\">\
4383             <title>Signed out</title></head><body><h1>You are signed out.</h1>\
4384             </body></html>",
4385        )
4386}

4388/// Build and dispatch logout tokens for a session's participants.
4389///
4390/// Spawned rather than awaited: the user's session is gone the moment this
4391/// request returns, and making logout wait on N external HTTP calls would make
4392/// it a hostage to the least reliable RP.
4393async fn dispatch_backchannel_logout<C: Connection + Clone>(
4394    state: &web::Data<AppState<C>>,
4395    tenant_id: Uuid,
4396    session_id: Uuid,
4397    subject_id: Uuid,
4398) {
4399    let participants = match state
4400        .oauth2
4401        .session_client_repo
4402        .list_for_session(tenant_id, session_id)
4403        .await
4404    {
4405        Ok(rows) => rows,
4406        Err(e) => {
4407            tracing::warn!(error = %e, "could not list session participants for logout");
4408            return;
4409        }
4410    };
4411    if participants.is_empty() {
4412        return;
4413    }
4414
4415    let mut clients = Vec::new();
4416    for client_id in participants
4417        .iter()
4418        .map(|p| p.client_id.clone())
4419        .collect::<std::collections::BTreeSet<_>>()
4420    {
4421        match state
4422            .oauth2_client_repo
4423            .get_by_client_id(tenant_id, &client_id)
4424            .await
4425        {
4426            Ok(c) => clients.push(c),
4427            // A participant whose registration cannot be loaded is dropped
4428            // from the fan-out — correct (a deleted client keeps its
4429            // participation rows), but silent until now. The RP that does not
4430            // get told sees nothing, and neither did anyone reading the logs.
4431            Err(e) => tracing::debug!(
4432                %client_id,
4433                error = %e,
4434                "back-channel logout: session participant is not a loadable client, skipping"
4435            ),
4436        }
4437    }
4438
4439    let issuer = state.auth_config.effective_issuer().to_string();
4440    let auth_config = state.auth_config.clone();
4441    let participant_ids: Vec<String> = participants.into_iter().map(|p| p.client_id).collect();
4442
4443    let deliveries = crate::backchannel_logout::select_targets(
4444        &participant_ids,
4445        &clients,
4446        |client_id| {
4447            match axiam_oauth2::logout::issue_logout_token(
4448                &issuer,
4449                client_id,
4450                session_id,
4451                subject_id,
4452                &auth_config,
4453            ) {
4454                Ok(token) => Some(token),
4455                // This used to be `.ok()`. A token that cannot be minted (a
4456                // misconfigured issuer, an unusable signing key) silently
4457                // removed the client from the fan-out, so a whole deployment
4458                // could stop notifying anyone with no signal anywhere — on a
4459                // session-termination path, where "nothing happened" is
4460                // exactly the outcome an attacker wants. WARN, not DEBUG: this
4461                // one is never routine.
4462                Err(e) => {
4463                    tracing::warn!(
4464                        %client_id,
4465                        error = %e,
4466                        "back-channel logout: could not issue a logout token, client will NOT be notified"
4467                    );
4468                    None
4469                }
4470            }
4471        },
4472    );
4473
4474    // One line naming every stage of the funnel, so a delivery that never
4475    // happens can be localised without a code read: how many clients joined
4476    // the session, how many of those are still loadable registrations, and how
4477    // many of those actually got a token and a URI to send it to. Each drop
4478    // between those numbers has its own line above (or is the deliberate
4479    // "client registered no backchannel_logout_uri" skip in `select_targets`).
4480    tracing::debug!(
4481        participants = participant_ids.len(),
4482        clients = clients.len(),
4483        deliveries = deliveries.len(),
4484        "back-channel logout fan-out computed"
4485    );
4486
4487    // Participation records are dropped after the fan-out list is built, not
4488    // before — the list is what the fan-out iterates.
4489    let _ = state
4490        .oauth2
4491        .session_client_repo
4492        .delete_for_session(tenant_id, session_id)
4493        .await;
4494
4495    if !deliveries.is_empty() {
4496        actix_web::rt::spawn(crate::backchannel_logout::deliver_all(deliveries));
4497    }
4498}
