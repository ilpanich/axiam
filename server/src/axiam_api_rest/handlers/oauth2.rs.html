42/// Cap for the caller-supplied `client_id` recorded in a failed-client-auth
43/// audit row (SEC-087). Server-generated ids are `oa_`/`sa_` + a 32-char
44/// encoding, so this leaves generous headroom for a legitimate value while
45/// bounding what an anonymous caller can push into the log.
46const MAX_CLIENT_ID_LEN: usize = 128;
47
48/// Truncate to at most [`MAX_CLIENT_ID_LEN`] **bytes**, cutting on a character
49/// boundary so the result is still valid UTF-8.
50///
51/// §22.3 residual 3: the previous `chars().take(N)` capped code points, not
52/// bytes, so a multibyte id could still store 4× the intended size. The cap
53/// exists to bound what an anonymous caller can push into an append-only log,
54/// and a bound expressed in the wrong unit is not the bound that was intended.
55fn truncate_bytes_on_char_boundary(s: &str) -> String {
56    if s.len() <= MAX_CLIENT_ID_LEN {
57        return s.to_owned();
58    }
59    let mut end = MAX_CLIENT_ID_LEN;
60    while end > 0 && !s.is_char_boundary(end) {
61        end -= 1;
62    }
63    s[..end].to_owned()
64}

190/// Read the authentication behind this request's session (X7.2, plan §4.3).
191///
192/// The evidence is snapshotted onto the authorization code, so it has to be
193/// read here, while the session still exists: refresh rotation replaces the
194/// row, and by the time the code is redeemed the session it was minted from
195/// may be a different row with a different id.
196///
197/// # It never fails the request
198///
199/// A session that cannot be read yields empty evidence and a `debug!` line,
200/// not an error. Two reasons, and both are invariant 4: an authorization
201/// request that works today must not start failing because of a column added
202/// this wave, and the principal on this path is not always a browser session
203/// at all (a bearer token minted for a service, a session already rotated
204/// away). Empty evidence is also the strict reading — a code with no
205/// `auth_time` can never present as fresh.
206async fn resolve_session_evidence<C: Connection + Clone>(
207    state: &AppState<C>,
208    tenant_id: Uuid,
209    session_id: Uuid,
210) -> SessionEvidence {
211    match state.session_repo.get_by_id(tenant_id, session_id).await {
212        Ok(session) => SessionEvidence {
213            auth_time: Some(session.authenticated_at),
214            amr: session.amr,
215        },
216        Err(e) => {
217            tracing::debug!(
218                error = %e,
219                %session_id,
220                "no session row behind this authorization request; the code is issued \
221                 without authentication evidence"
222            );
223            SessionEvidence::default()
224        }
225    }
226}

228/// Record the outcome of a `prompt=none` authorization request (W4, plan §4.2).
229///
230/// `prompt=none` is a **silent-authentication oracle**: a registered relying
231/// party learns, without any interaction, whether this browser is signed in.
232/// That is inherent to the parameter and it is accepted — it is bounded to
233/// clients registered in this tenant with exact `redirect_uri` matching, and
234/// nothing but one bit crosses — but "accepted" and "invisible" are different
235/// things. One audit row per outcome is what makes abuse of it something an
236/// operator can see and count rather than something they have to be told about
237/// by the party doing it.
238///
239/// Two actions are emitted, `oauth2.prompt_none.code` and
240/// `oauth2.prompt_none.login_required`. The plan names a third,
241/// `oauth2.prompt_none.consent_required`, which **cannot** be emitted in W4:
242/// it needs a consent-gated scope and there are none until W7 (plan §4.8). It
243/// is not written here rather than written into a branch that can never run.
244///
245/// Never fails the request. An audit sink that is down costs a row, not a
246/// login (T-15-04).
247async fn audit_prompt_none<C: Connection + Clone>(
248    state: &AppState<C>,
249    http_req: &HttpRequest,
250    tenant_id: Uuid,
251    client_id: &str,
252    refusal: Option<&OAuth2Error>,
253) {
254    let (action, result) = match refusal {
255        None => ("oauth2.prompt_none.code", AuditOutcome::Success),
256        Some(e) => (
257            match e.error_code() {
258                "login_required" => "oauth2.prompt_none.login_required",
259                _ => "oauth2.prompt_none.refused",
260            },
261            AuditOutcome::Failure,
262        ),
263    };
264    if let Err(e) = state
265        .audit_repo
266        .append(CreateAuditLogEntry {
267            tenant_id,
268            actor_id: Uuid::nil(),
269            actor_type: ActorType::System,
270            action: action.into(),
271            resource_id: None,
272            outcome: result,
273            ip_address: peer_ip(http_req),
274            metadata: Some(serde_json::json!({
275                // The `client_id` is the point of the row: the oracle is
276                // per-client, so counting it per client is what makes a
277                // relying party polling every few seconds visible.
278                "client_id": truncate_bytes_on_char_boundary(client_id),
279                "error": refusal.map(OAuth2Error::error_code),
280            })),
281        })
282        .await
283    {
284        tracing::error!(
285            error = %e,
286            %tenant_id,
287            "could not record a prompt=none outcome; the authorization request is unaffected"
288        );
289    }
290}

292/// Decode an `id_token_hint` presented at the authorization endpoint (W4,
293/// plan §4.2).
294///
295/// One decoder, shared with `/oauth2/end_session`
296/// ([`axiam_oauth2::logout::decode_id_token_hint`]): the signature is checked,
297/// the expiry deliberately is not — an `id_token_hint` has routinely expired,
298/// which is exactly why an RP is sending one rather than a live token — and
299/// `aud` is read out rather than validated against, because who the token was
300/// for is part of what the caller wants to learn. A second decoder here would
301/// be a second opinion about what a hint proves.
302///
303/// Returns `None` when no hint arrived and when one arrived and did not
304/// verify. The caller distinguishes them by
305/// [`AuthnRequestParams::id_token_hint`], and treats the second as a hint that
306/// names somebody else rather than as an absent one — dropping an unverifiable
307/// hint would let any string turn a `prompt=none` refusal into a code.
308///
309/// Run for every request carrying the parameter, not only for the honour lane.
310/// It costs one Ed25519 verification of a value the request already carried,
311/// it cannot change any response on the `ignore` lane (nothing reads the
312/// result there), and making it conditional would put the lane decision in a
313/// second place.
314fn decode_authorize_id_token_hint<C: Connection + Clone>(
315    state: &AppState<C>,
316    params: &AuthnRequestParams,
317) -> Option<axiam_oauth2::logout::IdTokenHint> {
318    let raw = params.id_token_hint.as_deref()?;
319    axiam_oauth2::logout::decode_id_token_hint(raw, &state.auth_config.jwt_public_key_pem)
320}

322/// Classify a request object on an authorization request (X7 G12, plan §4.10).
323///
324/// AXIAM accepts neither form. `request` is RFC 9101's request object by
325/// value; a `request_uri` that is not a `urn:ietf:params:oauth:request_uri:`
326/// handle is a request object by reference, which would have the authorization
327/// server fetch an attacker-chosen URL — an SSRF primitive that PAR (RFC 9126
328/// §1) made unnecessary and that FAPI 2.0 does not ask for.
329///
330/// `request` is checked first: when both arrive, the by-value object is the
331/// one the server would have had to *read*, so it is the one the refusal
332/// should name.
333///
334/// Returns `None` for a request carrying no request object at all, including
335/// one whose `request_uri` is a genuine PAR handle — that path is untouched.
336fn classify_request_object(
337    request: Option<&str>,
338    request_uri: Option<&str>,
339) -> Option<RequestObject> {
340    // A blank value is a client library filling in a template, not an object.
341    // Refusing it would break a request that works today and carries nothing.
342    fn present(v: Option<&str>) -> Option<&str> {
343        v.map(str::trim).filter(|s| !s.is_empty())
344    }
345
346    if present(request).is_some() {
347        return Some(RequestObject::ByValue);
348    }
349    match present(request_uri) {
350        Some(uri) if !uri.starts_with(axiam_oauth2::par::REQUEST_URI_PREFIX) => {
351            Some(RequestObject::ByReference)
352        }
353        _ => None,
354    }
355}

380/// Is this `return_to` a path on an origin this deployment owns?
381///
382/// The second of the two server-side checks the plan asks for. The first is
383/// [`axiam_oauth2::login_hop::validate_return_to`], which is syntactic and
384/// demands the exact authorize path; this one *resolves* the candidate against
385/// the issuer and hands the result to
386/// [`require_deployment_spa_origin`](crate::handlers::federation_login::require_deployment_spa_origin)
387/// — the rule that already decides where a federation SSO handoff code may be
388/// sent, rather than a second rule that could come to disagree with it.
389///
390/// While the syntactic check demands `/oauth2/authorize?…` this is redundant,
391/// and it is kept for the case that stops being true: `Url::join` is what turns
392/// `//evil.example` into a different host and `/a/../../b` into `/b`, so it is
393/// an independent opinion on the two attacks that are about *resolution*
394/// rather than about spelling.
395/// The authorization endpoint path **this** request arrived at (T21.6).
396///
397/// `/oauth2/authorize` for every request on the deployment-wide endpoint, which
398/// is every request on a deployment with `AXIAM__AUTH__TENANT_ISSUER_PATHS`
399/// unset — so a login or consent hop builds and validates exactly the value it
400/// built and validated before tenant paths existed.
401///
402/// `/t/{tenant_id}/oauth2/authorize` on a per-tenant path, built from the
403/// *parsed* tenant id the scope middleware recorded rather than from the path
404/// string the request carried, so the result cannot be anything the UUID parse
405/// did not already accept.
406fn authorize_path_of(req: &HttpRequest) -> String {
407    match crate::middleware::tenant_path::binding_of(req) {
408        None => axiam_oauth2::login_hop::AUTHORIZE_PATH.to_owned(),
409        Some(binding) => axiam_oauth2::login_hop::tenant_authorize_path(binding.tenant_id),
410    }
411}

413fn return_to_is_on_this_deployment<C: Connection + Clone>(
414    state: &AppState<C>,
415    return_to: &str,
416    authorize_path: &str,
417) -> bool {
418    // The deployment ROOT, never the tenant issuer: this resolves a path
419    // against an origin, and a tenant issuer carries a path of its own that
420    // `Url::join` would then treat as a base directory. `authorize_path` is
421    // where the tenant, if any, is expressed — and it is supplied by the caller
422    // from the request it is answering (T21.6), never read out of the candidate.
423    let Ok(base) = url::Url::parse(state.auth_config.root_issuer()) else {
424        return false;
425    };
426    let Ok(resolved) = base.join(return_to) else {
427        return false;
428    };
429    if resolved.path() != authorize_path {
430        return false;
431    }
432    crate::handlers::federation_login::require_deployment_spa_origin(state, resolved.as_str())
433        .is_ok()
434}

436/// Resolve who this authorization request is acting for (W3, plan §4.0).
437///
438/// Two ways in, in this order:
439///
440/// 1. **An access token** — the `axiam_access` cookie or a `Bearer`/`DPoP`
441///    header, via [`AuthenticatedUser`]. Unchanged, and it wins whenever it
442///    succeeds. Every request that works today takes this path and cannot tell
443///    that the other exists.
444/// 2. **The `axiam_op_session` cookie**, and only for a client registered
445///    `browser_sso`. This is the path that makes AXIAM usable as an OP for a
446///    third-party relying party at all: `axiam_access` is `SameSite=Strict`, so
447///    the cross-site top-level navigation an RP redirect *is* never carries it,
448///    and until this wave a signed-in user arrived here anonymous.
449///
450/// # Order of operations, and why the client is loaded first
451///
452/// The client must be read before the cookie is, because the client's
453/// registration is what decides whether the cookie is consulted. That is the
454/// reordering plan §4.0 describes, and it is not a new trust decision — the
455/// authorize service already loads this row before any redirectable error, so
456/// the lookup moves one step earlier rather than appearing.
457///
458/// # Every anonymous refusal is today's refusal
459///
460/// No tenant, no such client, a client that did not opt in, a repository that
461/// failed: all four return the **same** 401 the extractor would have produced,
462/// byte for byte, by returning the extractor's own error object. That is
463/// invariant 4 and it is also what keeps the endpoint from becoming an oracle:
464/// an anonymous caller cannot learn from the response whether a `client_id`
465/// exists in a tenant, only whether it opted into the hop.
466///
467/// # Errors
468///
469/// Returns a complete [`HttpResponse`] — the 401, the 302 into the login page,
470/// or the loop guard's terminal `login_required` — because each of the three
471/// is finished at the point it is decided and none is expressible as an
472/// [`OAuth2Error`] the caller could usefully re-render. Boxed: see
473/// [`AuthorizeRefusal`].
474async fn resolve_authorize_principal<C: Connection + Clone>(
475    state: &AppState<C>,
476    http_req: &HttpRequest,
477    q: &AuthorizeQuery,
478    maybe_user: MaybeAuthenticatedUser,
479) -> Result<AuthorizePrincipal, AuthorizeRefusal> {
480    use actix_web::ResponseError;
481
482    let auth_error = match maybe_user.into_result() {
483        Ok(user) => {
484            // Path 1. `tenant_id` on the query is ignored here, deliberately:
485            // the tenant a token was minted for is the tenant it acts in, and
486            // letting a query parameter move that would be a tenant-crossing
487            // primitive handed to whoever holds the browser.
488            return Ok(AuthorizePrincipal {
489                tenant_id: user.tenant_id,
490                user_id: user.user_id,
491                session_id: user.session_id,
492            });
493        }
494        Err(e) => e,
495    };
496
497    // ---- Path 2: anonymous ------------------------------------------------
498    let Some(tenant_id) = q.tenant_id else {
499        return Err(Box::new(auth_error.error_response()));
500    };
501    let Ok(client) = state
502        .oauth2_client_repo
503        .get_by_client_id(tenant_id, &q.client_id)
504        .await
505    else {
506        return Err(Box::new(auth_error.error_response()));
507    };
508    if !client.browser_sso {
509        // **T0.1 / T0.5 / M7's I4 twin.** Every client registered today lands
510        // here, including one whose browser happens to be carrying an
511        // `axiam_op_session` cookie: the cookie is not read, because reading it
512        // is what `browser_sso` opts into.
513        return Err(Box::new(auth_error.error_response()));
514    }
515
516    // The person said no.
517    //
518    // Answered here, before the session cookie is consulted, because the answer
519    // does not depend on it: somebody who cancels the sign-in page has by
520    // definition not signed in, and asking whether they might have a session
521    // from some earlier request would let a stale cookie turn a refusal into a
522    // grant. The two conditions above still apply first — an unknown client and
523    // a client that never opted into the hop are answered exactly as they are
524    // today, because this endpoint must not become an oracle for either.
525    //
526    // Delivered by redirect on the same terms as every other authorization
527    // error (RFC 6749 §4.1.2.1): only to a `redirect_uri` this client
528    // registered — as `any_redirect_uri_matches` reads "registered", so a
529    // desktop client's ephemeral loopback port is one — with the request's own
530    // `state`. An unregistered or absent one is answered directly, because a
531    // refusal is still not a licence to send a browser somewhere the client
532    // never named.
533    if axiam_oauth2::login_hop::user_declined(q.user_declined.as_deref()) {
534        let refusal = OAuth2Error::AccessDenied(
535            "the end user declined the authorization request at the sign-in page".into(),
536        );
537
538        // Where to answer, and with whose `state` — and for a pushed request
539        // neither comes from the query string.
540        //
541        // RFC 9126 §1: the pushed copy is the request. A FAPI client sends
542        // `client_id` and `request_uri` to the authorization endpoint and may
543        // send nothing else, so reading `state` off the query answered a
544        // refusal with no `state` at all — which the suite reports as
545        // `CheckStateInAuthorizationResponse: State was passed in request, but
546        // is missing from response`, and which a relying party would be right
547        // to discard as an unsolicited response (RFC 6749 §10.12).
548        //
549        // Consuming the `request_uri` here is correct rather than merely
550        // convenient: it is single-use, and this request has just been answered
551        // terminally. Leaving it spendable would let the same pushed request be
552        // presented again after its user said no.
553        let pushed = match q.request_uri.as_deref() {
554            Some(uri) => state
555                .oauth2
556                .par_service
557                .consume(tenant_id, &q.client_id, uri)
558                .await
559                .ok(),
560            None => None,
561        };
562        let (redirect_uri, echo_state) = match pushed.as_ref() {
563            Some(p) => (Some(p.redirect_uri.as_str()), p.state.as_deref()),
564            None => (q.redirect_uri.as_deref(), q.state.as_deref()),
565        };
566
567        return Err(Box::new(match redirect_uri {
568            Some(uri) if any_redirect_uri_matches(&client.redirect_uris, uri) => {
569                build_error_redirect(uri, &refusal, echo_state, &issuer_config(http_req, state))
570            }
571            _ => authorize_error_response(http_req, &refusal),
572        }));
573    }
574
575    // The cookie, and whether it still names a live session in this tenant.
576    let presented = http_req.cookie(crate::middleware::csrf::COOKIE_OP_SESSION);
577    let resolved = match presented.as_ref() {
578        Some(cookie) => {
579            let digest = axiam_auth::token::hash_browser_session_token(cookie.value());
580            match state
581                .session_repo
582                .get_by_browser_token_hash(tenant_id, &digest)
583                .await
584            {
585                Ok(found) => found,
586                Err(e) => {
587                    // A read failure is not a principal. Logged without the
588                    // cookie value or its digest — the first is a credential and
589                    // the second is enough to recognise one.
590                    tracing::warn!(
591                        error = %e,
592                        %tenant_id,
593                        "could not resolve the OP browser session; treating the \
594                         request as anonymous"
595                    );
596                    None
597                }
598            }
599        }
600        None => None,
601    };
602
603    if let Some(session) = resolved {
604        return Ok(AuthorizePrincipal {
605            tenant_id,
606            user_id: session.user_id,
607            session_id: session.id,
608        });
609    }
610
611    // ---- No principal: refuse silently, hop, or stop ---------------------
612    //
613    // W4 (plan §4.2, T1.1/T1.7). `prompt=none` says: answer without showing
614    // the end user anything. There is nothing to answer with — no session
615    // resolved — so the answer is `login_required`, **redirected to the
616    // relying party** rather than to a sign-in page it forbade.
617    //
618    // Read from the query string only. A pushed request's `prompt` cannot be
619    // seen here at all (the handle is consumed later, inside the handler), so
620    // a PAR client's `prompt=none` takes the hop first and is refused on the
621    // return leg instead, by the marker rule in `axiam_oauth2::honour` — the
622    // interaction is never converted into a code either way.
623    //
624    // Honour lane only, and that is what keeps T0.1 exact: a `browser_sso`
625    // client registered `ignore` — which is every client that exists — sending
626    // `prompt=none` gets today's answer, byte for byte.
627    //
628    // W5 (plan §4.5/§4.6) reads the same bundle for the cosmetic four, which is
629    // why it is parsed once here with every field rather than only `prompt`.
630    // Query string only, as before: a pushed request's parameters cannot be
631    // seen at this point (the handle is consumed inside the handler, after a
632    // principal exists), so a PAR client's presentation reaches the page on the
633    // *interaction* arm below instead. Both arms build their URL through the
634    // one builder, so the two carriers cannot come to disagree about what a
635    // `/login` URL looks like.
636    let honour = client.authn_request_params.is_honour();
637    let authn_params = honour.then(|| {
638        AuthnRequestParams::parse(&RawAuthnParams {
639            prompt: q.prompt.as_deref(),
640            login_hint: q.login_hint.as_deref(),
641            display: q.display.as_deref(),
642            ui_locales: q.ui_locales.as_deref(),
643            // `claims_locales` is deliberately not read: it selects the
644            // language of *claim values*, which AXIAM does not localise, and
645            // it must not reach the page's language. See
646            // `axiam_oauth2::login_hop::Cosmetic::from_params`.
647            ..Default::default()
648        })
649    });
650    if let Some(params) = authn_params.as_ref()
651        && params
652            .prompt
653            .contains(&axiam_oauth2::authn_params::Prompt::None)
654    {
655        let refusal = OAuth2Error::LoginRequired(
656            "no end user is authenticated at this authorization server, and prompt=none \
657                 forbids asking them to sign in"
658                .into(),
659        );
660        audit_prompt_none(state, http_req, tenant_id, &q.client_id, Some(&refusal)).await;
661        // RFC 6749 §4.1.2.1: an error is redirected only to a
662        // `redirect_uri` this client registered, and "registered" means what
663        // `any_redirect_uri_matches` means by it — string equality, plus RFC
664        // 8252 §7.3's port allowance for a registered `http` loopback URI.
665        // The same comparison `AuthorizeService::authorize` makes, which is
666        // the point: until T21.8 this arm compared exactly while authorize
667        // used the matcher, so a desktop client that registered
668        // `http://127.0.0.1/callback` and listened on an ephemeral port got
669        // its successes redirected and its refusals rendered as a page it
670        // could not read. An unregistered or absent URI is still answered
671        // directly.
672        return Err(Box::new(match q.redirect_uri.as_deref() {
673            Some(uri) if any_redirect_uri_matches(&client.redirect_uris, uri) => {
674                build_error_redirect(
675                    uri,
676                    &refusal,
677                    q.state.as_deref(),
678                    &issuer_config(http_req, state),
679                )
680            }
681            _ => authorize_error_response(http_req, &refusal),
682        }));
683    }
684
685    // The loop guard. A request that already carries the marker is one this
686    // server sent to the login page and got back; sending it there a second
687    // time is the non-terminating case, so it is answered instead. See
688    // `axiam_oauth2::login_hop` for the full argument.
689    if axiam_oauth2::login_hop::is_return_leg(q.login_hop.as_deref()) {
690        tracing::warn!(
691            %tenant_id,
692            client_id = %q.client_id,
693            "an authorization request returned from the login hop still \
694             carrying no OP session; refusing to redirect again"
695        );
696        return Err(Box::new(authorize_error_response(
697            http_req,
698            &OAuth2Error::LoginRequired(
699                "the sign-in did not establish a session for this tenant at this \
700                 origin; sign in again from the relying party, and check that the \
701                 browser accepts the axiam_op_session cookie"
702                    .into(),
703            ),
704        )));
705    }
706
707    // Two refusals stand here, and they are mutually exclusive by
708    // construction: the `response_type` check below runs only when there is NO
709    // `request_uri`, and the `peek` after it only when there is one. They can
710    // never both fire, so their order carries no meaning — the cheaper,
711    // datastore-free one is simply first.
712
713    // Do not send a user to sign in for a request that cannot succeed.
714    //
715    // `response_type` is the one parameter whose validity is decidable here,
716    // with no principal and without touching the pushed request: RFC 6749
717    // §3.1.1 makes it REQUIRED, AXIAM supports exactly `code`, and neither
718    // fact depends on who is signing in. Everything else the authorization
719    // endpoint refuses needs either the client's registration read against a
720    // resolved tenant or the pushed parameters, and the `request_uri` those
721    // live behind is single-use — consumed once, in the handler, after a
722    // principal exists. That constraint is why this check is narrow rather
723    // than a second copy of `AuthorizeService::authorize`'s validation.
724    //
725    // Skipped entirely when a `request_uri` is present: with PAR the real
726    // `response_type` is the pushed one, `ParService::push` already refused
727    // anything but `code` at push time, and the query string's copy is not
728    // authoritative — RFC 9126 §4 says the pushed parameters are used and the
729    // query's are ignored, so rejecting on the query's absence here would
730    // refuse a perfectly good pushed request.
731    //
732    // The refusal follows RFC 6749 §4.1.2.1 exactly as the `prompt=none` arm
733    // above does, and for the same reason: an error goes back to a
734    // `redirect_uri` only when the client registered it, as
735    // `any_redirect_uri_matches` decides "registered", because an unvalidated
736    // one is an open redirect. Anything else is answered in place, which is
737    // also the error page a certification reviewer is asked to see for
738    // `oidcc-response-type-missing` — unchanged, because that module registers
739    // an `https` URI and the matcher is string equality for those.
740    if q.request_uri.is_none() && q.response_type.as_deref() != Some("code") {
741        let refusal = match q.response_type.as_deref() {
742            None | Some("") => {
743                OAuth2Error::InvalidRequest("response_type is required (RFC 6749 §3.1.1)".into())
744            }
745            Some(_) => OAuth2Error::UnsupportedResponseType,
746        };
747        return Err(Box::new(match q.redirect_uri.as_deref() {
748            Some(uri) if any_redirect_uri_matches(&client.redirect_uris, uri) => {
749                build_error_redirect(
750                    uri,
751                    &refusal,
752                    q.state.as_deref(),
753                    &issuer_config(http_req, state),
754                )
755            }
756            _ => authorize_error_response(http_req, &refusal),
757        }));
758    }
759
760    // The handle, before a person is asked to sign in for it.
761    //
762    // `/oauth2/authorize` cannot consume a `request_uri` here: it is single-use
763    // and is spent inside the handler, after a principal exists — which is what
764    // the `prompt=none` comment above means by "a pushed request's parameters
765    // cannot be seen at this point". The consequence was that a browser
766    // presenting an already-used, expired or wrong-client handle was sent to
767    // `/login` without the handle being looked at at all, the user typed a
768    // password, and the request was refused on the return leg. The OIDF module
769    // `fapi2-security-profile-final-par-attempt-reuse-request_uri` reports that
770    // as a screenshot of a sign-in page against a condition asking for an error
771    // page about an invalid `request_uri`.
772    //
773    // `peek` is a READ. It does not consume, does not authorize, and carries
774    // nothing forward: the authoritative single-use decision stays in
775    // `ParService::consume`, in the handler, in one statement (see
776    // `par.rs`'s §2 rule 1). A handle that passes here may still lose the race
777    // to a concurrent authorize request, and must.
778    //
779    // It therefore refuses exactly what was already dead and nothing that is
780    // merely unfinished — which is the property
781    // `par-ensure-reused-request-uri-prior-to-auth-completion-succeeds`
782    // depends on: a `request_uri` re-presented BEFORE the first authorization
783    // completed is still unconsumed and unexpired, so the login page is shown,
784    // as that module requires.
785    //
786    // Only a genuine PAR handle is asked about. A `request` object by value, or
787    // a `request_uri` that is not a PAR handle, is a different refusal with its
788    // own OIDC Core §3.1.2.6 code, owned by `classify_request_object` in the
789    // handler; turning either into `invalid_request` here would lose the code a
790    // conformance suite matches on.
791    if classify_request_object(q.request.as_deref(), q.request_uri.as_deref()).is_none()
792        && let Some(request_uri) = q.request_uri.as_deref()
793        && let Err(refusal) = state
794            .oauth2
795            .par_service
796            .peek(tenant_id, &q.client_id, request_uri)
797            .await
798    {
799        // RFC 6749 §4.1.2.1, answered exactly as the `prompt=none` arm above
800        // does: redirected only to a `redirect_uri` this client registered,
801        // compared by `any_redirect_uri_matches`, and answered directly
802        // otherwise. The pushed copy's `redirect_uri` is not available and
803        // must not be — resolving it is the thing that just failed.
804        let reported = request_uri_error_for_client(&refusal);
805        let reported = reported.as_ref().unwrap_or(&refusal);
806        return Err(Box::new(match q.redirect_uri.as_deref() {
807            Some(uri) if any_redirect_uri_matches(&client.redirect_uris, uri) => {
808                build_error_redirect(
809                    uri,
810                    reported,
811                    q.state.as_deref(),
812                    &issuer_config(http_req, state),
813                )
814            }
815            // Rendered for the person, with the wording `ParService` produced.
816            // The OIDC code above is what a *relying party* can act on; a page
817            // saying `invalid_request_uri` to somebody who did not send the
818            // parameter tells them nothing they can use.
819            _ => authorize_error_response(http_req, &refusal),
820        }));
821    }
822
823    // T21.6 — the browser comes back to the endpoint it left. On the
824    // deployment-wide path that is `/oauth2/authorize`, exactly as before; on a
825    // per-tenant path it is `/t/{tenant_id}/oauth2/authorize`, which is what
826    // makes the authorization response the return leg produces carry the tenant
827    // issuer in its RFC 9207 `iss` — the issuer the client discovered.
828    let authorize_path = authorize_path_of(http_req);
829    let Some(return_to) =
830        axiam_oauth2::login_hop::build_return_to_at(&authorize_path, http_req.query_string())
831    else {
832        // Nothing safe to come back to. Answer as if the request had been
833        // anonymous with no `browser_sso` at all rather than send a browser
834        // somewhere on a value that did not validate.
835        return Err(Box::new(auth_error.error_response()));
836    };
837    if !return_to_is_on_this_deployment(state, &return_to, &authorize_path) {
838        return Err(Box::new(auth_error.error_response()));
839    }
840
841    // A cookie that was presented and did not resolve is stale: this browser
842    // believes it is signed in and is not. `reauth` tells the SPA to say so and
843    // to authenticate afresh instead of trusting what it already holds, and the
844    // dead cookie is removed on the way out so the next attempt starts clean.
845    let stale = presented.is_some();
846    // W5. The presentation the relying party asked for, on the honour lane and
847    // nowhere else: `authn_params` is `None` for every client registered
848    // `ignore`, so `Cosmetic::NONE` is what they get and their `/login` URL is
849    // byte-identical to the one W3 built. Invariant 4, as a type rather than as
850    // a branch somebody has to remember to write.
851    //
852    // W7 — the tenant default, which W5 deferred for want of a place to
853    // configure it. Read only when a bundle arrived at all: a client on the
854    // `ignore` lane gets `Cosmetic::NONE` and no settings read, so its
855    // `/login` URL is byte-identical to W3's and costs the same.
856    let tenant_default = match authn_params.as_ref() {
857        Some(_) => tenant_default_locale(state, tenant_id).await,
858        None => None,
859    };
860    let cosmetic = authn_params
861        .as_ref()
862        .map_or(axiam_oauth2::login_hop::Cosmetic::NONE, |params| {
863            axiam_oauth2::login_hop::Cosmetic::from_params(params, tenant_default)
864        });
865    let location =
866        axiam_oauth2::login_hop::build_login_redirect_for(&return_to, stale, None, &cosmetic);
867
868    let mut builder = HttpResponse::Found();
869    builder
870        .append_header((actix_web::http::header::LOCATION, location))
871        // The URL carries the whole authorization request; it must not be
872        // cached, and it must not travel onward as a referrer.
873        .append_header((actix_web::http::header::CACHE_CONTROL, "no-store"))
874        .append_header(("Referrer-Policy", "no-referrer"));
875    if stale {
876        builder.cookie(crate::middleware::csrf::clear_op_session_cookie());
877    }
878    Err(Box::new(builder.finish()))
879}

881/// W7 — the tenant's fallback UI language, for the last step of W5's
882/// `ui_locales` chain (plan §4.6's deferral, now picked up).
883///
884/// W5 shipped `select_ui_locale(requested, tenant_default)` and unit-tested it
885/// through all three steps, then passed `None` at both call sites because there
886/// was nowhere for an operator to set the value. There is now:
887/// `TenantSettingsOverride::default_locale`, on the settings surface every
888/// other per-tenant control lives on. This is the one argument those two call
889/// sites were waiting for.
890///
891/// Three ways to get `None`, and all three land on the deployment default
892/// (`en`), which is what every deployment did before this wave:
893///
894/// * the tenant has no preference;
895/// * the settings could not be read (a login page is not worth failing over a
896///   language);
897/// * the stored tag is one this build does not ship. `Locale::from_tag` is an
898///   **exact** match rather than an RFC 4647 lookup, deliberately: a stored
899///   `fr-CA` means "somebody wrote something this binary does not ship", and
900///   answering it with French would be a guess presented as a setting.
901async fn tenant_default_locale<C: Connection + Clone>(
902    state: &AppState<C>,
903    tenant_id: Uuid,
904) -> Option<axiam_oauth2::locale::Locale> {
905    let tenant = state.tenant_repo.get_by_id(tenant_id).await.ok()?;
906    let settings = axiam_core::repository::SettingsRepository::get_effective_settings(
907        &state.settings_repo,
908        tenant.organization_id,
909        tenant_id,
910    )
911    .await
912    .ok()?;
913    let tag = settings.oidc.default_locale?;
914    let locale = axiam_oauth2::locale::Locale::from_tag(&tag);
915    if locale.is_none() {
916        tracing::warn!(
917            %tenant_id,
918            configured = %tag,
919            "this tenant's default_locale is not a locale this build ships; the sign-in page \
920             falls back to the deployment default"
921        );
922    }
923    locale
924}

926/// W7 / X7 G8 — resolve what this authorization request's GDPR-sensitive
927/// scopes have earned, so `AuthorizeService` can decide without a repository.
928///
929/// Returns `(state, switch_is_off)`. Two values rather than one because they
930/// come from two reads and `axiam_oauth2::sensitive::decide` checks them
931/// against each other; collapsing them here would be collapsing the check.
932///
933/// # Costs nothing for a request that asks for nothing
934///
935/// The first line answers "did this request name `address` or `phone`", which
936/// is a scan of a short vector, and returns before either database read. Every
937/// client registered today asks for neither — they could not, the scopes were
938/// unregistrable — so this function adds one string comparison per scope to
939/// every authorization request in every existing deployment and nothing else.
940/// That is invariant 4 measured rather than asserted.
941///
942/// # A read that fails is not a release
943///
944/// Both reads fail closed and in different directions, which is deliberate:
945/// a settings read that fails reports the switch as **off** (refuse), and a
946/// consent read that fails reports the consent as **missing** (ask again).
947/// Neither answer can release data, and the second is recoverable by the end
948/// user rather than terminal.
949async fn resolve_sensitive_scopes<C: Connection + Clone>(
950    state: &AppState<C>,
951    tenant_id: Uuid,
952    user_id: Uuid,
953    client_id: &str,
954    scope: Option<&str>,
955) -> (axiam_oauth2::sensitive::Requested, bool) {
956    let scopes: Vec<String> = scope
957        .unwrap_or("")
958        .split_whitespace()
959        .map(String::from)
960        .collect();
961    let wanted = axiam_oauth2::sensitive::requested(&scopes);
962    if wanted.is_empty() {
963        return (axiam_oauth2::sensitive::Requested::None, false);
964    }
965
966    // The organization the settings baseline belongs to. Resolved from the
967    // tenant rather than taken from the caller's principal, the same way
968    // `axiam_auth::lockout` resolves it: the authorization endpoint's
969    // principal type carries no `org_id`, and adding one to it would put a
970    // field on the hot path for a branch almost no request takes.
971    let enabled = match state.tenant_repo.get_by_id(tenant_id).await {
972        Ok(tenant) => match axiam_core::repository::SettingsRepository::get_effective_settings(
973            &state.settings_repo,
974            tenant.organization_id,
975            tenant_id,
976        )
977        .await
978        {
979            Ok(settings) => settings.oidc.sensitive_scopes_enabled,
980            Err(e) => {
981                tracing::error!(
982                    error = %e,
983                    %tenant_id,
984                    "could not read the tenant's effective settings while deciding whether the \
985                     address/phone scopes are enabled; treating them as disabled"
986                );
987                false
988            }
989        },
990        Err(e) => {
991            tracing::error!(
992                error = %e,
993                %tenant_id,
994                "could not resolve the organization behind this tenant; treating the \
995                 address/phone scopes as disabled"
996            );
997            false
998        }
999    };
1000    if !enabled {
1001        return (axiam_oauth2::sensitive::Requested::Disabled, true);
1002    }
1003
1004    let consent_type = axiam_oauth2::sensitive::consent_type(client_id);
1005    let version = axiam_oauth2::sensitive::consent_version(&wanted);
1006    let consented = match axiam_core::repository::ConsentRepository::list_by_user(
1007        &state.gdpr.consent_repo,
1008        tenant_id,
1009        user_id,
1010    )
1011    .await
1012    {
1013        Ok(records) => records
1014            .iter()
1015            .any(|c| c.consent_type == consent_type && c.version == version),
1016        Err(e) => {
1017            tracing::error!(
1018                error = %e,
1019                %tenant_id,
1020                "could not read the end user's consent records; treating the release as \
1021                 unconsented, which asks again rather than releasing"
1022            );
1023            false
1024        }
1025    };
1026
1027    (
1028        if consented {
1029            axiam_oauth2::sensitive::Requested::Consented
1030        } else {
1031            axiam_oauth2::sensitive::Requested::ConsentMissing
1032        },
1033        false,
1034    )
1035}

1037/// T21.4 / D4 — resolve whether this authorization needs the external-client
1038/// consent hop.
1039///
1040/// Returns `NotApplicable` for a client an administrator created, which is
1041/// every client in most deployments and is the whole of I1 for this gate: the
1042/// function learns the client's `managed_by` and returns before touching the
1043/// consent repository.
1044///
1045/// # The cost, measured rather than assumed
1046///
1047/// Learning `managed_by` is an indexed read by `client_id` — a read
1048/// `AuthorizeService` repeats for its own validation. It is not avoidable at
1049/// this boundary: the service owns the client lookup and this handler owns
1050/// the consent repository, and the decision needs both. Passing a consent
1051/// repository into the service would put a repository into a crate whose
1052/// whole design is that it decides and does not fetch (see
1053/// `crate::sensitive`'s module docs), and returning early from the service so
1054/// the handler could resolve and re-call would run every validation twice.
1055///
1056/// The read was first shipped uncached, on the argument that a
1057/// browser-interactive endpoint would not feel one more round trip. The
1058/// 2026-09-18 benchmark A/B refuted that: `oauth2_authorize` lost ~19% of its
1059/// throughput to it. So the answer now comes from
1060/// [`crate::client_managed_by_cache`] after the first read — safe because
1061/// `managed_by` can never change for a `(tenant_id, client_id)`; the module
1062/// docs give the argument. Only the one field is cached; the service's own
1063/// read of the client, which validates everything that *can* change, is
1064/// untouched.
1065///
1066/// The **consent** read, which is the more expensive of the two, still
1067/// happens only for a client that is actually external.
1068///
1069/// # Both reads fail closed, in the recoverable direction
1070///
1071/// A client read that fails reports `NotApplicable` — there is no client, so
1072/// the authorization is about to fail anyway inside the service, with the
1073/// error that names the real problem rather than a consent question about a
1074/// client that does not exist. A consent read that fails reports the consent
1075/// as **missing**, which asks again rather than proceeding: the same direction
1076/// `resolve_sensitive_scopes` takes, for the same reason.
1077async fn resolve_external_consent<C: Connection + Clone>(
1078    state: &AppState<C>,
1079    tenant_id: Uuid,
1080    user_id: Uuid,
1081    client_id: &str,
1082    scope: Option<&str>,
1083) -> axiam_oauth2::external_consent::Requested {
1084    use axiam_oauth2::external_consent::Requested;
1085
1086    let managed_by = match state.client_managed_by_cache.get(tenant_id, client_id) {
1087        Some(managed_by) => managed_by,
1088        None => {
1089            let Ok(client) = axiam_core::repository::OAuth2ClientRepository::get_by_client_id(
1090                &state.oauth2_client_repo,
1091                tenant_id,
1092                client_id,
1093            )
1094            .await
1095            else {
1096                // Not cached: a missing client is never remembered.
1097                return Requested::NotApplicable;
1098            };
1099            state
1100                .client_managed_by_cache
1101                .insert(tenant_id, client_id, client.managed_by);
1102            client.managed_by
1103        }
1104    };
1105    if !axiam_oauth2::external_consent::applies_to(managed_by) {
1106        return Requested::NotApplicable;
1107    }
1108
1109    let scopes: Vec<String> = scope
1110        .unwrap_or("")
1111        .split_whitespace()
1112        .map(String::from)
1113        .collect();
1114    let consent_type = axiam_oauth2::external_consent::consent_type(client_id);
1115    let version = axiam_oauth2::external_consent::version(&scopes);
1116
1117    match axiam_core::repository::ConsentRepository::list_by_user(
1118        &state.gdpr.consent_repo,
1119        tenant_id,
1120        user_id,
1121    )
1122    .await
1123    {
1124        Ok(records)
1125            if records
1126                .iter()
1127                .any(|c| c.consent_type == consent_type && c.version == version) =>
1128        {
1129            Requested::Consented
1130        }
1131        Ok(_) => Requested::ConsentMissing,
1132        Err(e) => {
1133            tracing::error!(
1134                error = %e,
1135                %tenant_id,
1136                "could not read the end user's consent records for an externally registered \
1137                 client; treating the client as unconsented, which asks again rather than \
1138                 letting it act as the user"
1139            );
1140            Requested::ConsentMissing
1141        }
1142    }
1143}

1173    // today's refusals is preserved. `AuthenticatedUser` used to be the first
1174    // extractor on this handler, so an unauthenticated request with an
1175    // unparseable query was answered 401 and never reached the body.
1176    // `MaybeAuthenticatedUser` cannot fail, which would have promoted the query
1177    // error to first place and turned that 401 into a 400 — a behaviour change

1200    // never seen. Here, before anything looks the client up, so that everything

1203    // tenant that has enabled the mechanism. It returns nothing and cannot
1204    // refuse the request — see `crate::cimd`.

1210    // `invalid_request`, losing the code OIDC Core §3.1.2.6 defines and a
1211    // conformance suite matches on.
1212    //
1213    // The marker travels on the `AuthorizeRequest` rather than being answered

1225    // merged and not refused.
1226    //
1227    // The distinction is the whole of RFC 9101 §6.3, which RFC 9126 §4 adopts
1228    // by reference ("build an authorization request as defined in [RFC9101]"):
1229    //
1230    //   The authorization server MUST extract the set of authorization request
1231    //   parameters from the Request Object value. The authorization server MUST
1232    //   only use the parameters in the Request Object, even if the same
1233    //   parameter is provided in the query parameter.

1235    // and §5 says in terms that a client MAY send them duplicated. So a
1236    // duplicated `response_type`/`redirect_uri`/`scope`/`code_challenge` is a
1237    // conformant request, and an earlier revision of this handler refused it —
1238    // which failed every FAPI 2.0 authorization module, since the OIDF suite
1239    // sends exactly that shape.
1240    //
1241    // The security argument the refusal was built on is sound and is satisfied
1242    // by ignoring rather than refusing: parameter confusion needs the inline
1243    // value to be *read* by something, and nothing below reads it. Every field

1251    // there is nothing to consume, and the branch exists only to reach the
1252    // validation that decides how the refusal is reported.

1254        Some(_) => None,

1261                .par_service
1262                .consume(user.tenant_id, &q.client_id, &request_uri)
1263                .await
1264            {
1265                Ok(params) => params,
1266                // W3 (plan §4.0, F10). A pushed request lives 60 seconds. A
1267                // user who took longer than that to type a password comes back
1268                // here to a handle that no longer exists — and telling them

1270                // describes a client bug that did not happen. On a return leg,
1271                // and only there, the recoverable code is used instead: the
1272                // relying party pushes again and restarts, which is RFC 9126
1273                // §2.2's own design rather than a workaround for it.
1274                //

1292                }

1298                // `redirect_uri` this client registered, and only then. RFC

1340            // be substitutable by the browser that carries the handle.

1598        .authn_params
1599        .prompt
1600        .contains(&axiam_oauth2::authn_params::Prompt::None);
1601
1602    // W5 — the presentation the sign-in page will be asked for, if it is
1603    // shown. Captured before `req` moves, from the bundle whichever carrier
1604    // delivered it, and used **only** inside the interaction arm below.
1605    //
1606    // There is no honour-lane condition here, and W7 is why it now needs
1607    // saying rather than assuming. W4 produced `AuthorizeOutcome::Interact` at
1608    // exactly one place — step 6b, inside `if fapi::honours_authn_params` — so
1609    // a client on the `ignore` lane could not reach the arm that reads this.
1610    // W7 added a second producer, step 6c's consent gate, which is *not*
1611    // lane-conditional: the tenant switch and the consent record decide it,
1612    // not the client's `authn_request_params`.
1613    //
1614    // That is still invariant 4, and for a reason stronger than a lane check:
1615    // reaching step 6c at all requires the request to carry `address` or
1616    // `phone`, and requires the client to have them registered. No client in
1617    // any existing deployment does — the scopes were unregistrable before this
1618    // wave, which is the whole of I4 for G8. A client that has them was
1619    // registered after W7 shipped, by an operator who turned the tenant switch
1620    // on to do it.
1621    //
1622    // `build_interaction_redirect` then decides which page and which of these
1623    // values it may carry; `login_hint` reaches the consent page from neither
1624    // lane.
1625    //
1626    // Assembled here rather than inside `honour::evaluate` because the cosmetic
1627    // four decide no `Outcome` — see that module's "W5's cosmetic four are not
1628    // here".
1629    //
1630    // W7 supplies the tenant default W5 left as `None`, and reads it only when
1631    // the request carried a bundle: an empty bundle selects nothing whatever
1632    // the tenant prefers, so the read would be a database round trip on every
1633    // authorization request in the deployment to reach the same answer.
1634    let cosmetic_owned = {
1635        let tenant_default = if req.authn_params.is_empty() {
1636            None
1637        } else {
1638            tenant_default_locale(&state, user.tenant_id).await
1639        };
1640        let c = axiam_oauth2::login_hop::Cosmetic::from_params(&req.authn_params, tenant_default);
1641        (c.login_hint.map(str::to_owned), c.display, c.ui_locale)
1642    };
1643
1644    // T21.6 — see `resolve_authorize_principal`'s use of the same value: an
1645    // interaction hop comes back to the endpoint the request arrived at.
1646    let authorize_path = authorize_path_of(&http_req);
1647
1648    let outcome = match state.oauth2.authorize_service.authorize(req).await {
1649        Ok(axiam_oauth2::authorize::AuthorizeOutcome::Interact(interaction)) => {
1650            // W4 — the honour lane asked for an interaction. It rides W3's
1651            // login hop: same `return_to`, same validation on both sides, same
1652            // marker, so the chain is still bounded at one redirect.
1653            //
1654            // `reauth` is set for every reason the *honour lane* interacts —
1655            // `prompt=login`, a `max_age` this session cannot satisfy, a
1656            // step-up, an `id_token_hint` naming somebody else — because each
1657            // is a reason not to trust what the browser already holds; a
1658            // sign-in page that silently reused the current session would
1659            // return the same unsatisfying session and the loop guard would
1660            // have to catch it.
1661            //
1662            // W7 adds the one reason that is not: a consent question. The end
1663            // user is signed in, and what is missing is their answer, not
1664            // their password. `Reason::requires_reauthentication` is where
1665            // that distinction lives, so the two callers of this redirect
1666            // cannot come to disagree about it.
1667            tracing::debug!(
1668                client_id = %authorized_client_id,
1669                reason = ?interaction.reason,
1670                required_acr = ?interaction.required_acr,
1671                "an authorization request on the honour lane needs an interaction"
1672            );
1673            // W7 — a consent hop marks its own return leg. Same `return_to`
1674            // machinery, same validation on both sides, one more marker; see
1675            // `login_hop::CONSENT_HOP_MARKER` for why sharing the login one
1676            // would answer the consent question with the login hop's evidence.
1677            let built = if interaction.reason.requires_reauthentication() {
1678                axiam_oauth2::login_hop::build_return_to_at(
1679                    &authorize_path,
1680                    http_req.query_string(),
1681                )
1682            } else {
1683                axiam_oauth2::login_hop::build_consent_return_to_at(
1684                    &authorize_path,
1685                    http_req.query_string(),
1686                )
1687            };
1688            let Some(return_to) = built else {
1689                // Nothing safe to come back to, so there is nothing to send
1690                // the browser away for. The relying party is told what is
1691                // missing instead.
1692                return build_error_redirect(
1693                    &resolved_redirect_uri,
1694                    &OAuth2Error::LoginRequired(
1695                        "this authorization request needs the end user to authenticate, and \
1696                         could not be resumed after a sign-in"
1697                            .into(),
1698                    ),
1699                    resolved_state.as_deref(),
1700                    &issuer_config(&http_req, &state),
1701                );
1702            };
1703            if !return_to_is_on_this_deployment(&state, &return_to, &authorize_path) {
1704                return build_error_redirect(
1705                    &resolved_redirect_uri,
1706                    &OAuth2Error::LoginRequired(
1707                        "this authorization request needs the end user to authenticate, and \
1708                         could not be resumed after a sign-in"
1709                            .into(),
1710                    ),
1711                    resolved_state.as_deref(),
1712                    &issuer_config(&http_req, &state),
1713                );
1714            }
1715            let (login_hint, display, ui_locale) = &cosmetic_owned;
1716            let cosmetic = axiam_oauth2::login_hop::Cosmetic {
1717                login_hint: login_hint.as_deref(),
1718                display: *display,
1719                ui_locale: *ui_locale,
1720            };
1721            let location = axiam_oauth2::login_hop::build_interaction_redirect(
1722                &return_to,
1723                interaction.reason,
1724                interaction.required_acr,
1725                &cosmetic,
1726            );
1727            return HttpResponse::Found()
1728                .append_header((actix_web::http::header::LOCATION, location))
1729                .append_header((actix_web::http::header::CACHE_CONTROL, "no-store"))
1730                .append_header(("Referrer-Policy", "no-referrer"))
1731                .finish();
1732        }
1733        other => other.map(|o| match o {
1734            axiam_oauth2::authorize::AuthorizeOutcome::Code(resp) => resp,
1735            axiam_oauth2::authorize::AuthorizeOutcome::Interact(_) => {
1736                unreachable!("the interaction arm returned above")
1737            }
1738        }),
1739    };
1740
1741    if silent {
1742        audit_prompt_none(
1743            &state,
1744            &http_req,
1745            user.tenant_id,
1746            &authorized_client_id,
1747            outcome.as_ref().err(),
1748        )
1749        .await;
1750    }
1751
1752    match outcome {
1753        Ok(resp) => {
1754            // B5: the client has just joined this session. Recorded here —
1755            // the moment a code is issued — because that is when
1756            // participation becomes true, and back-channel logout iterates
1757            // exactly these rows. A failure is logged and swallowed: losing a
1758            // logout notification is bad, but failing the login that would
1759            // have created the session is worse.
1760            if let Err(e) = state
1761                .oauth2
1762                .session_client_repo
1763                .record(axiam_core::models::oauth2_client::CreateSessionClient {
1764                    tenant_id: user.tenant_id,
1765                    session_id: user.session_id,
1766                    client_id: authorized_client_id.clone(),
1767                    user_id: user.user_id,
1768                })
1769                .await
1770            {
1771                tracing::warn!(
1772                    error = %e,
1773                    "could not record session participation; back-channel \
1774                     logout will not reach this client"
1775                );
1776            }
1777
1778            match url::Url::parse(&resp.redirect_uri) {
1779                Ok(mut url) => {
1780                    url.query_pairs_mut().append_pair("code", &resp.code);
1781                    if let Some(ref state) = resp.state {
1782                        url.query_pairs_mut().append_pair("state", state);
1783                    }
1784                    append_issuer(&mut url, &issuer_config(&http_req, &state));
1785                    HttpResponse::Found()
1786                        .append_header(("Location", url.to_string()))
1787                        .finish()
1788                }
1789                Err(_) => {
1790                    // Never leak the authorization code to an
1791                    // unknown host.
1792                    HttpResponse::InternalServerError().json(OAuth2ErrorResponse {
1793                        error: "server_error".into(),
1794                        error_description: "invalid redirect_uri in \
1795                                 authorization response"
1796                            .into(),
1797                    })
1798                }
1799            }
1800        }
1801        Err(e) => {
1802            // Per RFC 6749: only redirect when the redirect_uri has
1803            // been validated. For InvalidClient / redirect_uri
1804            // errors, return a direct HTTP error instead.
1805            match &e {
1806                // B5: `ParRequired` joins these two because it is raised
1807                // BEFORE redirect_uri validation — redirecting would bounce
1808                // the user agent to an unvalidated URI that arrived by the
1809                // very channel the setting forbids.
1810                OAuth2Error::InvalidClient(_)
1811                | OAuth2Error::InvalidRedirectUri(_)
1812                | OAuth2Error::ParRequired(_) => authorize_error_response(&http_req, &e),
1813                _ => {
1814                    // These errors occur after client+redirect_uri
1815                    // were validated — safe to redirect.
1816                    build_error_redirect(
1817                        &resolved_redirect_uri,
1818                        &e,
1819                        resolved_state.as_deref(),
1820                        &issuer_config(&http_req, &state),
1821                    )
1822                }
1823            }
1824        }
1825    }
1826}

1828/// `POST /oauth2/token` -- OAuth2 token endpoint.
1829///
1830/// Accepts form-encoded body per RFC 6749. Supports authorization_code,
1831/// client_credentials, and refresh_token grant types. The `tenant_id` is
1832/// passed as a query parameter since the token endpoint is unauthenticated
1833/// (the client is authenticating itself here).
1834#[utoipa::path(
1835    post,
1836    path = "/oauth2/token",
1837    tag = "oauth2",
1838    params(TenantQuery),
1839    request_body(
1840        content_type = "application/x-www-form-urlencoded",
1841        content = TokenRequest,
1842    ),
1843    responses(
1844        (status = 200, description = "Token response",
1845         body = TokenResponse),
1846        (status = 400, description = "OAuth2 error",
1847         body = OAuth2ErrorResponse),
1848    ),
1849)]
1850pub async fn token<C: Connection + Clone>(
1851    req: HttpRequest,
1852    tenant_query: web::Query<TenantQuery>,
1853    form: web::Form<TokenRequest>,
1854    state: web::Data<AppState<C>>,
1855) -> HttpResponse {
1856    // W8 / RFC 6749 §5.2 — the challenge is decided from the request,
1857    // before the handler runs, and applied to whatever 401 it produces.
1858    let challenge = client_auth_challenge(&req);
1859    with_client_auth_challenge(token_inner(req, tenant_query, form, state).await, challenge)
1860}

1862/// The token endpoint proper. Split from [`token`] only so that the RFC 6749
1863/// §5.2 challenge is applied to every one of its exits.
1864async fn token_inner<C: Connection + Clone>(
1865    req: HttpRequest,
1866    tenant_query: web::Query<TenantQuery>,
1867    form: web::Form<TokenRequest>,
1868    state: web::Data<AppState<C>>,
1869) -> HttpResponse {
1870    let tenant_id = tenant_query.into_inner().tenant_id;
1871
1872    let form = form.into_inner();
1873    // I5: keep the grant type for the stage-timing event below. The token
1874    // service moves `form`, so copy the (short) discriminator first.
1875    let grant_type = form.grant_type.clone();
1876    // §17.2 residual 3: same reason, for the failed-client-auth audit event
1877    // below. Never the secret — only the client id, which is the caller's own
1878    // input and is what an operator needs to correlate an attack.
1879    let attempted_client_id = form.client_id.clone();
1880    let started = std::time::Instant::now();
1881
1882    // B2 / RFC 8628 §3.4: the device-code grant is served by its own service.
1883    // Dispatching here, before `TokenService::exchange`, is the single `match`
1884    // arm the device flow costs the rest of the codebase — `TokenService`
1885    // never learns that this grant exists.
1886    //
1887    // Note what is deliberately NOT done: no client authentication. RFC 8628
1888    // targets public clients (a television cannot keep a secret), so the
1889    // device presents only its `device_code`, which is a 256-bit CSPRNG value
1890    // stored as a hash. The device code IS the credential.
1891    if grant_type == DEVICE_CODE_GRANT_TYPE {
1892        let device_code = match form.device_code.as_deref() {
1893            Some(code) if !code.is_empty() => code.to_owned(),
1894            _ => {
1895                return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
1896                    "device_code is required for the device_code grant".into(),
1897                ));
1898            }
1899        };
1900        return match state
1901            .oauth2
1902            .device_authorization_service
1903            .poll_at(
1904                tenant_id,
1905                &device_code,
1906                form.resource.as_deref(),
1907                // T21.6. The device-code grant runs above the point where a
1908                // `TokenRequestContext` is built (RFC 8628 performs no client
1909                // authentication), so the binding is read directly here.
1910                crate::middleware::tenant_path::binding_of(&req)
1911                    .map(|binding| binding.issuer().to_owned())
1912                    .as_deref(),
1913            )
1914            .await
1915        {
1916            Ok(resp) => HttpResponse::Ok()
1917                .append_header(("Cache-Control", "no-store"))
1918                .append_header(("Pragma", "no-cache"))
1919                .json(resp),
1920            // `authorization_pending` and `slow_down` are the NORMAL answers
1921            // here — a device polls for as long as the user takes to walk to
1922            // another screen — so they are not logged as failures. Treating
1923            // the expected path as an error condition would bury the real
1924            // ones under it.
1925            Err(e) => build_oauth2_error_response(&e),
1926        };
1927    }
1928
1929    // T21.5 — before client authentication, for the reason the authorize
1930    // endpoint calls it before the client lookup: what follows must act on an
1931    // ordinary row. Placed after the device-code branch because RFC 8628's
1932    // grant authenticates no client at all and carries no `client_id` worth
1933    // resolving.
1934    if let Some(client_id) = form.client_id.as_deref() {
1935        crate::cimd::materialise_if_cimd(&state, &req, tenant_id, client_id).await;
1936    }
1937
1938    // SEC-096: ONE context construction and ONE DPoP verification, ahead of
1939    // every grant that authenticates a client. Before this, the
1940    // token-exchange and uma-ticket grants each `return`ed above this point,
1941    // so a DPoP proof on those requests was never verified, never recorded as
1942    // single-use, and never available to the binding decision — which meant a
1943    // client holding a `cnf`-bound token could exchange it for a plain bearer
1944    // one, and a `fapi2` client could obtain an unconstrained token from a
1945    // grant `fapi::enforce_token_request` never saw.
1946    //
1947    // The device-code grant stays above this line deliberately: RFC 8628
1948    // performs no client authentication at all, so there is no registration
1949    // in hand for the profile gate to enforce and no party for a proof to
1950    // bind to.
1951    let ctx = match token_request_context(&req) {
1952        Ok(ctx) => ctx.with_assertion_from(&form),
1953        Err(response) => return *response,
1954    };
1955    let ctx = match dpop_from_request(&req, &state, tenant_id, ctx).await {
1956        Ok(ctx) => ctx,
1957        Err(response) => return *response,
1958    };
1959
1960    // T21.2 — the public-client bucket.
1961    //
1962    // Every request to this endpoint already passes the endpoint's own
1963    // rate limit, keyed by `AXIAM__RATE_LIMIT__KEY` (`ip` by default,
1964    // `client_id` or `ip_client_id` on a NAT'd fleet — see
1965    // `config::rate_limit`). This is a second, narrower bucket for the one
1966    // request shape T21.2 makes reachable: a token request carrying **no
1967    // client credential at all**.
1968    //
1969    // Keyed on `client_id` AND the transport peer address, never on either
1970    // alone. `client_id` alone would be a bucket an unauthenticated caller
1971    // mints by rotating a value they choose (`KeyMintability::Full`); the peer
1972    // address alone would put every public client behind one NAT gateway —
1973    // every developer in an office, every desktop client on one residential
1974    // connection — into a single allowance. The pair is what the
1975    // `ip_client_id` mode was added for, applied here unconditionally because
1976    // for a credential-less request it is the only defensible key.
1977    //
1978    // Decidable from the request alone, so it costs no lookup and creates no
1979    // client-existence oracle: it applies equally to a real public client, an
1980    // unknown `client_id`, and a confidential client whose caller forgot the
1981    // secret. All three were refused `invalid_client` before T21.2 and two of
1982    // them still are — the only thing that changes for them is that a caller
1983    // hammering the endpoint with no credential now gets `slow_down` first.
1984    // `token_per_min` is the limit, deliberately reusing the endpoint's own
1985    // number rather than adding a knob: a public client's token request is a
1986    // token request, and an operator who has retuned one has retuned both.
1987    if ctx.carries_no_client_credential(form.client_secret.as_deref()) {
1988        let bucket = format!(
1989            "oauth2_token_public:{}:{}:{}",
1990            tenant_id,
1991            form.client_id.as_deref().unwrap_or("-"),
1992            peer_ip(&req).unwrap_or_else(|| "-".into()),
1993        );
1994        if !state.shared_rate_limit.check_at(
1995            &bucket,
1996            chrono::Utc::now(),
1997            state.rate_limit_cfg.token_per_min,
1998        ) {
1999            return HttpResponse::TooManyRequests()
2000                .append_header(("Cache-Control", "no-store"))
2001                .json(OAuth2ErrorResponse {
2002                    error: "slow_down".into(),
2003                    error_description: "token rate limit exceeded for this client and address"
2004                        .into(),
2005                });
2006        }
2007    }
2008
2009    // B3 / RFC 8693. Like the device grant, its own service behind one match
2010    // arm. Unlike the device grant, the exchanging client DOES authenticate —
2011    // it is a confidential service, not a television, and an exchange is
2012    // precisely the operation that should be attributable.
2013    if grant_type == TOKEN_EXCHANGE_GRANT_TYPE {
2014        // SEC-093: the transport context travels with the grant, so a client
2015        // registered for `tls_client_auth` / `private_key_jwt` is
2016        // authenticated by the credential its registration names rather than
2017        // by the secret it was also, unavoidably, issued.
2018        return handle_token_exchange(tenant_id, form, &state, &ctx).await;
2019    }
2020
2021    // X2 / UMA 2.0 §3.3.1. Third grant dispatched here for the same structural
2022    // reason as the other two: the body is deserialized once, so the grant is
2023    // only knowable after `grant_type` has been read out of it.
2024    if grant_type == UMA_TICKET_GRANT_TYPE {
2025        // The authz checker is pulled from the request here rather than taken
2026        // as a handler parameter on purpose. `/oauth2/token` serves five
2027        // grants, and only this one evaluates anything against the RBAC
2028        // engine; making it an extractor would make the checker a hard
2029        // requirement of the whole endpoint, so a deployment (or a test
2030        // harness) that never uses UMA would start answering 500 to ordinary
2031        // token requests because of a dependency none of them use.
2032        let Some(authz) = req.app_data::<crate::authz::AuthzData>().cloned() else {
2033            return build_oauth2_error_response(&OAuth2Error::ServerError(
2034                "the uma-ticket grant requires an authorization checker".into(),
2035            ));
2036        };
2037        // SEC-093, as for the token-exchange grant above.
2038        return handle_uma_ticket(tenant_id, form, &state, &authz, &ctx).await;
2039    }
2040
2041    match state
2042        .oauth2
2043        .token_service
2044        .exchange(tenant_id, form, &ctx)
2045        .await
2046    {
2047        Ok(resp) => {
2048            let exchange_us = started.elapsed().as_micros() as u64;
2049
2050            // I5: serialize explicitly rather than via `.json()` so the
2051            // serialization cost and the exact response body size are
2052            // observable. Body size is the load-bearing number for the
2053            // Nagle/delayed-ACK hypothesis behind the TLS client-credentials
2054            // plateau: the pathology only fires when a response reaches the
2055            // socket as a full segment plus a small trailing fragment, so run 5
2056            // needs to know where this body sits relative to the path MSS
2057            // (1448 bytes on a standard 1500-byte-MTU docker veth) once TLS
2058            // record and HTTP/2 frame overhead are added.
2059            let t_serialize = std::time::Instant::now();
2060            let body = match serde_json::to_vec(&resp) {
2061                Ok(body) => body,
2062                Err(e) => {
2063                    tracing::error!(error = %e, "failed to serialize token response");
2064                    return HttpResponse::InternalServerError().finish();
2065                }
2066            };
2067            let serialize_us = t_serialize.elapsed().as_micros() as u64;
2068
2069            tracing::debug!(
2070                target: "axiam::perf",
2071                stage = "oauth2.token_endpoint",
2072                %grant_type,
2073                exchange_us,
2074                serialize_us,
2075                response_body_bytes = body.len(),
2076                handler_total_us = started.elapsed().as_micros() as u64,
2077                "token endpoint stage timings (I5)"
2078            );
2079
2080            HttpResponse::Ok()
2081                .append_header(("Cache-Control", "no-store"))
2082                .append_header(("Pragma", "no-cache"))
2083                .content_type("application/json")
2084                .body(body)
2085        }
2086        Err(e) => {
2087            if matches!(e, OAuth2Error::InvalidClient(_)) {
2088                // §22.3 residual 1: this used to be awaited, which put a tenant
2089                // read (and, when it succeeded, an insert) on the response path
2090                // — so a failed client auth returned measurably faster for a
2091                // NONEXISTENT tenant than a real one, on an unauthenticated
2092                // endpoint. Detaching it removes the differential at its source
2093                // rather than trying to balance the two branches, and makes the
2094                // "fire-and-forget" this function has always claimed actually
2095                // true. The response below no longer waits on the audit sink at
2096                // all.
2097                let tenant_repo = state.tenant_repo.clone();
2098                let audit_repo = state.audit_repo.clone();
2099                let attempted = attempted_client_id.clone();
2100                let grant = grant_type.clone();
2101                let peer = peer_ip(&req);
2102                let forwarded = client_ip(&req);
2103                actix_web::rt::spawn(async move {
2104                    append_client_auth_failure_audit(
2105                        &tenant_repo,
2106                        &audit_repo,
2107                        tenant_id,
2108                        attempted.as_deref(),
2109                        &grant,
2110                        peer,
2111                        forwarded,
2112                    )
2113                    .await;
2114                });
2115            }
2116            build_oauth2_error_response(&e)
2117        }
2118    }
2119}

2121/// Build the token endpoint's transport context from the connection (X5.1).
2122///
2123/// The certificate comes from [`VerifiedClientCert`], which `axiam-server`'s
2124/// `on_connect` hook populates from the peer chain **rustls verified during
2125/// the TLS 1.3 handshake**. Nothing else is consulted: in particular the
2126/// `X-Client-Certificate` header that `CertificateAuthenticated` accepts as a
2127/// proxy fallback is deliberately not read here, because an OAuth2 client
2128/// credential must not be assertable by anything that can set a header. See
2129/// `axiam_oauth2::mtls`'s module docs for the full argument.
2130///
2131/// Cost on the ordinary path — a connection with no client certificate, which
2132/// is every non-mTLS deployment — is one `conn_data` lookup that misses and no
2133/// allocation at all. On an mTLS connection it is one copy of the DER plus one
2134/// SHA-256 over it. The X.509 *parse* is deliberately not done here: it is
2135/// deferred to `PresentedCertificate::identity`, which only the
2136/// `tls_client_auth` branch calls, so a deployment running mTLS with ordinary
2137/// secret-authenticating clients does not pay for a DN nobody reads.
2138fn token_request_context(req: &HttpRequest) -> Result<TokenRequestContext, Box<HttpResponse>> {
2139    // W8 — RFC 6749 §2.3.1. The header is read here and decoded by
2140    // `axiam_oauth2::client_secret_basic`, so this crate owns "where the bytes
2141    // came from" and that crate owns "what they mean". A malformed header is
2142    // an error rather than an absence, for the same reason a bad DPoP proof is:
2143    // treating it as "no credentials presented" would make a corrupt Basic
2144    // header silently equivalent to sending none.
2145    let basic = basic_credentials_from_request(req)?;
2146
2147    // The trust level travels with the certificate rather than being inferred
2148    // here: `authenticate_mtls_client` refuses a self-asserted certificate for
2149    // `tls_client_auth` (§2.1) and accepts one for `self_signed_tls_client_auth`
2150    // (§2.2), and this is the seam that lets it tell them apart.
2151    let certificate = req
2152        .conn_data::<VerifiedClientCert>()
2153        .map(|verified| PresentedCertificate::from_der(&verified.der, verified.trust));
2154
2155    Ok(TokenRequestContext {
2156        client_certificate: certificate,
2157        // T21.6 — the issuer this request arrived under, read from the scope
2158        // middleware's binding and therefore from the PATH, never from the
2159        // form. `None` on `/oauth2/token`, which is every request on a
2160        // deployment with `AXIAM__AUTH__TENANT_ISSUER_PATHS` unset.
2161        issuer: crate::middleware::tenant_path::binding_of(req)
2162            .map(|binding| binding.issuer().to_owned()),
2163        ..TokenRequestContext::default()
2164    }
2165    .with_basic_credentials(basic))
2166}

2168/// Decode an `Authorization: Basic` header, if one is present (W8, RFC 6749
2169/// §2.3.1).
2170///
2171/// `Ok(None)` — no `Authorization` header, or one naming a different scheme.
2172/// Not an error: four of AXIAM's five client-authentication methods put
2173/// nothing in this header.
2174///
2175/// `Err(_)` — the client *did* attempt Basic authentication and got the
2176/// encoding wrong. Answered with `invalid_client`, uniform with every other
2177/// client-authentication failure (SEC-086): the four distinguishable causes in
2178/// [`BasicAuthError`] are a `debug!` for the operator and nothing at all for
2179/// the caller. RFC 6749 §5.2's `WWW-Authenticate: Basic` challenge is attached
2180/// by [`with_client_auth_challenge`] on the way out, so it is impossible for
2181/// one of these endpoints to answer a Basic attempt with a `Bearer` challenge.
2182///
2183/// A second `Authorization` header is refused rather than resolved. Taking the
2184/// first would let an intermediary that appends a header decide which
2185/// credential authenticates — the request-smuggling shape `dpop_from_request`
2186/// refuses for the same reason.
2187fn basic_credentials_from_request(
2188    req: &HttpRequest,
2189) -> Result<Option<BasicCredentials>, Box<HttpResponse>> {
2190    use actix_web::http::header::AUTHORIZATION;
2191
2192    let Some(raw) = req
2193        .headers()
2194        .get(AUTHORIZATION)
2195        .and_then(|v| v.to_str().ok())
2196    else {
2197        return Ok(None);
2198    };
2199
2200    match axiam_oauth2::client_secret_basic::parse_authorization_header(raw) {
2201        // Not a Basic attempt. Nothing is refused here, including a request
2202        // carrying several `Authorization` headers of some other scheme: the
2203        // duplicate-header rule below exists to stop an intermediary choosing
2204        // between two *credentials*, and this endpoint reads no credential
2205        // from any other scheme. Refusing more widely would change the
2206        // behaviour of requests no client sends today, which is I4's whole
2207        // point.
2208        None => Ok(None),
2209        Some(Ok(credentials)) => {
2210            // RFC 9110 §5.2 lets a recipient combine repeated field lines, and
2211            // `Authorization` is not a list-valued field — so two of them is a
2212            // malformed request, not a choice of credentials. Taking the first
2213            // would let an intermediary that appends a header decide which
2214            // credential authenticates, the request-smuggling shape
2215            // `dpop_from_request` refuses for the same reason.
2216            if req.headers().get_all(AUTHORIZATION).count() > 1 {
2217                tracing::debug!(
2218                    "more than one Authorization header on a request attempting Basic client \
2219                     authentication; refusing rather than choosing which credential counts"
2220                );
2221                return Err(Box::new(build_oauth2_error_response(
2222                    &OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()),
2223                )));
2224            }
2225            Ok(Some(credentials))
2226        }
2227        Some(Err(cause)) => {
2228            // The cause, never the header. `BasicAuthError` is four unit
2229            // variants and carries no part of the credential, which is why it
2230            // is safe to log at all.
2231            tracing::debug!(
2232                ?cause,
2233                "an Authorization header using the Basic scheme could not be decoded (RFC 6749 \
2234                 §2.3.1)"
2235            );
2236            Err(Box::new(build_oauth2_error_response(
2237                &OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()),
2238            )))
2239        }
2240    }
2241}

2243/// The `WWW-Authenticate` challenge this request must be answered with if it
2244/// ends in a 401 (RFC 6749 §5.2).
2245///
2246/// The RFC's rule is that the challenge names *the scheme the client used*, so
2247/// it is a property of the request and is computed before the handler runs —
2248/// which is also why it survives every early return inside one.
2249/// `None` means "the client did not authenticate through the `Authorization`
2250/// header", and the endpoint's existing `Bearer realm="axiam"` stands.
2251fn client_auth_challenge(req: &HttpRequest) -> Option<&'static str> {
2252    use actix_web::http::header::AUTHORIZATION;
2253    let raw = req.headers().get(AUTHORIZATION)?.to_str().ok()?;
2254    // Deliberately keyed on the *scheme*, not on whether the credentials
2255    // decoded: a client that sent a corrupt Basic header still used Basic, and
2256    // RFC 6749 §5.2 says the challenge must tell it so.
2257    axiam_oauth2::client_secret_basic::names_basic_scheme(raw)
2258        .then_some(axiam_oauth2::client_secret_basic::BASIC_CHALLENGE)
2259}

2261/// Replace the `WWW-Authenticate` challenge on a 401 with the scheme the
2262/// client actually used (RFC 6749 §5.2).
2263///
2264/// Applied at the four client-authenticating endpoints' single exit rather
2265/// than at each of the thirty-odd `build_oauth2_error_response` call sites
2266/// inside them. That placement is the point: a 401 raised by a path nobody
2267/// thought about while writing this still gets the right challenge.
2268///
2269/// Nothing else about the response is touched, and a non-401 is returned
2270/// unchanged — a challenge on a 200 or a 400 would be a protocol error of its
2271/// own.
2272fn with_client_auth_challenge(
2273    mut resp: HttpResponse,
2274    challenge: Option<&'static str>,
2275) -> HttpResponse {
2276    let Some(challenge) = challenge else {
2277        return resp;
2278    };
2279    if resp.status() != actix_web::http::StatusCode::UNAUTHORIZED {
2280        return resp;
2281    }
2282    resp.headers_mut().insert(
2283        actix_web::http::header::WWW_AUTHENTICATE,
2284        actix_web::http::header::HeaderValue::from_static(challenge),
2285    );
2286    resp
2287}

2289/// The `htu` a DPoP proof must name for this request (SEC-102, RFC 9449 §4.3
2290/// step 9).
2291///
2292/// Built from the **configured** issuer plus the request's path. RFC 9449 says
2293/// `htu` is compared against "the HTTP URI used for the request, without query
2294/// and fragment parts" — meaning the server's own view of its own endpoint.
2295/// `HttpRequest::full_url()` does not give that: it takes the authority from
2296/// `ConnectionInfo`, which prefers `Forwarded`, then `X-Forwarded-Host`, then
2297/// `Host`, and this deployment has no trusted-proxy layer that normalises any
2298/// of the three. With the authority sourced from the request, a caller chooses
2299/// *both* sides of the comparison and the check degrades to "the proof names
2300/// some authority consistently" — so a client phished into signing a proof for
2301/// `https://attacker.example/oauth2/token` could have it accepted here by
2302/// sending a matching `Host`.
2303///
2304/// The path still comes from the request, and must: it is what keeps a
2305/// `/oauth2/par` proof from being replayed at `/oauth2/token`. `req.path()` is
2306/// actix's already-normalised path and carries neither query nor fragment,
2307/// which is exactly the string the RFC compares.
2308///
2309/// `effective_issuer()` is trimmed of any trailing slash by `AuthConfig`, and
2310/// `req.path()` always begins with one, so the join never doubles it.
2311fn dpop_htu<C: Connection + Clone>(state: &AppState<C>, req: &HttpRequest) -> String {
2312    format!("{}{}", state.auth_config.effective_issuer(), req.path())
2313}

2321/// Fold a verified `DPoP` proof into the token-endpoint context (X5.1).
2322///
2323/// All of the verification is [`verify_dpop_header`]; this is the one line of

2325/// which needs the proof's thumbprint but has no `TokenRequestContext` to put
2326/// it in — reaches the same verification rather than a second copy of it.
2327async fn dpop_from_request<C: Connection + Clone>(
2328    req: &HttpRequest,
2329    state: &AppState<C>,
2330    tenant_id: Uuid,
2331    mut ctx: TokenRequestContext,
2332) -> Result<TokenRequestContext, Box<HttpResponse>> {

2334        return Ok(ctx);
2335    };
2336    ctx.dpop_proof = Some(verified);
2337    Ok(ctx)
2338}

2340/// Verify the `DPoP` header, if one is present (X5.1, RFC 9449 §4.3).
2341///
2342/// `Ok(None)` means no header arrived. `Err` means one arrived and did not
2343/// verify, and carries the response to send.
2344///
2345/// Two properties this function exists to hold, both of which are easy to lose
2346/// by writing the obvious thing instead:
2347///
2348/// 1. **A proof that fails verification is an error, not an absence.** Returning
2349///    `Ok(None)` for a *bad* proof would be silently equivalent to not sending
2350///    one — so a client that sent a forged proof would get whatever an unbound
2351///    client gets. The `Err` arm is what stops that.
2352/// 2. **An absent header is not an error here.** Whether this particular client
2353///    needed a proof is `fapi::enforce_token_request`'s question and
2354///    `certificate_binding_for`'s, both of which read the *registration*. Making
2355///    the decision here would require loading the client twice, and would put
2356///    the "does this client need a proof" rule in two places.
2357///
2358/// The `jti` is recorded on success, through the same `UNIQUE`-index guard the
2359/// client-assertion path uses. A proof whose `jti` cannot be recorded is
2360/// refused rather than accepted: failing open would turn a database blip into
2361/// an unlimited replay window.
2362///
2363/// Shared with `/oauth2/par` since RFC 9449 §10.1, which requires the PAR
2364/// endpoint to check a proof "as defined in Section 4.3" — the same check,
2365/// with `htu` naming the PAR path because [`dpop_htu`] reads it from the
2366/// request.
2367async fn verify_dpop_header<C: Connection + Clone>(
2368    req: &HttpRequest,
2369    state: &AppState<C>,
2370    tenant_id: Uuid,
2371) -> Result<Option<axiam_oauth2::dpop::VerifiedDpopProof>, Box<HttpResponse>> {
2372    use axiam_core::repository::{ProofKind, ProofReplayRepository};
2373    use axiam_oauth2::dpop::{self, DpopExpectation};
2374
2375    let Some(raw) = req
2376        .headers()
2377        .get(DPOP_HEADER)
2378        .and_then(|v| v.to_str().ok())
2379        .map(str::trim)
2380        .filter(|v| !v.is_empty())
2381    else {
2382        return Ok(None);
2383    };
2384
2385    // RFC 9449 §4.3 step 2: more than one DPoP header is a malformed request,
2386    // not a choice of proofs. Refusing rather than taking the first is what
2387    // stops a request-smuggling intermediary from deciding which proof counts.
2388    if req.headers().get_all(DPOP_HEADER).count() > 1 {
2389        return Err(dpop_error_response(
2390            "invalid_dpop_proof",
2391            "a request may carry at most one DPoP header",
2392            None,
2393        ));
2394    }
2395
2396    // SEC-102: the authority comes from the CONFIGURED issuer, never from the
2397    // request. `HttpRequest::full_url()` builds it from `ConnectionInfo`,
2398    // which prefers `Forwarded`, then `X-Forwarded-Host`, then `Host` — all
2399    // three request-controlled. Sourcing it there let the caller choose both
2400    // sides of RFC 9449 §4.3 step 9's comparison, reducing the `htu` check to
2401    // "the proof names *some* authority consistently". Only the path is taken
2402    // from the request, and only `req.path()`, which actix has already
2403    // normalised and which carries no query or fragment — exactly what the
2404    // RFC compares against.
2405    let htu = dpop_htu(state, req);
2406    let expect = DpopExpectation {
2407        htm: req.method().as_str(),
2408        htu: &htu,
2409        // SEC-097: `expected_nonce` stays `None` because this deployment
2410        // stores no per-client nonce to compare against — see
2411        // `dpop_nonce_required_for`, which is what decides whether a nonce is
2412        // demanded at all, and `docs/security-profiles.md` for what v1's
2413        // nonce challenge does and does not prove.
2414        expected_nonce: None,
2415        require_nonce: false,
2416        access_token: None,
2417        now: chrono::Utc::now().timestamp(),
2418        max_age_secs: dpop::DEFAULT_PROOF_MAX_AGE_SECS,
2419    };
2420
2421    let verified = match dpop::verify_dpop_proof(raw, &expect) {
2422        Ok(v) => v,
2423        Err(e) if e.is_nonce_challenge() => {
2424            return Err(dpop_error_response(
2425                e.error_code(),
2426                "a DPoP nonce is required",
2427                Some(dpop::new_nonce()),
2428            ));
2429        }
2430        Err(e) => {
2431            tracing::debug!(error = %e, "DPoP proof verification failed at the token endpoint");
2432            return Err(dpop_error_response(
2433                e.error_code(),
2434                "the DPoP proof could not be verified",
2435                None,
2436            ));
2437        }
2438    };
2439
2440    match state
2441        .oauth2
2442        .proof_replay_repo
2443        .insert_proof_jti(
2444            tenant_id,
2445            ProofKind::DpopProof,
2446            &verified.jkt,
2447            &verified.jti,
2448            verified.replay_expiry(dpop::DEFAULT_PROOF_MAX_AGE_SECS),
2449        )
2450        .await
2451    {
2452        Ok(()) => {}
2453        Err(axiam_core::error::AxiamError::ReplayDetected) => {
2454            tracing::warn!(jkt = %verified.jkt, "a DPoP proof was replayed; refusing");
2455            return Err(dpop_error_response(
2456                "invalid_dpop_proof",
2457                "this DPoP proof has already been used",
2458                None,
2459            ));
2460        }
2461        Err(e) => {
2462            tracing::error!(
2463                error = %e,
2464                "could not record a DPoP proof's jti; refusing rather than accepting a proof \
2465                 that cannot be made single-use"
2466            );
2467            return Err(dpop_error_response(
2468                "invalid_dpop_proof",
2469                "the DPoP proof could not be verified",
2470                None,
2471            ));
2472        }
2473    }
2474
2475    Ok(Some(verified))
2476}

2502/// Append an `oauth2.client_auth_failed` audit entry, fire-and-forget.
2503///
2504/// §17.2 residual 3: a failed client authentication at the token endpoint used
2505/// to accrue **nothing** — no counter, no lockout, no audit row, no metric —
2506/// and `/oauth2` is not covered by the audit middleware, so nothing else
2507/// filled the gap either. Contrast the interactive login path, which records
2508/// every failure and applies exponential backoff.
2509///
2510/// This closes the **detection** half only, which is the half that was
2511/// genuinely missing. Lockout is deliberately *not* added here: client
2512/// credentials are 256-bit CSPRNG values, so online guessing is not the threat
2513/// — and a lockout keyed on a caller-supplied `client_id` would hand any
2514/// unauthenticated party a denial-of-service against a known client. The
2515/// endpoint's existing rate limit remains the volumetric control.
2516///
2517/// Recorded with `actor_id = Uuid::nil()` and [`ActorType::System`] because
2518/// the caller is by definition unauthenticated: the presented `client_id`
2519/// either does not exist or did not prove possession of its secret, so it must
2520/// not be promoted into the actor field as though it had. It goes in the
2521/// metadata instead, where it is evidence rather than identity.
2522///
2523/// The secret is never recorded, in any form.
2524///
2525/// # SEC-087 — this endpoint is unauthenticated, so nothing here may be trusted
2526///
2527/// `/oauth2/token` takes its `tenant_id` from a **query parameter** and requires
2528/// no credential to reach. The first version of this function wrote the row
2529/// unconditionally, which let any anonymous caller append rows into an arbitrary
2530/// tenant's append-only log — or into a tenant id that never existed — at
2531/// request rate. Three separate controls apply here as a result:
2532///
2533/// 1. **The tenant is resolved before the write**, and a row that cannot be
2534///    attributed to a real tenant goes to the **system partition**
2535///    (`tenant_id = nil`, the one [`AuditLogRepository::list_system`] serves)
2536///    rather than to a caller-chosen id. This bounds what an anonymous caller
2537///    can steer: they may cause a row, but never choose which real tenant's
2538///    append-only log it lands in. It does *not* stop a caller from generating
2539///    failures against a real tenant they do not belong to — and deliberately
2540///    doesn't try to, because that event is **true**: a failed client
2541///    authentication did occur against that tenant, and suppressing it would
2542///    blind the operator to a real attack. The volumetric control is the
2543///    endpoint's rate limit, not the audit layer.
2544///
2545///    §22.3 residual 2: an earlier version *dropped* the row whenever the
2546///    tenant read failed for any reason other than `NotFound`. The reasoning
2547///    (a DB outage must not turn a 401 into a 500) was right, but the effect
2548///    was that a database fault cost exactly the telemetry this row exists to
2549///    provide — and a database fault is plausibly correlated with an incident.
2550///    Routing to the system partition keeps the signal in precisely the case
2551///    where losing it hurts most, while still refusing to guess a tenant.
2552/// 2. **The client id is truncated on a character boundary, by bytes.**
2553///    §22.3 residual 3: the cap was applied with `chars().take(N)`, so a
2554///    multibyte id still stored up to 4N bytes. It is caller-supplied and
2555///    otherwise unbounded, unlike the `client_ip`/`user_agent` helpers which
2556///    have always capped their inputs (and which share the same char-vs-byte
2557///    looseness, bounded there by much smaller limits).
2558/// 3. **The IP is recorded at two trust levels.** `ip_address` carries the
2559///    transport peer address, which a caller cannot forge. The
2560///    `X-Forwarded-For`-derived value goes to metadata under a name that says
2561///    it is untrusted: on an authenticated path behind a trusted proxy the
2562///    forwarded value is the useful one, but here anyone can assert it, and an
2563///    operator who blocks a forged IP has been made to act against the wrong
2564///    host.
2565async fn append_client_auth_failure_audit<C: Connection + Clone>(
2566    tenant_repo: &SurrealTenantRepository<C>,
2567    audit_repo: &SurrealAuditLogRepository<C>,
2568    tenant_id: Uuid,
2569    attempted_client_id: Option<&str>,
2570    grant_type: &str,
2571    peer_ip: Option<String>,
2572    forwarded_ip_untrusted: Option<String>,
2573) {
2574    // Control 1 — attribute the row to the request's tenant only when that
2575    // tenant demonstrably exists; otherwise fall back to the system partition.
2576    // `attributed_tenant` is nil in both the unknown and the indeterminate
2577    // case, so a caller can never place a row in a real tenant's log by naming
2578    // one.
2579    let (attributed_tenant, tenant_note) = match tenant_repo.get_by_id(tenant_id).await {
2580        Ok(_) => (tenant_id, None),
2581        Err(AxiamError::NotFound { .. }) => {
2582            tracing::warn!(
2583                claimed_tenant_id = %tenant_id,
2584                %grant_type,
2585                "oauth2: client-auth-failure audit routed to the system partition — unknown tenant (SEC-087)"
2586            );
2587            (Uuid::nil(), Some("unknown_tenant"))
2588        }
2589        Err(e) => {
2590            // A DB fault must not turn a 401 into a 500 (T-15-04), and must not
2591            // cost the telemetry either (§22.3 residual 2). Record it as
2592            // unattributed rather than dropping it.
2593            tracing::error!(
2594                error = %e,
2595                claimed_tenant_id = %tenant_id,
2596                "oauth2: tenant lookup failed for client-auth-failure audit; \
2597                 routing to the system partition"
2598            );
2599            (Uuid::nil(), Some("tenant_lookup_failed"))
2600        }
2601    };
2602
2603    // Control 2 — bound the one remaining caller-controlled string, by BYTES,
2604    // truncating on a character boundary so the stored value stays valid UTF-8.
2605    let attempted_client_id: Option<String> =
2606        attempted_client_id.map(truncate_bytes_on_char_boundary);
2607
2608    if let Err(e) = audit_repo
2609        .append(CreateAuditLogEntry {
2610            tenant_id: attributed_tenant,
2611            actor_id: Uuid::nil(),
2612            actor_type: ActorType::System,
2613            action: "oauth2.client_auth_failed".into(),
2614            resource_id: None,
2615            outcome: AuditOutcome::Failure,
2616            // Control 3 — the unforgeable half.
2617            ip_address: peer_ip,
2618            metadata: Some(serde_json::json!({
2619                "client_id": attempted_client_id,
2620                "grant_type": grant_type,
2621                // Named so nobody downstream mistakes it for verified input.
2622                "forwarded_for_untrusted": forwarded_ip_untrusted,
2623                // Present only on a system-partition row: the tenant the caller
2624                // named, and why it could not be attributed. Also caller-supplied.
2625                "claimed_tenant_id_untrusted": tenant_note.map(|_| tenant_id.to_string()),
2626                "unattributed_reason": tenant_note,
2627            })),
2628        })
2629        .await
2630    {
2631        // Never propagated: a token-endpoint 401 must not become a 500 because
2632        // the audit sink is unavailable (T-15-04).
2633        tracing::error!(
2634            error = %e,
2635            tenant_id = %attributed_tenant,
2636            %grant_type,
2637            "oauth2: failed to write oauth2.client_auth_failed audit log"
2638        );
2639    }
2640}

2939/// Build and serve a discovery document.
2940///
2941/// `tenant_issuer` is `Some` only for the three T21.6 tenant-path routes. It
2942/// changes exactly two things and nothing else:
2943///
2944/// * the document's `issuer`, and with it every endpoint URL, which is derived
2945///   from it;
2946/// * whether the endpoints carry `?tenant_id=` — they do not under a tenant
2947///   issuer, because the tenant is already in the path, and an issuer with a
2948///   query is not an issuer RFC 8414 §2 allows.
2949///
2950/// Everything else — the capability members, the sensitive-scope rows, the
2951/// mTLS aliases, the error posture — is the same code answering the same way,
2952/// which is what the "the three forms agree" test asserts.
2953async fn discovery_document<C: Connection + Clone>(
2954    described_tenant: Option<Uuid>,
2955    tenant_issuer: Option<String>,
2956    state: &web::Data<AppState<C>>,
2957) -> HttpResponse {
2958    let auth_config = &state.auth_config;
2959    let issuer = tenant_issuer
2960        .as_deref()
2961        .unwrap_or_else(|| auth_config.effective_issuer());
2962    // Guard: effective_issuer must be a valid URL for a compliant
2963    // discovery document.  Startup validation should catch this, but
2964    // defend in depth at the endpoint level.
2965    if url::Url::parse(issuer).is_err() {
2966        return HttpResponse::InternalServerError().json(OAuth2ErrorResponse {
2967            error: "server_error".into(),
2968            error_description: "OIDC issuer is not configured as a \
2969                valid URL"
2970                .into(),
2971        });
2972    }
2973    // A misconfigured mTLS base URL fails the request rather than dropping the
2974    // aliases. Serving the document without them would tell an mTLS client the
2975    // conventional endpoints are the ones to use — a silent downgrade to
2976    // exactly the topology RFC 8705 §5 exists to steer it away from — and it
2977    // would look, from the client's side, indistinguishable from a deployment
2978    // that has no mTLS host at all.
2979    // W7 / X7 G8 — whether this tenant may use the sensitive scopes at all.
2980    // `false` when no tenant was named, which is every call made before this
2981    // wave: the document such a caller receives is byte-identical to the one
2982    // W6 served. See `build_discovery_document_for` for why omitting is the
2983    // truthful answer rather than the cautious one.
2984    // The tenant this document describes: the one the caller named, else the
2985    // deployment's configured default (`AXIAM__AUTH__OAUTH2_DEFAULT_TENANT_ID`,
2986    // unset on a multi-tenant deployment). `None` reproduces the pre-existing
2987    // document exactly, endpoint for endpoint.
2988    //
2989    // ONE value for both halves of the document, deliberately: the endpoints it
2990    // publishes and the scopes it advertises must describe the same tenant. A
2991    // document offering `address` for tenant X while pointing its token
2992    // endpoint at tenant Y would be internally inconsistent in a way no client
2993    // could detect and every client would trust.
2994    let described_tenant = described_tenant.or_else(|| auth_config.default_tenant_id());
2995
2996    // T21.4 — the two per-tenant capability rows are resolved from **one**
2997    // settings read, for the reason `described_tenant` is one value: a
2998    // document that advertised `address` for tenant X and a registration
2999    // endpoint for tenant Y would be internally inconsistent in a way no
3000    // client could detect. Both default to `false` when no tenant was named,
3001    // which is every caller written before W7 and every conformance run — such
3002    // a caller receives a document byte-identical to the one served before
3003    // this task (I1).
3004    let capabilities = match described_tenant {
3005        None => axiam_oauth2::oidc::TenantCapabilities::default(),
3006        Some(tenant_id) => match state.tenant_repo.get_by_id(tenant_id).await {
3007            Ok(tenant) => axiam_core::repository::SettingsRepository::get_effective_settings(
3008                &state.settings_repo,
3009                tenant.organization_id,
3010                tenant_id,
3011            )
3012            .await
3013            .map(|s| axiam_oauth2::oidc::TenantCapabilities {
3014                sensitive_scopes_enabled: s.oidc.sensitive_scopes_enabled,
3015                dynamic_registration_enabled: s.oidc.dynamic_registration.is_enabled(),
3016                // T21.5 — the third row the `TenantCapabilities` struct was
3017                // built to hold, resolved from the same one settings read for
3018                // the same reason: one document, one tenant.
3019                cimd_enabled: s.oidc.cimd.enabled,
3020            })
3021            // A settings read that fails advertises less rather than more.
3022            // A relying party told a scope exists and then refused it has a
3023            // worse day than one that was never told. The same reasoning
3024            // covers the registration endpoint: an MCP client told it may
3025            // register and then refused `403` is worse off than one that was
3026            // never told.
3027            .unwrap_or_default(),
3028            // An unknown tenant is not an error here: discovery is public and
3029            // unauthenticated, and answering `404` for a tenant id would make
3030            // this endpoint a tenant-enumeration oracle. The deployment-wide
3031            // document is served instead, which is what a caller that named no
3032            // tenant gets.
3033            Err(_) => axiam_oauth2::oidc::TenantCapabilities::default(),
3034        },
3035    };
3036
3037    // RFC 8705 §5 — the alias host is re-based on the tenant path too, so an
3038    // mTLS client following an alias reaches the same tenant it discovered. It
3039    // must be: §5 tells such a client it MUST prefer the alias, so an alias
3040    // pointing at the deployment-wide endpoint would silently move it to
3041    // another tenant.
3042    let tenant_path_suffix = tenant_issuer
3043        .as_ref()
3044        .and(described_tenant)
3045        .map(|tenant| format!("{}{}", axiam_auth::config::TENANT_PATH_PREFIX, tenant));
3046    let mtls_base = match tenant_path_suffix {
3047        None => auth_config.mtls_base_url().map(str::to_owned),
3048        Some(ref suffix) => auth_config
3049            .mtls_base_url()
3050            .map(|base| format!("{base}{suffix}")),
3051    };
3052    let doc = match build_discovery_document_for(
3053        issuer,
3054        mtls_base.as_deref(),
3055        capabilities,
3056        // `None` under a tenant issuer: the endpoints are already tenant-scoped
3057        // by their path, and `?tenant_id=` on top would be a second selector
3058        // saying the same thing — which `middleware::tenant_path` refuses on
3059        // arrival anyway. `capabilities` above is still resolved from the real
3060        // `described_tenant`, so the document says what THAT tenant can do.
3061        if tenant_issuer.is_some() {
3062            None
3063        } else {
3064            described_tenant
3065        },
3066    ) {
3067        Ok(doc) => doc,
3068        Err(e) => {
3069            tracing::error!(
3070                error = %e,
3071                "AXIAM__AUTH__OAUTH2_MTLS_BASE_URL is set but unusable; refusing to serve a \
3072                 discovery document without its RFC 8705 §5 mtls_endpoint_aliases"
3073            );
3074            return HttpResponse::InternalServerError().json(OAuth2ErrorResponse {
3075                error: "server_error".into(),
3076                error_description: "mTLS endpoint alias base URL is not configured as a \
3077                    valid URL"
3078                    .into(),
3079            });
3080        }
3081    };
3082    HttpResponse::Ok().json(doc)
3083}

3085/// The session-revocation feed document (T-39, T-143).
3086#[derive(serde::Serialize, utoipa::ToSchema)]
3087pub struct RevocationFeedDocument {
3088    /// The digest algorithm the entries are computed with. Always `SHA-256`.
3089    /// A client that does not recognise the value MUST treat the document as
3090    /// unusable and behave exactly as it does when the feed is unreachable —
3091    /// never as though the list were empty, which is a guard that silently
3092    /// honours no revocations at all.
3093    pub alg: String,
3094    /// When this document was produced, as a Unix timestamp.
3095    pub issued_at: i64,
3096    /// How long an entry is published for, in seconds: one access-token
3097    /// lifetime. After that every token naming the session has expired on its
3098    /// own `exp`, so the entry proves nothing and is dropped.
3099    pub ttl: u64,
3100    /// Base64url-unpadded SHA-256 of each revoked session id, in no
3101    /// security-significant order. Never a session id, a subject or a tenant.
3102    pub revoked: Vec<String>,
3103}
3104
3105/// `GET /oauth2/revocations` -- sessions revoked within the last access-token
3106/// lifetime, as unlinkable hashes (T-39, T-143).
3107///
3108/// Mounted only where `AXIAM__AUTH__REVOCATION_FEED_ENABLED` is set; a
3109/// deployment that has not opted in does not serve this path at all.
3110///
3111/// An SDK route guard that polls this rejects a revoked session within one
3112/// poll interval rather than within one access-token lifetime. It is a
3113/// narrowing of a residual window and **not** a control: a guard that cannot
3114/// fetch the document, or that cannot interpret it, must behave exactly as it
3115/// does today. See `sdks/CONTRACT.md` §10.4.
3116//
3117// F3 note, deliberately not a doc comment: the reasoning for the entry format,
3118// the bound and the five properties lives in `axiam_core::revocation_feed`,
3119// where whoever changes this will read it. utoipa lifts a doc block verbatim
3120// into the OpenAPI `description` and eleven SDK repositories vendor that file
3121// byte-for-byte, so a design argument here becomes a re-vendor round.
3122#[utoipa::path(
3123    get,
3124    path = "/oauth2/revocations",
3125    tag = "oidc",
3126    responses(
3127        (status = 200, description = "Revocation feed", body = RevocationFeedDocument),
3128        (status = 304, description = "Not Modified -- ETag matches If-None-Match"),
3129    ),
3130)]
3131pub async fn revocations<C: Connection + Clone>(
3132    req: HttpRequest,
3133    state: web::Data<AppState<C>>,
3134) -> HttpResponse {
3135    let ttl = state.auth_config.access_token_lifetime_secs;
3136    let repo = axiam_db::SurrealRevokedSessionRepository::new(state.db.clone());
3137    let revoked = match repo.list_live(chrono::Utc::now()).await {
3138        Ok(hashes) => hashes,
3139        Err(e) => {
3140            // A guard treats an unreachable feed exactly as it treats one it
3141            // never fetched, so failing loudly here costs nothing a client can
3142            // act on. Answering `200` with an empty list would be worse than
3143            // failing: it asserts that nothing has been revoked.
3144            tracing::warn!(error = %e, "revocation feed could not be read");
3145            return HttpResponse::ServiceUnavailable().finish();
3146        }
3147    };
3148
3149    let document = RevocationFeedDocument {
3150        alg: axiam_core::revocation_feed::REVOCATION_HASH_ALG.to_owned(),
3151        issued_at: chrono::Utc::now().timestamp(),
3152        ttl,
3153        revoked,
3154    };
3155
3156    // The ETag covers the entry list and nothing else. `issued_at` changes on
3157    // every call by construction, so including it would make every poll a
3158    // full transfer — which is the one thing a feed meant to be polled must
3159    // not do.
3160    let etag = {
3161        use sha2::Digest as _;
3162        let mut hasher = sha2::Sha256::new();
3163        for entry in &document.revoked {
3164            hasher.update(entry.as_bytes());
3165            hasher.update(b"\n");
3166        }
3167        format!(
3168            "\"{}\"",
3169            base64::Engine::encode(
3170                &base64::engine::general_purpose::URL_SAFE_NO_PAD,
3171                hasher.finalize()
3172            )
3173        )
3174    };
3175
3176    // Half a poll interval's worth of caching at the shortest interval the
3177    // contract permits: long enough that a misbehaving poller cannot turn the
3178    // feed into a load source, short enough that it cannot widen the window
3179    // the feed exists to narrow.
3180    let cache_control = "public, max-age=15";
3181

3503/// Decide, and record, what `address` and `phone` release for this call
3504/// (X7 G8, plan §4.8).
3505///
3506/// # Every gate is re-asked here, at the moment of release
3507///
3508/// The authorization endpoint asked all of them too, and none of its answers
3509/// is reused. That is the whole design: an access token lives fifteen minutes
3510/// and a refresh of it lives thirty days, so a decision taken when the token
3511/// was minted is a decision that outlives the facts it rested on. Asking again
3512/// is what makes withdrawal immediate (T8.4) rather than effective on the next
3513/// token.
3514///
3515/// In order, cheapest and most decisive first:
3516///
3517/// 1. **The token's scopes.** No `phone`, no telephone number. A scope the
3518///    relying party never asked for cannot be released by any later gate
3519///    saying yes.
3520/// 2. **The relying party.** Taken from the token's `client_id`
3521///    (RFC 9068 §2.2). A token that names none — every token issued before W7,
3522///    and every token minted by a login, a device flow or an exchange — cannot
3523///    have a consent record looked up for it, and releases nothing. Fail
3524///    closed, and the direction matters: the alternative is "any consent this
3525///    subject ever gave", which would hand a postal address to a client the
3526///    subject consented to a *different* client receiving.
3527/// 3. **The profile (M10).** A `fapi2` client is refused here regardless of
3528///    scope, switch or record — the third of the three places the FAPI gate
3529///    runs, and the only one where no authorization request is in hand.
3530/// 4. **The tenant switch.** The operator's decision outranks the subject's.
3531/// 5. **The consent record**, matched on client *and* on the exact scope set.
3532///
3533/// # A read that fails releases nothing

3535/// Every fallible step below resolves to "release nothing" and logs. A
3536/// UserInfo response missing an optional claim is what OIDC Core §5.3.2 calls
3537/// an omitted claim and what the conformance suite treats as a WARNING; a
3538/// UserInfo response carrying a postal address the subject withdrew is a
3539/// personal-data breach. Those are not comparable failures and the code does
3540/// not treat them as if they were.

3744/// read and must not start reading: that impl is on the path `GET` takes, and
3745/// on the path every other authenticated route in the tree takes. Resolving the
3746/// token here instead leaves all of them untouched, which is what makes "GET
3747/// behaviour is unchanged" a property of the routing table rather than a claim
3748/// about a shared code path. Both arms then hand the same
3749/// [`AuthenticatedUser`] to the same [`userinfo_claims`].
3750///
3751/// # The refusals, and where each is decided
3752///
3753/// * **Two carriers, one request.** RFC 6750 §2: *"Clients MUST NOT use more
3754///   than one method to transmit the token in each request."* A form field
3755///   presented together with an `Authorization` header is refused here with
3756///   `400 invalid_request`, and the refusal names neither token — it does not
3757///   read them at all, so there is nothing to leak into the body or the log.
3758///   The `axiam_access` cookie is refused alongside the form field for the same
3759///   reason even though it is not one of RFC 6750's methods: it is a second
3760///   credential naming a possibly different subject, and the alternative to
3761///   refusing is choosing silently between two identities.
3762/// * **The cookie alone still authenticates, exactly as on GET.** That is
3763///   today's behaviour and invariant 4 keeps it. It is a CSRF surface in
3764///   principle — `/oauth2` is not wrapped in `CsrfMiddleware`, only `/api/v1`
3765///   is — and it fails closed in practice because `axiam_access` is
3766///   `SameSite=Strict` (`middleware::csrf::access_cookie`), so a cross-site
3767///   form POST carries no cookie and lands here unauthenticated. That is a
3768///   property of the cookie, not of this handler, so it is pinned by a test
3769///   rather than asserted by this comment
3770///   (`oauth2_userinfo_post_test.rs::the_access_cookie_is_strict_so_a_cross_site_post_cannot_be_authenticated_by_it`).
3771/// * **`access_token` in the query string is never read** — on either method.
3772///   RFC 6750 §2.3 deprecates that form because it puts a credential in a URL,
3773///   a `Referer` and every access log in the path. Refusing it would mean
3774///   reading it first, and the parameter's problem is that it exists at all,
3775///   not that a server honours it. So there is no code here that looks at the
3776///   query string, and a `GET /oauth2/userinfo?access_token=…` is simply an
3777///   unauthenticated request with an odd URL.
3778///
3779/// # An empty field is not a carrier
3780///
3781/// `access_token=` transmits no token, so it is treated as absent rather than
3782/// as a second method or as an invalid credential. A request carrying an empty
3783/// field and a real `Authorization` header is answered, not refused.
3784#[utoipa::path(
3785    post,
3786    path = "/oauth2/userinfo",
3787    tag = "oidc",
3788    request_body(
3789        content = UserInfoPostForm,
3790        content_type = "application/x-www-form-urlencoded",
3791        description = "RFC 6750 §2.2 form-encoded access token. Omit the body \
3792                       entirely to present the token in the Authorization header \
3793                       instead; presenting it both ways is refused."
3794    ),
3795    responses(
3796        (status = 200, description = "UserInfo response — byte-identical to what \
3797                                      GET answers for the same token",
3798         body = UserInfoResponse),
3799        (status = 400, description = "The access token was presented by more than \
3800                                      one method (RFC 6750 §2)"),
3801        (status = 401, description = "Invalid or missing access token"),
3802    ),
3803    security(("bearer" = []))
3804)]
3805pub async fn userinfo_post<C: Connection + Clone>(
3806    req: HttpRequest,
3807    form: Option<web::Form<UserInfoPostForm>>,
3808    maybe_user: MaybeAuthenticatedUser,
3809    state: web::Data<AppState<C>>,
3810) -> HttpResponse {
3811    use actix_web::ResponseError as _;
3812
3813    let body_token = form
3814        .as_ref()
3815        .and_then(|f| f.access_token.as_deref())
3816        .map(str::trim)
3817        .filter(|t| !t.is_empty());
3818
3819    let Some(body_token) = body_token else {
3820        // No §2.2 carrier: this is a header- or cookie-authenticated request,
3821        // and `MaybeAuthenticatedUser` holds precisely what actix would have
3822        // handed a `AuthenticatedUser` handler — the same principal, or the
3823        // same 401.
3824        return match maybe_user.into_result() {
3825            Ok(user) => userinfo_claims_for(&user, &state, Some(&req)).await,
3826            Err(e) => e.error_response(),
3827        };
3828    };
3829
3830    let header_present = req
3831        .headers()
3832        .contains_key(actix_web::http::header::AUTHORIZATION);
3833    let cookie_present = req.cookie("axiam_access").is_some();
3834    if header_present || cookie_present {
3835        return HttpResponse::BadRequest().json(OAuth2ErrorResponse {
3836            error: "invalid_request".into(),
3837            error_description: "the access token was presented by more than one method; \
3838                 RFC 6750 section 2 permits exactly one per request"
3839                .into(),
3840        });
3841    }
3842
3843    match crate::extractors::auth::authenticate_presented_token(&req, body_token).await {
3844        Ok(user) => userinfo_claims_for(&user, &state, Some(&req)).await,
3845        Err(e) => e.error_response(),
3846    }
3847}

3853/// Build a redirect response with error parameters per RFC 6749
3854/// section 4.1.2.1.
3855/// Append the RFC 9207 `iss` authorization-response parameter (X5.1).
3856///
3857/// # Why this matters, and why it is unconditional
3858///
3859/// RFC 9207 exists because of the **mix-up attack**: a client that talks to
3860/// more than one authorization server, and receives an authorization response
3861/// on a shared redirect URI, cannot otherwise tell which server sent it. An
3862/// attacker who controls one of those servers can therefore have a code minted
3863/// by an honest server delivered to the attacker's own token endpoint — or the
3864/// reverse. `iss` names the sender, so the client can check that the response
3865/// came from the server it started the flow with.
3866///
3867/// Emitted for **every** client, not just FAPI ones. RFC 9207 §2.4 says an AS
3868/// that supports the parameter SHOULD include it in every authorization
3869/// response; a client that does not understand it ignores an unknown query
3870/// parameter, which is the behaviour RFC 6749 §4.1.2 already requires of it.
3871/// Making it conditional would mean a client only gets mix-up protection when
3872/// somebody remembered to switch it on — and mix-up is precisely the attack a
3873/// client does not know it is under.
3874///
3875/// It goes on the **error** redirect too. RFC 9207 §2.4 is explicit about
3876/// this, and it is not a formality: one variant of the mix-up attack works by
3877/// injecting an error response, so a client that validates `iss` on success
3878/// and skips it on failure has left the door it just closed ajar.
3879/// The config this request stamps an `iss` with (T21.6).
3880///
3881/// The deployment's own, borrowed and free, for every request that did not
3882/// arrive on a `/t/{tenant_id}` path — which is every request on a deployment
3883/// with `AXIAM__AUTH__TENANT_ISSUER_PATHS` unset. A clone carrying the tenant
3884/// issuer for one that did, so that the RFC 9207 `iss` parameter, the access
3885/// token, the ID token and the logout token all say what the discovery document
3886/// the client read says.
3887fn issuer_config<'a, C: Connection + Clone>(
3888    req: &HttpRequest,
3889    state: &'a AppState<C>,
3890) -> std::borrow::Cow<'a, AuthConfig> {
3891    crate::middleware::tenant_path::minting_config(req, &state.auth_config)
3892}

3894/// [`issuer_config`] for a token-endpoint grant, which carries the issuer on
3895/// its [`TokenRequestContext`] rather than reading the request again (T21.6).
3896fn uma_issuer_config<'a, C: Connection + Clone>(
3897    ctx: &TokenRequestContext,
3898    state: &'a AppState<C>,
3899) -> std::borrow::Cow<'a, AuthConfig> {
3900    match ctx.issuer.as_deref() {
3901        None => std::borrow::Cow::Borrowed(&state.auth_config),
3902        Some(issuer) => std::borrow::Cow::Owned(AuthConfig {
3903            request_issuer: Some(issuer.to_owned()),
3904            ..state.auth_config.clone()
3905        }),
3906    }
3907}

3909fn append_issuer(url: &mut url::Url, auth_config: &AuthConfig) {
3910    let issuer = auth_config.effective_issuer();
3911    if issuer.is_empty() {
3912        // Nothing sensible to assert. Emitting an empty `iss` would be worse
3913        // than omitting it: a client comparing it against its configured
3914        // issuer would fail every flow, and one comparing loosely might pass
3915        // anything.
3916        tracing::warn!(
3917            "no issuer is configured, so the RFC 9207 `iss` authorization-response              parameter cannot be emitted; clients cannot detect a mix-up attack"
3918        );
3919        return;
3920    }
3921    url.query_pairs_mut().append_pair("iss", issuer);
3922}

3924fn build_error_redirect(
3925    redirect_uri: &str,
3926    error: &OAuth2Error,
3927    state: Option<&str>,
3928    auth_config: &AuthConfig,
3929) -> HttpResponse {
3930    let mut url = match url::Url::parse(redirect_uri) {
3931        Ok(u) => u,
3932        Err(_) => {
3933            // Unparseable redirect_uri — return direct error
3934            return HttpResponse::BadRequest().json(OAuth2ErrorResponse {
3935                error: error.error_code().to_string(),
3936                error_description: error.error_description(),
3937            });
3938        }
3939    };
3940    url.query_pairs_mut()
3941        .append_pair("error", error.error_code())
3942        .append_pair("error_description", &error.error_description());
3943    if let Some(state) = state {
3944        url.query_pairs_mut().append_pair("state", state);
3945    }
3946    // RFC 9207 §2.4 — see `append_issuer`. Error responses carry `iss` too,
3947    // because one form of the mix-up attack is an injected error response.
3948    append_issuer(&mut url, auth_config);
3949    HttpResponse::Found()
3950        .append_header(("Location", url.to_string()))
3951        .finish()
3952}

4009/// The `urn:ietf:params:oauth:grant-type:token-exchange` arm of the token
4010/// endpoint (B3).
4011///
4012/// The form is re-read rather than typed at the route because RFC 8693's
4013/// parameters are disjoint from RFC 6749's and actix deserializes one body
4014/// once — so `TokenRequest` carries the shared fields and the
4015/// exchange-specific ones are pulled from the raw form here.
4016/// X2 — `grant_type=urn:ietf:params:oauth:grant-type:uma-ticket`.
4017///
4018/// Redeems a permission ticket for an RPT. Four things are worth naming.
4019///
4020/// **The client authenticates, and the ticket is bound to it.** A ticket is
4021/// minted by a resource server through the Protection API and redeemed by that
4022/// same resource server; `consume` matches on `client_id`, so a ticket leaked
4023/// to another client is not a usable credential — and a wrong-client attempt
4024/// changes nothing rather than burning the rightful holder's ticket.
4025///
4026/// **`claim_token` is required, though UMA 2.0 calls it optional.** The spec's
4027/// other two ways to name a requesting party — an RPT presented for
4028/// incremental authorization, and interactive claims-gathering — are both
4029/// deferred to v2, so this is the only channel that exists. Requiring it and
4030/// saying so beats resolving to some default subject, which would mint an RPT
4031/// for a party nobody named.
4032///
4033/// **The subject token is validated, not trusted.** It is a signed AXIAM access
4034/// token; its remaining life bounds the RPT's, which is what stops an RPT from
4035/// outliving the authorization it rests on.
4036///
4037/// **Refusals are uniform.** Every ticket rejection — unknown, consumed,
4038/// expired, wrong client — answers `invalid_grant` with one message, because a
4039/// caller that could tell them apart could probe for live ticket handles.
4040async fn handle_uma_ticket<C: Connection + Clone>(
4041    tenant_id: Uuid,
4042    form: TokenRequest,
4043    state: &web::Data<AppState<C>>,
4044    authz: &crate::authz::AuthzData,
4045    ctx: &TokenRequestContext,
4046) -> HttpResponse {
4047    // SEC-093: only `client_id` is required up front. Whether the credential
4048    // that must accompany it is a secret, a certificate or a signed assertion
4049    // is a property of the REGISTRATION, so it cannot be decided before the
4050    // lookup — `authenticate_client` decides it, and answers the same
4051    // `invalid_client` for "no credential" as for "wrong credential".
4052    let Some(client_id) = form.client_id.as_deref() else {
4053        return build_oauth2_error_response(&OAuth2Error::InvalidClient(
4054            "the uma-ticket grant requires client authentication".into(),
4055        ));
4056    };
4057
4058    let client = match state
4059        .oauth2
4060        .token_service
4061        .authenticate_client(tenant_id, client_id, form.client_secret.as_deref(), ctx)
4062        .await
4063    {
4064        Ok(client) => client,
4065        Err(e) => return build_oauth2_error_response(&e),
4066    };
4067
4068    // T21.2 — a public client may not obtain an RPT. The uma-ticket grant
4069    // mints a token naming a requesting party on the strength of the calling
4070    // client's identity, and a public client's identity is a `client_id`
4071    // anybody who has seen one of its authorization requests knows. The admin
4072    // API cannot refuse this combination at registration — `uma-ticket` is not
4073    // a registrable grant type — so the refusal has to be here.
4074    if client.token_endpoint_auth_method.is_public() {
4075        return build_oauth2_error_response(&OAuth2Error::UnauthorizedClient(
4076            "a public client (token_endpoint_auth_method: none) cannot use the uma-ticket \
4077             grant: it presents no credential, and an RPT must be attributable"
4078                .into(),
4079        ));
4080    }
4081
4082    // Counted after authentication, keyed by the authenticated client — the
4083    // same reasoning as the token-exchange bucket: an unauthenticated caller
4084    // must not be able to consume a real client's allowance, and per-IP would
4085    // collapse a whole mesh behind one NAT into a single bucket.
4086    if !state.shared_rate_limit.check_at(
4087        &format!("oauth2_uma_ticket:{}:{}", tenant_id, client.client_id),
4088        chrono::Utc::now(),
4089        state.rate_limit_cfg.uma_ticket_per_min,
4090    ) {
4091        return HttpResponse::TooManyRequests()
4092            .append_header(("Cache-Control", "no-store"))
4093            .json(OAuth2ErrorResponse {
4094                error: "slow_down".into(),
4095                error_description: "uma-ticket rate limit exceeded for this client".into(),
4096            });
4097    }
4098
4099    // SEC-096: the same two gates the token-exchange grant now runs, for the
4100    // same reason — an RPT is an access token, and a `fapi2` client must not
4101    // be able to obtain an unconstrained one through the grant nobody wired
4102    // the profile gate into.
4103    if let Err(e) = axiam_oauth2::fapi::enforce_token_request(&client, ctx.evidence()) {
4104        return build_oauth2_error_response(&e);
4105    }
4106    let cnf = match state
4107        .oauth2
4108        .token_service
4109        .certificate_binding_for(&client, ctx)
4110    {
4111        Ok(cnf) => cnf,
4112        Err(e) => return build_oauth2_error_response(&e),
4113    };
4114
4115    let Some(ticket) = form.ticket.as_deref().filter(|t| !t.is_empty()) else {
4116        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
4117            "ticket is required for the uma-ticket grant".into(),
4118        ));
4119    };
4120
4121    let Some(claim_token) = form.claim_token.as_deref().filter(|t| !t.is_empty()) else {
4122        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
4123            "claim_token is required for the uma-ticket grant: it names the \
4124             requesting party, and v1 implements neither incremental \
4125             authorization nor claims-gathering"
4126                .into(),
4127        ));
4128    };
4129
4130    // An unrecognised format is refused rather than assumed. Guessing would
4131    // mean validating a foreign token with the wrong verifier.
4132    if let Some(format) = form.claim_token_format.as_deref().filter(|f| !f.is_empty())
4133        && format != UMA_CLAIM_TOKEN_FORMAT
4134    {
4135        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(format!(
4136            "unsupported claim_token_format '{format}'; v1 accepts only \
4137             '{UMA_CLAIM_TOKEN_FORMAT}'"
4138        )));
4139    }
4140
4141    let claims = match axiam_auth::token::validate_access_token(claim_token, &state.auth_config) {
4142        Ok(validated) => validated.0,
4143        Err(_) => {
4144            return build_oauth2_error_response(&OAuth2Error::InvalidGrant(
4145                "claim_token is not a valid access token".into(),
4146            ));
4147        }
4148    };
4149
4150    let Ok(subject_id) = claims.sub.parse::<Uuid>() else {
4151        return build_oauth2_error_response(&OAuth2Error::InvalidGrant(
4152            "claim_token does not name a user as its subject".into(),
4153        ));
4154    };
4155
4156    // The requesting party must belong to the tenant the ticket was minted in.
4157    // Without this a token from tenant A could redeem a ticket in tenant B and
4158    // be evaluated against B's grants under A's subject id.
4159    if claims.tenant_id != tenant_id.to_string() {
4160        return build_oauth2_error_response(&OAuth2Error::InvalidGrant(
4161            "claim_token belongs to a different tenant".into(),
4162        ));
4163    }
4164
4165    let subject_remaining_secs = claims.exp - chrono::Utc::now().timestamp();
4166
4167    let granted = match state
4168        .uma_service(authz.get_ref().clone())
4169        .exchange_ticket(
4170            tenant_id,
4171            &client.client_id,
4172            ticket,
4173            subject_id,
4174            subject_remaining_secs,
4175        )
4176        .await
4177    {
4178        Ok(granted) => granted,
4179        Err(e) => {
4180            tracing::info!(
4181                target: "axiam::audit",
4182                event = "uma.rpt_refused",
4183                tenant_id = %tenant_id,
4184                client_id = %client.client_id,
4185                subject = %subject_id,
4186                error = e.wire_error(),
4187                "uma-ticket grant refused"
4188            );
4189            return uma_refusal_response(e);
4190        }
4191    };
4192
4193    let org_id = match claims.org_id.parse::<Uuid>() {
4194        Ok(id) => id,
4195        Err(_) => {
4196            return build_oauth2_error_response(&OAuth2Error::ServerError(
4197                "claim_token carries an unparseable org_id".into(),
4198            ));
4199        }
4200    };
4201
4202    let rpt = match axiam_auth::token::issue_rpt(
4203        subject_id,
4204        tenant_id,
4205        org_id,
4206        granted.permissions.clone(),
4207        granted.lifetime_secs,
4208        // T21.6 — the RPT is a token this request minted, so it names the
4209        // issuer the request arrived under like every other one.
4210        &uma_issuer_config(ctx, state),
4211        cnf.clone(),
4212    ) {
4213        Ok(token) => token,
4214        Err(e) => return build_oauth2_error_response(&OAuth2Error::ServerError(e.to_string())),
4215    };
4216
4217    // Audited before the token is returned: the ticket is already consumed at
4218    // this point, so a crash between issuing and logging would leave a spent
4219    // ticket with no record of what it bought.
4220    tracing::info!(
4221        target: "axiam::audit",
4222        event = "uma.rpt_issued",
4223        tenant_id = %tenant_id,
4224        client_id = %client.client_id,
4225        subject = %subject_id,
4226        pairs = granted.permissions.len(),
4227        lifetime_secs = granted.lifetime_secs,
4228        "RPT issued"
4229    );
4230
4231    HttpResponse::Ok()
4232        .append_header(("Cache-Control", "no-store"))
4233        .append_header(("Pragma", "no-cache"))
4234        .json(TokenResponse {
4235            access_token: rpt,
4236            // SEC-096 / RFC 9449 §5: `DPoP` when the RPT is DPoP-bound,
4237            // `Bearer` for everything else — which is every client that did
4238            // not register DPoP binding.
4239            token_type: axiam_oauth2::dpop::token_type_for(cnf.as_ref()),
4240            expires_in: granted.lifetime_secs as u64,
4241            refresh_token: None,
4242            scope: None,
4243            id_token: None,
4244        })
4245}

4247/// Render a [`UmaError`] as the token endpoint's error response.
4248///
4249/// # Why `access_denied` does not go through `build_oauth2_error_response`
4250///
4251/// That helper maps everything except `invalid_client` and `server_error` to
4252/// **400**, which is right for RFC 6749 and right for RFC 8628 — the device
4253/// grant's `access_denied` (the user said no) is a 400 and must stay one.
4254///
4255/// UMA 2.0 §3.3.6 specifies **403** for a request the authorization server
4256/// refuses on the merits, and a UMA-conforming resource server distinguishes
4257/// "you may not have this" from "your request was malformed" by exactly that
4258/// status. Routing the UMA refusal through the shared helper would answer 400
4259/// and quietly fail conformance, so this one status is built here. Every other
4260/// UMA error keeps the shared mapping.
4261fn uma_refusal_response(e: UmaError) -> HttpResponse {
4262    if matches!(e, UmaError::AccessDenied) {
4263        return HttpResponse::Forbidden()
4264            .append_header(("Cache-Control", "no-store"))
4265            .append_header(("Pragma", "no-cache"))
4266            .json(OAuth2ErrorResponse {
4267                // v1 sends no `need_info` alongside it: that field exists to
4268                // steer a client into claims-gathering, which is deferred, so
4269                // sending it would point at a flow that does not run.
4270                error: "access_denied".into(),
4271                error_description:
4272                    "the requesting party is not authorized for every requested permission".into(),
4273            });
4274    }

4293async fn handle_token_exchange<C: Connection + Clone>(
4294    tenant_id: Uuid,
4295    form: TokenRequest,
4296    state: &web::Data<AppState<C>>,
4297    ctx: &TokenRequestContext,
4298) -> HttpResponse {
4299    // SEC-093 — see `handle_uma_ticket` for why only `client_id` is required
4300    // before the lookup.
4301    let Some(client_id) = form.client_id.as_deref() else {
4302        return build_oauth2_error_response(&OAuth2Error::InvalidClient(
4303            "the token-exchange grant requires client authentication".into(),
4304        ));
4305    };
4306
4307    let client = match state
4308        .oauth2
4309        .token_service
4310        .authenticate_client(tenant_id, client_id, form.client_secret.as_deref(), ctx)
4311        .await
4312    {
4313        Ok(client) => client,
4314        Err(e) => return build_oauth2_error_response(&e),
4315    };
4316
4317    // T21.2 — and a public client may not exchange. The admin API refuses the
4318    // grant at registration; this is the request-time half, so a row that
4319    // acquired the combination another way is refused here too rather than
4320    // minting a token for an audience on the strength of a `client_id`.
4321    if client.token_endpoint_auth_method.is_public() {
4322        return build_oauth2_error_response(&OAuth2Error::UnauthorizedClient(
4323            "a public client (token_endpoint_auth_method: none) cannot use the token-exchange \
4324             grant: it presents no credential, and an exchange must be attributable"
4325                .into(),
4326        ));
4327    }
4328
4329    // B3: the exchange's own bucket, applied here rather than as route
4330    // middleware because `/oauth2/token` serves every grant and actix
4331    // deserializes the body once — the grant is not knowable until the form
4332    // has been read, which is after the middleware chain has run.
4333    //
4334    // Keyed by the authenticated client, deliberately: an exchange always
4335    // carries client credentials, so unlike the device endpoints there is a
4336    // real identity to key on, and per-IP would collapse a whole mesh behind
4337    // one NAT into a single bucket. Counted AFTER authentication so an
4338    // unauthenticated caller cannot consume a real client's allowance.
4339    //
4340    // `check_at` rather than `check`: it derives the window itself so it can
4341    // also see the OFFSET into it, which is the sliding-window bound the J1
4342    // fix added after run 5 measured boundary over-admission. Re-deriving a
4343    // truncated window here would quietly opt this endpoint out of that fix.
4344    if !state.shared_rate_limit.check_at(
4345        &format!("oauth2_token_exchange:{}:{}", tenant_id, client.client_id),
4346        chrono::Utc::now(),
4347        state.rate_limit_cfg.token_exchange_per_min,
4348    ) {
4349        return HttpResponse::TooManyRequests()
4350            .append_header(("Cache-Control", "no-store"))
4351            .json(OAuth2ErrorResponse {
4352                error: "slow_down".into(),
4353                error_description: "token-exchange rate limit exceeded for this client".into(),
4354            });
4355    }
4356
4357    // SEC-096, gate 1: the profile the REGISTRATION declares, re-checked at
4358    // the moment it matters — the same call the authorization_code,
4359    // client_credentials and refresh_token grants make. Without it a `fapi2`
4360    // client could obtain an unconstrained token through this grant while
4361    // being refused one through the other three, which is a conformance
4362    // failure and an authentication downgrade in the same request.
4363    if let Err(e) = axiam_oauth2::fapi::enforce_token_request(&client, ctx.evidence()) {
4364        return build_oauth2_error_response(&e);
4365    }
4366
4367    // SEC-096, gate 2: the confirmation claim the EXCHANGING client earned on
4368    // this request. `None` for every client that registered no binding, so
4369    // this is byte-identical to the pre-SEC-096 response for them; for a
4370    // client that registered binding and presented nothing it refuses rather
4371    // than laundering a bound token into an unbound one.
4372    let cnf = match state
4373        .oauth2
4374        .token_service
4375        .certificate_binding_for(&client, ctx)
4376    {
4377        Ok(cnf) => cnf,
4378        Err(e) => return build_oauth2_error_response(&e),
4379    };
4380
4381    let Some(exchange_req) = form.exchange_request() else {
4382        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
4383            "subject_token and subject_token_type are required for the \
4384             token-exchange grant"
4385                .into(),
4386        ));
4387    };
4388
4389    match state
4390        .oauth2
4391        .token_exchange_service
4392        .exchange_at(tenant_id, &client, exchange_req, cnf, ctx.issuer.as_deref())
4393        .await
4394    {
4395        Ok(outcome) => {
4396            // Audited BEFORE the token is returned, not after. For an
4397            // impersonation this record is the only surviving evidence that
4398            // the acting party was not the subject — the token deliberately
4399            // carries no `act` claim — so it must not be lost to a crash
4400            // between issuing and logging.
4401            // X4: an external exchange's provenance is recorded in the same
4402            // entry, not a second one. A reader correlating two lines by
4403            // timestamp is a reader who can be made to correlate the wrong
4404            // two, and for a JIT provision this is the only record that a
4405            // partner's IdP created an AXIAM user.
4406            let ext = outcome.external.as_ref();
4407            tracing::info!(
4408                target: "axiam::audit",
4409                event = "oauth2.token_exchange",
4410                tenant_id = %tenant_id,
4411                client_id = %client.client_id,
4412                kind = outcome.kind.as_str(),
4413                subject = %outcome.subject,
4414                actor = outcome.actor.as_deref().unwrap_or("-"),
4415                granted_scopes = %outcome.granted_scopes.join(" "),
4416                audience = %outcome.audience,
4417                external_issuer = ext.map(|e| e.issuer.as_str()).unwrap_or("-"),
4418                external_subject = ext.map(|e| e.external_subject.as_str()).unwrap_or("-"),
4419                federation_provider = ext.map(|e| e.provider_name.as_str()).unwrap_or("-"),
4420                federation_config_id = ext
4421                    .map(|e| e.provider_id.to_string())
4422                    .unwrap_or_else(|| "-".into()),
4423                jit_provisioned = ext.is_some_and(|e| e.newly_provisioned),
4424                "token exchange"
4425            );
4426            HttpResponse::Ok()
4427                .append_header(("Cache-Control", "no-store"))
4428                .append_header(("Pragma", "no-cache"))
4429                .json(outcome.response)
4430        }
4431        Err(e) => {
4432            tracing::info!(
4433                target: "axiam::audit",
4434                event = "oauth2.token_exchange",
4435                tenant_id = %tenant_id,
4436                client_id = %client.client_id,
4437                outcome = "refused",
4438                error = e.error_code(),
4439                "token exchange refused"
4440            );
4441            build_oauth2_error_response(&e)
4442        }
4443    }
4444}

4446/// `invalid_client` returns 401 with a `WWW-Authenticate` header per
4447/// RFC 6749 §5.2.  Although the token endpoint uses `client_secret_post`,
4448/// RFC 6749 §5.2 still requires the 401 response to include
4449/// `WWW-Authenticate` indicating the authentication scheme.
4450/// Whether this request came from something that will *render* the answer.
4451///
4452/// Only an explicit `text/html` in `Accept` counts. A missing header, `*/*`
4453/// (curl's default) or an API client's `application/json` all keep the JSON
4454/// body byte-for-byte, so nothing that integrates against AXIAM today can
4455/// observe this.
4456fn prefers_html(req: &HttpRequest) -> bool {
4457    req.headers()
4458        .get(actix_web::http::header::ACCEPT)
4459        .and_then(|v| v.to_str().ok())
4460        .is_some_and(|accept| {
4461            accept
4462                .split(',')
4463                .any(|part| part.trim().starts_with("text/html"))
4464        })
4465}

4467/// The five characters that turn text into markup.
4468///
4469/// Written out rather than pulled from a crate because the set is closed and
4470/// the reason it exists should be visible at the point of use: an
4471/// `error_description` is not always AXIAM's own prose — `par_form_error`
4472/// forwards actix's message, which quotes a field name the caller chose.
4473fn escape_html(raw: &str) -> String {
4474    let mut out = String::with_capacity(raw.len());
4475    for c in raw.chars() {
4476        match c {
4477            '&' => out.push_str("&amp;"),
4478            '<' => out.push_str("&lt;"),
4479            '>' => out.push_str("&gt;"),
4480            '"' => out.push_str("&quot;"),
4481            '\'' => out.push_str("&#39;"),
4482            _ => out.push(c),
4483        }
4484    }
4485    out
4486}

4488/// An authorization-endpoint refusal, rendered for whoever asked.
4489///
4490/// # Why the authorization endpoint gets its own
4491///
4492/// `/oauth2/authorize` is the one OAuth2 endpoint a **person** arrives at.
4493/// When a refusal cannot be redirected to the client — RFC 6749 §4.1.2.1 for a
4494/// bad `redirect_uri` or `client_id`, and every `request_uri` failure, where
4495/// the redirect target is precisely what could not be resolved — the
4496/// specification says the server "SHOULD inform the resource owner of the
4497/// error". A `Content-Type: application/json` body informs a developer reading
4498/// a browser window; it does not inform a resource owner.
4499///
4500/// It is also what a certification reviewer is looking at. Five OIDF modules
4501/// end in REVIEW with instructions of the form "it must show an error page
4502/// saying the request_uri is invalid - upload a screenshot of the error page",
4503/// and the screenshot they were being handed was a raw JSON object.
4504///
4505/// # What it does not do
4506///
4507/// It does not change a single byte for a non-browser caller, and it does not
4508/// invent content: the page renders the same `error` and `error_description`
4509/// the JSON carries, escaped, and nothing else. In particular it echoes no
4510/// `state`, no `redirect_uri` and no `request_uri` — the rule
4511/// [`logged_out_page`] states and for the same reason, since those are
4512/// attacker-supplied strings and this page is served from AXIAM's own origin.
4513/// The code a `request_uri` refusal carries **when it is reported to the relying
4514/// party**, as opposed to rendered for the person in front of the browser.
4515///
4516/// OIDC Core §3.1.2.6 defines `invalid_request_uri` for exactly this state, and
4517/// the OIDF suite asserts it by name: `EnsureInvalidRequestUriError` accepts
4518/// that code and no other, so `invalid_request` — which is what RFC 6749 leaves
4519/// you with and what [`ParService::consume`](axiam_oauth2::par) produces —
4520/// fails two FAPI 2.0 modules the moment the refusal starts reaching the client
4521/// at all. It is also the more useful of the two: a relying party told
4522/// `invalid_request` has been handed a description of a client bug that did not
4523/// happen, while `invalid_request_uri` says the recoverable thing — push again
4524/// and restart.
4525///
4526/// **Narrow on purpose**, and narrow in the same place `is_request_uri_gone` is
4527/// narrow: `request_uri was not issued to this client` is a different failure
4528/// and keeps its own answer, exactly as it does on the login-hop return leg.
4529///
4530/// Returns `None` when the refusal is not about a spent handle, meaning "report
4531/// it as it is". Nothing here changes what a request with no registered
4532/// `redirect_uri` is answered — that is still
4533/// [`authorize_error_response`] rendering the original error.
4534fn request_uri_error_for_client(e: &OAuth2Error) -> Option<OAuth2Error> {
4535    axiam_oauth2::par::is_request_uri_gone(e).then(|| {
4536        OAuth2Error::InvalidRequestUri(
4537            "the pushed authorization request is unknown, expired, or has already been \
4538             used (a request_uri lives 60 seconds and may be spent once); push it again \
4539             and restart the authorization request"
4540                .into(),
4541        )
4542    })
4543}

4545/// Report a `request_uri` refusal to the relying party when the request named a
4546/// `redirect_uri` that client registered, and to the person otherwise.
4547///
4548/// The client is looked up *here* rather than passed in because the caller is
4549/// past [`resolve_authorize_principal`] and holds no registration: this path is
4550/// reached with a principal already resolved, and the pushed request that would
4551/// have carried the client's parameters is precisely what could not be read.
4552/// One repository read, on a refusal path only.
4553///
4554/// A lookup that fails is not a licence to redirect. Anything other than a
4555/// client whose registration lists the exact `redirect_uri` asked for falls
4556/// through to [`authorize_error_response`], which is the answer this endpoint
4557/// gave before the redirect existed.
4558async fn refuse_request_uri_to_client<C: Connection + Clone>(
4559    state: &AppState<C>,
4560    http_req: &HttpRequest,
4561    tenant_id: Uuid,
4562    client_id: &str,
4563    redirect_uri: Option<&str>,
4564    state_param: Option<&str>,
4565    e: &OAuth2Error,
4566) -> HttpResponse {
4567    let Some(uri) = redirect_uri else {
4568        return authorize_error_response(http_req, e);
4569    };
4570    let Ok(client) = state
4571        .oauth2_client_repo
4572        .get_by_client_id(tenant_id, client_id)
4573        .await
4574    else {
4575        return authorize_error_response(http_req, e);
4576    };
4577    if any_redirect_uri_matches(&client.redirect_uris, uri) {
4578        let mapped = request_uri_error_for_client(e);
4579        build_error_redirect(
4580            uri,
4581            mapped.as_ref().unwrap_or(e),
4582            state_param,
4583            &issuer_config(http_req, state),
4584        )
4585    } else {
4586        authorize_error_response(http_req, e)
4587    }
4588}

4590fn authorize_error_response(req: &HttpRequest, e: &OAuth2Error) -> HttpResponse {
4591    if !prefers_html(req) {
4592        return build_oauth2_error_response(e);
4593    }
4594    let status = match e {
4595        OAuth2Error::InvalidClient(_) => actix_web::http::StatusCode::UNAUTHORIZED,
4596        OAuth2Error::ServerError(_) => actix_web::http::StatusCode::INTERNAL_SERVER_ERROR,
4597        _ => actix_web::http::StatusCode::BAD_REQUEST,
4598    };
4599    let code = escape_html(e.error_code());
4600    let description = escape_html(&e.error_description());
4601    HttpResponse::build(status)
4602        .append_header(("Cache-Control", "no-store"))
4603        .append_header(("Pragma", "no-cache"))
4604        .content_type("text/html; charset=utf-8")
4605        .body(format!(
4606            "<!doctype html><html lang=\"en\"><head><meta charset=\"utf-8\">\
4607             <meta name=\"viewport\" content=\"width=device-width,initial-scale=1\">\
4608             <title>This authorization request cannot be completed</title></head>\
4609             <body><h1>This authorization request cannot be completed</h1>\
4610             <p>{description}</p><p><small>Error code: <code>{code}</code></small></p>\
4611             <p>Nothing has been shared with the application that sent you here. \
4612             Return to it and start again.</p></body></html>"
4613        ))
4614}

4616fn build_oauth2_error_response(e: &OAuth2Error) -> HttpResponse {
4617    let status = match e {
4618        OAuth2Error::InvalidClient(_) => actix_web::http::StatusCode::UNAUTHORIZED,
4619        OAuth2Error::ServerError(_) => actix_web::http::StatusCode::INTERNAL_SERVER_ERROR,
4620        _ => actix_web::http::StatusCode::BAD_REQUEST,
4621    };
4622    let mut builder = HttpResponse::build(status);
4623    builder
4624        .append_header(("Cache-Control", "no-store"))
4625        .append_header(("Pragma", "no-cache"));
4626    // RFC 6749 §5.2: 401 responses MUST include WWW-Authenticate
4627    if status == actix_web::http::StatusCode::UNAUTHORIZED {
4628        builder.append_header(("WWW-Authenticate", "Bearer realm=\"axiam\""));
4629    }
4630    builder.json(OAuth2ErrorResponse {
4631        error: e.error_code().to_string(),
4632        error_description: e.error_description(),
4633    })
4634}

4885/// Render a malformed PAR body as an OAuth2 error object (RFC 9126 §2.3).
4886///
4887/// # Why this exists
4888///
4889/// `web::Form` rejects a body it cannot deserialize *before* the handler runs,
4890/// and actix's default rendering is `400` with a `text/plain` body — for a
4891/// missing `redirect_uri`, literally:
4892///
4893/// ```text
4894/// Parse error: missing field `redirect_uri`.
4895/// ```
4896///
4897/// RFC 9126 §2.3 says the PAR endpoint's error response is the token
4898/// endpoint's: a JSON object carrying `error` and optionally
4899/// `error_description`. A client cannot act on prose, and a conformance suite
4900/// reports it as "Pushed Authorization did not return a JSON object" — which
4901/// is what two FAPI 2.0 modules did.
4902///
4903/// The description is deliberately actix's own message: the caller already
4904/// controls every byte of the body being described, so naming the field it got
4905/// wrong tells an attacker nothing and saves an integrator an afternoon.
4906///
4907/// Wired at the route rather than inside the handler because the handler is
4908/// never reached — see `crate::server`'s `/par` resource.
4909/// The message `serde` produces for a repeated field, for the one field this
4910/// server has to answer for specially (T21.3 / D1).
4911///
4912/// Pinned by [`the_duplicate_field_message_is_what_serde_still_says`] so that
4913/// a `serde` upgrade which reworded it fails a test rather than silently
4914/// reverting the error code below to `invalid_request`.
4915const DUPLICATE_RESOURCE_MESSAGE: &str = "duplicate field `resource`";
4916
4917/// Answer a repeated `resource` parameter with `invalid_target` (D1), or
4918/// `None` when the failure is anything else.
4919///
4920/// # Why a string comparison, and why that is not load-bearing
4921///
4922/// RFC 8707 §2 lets `resource` repeat; D1 of the MCP plan accepts one value
4923/// and refuses a second, because AXIAM's access token carries a single `aud`
4924/// (`AccessTokenClaims::aud`) and there is no honest way to serve two targets
4925/// with one string.
4926///
4927/// The refusal itself is **not** implemented here. `serde_urlencoded`
4928/// deserializes a repeated key into a non-sequence field by failing, so a
4929/// second `resource` never reaches a handler at all: the request is refused
4930/// before any of this runs, and it is refused whatever this function returns.
4931/// That is the fail-closed property, and it holds by construction — there is
4932/// no path on which two values are silently narrowed to one.
4933///
4934/// What this function buys is the *code*: a client that repeats the parameter
4935/// is told `invalid_target`, which is what RFC 8707 defines and what a client
4936/// can act on, instead of a `400` describing a deserializer. If the string
4937/// ever stops matching, the answer degrades to that `400` — still a refusal,
4938/// merely a less useful one.
4939fn duplicate_resource_response(message: &str) -> Option<HttpResponse> {
4940    message.contains(DUPLICATE_RESOURCE_MESSAGE).then(|| {
4941        build_oauth2_error_response(&OAuth2Error::InvalidTarget(
4942            "resource may be given at most once: AXIAM issues a token for one target service              (RFC 8707 section 2). Start a separate authorization for each resource"
4943                .into(),
4944        ))
4945    })
4946}

5049/// See [`token_inner`] for why this split exists.
5050async fn pushed_authorization_request_inner<C: Connection + Clone>(
5051    http_req: HttpRequest,
5052    tenant_query: web::Query<TenantQuery>,
5053    form: web::Form<PushedAuthorizationRequest>,
5054    state: web::Data<AppState<C>>,
5055) -> HttpResponse {
5056    let tenant_id = tenant_query.into_inner().tenant_id;
5057    let req = form.into_inner();
5058
5059    // RFC 9126 §2.1-2, answered before anything else is looked at.
5060    //
5061    // Ahead of client authentication deliberately, and it costs nothing: the
5062    // refusal names a parameter the caller sent rather than anything about the
5063    // client, so it is not an oracle, and a request that cannot be honoured
5064    // whoever sent it should not first consume an authentication.
5065    if req.request_uri.is_some() {
5066        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
5067            "request_uri must not be provided to the pushed authorization request endpoint \
5068             (RFC 9126 §2.1)"
5069                .into(),
5070        ));
5071    }
5072
5073    // T21.5 — before client authentication, as at the token endpoint.
5074    crate::cimd::materialise_if_cimd(&state, &http_req, tenant_id, &req.client_id).await;
5075
5076    // One client-authentication path in the codebase, shared with the token
5077    // endpoint, rather than a second one to keep correct — and since SEC-093
5078    // that path honours the registered `token_endpoint_auth_method`. PAR is
5079    // the sharpest instance of the old bug: FAPI 2.0 §5.3.1.1 requires strong
5080    // client authentication AND requires PAR, so a FAPI deployment whose PAR
5081    // endpoint accepted a shared secret was both an authentication downgrade
5082    // and a conformance failure.
5083    let ctx = match token_request_context(&http_req) {
5084        Ok(ctx) => ctx.with_assertion(
5085            req.client_assertion.as_deref(),
5086            req.client_assertion_type.as_deref(),
5087        ),
5088        Err(response) => return *response,
5089    };
5090    let client = match state
5091        .oauth2
5092        .token_service
5093        .authenticate_client(
5094            tenant_id,
5095            &req.client_id,
5096            req.client_secret.as_deref(),
5097            &ctx,
5098        )
5099        .await
5100    {
5101        Ok(client) => client,
5102        Err(e) => return build_oauth2_error_response(&e),
5103    };
5104    // Cloned because `client.client_id` is moved into the pushed request
5105    // below, and the §10.1 refusal between here and there wants to name the
5106    // client it is refusing.
5107    let client_id_for_log = client.client_id.clone();
5108
5109    // Keyed by the authenticated client and counted AFTER authentication, for
5110    // the same reasons as the token exchange: PAR always carries credentials
5111    // so there is a real identity to key on, per-IP would collapse a whole
5112    // deployment behind one NAT into a single bucket, and counting before
5113    // authentication would let an unauthenticated caller burn a real client's
5114    // allowance.
5115    if !state.shared_rate_limit.check_at(
5116        &format!("oauth2_par:{}:{}", tenant_id, client.client_id),
5117        chrono::Utc::now(),
5118        state.rate_limit_cfg.par_per_min,
5119    ) {
5120        return HttpResponse::TooManyRequests()
5121            .append_header(("Cache-Control", "no-store"))
5122            .json(OAuth2ErrorResponse {
5123                error: "slow_down".into(),
5124                error_description: "PAR rate limit exceeded for this client".into(),
5125            });
5126    }
5127
5128    // RFC 9449 §10.1 — resolve the DPoP key this authorization binds to.
5129    //
5130    //   Both mechanisms MUST be supported by an authorization server that
5131    //   supports PAR and DPoP.  If both mechanisms are used at the same time,
5132    //   the authorization server MUST reject the request if the JWK Thumbprint
5133    //   in dpop_jkt does not match the public key in the DPoP header.
5134    //
5135    // Deliberately after client authentication and after the rate limit.
5136    // Verifying a proof *records its `jti`*, so doing it earlier would let an
5137    // unauthenticated caller write to the replay table at will — and the
5138    // refusals below describe a key, which is a fact about the caller and not
5139    // about the client registration, so nothing is leaked by the wait.
5140    //
5141    // The header alone binds just as firmly as the parameter alone: §10.1 says
5142    // the server "MUST further behave as if the contained public key's
5143    // thumbprint was provided using dpop_jkt". That is the whole reason this
5144    // resolves to a single value rather than carrying both onward — by the
5145    // time the token endpoint asks, there is only one right answer and no
5146    // benefit in it having to reconstruct which carrier produced it.
5147    let proof_jkt = match verify_dpop_header(&http_req, &state, tenant_id).await {
5148        Ok(verified) => verified.map(|v| v.jkt),
5149        Err(response) => return *response,
5150    };
5151    let dpop_jkt = match (req.dpop_jkt.as_deref(), proof_jkt.as_deref()) {
5152        (Some(param), Some(proof)) if param != proof => {
5153            tracing::debug!(
5154                client_id = %client_id_for_log,
5155                "PAR carried a dpop_jkt naming a different key from the DPoP proof; refusing"
5156            );
5157            return *dpop_error_response(
5158                "invalid_dpop_proof",
5159                "dpop_jkt does not match the key of the DPoP proof on this request \
5160                 (RFC 9449 §10.1)",
5161                None,
5162            );
5163        }
5164        // Either they agree, or exactly one arrived, or neither did. In all
5165        // three the parameter wins where it exists and the proof supplies the
5166        // binding where it does not — which is the same value in the agreeing
5167        // case, so the arms need not be distinguished.
5168        (Some(param), _) => Some(param.to_owned()),
5169        (None, proof) => proof.map(str::to_owned),
5170    };
5171
5172    match state
5173        .oauth2
5174        .par_service
5175        .push(axiam_oauth2::par::PushedRequest {
5176            tenant_id,
5177            client_id: client.client_id,
5178            response_type: req.response_type,
5179            redirect_uri: req.redirect_uri,
5180            scope: req.scope,
5181            state: req.state,
5182            code_challenge: req.code_challenge,
5183            code_challenge_method: req.code_challenge_method,
5184            nonce: req.nonce,
5185            prompt: req.prompt,
5186            max_age: req.max_age,
5187            acr_values: req.acr_values,
5188            claims: req.claims,
5189            id_token_hint: req.id_token_hint,
5190            login_hint: req.login_hint,
5191            display: req.display,
5192            ui_locales: req.ui_locales,
5193            claims_locales: req.claims_locales,
5194            dpop_jkt,
5195            resource: req.resource,
5196        })
5197        .await
5198    {
5199        Ok(resp) => HttpResponse::Created()
5200            .append_header(("Cache-Control", "no-store"))
5201            .append_header(("Pragma", "no-cache"))
5202            .json(PushedAuthorizationResponse {
5203                request_uri: resp.request_uri,
5204                expires_in: resp.expires_in,
5205            }),
5206        Err(e) => build_oauth2_error_response(&e),
5207    }
5208}

5229
5230/// `GET`/`POST /oauth2/end_session` — OIDC RP-Initiated Logout 1.0 (B5).
5231///
5232/// Ends the session named by `id_token_hint` and notifies every client that
5233/// participated in it (Back-Channel Logout 1.0), then either redirects to an
5234/// allow-listed `post_logout_redirect_uri` or renders AXIAM's own logged-out
5235/// page.
5236///
5237/// # The redirect is the whole security surface
5238///
5239/// This endpoint is reachable without authentication by design — a user whose
5240/// session has already expired must still be able to complete a logout — so an
5241/// unvalidated `post_logout_redirect_uri` would be an open redirect on an
5242/// unauthenticated endpoint. It is honoured **only** on exact match against
5243/// the identified client's `post_logout_redirect_uris`.
5244///
5245/// A non-matching URI does not fail the request: the session is still ended
5246/// and AXIAM renders its own page. Refusing to log a user out because their RP
5247/// sent a bad parameter would be the wrong failure — they asked to log out.
5248///
5249/// # Why an unverifiable hint cannot end a session
5250///
5251/// Without a verifiable `id_token_hint` there is nothing to identify but the
5252/// browser's own AXIAM cookie, so the endpoint ends the cookie session if
5253/// there is one and does nothing otherwise. It deliberately does **not** fall
5254/// back to "end every session for the subject named in an unverified
5255/// parameter": that is a denial-of-service primitive handed to anyone who
5256/// knows a user id.
5257///
5258/// # No confirmation prompt
5259///
5260/// RP-Initiated Logout 1.0 §2 permits one, and a prompt is the mitigation for
5261/// logout CSRF. We take the other side deliberately: a forced logout is a
5262/// nuisance, not a privilege escalation, and a prompt shown on every logout is
5263/// trained away within a week.
5264#[utoipa::path(
5265    get,
5266    path = "/oauth2/end_session",
5267    tag = "oauth2",
5268    params(EndSessionQuery),
5269    responses(
5270        (status = 302, description = "Session ended; redirected to an allow-listed URI"),
5271        (status = 200, description = "Session ended; AXIAM's logged-out page"),
5272        (status = 400, description = "OAuth2 error", body = OAuth2ErrorResponse),
5273    ),
5274)]
5275pub async fn end_session<C: Connection + Clone>(
5276    req: HttpRequest,
5277    query: web::Query<EndSessionQuery>,
5278    state: web::Data<AppState<C>>,
5279) -> HttpResponse {
5280    let q = query.into_inner();
5281    let tenant_id = q.tenant_id;
5282    // T21.6 — `None` on the deployment-wide endpoint. Only the logout token's
5283    // `iss` reads it; which session is ended and where the browser may be sent
5284    // afterwards are decided exactly as before.
5285    let tenant_issuer =
5286        crate::middleware::tenant_path::binding_of(&req).map(|binding| binding.issuer().to_owned());
5287
5288    // 1. Identify the session and client. The hint is a *signed* statement of
5289    //    both; `client_id` is an unauthenticated parameter and is only a
5290    //    fallback for choosing the redirect allow-list.
5291    let hinted = q.id_token_hint.as_deref().and_then(|t| {
5292        axiam_oauth2::logout::decode_id_token_hint(t, &state.auth_config.jwt_public_key_pem)
5293    });
5294
5295    let (session_id, hint_client_id, subject_id) = match hinted {
5296        Some(h) => (h.session_id, Some(h.client_id), h.subject_id),
5297        None => (None, None, None),
5298    };
5299
5300    // 2. When both are present and disagree, refuse rather than resolve in
5301    //    favour of either. Picking the signed one would silently ignore what
5302    //    the caller asked; picking the unsigned one would let a parameter
5303    //    override a signature.
5304    if let (Some(hint_client), Some(param_client)) =
5305        (hint_client_id.as_deref(), q.client_id.as_deref())
5306        && hint_client != param_client
5307    {
5308        return build_oauth2_error_response(&OAuth2Error::InvalidRequest(
5309            "id_token_hint and client_id identify different clients".into(),
5310        ));
5311    }
5312
5313    let effective_client_id = hint_client_id.or_else(|| q.client_id.clone());
5314
5315    // 3. End the session, and notify the clients that were in it.
5316    if let (Some(session_id), Some(subject_id)) = (session_id, subject_id) {
5317        dispatch_backchannel_logout(
5318            &state,
5319            tenant_id,
5320            session_id,
5321            subject_id,
5322            tenant_issuer.as_deref(),
5323        )
5324        .await;
5325        // Best-effort: a session that has already expired is not an error —
5326        // the user asked to be logged out and they are.
5327        let _ = state.auth_service.logout(tenant_id, session_id).await;
5328    }
5329
5330    // 4. Resolve the redirect against the identified client's allow-list.
5331    let allow_list = match effective_client_id.as_deref() {
5332        Some(client_id) => state
5333            .oauth2_client_repo
5334            .get_by_client_id(tenant_id, client_id)
5335            .await
5336            .map(|c| c.post_logout_redirect_uris)
5337            .unwrap_or_default(),
5338        None => Vec::new(),
5339    };
5340
5341    let cookie_secure = state.auth_config.cookie_secure;
5342
5343    match axiam_oauth2::logout::resolve_post_logout_redirect(
5344        q.post_logout_redirect_uri.as_deref(),
5345        &allow_list,
5346        q.state.as_deref(),
5347    ) {
5348        axiam_oauth2::logout::LogoutOutcome::Redirect { uri, state: st } => {
5349            let location = match (url::Url::parse(&uri), st) {
5350                (Ok(mut url), Some(st)) => {
5351                    url.query_pairs_mut().append_pair("state", &st);
5352                    url.to_string()
5353                }
5354                (Ok(url), None) => url.to_string(),
5355                // An allow-listed entry that will not parse is an operator
5356                // error, not a user-facing one; render rather than emit a
5357                // Location header we could not construct.
5358                (Err(_), _) => {
5359                    return logged_out_page(cookie_secure);
5360                }
5361            };
5362            HttpResponse::Found()
5363                .append_header(("Location", location))
5364                .append_header(("Cache-Control", "no-store"))
5365                .cookie(crate::middleware::csrf::clear_access_cookie(cookie_secure))
5366                .cookie(crate::middleware::csrf::clear_refresh_cookie(cookie_secure))
5367                .cookie(crate::middleware::csrf::clear_csrf_cookie(cookie_secure))
5368                // W3: RP-initiated logout clears the OP browser session too.
5369                // Leaving it would mean a user who logged out through one
5370                // relying party is still recognised, silently, by the next.
5371                .cookie(crate::middleware::csrf::clear_op_session_cookie())
5372                .finish()
5373        }
5374        axiam_oauth2::logout::LogoutOutcome::Rendered => logged_out_page(cookie_secure),
5375    }
5376}

5378/// AXIAM's own logged-out page.
5379///
5380/// Deliberately carries no RP-supplied content — not the `state`, not the
5381/// rejected URI. Echoing either would put an attacker-controlled string into
5382/// a page served from AXIAM's own origin.
5383///
5384/// `cookie_secure` is `AuthConfig::cookie_secure` (D-18): the removal cookies
5385/// must carry the same attributes as the ones they clear.
5386fn logged_out_page(cookie_secure: bool) -> HttpResponse {
5387    HttpResponse::Ok()
5388        .append_header(("Cache-Control", "no-store"))
5389        .content_type("text/html; charset=utf-8")
5390        .cookie(crate::middleware::csrf::clear_access_cookie(cookie_secure))
5391        .cookie(crate::middleware::csrf::clear_refresh_cookie(cookie_secure))
5392        .cookie(crate::middleware::csrf::clear_csrf_cookie(cookie_secure))
5393        .cookie(crate::middleware::csrf::clear_op_session_cookie())
5394        .body(
5395            "<!doctype html><html lang=\"en\"><head><meta charset=\"utf-8\">\
5396             <title>Signed out</title></head><body><h1>You are signed out.</h1>\
5397             </body></html>",
5398        )
5399}

5401/// Build and dispatch logout tokens for a session's participants.
5402///
5403/// Spawned rather than awaited: the user's session is gone the moment this
5404/// request returns, and making logout wait on N external HTTP calls would make
5405/// it a hostage to the least reliable RP.
5406async fn dispatch_backchannel_logout<C: Connection + Clone>(
5407    state: &web::Data<AppState<C>>,
5408    tenant_id: Uuid,
5409    session_id: Uuid,
5410    subject_id: Uuid,
5411    // T21.6 — the issuer the end-session request arrived under. `None` on the
5412    // deployment-wide endpoint, which is every request on a deployment with
5413    // `AXIAM__AUTH__TENANT_ISSUER_PATHS` unset.
5414    //
5415    // Back-Channel Logout 1.0 §2.4 requires the logout token's `iss` to be the
5416    // OP's issuer, and the relying party validates it against the `issuer` of
5417    // the discovery document it registered with — so a logout token minted
5418    // under the root issuer for a session established under a tenant issuer
5419    // would be dropped by every conforming RP.
5420    request_issuer: Option<&str>,
5421) {
5422    let participants = match state
5423        .oauth2
5424        .session_client_repo
5425        .list_for_session(tenant_id, session_id)
5426        .await
5427    {
5428        Ok(rows) => rows,
5429        Err(e) => {
5430            tracing::warn!(error = %e, "could not list session participants for logout");
5431            return;
5432        }
5433    };
5434    if participants.is_empty() {
5435        return;
5436    }
5437
5438    let mut clients = Vec::new();
5439    for client_id in participants
5440        .iter()
5441        .map(|p| p.client_id.clone())
5442        .collect::<std::collections::BTreeSet<_>>()
5443    {
5444        match state
5445            .oauth2_client_repo
5446            .get_by_client_id(tenant_id, &client_id)
5447            .await
5448        {
5449            Ok(c) => clients.push(c),
5450            // A participant whose registration cannot be loaded is dropped
5451            // from the fan-out — correct (a deleted client keeps its
5452            // participation rows), but silent until now. The RP that does not
5453            // get told sees nothing, and neither did anyone reading the logs.
5454            Err(e) => tracing::debug!(
5455                %client_id,
5456                error = %e,
5457                "back-channel logout: session participant is not a loadable client, skipping"
5458            ),
5459        }
5460    }
5461
5462    let issuer = request_issuer
5463        .unwrap_or_else(|| state.auth_config.effective_issuer())
5464        .to_string();
5465    let auth_config = state.auth_config.clone();
5466    let participant_ids: Vec<String> = participants.into_iter().map(|p| p.client_id).collect();
5467
5468    let deliveries = crate::backchannel_logout::select_targets(
5469        &participant_ids,
5470        &clients,
5471        |client_id| {
5472            match axiam_oauth2::logout::issue_logout_token(
5473                &issuer,
5474                client_id,
5475                session_id,
5476                subject_id,
5477                &auth_config,
5478            ) {
5479                Ok(token) => Some(token),
5480                // This used to be `.ok()`. A token that cannot be minted (a
5481                // misconfigured issuer, an unusable signing key) silently
5482                // removed the client from the fan-out, so a whole deployment
5483                // could stop notifying anyone with no signal anywhere — on a
5484                // session-termination path, where "nothing happened" is
5485                // exactly the outcome an attacker wants. WARN, not DEBUG: this
5486                // one is never routine.
5487                Err(e) => {
5488                    tracing::warn!(
5489                        %client_id,
5490                        error = %e,
5491                        "back-channel logout: could not issue a logout token, client will NOT be notified"
5492                    );
5493                    None
5494                }
5495            }
5496        },
5497    );
5498
5499    // One line naming every stage of the funnel, so a delivery that never
5500    // happens can be localised without a code read: how many clients joined
5501    // the session, how many of those are still loadable registrations, and how
5502    // many of those actually got a token and a URI to send it to. Each drop
5503    // between those numbers has its own line above (or is the deliberate
5504    // "client registered no backchannel_logout_uri" skip in `select_targets`).
5505    tracing::debug!(
5506        participants = participant_ids.len(),
5507        clients = clients.len(),
5508        deliveries = deliveries.len(),
5509        "back-channel logout fan-out computed"
5510    );
5511
5512    // Participation records are dropped after the fan-out list is built, not
5513    // before — the list is what the fan-out iterates.
5514    let _ = state
5515        .oauth2
5516        .session_client_repo
5517        .delete_for_session(tenant_id, session_id)
5518        .await;
5519
5520    if !deliveries.is_empty() {
5521        actix_web::rt::spawn(crate::backchannel_logout::deliver_all(deliveries));
5522    }
5523}
