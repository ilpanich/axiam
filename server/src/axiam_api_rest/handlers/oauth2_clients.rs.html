22#[derive(Debug, Deserialize, utoipa::ToSchema)]
23pub struct CreateOAuth2ClientRequest {
24    /// Human-readable name for the client.
25    pub name: String,
26    /// Allowed redirect URIs (must be HTTPS, except localhost for dev).
27    /// SEC-089: this list doubles as the token-exchange audience allow-list
28    /// — adding a URI here also authorises it as a token audience for this
29    /// client, so review additions on exchange-capable clients with that in
30    /// mind (see `docs/api/token-exchange.md#audience`).
31    pub redirect_uris: Vec<String>,
32    /// Grant types this client is authorized to use.
33    pub grant_types: Vec<String>,
34    /// Scopes the client may request.
35    pub scopes: Vec<String>,
36    /// B5 — allow-list for RP-initiated logout's `post_logout_redirect_uri`.
37    /// Separate from `redirect_uris` on purpose: that list receives
38    /// authorization codes, this one receives a browser after logout.
39    #[serde(default)]
40    pub post_logout_redirect_uris: Vec<String>,
41    /// B5 — where OIDC back-channel logout tokens are delivered. Omit for a
42    /// client that does not participate.
43    #[serde(default)]
44    pub backchannel_logout_uri: Option<String>,
45    /// B5 — require this client to push its authorization parameters to
46    /// `/oauth2/par` (RFC 9126) rather than sending them through the browser.
47    #[serde(default)]
48    pub require_par: bool,
49    /// X5.1 — the security posture this client is registered under.
50    ///
51    /// `"standard"` (the default) is every AXIAM client that has ever existed.
52    /// `"fapi2"` turns on the whole FAPI 2.0 constraint bundle at once, and
53    /// the registration is refused unless it also sets `require_par`, a strong
54    /// `token_endpoint_auth_method` (either mTLS method or `private_key_jwt`),
55    /// and at least one sender-constraining mechanism
56    /// (`tls_client_certificate_bound_access_tokens` or
57    /// `dpop_bound_access_tokens`). See the FAPI operator guide.
58    #[serde(default)]
59    pub profile: ClientProfile,
60    /// X5.1 — how this client authenticates at the token endpoint
61    /// (RFC 8705 §2). Defaults to `client_secret_post`.
62    #[serde(default)]
63    pub token_endpoint_auth_method: ClientAuthMethod,
64    /// RFC 8705 §2.1.2 — expected certificate subject DN, RFC 4514 form.
65    /// Exactly one of the three `tls_client_auth_*` parameters may be set.
66    #[serde(default)]
67    pub tls_client_auth_subject_dn: Option<String>,
68    /// RFC 8705 §2.1.2 — expected `dNSName` SAN.
69    #[serde(default)]
70    pub tls_client_auth_san_dns: Option<String>,
71    /// RFC 8705 §2.1.2 — expected `uniformResourceIdentifier` SAN.
72    #[serde(default)]
73    pub tls_client_auth_san_uri: Option<String>,
74    /// Accepted certificate thumbprints for `self_signed_tls_client_auth`,
75    /// as base64url-unpadded SHA-256 digests of the DER certificate (the same
76    /// `x5t#S256` encoding as the `cnf` claim). More than one permits an
77    /// overlapping rotation.
78    #[serde(default)]
79    pub self_signed_tls_client_auth_thumbprints: Vec<String>,
80    /// RFC 8705 §3.4 — issue certificate-bound (sender-constrained) access
81    /// tokens to this client. Independent of the authentication method.
82    #[serde(default)]
83    pub tls_client_certificate_bound_access_tokens: bool,
84    /// RFC 7591 §2 — the client's public key set, inline, for
85    /// `private_key_jwt`. Exactly one of `jwks` and `jwks_uri` may be set.
86    #[serde(default)]
87    pub jwks: Option<String>,
88    /// RFC 7591 §2 — where the client publishes its public key set. Must be an
89    /// absolute `https` URL, and is fetched through the SSRF-guarded JWKS
90    /// cache, which refuses private and loopback addresses.
91    #[serde(default)]
92    pub jwks_uri: Option<String>,
93    /// RFC 9449 §5.2 — issue DPoP-bound (sender-constrained) access tokens to
94    /// this client. Independent of both the authentication method and
95    /// `tls_client_certificate_bound_access_tokens`; a client may ask for both
96    /// constraints, and a token carrying both must satisfy both.
97    #[serde(default)]
98    pub dpop_bound_access_tokens: bool,
99    /// RFC 9449 §8 — require this client's DPoP proofs to carry a
100    /// server-issued nonce.
101    ///
102    /// **Not implemented in this build (SEC-097).** `true` is refused with
103    /// `400`; only `false` (the default) is accepted. Nothing reads the stored
104    /// value, so accepting `true` would persist and echo back a security
105    /// switch that does nothing. DPoP proofs are made single-use at the token
106    /// endpoint by `jti` replay detection instead — see
107    /// `docs/security-profiles.md`.
108    #[serde(default)]
109    pub dpop_require_nonce: bool,
110}

226/// SEC-097 — refuse `dpop_require_nonce: true` while nothing reads it.
227///
228/// The field is persisted, echoed back by `GET`, and documented in the OpenAPI
229/// schema, and **no code path consults it**: the token endpoint passes
230/// `require_nonce: false` unconditionally because this deployment stores no
231/// per-client nonce to compare an echoed one against, so a challenge would
232/// prove liveness and nothing else.
233///
234/// An operator who sets a security switch, sees it persisted and echoed back,
235/// and gets nothing has been told a lie by the API. Refusing at the point of
236/// action is the same shape as SEC-101's transport check and is the only
237/// answer that does not require the operator to read `handlers/oauth2.rs` to
238/// discover the truth. The field itself is kept — removing it would be a wire
239/// break across seven SDKs for a value that is always `false` anyway — and
240/// this check is what a future nonce implementation deletes.
241fn reject_unimplemented_dpop_nonce(requested: Option<bool>) -> Result<(), AxiamApiError> {
242    if requested == Some(true) {
243        return Err(validation_err(
244            "dpop_require_nonce is not implemented in this build: nothing reads the stored \
245             value, so setting it would persist a security switch that does nothing. DPoP \
246             proofs are already made single-use at the token endpoint by jti replay \
247             detection (RFC 9449 §11.1). See docs/security-profiles.md.",
248        ));
249    }
250    Ok(())
251}
252
253/// Grant types a client may be registered for through this API.
254///
255/// This list MUST stay in step with what the token endpoint actually honours.
256/// Until 2026-08-15 it omitted the device-code and token-exchange grants, so
257/// `device_service::DEVICE_CODE_GRANT_TYPE` and
258/// `token_exchange::TOKEN_EXCHANGE_GRANT_TYPE` — both required by their
259/// respective flows, and both shipped — were rejected here as
260/// "unknown grant_type". That made B2 (device flow) and B3 (token exchange)
261/// unreachable through the admin API: their tests pass only because they build
262/// `OAuth2Client` values directly and never cross this validator.
263///
264/// `device_code` (the bare form) is accepted alongside the URN because
265/// `device_service` honours both.
266///
267/// Deliberately NOT included: `urn:axiam:params:oauth:grant-type:may-impersonate`.
268/// It is a capability marker, not a grant, and whether an admin-API caller may
269/// confer impersonation is a security decision that has not been taken. Leaving
270/// it out preserves the existing posture — impersonation stays settable only
271/// out-of-band.
272const KNOWN_GRANT_TYPES: &[&str] = &[
273    "authorization_code",
274    "client_credentials",
275    "refresh_token",
276    axiam_oauth2::device_service::DEVICE_CODE_GRANT_TYPE,
277    "device_code",
278    axiam_oauth2::token_exchange::TOKEN_EXCHANGE_GRANT_TYPE,
279];
280
281fn validation_err(msg: impl Into<String>) -> AxiamApiError {
282    axiam_core::error::AxiamError::Validation {
283        message: msg.into(),
284    }
285    .into()
286}

288fn validate_redirect_uris(uris: &[String]) -> Result<(), AxiamApiError> {
289    if uris.is_empty() {
290        return Err(validation_err("redirect_uris must not be empty"));
291    }
292    for uri in uris {
293        let parsed: url::Url = uri
294            .parse()
295            .map_err(|_| validation_err(format!("invalid redirect_uri: {uri}")))?;
296        // Redirect URIs must be absolute with an authority (host)
297        let host = parsed.host_str().ok_or_else(|| {
298            validation_err(format!(
299                "redirect_uri must be an absolute URL with a host: {uri}"
300            ))
301        })?;
302        // Allow http for localhost/loopback only, require HTTPS otherwise
303        let is_localhost = host == "localhost" || host == "127.0.0.1" || host == "::1";
304        if parsed.scheme() != "https" && !(parsed.scheme() == "http" && is_localhost) {
305            return Err(validation_err(format!(
306                "redirect_uri must use https (http is only allowed for localhost/127.0.0.1/::1): {uri}"
307            )));
308        }
309        // RFC 6749 §3.1.2: redirect URIs must not include a fragment
310        if parsed.fragment().is_some() {
311            return Err(validation_err(format!(
312                "redirect_uri must not contain a fragment: {uri}"
313            )));
314        }
315    }
316    Ok(())
317}
318
319fn validate_grant_types(grant_types: &[String]) -> Result<(), AxiamApiError> {
320    if grant_types.is_empty() {
321        return Err(validation_err("grant_types must not be empty"));
322    }
323    for gt in grant_types {
324        if !KNOWN_GRANT_TYPES.contains(&gt.as_str()) {
325            return Err(validation_err(format!(
326                "unknown grant_type: {gt} (allowed: {})",
327                KNOWN_GRANT_TYPES.join(", ")
328            )));
329        }
330    }
331    Ok(())
332}

338/// `POST /api/v1/oauth2-clients`
339#[utoipa::path(
340    post,
341    path = "/api/v1/oauth2-clients",
342    tag = "oauth2-clients",
343    request_body = CreateOAuth2ClientRequest,
344    responses(
345        (status = 201, description = "OAuth2 client created (secret shown once)",
346         body = OAuth2ClientCreatedResponse),
347    ),
348    security(("bearer" = []))
349)]
350pub async fn create<C: Connection + Clone>(
351    user: AuthenticatedUser,
352    authz: AuthzData,
353    state: web::Data<AppState<C>>,
354    body: web::Json<CreateOAuth2ClientRequest>,
355) -> Result<HttpResponse, AxiamApiError> {
356    RequirePermission::new("oauth2_clients:create", Uuid::nil())
357        .check(&user, authz.get_ref().as_ref())
358        .await?;
359    let req = body.into_inner();
360
361    if req.name.is_empty() {
362        return Err(validation_err("name must not be empty"));
363    }
364    validate_grant_types(&req.grant_types)?;
365    reject_unimplemented_dpop_nonce(Some(req.dpop_require_nonce))?;
366    // redirect_uris are only required when the client uses
367    // authorization_code (which involves user-agent redirects).
368    // M2M clients (client_credentials only) don't need them.
369    let needs_redirects = req.grant_types.iter().any(|g| g == "authorization_code");
370    if needs_redirects {
371        validate_redirect_uris(&req.redirect_uris)?;
372    }
373
374    let create = CreateOAuth2Client {
375        tenant_id: user.tenant_id,
376        name: req.name,
377        redirect_uris: req.redirect_uris,
378        grant_types: req.grant_types,
379        scopes: req.scopes,
380        post_logout_redirect_uris: req.post_logout_redirect_uris,
381        backchannel_logout_uri: req.backchannel_logout_uri,
382        require_par: req.require_par,
383        profile: req.profile,
384        token_endpoint_auth_method: req.token_endpoint_auth_method,
385        tls_client_auth_subject_dn: req.tls_client_auth_subject_dn,
386        tls_client_auth_san_dns: req.tls_client_auth_san_dns,
387        tls_client_auth_san_uri: req.tls_client_auth_san_uri,
388        self_signed_tls_client_auth_thumbprints: req.self_signed_tls_client_auth_thumbprints,
389        tls_client_certificate_bound_access_tokens: req.tls_client_certificate_bound_access_tokens,
390        jwks: req.jwks,
391        jwks_uri: req.jwks_uri,
392        dpop_bound_access_tokens: req.dpop_bound_access_tokens,
393        dpop_require_nonce: req.dpop_require_nonce,
394    };
395
396    // X5.1 — refuse a registration that could not satisfy the profile it
397    // declares, BEFORE it is written. A `fapi2` client missing PAR or mTLS
398    // would otherwise be created and then fail every request it ever made,
399    // which is a configuration error discovered by the client's users rather
400    // than by the operator making it. A `standard` client with a secret — every
401    // client that predates X5.1 — passes without a check running.
402    axiam_oauth2::fapi::validate_registration(&create)
403        .map_err(|e| validation_err(e.to_string()))?;
404
405    let (client, raw_secret) = state.oauth2_client_repo.create(create).await?;
406
407    Ok(HttpResponse::Created().json(OAuth2ClientCreatedResponse {
408        id: client.id,
409        tenant_id: client.tenant_id,
410        client_id: client.client_id,
411        client_secret: raw_secret,
412        name: client.name,
413        redirect_uris: client.redirect_uris,
414        grant_types: client.grant_types,
415        scopes: client.scopes,
416        created_at: client.created_at,
417        updated_at: client.updated_at,
418    }))
419}

489/// `PUT /api/v1/oauth2-clients/{id}`
490#[utoipa::path(
491    put,
492    path = "/api/v1/oauth2-clients/{id}",
493    tag = "oauth2-clients",
494    params(("id" = Uuid, Path, description = "OAuth2 client ID")),
495    request_body = UpdateOAuth2ClientRequest,
496    responses(
497        (status = 200, description = "OAuth2 client updated",
498         body = OAuth2ClientResponse),
499        (status = 404, description = "OAuth2 client not found"),
500    ),
501    security(("bearer" = []))
502)]
503pub async fn update<C: Connection + Clone>(
504    user: AuthenticatedUser,
505    authz: AuthzData,
506    path: web::Path<Uuid>,
507    state: web::Data<AppState<C>>,
508    body: web::Json<UpdateOAuth2ClientRequest>,
509) -> Result<HttpResponse, AxiamApiError> {
510    RequirePermission::new("oauth2_clients:update", Uuid::nil())
511        .check(&user, authz.get_ref().as_ref())
512        .await?;
513    let id = path.into_inner();
514    let req = body.into_inner();
515
516    if let Some(ref name) = req.name
517        && name.is_empty()
518    {
519        return Err(validation_err("name must not be empty"));
520    }
521    if let Some(ref gts) = req.grant_types {
522        validate_grant_types(gts)?;
523    }
524    reject_unimplemented_dpop_nonce(req.dpop_require_nonce)?;
525    if let Some(ref uris) = req.redirect_uris {
526        let needs_redirects = req
527            .grant_types
528            .as_ref()
529            .map(|gts| gts.iter().any(|g| g == "authorization_code"))
530            .unwrap_or(true);
531        if needs_redirects {
532            validate_redirect_uris(uris)?;
533        }
534    }
535
536    // When adding authorization_code to grant_types, ensure the
537    // client will have valid redirect_uris (either supplied in this
538    // request or already stored).
539    let adding_auth_code = req
540        .grant_types
541        .as_ref()
542        .is_some_and(|gts| gts.iter().any(|g| g == "authorization_code"));
543    if adding_auth_code && req.redirect_uris.is_none() {
544        let existing = state
545            .oauth2_client_repo
546            .get_by_id(user.tenant_id, id)
547            .await?;
548        if existing.redirect_uris.is_empty() {
549            return Err(validation_err(
550                "redirect_uris are required when enabling \
551                 authorization_code grant",
552            ));
553        }
554        validate_redirect_uris(&existing.redirect_uris)?;
555    }
556
557    let update = UpdateOAuth2Client {
558        name: req.name,
559        redirect_uris: req.redirect_uris,
560        grant_types: req.grant_types,
561        scopes: req.scopes,
562        post_logout_redirect_uris: req.post_logout_redirect_uris,
563        backchannel_logout_uri: req.backchannel_logout_uri,
564        require_par: req.require_par,
565        profile: req.profile,
566        token_endpoint_auth_method: req.token_endpoint_auth_method,
567        tls_client_auth_subject_dn: req.tls_client_auth_subject_dn,
568        tls_client_auth_san_dns: req.tls_client_auth_san_dns,
569        tls_client_auth_san_uri: req.tls_client_auth_san_uri,
570        self_signed_tls_client_auth_thumbprints: req.self_signed_tls_client_auth_thumbprints,
571        tls_client_certificate_bound_access_tokens: req.tls_client_certificate_bound_access_tokens,
572        jwks: req.jwks,
573        jwks_uri: req.jwks_uri,
574        dpop_bound_access_tokens: req.dpop_bound_access_tokens,
575        dpop_require_nonce: req.dpop_require_nonce,
576    };
577
578    // X5.1 — validate the MERGED result, not the patch. Flipping `profile` to
579    // `fapi2` on a client that has none of the rest is a perfectly well-formed
580    // patch and a completely broken client; validating the patch alone would
581    // wave it through. The extra read happens only when the patch touches a
582    // profile-relevant field, so an ordinary rename still costs one write.
583    if update.touches_security_profile() {
584        let merged = state
585            .oauth2_client_repo
586            .get_by_id(user.tenant_id, id)
587            .await?
588            .with_update_applied(&update);
589        axiam_oauth2::fapi::validate_registration(&merged)
590            .map_err(|e| validation_err(e.to_string()))?;
591    }
592
593    let client = state
594        .oauth2_client_repo
595        .update(user.tenant_id, id, update)
596        .await?;
597    Ok(HttpResponse::Ok().json(OAuth2ClientResponse::from(client)))
598}

600/// `DELETE /api/v1/oauth2-clients/{id}`
601#[utoipa::path(
602    delete,
603    path = "/api/v1/oauth2-clients/{id}",
604    tag = "oauth2-clients",
605    params(("id" = Uuid, Path, description = "OAuth2 client ID")),
606    responses(
607        (status = 204, description = "OAuth2 client deleted"),
608        (status = 404, description = "OAuth2 client not found"),
609    ),
610    security(("bearer" = []))
611)]
612pub async fn delete<C: Connection + Clone>(
613    user: AuthenticatedUser,
614    authz: AuthzData,
615    path: web::Path<Uuid>,
616    state: web::Data<AppState<C>>,
617) -> Result<HttpResponse, AxiamApiError> {
618    RequirePermission::new("oauth2_clients:delete", Uuid::nil())
619        .check(&user, authz.get_ref().as_ref())
620        .await?;
621    let id = path.into_inner();
622    state.oauth2_client_repo.delete(user.tenant_id, id).await?;
623    Ok(HttpResponse::NoContent().finish())
624}
