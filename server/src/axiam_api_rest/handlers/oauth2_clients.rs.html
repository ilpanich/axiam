134    /// Accepted and stored, but **nothing reads it yet**: the login hop it

137    /// anonymous browser is answered.
138    #[serde(default)]
139    pub browser_sso: bool,
140    /// T21.3 / RFC 8707 — the target services this client may name in a

143    ///
144    /// Each entry must be an absolute URI without a fragment (RFC 8707 §2).
145    /// Entries are stored in their RFC 3986 §6.2.2 normalised form, which is

148    ///
149    /// Empty (the default) means the client may name no resource, so every

167    pub backchannel_client_notification_endpoint: Option<String>,

375/// SEC-097 — refuse `dpop_require_nonce: true` while nothing reads it.
376///
377/// The field is persisted, echoed back by `GET`, and documented in the OpenAPI
378/// schema, and **no code path consults it**: the token endpoint passes
379/// `require_nonce: false` unconditionally because this deployment stores no
380/// per-client nonce to compare an echoed one against, so a challenge would
381/// prove liveness and nothing else.
382///
383/// An operator who sets a security switch, sees it persisted and echoed back,
384/// and gets nothing has been told a lie by the API. Refusing at the point of
385/// action is the same shape as SEC-101's transport check and is the only
386/// answer that does not require the operator to read `handlers/oauth2.rs` to
387/// discover the truth. The field itself is kept — removing it would be a wire
388/// break across seven SDKs for a value that is always `false` anyway — and
389/// this check is what a future nonce implementation deletes.
390fn reject_unimplemented_dpop_nonce(requested: Option<bool>) -> Result<(), AxiamApiError> {
391    if requested == Some(true) {
392        return Err(validation_err(
393            "dpop_require_nonce is not implemented in this build: nothing reads the stored \
394             value, so setting it would persist a security switch that does nothing. DPoP \
395             proofs are already made single-use at the token endpoint by jti replay \
396             detection (RFC 9449 §11.1). See docs/security-profiles.md.",
397        ));
398    }
399    Ok(())
400}

402/// Grant types a client may be registered for through this API.
403///
404/// This list MUST stay in step with what the token endpoint actually honours.
405/// Until 2026-08-15 it omitted the device-code and token-exchange grants, so
406/// `device_service::DEVICE_CODE_GRANT_TYPE` and
407/// `token_exchange::TOKEN_EXCHANGE_GRANT_TYPE` — both required by their
408/// respective flows, and both shipped — were rejected here as
409/// "unknown grant_type". That made B2 (device flow) and B3 (token exchange)
410/// unreachable through the admin API: their tests pass only because they build
411/// `OAuth2Client` values directly and never cross this validator.
412///
413/// `device_code` (the bare form) is accepted alongside the URN because
414/// `device_service` honours both.
415///
416/// Deliberately NOT included: `urn:axiam:params:oauth:grant-type:may-impersonate`.
417/// It is a capability marker, not a grant, and whether an admin-API caller may
418/// confer impersonation is a security decision that has not been taken. Leaving
419/// it out preserves the existing posture — impersonation stays settable only
420/// out-of-band.
421const KNOWN_GRANT_TYPES: &[&str] = &[
422    "authorization_code",
423    "client_credentials",
424    "refresh_token",
425    axiam_oauth2::device_service::DEVICE_CODE_GRANT_TYPE,
426    "device_code",
427    axiam_oauth2::token_exchange::TOKEN_EXCHANGE_GRANT_TYPE,
428    // G-7 — CIBA; its metadata is validated by `validate_ciba`.
429    CIBA_GRANT_TYPE,
430];

432/// G-7 — validate a registration's CIBA metadata (the rules are
433/// `axiam_oauth2::ciba::validate_client_registration`'s, shared with
434/// `POST /oauth2/register`) and return what to store.
435fn validate_ciba(
436    view: axiam_oauth2::ciba::CibaRegistrationView<'_>,
437) -> Result<CibaClientMetadata, AxiamApiError> {
438    axiam_oauth2::ciba::validate_client_registration(view)
439        .map_err(|e| validation_err(e.to_string()))
440}

442fn validation_err(msg: impl Into<String>) -> AxiamApiError {
443    axiam_core::error::AxiamError::Validation {
444        message: msg.into(),
445    }
446    .into()
447}

449/// Structural rules for a redirect URI, shared by the admin registration API
450/// and by `POST /oauth2/register` (T21.4).
451///
452/// `pub(crate)` rather than private since T21.4: dynamic registration applies
453/// the same rules and must not grow a second copy of them. What the DCR path
454/// adds on top is the tenant's host allow-list, which has no admin-API
455/// equivalent because an administrator registering a URI has already decided
456/// it is acceptable.
457pub(crate) fn validate_redirect_uris(uris: &[String]) -> Result<(), AxiamApiError> {
458    if uris.is_empty() {
459        return Err(validation_err("redirect_uris must not be empty"));
460    }
461    // The per-URI rule (absolute, https except loopback, no fragment) is the
462    // one copy in `axiam_federation::saml_sp`, shared with the SAML ACS
463    // allow-list (G-2, T23.2.1) so the two registrations can never disagree
464    // about what a registrable redirect is.
465    for uri in uris {
466        axiam_federation::saml_sp::check_redirect_uri_registration(uri).map_err(validation_err)?;
467    }
468    Ok(())
469}

471/// Validate and normalise `allowed_resources` (T21.3 / RFC 8707 §2).
472///
473/// Returns the list to store, which is the **normalised** form rather than
474/// what the operator typed. Two reasons, and the second is the one that
475/// matters: the read-back then shows the strings the server actually compares,
476/// so an operator auditing a client sees the allow-list as the request path
477/// sees it; and a duplicate that differs only in case or default port shows up
478/// as a duplicate rather than hiding as a second entry.
479///
480/// A malformed entry is a `400` naming that entry. The alternative — storing
481/// it and refusing it at request time — would give the operator a registration
482/// that looks complete and a client that gets `invalid_target` from a value
483/// the admin API accepted.
484fn validate_allowed_resources(entries: &[String]) -> Result<Vec<String>, AxiamApiError> {
485    axiam_oauth2::resource::normalise_registration(entries).map_err(|(entry, err)| {
486        validation_err(format!(
487            "invalid allowed_resources entry {entry:?}: {}",
488            err.message()
489        ))
490    })
491}

493fn validate_grant_types(grant_types: &[String]) -> Result<(), AxiamApiError> {
494    if grant_types.is_empty() {
495        return Err(validation_err("grant_types must not be empty"));
496    }
497    for gt in grant_types {
498        if !KNOWN_GRANT_TYPES.contains(&gt.as_str()) {
499            return Err(validation_err(format!(
500                "unknown grant_type: {gt} (allowed: {})",
501                KNOWN_GRANT_TYPES.join(", ")
502            )));
503        }
504    }
505    Ok(())
506}

512/// Refuse a registration that names a GDPR-sensitive scope while the tenant's
513/// capability is off (W7 / X7 G8, plan §4.8, test T8.1).
514///
515/// `axiam_oauth2::fapi::validate_registration` cannot make this check — it is
516/// a pure function of the registration, it lives four layers below this crate,
517/// and the answer depends on a settings row. So this is a second gate rather
518/// than a widening of the first, and the two refuse different things: that one
519/// refuses `address` on a `fapi2` row whatever the tenant says, this one
520/// refuses it on **any** row whose tenant has not enabled the capability.
521///
522/// Refusing at registration rather than only at the authorization endpoint is
523/// what makes the switch a switch. Without it, an operator could register the
524/// scope on a client while the capability was off and discover the refusal
525/// only when a user tried to sign in — and the request-time gate would be
526/// carrying the whole policy on its own.
527///
528/// It is *not* the only gate, and deliberately so: an operator who enables the
529/// capability, registers clients, and then disables it again leaves rows this
530/// check has already passed. `axiam_oauth2::sensitive::decide` refuses those at
531/// request time, and `release_sensitive_claims` refuses them again at UserInfo.
532async fn reject_sensitive_scopes_when_disabled<C: Connection + Clone>(
533    state: &AppState<C>,
534    tenant_id: Uuid,
535    scopes: &[String],
536) -> Result<(), AxiamApiError> {
537    let wanted = axiam_oauth2::sensitive::requested(scopes);
538    if wanted.is_empty() {
539        return Ok(());
540    }
541    let tenant =
542        axiam_core::repository::TenantRepository::get_by_id(&state.tenant_repo, tenant_id).await?;
543    let settings = axiam_core::repository::SettingsRepository::get_effective_settings(
544        &state.settings_repo,
545        tenant.organization_id,
546        tenant_id,
547    )
548    .await?;
549    if settings.oidc.sensitive_scopes_enabled {
550        return Ok(());
551    }
552    Err(validation_err(format!(
553        "the scope(s) {} release personal data and are not enabled for this tenant; \
554         an organization administrator must set sensitive_scopes_enabled before a \
555         client may register them",
556        wanted.join(", ")
557    )))
558}

560/// Grants a public client may never be registered for (T21.2).
561///
562/// Both mint a token on the strength of the client's own credential, and a
563/// public client's credential is its `client_id` — a value that travels in
564/// every authorization request it makes, in the clear, through a browser.
565/// `client_credentials` would turn that into a machine token; the RFC 8693
566/// exchange would turn it into somebody else's.
567const GRANTS_FORBIDDEN_TO_PUBLIC_CLIENTS: &[&str] = &[
568    "client_credentials",
569    axiam_oauth2::token_exchange::TOKEN_EXCHANGE_GRANT_TYPE,
570    // G-7 — CIBA Core §7.1: the client authenticates at `bc-authorize`.
571    CIBA_GRANT_TYPE,
572];
573
574/// Refuse a registration that says `none` and something else at the same time
575/// (T21.2).
576///
577/// The refusals that are *about the registration's own consistency* — a
578/// credential registered alongside `none` — live in
579/// `axiam_oauth2::fapi::validate_registration` beside the other methods'
580/// consistency rules, and the FAPI refusal needs no code at all (`none` is not
581/// `is_strong()`, so a `fapi2` profile rejects it through the gate that has
582/// always rejected `client_secret_post`, I5). What is left here is the one
583/// question that is about this API's grant vocabulary rather than about the
584/// client: which grants a public client may hold.
585fn reject_public_client_grants(
586    method: ClientAuthMethod,
587    grant_types: &[String],
588) -> Result<(), AxiamApiError> {
589    if !method.is_public() {
590        return Ok(());
591    }
592    for gt in grant_types {
593        if GRANTS_FORBIDDEN_TO_PUBLIC_CLIENTS.contains(&gt.as_str()) {
594            return Err(validation_err(format!(
595                "a public client (token_endpoint_auth_method: none) may not be registered for \
596                 the {gt} grant: it authenticates with nothing, so the only thing that grant \
597                 could rest on is a client_id that travels in every authorization request this \
598                 client makes. Register a confidential client for it instead"
599            )));
600        }
601    }
602    Ok(())
603}

619    authz: AuthzData,
620    state: web::Data<AppState<C>>,
621    body: web::Json<CreateOAuth2ClientRequest>,

624        .check(&user, authz.get_ref().as_ref())
625        .await?;
626    let req = body.into_inner();
627
628    if req.name.is_empty() {
629        return Err(validation_err("name must not be empty"));
630    }
631    validate_grant_types(&req.grant_types)?;
632    reject_public_client_grants(req.token_endpoint_auth_method, &req.grant_types)?;
633    let allowed_resources = validate_allowed_resources(&req.allowed_resources)?;
634    reject_unimplemented_dpop_nonce(Some(req.dpop_require_nonce))?;

636    // authorization_code (which involves user-agent redirects).
637    // M2M clients (client_credentials only) don't need them.

639    if needs_redirects {
640        validate_redirect_uris(&req.redirect_uris)?;
641    }

643        grant_types: &req.grant_types,
644        token_endpoint_auth_method: req.token_endpoint_auth_method,
645        profile: req.profile,
646        delivery_mode: req.backchannel_token_delivery_mode.as_deref(),
647        notification_endpoint: req.backchannel_client_notification_endpoint.as_deref(),
648        signing_alg: req
649            .backchannel_authentication_request_signing_alg
650            .as_deref(),
651        user_code_parameter: req.backchannel_user_code_parameter,
652        jwks: req.jwks.as_deref(),
653        jwks_uri: req.jwks_uri.as_deref(),
654    })?;
655
656    let create = CreateOAuth2Client {
657        tenant_id: user.tenant_id,
658        name: req.name,
659        redirect_uris: req.redirect_uris,
660        grant_types: req.grant_types,
661        scopes: req.scopes,
662        post_logout_redirect_uris: req.post_logout_redirect_uris,
663        backchannel_logout_uri: req.backchannel_logout_uri,
664        require_par: req.require_par,
665        profile: req.profile,
666        token_endpoint_auth_method: req.token_endpoint_auth_method,
667        tls_client_auth_subject_dn: req.tls_client_auth_subject_dn,
668        tls_client_auth_san_dns: req.tls_client_auth_san_dns,
669        tls_client_auth_san_uri: req.tls_client_auth_san_uri,
670        self_signed_tls_client_auth_thumbprints: req.self_signed_tls_client_auth_thumbprints,
671        tls_client_certificate_bound_access_tokens: req.tls_client_certificate_bound_access_tokens,
672        jwks: req.jwks,
673        jwks_uri: req.jwks_uri,
674        dpop_bound_access_tokens: req.dpop_bound_access_tokens,
675        dpop_require_nonce: req.dpop_require_nonce,
676        authn_request_params: req.authn_request_params,
677        browser_sso: req.browser_sso,
678        allowed_resources,
679        // T21.4 / D5 — this endpoint is the definition of an administrator's
680        // client. Not read from the request body: `CreateOAuth2ClientRequest`
681        // has no such member, so an API caller cannot claim a provenance, and
682        // this is the one handler entitled to assert `admin`.
683        managed_by: axiam_core::models::oauth2_client::ManagedBy::Admin,
684        ciba,
685    };
686
687    // X5.1 — refuse a registration that could not satisfy the profile it
688    // declares, BEFORE it is written. A `fapi2` client missing PAR or mTLS
689    // would otherwise be created and then fail every request it ever made,
690    // which is a configuration error discovered by the client's users rather
691    // than by the operator making it. A `standard` client with a secret — every
692    // client that predates X5.1 — passes without a check running.
693    axiam_oauth2::fapi::validate_registration(&create)
694        .map_err(|e| validation_err(e.to_string()))?;
695    // W7 — the tenant switch's registration half.
696    reject_sensitive_scopes_when_disabled(&state, user.tenant_id, &create.scopes).await?;
697
698    let (client, raw_secret) = state.oauth2_client_repo.create(create).await?;
699
700    // T21.2 — a public client has no secret to show. The decision is read off
701    // the stored method rather than off `raw_secret.is_empty()`: the two agree
702    // by construction (the repository mints nothing for `none`), and asking
703    // the registration means a future bug in the repository surfaces as an
704    // operator who cannot find their secret, not as a client silently treated
705    // as public.
706    let client_secret = (!client.token_endpoint_auth_method.is_public()).then_some(raw_secret);
707
708    Ok(HttpResponse::Created().json(OAuth2ClientCreatedResponse {
709        id: client.id,
710        tenant_id: client.tenant_id,
711        client_id: client.client_id,
712        client_secret,
713        name: client.name,
714        redirect_uris: client.redirect_uris,
715        grant_types: client.grant_types,
716        scopes: client.scopes,
717        created_at: client.created_at,
718        updated_at: client.updated_at,
719    }))
720}

790/// `PUT /api/v1/oauth2-clients/{id}`
791#[utoipa::path(
792    put,
793    path = "/api/v1/oauth2-clients/{id}",
794    tag = "oauth2-clients",
795    params(("id" = Uuid, Path, description = "OAuth2 client ID")),
796    request_body = UpdateOAuth2ClientRequest,
797    responses(
798        (status = 200, description = "OAuth2 client updated",
799         body = OAuth2ClientResponse),
800        (status = 404, description = "OAuth2 client not found"),
801    ),
802    security(("bearer" = []))
803)]
804pub async fn update<C: Connection + Clone>(
805    user: AuthenticatedUser,
806    authz: AuthzData,
807    path: web::Path<Uuid>,
808    state: web::Data<AppState<C>>,
809    body: web::Json<UpdateOAuth2ClientRequest>,
810) -> Result<HttpResponse, AxiamApiError> {
811    RequirePermission::new("oauth2_clients:update", Uuid::nil())
812        .check(&user, authz.get_ref().as_ref())
813        .await?;
814    let id = path.into_inner();
815    let req = body.into_inner();
816
817    if let Some(ref name) = req.name
818        && name.is_empty()
819    {
820        return Err(validation_err("name must not be empty"));
821    }
822    if let Some(ref gts) = req.grant_types {
823        validate_grant_types(gts)?;
824    }
825    reject_unimplemented_dpop_nonce(req.dpop_require_nonce)?;
826    let allowed_resources = req
827        .allowed_resources
828        .as_deref()
829        .map(validate_allowed_resources)
830        .transpose()?;
831
832    // T21.2 — `none` is a registration decision, and it stays one (I4).
833    //
834    // A PATCH may not move a client across the public/confidential line in
835    // either direction, because the secret cannot follow it. Going public
836    // would leave a live secret hash on a row whose token-endpoint arm never
837    // reads it — a credential an operator believes they revoked by changing
838    // the method, and SEC-093's exact shape. Going confidential would leave a
839    // row with no hash at all, so every authentication attempt fails on a
840    // secret that was never minted and cannot be minted by this endpoint,
841    // which only ever shows one at creation.
842    //
843    // Both directions are refused with a message that names the remedy:
844    // register the client the operator actually wants. The refusal is cheap to
845    // relax later — a rotate-secret endpoint would make the second direction
846    // meaningful — and impossible to take back once a deployment has rows that
847    // relied on it.
848    if let Some(requested_method) = req.token_endpoint_auth_method {
849        let existing = state
850            .oauth2_client_repo
851            .get_by_id(user.tenant_id, id)
852            .await?;
853        let existing_method = existing.token_endpoint_auth_method;
854        if requested_method.is_public() != existing_method.is_public() {
855            return Err(validation_err(format!(
856                "token_endpoint_auth_method cannot be changed from {} to {}: whether a client \
857                 is public is decided at registration, because the client secret cannot follow \
858                 the change — this endpoint mints one only at creation, and a stale one on a \
859                 public row would be a credential AXIAM never checks. Register a new client \
860                 with the method you want",
861                existing_method.as_str(),
862                requested_method.as_str(),
863            )));
864        }
865    }
866    if let Some(ref uris) = req.redirect_uris {
867        let needs_redirects = req
868            .grant_types
869            .as_ref()
870            .map(|gts| gts.iter().any(|g| g == "authorization_code"))
871            .unwrap_or(true);
872        if needs_redirects {
873            validate_redirect_uris(uris)?;
874        }
875    }
876
877    // When adding authorization_code to grant_types, ensure the
878    // client will have valid redirect_uris (either supplied in this
879    // request or already stored).
880    let adding_auth_code = req
881        .grant_types
882        .as_ref()
883        .is_some_and(|gts| gts.iter().any(|g| g == "authorization_code"));
884    if adding_auth_code && req.redirect_uris.is_none() {
885        let existing = state
886            .oauth2_client_repo
887            .get_by_id(user.tenant_id, id)
888            .await?;
889        if existing.redirect_uris.is_empty() {
890            return Err(validation_err(
891                "redirect_uris are required when enabling \
892                 authorization_code grant",
893            ));
894        }
895        validate_redirect_uris(&existing.redirect_uris)?;
896    }
897
898    // G-7 — the CIBA metadata is validated against the row the patch will
899    // produce whenever the patch touches anything the rules read: the grants,
900    // the method, the profile or the metadata itself. Untouched metadata is
901    // carried only while the merged grants still hold CIBA; a patch that
902    // removes the grant drops it rather than leaving it unread on the row.
903    let ciba_touched = req.backchannel_token_delivery_mode.is_some()
904        || req.backchannel_client_notification_endpoint.is_some()
905        || req.backchannel_authentication_request_signing_alg.is_some()
906        || req.backchannel_user_code_parameter.is_some();
907    // D-61: the key source is read too — removing the keys a signing
908    // algorithm is verified against must not leave the algorithm stranded.
909    let ciba = if ciba_touched
910        || req.grant_types.is_some()
911        || req.profile.is_some()
912        || req.token_endpoint_auth_method.is_some()
913        || req.jwks.is_some()
914        || req.jwks_uri.is_some()
915    {
916        let stored = state
917            .oauth2_client_repo
918            .get_by_id(user.tenant_id, id)
919            .await?;
920        let grants = req.grant_types.as_deref().unwrap_or(&stored.grant_types);
921        let keep_stored = axiam_oauth2::ciba::holds_ciba_grant(grants);
922        let stored_mode = stored
923            .ciba
924            .backchannel_token_delivery_mode
925            .filter(|_| keep_stored)
926            .map(|m| m.as_str().to_owned());
927        let stored_endpoint = stored
928            .ciba
929            .backchannel_client_notification_endpoint
930            .clone()
931            .filter(|_| keep_stored);
932        let stored_signing_alg = stored
933            .ciba
934            .backchannel_authentication_request_signing_alg
935            .filter(|_| keep_stored)
936            .map(|a| a.as_str().to_owned());
937        let mode = req.backchannel_token_delivery_mode.clone().or(stored_mode);
938        let endpoint = req
939            .backchannel_client_notification_endpoint
940            .clone()
941            .or(stored_endpoint);
942        let signing_alg = req
943            .backchannel_authentication_request_signing_alg
944            .clone()
945            .or(stored_signing_alg);
946        // `Some("")` clears, as the repository will store it.
947        let jwks = req.jwks.clone().or(stored.jwks.clone());
948        let jwks_uri = req.jwks_uri.clone().or(stored.jwks_uri.clone());
949        Some(validate_ciba(axiam_oauth2::ciba::CibaRegistrationView {
950            grant_types: grants,
951            token_endpoint_auth_method: req
952                .token_endpoint_auth_method
953                .unwrap_or(stored.token_endpoint_auth_method),
954            profile: req.profile.unwrap_or(stored.profile),
955            delivery_mode: mode.as_deref(),
956            notification_endpoint: endpoint.as_deref(),
957            signing_alg: signing_alg.as_deref(),
958            user_code_parameter: req.backchannel_user_code_parameter,
959            jwks: jwks.as_deref(),
960            jwks_uri: jwks_uri.as_deref(),
961        })?)
962    } else {
963        None
964    };
965
966    let update = UpdateOAuth2Client {
967        name: req.name,
968        redirect_uris: req.redirect_uris,
969        grant_types: req.grant_types,
970        scopes: req.scopes,
971        post_logout_redirect_uris: req.post_logout_redirect_uris,
972        backchannel_logout_uri: req.backchannel_logout_uri,
973        require_par: req.require_par,
974        profile: req.profile,
975        token_endpoint_auth_method: req.token_endpoint_auth_method,
976        tls_client_auth_subject_dn: req.tls_client_auth_subject_dn,
977        tls_client_auth_san_dns: req.tls_client_auth_san_dns,
978        tls_client_auth_san_uri: req.tls_client_auth_san_uri,
979        self_signed_tls_client_auth_thumbprints: req.self_signed_tls_client_auth_thumbprints,
980        tls_client_certificate_bound_access_tokens: req.tls_client_certificate_bound_access_tokens,
981        jwks: req.jwks,
982        jwks_uri: req.jwks_uri,
983        dpop_bound_access_tokens: req.dpop_bound_access_tokens,
984        dpop_require_nonce: req.dpop_require_nonce,
985        authn_request_params: req.authn_request_params,
986        browser_sso: req.browser_sso,
987        allowed_resources,
988        ciba,
989    };
990
991    // X5.1 — validate the MERGED result, not the patch. Flipping `profile` to
992    // `fapi2` on a client that has none of the rest is a perfectly well-formed
993    // patch and a completely broken client; validating the patch alone would
994    // wave it through. The extra read happens only when the patch touches a
995    // profile-relevant field, so an ordinary rename still costs one write.
996    if update.touches_security_profile() {
997        let merged = state
998            .oauth2_client_repo
999            .get_by_id(user.tenant_id, id)
1000            .await?
1001            .with_update_applied(&update);
1002        axiam_oauth2::fapi::validate_registration(&merged)
1003            .map_err(|e| validation_err(e.to_string()))?;
1004        // W7 — and on the update path too, on the **merged** row, so a patch
1005        // that adds `address` to a client registered without it is refused by
1006        // the same rule that would have refused it at creation. A gate that
1007        // only ran on create is a gate with an `PATCH` around it.
1008        reject_sensitive_scopes_when_disabled(&state, user.tenant_id, &merged.scopes).await?;
1009    }
1010
1011    let client = state
1012        .oauth2_client_repo
1013        .update(user.tenant_id, id, update)
1014        .await?;
1015    Ok(HttpResponse::Ok().json(OAuth2ClientResponse::from(client)))
1016}
