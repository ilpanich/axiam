391/// SEC-097 — refuse `dpop_require_nonce: true` while nothing reads it.
392///
393/// The field is persisted, echoed back by `GET`, and documented in the OpenAPI
394/// schema, and **no code path consults it**: the token endpoint passes
395/// `require_nonce: false` unconditionally because this deployment stores no
396/// per-client nonce to compare an echoed one against, so a challenge would
397/// prove liveness and nothing else.
398///
399/// An operator who sets a security switch, sees it persisted and echoed back,
400/// and gets nothing has been told a lie by the API. Refusing at the point of
401/// action is the same shape as SEC-101's transport check and is the only
402/// answer that does not require the operator to read `handlers/oauth2.rs` to
403/// discover the truth. The field itself is kept — removing it would be a wire
404/// break across seven SDKs for a value that is always `false` anyway — and
405/// this check is what a future nonce implementation deletes.
406fn reject_unimplemented_dpop_nonce(requested: Option<bool>) -> Result<(), AxiamApiError> {
407    if requested == Some(true) {
408        return Err(validation_err(
409            "dpop_require_nonce is not implemented in this build: nothing reads the stored \
410             value, so setting it would persist a security switch that does nothing. DPoP \
411             proofs are already made single-use at the token endpoint by jti replay \
412             detection (RFC 9449 §11.1). See docs/security-profiles.md.",
413        ));
414    }
415    Ok(())
416}

465/// Structural rules for a redirect URI, shared by the admin registration API
466/// and by `POST /oauth2/register` (T21.4).
467///
468/// `pub(crate)` rather than private since T21.4: dynamic registration applies
469/// the same rules and must not grow a second copy of them. What the DCR path
470/// adds on top is the tenant's host allow-list, which has no admin-API
471/// equivalent because an administrator registering a URI has already decided
472/// it is acceptable.
473pub(crate) fn validate_redirect_uris(uris: &[String]) -> Result<(), AxiamApiError> {
474    if uris.is_empty() {
475        return Err(validation_err("redirect_uris must not be empty"));
476    }
477    // The per-URI rule (absolute, https except loopback, no fragment) is the
478    // one copy in `axiam_federation::saml_sp`, shared with the SAML ACS
479    // allow-list (G-2, T23.2.1) so the two registrations can never disagree
480    // about what a registrable redirect is.
481    for uri in uris {
482        axiam_federation::saml_sp::check_redirect_uri_registration(uri).map_err(validation_err)?;
483    }
484    Ok(())
485}

488///
489/// The URI receives a signed logout token naming a user's session, so it is
490/// held to the outbound-endpoint policy every other credential-bearing target
491/// uses (D-49, `axiam_oauth2::ssf::validate_push_endpoint`, also CIBA's ping
492/// endpoint): absolute `https` with a host, no credentials, no fragment, no IP
493/// literal that is not globally routable (loopback, private, link-local, the
494/// cloud metadata address) and no `localhost`, `*.local` or `*.internal` name.
495///
496/// That is the redirect-URI rule (absolute, host, `https`, no fragment) plus
497/// the address policy, and **without** the redirect-URI rule's `http` exception
498/// for loopback: delivery goes through `guarded_fetch_no_redirect`, which
499/// requires `https` and refuses loopback at connect time, so a loopback `http`
500/// URI would be accepted here and could never be delivered. A name that
501/// *resolves* to a non-public address passes this check and is refused at
502/// delivery, which pins the address it validated.
503///
504/// An empty string is the "clear it" sentinel of the update request and is
505/// never validated; callers filter it first.
506pub(crate) fn validate_backchannel_logout_uri(uri: &str) -> Result<(), AxiamApiError> {
507    axiam_oauth2::ssf::validate_push_endpoint(uri)
508        .map(|_| ())
509        .map_err(|reason| {
510            validation_err(format!(
511                "backchannel_logout_uri: {}",
512                reason.replace("endpoint_url", "the URI")
513            ))
514        })
515}

517/// Validate and normalise `allowed_resources` (T21.3 / RFC 8707 §2).
518///
519/// Returns the list to store, which is the **normalised** form rather than
520/// what the operator typed. Two reasons, and the second is the one that
521/// matters: the read-back then shows the strings the server actually compares,
522/// so an operator auditing a client sees the allow-list as the request path
523/// sees it; and a duplicate that differs only in case or default port shows up
524/// as a duplicate rather than hiding as a second entry.
525///
526/// A malformed entry is a `400` naming that entry. The alternative — storing
527/// it and refusing it at request time — would give the operator a registration
528/// that looks complete and a client that gets `invalid_target` from a value
529/// the admin API accepted.
530fn validate_allowed_resources(entries: &[String]) -> Result<Vec<String>, AxiamApiError> {
531    axiam_oauth2::resource::normalise_registration(entries).map_err(|(entry, err)| {
532        validation_err(format!(
533            "invalid allowed_resources entry {entry:?}: {}",
534            err.message()
535        ))
536    })
537}

539fn validate_grant_types(grant_types: &[String]) -> Result<(), AxiamApiError> {
540    if grant_types.is_empty() {
541        return Err(validation_err("grant_types must not be empty"));
542    }
543    for gt in grant_types {
544        if !KNOWN_GRANT_TYPES.contains(&gt.as_str()) {
545            return Err(validation_err(format!(
546                "unknown grant_type: {gt} (allowed: {})",
547                KNOWN_GRANT_TYPES.join(", ")
548            )));
549        }
550    }
551    Ok(())
552}

558/// Refuse a registration that names a GDPR-sensitive scope while the tenant's
559/// capability is off (W7 / X7 G8, plan §4.8, test T8.1).
560///
561/// `axiam_oauth2::fapi::validate_registration` cannot make this check — it is
562/// a pure function of the registration, it lives four layers below this crate,
563/// and the answer depends on a settings row. So this is a second gate rather
564/// than a widening of the first, and the two refuse different things: that one
565/// refuses `address` on a `fapi2` row whatever the tenant says, this one
566/// refuses it on **any** row whose tenant has not enabled the capability.
567///
568/// Refusing at registration rather than only at the authorization endpoint is
569/// what makes the switch a switch. Without it, an operator could register the
570/// scope on a client while the capability was off and discover the refusal
571/// only when a user tried to sign in — and the request-time gate would be
572/// carrying the whole policy on its own.
573///
574/// It is *not* the only gate, and deliberately so: an operator who enables the
575/// capability, registers clients, and then disables it again leaves rows this
576/// check has already passed. `axiam_oauth2::sensitive::decide` refuses those at
577/// request time, and `release_sensitive_claims` refuses them again at UserInfo.
578async fn reject_sensitive_scopes_when_disabled<C: Connection + Clone>(
579    state: &AppState<C>,
580    tenant_id: Uuid,
581    scopes: &[String],
582) -> Result<(), AxiamApiError> {
583    let wanted = axiam_oauth2::sensitive::requested(scopes);
584    if wanted.is_empty() {
585        return Ok(());
586    }
587    let tenant =
588        axiam_core::repository::TenantRepository::get_by_id(&state.tenant_repo, tenant_id).await?;
589    let settings = axiam_core::repository::SettingsRepository::get_effective_settings(
590        &state.settings_repo,
591        tenant.organization_id,
592        tenant_id,
593    )
594    .await?;
595    if settings.oidc.sensitive_scopes_enabled {
596        return Ok(());
597    }
598    Err(validation_err(format!(
599        "the scope(s) {} release personal data and are not enabled for this tenant; \
600         an organization administrator must set sensitive_scopes_enabled before a \
601         client may register them",
602        wanted.join(", ")
603    )))
604}

606/// #636 (R101W3-02) — the scopes `requested` would **newly** put on a
607/// `managed_by: dcr | cimd` client that lie outside the tenant's effective
608/// `dcr_allowed_scopes`.
609///
610/// The ceiling (#613) withholds such a scope at every read point, so letting
611/// the administrator API write one is accepting what will never be served.
612/// Shares [`axiam_oauth2::scope_ceiling::applies_to`] with the read points (an
613/// administrator's client is not bounded) and the same
614/// [`SettingsScopePolicy`](crate::scope_policy::SettingsScopePolicy) that
615/// resolves organization baseline and tenant override for them, so the
616/// write-time and grant-time answers cannot disagree.
617///
618/// Only scopes the row does **not already hold** count. A scope the stored row
619/// holds beyond a ceiling that has since narrowed is already withheld at grant
620/// time — the row keeps working with the scopes inside the ceiling — so a
621/// request that merely re-sends it (a form posting the whole record back) adds
622/// nothing, and refusing it would strand an otherwise editable row. An edit
623/// that does not send `scopes` never reaches this function.
624///
625/// A failed read of the policy is an error, never "no ceiling": fail closed.
626async fn scopes_outside_ceiling<C: Connection + Clone>(
627    state: &AppState<C>,
628    stored: &OAuth2Client,
629    requested: &[String],
630) -> Result<Vec<String>, AxiamApiError> {
631    if !axiam_oauth2::scope_ceiling::applies_to(stored.managed_by) {
632        return Ok(Vec::new());
633    }
634    let policy = crate::scope_policy::SettingsScopePolicy::new(
635        state.tenant_repo.clone(),
636        state.settings_repo.clone(),
637        std::sync::Arc::clone(&state.tenant_org_cache),
638    );
639    let ceiling = axiam_oauth2::scope_ceiling::ExternalClientScopePolicy::allowed_scopes(
640        &policy,
641        stored.tenant_id,
642    )
643    .await?;
644    let mut outside: Vec<String> = Vec::new();
645    for scope in requested {
646        if !ceiling.contains(scope) && !stored.scopes.contains(scope) && !outside.contains(scope) {
647            outside.push(scope.clone());
648        }
649    }
650    Ok(outside)
651}

653/// #680 (R110W3-07) — the grants `requested` would **newly** put on a
654/// `managed_by: dcr | cimd` client that a self-registration would refuse, each
655/// with the reason.
656///
657/// Asks [`axiam_oauth2::dcr::grant_refusal`], the predicate `dcr::validate`
658/// itself runs and the one `cimd::validate` shares its grant list with, so the
659/// admin `PUT` and the registration endpoints cannot disagree. Same
660/// held-grant exemption as [`scopes_outside_ceiling`]: a grant the stored row
661/// already holds is never counted, so a form posting the whole record back
662/// still saves; the exemption keeps a grant and never adds one. An
663/// administrator's own client is not bounded. A failed read of the tenant's
664/// policy is an error, never "no policy": fail closed.
665async fn grants_refused_by_registration_policy<C: Connection + Clone>(
666    state: &AppState<C>,
667    stored: &OAuth2Client,
668    requested: &[String],
669) -> Result<Vec<String>, AxiamApiError> {
670    use axiam_core::models::oauth2_client::ManagedBy;
671    if !axiam_oauth2::scope_ceiling::applies_to(stored.managed_by) {
672        return Ok(Vec::new());
673    }
674    let mode = if stored.managed_by == ManagedBy::Dcr {
675        let tenant = axiam_core::repository::TenantRepository::get_by_id(
676            &state.tenant_repo,
677            stored.tenant_id,
678        )
679        .await?;
680        let settings = axiam_core::repository::SettingsRepository::get_effective_settings(
681            &state.settings_repo,
682            tenant.organization_id,
683            stored.tenant_id,
684        )
685        .await?;
686        Some(settings.oidc.dynamic_registration)
687    } else {
688        None
689    };
690    let mut refused: Vec<String> = Vec::new();
691    for grant in requested {
692        if stored.grant_types.contains(grant) {
693            continue;
694        }
695        if let Some(reason) = axiam_oauth2::dcr::grant_refusal(grant, mode) {
696            refused.push(reason);
697        }
698    }
699    Ok(refused)
700}

702/// Grants a public client may never be registered for (T21.2).
703///
704/// Both mint a token on the strength of the client's own credential, and a
705/// public client's credential is its `client_id` — a value that travels in
706/// every authorization request it makes, in the clear, through a browser.
707/// `client_credentials` would turn that into a machine token; the RFC 8693
708/// exchange would turn it into somebody else's.
709const GRANTS_FORBIDDEN_TO_PUBLIC_CLIENTS: &[&str] = &[
710    "client_credentials",
711    axiam_oauth2::token_exchange::TOKEN_EXCHANGE_GRANT_TYPE,
712    // G-7 — CIBA Core §7.1: the client authenticates at `bc-authorize`.
713    CIBA_GRANT_TYPE,
714];
715
716/// Refuse a registration that says `none` and something else at the same time
717/// (T21.2).
718///
719/// The refusals that are *about the registration's own consistency* — a
720/// credential registered alongside `none` — live in
721/// `axiam_oauth2::fapi::validate_registration` beside the other methods'
722/// consistency rules, and the FAPI refusal needs no code at all (`none` is not
723/// `is_strong()`, so a `fapi2` profile rejects it through the gate that has
724/// always rejected `client_secret_post`, I5). What is left here is the one
725/// question that is about this API's grant vocabulary rather than about the
726/// client: which grants a public client may hold.
727fn reject_public_client_grants(
728    method: ClientAuthMethod,
729    grant_types: &[String],
730) -> Result<(), AxiamApiError> {
731    if !method.is_public() {
732        return Ok(());
733    }
734    for gt in grant_types {
735        if GRANTS_FORBIDDEN_TO_PUBLIC_CLIENTS.contains(&gt.as_str()) {
736            return Err(validation_err(format!(
737                "a public client (token_endpoint_auth_method: none) may not be registered for \
738                 the {gt} grant: it authenticates with nothing, so the only thing that grant \
739                 could rest on is a client_id that travels in every authorization request this \
740                 client makes. Register a confidential client for it instead"
741            )));
742        }
743    }
744    Ok(())
745}

775    let allowed_resources = validate_allowed_resources(&req.allowed_resources)?;
776    reject_unimplemented_dpop_nonce(Some(req.dpop_require_nonce))?;

794        notification_endpoint: req.backchannel_client_notification_endpoint.as_deref(),
795        signing_alg: req

797            .as_deref(),
798        user_code_parameter: req.backchannel_user_code_parameter,
799        jwks: req.jwks.as_deref(),
800        jwks_uri: req.jwks_uri.as_deref(),
801    })?;
802
803    let create = CreateOAuth2Client {
804        tenant_id: user.tenant_id,
805        name: req.name,
806        redirect_uris: req.redirect_uris,
807        grant_types: req.grant_types,
808        scopes: req.scopes,
809        post_logout_redirect_uris: req.post_logout_redirect_uris,
810        backchannel_logout_uri,
811        require_par: req.require_par,
812        profile: req.profile,
813        token_endpoint_auth_method: req.token_endpoint_auth_method,

815        tls_client_auth_san_dns: req.tls_client_auth_san_dns,
816        tls_client_auth_san_uri: req.tls_client_auth_san_uri,
817        self_signed_tls_client_auth_thumbprints: req.self_signed_tls_client_auth_thumbprints,
818        tls_client_certificate_bound_access_tokens: req.tls_client_certificate_bound_access_tokens,
819        jwks: req.jwks,
820        jwks_uri: req.jwks_uri,
821        dpop_bound_access_tokens: req.dpop_bound_access_tokens,
822        dpop_require_nonce: req.dpop_require_nonce,
823        authn_request_params: req.authn_request_params,
824        browser_sso: req.browser_sso,
825        allowed_resources,
826        // T21.4 / D5 — this endpoint is the definition of an administrator's
827        // client. Not read from the request body: `CreateOAuth2ClientRequest`
828        // has no such member, so an API caller cannot claim a provenance, and
829        // this is the one handler entitled to assert `admin`.

831        ciba,
832    };

835    // declares, BEFORE it is written. A `fapi2` client missing PAR or mTLS
836    // would otherwise be created and then fail every request it ever made,
837    // which is a configuration error discovered by the client's users rather
838    // than by the operator making it. A `standard` client with a secret — every

850    // the registration means a future bug in the repository surfaces as an
851    // operator who cannot find their secret, not as a client silently treated
852    // as public.

855    Ok(HttpResponse::Created().json(OAuth2ClientCreatedResponse {

A scope or grant the row already holds is not counted. An administrator-managed client is not bounded."),
950    ),

952)]
953pub async fn update<C: Connection + Clone>(
954    http: HttpRequest,
955    user: AuthenticatedUser,
956    authz: AuthzData,
957    path: web::Path<Uuid>,
958    state: web::Data<AppState<C>>,
959    body: web::Json<UpdateOAuth2ClientRequest>,
960) -> Result<HttpResponse, AxiamApiError> {
961    RequirePermission::new("oauth2_clients:update", Uuid::nil())
962        .check(&user, authz.get_ref().as_ref())
963        .await?;
964    let id = path.into_inner();
965    let req = body.into_inner();
966
967    if let Some(ref name) = req.name
968        && name.is_empty()
969    {
970        return Err(validation_err("name must not be empty"));
971    }
972    if let Some(ref gts) = req.grant_types {
973        validate_grant_types(gts)?;
974    }
975    reject_unimplemented_dpop_nonce(req.dpop_require_nonce)?;
976    // R101W2-04 — an empty string clears the URI and is not validated; any
977    // other value is held to the same rule as on create.
978    if let Some(ref uri) = req.backchannel_logout_uri
979        && !uri.is_empty()
980    {
981        validate_backchannel_logout_uri(uri)?;
982    }
983    let allowed_resources = req
984        .allowed_resources
985        .as_deref()
986        .map(validate_allowed_resources)
987        .transpose()?;
988
989    // Item 9 of the 1.1.0 atomicity audit: every rule below is checked against
990    // ONE read of the row, and the write is conditional on that read
991    // (`update_if_unchanged`). Each rule used to read the row again and the
992    // write set only the patch's own fields, so two patches each validated
993    // against the row before the other could both land, leaving a merged row
994    // neither had validated. A row written since this read now answers `409`,
995    // and the caller reads it again. The version is the one read here, so the
996    // request carries nothing new.
997    let stored = state
998        .oauth2_client_repo
999        .get_by_id(user.tenant_id, id)
1000        .await?;
1001
1002    // #680 (R110W3-07) — a self-registered (`dcr`) or CIMD client holds no grant a
1003    // registration would refuse. Asked first, before the structural rules below, so
1004    // the answer is the policy's rather than a `400` about the grant's own shape.
1005    // It composes with the #675 public-conversion rule: that rule needs a
1006    // device-only stored row, which a `dcr`/`cimd` row can never be.
1007    if let Some(ref requested) = req.grant_types {
1008        let refused = grants_refused_by_registration_policy(&state, &stored, requested).await?;
1009        if !refused.is_empty() {
1010            return Ok(crate::error::unprocessable_entity(
1011                "grant_exceeds_registration_policy",
1012                format!(
1013                    "this client is managed_by {} and the administrator API may not give it \
1014                     a grant its registration would be refused: {}; register an \
1015                     administrator-managed client for it instead",
1016                    stored.managed_by,
1017                    refused.join("; "),
1018                ),
1019            ));
1020        }
1021    }
1022
1023    // T21.2 — `none` is a registration decision, and it stays one (I4).
1024    //
1025    // A PATCH may not move a client across the public/confidential line,
1026    // because the secret cannot follow it. Going public would leave a live
1027    // secret hash on a row whose token-endpoint arm never reads it — a
1028    // credential an operator believes they revoked by changing the method, and
1029    // SEC-093's exact shape. Going confidential would leave a row with no hash
1030    // at all, so every authentication attempt fails on a secret that was never
1031    // minted and cannot be minted by this endpoint, which only ever shows one
1032    // at creation.
1033    //
1034    // One exception (#675, D-4): a confidential client whose grants are only
1035    // the device grant and `refresh_token` may become public. Since #635 both
1036    // device endpoints authenticate the client by its registered method, so a
1037    // 1.0.x device client registered with the default `client_secret_post`
1038    // is refused at the upgrade while its devices — built to CONTRACT §14.1 —
1039    // send `client_id` alone. The conversion lets those devices keep their
1040    // `client_id`, and the objection above does not apply to it: the
1041    // repository clears `client_secret_hash` in the same write that sets
1042    // `none`, so no stale secret survives. Both the stored and the patched
1043    // grants must be device-only, so the conversion never lands on a client
1044    // that holds `client_credentials`, token exchange or CIBA, and the
1045    // operator cannot strip such a grant and go public in one request.
1046    // Every other conversion, and every conversion to confidential, is still
1047    // refused with a message that names the remedy: register the client the
1048    // operator actually wants.
1049    let mut converted_from = None;
1050    if let Some(requested_method) = req.token_endpoint_auth_method {
1051        let existing_method = stored.token_endpoint_auth_method;
1052        if requested_method.is_public() != existing_method.is_public() {
1053            let patched_grants = req.grant_types.as_deref().unwrap_or(&stored.grant_types);
1054            if requested_method.is_public()
1055                && holds_only_device_grants(&stored.grant_types)
1056                && holds_only_device_grants(patched_grants)
1057            {
1058                converted_from = Some(existing_method);
1059            } else {
1060                return Err(validation_err(format!(
1061                    "token_endpoint_auth_method cannot be changed from {} to {}: whether a \
1062                     client is public is decided at registration, because the client secret \
1063                     cannot follow the change — this endpoint mints one only at creation, and a \
1064                     stale one on a public row would be a credential AXIAM never checks. The one \
1065                     exception is a confidential client whose grant_types are only the device \
1066                     grant and refresh_token, which may be made public (its secret is cleared). \
1067                     Register a new client with the method you want",
1068                    existing_method.as_str(),
1069                    requested_method.as_str(),
1070                )));
1071            }
1072        }
1073    }
1074    if let Some(ref uris) = req.redirect_uris {
1075        let needs_redirects = req
1076            .grant_types
1077            .as_ref()
1078            .map(|gts| gts.iter().any(|g| g == "authorization_code"))
1079            .unwrap_or(true);
1080        if needs_redirects {
1081            validate_redirect_uris(uris)?;
1082        }
1083    }
1084
1085    // When adding authorization_code to grant_types, ensure the
1086    // client will have valid redirect_uris (either supplied in this
1087    // request or already stored).
1088    let adding_auth_code = req
1089        .grant_types
1090        .as_ref()
1091        .is_some_and(|gts| gts.iter().any(|g| g == "authorization_code"));
1092    if adding_auth_code && req.redirect_uris.is_none() {
1093        if stored.redirect_uris.is_empty() {
1094            return Err(validation_err(
1095                "redirect_uris are required when enabling \
1096                 authorization_code grant",
1097            ));
1098        }
1099        validate_redirect_uris(&stored.redirect_uris)?;
1100    }
1101
1102    // G-7 — the CIBA metadata is validated against the row the patch will
1103    // produce whenever the patch touches anything the rules read: the grants,
1104    // the method, the profile or the metadata itself. Untouched metadata is
1105    // carried only while the merged grants still hold CIBA; a patch that
1106    // removes the grant drops it rather than leaving it unread on the row.
1107    let ciba_touched = req.backchannel_token_delivery_mode.is_some()
1108        || req.backchannel_client_notification_endpoint.is_some()
1109        || req.backchannel_authentication_request_signing_alg.is_some()
1110        || req.backchannel_user_code_parameter.is_some();
1111    // D-61: the key source is read too — removing the keys a signing
1112    // algorithm is verified against must not leave the algorithm stranded.
1113    let ciba = if ciba_touched
1114        || req.grant_types.is_some()
1115        || req.profile.is_some()
1116        || req.token_endpoint_auth_method.is_some()
1117        || req.jwks.is_some()
1118        || req.jwks_uri.is_some()
1119    {
1120        let grants = req.grant_types.as_deref().unwrap_or(&stored.grant_types);
1121        let keep_stored = axiam_oauth2::ciba::holds_ciba_grant(grants);
1122        let stored_mode = stored
1123            .ciba
1124            .backchannel_token_delivery_mode
1125            .filter(|_| keep_stored)
1126            .map(|m| m.as_str().to_owned());
1127        let stored_endpoint = stored
1128            .ciba
1129            .backchannel_client_notification_endpoint
1130            .clone()
1131            .filter(|_| keep_stored);
1132        let stored_signing_alg = stored
1133            .ciba
1134            .backchannel_authentication_request_signing_alg
1135            .filter(|_| keep_stored)
1136            .map(|a| a.as_str().to_owned());
1137        let mode = req.backchannel_token_delivery_mode.clone().or(stored_mode);
1138        let endpoint = req
1139            .backchannel_client_notification_endpoint
1140            .clone()
1141            .or(stored_endpoint);
1142        let signing_alg = req
1143            .backchannel_authentication_request_signing_alg
1144            .clone()
1145            .or(stored_signing_alg);
1146        // `Some("")` clears, as the repository will store it.
1147        let jwks = req.jwks.clone().or(stored.jwks.clone());
1148        let jwks_uri = req.jwks_uri.clone().or(stored.jwks_uri.clone());
1149        Some(validate_ciba(axiam_oauth2::ciba::CibaRegistrationView {
1150            grant_types: grants,
1151            token_endpoint_auth_method: req
1152                .token_endpoint_auth_method
1153                .unwrap_or(stored.token_endpoint_auth_method),
1154            profile: req.profile.unwrap_or(stored.profile),
1155            delivery_mode: mode.as_deref(),
1156            notification_endpoint: endpoint.as_deref(),
1157            signing_alg: signing_alg.as_deref(),
1158            user_code_parameter: req.backchannel_user_code_parameter,
1159            jwks: jwks.as_deref(),
1160            jwks_uri: jwks_uri.as_deref(),
1161        })?)
1162    } else {
1163        None
1164    };
1165
1166    // #636 (R101W3-02) — a self-registered (`dcr`) or CIMD client holds no scope
1167    // the tenant's `dcr_allowed_scopes` does not offer. Checked before anything
1168    // is written, so a refusal leaves the row as it was.
1169    if let Some(ref requested) = req.scopes {
1170        let outside = scopes_outside_ceiling(&state, &stored, requested).await?;
1171        if !outside.is_empty() {
1172            return Ok(crate::error::unprocessable_entity(
1173                "scope_exceeds_ceiling",
1174                format!(
1175                    "the scope(s) {} are outside the tenant's dcr_allowed_scopes, and this \
1176                     client is managed_by {} so it can never be served them; add them to \
1177                     dcr_allowed_scopes first, or register an administrator-managed client",
1178                    outside.join(", "),
1179                    stored.managed_by,
1180                ),
1181            ));
1182        }
1183    }
1184
1185    let update = UpdateOAuth2Client {
1186        name: req.name,
1187        redirect_uris: req.redirect_uris,
1188        grant_types: req.grant_types,
1189        scopes: req.scopes,
1190        post_logout_redirect_uris: req.post_logout_redirect_uris,
1191        backchannel_logout_uri: req.backchannel_logout_uri,
1192        require_par: req.require_par,
1193        profile: req.profile,
1194        token_endpoint_auth_method: req.token_endpoint_auth_method,
1195        tls_client_auth_subject_dn: req.tls_client_auth_subject_dn,
1196        tls_client_auth_san_dns: req.tls_client_auth_san_dns,
1197        tls_client_auth_san_uri: req.tls_client_auth_san_uri,
1198        self_signed_tls_client_auth_thumbprints: req.self_signed_tls_client_auth_thumbprints,
1199        tls_client_certificate_bound_access_tokens: req.tls_client_certificate_bound_access_tokens,
1200        jwks: req.jwks,
1201        jwks_uri: req.jwks_uri,
1202        dpop_bound_access_tokens: req.dpop_bound_access_tokens,
1203        dpop_require_nonce: req.dpop_require_nonce,
1204        authn_request_params: req.authn_request_params,
1205        browser_sso: req.browser_sso,
1206        allowed_resources,
1207        ciba,
1208    };
1209
1210    // X5.1 — validate the MERGED result, not the patch. Flipping `profile` to
1211    // `fapi2` on a client that has none of the rest is a perfectly well-formed
1212    // patch and a completely broken client; validating the patch alone would
1213    // wave it through. The merge is onto the row read above, the one the write
1214    // is conditional on, so what is validated here is what will be stored.
1215    if update.touches_security_profile() {
1216        let merged = stored.clone().with_update_applied(&update);
1217        axiam_oauth2::fapi::validate_registration(&merged)
1218            .map_err(|e| validation_err(e.to_string()))?;
1219        // W7 — and on the update path too, on the **merged** row, so a patch
1220        // that adds `address` to a client registered without it is refused by
1221        // the same rule that would have refused it at creation. A gate that
1222        // only ran on create is a gate with an `PATCH` around it.
1223        reject_sensitive_scopes_when_disabled(&state, user.tenant_id, &merged.scopes).await?;
1224    }
1225
1226    // #749 (R111W3-01) — a conversion revokes what the client was issued while
1227    // confidential. Its refresh tokens and device grants could be redeemed only
1228    // with the secret; once the client is public the same ones redeem on
1229    // `client_id` alone. First pass before the write, and its failure refuses
1230    // the request: nothing is converted, and the retry is the whole request.
1231    // The other order could not be retried, because the repeat `PUT` finds the
1232    // client already public and is no conversion.
1233    let mut revoked = if converted_from.is_some() {
1234        revoke_confidential_era_grants(&state, user.tenant_id, &stored.client_id).await?
1235    } else {
1236        RevokedGrants::default()
1237    };
1238
1239    let client = state
1240        .oauth2_client_repo
1241        .update_if_unchanged(user.tenant_id, id, update, stored.updated_at)
1242        .await?;
1243    if let Some(from) = converted_from {
1244        // Second pass: a refresh or a device request that read the
1245        // confidential row before the write may have issued a grant since the
1246        // first. Logged and recorded rather than returned, as `delete` does:
1247        // the conversion is durable, and a `5xx` for it would be retried as a
1248        // no-op that answers `200`.
1249        let complete = match revoke_confidential_era_grants(
1250            &state,
1251            user.tenant_id,
1252            &client.client_id,
1253        )
1254        .await
1255        {
1256            Ok(again) => {
1257                revoked.refresh_tokens += again.refresh_tokens;
1258                revoked.device_grants += again.device_grants;
1259                true
1260            }
1261            Err(e) => {
1262                tracing::error!(
1263                    error = %e,
1264                    tenant_id = %user.tenant_id,
1265                    client_id = %client.client_id,
1266                    "a client converted to public could not have its grants revoked a second time"
1267                );
1268                false
1269            }
1270        };
1271        audit_conversion_to_public(&state, &http, &user, &client, from, revoked, complete).await;
1272    }
1273    Ok(HttpResponse::Ok().json(OAuth2ClientResponse::from(client)))
1274}

1285/// pushed request or CIBA request), so [`revoke_client_grants`]'s other stores
1286/// are not visited. Idempotent, so it runs before and again after the write.

1305/// #675 (D-4) — the grants a confidential client may hold and still be made
1306/// public by `PUT`: the device grant, in either spelling, and `refresh_token`.
1307const GRANTS_CONVERTIBLE_TO_PUBLIC: &[&str] = &[
1308    axiam_oauth2::device_service::DEVICE_CODE_GRANT_SPELLINGS[0],
1309    axiam_oauth2::device_service::DEVICE_CODE_GRANT_SPELLINGS[1],
1310    "refresh_token",
1311];

1313/// Whether `grant_types` holds the device grant and nothing but
1314/// [`GRANTS_CONVERTIBLE_TO_PUBLIC`].

1316    grant_types
1317        .iter()
1318        .any(|g| axiam_oauth2::device_service::DEVICE_CODE_GRANT_SPELLINGS.contains(&g.as_str()))
1319        && grant_types
1320            .iter()
1321            .all(|g| GRANTS_CONVERTIBLE_TO_PUBLIC.contains(&g.as_str()))
1322}

1368/// `DELETE /api/v1/oauth2-clients/{id}`
1369#[utoipa::path(
1370    delete,
1371    path = "/api/v1/oauth2-clients/{id}",
1372    tag = "oauth2-clients",
1373    params(("id" = Uuid, Path, description = "OAuth2 client ID")),
1374    responses(
1375        (status = 204, description = "OAuth2 client deleted; its refresh tokens, authorization codes, pushed requests, device grants and CIBA requests are revoked or deleted first"),
1376        (status = 404, description = "OAuth2 client not found"),
1377    ),
1378    security(("bearer" = []))
1379)]
1380pub async fn delete<C: Connection + Clone>(
1381    user: AuthenticatedUser,
1382    authz: AuthzData,
1383    path: web::Path<Uuid>,
1384    state: web::Data<AppState<C>>,
1385) -> Result<HttpResponse, AxiamApiError> {
1386    RequirePermission::new("oauth2_clients:delete", Uuid::nil())
1387        .check(&user, authz.get_ref().as_ref())
1388        .await?;
1389    let id = path.into_inner();
1390    let client = state
1391        .oauth2_client_repo
1392        .get_by_id(user.tenant_id, id)
1393        .await?;
1394
1395    // #517 — revoke first, and refuse the delete if that fails. The other
1396    // order would leave a row-less client whose grants are still live, and a
1397    // retry would answer `404`: for a `managed_by: cimd` client the next
1398    // request re-materialises the row under the same `client_id` and every one
1399    // of those refresh tokens, codes and pushed requests works again.
1400    revoke_client_grants(&state, user.tenant_id, &client.client_id).await?;
1401    state.oauth2_client_repo.delete(user.tenant_id, id).await?;
1402
1403    // A refresh or an authorization that read the row before the delete can
1404    // still have written a token or a code between the two writes above; one
1405    // more pass leaves nothing live.
1406    // Logged rather than returned, because the client is already gone.
1407    if let Err(e) = revoke_client_grants(&state, user.tenant_id, &client.client_id).await {
1408        tracing::error!(
1409            error = %e,
1410            tenant_id = %user.tenant_id,
1411            client_id = %client.client_id,
1412            "a deleted client's grants could not be revoked a second time"
1413        );
1414    }
1415    Ok(HttpResponse::NoContent().finish())
1416}

1419/// authorization codes, its pushed requests, and (#610 (d)) its device grants
1420/// and CIBA requests (#517). Both deletion paths

1437        axiam_db::SurrealCibaRequestRepository::new(state.db.clone(), None),
1438    )
1439    .revoke(tenant_id, client_id)
1440    .await
1441}
