84    pub self_signed_tls_client_auth_thumbprints: Vec<String>,
85    /// RFC 8705 §3.4 — issue certificate-bound (sender-constrained) access

87    #[serde(default)]
88    pub tls_client_certificate_bound_access_tokens: bool,
89    /// RFC 7591 §2 — the client's public key set, inline, for

91    #[serde(default)]
92    pub jwks: Option<String>,
93    /// RFC 7591 §2 — where the client publishes its public key set. Must be an
94    /// absolute `https` URL, and is fetched through the SSRF-guarded JWKS
95    /// cache, which refuses private and loopback addresses.
96    #[serde(default)]
97    pub jwks_uri: Option<String>,
98    /// RFC 9449 §5.2 — issue DPoP-bound (sender-constrained) access tokens to
99    /// this client. Independent of both the authentication method and
100    /// `tls_client_certificate_bound_access_tokens`; a client may ask for both
101    /// constraints, and a token carrying both must satisfy both.
102    #[serde(default)]
103    pub dpop_bound_access_tokens: bool,
104    /// RFC 9449 §8 — require this client's DPoP proofs to carry a
105    /// server-issued nonce.
106    ///
107    /// **Not implemented in this build (SEC-097).** `true` is refused with
108    /// `400`; only `false` (the default) is accepted. Nothing reads the stored
109    /// value, so accepting `true` would persist and echo back a security
110    /// switch that does nothing. DPoP proofs are made single-use at the token
111    /// endpoint by `jti` replay detection instead — see
112    /// `docs/security-profiles.md`.
113    #[serde(default)]
114    pub dpop_require_nonce: bool,
115    /// X7.1 — whether this client's authorization requests may carry the
116    /// OpenID Connect authentication-request parameters (`prompt`, `max_age`,

118    /// `ui_locales`, `claims_locales`).
119    ///
120    /// `"ignore"` (the default) is what every AXIAM client has always done:
121    /// they are dropped and reach no decision. `"honour"` opts in, and is
122    /// **refused on a `fapi2` client** at both this gate and the authorization
123    /// endpoint — the two are different answers to the same question about
124    /// what a request from this client means.
125    #[serde(default)]
126    pub authn_request_params: AuthnRequestParamsMode,
127    /// X7.3 — whether an unauthenticated authorization request from this
128    /// client may be answered with a redirect to the login page rather than
129    /// the `401` AXIAM answers today.
130    ///
131    /// Accepted and stored, but **nothing reads it yet**: the login hop it
132    /// gates is a later wave. Unlike `authn_request_params` it is permitted on
133    /// a `fapi2` client, because it relaxes nothing — it decides only how an
134    /// anonymous browser is answered.
135    #[serde(default)]
136    pub browser_sso: bool,
137    /// T21.3 / RFC 8707 — the target services this client may name in a
138    /// `resource` parameter, at `/oauth2/authorize`, `/oauth2/par`,
139    /// `/oauth2/device_authorization` and `/oauth2/token`.
140    ///
141    /// Each entry must be an absolute URI without a fragment (RFC 8707 §2).
142    /// Entries are stored in their RFC 3986 §6.2.2 normalised form, which is
143    /// what the read-back shows and what every comparison uses; matching is by
144    /// equivalence and **never by prefix**.
145    ///
146    /// Empty (the default) means the client may name no resource, so every
147    /// token it obtains carries `axiam:user` or `axiam:m2m` exactly as before
148    /// RFC 8707 support existed. This is also the list the RFC 8693 token
149    /// exchange consults for its `audience`/`resource` target.
150    #[serde(default)]
151    pub allowed_resources: Vec<String>,
152}

323/// SEC-097 — refuse `dpop_require_nonce: true` while nothing reads it.
324///
325/// The field is persisted, echoed back by `GET`, and documented in the OpenAPI
326/// schema, and **no code path consults it**: the token endpoint passes
327/// `require_nonce: false` unconditionally because this deployment stores no
328/// per-client nonce to compare an echoed one against, so a challenge would
329/// prove liveness and nothing else.
330///
331/// An operator who sets a security switch, sees it persisted and echoed back,
332/// and gets nothing has been told a lie by the API. Refusing at the point of
333/// action is the same shape as SEC-101's transport check and is the only
334/// answer that does not require the operator to read `handlers/oauth2.rs` to
335/// discover the truth. The field itself is kept — removing it would be a wire
336/// break across seven SDKs for a value that is always `false` anyway — and
337/// this check is what a future nonce implementation deletes.
338fn reject_unimplemented_dpop_nonce(requested: Option<bool>) -> Result<(), AxiamApiError> {
339    if requested == Some(true) {
340        return Err(validation_err(
341            "dpop_require_nonce is not implemented in this build: nothing reads the stored \
342             value, so setting it would persist a security switch that does nothing. DPoP \
343             proofs are already made single-use at the token endpoint by jti replay \
344             detection (RFC 9449 §11.1). See docs/security-profiles.md.",
345        ));
346    }
347    Ok(())
348}

350/// Grant types a client may be registered for through this API.
351///
352/// This list MUST stay in step with what the token endpoint actually honours.
353/// Until 2026-08-15 it omitted the device-code and token-exchange grants, so
354/// `device_service::DEVICE_CODE_GRANT_TYPE` and
355/// `token_exchange::TOKEN_EXCHANGE_GRANT_TYPE` — both required by their
356/// respective flows, and both shipped — were rejected here as
357/// "unknown grant_type". That made B2 (device flow) and B3 (token exchange)
358/// unreachable through the admin API: their tests pass only because they build
359/// `OAuth2Client` values directly and never cross this validator.
360///
361/// `device_code` (the bare form) is accepted alongside the URN because
362/// `device_service` honours both.
363///
364/// Deliberately NOT included: `urn:axiam:params:oauth:grant-type:may-impersonate`.
365/// It is a capability marker, not a grant, and whether an admin-API caller may
366/// confer impersonation is a security decision that has not been taken. Leaving
367/// it out preserves the existing posture — impersonation stays settable only
368/// out-of-band.
369const KNOWN_GRANT_TYPES: &[&str] = &[
370    "authorization_code",
371    "client_credentials",
372    "refresh_token",
373    axiam_oauth2::device_service::DEVICE_CODE_GRANT_TYPE,
374    "device_code",
375    axiam_oauth2::token_exchange::TOKEN_EXCHANGE_GRANT_TYPE,
376];
377
378fn validation_err(msg: impl Into<String>) -> AxiamApiError {
379    axiam_core::error::AxiamError::Validation {
380        message: msg.into(),
381    }
382    .into()
383}

385/// Structural rules for a redirect URI, shared by the admin registration API
386/// and by `POST /oauth2/register` (T21.4).
387///
388/// `pub(crate)` rather than private since T21.4: dynamic registration applies
389/// the same rules and must not grow a second copy of them. What the DCR path
390/// adds on top is the tenant's host allow-list, which has no admin-API
391/// equivalent because an administrator registering a URI has already decided
392/// it is acceptable.
393pub(crate) fn validate_redirect_uris(uris: &[String]) -> Result<(), AxiamApiError> {
394    if uris.is_empty() {
395        return Err(validation_err("redirect_uris must not be empty"));
396    }
397    for uri in uris {
398        let parsed: url::Url = uri
399            .parse()
400            .map_err(|_| validation_err(format!("invalid redirect_uri: {uri}")))?;
401        // Redirect URIs must be absolute with an authority (host)
402        let host = parsed.host_str().ok_or_else(|| {
403            validation_err(format!(
404                "redirect_uri must be an absolute URL with a host: {uri}"
405            ))
406        })?;
407        // Allow http for localhost/loopback only, require HTTPS otherwise.
408        //
409        // `[::1]`, with the brackets, is what `Url::host_str` returns for an
410        // IPv6 literal — the bare `::1` this line compared against until T21.8
411        // is a spelling no parser produces, so the IPv6 arm was unreachable and
412        // `http://[::1]/callback` was refused by both registration endpoints
413        // while the error message named it as allowed. The rest of the stack
414        // always spelled it with brackets: the redirect matcher's loopback arm
415        // (`axiam_oauth2::redirect_uri`), the DCR host allow-list
416        // (`axiam_oauth2::dcr::ALWAYS_ALLOWED_REDIRECT_HOSTS`) and the CIMD
417        // document validator all do, and all are tested on it. RFC 8252 §7.3
418        // lists the IPv6 loopback beside `127.0.0.1`.
419        let is_localhost = host == "localhost" || host == "127.0.0.1" || host == "[::1]";
420        if parsed.scheme() != "https" && !(parsed.scheme() == "http" && is_localhost) {
421            return Err(validation_err(format!(
422                "redirect_uri must use https (http is only allowed for localhost/127.0.0.1/[::1]): {uri}"
423            )));
424        }
425        // RFC 6749 §3.1.2: redirect URIs must not include a fragment
426        if parsed.fragment().is_some() {
427            return Err(validation_err(format!(
428                "redirect_uri must not contain a fragment: {uri}"
429            )));
430        }
431    }
432    Ok(())
433}

435/// Validate and normalise `allowed_resources` (T21.3 / RFC 8707 §2).
436///
437/// Returns the list to store, which is the **normalised** form rather than
438/// what the operator typed. Two reasons, and the second is the one that
439/// matters: the read-back then shows the strings the server actually compares,
440/// so an operator auditing a client sees the allow-list as the request path
441/// sees it; and a duplicate that differs only in case or default port shows up
442/// as a duplicate rather than hiding as a second entry.
443///
444/// A malformed entry is a `400` naming that entry. The alternative — storing
445/// it and refusing it at request time — would give the operator a registration
446/// that looks complete and a client that gets `invalid_target` from a value
447/// the admin API accepted.
448fn validate_allowed_resources(entries: &[String]) -> Result<Vec<String>, AxiamApiError> {
449    axiam_oauth2::resource::normalise_registration(entries).map_err(|(entry, err)| {
450        validation_err(format!(
451            "invalid allowed_resources entry {entry:?}: {}",
452            err.message()
453        ))
454    })
455}
456
457fn validate_grant_types(grant_types: &[String]) -> Result<(), AxiamApiError> {
458    if grant_types.is_empty() {
459        return Err(validation_err("grant_types must not be empty"));
460    }
461    for gt in grant_types {
462        if !KNOWN_GRANT_TYPES.contains(&gt.as_str()) {
463            return Err(validation_err(format!(
464                "unknown grant_type: {gt} (allowed: {})",
465                KNOWN_GRANT_TYPES.join(", ")
466            )));
467        }
468    }
469    Ok(())
470}

476/// Refuse a registration that names a GDPR-sensitive scope while the tenant's
477/// capability is off (W7 / X7 G8, plan §4.8, test T8.1).
478///
479/// `axiam_oauth2::fapi::validate_registration` cannot make this check — it is
480/// a pure function of the registration, it lives four layers below this crate,
481/// and the answer depends on a settings row. So this is a second gate rather
482/// than a widening of the first, and the two refuse different things: that one
483/// refuses `address` on a `fapi2` row whatever the tenant says, this one
484/// refuses it on **any** row whose tenant has not enabled the capability.
485///
486/// Refusing at registration rather than only at the authorization endpoint is
487/// what makes the switch a switch. Without it, an operator could register the
488/// scope on a client while the capability was off and discover the refusal
489/// only when a user tried to sign in — and the request-time gate would be
490/// carrying the whole policy on its own.
491///
492/// It is *not* the only gate, and deliberately so: an operator who enables the
493/// capability, registers clients, and then disables it again leaves rows this
494/// check has already passed. `axiam_oauth2::sensitive::decide` refuses those at
495/// request time, and `release_sensitive_claims` refuses them again at UserInfo.
496async fn reject_sensitive_scopes_when_disabled<C: Connection + Clone>(
497    state: &AppState<C>,
498    tenant_id: Uuid,
499    scopes: &[String],
500) -> Result<(), AxiamApiError> {
501    let wanted = axiam_oauth2::sensitive::requested(scopes);
502    if wanted.is_empty() {
503        return Ok(());
504    }
505    let tenant =
506        axiam_core::repository::TenantRepository::get_by_id(&state.tenant_repo, tenant_id).await?;
507    let settings = axiam_core::repository::SettingsRepository::get_effective_settings(
508        &state.settings_repo,
509        tenant.organization_id,
510        tenant_id,
511    )
512    .await?;
513    if settings.oidc.sensitive_scopes_enabled {
514        return Ok(());
515    }
516    Err(validation_err(format!(
517        "the scope(s) {} release personal data and are not enabled for this tenant; \
518         an organization administrator must set sensitive_scopes_enabled before a \
519         client may register them",
520        wanted.join(", ")
521    )))
522}

524/// Grants a public client may never be registered for (T21.2).
525///
526/// Both mint a token on the strength of the client's own credential, and a
527/// public client's credential is its `client_id` — a value that travels in
528/// every authorization request it makes, in the clear, through a browser.
529/// `client_credentials` would turn that into a machine token; the RFC 8693
530/// exchange would turn it into somebody else's.
531const GRANTS_FORBIDDEN_TO_PUBLIC_CLIENTS: &[&str] = &[
532    "client_credentials",
533    axiam_oauth2::token_exchange::TOKEN_EXCHANGE_GRANT_TYPE,
534];
535
536/// Refuse a registration that says `none` and something else at the same time
537/// (T21.2).
538///
539/// The refusals that are *about the registration's own consistency* — a
540/// credential registered alongside `none` — live in
541/// `axiam_oauth2::fapi::validate_registration` beside the other methods'
542/// consistency rules, and the FAPI refusal needs no code at all (`none` is not
543/// `is_strong()`, so a `fapi2` profile rejects it through the gate that has
544/// always rejected `client_secret_post`, I5). What is left here is the one
545/// question that is about this API's grant vocabulary rather than about the
546/// client: which grants a public client may hold.
547fn reject_public_client_grants(
548    method: ClientAuthMethod,
549    grant_types: &[String],
550) -> Result<(), AxiamApiError> {
551    if !method.is_public() {
552        return Ok(());
553    }
554    for gt in grant_types {
555        if GRANTS_FORBIDDEN_TO_PUBLIC_CLIENTS.contains(&gt.as_str()) {
556            return Err(validation_err(format!(
557                "a public client (token_endpoint_auth_method: none) may not be registered for \
558                 the {gt} grant: it authenticates with nothing, so the only thing that grant \
559                 could rest on is a client_id that travels in every authorization request this \
560                 client makes. Register a confidential client for it instead"
561            )));
562        }
563    }
564    Ok(())
565}

567/// `POST /api/v1/oauth2-clients`
568#[utoipa::path(
569    post,
570    path = "/api/v1/oauth2-clients",
571    tag = "oauth2-clients",
572    request_body = CreateOAuth2ClientRequest,
573    responses(
574        (status = 201, description = "OAuth2 client created (secret shown once)",
575         body = OAuth2ClientCreatedResponse),
576    ),
577    security(("bearer" = []))
578)]
579pub async fn create<C: Connection + Clone>(
580    user: AuthenticatedUser,
581    authz: AuthzData,
582    state: web::Data<AppState<C>>,
583    body: web::Json<CreateOAuth2ClientRequest>,
584) -> Result<HttpResponse, AxiamApiError> {
585    RequirePermission::new("oauth2_clients:create", Uuid::nil())
586        .check(&user, authz.get_ref().as_ref())
587        .await?;
588    let req = body.into_inner();
589
590    if req.name.is_empty() {
591        return Err(validation_err("name must not be empty"));
592    }
593    validate_grant_types(&req.grant_types)?;
594    reject_public_client_grants(req.token_endpoint_auth_method, &req.grant_types)?;
595    let allowed_resources = validate_allowed_resources(&req.allowed_resources)?;
596    reject_unimplemented_dpop_nonce(Some(req.dpop_require_nonce))?;
597    // redirect_uris are only required when the client uses
598    // authorization_code (which involves user-agent redirects).
599    // M2M clients (client_credentials only) don't need them.
600    let needs_redirects = req.grant_types.iter().any(|g| g == "authorization_code");
601    if needs_redirects {
602        validate_redirect_uris(&req.redirect_uris)?;
603    }
604
605    let create = CreateOAuth2Client {
606        tenant_id: user.tenant_id,
607        name: req.name,
608        redirect_uris: req.redirect_uris,
609        grant_types: req.grant_types,
610        scopes: req.scopes,
611        post_logout_redirect_uris: req.post_logout_redirect_uris,
612        backchannel_logout_uri: req.backchannel_logout_uri,
613        require_par: req.require_par,
614        profile: req.profile,
615        token_endpoint_auth_method: req.token_endpoint_auth_method,
616        tls_client_auth_subject_dn: req.tls_client_auth_subject_dn,
617        tls_client_auth_san_dns: req.tls_client_auth_san_dns,
618        tls_client_auth_san_uri: req.tls_client_auth_san_uri,
619        self_signed_tls_client_auth_thumbprints: req.self_signed_tls_client_auth_thumbprints,
620        tls_client_certificate_bound_access_tokens: req.tls_client_certificate_bound_access_tokens,
621        jwks: req.jwks,
622        jwks_uri: req.jwks_uri,
623        dpop_bound_access_tokens: req.dpop_bound_access_tokens,
624        dpop_require_nonce: req.dpop_require_nonce,
625        authn_request_params: req.authn_request_params,
626        browser_sso: req.browser_sso,
627        allowed_resources,
628        // T21.4 / D5 — this endpoint is the definition of an administrator's
629        // client. Not read from the request body: `CreateOAuth2ClientRequest`
630        // has no such member, so an API caller cannot claim a provenance, and
631        // this is the one handler entitled to assert `admin`.
632        managed_by: axiam_core::models::oauth2_client::ManagedBy::Admin,
633    };
634
635    // X5.1 — refuse a registration that could not satisfy the profile it
636    // declares, BEFORE it is written. A `fapi2` client missing PAR or mTLS
637    // would otherwise be created and then fail every request it ever made,
638    // which is a configuration error discovered by the client's users rather
639    // than by the operator making it. A `standard` client with a secret — every
640    // client that predates X5.1 — passes without a check running.
641    axiam_oauth2::fapi::validate_registration(&create)
642        .map_err(|e| validation_err(e.to_string()))?;
643    // W7 — the tenant switch's registration half.
644    reject_sensitive_scopes_when_disabled(&state, user.tenant_id, &create.scopes).await?;
645
646    let (client, raw_secret) = state.oauth2_client_repo.create(create).await?;
647
648    // T21.2 — a public client has no secret to show. The decision is read off
649    // the stored method rather than off `raw_secret.is_empty()`: the two agree
650    // by construction (the repository mints nothing for `none`), and asking
651    // the registration means a future bug in the repository surfaces as an
652    // operator who cannot find their secret, not as a client silently treated
653    // as public.
654    let client_secret = (!client.token_endpoint_auth_method.is_public()).then_some(raw_secret);
655
656    Ok(HttpResponse::Created().json(OAuth2ClientCreatedResponse {
657        id: client.id,
658        tenant_id: client.tenant_id,
659        client_id: client.client_id,
660        client_secret,
661        name: client.name,
662        redirect_uris: client.redirect_uris,
663        grant_types: client.grant_types,
664        scopes: client.scopes,
665        created_at: client.created_at,
666        updated_at: client.updated_at,
667    }))
668}

738/// `PUT /api/v1/oauth2-clients/{id}`
739#[utoipa::path(
740    put,
741    path = "/api/v1/oauth2-clients/{id}",
742    tag = "oauth2-clients",
743    params(("id" = Uuid, Path, description = "OAuth2 client ID")),
744    request_body = UpdateOAuth2ClientRequest,
745    responses(
746        (status = 200, description = "OAuth2 client updated",
747         body = OAuth2ClientResponse),
748        (status = 404, description = "OAuth2 client not found"),
749    ),
750    security(("bearer" = []))
751)]
752pub async fn update<C: Connection + Clone>(
753    user: AuthenticatedUser,
754    authz: AuthzData,
755    path: web::Path<Uuid>,
756    state: web::Data<AppState<C>>,
757    body: web::Json<UpdateOAuth2ClientRequest>,
758) -> Result<HttpResponse, AxiamApiError> {
759    RequirePermission::new("oauth2_clients:update", Uuid::nil())
760        .check(&user, authz.get_ref().as_ref())
761        .await?;
762    let id = path.into_inner();
763    let req = body.into_inner();
764
765    if let Some(ref name) = req.name
766        && name.is_empty()
767    {
768        return Err(validation_err("name must not be empty"));
769    }
770    if let Some(ref gts) = req.grant_types {
771        validate_grant_types(gts)?;
772    }
773    reject_unimplemented_dpop_nonce(req.dpop_require_nonce)?;
774    let allowed_resources = req
775        .allowed_resources
776        .as_deref()
777        .map(validate_allowed_resources)
778        .transpose()?;
779
780    // T21.2 — `none` is a registration decision, and it stays one (I4).
781    //
782    // A PATCH may not move a client across the public/confidential line in
783    // either direction, because the secret cannot follow it. Going public
784    // would leave a live secret hash on a row whose token-endpoint arm never
785    // reads it — a credential an operator believes they revoked by changing
786    // the method, and SEC-093's exact shape. Going confidential would leave a
787    // row with no hash at all, so every authentication attempt fails on a
788    // secret that was never minted and cannot be minted by this endpoint,
789    // which only ever shows one at creation.
790    //
791    // Both directions are refused with a message that names the remedy:
792    // register the client the operator actually wants. The refusal is cheap to
793    // relax later — a rotate-secret endpoint would make the second direction
794    // meaningful — and impossible to take back once a deployment has rows that
795    // relied on it.
796    if let Some(requested_method) = req.token_endpoint_auth_method {
797        let existing = state
798            .oauth2_client_repo
799            .get_by_id(user.tenant_id, id)
800            .await?;
801        let existing_method = existing.token_endpoint_auth_method;
802        if requested_method.is_public() != existing_method.is_public() {
803            return Err(validation_err(format!(
804                "token_endpoint_auth_method cannot be changed from {} to {}: whether a client \
805                 is public is decided at registration, because the client secret cannot follow \
806                 the change — this endpoint mints one only at creation, and a stale one on a \
807                 public row would be a credential AXIAM never checks. Register a new client \
808                 with the method you want",
809                existing_method.as_str(),
810                requested_method.as_str(),
811            )));
812        }
813    }
814    if let Some(ref uris) = req.redirect_uris {
815        let needs_redirects = req
816            .grant_types
817            .as_ref()
818            .map(|gts| gts.iter().any(|g| g == "authorization_code"))
819            .unwrap_or(true);
820        if needs_redirects {
821            validate_redirect_uris(uris)?;
822        }
823    }
824
825    // When adding authorization_code to grant_types, ensure the
826    // client will have valid redirect_uris (either supplied in this
827    // request or already stored).
828    let adding_auth_code = req
829        .grant_types
830        .as_ref()
831        .is_some_and(|gts| gts.iter().any(|g| g == "authorization_code"));
832    if adding_auth_code && req.redirect_uris.is_none() {
833        let existing = state
834            .oauth2_client_repo
835            .get_by_id(user.tenant_id, id)
836            .await?;
837        if existing.redirect_uris.is_empty() {
838            return Err(validation_err(
839                "redirect_uris are required when enabling \
840                 authorization_code grant",
841            ));
842        }
843        validate_redirect_uris(&existing.redirect_uris)?;
844    }
845
846    let update = UpdateOAuth2Client {
847        name: req.name,
848        redirect_uris: req.redirect_uris,
849        grant_types: req.grant_types,
850        scopes: req.scopes,
851        post_logout_redirect_uris: req.post_logout_redirect_uris,
852        backchannel_logout_uri: req.backchannel_logout_uri,
853        require_par: req.require_par,
854        profile: req.profile,
855        token_endpoint_auth_method: req.token_endpoint_auth_method,
856        tls_client_auth_subject_dn: req.tls_client_auth_subject_dn,
857        tls_client_auth_san_dns: req.tls_client_auth_san_dns,
858        tls_client_auth_san_uri: req.tls_client_auth_san_uri,
859        self_signed_tls_client_auth_thumbprints: req.self_signed_tls_client_auth_thumbprints,
860        tls_client_certificate_bound_access_tokens: req.tls_client_certificate_bound_access_tokens,
861        jwks: req.jwks,
862        jwks_uri: req.jwks_uri,
863        dpop_bound_access_tokens: req.dpop_bound_access_tokens,
864        dpop_require_nonce: req.dpop_require_nonce,
865        authn_request_params: req.authn_request_params,
866        browser_sso: req.browser_sso,
867        allowed_resources,
868    };
869
870    // X5.1 — validate the MERGED result, not the patch. Flipping `profile` to
871    // `fapi2` on a client that has none of the rest is a perfectly well-formed
872    // patch and a completely broken client; validating the patch alone would
873    // wave it through. The extra read happens only when the patch touches a
874    // profile-relevant field, so an ordinary rename still costs one write.
875    if update.touches_security_profile() {
876        let merged = state
877            .oauth2_client_repo
878            .get_by_id(user.tenant_id, id)
879            .await?
880            .with_update_applied(&update);
881        axiam_oauth2::fapi::validate_registration(&merged)
882            .map_err(|e| validation_err(e.to_string()))?;
883        // W7 — and on the update path too, on the **merged** row, so a patch
884        // that adds `address` to a client registered without it is refused by
885        // the same rule that would have refused it at creation. A gate that
886        // only ran on create is a gate with an `PATCH` around it.
887        reject_sensitive_scopes_when_disabled(&state, user.tenant_id, &merged.scopes).await?;
888    }
889
890    let client = state
891        .oauth2_client_repo
892        .update(user.tenant_id, id, update)
893        .await?;
894    Ok(HttpResponse::Ok().json(OAuth2ClientResponse::from(client)))
895}
