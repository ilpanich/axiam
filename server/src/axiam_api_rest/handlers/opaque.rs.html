1//! OPAQUE endpoints — `POST /api/v1/auth/opaque/register/start`,
2//! `POST /api/v1/auth/opaque/login/start` and
3//! `POST /api/v1/auth/opaque/login/finish`.
4//!
5//! The login pair is a *sibling* of `POST /api/v1/auth/login`, not a
6//! replacement. They establish the same fact — this principal knows the
7//! password — by a route that never puts the password on the wire, and then
8//! hand off to the exact same post-credential path
9//! ([`axiam_auth::AuthService::complete_authenticated_login`]), so MFA policy,
10//! account status, the `login.post_auth` reactor hook, lockout accounting and
11//! session issuance are identical on both. That shared tail is the point: two
12//! parallel implementations of "what happens after a successful login" is how
13//! a deployment ends up with MFA enforced on one path and not the other.
14//!
15//! # Why there is no `register/finish` endpoint
16//!
17//! A registration record can only be built at a moment when the plaintext
18//! password legitimately exists on the client, and every one of those moments
19//! — user creation, an authenticated password change, reset completion,
20//! first-run bootstrap — is already an endpoint that takes a password. Each of
21//! those grew an `opaque` field carrying the finished record, exactly where
22//! the SRP implementation carried an `srp` verifier. A free-standing finish
23//! would be an endpoint whose only job is to attach a credential to an
24//! account, which is a thing worth not having.
25//!
26//! # Enumeration
27//!
28//! `/auth/opaque/login/start` answers `200` for every syntactically valid
29//! request, including one naming a user who does not exist, a user in another
30//! tenant, or a user with no record. The decoy exchange is described in
31//! [`axiam_auth::opaque`]. Anything else here would make this endpoint a
32//! faster account-enumeration oracle than the rest of the API, which is
33//! exactly the kind of regression that "we added a security feature" is
34//! supposed to preclude.
35//!
36//! The one case that is *not* disguised is `opaque_mode = disabled`, which
37//! returns `404`: that is a property of the tenant, not of any user, so
38//! revealing it leaks nothing about who exists.
39

65/// `POST /api/v1/auth/opaque/register/start`
66#[utoipa::path(
67    post,
68    path = "/api/v1/auth/opaque/register/start",
69    tag = "auth",
70    request_body = OpaqueRegisterStartRequest,
71    responses(
72        (status = 200, description = "OPRF evaluation and the KSF parameters to stretch with", body = axiam_core::models::opaque::OpaqueRegisterStartResponse),
73        (status = 400, description = "Malformed registration request or missing workspace identity"),
74        (status = 404, description = "OPAQUE is not enabled for this tenant"),
75        (status = 503, description = "OPAQUE is enabled but the server has no keys configured"),
76    )
77)]
78pub async fn opaque_register_start<C: Connection + Clone>(
79    state: web::Data<AppState<C>>,
80    body: web::Json<OpaqueRegisterStartRequest>,
81) -> Result<HttpResponse, AxiamApiError> {
82    let b = body.into_inner();
83    let (org_id, tenant_id) = resolve_workspace(
84        &state,
85        b.org_id,
86        b.org_slug.as_deref(),
87        b.tenant_id,
88        b.tenant_slug.as_deref(),
89    )
90    .await?;
91
92    let settings = state
93        .settings_repo
94        .get_effective_settings(org_id, tenant_id)
95        .await?;
96    let policy = &settings.opaque;
97
98    if policy.opaque_mode == OpaqueMode::Disabled {
99        return Err(AxiamApiError(AxiamError::NotFound {
100            entity: "opaque".into(),
101            id: "disabled".into(),
102        }));
103    }
104
105    let opaque = opaque_server(&state)?;
106    let setup = server_setup(&state, tenant_id, policy.opaque_suite).await?;
107
108    let response = opaque
109        .register_start(
110            &setup,
111            OpaqueKsfParams::defaults_for(policy.opaque_ksf),
112            &b.registration_request,
113        )
114        .map_err(AxiamError::from)?;
115
116    Ok(HttpResponse::Ok().json(response))
117}

119/// `POST /api/v1/auth/opaque/login/start`
120#[utoipa::path(
121    post,
122    path = "/api/v1/auth/opaque/login/start",
123    tag = "auth",
124    request_body = OpaqueLoginStartRequest,
125    responses(
126        (status = 200, description = "KE2 (identical shape for unknown identities)", body = axiam_core::models::opaque::OpaqueLoginStartResponse),
127        (status = 400, description = "Malformed KE1 or missing workspace identity"),
128        (status = 404, description = "OPAQUE is not enabled for this tenant"),
129        (status = 503, description = "OPAQUE is enabled but the server has no keys configured"),
130    )
131)]
132pub async fn opaque_login_start<C: Connection + Clone>(
133    state: web::Data<AppState<C>>,
134    body: web::Json<OpaqueLoginStartRequest>,
135) -> Result<HttpResponse, AxiamApiError> {
136    let b = body.into_inner();
137    let (org_id, tenant_id) = resolve_workspace(
138        &state,
139        b.org_id,
140        b.org_slug.as_deref(),
141        b.tenant_id,
142        b.tenant_slug.as_deref(),
143    )
144    .await?;
145
146    let settings = state
147        .settings_repo
148        .get_effective_settings(org_id, tenant_id)
149        .await?;
150    let policy = &settings.opaque;
151
152    if policy.opaque_mode == OpaqueMode::Disabled {
153        return Err(AxiamApiError(AxiamError::NotFound {
154            entity: "opaque".into(),
155            id: "disabled".into(),
156        }));
157    }
158
159    let opaque = opaque_server(&state)?;
160    let setup = server_setup(&state, tenant_id, policy.opaque_suite).await?;
161
162    // Look up the user; a miss at any step falls through to the decoy branch
163    // rather than returning an error, so all four outcomes — unknown name,
164    // wrong tenant, known user without a record, known user with one — are
165    // indistinguishable from outside.
166    let credential = match lookup_user_id(&state, tenant_id, &b.username_or_email).await {
167        Some(user_id) => state
168            .opaque_credential_repo
169            .get_by_user(tenant_id, user_id)
170            .await
171            .ok(),
172        None => None,
173    };
174
175    let started = match credential {
176        Some(cred) => opaque
177            .login_start(&setup, &cred, org_id, &b.ke1)
178            .map_err(AxiamError::from)?,
179        None => opaque
180            .login_start_decoy(
181                &setup,
182                OpaqueKsfParams::defaults_for(policy.opaque_ksf),
183                &b.username_or_email,
184                tenant_id,
185                org_id,
186                &b.ke1,
187            )
188            .map_err(AxiamError::from)?,
189    };
190
191    Ok(HttpResponse::Ok().json(started))
192}

194/// `POST /api/v1/auth/opaque/login/finish`
195///
196/// Returns the same three outcomes as `POST /api/v1/auth/login` — `200` with
197/// session cookies, `202` for an MFA challenge, `403` for MFA setup — so a
198/// client can share one result handler across both login paths.
199///
200/// Note what this response does *not* carry, where the SRP one did: a server
201/// proof. Under SRP the client had to verify `M2` itself or it would have
202/// authenticated to a server that never knew the verifier. RFC 9807's AKE
203/// authenticates the server as part of the handshake — a client that
204/// successfully opens `KE2` has already proved the server holds the record —
205/// so there is nothing left for the client to check afterwards, and an SDK
206/// cannot forget to.
207#[utoipa::path(
208    post,
209    path = "/api/v1/auth/opaque/login/finish",
210    tag = "auth",
211    request_body = OpaqueLoginFinishRequest,
212    responses(
213        (status = 200, description = "Login successful; tokens arrive as cookies"),
214        (status = 202, description = "MFA challenge required", body = MfaRequiredResponse),
215        (status = 403, description = "MFA setup required", body = MfaSetupRequiredResponse),
216        (status = 401, description = "Invalid credentials, expired session, or unknown identity"),
217    )
218)]
219pub async fn opaque_login_finish<C: Connection + Clone>(
220    req: HttpRequest,
221    state: web::Data<AppState<C>>,
222    body: web::Json<OpaqueLoginFinishRequest>,
223) -> Result<HttpResponse, AxiamApiError> {
224    let b = body.into_inner();
225    let opaque = opaque_server(&state)?;
226
227    let verified = match opaque.login_finish(&b.opaque_session, &b.ke3) {
228        Ok(verified) => verified,
229        Err(rejection) => {
230            // A failed KE3 is a wrong password: it must move the lockout
231            // counter exactly as a failed Argon2id verify on `/auth/login`
232            // does. Without this, enabling OPAQUE would quietly remove
233            // brute-force protection from the accounts that adopted it.
234            if let axiam_auth::OpaqueRejection::BadCredentials {
235                tenant_id,
236                user_id: Some(user_id),
237            } = rejection
238            {
239                // Best-effort: a bookkeeping failure must not turn a 401 into
240                // a 500 and hand the caller a way to tell the two apart.
241                if let Err(e) = state
242                    .auth_service
243                    .record_failed_opaque_attempt(tenant_id, user_id)
244                    .await
245                {
246                    tracing::warn!(
247                        target: "axiam::auth",
248                        tenant_id = %tenant_id,
249                        user_id = %user_id,
250                        error = %e,
251                        "failed to record an OPAQUE lockout attempt"
252                    );
253                }
254            }
255            // Every rejection reason collapses to one 401 here; the variants
256            // exist for the accounting above, never for the caller.
257            return Err(AxiamApiError(AxiamError::AuthenticationFailed {
258                reason: "invalid credentials".into(),
259            }));
260        }
261    };
262
263    let user = state
264        .user_repo
265        .get_by_id(verified.tenant_id, verified.user_id)
266        .await
267        .map_err(|_| AxiamError::AuthenticationFailed {
268            reason: "invalid credentials".into(),
269        })?;
270
271    // Lockout is checked here rather than at login/start on purpose: the start
272    // response must stay indistinguishable for every identity, and refusing it
273    // for a locked account would announce that the account exists and is under
274    // attack.
275    if axiam_auth::lockout::is_locked_out(&user) {
276        return Err(AxiamApiError(AxiamError::AuthenticationFailed {
277            reason: "invalid credentials".into(),
278        }));
279    }
280
281    let mfa_policy = Some(
282        state
283            .settings_repo
284            .get_effective_settings(verified.org_id, verified.tenant_id)
285            .await
286            .map(|s| s.mfa)?,
287    );
288
289    let result = state
290        .auth_service
291        .complete_authenticated_login(
292            user,
293            verified.tenant_id,
294            verified.org_id,
295            client_ip(&req),
296            user_agent(&req),
297            mfa_policy,
298        )
299        .await?;
300
301    match result {
302        LoginResult::Success(out) => {
303            // The same builder the password path uses, so the cookie set, the
304            // CSRF header and the body shape cannot drift between the two.
305            cookie_response_from_output(
306                &out,
307                &state.auth_config,
308                &state.user_repo,
309                &state.tenant_repo,
310                &state.org_repo,
311            )
312            .await
313        }
314        LoginResult::MfaRequired(mut challenge) => {
315            if let Ok((user_id, tenant_id, _org_id)) = state
316                .auth_service
317                .decode_mfa_challenge_ids(&challenge.challenge_token)
318                && let Ok(types) = state
319                    .mfa_method_service
320                    .available_method_types(tenant_id, user_id)
321                    .await
322            {
323                challenge.available_methods = types;
324            }
325            Ok(HttpResponse::Accepted().json(serde_json::json!({
326                "mfa_required": true,
327                "challenge_token": challenge.challenge_token,
328                "available_methods": challenge.available_methods,
329            })))
330        }
331        LoginResult::MfaSetupRequired(setup) => {
332            Ok(HttpResponse::Forbidden().json(serde_json::json!({
333                "mfa_setup_required": true,
334                "setup_token": setup.setup_token,
335            })))
336        }
337    }
338}

344/// Fetch the configured OPAQUE server, or fail loudly.
345///
346/// A configured policy with no keys is a misconfiguration, and it must look
347/// like one. Falling through to "no OPAQUE available" would let an operator
348/// believe a control is on while every client silently uses password login.
349pub(crate) fn opaque_server<C: Connection + Clone>(
350    state: &AppState<C>,
351) -> Result<&axiam_auth::OpaqueServer, AxiamApiError> {
352    state.opaque_server.as_ref().ok_or_else(|| {
353        AxiamApiError(AxiamError::Internal(
354            "OPAQUE is enabled but AXIAM__AUTH__OPAQUE_SESSION_KEY / \
355             AXIAM__AUTH__OPAQUE_SETUP_KEY are not configured"
356                .into(),
357        ))
358    })
359}

361/// Fetch a tenant's OPAQUE key material, minting it on first use.
362///
363/// The mint-on-first-use is why this is a helper rather than an inline read:
364/// `get_or_create` must be the only way key material comes into existence, so
365/// that two concurrent first logins cannot end up writing two different OPRF
366/// seeds for one tenant. A tenant with two seeds has records that only
367/// sometimes open, which presents as an intermittent wrong password.
368pub(crate) async fn server_setup<C: Connection + Clone>(
369    state: &AppState<C>,
370    tenant_id: Uuid,
371    suite: axiam_core::models::opaque::OpaqueSuite,
372) -> Result<axiam_core::models::opaque::OpaqueServerSetup, AxiamApiError> {
373    let opaque = opaque_server(state)?;
374    let fresh = opaque
375        .create_server_setup(tenant_id, suite)
376        .map_err(AxiamError::from)?;
377    Ok(state.opaque_setup_repo.get_or_create(fresh).await?)
378}

380/// Resolve `(org_id, tenant_id)` from UUIDs or slugs and prove the tenant
381/// belongs to the organization.
382///
383/// The binding check is the same one `POST /auth/login` performs (NEW-1): a
384/// client passing raw UUIDs could otherwise bind their own tenant to a foreign
385/// `org_id` and get a token scoped to that organization. Every failure maps to
386/// an enumeration-safe `Validation`/`NotFound` that says nothing about which
387/// part was wrong.
388async fn resolve_workspace<C: Connection + Clone>(
389    state: &AppState<C>,
390    org_id: Option<Uuid>,
391    org_slug: Option<&str>,
392    tenant_id: Option<Uuid>,
393    tenant_slug: Option<&str>,
394) -> Result<(Uuid, Uuid), AxiamApiError> {
395    let org_id = match (org_id, org_slug) {
396        (Some(id), _) => id,
397        (None, Some(slug)) => {
398            state
399                .org_repo
400                .get_by_slug(slug)
401                .await
402                .map_err(|_| AxiamError::AuthenticationFailed {
403                    reason: "invalid credentials".into(),
404                })?
405                .id
406        }
407        (None, None) => {
408            return Err(AxiamApiError(AxiamError::Validation {
409                message: "must provide org_id or org_slug".into(),
410            }));
411        }
412    };
413
414    let tenant_id = match (tenant_id, tenant_slug) {
415        (Some(id), _) => id,
416        (None, Some(slug)) => {
417            state
418                .tenant_repo
419                .get_by_slug(org_id, slug)
420                .await
421                .map_err(|_| AxiamError::AuthenticationFailed {
422                    reason: "invalid credentials".into(),
423                })?
424                .id
425        }
426        (None, None) => {
427            return Err(AxiamApiError(AxiamError::Validation {
428                message: "must provide tenant_id or tenant_slug".into(),
429            }));
430        }
431    };
432
433    let tenant = state.tenant_repo.get_by_id(tenant_id).await.map_err(|_| {
434        AxiamError::AuthenticationFailed {
435            reason: "invalid credentials".into(),
436        }
437    })?;
438    if tenant.organization_id != org_id {
439        return Err(AxiamApiError(AxiamError::AuthenticationFailed {
440            reason: "invalid credentials".into(),
441        }));
442    }
443
444    Ok((tenant.organization_id, tenant_id))
445}

447/// Resolve a username-or-email to a user id, or `None`.
448///
449/// Deliberately swallows every error into `None`: the caller's next step is the
450/// enumeration-safe decoy exchange, and surfacing "no such user" as an error
451/// here would defeat it.
452async fn lookup_user_id<C: Connection + Clone>(
453    state: &AppState<C>,
454    tenant_id: Uuid,
455    username_or_email: &str,
456) -> Option<Uuid> {
457    if let Ok(user) = state
458        .user_repo
459        .get_by_username(tenant_id, username_or_email)
460        .await
461    {
462        return Some(user.id);
463    }
464    state
465        .user_repo
466        .get_by_email(tenant_id, username_or_email)
467        .await
468        .ok()
469        .map(|user| user.id)
470}
