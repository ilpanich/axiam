23/// Body for the request-reset endpoint.
24///
25/// `tenant_id` is optional (Open Question 1 / Assumption A4): the public
26/// forgot-password page has no prior tenant context, so it may instead
27/// supply `org_slug`/`tenant_slug` (mirroring `handlers/auth.rs`'s
28/// `(Option<Uuid>, Option<&str>)` login pattern). Slug-resolution failure
29/// is enumeration-safe (D-05) — it funnels into the SAME uniform
30/// `{"sent": true}` response as an unknown account, never a distinct
31/// error status.
32#[derive(Debug, Deserialize, utoipa::ToSchema)]
33pub struct RequestResetBody {
34    #[serde(default)]
35    pub tenant_id: Option<Uuid>,
36    pub email: String,
37    #[serde(default)]
38    pub org_slug: Option<String>,
39    #[serde(default)]
40    pub tenant_slug: Option<String>,
41}
42
43/// Body for the confirm-reset endpoint.
44#[derive(Debug, Deserialize, utoipa::ToSchema)]
45pub struct ConfirmResetBody {
46    pub tenant_id: Uuid,
47    pub token: String,
48    pub new_password: String,
49    /// OPAQUE registration record for `new_password`, built client-side.
50    ///
51    /// Without this, reset would be the hole in OPAQUE coverage: a tenant could
52    /// run `opaque_mode = required`, and every "forgot password" would still put a
53    /// plaintext password on the wire and leave the account with a verifier for
54    /// a password it no longer has.
55    #[serde(default, skip_serializing_if = "Option::is_none")]
56    pub opaque: Option<axiam_core::models::opaque::OpaqueEnrollment>,
57}

63/// Resolve the effective tenant context for `request_reset`.
64///
65/// Accepts either a raw `tenant_id` UUID or an `(org_slug, tenant_slug)`
66/// pair, mirroring `handlers/auth.rs`'s login resolution
67/// (`(Option<Uuid>, Option<&str>)`). UNLIKE login, a resolution failure is
68/// represented here as `None` rather than a distinct error — the caller
69/// MUST treat `None` as an enumeration-safe no-op (D-05 / Pitfall 4 /
70/// T-23-06-A), funneling into the SAME uniform `{"sent": true}` response
71/// as an unknown account, NEVER a distinct 400/404.
72///
73/// # Naming an organization and no tenant
74///
75/// Resolves the organization's own reserved scope, exactly as `POST /auth/login`
76/// and `POST /auth/opaque/*` do. That parity is the whole fix: bootstrap creates
77/// an **organization-level** administrator, and the "Forgot password?" link on an
78/// organization-level sign-in therefore carries only `?org=<slug>` — there is no
79/// tenant to carry. This function used to give up on that pair and return
80/// `None`, which the enumeration-safe contract above then turned into a silent
81/// success: `{"sent": true}`, no mail, nothing logged above `debug`. The only
82/// administrator of a freshly bootstrapped deployment had no way to recover a
83/// forgotten password, and no way to find out why.
84///
85/// A blank slug is not a slug, for the same reason it is not one on the login
86/// path: the browser posts `tenant_slug: ""` for an organization-level form, and
87/// so may an SDK that serializes an empty field rather than omitting it.
88///
89/// Naming *nothing* stays `None`. With no organization there is nothing to
90/// resolve, and inventing one would make this endpoint an oracle for which
91/// organizations exist — the precise thing the uniform response protects.
92async fn resolve_reset_tenant_id<O, T>(
93    org_repo: &O,
94    tenant_repo: &T,
95    tenant_id: Option<Uuid>,
96    org_slug: Option<&str>,
97    tenant_slug: Option<&str>,
98) -> Option<Uuid>
99where
100    O: OrganizationRepository,
101    T: TenantRepository,
102{
103    if let Some(id) = tenant_id {
104        return Some(id);
105    }
106
107    let org_slug = crate::handlers::auth::blank_is_unnamed(org_slug)?;
108    let tenant_slug = crate::handlers::auth::blank_is_unnamed(tenant_slug);
109    let org = org_repo.get_by_slug(org_slug).await.ok()?;
110
111    match tenant_slug {
112        Some(slug) => tenant_repo
113            .get_by_slug(org.id, slug)
114            .await
115            .ok()
116            .map(|t| t.id),
117        None => tenant_repo
118            .get_organization_tenant(org.id)
119            .await
120            .ok()
121            .map(|t| t.id),
122    }
123}

129/// `POST /api/v1/auth/reset`
130///
131/// Initiates a password reset by enqueuing an `OutboundMailMessage` to the
132/// async mail queue.  Always returns `{"sent": true}` to prevent email
133/// enumeration (D-15) — regardless of whether the email exists, the user is
134/// federated, delivery later succeeds, or the rate limit is exceeded.
135#[utoipa::path(
136    post,
137    path = "/api/v1/auth/reset",
138    tag = "auth",
139    request_body = RequestResetBody,
140    responses(
141        (status = 200, description = "Password reset email enqueued"),
142    )
143)]
144pub async fn request_reset<C: Connection + Clone>(
145    http_req: HttpRequest,
146    state: web::Data<AppState<C>>,
147    body: web::Json<RequestResetBody>,
148) -> Result<HttpResponse, AxiamApiError> {
149    let req = body.into_inner();
150
151    // Resolve tenant context — accept either a raw tenant_id UUID or a
152    // (org_slug, tenant_slug) pair (Open Question 1 / Assumption A4).
153    // Resolution failure NEVER `?`-propagates: it funnels into the SAME
154    // uniform enumeration-safe `{"sent": true}` response as an unknown
155    // account (D-05 / Pitfall 4 / T-23-06-A).
156    let tenant_id = resolve_reset_tenant_id(
157        &state.org_repo,
158        &state.tenant_repo,
159        req.tenant_id,
160        req.org_slug.as_deref(),
161        req.tenant_slug.as_deref(),
162    )
163    .await;
164
165    let Some(tenant_id) = tenant_id else {
166        // D-05: unresolvable/missing tenant context is enumeration-safe —
167        // identical to the unknown-account branch below. Never disclose
168        // whether an org/tenant slug exists.
169        tracing::debug!(
170            email = %req.email,
171            "password-reset: no action (tenant unresolved)"
172        );
173        return Ok(HttpResponse::Ok().json(serde_json::json!({ "sent": true })));
174    };
175
176    // Attribute the audit entry to the tenant just resolved, so a
177    // `password_reset_requested` notification rule can match it. Set here rather
178    // than in the unresolved branch above on purpose: an unresolvable tenant
179    // must stay indistinguishable from an unknown account (D-05), and that
180    // includes leaving no attributed audit trail behind for one and not the
181    // other.
182    //
183    // The lookup is resolved BEFORE the extensions borrow is taken. Awaiting
184    // inside the `insert(...)` argument holds a `RefCell` borrow of the
185    // extensions map across the await — `clippy::await_holding_refcell_ref` —
186    // and anything that borrowed the same map while this task was suspended
187    // would panic at runtime rather than fail to compile.
188    let org_id = state
189        .tenant_repo
190        .get_by_id(tenant_id)
191        .await
192        .map(|t| t.organization_id)
193        .unwrap_or_else(|_| Uuid::nil());
194    http_req
195        .extensions_mut()
196        .insert(axiam_audit::AuditAttribution { tenant_id, org_id });
197
198    // QUAL-07: PasswordResetService is now a hoisted AppState singleton
199    // (was constructed per-request here).
200    let expiry_hours = state.auth_config.password_reset_token_expiry_hours;
201    let pepper = state.auth_config.pepper.as_ref().map(|p| p.expose_secret());
202
203    match state
204        .mail
205        .password_reset_service
206        .initiate_reset(tenant_id, &req.email, expiry_hours, pepper)
207        .await
208    {
209        Ok(Some((raw_token, user_id, expires_at))) => {
210            // Resolve org_id from tenant for the mail message.
211            // On failure, log and continue — D-15: never propagate to client.
212            let org_id = match state.tenant_repo.get_by_id(tenant_id).await {
213                Ok(tenant) => tenant.organization_id,
214                Err(e) => {
215                    tracing::warn!(
216                        error = %e,
217                        tenant_id = %tenant_id,
218                        "failed to resolve org_id for password-reset mail; using nil"
219                    );
220                    Uuid::nil()
221                }
222            };
223
224            // PHASE-DEFINING (23-RESEARCH Pattern 6 / Pitfall 3): build the
225            // action_url the same way gdpr.rs builds cancel_url — a
226            // relative-path frontend route carrying the raw token plus the
227            // resolved tenant_id UUID (Open Question 2, mirroring the
228            // already-shipped VerifyEmailPage `?token=…&tenant_id=…`).
229            let reset_url = format!("/auth/reset-password?token={raw_token}&tenant_id={tenant_id}");
230
231            let msg = OutboundMailMessage {
232                mail_type: MailType::PasswordReset,
233                tenant_id,
234                org_id,
235                user_id,
236                to_address: req.email.clone(),
237                template_context: serde_json::json!({
238                    "token": raw_token,
239                    "action_url": reset_url,
240                    "expiry_time": expires_at.to_rfc3339(),
241                }),
242                attempt_count: 0,
243                enqueued_at: Utc::now(),
244            };
245
246            if let Err(e) = state.mail.mail_outbound_publisher.publish(msg).await {
247                // D-15: log warn but do NOT propagate — uniform 200 regardless
248                tracing::warn!(
249                    error = %e,
250                    "failed to enqueue password-reset email; continuing"
251                );
252            } else {
253                tracing::debug!(email = %req.email, "password reset email enqueued");
254            }
255        }
256        Ok(None) => {
257            // User not found or federated — silently ignore (D-15).
258            tracing::debug!(
259                email = %req.email,
260                "password-reset: no action (unknown or federated)"
261            );
262        }
263        Err(AxiamError::RateLimited) => {
264            // Swallow rate-limit to prevent user enumeration via
265            // differential 429 responses (D-15).
266            tracing::debug!(
267                email = %req.email,
268                "password-reset: rate-limited (suppressed)"
269            );
270        }
271        Err(e) => return Err(e.into()),
272    }
273
274    // Always return identical 200 regardless of outcome (D-15).
275    Ok(HttpResponse::Ok().json(serde_json::json!({ "sent": true })))
276}

278/// `POST /api/v1/auth/reset/confirm`
279///
280/// Confirms a password reset using a one-time token and a new
281/// password. The token is consumed atomically.
282///
283/// Returns `{"reset": true}` on success, or 400 with policy
284/// violations if the new password is too weak.
285#[utoipa::path(
286    post,
287    path = "/api/v1/auth/reset/confirm",
288    tag = "auth",
289    request_body = ConfirmResetBody,
290    responses(
291        (status = 200, description = "Password reset successfully"),
292        (status = 400, description = "Invalid token or password policy violation"),
293    )
294)]
295pub async fn confirm_reset<C: Connection + Clone>(
296    http_req: HttpRequest,
297    state: web::Data<AppState<C>>,
298    body: web::Json<ConfirmResetBody>,
299) -> Result<HttpResponse, AxiamApiError> {
300    use axiam_core::repository::{SettingsRepository, TenantRepository};
301
302    let req = body.into_inner();
303
304    // Resolve the tenant to get its org_id for settings.
305    let tenant = state.tenant_repo.get_by_id(req.tenant_id).await?;
306
307    // Attribute the audit entry, so a `password_changed` notification rule can
308    // match it. This endpoint is unauthenticated by nature — the caller proves
309    // themselves with the reset token, not a session — so the middleware has no
310    // JWT to read the tenant from. See `axiam_audit::AuditAttribution`.
311    http_req
312        .extensions_mut()
313        .insert(axiam_audit::AuditAttribution {
314            tenant_id: req.tenant_id,
315            org_id: tenant.organization_id,
316        });
317
318    // Resolve effective password policy.
319    let settings = state
320        .settings_repo
321        .get_effective_settings(tenant.organization_id, req.tenant_id)
322        .await?;
323
324    // Reject a malformed verifier before the token is consumed. `confirm_reset`
325    // burns the single-use token, so failing after it would leave the user with
326    // a changed password, no record, and no way to retry without requesting a
327    // second reset email.
328    let enrolled = crate::handlers::opaque_enrollment::validate_enrollment(
329        &state,
330        &settings.opaque,
331        req.tenant_id,
332        req.opaque.as_ref(),
333    )?;
334
335    // QUAL-07: PasswordResetService is now a hoisted AppState singleton.
336    let user_id = state
337        .mail
338        .password_reset_service
339        .confirm_reset(
340            req.tenant_id,
341            &req.token,
342            &req.new_password,
343            &settings.password,
344            state.auth_config.pepper.as_ref().map(|p| p.expose_secret()),
345            Some(&state.http_client),
346        )
347        .await?;
348
349    crate::handlers::opaque_enrollment::store_credential(&state, req.tenant_id, user_id, enrolled)
350        .await?;
351
352    Ok(HttpResponse::Ok().json(serde_json::json!({ "reset": true })))
353}

355/// Query for `GET /api/v1/auth/reset/context`.
356#[derive(Debug, Deserialize, utoipa::ToSchema)]
357pub struct ResetContextQuery {
358    /// Tenant the reset link was issued for.
359    pub tenant_id: Uuid,
360    /// The raw reset token from the emailed link. Not consumed by this call.
361    pub token: String,
362}
363
364/// What a valid reset-token holder needs in order to build an OPAQUE record.
365///
366/// The SRP version of this also returned the account's `identity`, because
367/// SRP derived its private key over `username ":" password` and the reset page
368/// had no other way to learn the username. OPAQUE binds to a server-chosen
369/// credential identifier that the client never handles, so the username is not
370/// needed and is no longer disclosed. The endpoint went from "tell an
371/// unauthenticated caller who this account is" to "tell them whether this
372/// tenant uses OPAQUE", which is strictly less.
373#[derive(Debug, Serialize, utoipa::ToSchema)]
374pub struct ResetContextResponse {
375    /// The tenant's effective OPAQUE policy, so the client knows whether to
376    /// run an enrolment at all. Absent when the tenant does not use OPAQUE.
377    #[serde(skip_serializing_if = "Option::is_none")]
378    pub opaque: Option<axiam_core::models::settings::OpaquePolicy>,
379}
380
381/// `GET /api/v1/auth/reset/context`
382///
383/// Password reset would otherwise be a permanent hole in OPAQUE coverage. The
384/// reset page is unauthenticated and holds only a token and a tenant id, so it
385/// cannot tell whether the tenant expects an enrolment. Without that, every
386/// "forgot password" would leave the account with a record for a password it
387/// no longer has, and under `opaque_mode: required` would leave it unable to
388/// log in at all.
389///
390/// # What this no longer discloses
391///
392/// The SRP version returned the account's username, because that string was
393/// inside the client-side key derivation and the reset page had no other way
394/// to learn it. That was defensible — the caller holds a token delivered to
395/// the account's own email — but it is better not to need the argument.
396/// OPAQUE does not bind to the username, so it is not returned.
397///
398/// An attacker without a valid token still gets `404` and learns nothing — the
399/// same answer for a wrong token, an expired token, and a token for a tenant that
400/// does not exist.
401///
402/// The token is **read, not consumed**: this is a lookup, and burning the
403/// single-use token here would break the reset it is meant to enable.
404#[utoipa::path(
405    get,
406    path = "/api/v1/auth/reset/context",
407    tag = "auth",
408    params(("tenant_id" = Uuid, Query, description = "Tenant the reset link was issued for"),
409           ("token" = String, Query, description = "Raw reset token from the emailed link")),
410    responses(
411        (status = 200, description = "OPAQUE policy for this reset", body = ResetContextResponse),
412        (status = 404, description = "Unknown, expired or already-consumed token"),
413    )
414)]
415pub async fn reset_context<C: Connection + Clone>(
416    state: web::Data<AppState<C>>,
417    query: web::Query<ResetContextQuery>,
418) -> Result<HttpResponse, AxiamApiError> {
419    use axiam_core::repository::{
420        PasswordResetTokenRepository as _, SettingsRepository as _, TenantRepository as _,
421        UserRepository as _,
422    };
423
424    let q = query.into_inner();
425    let token_hash = axiam_auth::token::hash_refresh_token(&q.token);
426
427    // Every failure below is the same `404`, so a caller cannot distinguish a
428    // wrong token from an expired one from a non-existent tenant.
429    let not_found = || {
430        AxiamApiError(AxiamError::NotFound {
431            entity: "password_reset_token".into(),
432            id: "invalid".into(),
433        })
434    };
435
436    // Built here rather than held on `AppState`: this is the only reader of the
437    // raw token rows outside `PasswordResetService`, and the repository is a
438    // cheap handle over the already-pooled connection.
439    let token_repo = axiam_db::SurrealPasswordResetTokenRepository::new(state.db.clone());
440    let token = token_repo
441        .get_by_token_hash(q.tenant_id, &token_hash)
442        .await
443        .map_err(|_| not_found())?;
444
445    // The token's user must still exist. Nothing from the row is returned any
446    // more — the SRP version needed the username from it — but a token
447    // pointing at a deleted account must answer `404` like any other invalid
448    // token rather than reporting the tenant's policy.
449    state
450        .user_repo
451        .get_by_id(q.tenant_id, token.user_id)
452        .await
453        .map_err(|_| not_found())?;
454
455    let tenant = state
456        .tenant_repo
457        .get_by_id(q.tenant_id)
458        .await
459        .map_err(|_| not_found())?;
460
461    let opaque = state
462        .settings_repo
463        .get_effective_settings(tenant.organization_id, q.tenant_id)
464        .await
465        .ok()
466        .map(|s| s.opaque)
467        .filter(|p| p.opaque_mode != axiam_core::models::opaque::OpaqueMode::Disabled);
468
469    Ok(HttpResponse::Ok().json(ResetContextResponse { opaque }))
470}

476#[cfg(test)]
477mod tests {
478    use super::*;
479    use axiam_core::error::AxiamResult;
480    use axiam_core::models::mail::OutboundMailMessage;
481    use axiam_core::repository::MailPublisher;
482    use std::sync::{Arc, Mutex};
483
484    /// Fake mail publisher that records messages for test assertions.
485    #[derive(Clone, Default)]
486    struct RecordingPublisher {
487        sent: Arc<Mutex<Vec<OutboundMailMessage>>>,
488        fail: bool,
489    }
490
491    impl RecordingPublisher {
492        fn new() -> Self {
493            Self {
494                sent: Arc::new(Mutex::new(Vec::new())),
495                fail: false,
496            }
497        }
498
499        fn failing() -> Self {
500            Self {
501                sent: Arc::new(Mutex::new(Vec::new())),
502                fail: true,
503            }
504        }
505
506        fn messages(&self) -> Vec<OutboundMailMessage> {
507            self.sent.lock().unwrap().clone()
508        }
509
510        fn count(&self) -> usize {
511            self.sent.lock().unwrap().len()
512        }
513    }
514
515    impl MailPublisher for RecordingPublisher {
516        async fn publish(&self, msg: OutboundMailMessage) -> AxiamResult<()> {
517            if self.fail {
518                return Err(AxiamError::Internal("mock publish failure".into()));
519            }
520            self.sent.lock().unwrap().push(msg);
521            Ok(())
522        }
523    }
524
525    // -----------------------------------------------------------------------
526    // D-15 tests: unknown email returns {"sent": true} — same as known email
527    // -----------------------------------------------------------------------
528
529    /// Unknown email → response body is `{"sent": true}` with no token field.
530    ///
531    /// This test validates the D-15 enumeration-safe contract: the handler
532    /// MUST return a uniform 200 regardless of whether the address exists.
533    #[tokio::test]
534    async fn unknown_email_enqueues_and_returns_sent() {
535        // Handler logic for the unknown-address branch:
536        // `svc.initiate_reset` returns `Ok(None)` → no enqueue, but response is
537        // still `{"sent": true}`.
538        //
539        // We simulate the handler's conditional logic directly since we can't
540        // easily spin up a full Actix stack in a unit test without a live DB.
541        // The key invariant is: the response body is `{"sent": true}` in ALL
542        // branches of the match and it NEVER contains a `token` field.
543
544        let response_body = serde_json::json!({ "sent": true });
545        // Confirm no token field in the body regardless of branch taken.
546        assert!(
547            response_body.get("token").is_none(),
548            "unknown-email response MUST NOT contain a token field (D-15)"
549        );
550        assert_eq!(
551            response_body.get("sent").and_then(|v| v.as_bool()),
552            Some(true),
553            "unknown-email response must be {{\"sent\": true}}"
554        );
555    }
556
557    /// Known email → response is `{"sent": true}` with NO token in the body.
558    ///
559    /// Enqueue path: an `OutboundMailMessage` is queued, but the raw token
560    /// is NEVER placed in the HTTP response body.
561    #[tokio::test]
562    async fn known_email_never_returns_token() {
563        let publisher = RecordingPublisher::new();
564
565        // Simulate the known-email branch of the handler.
566        let raw_token = "secret-reset-token-abc123".to_string();
567        let user_id = Uuid::new_v4();
568        let tenant_id = Uuid::new_v4();
569        let expires_at = Utc::now() + chrono::Duration::hours(1);
570        let org_id = Uuid::new_v4();
571
572        // This is exactly what the handler does in the Ok(Some(...)) branch.
573        let msg = OutboundMailMessage {
574            mail_type: MailType::PasswordReset,
575            tenant_id,
576            org_id,
577            user_id,
578            to_address: "user@example.com".to_string(),
579            template_context: serde_json::json!({
580                "token": raw_token.clone(),
581                "expiry_time": expires_at.to_rfc3339(),
582            }),
583            attempt_count: 0,
584            enqueued_at: Utc::now(),
585        };
586        publisher.publish(msg).await.unwrap();
587
588        // One message enqueued.
589        assert_eq!(publisher.count(), 1, "expected exactly one enqueued mail");
590        let enqueued = &publisher.messages()[0];
591        assert!(
592            matches!(enqueued.mail_type, MailType::PasswordReset),
593            "mail_type must be PasswordReset"
594        );
595
596        // The HTTP response body is ALWAYS {"sent": true} — the token is
597        // only in the enqueued message's template_context, never in the body.
598        let response_body = serde_json::json!({ "sent": true });
599        assert!(
600            response_body.get("token").is_none(),
601            "response body MUST NOT contain token (D-15 / T-5-token-leak)"
602        );
603        assert_eq!(
604            response_body.get("sent").and_then(|v| v.as_bool()),
605            Some(true)
606        );
607
608        // Publish-failure path: response still {"sent": true} (D-15).
609        let failing_publisher = RecordingPublisher::failing();
610        let msg2 = OutboundMailMessage {
611            mail_type: MailType::PasswordReset,
612            tenant_id,
613            org_id,
614            user_id,
615            to_address: "user@example.com".to_string(),
616            template_context: serde_json::json!({"token": "t", "expiry_time": "e"}),
617            attempt_count: 0,
618            enqueued_at: Utc::now(),
619        };
620        // Publish error is swallowed; response is still sent: true.
621        let result = failing_publisher.publish(msg2).await;
622        assert!(result.is_err(), "failing publisher should return error");

635    /// The action_url built by `request_reset` (mirroring gdpr.rs's
636    /// cancel_url) fully substitutes into the rendered PasswordReset email:
637    /// the rendered link contains the token + tenant_id, and the literal
638    /// `{{action_url}}` mustache placeholder is gone.
639    #[tokio::test]
640    async fn action_url_is_substituted_in_rendered_password_reset_email() {
641        use axiam_core::models::email_template::TemplateKind;
642        use axiam_email::template::{TemplateContext, render_email, resolve_template};
643
644        let raw_token = "reset-token-substitution-check";
645        let tenant_id = Uuid::new_v4();
646        let expires_at = Utc::now() + chrono::Duration::hours(1);
647
648        // Exactly what `request_reset` now builds into template_context.
649        let reset_url = format!("/auth/reset-password?token={raw_token}&tenant_id={tenant_id}");
650        let template_context = serde_json::json!({
651            "token": raw_token,
652            "action_url": reset_url,
653            "expiry_time": expires_at.to_rfc3339(),
654        });
655
656        // Mirror `axiam-amqp::mail_consumer::build_template_context`'s
657        // JSON-object -> TemplateContext conversion (string values as-is).
658        let mut ctx = TemplateContext::new();
659        if let serde_json::Value::Object(obj) = &template_context {
660            for (k, v) in obj {
661                if let serde_json::Value::String(s) = v {
662                    ctx.insert(k.clone(), s.clone());
663                }
664            }
665        }
666
667        let template = resolve_template(TemplateKind::PasswordReset, None, None);
668        let rendered = render_email(&template, "user@example.com", &ctx);
669        let html = rendered.html_body.expect("html body must be rendered");
670        let text = rendered.text_body.expect("text body must be rendered");
671
672        for body in [&html, &text] {
673            assert!(
674                body.contains(raw_token),
675                "rendered email must contain the raw reset token: {body}"
676            );
677            assert!(
678                body.contains(&tenant_id.to_string()),
679                "rendered email must contain the resolved tenant_id: {body}"
680            );
681            assert!(
682                !body.contains("{{action_url}}"),
683                "rendered email MUST NOT contain the unsubstituted action_url \
684                 placeholder (23-RESEARCH Pitfall 3): {body}"
685            );
686        }
687    }
