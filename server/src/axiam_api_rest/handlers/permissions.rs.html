35#[derive(Debug, Deserialize, utoipa::ToSchema)]
36pub struct GrantPermissionRequest {
37    pub permission_id: Uuid,
38    #[serde(default)]
39    pub scope_ids: Vec<Uuid>,
40    /// B1: `"allow"` (the default) or `"deny"`.
41    ///
42    /// A deny grant **overrides every allow**, at any depth of the resource
43    /// hierarchy and at equal specificity — it is not most-specific-wins. See
44    /// `claude_dev/deny-override-design.md` for the precedence table and for
45    /// why that trade was made.
46    ///
47    /// Omitting the field means `"allow"`, so every existing client keeps
48    /// working unchanged and no migration is required.
49    #[serde(default)]
50    pub effect: PermissionEffect,
51}

67/// `POST /api/v1/permissions`
68#[utoipa::path(
69    post,
70    path = "/api/v1/permissions",
71    tag = "permissions",
72    request_body = CreatePermissionRequest,
73    responses(
74        (status = 201, description = "Permission created", body = Permission),
75    ),
76    security(("bearer" = []), ("service_account" = []))
77)]
78pub async fn create<C: Connection + Clone>(
79    principal: AuthenticatedPrincipal,
80    authz: AuthzData,
81    state: web::Data<AppState<C>>,
82    body: web::Json<CreatePermissionRequest>,
83) -> Result<HttpResponse, AxiamApiError> {
84    RequirePermission::new("permissions:create", Uuid::nil())
85        .check(&principal, authz.get_ref().as_ref())
86        .await?;
87    let req = body.into_inner();
88    // §4.2: the `network:` namespace is reserved, so a tenant cannot mint a
89    // look-alike of the nine.
90    let _cleared = network_gate::authorize(
91        &principal,
92        authz.get_ref().as_ref(),
93        &state,
94        NetworkWrite::PermissionCreate {
95            action: &req.action,
96        },
97    )
98    .await?;
99    let input = CreatePermission {
100        tenant_id: principal.tenant_id,
101        action: req.action,
102        description: req.description,
103    };
104    let permission = state.permission_repo.create(input).await?;
105    Ok(HttpResponse::Created().json(permission))
106}

163/// `PUT /api/v1/permissions/{permission_id}`
164#[utoipa::path(
165    put,
166    path = "/api/v1/permissions/{permission_id}",
167    tag = "permissions",
168    params(("permission_id" = Uuid, Path, description = "Permission ID")),
169    request_body = UpdatePermissionRequest,
170    responses(
171        (status = 200, description = "Permission updated", body = Permission),
172        (status = 404, description = "Permission not found"),
173    ),
174    security(("bearer" = []), ("service_account" = []))
175)]
176pub async fn update<C: Connection + Clone>(
177    principal: AuthenticatedPrincipal,
178    authz: AuthzData,
179    state: web::Data<AppState<C>>,
180    path: web::Path<Uuid>,
181    body: web::Json<UpdatePermissionRequest>,
182) -> Result<HttpResponse, AxiamApiError> {
183    RequirePermission::new("permissions:update", Uuid::nil())
184        .check(&principal, authz.get_ref().as_ref())
185        .await?;
186    let req = body.into_inner();
187    let permission_id = path.into_inner();
188    // No rename into or out of the reserved namespace, and no change to one of
189    // its permissions: renaming `network:join` would release every
190    // quarantine at once.
191    let _cleared = network_gate::authorize(
192        &principal,
193        authz.get_ref().as_ref(),
194        &state,
195        NetworkWrite::PermissionUpdate {
196            permission_id,
197            action: req.action.as_deref(),
198        },
199    )
200    .await?;
201    let input = UpdatePermission {
202        action: req.action,
203        description: req.description,
204    };
205    let permission = state
206        .permission_repo
207        .update(principal.tenant_id, permission_id, input)
208        .await?;
209    // D7 (REVOCATION — security critical): changing a permission's `action`
210    // narrows access (a subject allowed for the old action is now denied).
211    // Flush the tenant.
212    authz
213        .get_ref()
214        .as_ref()
215        .invalidate_tenant(principal.tenant_id)
216        .await?;
217    Ok(HttpResponse::Ok().json(permission))
218}

220/// `DELETE /api/v1/permissions/{permission_id}`
221#[utoipa::path(
222    delete,
223    path = "/api/v1/permissions/{permission_id}",
224    tag = "permissions",
225    params(("permission_id" = Uuid, Path, description = "Permission ID")),
226    responses(
227        (status = 204, description = "Permission deleted"),
228        (status = 404, description = "Permission not found"),
229    ),
230    security(("bearer" = []), ("service_account" = []))
231)]
232pub async fn delete<C: Connection + Clone>(
233    principal: AuthenticatedPrincipal,
234    authz: AuthzData,
235    state: web::Data<AppState<C>>,
236    path: web::Path<Uuid>,
237) -> Result<HttpResponse, AxiamApiError> {
238    RequirePermission::new("permissions:delete", Uuid::nil())
239        .check(&principal, authz.get_ref().as_ref())
240        .await?;
241    let permission_id = path.into_inner();
242    // The reserved catalogue is not deleted: dropping `network:join` would
243    // drop every quarantine deny with it.
244    let _cleared = network_gate::authorize(
245        &principal,
246        authz.get_ref().as_ref(),
247        &state,
248        NetworkWrite::PermissionDelete { permission_id },
249    )
250    .await?;
251    state
252        .permission_repo
253        .delete(principal.tenant_id, permission_id)
254        .await?;
255    // D7 (REVOCATION — security critical): deleting a permission removes it
256    // from every role/subject that was granted it. Flush the tenant.
257    authz
258        .get_ref()
259        .as_ref()
260        .invalidate_tenant(principal.tenant_id)
261        .await?;
262    Ok(HttpResponse::NoContent().finish())
263}

269/// `POST /api/v1/roles/{role_id}/permissions`
270#[utoipa::path(
271    post,
272    path = "/api/v1/roles/{role_id}/permissions",
273    tag = "permissions",
274    params(("role_id" = Uuid, Path, description = "Role ID")),
275    request_body = GrantPermissionRequest,
276    responses(
277        (status = 204, description = "Permission granted to role"),
278    ),
279    security(("bearer" = []), ("service_account" = []))
280)]
281pub async fn grant_to_role<C: Connection + Clone>(
282    principal: AuthenticatedPrincipal,
283    authz: AuthzData,
284    state: web::Data<AppState<C>>,
285    path: web::Path<Uuid>,
286    body: web::Json<GrantPermissionRequest>,
287) -> Result<HttpResponse, AxiamApiError> {
288    RequirePermission::new("permissions:grant", Uuid::nil())
289        .check(&principal, authz.get_ref().as_ref())
290        .await?;
291    let req = body.into_inner();
292    let role_id = path.into_inner();
293    // §5.3 / §4.2: a `network:` grant changes every assignment of the role at
294    // once, so it takes `network:grant` at /network (and every meta permission
295    // it confers), and never lands on a global or group-held role.
296    let scope_ids = req.scope_ids.clone();
297    let write = NetworkWrite::PermissionGrant {
298        role_id,
299        permission_id: req.permission_id,
300        effect: req.effect,
301        scope_ids: &scope_ids,
302    };
303    let cleared =
304        network_gate::authorize(&principal, authz.get_ref().as_ref(), &state, write).await?;
305    if let Err(error) = state
306        .permission_repo
307        .grant_to_role_claimed(
308            principal.tenant_id,
309            role_id,
310            req.permission_id,
311            req.scope_ids,
312            req.effect,
313            cleared.role_claim(),
314        )
315        .await
316    {
317        return network_gate::refused_after_clearance(&principal, &state, write, error).await;
318    }
319    // D7: an ALLOW grant widens access (safe direction) for every subject
320    // holding the role (set unknown here) — flush the tenant so the new grant
321    // is visible immediately.
322    //
323    // B1: a DENY grant *narrows* access, which is the direction where a stale
324    // cache is a security problem rather than an inconvenience. The same
325    // tenant flush covers it, on the same event path and with the same
326    // measured 262 ms contract — a deny rule is a grant, not a special case.
327    authz
328        .get_ref()
329        .as_ref()
330        .invalidate_tenant(principal.tenant_id)
331        .await?;
332    Ok(HttpResponse::NoContent().finish())
333}

335/// `GET /api/v1/roles/{role_id}/permissions`
336#[utoipa::path(
337    get,
338    path = "/api/v1/roles/{role_id}/permissions",
339    tag = "permissions",
340    params(("role_id" = Uuid, Path, description = "Role ID")),
341    responses(
342        (status = 200, description = "Permission grants for role, with scopes resolved",
343         body = Vec<ResolvedPermissionGrant>),
344    ),
345    security(("bearer" = []), ("service_account" = []))
346)]
347pub async fn list_role_permissions<C: Connection + Clone>(
348    principal: AuthenticatedPrincipal,
349    authz: AuthzData,
350    state: web::Data<AppState<C>>,
351    path: web::Path<Uuid>,
352) -> Result<HttpResponse, AxiamApiError> {
353    RequirePermission::new("permissions:list", Uuid::nil())
354        .check(&principal, authz.get_ref().as_ref())
355        .await?;
356    let grants = state
357        .permission_repo
358        .get_role_permission_grants(principal.tenant_id, path.into_inner())
359        .await?;
360
361    // Resolve each grant's `scope_ids` to the scopes they name.
362    //
363    // The grant edge stores scope UUIDs and nothing else, so the admin UI could
364    // only ever render "3 scopes" — a number with no way to find out which
365    // three. Answering "which scopes does this grant cover?" from the client
366    // would have meant listing every resource in the tenant and then every
367    // resource's scopes, just to turn three UUIDs into three names. The ids are
368    // resolved here instead, where one lookup answers it.
369    let resolved = resolve_grant_scopes(&state, principal.tenant_id, grants).await;
370    Ok(HttpResponse::Ok().json(resolved))
371}

373/// A scope named by a grant, resolved to something a human can read.
374#[derive(Debug, Clone, serde::Serialize, utoipa::ToSchema)]
375pub struct GrantedScope {
376    /// The scope's id, as it appears in the grant's `scope_ids`.
377    pub id: Uuid,
378    /// The scope's name, e.g. `invoices`.
379    pub name: String,
380    /// The resource the scope belongs to.
381    pub resource_id: Uuid,
382}
383
384/// A permission grant with its scopes resolved.
385///
386/// A superset of [`PermissionGrant`]: `scope_ids` is still present and still
387/// authoritative, so a client written before `scopes` existed is unaffected.
388#[derive(Debug, Clone, serde::Serialize, utoipa::ToSchema)]
389pub struct ResolvedPermissionGrant {
390    /// The permission this grant is about.
391    pub permission: Permission,
392    /// The scope ids the grant is constrained to. Empty is the wildcard.
393    pub scope_ids: Vec<Uuid>,
394    /// Whether the grant permits or refuses.
395    pub effect: PermissionEffect,
396    /// The scopes `scope_ids` names, in the same order where resolvable.
397    ///
398    /// A scope that cannot be resolved is **omitted** rather than represented
399    /// by a placeholder, so this can be shorter than `scope_ids`. That happens
400    /// when a scope was deleted while a grant still referenced it, and showing
401    /// a name for something that no longer exists would be worse than showing
402    /// one fewer chip beside a count that still says how many ids the grant
403    /// carries.
404    pub scopes: Vec<GrantedScope>,
405}
406
407/// Turn each grant's `scope_ids` into [`GrantedScope`]s.
408///
409/// One lookup per distinct scope across the whole response, not per grant:
410/// several grants on the same role commonly name the same scopes, and issuing
411/// the same read once per grant would make a role page's cost quadratic in how
412/// carefully its grants were narrowed — penalising exactly the operators doing
413/// the right thing.
414///
415/// A failed lookup drops the scope and logs; it never fails the request. The
416/// caller asked for the role's grants, and `scope_ids` — the authoritative
417/// part — is already in hand.
418async fn resolve_grant_scopes<C: Connection + Clone>(
419    state: &AppState<C>,
420    tenant_id: Uuid,
421    grants: Vec<PermissionGrant>,
422) -> Vec<ResolvedPermissionGrant> {
423    use axiam_core::repository::ScopeRepository as _;
424    use std::collections::HashMap;
425
426    let mut cache: HashMap<Uuid, Option<GrantedScope>> = HashMap::new();
427    for grant in &grants {
428        for scope_id in &grant.scope_ids {
429            if cache.contains_key(scope_id) {
430                continue;
431            }
432            let resolved = match state.scope_repo.get_by_id(tenant_id, *scope_id).await {
433                Ok(scope) => Some(GrantedScope {
434                    id: scope.id,
435                    name: scope.name,
436                    resource_id: scope.resource_id,
437                }),
438                Err(e) => {
439                    tracing::debug!(
440                        error = %e,
441                        %tenant_id,
442                        %scope_id,
443                        "a grant names a scope that no longer resolves"
444                    );
445                    None
446                }
447            };
448            cache.insert(*scope_id, resolved);
449        }
450    }
451
452    grants
453        .into_iter()
454        .map(|grant| {
455            let scopes = grant
456                .scope_ids
457                .iter()
458                .filter_map(|id| cache.get(id).and_then(|s| s.clone()))
459                .collect();
460            ResolvedPermissionGrant {
461                permission: grant.permission,
462                scope_ids: grant.scope_ids,
463                effect: grant.effect,
464                scopes,
465            }
466        })
467        .collect()
468}

470/// `DELETE /api/v1/roles/{role_id}/permissions/{permission_id}`
471#[utoipa::path(
472    delete,
473    path = "/api/v1/roles/{role_id}/permissions/{permission_id}",
474    tag = "permissions",
475    params(
476        ("role_id" = Uuid, Path, description = "Role ID"),
477        ("permission_id" = Uuid, Path, description = "Permission ID"),
478    ),
479    responses(
480        (status = 204, description = "Permission revoked from role"),
481    ),
482    security(("bearer" = []), ("service_account" = []))
483)]
484pub async fn revoke_from_role<C: Connection + Clone>(
485    principal: AuthenticatedPrincipal,
486    authz: AuthzData,
487    state: web::Data<AppState<C>>,
488    path: web::Path<RolePermissionPath>,
489) -> Result<HttpResponse, AxiamApiError> {
490    RequirePermission::new("permissions:revoke", Uuid::nil())
491        .check(&principal, authz.get_ref().as_ref())
492        .await?;
493    let p = path.into_inner();
494    // §5.3: revoking a `network:` grant of either effect (a deny's revoke
495    // releases a quarantine) takes `network:grant` at /network.
496    let write = NetworkWrite::PermissionRevoke {
497        role_id: p.role_id,
498        permission_id: p.permission_id,
499    };
500    let cleared =
501        network_gate::authorize(&principal, authz.get_ref().as_ref(), &state, write).await?;
502    if let Err(error) = state
503        .permission_repo
504        .revoke_from_role_claimed(
505            principal.tenant_id,
506            p.role_id,
507            p.permission_id,
508            cleared.role_claim(),
509        )
510        .await
511    {
512        return network_gate::refused_after_clearance(&principal, &state, write, error).await;
513    }
514    // D7 (REVOCATION — security critical): revoking a grant removes access for
515    // every subject holding this role. The subject set isn't known here without
516    // a query, so flush the whole tenant — this must never leave a stale allow.
517    authz
518        .get_ref()
519        .as_ref()
520        .invalidate_tenant(principal.tenant_id)
521        .await?;
522    Ok(HttpResponse::NoContent().finish())
