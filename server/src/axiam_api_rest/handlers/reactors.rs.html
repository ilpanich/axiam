1//! Reactor management endpoints (X1).
2//!
3//! A registration decides whether third-party code can veto a login or add a
4//! claim to a token, so every write goes through
5//! [`axiam_core::models::reactor::validate_registration`] against the same
6//! `EVENT_REGISTRY` the dispatcher reads. Restating the valid event set here
7//! would create a second source of truth, and the one that drifted would be
8//! whichever nobody was looking at.
9

30// ---------------------------------------------------------------------------
31// Per-reactor health (R2.3)
32// ---------------------------------------------------------------------------
33//
34// R2.2 started writing `ChainResult.failures` and a chain-level deny to the
35// audit trail; `axiam_amqp::reactor::gate::recent_health` is where that
36// finally becomes visible to an operator, in the same response the admin
37// console already renders `last_seen_at` from. It is shared with the gRPC
38// admin service (`axiam-api-grpc`'s `ReactorAdminServiceImpl`) rather than
39// duplicated, so the two transports report the same numbers from the same
40// query.
41
42/// Read one reactor's health using this handler's fixed lookback + sample
43/// size — a thin wrapper so call sites below read `reactor_health(..)`
44/// rather than the longer shared-function call with its two tuning
45/// parameters repeated at every site.
46async fn reactor_health<A: axiam_core::repository::AuditLogRepository>(
47    audit_repo: &A,
48    tenant_id: Uuid,
49    reactor_id: Uuid,
50) -> ReactorHealth {
51    recent_health(
52        audit_repo,
53        tenant_id,
54        reactor_id,
55        chrono::Duration::hours(DEFAULT_HEALTH_LOOKBACK_HOURS),
56        DEFAULT_HEALTH_FAILURE_SAMPLE_LIMIT,
57    )
58    .await
59}

99#[derive(Debug, Serialize, utoipa::ToSchema)]
100pub struct ReactorResponse {
101    pub id: Uuid,
102    pub tenant_id: Uuid,
103    pub name: String,
104    pub description: String,
105    pub events: Vec<String>,
106    pub mode: ReactorMode,
107    pub priority: i32,
108    pub timeout_ms: u32,
109    pub failure_policy: FailurePolicy,
110    pub enabled: bool,
111    pub created_at: DateTime<Utc>,
112    pub updated_at: DateTime<Utc>,
113    /// When this reactor last consumed from its queue. `null` means it has
114    /// never connected — which the admin UI shows differently from "connected
115    /// once, silent since".
116    pub last_seen_at: Option<DateTime<Utc>>,
117    /// R2.3: dispatch failures against this registration in the last 24h
118    /// whose cause was a timeout (as opposed to a rejected reply, a transport
119    /// failure, or overload), capped at 100 — a health signal read from the
120    /// audit trail R2.2 started writing, not a replacement for the audit log
121    /// itself (`GET /api/v1/audit-log` is that).
122    pub recent_timeout_count: u32,
123    /// R2.3: operations this reactor's own reply *denied* in the last 24h,
124    /// capped at 100. Distinct from `recent_timeout_count`: a veto is the
125    /// reactor working as designed; a timeout is the reactor not answering.
126    pub recent_veto_count: u32,
127}
128
129impl From<Reactor> for ReactorResponse {
130    /// Health defaults to zero. Call sites that have an audit repository in
131    /// scope (`list`, `get`) overwrite it via [`ReactorResponse::with_health`]
132    /// — a freshly created or just-updated reactor has nothing in the
133    /// lookback window to report anyway, so `create`/`update`/`delete` do not
134    /// pay the extra audit-trail round trips.
135    fn from(r: Reactor) -> Self {
136        Self {
137            id: r.id,
138            tenant_id: r.tenant_id,
139            name: r.name,
140            description: r.description,
141            events: r.events,
142            mode: r.mode,
143            priority: r.priority,
144            timeout_ms: r.timeout_ms,
145            failure_policy: r.failure_policy,
146            enabled: r.enabled,
147            created_at: r.created_at,
148            updated_at: r.updated_at,
149            last_seen_at: r.last_seen_at,
150            recent_timeout_count: 0,
151            recent_veto_count: 0,
152        }
153    }
154}
155
156impl ReactorResponse {
157    fn with_health(mut self, health: ReactorHealth) -> Self {
158        self.recent_timeout_count = health.recent_timeout_count;
159        self.recent_veto_count = health.recent_veto_count;
160        self
161    }
162}

181/// SEC-101 — refuse a registration that could never be delivered to.
182///
183/// Two independent reasons, both of which make an accepted registration a
184/// lie, and both of which are checked before anything is validated or written.
185///
186/// # 1. The composed transport cannot dispatch at all
187///
188/// Registering a reactor against a transport that fails every dispatch is a
189/// **self-service, tenant-wide, complete login outage**: `login.post_auth`,
190/// `user.pre_create`, `user.pre_update` and `grant.pre_assign` all default to
191/// `fail_closed`, so the first login after the registration is denied and so
192/// is every one after that. The only thing standing between an operator and
193/// that would be one `tracing::warn!` emitted once at boot — for every
194/// deployment, including the overwhelming majority that will never register a
195/// reactor, which is how a warning becomes a line that is filtered out. The
196/// action that causes the outage happens hours or weeks later, in an admin UI
197/// that gives no indication anything is wrong.
198///
199/// The fail-closed posture itself is correct and is not changed here: a
200/// registered fail-closed fraud check that silently did nothing would be
201/// worse, and is the failure mode reactors exist to prevent. What changes is
202/// *when* the operator is told — at the point of the action, in the response
203/// to it, instead of in a boot log.
204///
205/// Since R2.4 merged the lapin transport this arm no longer fires for
206/// `axiam-server`, which composes a transport that reports `can_dispatch()`.
207/// It is **not** dead: `can_dispatch` reports the transport's capability, and
208/// a build that composes `UnavailableReactorTransport` (or `NoopReactorGate`)
209/// still answers `false`. It deliberately does **not** fire on a broker
210/// outage — see `ReactorTransport::can_dispatch`'s doc for why a blip must not
211/// become a registration outage.
212///
213/// # 2. The registration is `listen`, whose fan-out has no caller
214///
215/// `ReactorTransport::publish_listen` is implemented by the lapin transport,
216/// but nothing calls it: `run_chain` filters listeners out and
217/// `DispatchingReactorGate::intercept` returns early once the interceptor list
218/// is empty. A `listen` registration would therefore receive nothing at all —
219/// the inverse of its contract, and a silent one, since a listener produces no
220/// outcome to notice its absence in. Refusing is the honest answer until the
221/// gate grows the fan-out.
222///
223/// # Why 503, and why only enabled registrations
224///
225/// 503 rather than 400: nothing is wrong with the request. The dependency the
226/// resource needs is absent from this build, which is exactly what
227/// `AxiamError::ServiceUnavailable` means everywhere else in AXIAM.
228/// `would_be_enabled` is the merged result, not the request: an operator must
229/// always be able to **disable** or delete a registration, especially when the
230/// transport is missing, so only a write that would leave an enabled
231/// registration in place is refused. `DELETE` is never refused for the same
232/// reason.
233fn require_dispatchable_transport<C: Connection + Clone>(
234    state: &AppState<C>,
235    would_be_enabled: bool,
236    mode: ReactorMode,
237) -> Result<(), AxiamApiError> {
238    use axiam_core::models::reactor::ReactorGate;
239
240    if !would_be_enabled {
241        return Ok(());
242    }
243
244    if !state.events.reactor_gate.can_dispatch() {
245        return Err(axiam_core::error::AxiamError::ServiceUnavailable(
246            "the AMQP reactor transport is not available in this build, so a registered \
247             reactor could never be reached. Registering one would apply its failure_policy \
248             to every dispatch instead — and login.post_auth, user.pre_create, \
249             user.pre_update and grant.pre_assign all default to fail_closed, so the \
250             registration would deny those operations for this tenant. Refusing rather than \
251             accepting a registration that causes an outage."
252                .into(),
253        )
254        .into());
255    }
256
257    if mode == ReactorMode::Listen {
258        return Err(axiam_core::error::AxiamError::ServiceUnavailable(
259            "listen-mode reactors are not dispatched to yet: the transport can publish a \
260             listen event, but no hook site fans out to listeners, so this registration \
261             would receive nothing and — being a listener — would produce no outcome in \
262             which you could notice. Register the reactor with mode 'intercept', or create \
263             it with enabled: false until the listener fan-out ships."
264                .into(),
265        )
266        .into());
267    }
268
269    Ok(())
270}

279/// `GET /api/v1/reactors/events` — the registry, verbatim.
280///
281/// Served rather than documented so an admin UI, an SDK generator and an
282/// operator all read the same list the dispatcher enforces. A hand-maintained
283/// copy in the docs is a copy that goes stale on the next hook added.
284#[utoipa::path(
285    get,
286    path = "/api/v1/reactors/events",
287    tag = "reactors",
288    responses(
289        (status = 200, description = "Hookable events", body = Vec<ReactorEventDescriptor>),
290    ),
291    security(("bearer" = []))
292)]
293pub async fn list_events<C: Connection + Clone>(
294    user: AuthenticatedUser,
295    authz: AuthzData,
296    _state: web::Data<AppState<C>>,
297) -> Result<HttpResponse, AxiamApiError> {
298    RequirePermission::new("reactors:list", Uuid::nil())
299        .check(&user, authz.get_ref().as_ref())
300        .await?;
301
302    let events: Vec<ReactorEventDescriptor> = EVENT_REGISTRY
303        .iter()
304        .map(|spec| ReactorEventDescriptor {
305            name: spec.name.to_string(),
306            interceptable: spec.interceptable,
307            mutable: spec.mutable,
308            mutable_fields: spec.mutable_fields.iter().map(|f| f.to_string()).collect(),
309            default_failure_policy: spec.default_failure_policy,
310            description: spec.description.to_string(),
311        })
312        .collect();
313
314    Ok(HttpResponse::Ok().json(events))
315}

317/// `POST /api/v1/reactors`
318#[utoipa::path(
319    post,
320    path = "/api/v1/reactors",
321    tag = "reactors",
322    request_body = CreateReactorRequest,
323    responses(
324        (status = 201, description = "Reactor registered", body = ReactorResponse),
325        (status = 400, description = "Unknown event, non-interceptable event, or out-of-range timeout"),
326    ),
327    security(("bearer" = []))
328)]
329pub async fn create<C: Connection + Clone>(
330    user: AuthenticatedUser,
331    authz: AuthzData,
332    state: web::Data<AppState<C>>,
333    body: web::Json<CreateReactorRequest>,
334) -> Result<HttpResponse, AxiamApiError> {
335    RequirePermission::new("reactors:create", Uuid::nil())
336        .check(&user, authz.get_ref().as_ref())
337        .await?;
338    let req = body.into_inner();
339
340    // SEC-101: before anything is validated or written.
341    require_dispatchable_transport(state.get_ref(), req.enabled, req.mode)?;
342
343    // Validate the timeout that will actually be stored, not the one the
344    // request happened to carry — an omitted value still has to be in range,
345    // and a default that was not is a bug worth catching here.
346    let timeout_ms = req.timeout_ms.unwrap_or(DEFAULT_TIMEOUT_MS);
347    validate_registration(&req.name, &req.events, req.mode, timeout_ms)
348        .map_err(|e| validation_err(e.to_string()))?;
349
350    let reactor = state
351        .events
352        .reactor_repo
353        .create(CreateReactor {
354            tenant_id: user.tenant_id,
355            name: req.name,
356            description: req.description,
357            events: req.events,
358            mode: req.mode,
359            priority: req.priority,
360            timeout_ms: Some(timeout_ms),
361            failure_policy: req.failure_policy,
362            enabled: req.enabled,
363        })
364        .await?;
365
366    // X1 — the routing table this replica dispatches from is a TTL cache, so a
367    // new registration would otherwise not fire for up to the TTL. Invalidate
368    // the tenant's entries here so it is live at once; the TTL remains the
369    // guarantee for every OTHER replica, which did not serve this request.
370    state.invalidate_reactor_routing(user.tenant_id);
371
372    Ok(HttpResponse::Created().json(ReactorResponse::from(reactor)))
373}

375/// `GET /api/v1/reactors`
376#[utoipa::path(
377    get,
378    path = "/api/v1/reactors",
379    tag = "reactors",
380    params(Pagination),
381    responses(
382        (status = 200, description = "Registered reactors",
383         body = inline(PaginatedResult<ReactorResponse>)),
384    ),
385    security(("bearer" = []))
386)]
387pub async fn list<C: Connection + Clone>(
388    user: AuthenticatedUser,
389    authz: AuthzData,
390    state: web::Data<AppState<C>>,
391    pagination: web::Query<Pagination>,
392) -> Result<HttpResponse, AxiamApiError> {
393    RequirePermission::new("reactors:list", Uuid::nil())
394        .check(&user, authz.get_ref().as_ref())
395        .await?;
396    let result = state
397        .events
398        .reactor_repo
399        .list(user.tenant_id, pagination.into_inner())
400        .await?;
401
402    // R2.3: one page of reactors is small (an admin console's page size, not
403    // the event-path scale reactors themselves dispatch at), so a sequential
404    // per-reactor health read is the honest cost of "current health" — the
405    // alternative is a stale cached number a fail_closed veto going quiet
406    // would not show up in promptly.
407    let mut items = Vec::with_capacity(result.items.len());
408    for reactor in result.items {
409        let health = reactor_health(&state.audit_repo, user.tenant_id, reactor.id).await;
410        items.push(ReactorResponse::from(reactor).with_health(health));
411    }
412
413    Ok(HttpResponse::Ok().json(PaginatedResult {
414        items,
415        total: result.total,
416        offset: result.offset,
417        limit: result.limit,
418    }))
419}

448/// `PUT /api/v1/reactors/{id}`
449#[utoipa::path(
450    put,
451    path = "/api/v1/reactors/{id}",
452    tag = "reactors",
453    params(("id" = Uuid, Path, description = "Reactor ID")),
454    request_body = UpdateReactorRequest,
455    responses(
456        (status = 200, description = "Reactor updated", body = ReactorResponse),
457        (status = 400, description = "The merged registration is invalid"),
458    ),
459    security(("bearer" = []))
460)]
461pub async fn update<C: Connection + Clone>(
462    user: AuthenticatedUser,
463    authz: AuthzData,
464    path: web::Path<Uuid>,
465    state: web::Data<AppState<C>>,
466    body: web::Json<UpdateReactorRequest>,
467) -> Result<HttpResponse, AxiamApiError> {
468    RequirePermission::new("reactors:update", Uuid::nil())
469        .check(&user, authz.get_ref().as_ref())
470        .await?;
471    let id = path.into_inner();
472    let req = body.into_inner();
473
474    // Validate the MERGED registration, not the request.
475    //
476    // A PUT that only sets `mode: intercept` is valid in isolation and
477    // invalid against a stored `events` list containing a listen-only event.
478    // Validating the request alone would let exactly that combination through
479    // — the request names one field and the violation is in the pair.
480    let current = state
481        .events
482        .reactor_repo
483        .get_by_id(user.tenant_id, id)
484        .await?;
485    let merged_name = req.name.clone().unwrap_or_else(|| current.name.clone());
486    let merged_events = req.events.clone().unwrap_or_else(|| current.events.clone());
487    let merged_mode = req.mode.unwrap_or(current.mode);
488    let merged_timeout = req.timeout_ms.unwrap_or(current.timeout_ms);
489    let merged_enabled = req.enabled.unwrap_or(current.enabled);
490
491    // SEC-101, on the merged result: an update that leaves the registration
492    // enabled is refused while the transport cannot deliver, and one that
493    // disables it is always allowed — that is the operator's way out.
494    require_dispatchable_transport(state.get_ref(), merged_enabled, merged_mode)?;
495
496    validate_registration(&merged_name, &merged_events, merged_mode, merged_timeout)
497        .map_err(|e| validation_err(e.to_string()))?;
498
499    // Changing the event set can change what the STORED failure_policy means.
500    // If the caller replaces the events and does not name a policy, re-derive
501    // it: a reactor that was enrichment-only (fail_open) and is now also
502    // registered for `login.post_auth` must not keep passing when unreachable.
503    let failure_policy = match (req.failure_policy, req.events.is_some()) {
504        (Some(explicit), _) => Some(explicit),
505        (None, true) => Some(default_failure_policy_for(&merged_events)),
506        (None, false) => None,
507    };
508
509    let reactor = state
510        .events
511        .reactor_repo
512        .update(
513            user.tenant_id,
514            id,
515            UpdateReactor {
516                name: req.name,
517                description: req.description,
518                events: req.events,
519                mode: req.mode,
520                priority: req.priority,
521                timeout_ms: req.timeout_ms,
522                failure_policy,
523                enabled: req.enabled,
524            },
525        )
526        .await?;
527
528    state.invalidate_reactor_routing(user.tenant_id);
529
530    Ok(HttpResponse::Ok().json(ReactorResponse::from(reactor)))
531}

546    state: web::Data<AppState<C>>,
547) -> Result<HttpResponse, AxiamApiError> {

549        .check(&user, authz.get_ref().as_ref())

551    state

554        .delete(user.tenant_id, path.into_inner())
555        .await?;
556
557    // Deleting matters most of the three: an interceptor that no longer exists
558    // must stop being dispatched to, and a `fail_closed` one that is still in
559    // the routing table would deny every login it was registered for until the
560    // TTL expired.
561    state.invalidate_reactor_routing(user.tenant_id);
562
563    Ok(HttpResponse::NoContent().finish())
