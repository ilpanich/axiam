338/// Check a requested `tenant_scope` and turn it into an [`AssignmentScope`].
339///
340/// # What a tenant scope means, and where it is legal
341///
342/// It answers "which of this organization's tenants does this assignment
343/// reach", and that is only a question worth asking where the answer could be
344/// more than one — an assignment made in an **organization scope** tenant,
345/// whose global roles otherwise carry into every tenant of the organization.
346///
347/// In an ordinary tenant the only tenant an assignment could name is that
348/// tenant itself, so a scope there is either a no-op or a contradiction. Both
349/// are refused rather than accepted and quietly ignored: an operator who wrote
350/// a restriction and got a 204 would reasonably believe one was applied.
351///
352/// # The three refusals
353///
354/// * **Not an organization scope.** As above.
355/// * **Empty list.** An assignment that reaches no tenant contributes nothing
356///   anywhere. It is not a restriction, it is a grant that does not exist, and
357///   writing one produces a role that appears on the user's page and does
358///   nothing — the least debuggable outcome available.
359/// * **A tenant outside the organization, or the organization's own scope
360///   tenant.** The first would cross the boundary an organization *is*. The
361///   second would hand back the organization-wide reach the restriction exists
362///   to remove, in the one place `require_organization_principal` looks for
363///   it.
364async fn validate_tenant_scope<C: Connection + Clone>(
365    state: &AppState<C>,
366    assignment_tenant_id: Uuid,
367    resource_id: Option<Uuid>,
368    tenant_scope: Option<Vec<Uuid>>,
369) -> Result<AssignmentScope, AxiamApiError> {
370    let Some(tenants) = tenant_scope else {
371        return Ok(AssignmentScope {
372            resource_id,
373            tenant_scope: None,
374        });
375    };
376
377    let invalid = |reason: &str| -> AxiamApiError {
378        AxiamApiError(AxiamError::Validation {
379            message: format!("tenant_scope: {reason}"),
380        })
381    };
382
383    let home = state
384        .tenant_repo
385        .get_by_id(assignment_tenant_id)
386        .await
387        .map_err(|_| invalid("the tenant this assignment is made in could not be resolved"))?;
388    if !home.is_organization_scope() {
389        return Err(invalid(
390            "a tenant scope only applies to an assignment made in an organization's \
391             scope; a role in an ordinary tenant already reaches that tenant and \
392             no other",
393        ));
394    }
395    if tenants.is_empty() {
396        return Err(invalid(
397            "an empty tenant scope reaches no tenant at all — omit the field to \
398             reach every tenant of the organization",
399        ));
400    }
401
402    let mut seen = std::collections::BTreeSet::new();
403    for tenant_id in &tenants {
404        if tenant_id == &assignment_tenant_id {
405            return Err(invalid(
406                "the organization's own scope tenant cannot be named: an \
407                 assignment restricted to particular tenants is deliberately not \
408                 an organization-wide one",
409            ));
410        }
411        let target = state
412            .tenant_repo
413            .get_by_id(*tenant_id)
414            .await
415            .map_err(|_| invalid("a tenant named in the scope does not exist"))?;
416        if target.organization_id != home.organization_id {
417            return Err(invalid(
418                "a tenant named in the scope belongs to a different organization",
419            ));
420        }
421        seen.insert(*tenant_id);
422    }
423
424    Ok(AssignmentScope {
425        resource_id,
426        // Deduplicated and ordered, so two requests naming the same tenants in
427        // different orders produce the same stored edge.
428        tenant_scope: Some(seen.into_iter().collect()),
429    })
430}

436/// `POST /api/v1/roles/{role_id}/users`
437#[utoipa::path(
438    post,
439    path = "/api/v1/roles/{role_id}/users",
440    tag = "roles",
441    params(("role_id" = Uuid, Path, description = "Role ID")),
442    request_body = AssignRoleToUserRequest,
443    responses(
444        (status = 204, description = "Role assigned to user"),
445    ),
446    security(("bearer" = []))
447)]
448pub async fn assign_to_user<C: Connection + Clone>(
449    user: AuthenticatedUser,
450    authz: AuthzData,
451    state: web::Data<AppState<C>>,
452    path: web::Path<Uuid>,
453    body: web::Json<AssignRoleToUserRequest>,
454) -> Result<HttpResponse, AxiamApiError> {
455    RequirePermission::new("roles:assign", Uuid::nil())
456        .check(&user, authz.get_ref().as_ref())
457        .await?;
458    let req = body.into_inner();
459    let target_user = req.user_id;
460    let role_id = path.into_inner();
461
462    // X1 `grant.pre_assign` — the four-eyes hook, veto-only. Before the write:
463    // an approval workflow that ran after the grant existed would be reviewing
464    // access the subject already has.
465    crate::reactor_hooks::grant_pre_assign(
466        &state.events.reactor_gate,
467        user.tenant_id,
468        serde_json::json!({
469            "grantee_kind": "user",
470            "user_id": target_user,
471            "role_id": role_id,
472            "resource_id": req.resource_id,
473            // How far the grant reaches, not just what it covers: a four-eyes
474            // rule that cannot see whether a grant spans every tenant of the
475            // organization or two of them is reviewing half of it.
476            "tenant_scope": req.tenant_scope,
477            "tenant_id": user.tenant_id,
478            // Who is asking — the half a four-eyes rule is actually about.
479            "actor_id": user.user_id,
480        }),
481    )
482    .await?;
483
484    let scope = validate_tenant_scope(
485        state.get_ref(),
486        user.tenant_id,
487        req.resource_id,
488        req.tenant_scope,
489    )
490    .await?;
491    state
492        .role_repo
493        .assign_to_user(user.tenant_id, req.user_id, role_id, scope)
494        .await?;
495    // D7: only this subject's effective permissions change — targeted flush.
496    // (Assignment widens access, the safe direction, but we invalidate anyway
497    // so the new grant is visible immediately rather than after the TTL.)
498    authz
499        .get_ref()
500        .as_ref()
501        .invalidate_subject(user.tenant_id, target_user)
502        .await?;
503    Ok(HttpResponse::NoContent().finish())
504}

508    delete,
509    path = "/api/v1/roles/{role_id}/users/{user_id}",

512        ("role_id" = Uuid, Path, description = "Role ID"),
513        ("user_id" = Uuid, Path, description = "User ID"),

515    ),
516    responses(
517        (status = 204, description = "Role unassigned from user"),
518    ),
519    security(("bearer" = []))
520)]
521pub async fn unassign_from_user<C: Connection + Clone>(
522    user: AuthenticatedUser,
523    authz: AuthzData,
524    state: web::Data<AppState<C>>,
525    path: web::Path<RoleUserPath>,
526    query: web::Query<UnassignQuery>,
527) -> Result<HttpResponse, AxiamApiError> {
528    RequirePermission::new("roles:unassign", Uuid::nil())
529        .check(&user, authz.get_ref().as_ref())
530        .await?;
531    let p = path.into_inner();
532    state
533        .role_repo
534        .unassign_from_user(user.tenant_id, p.user_id, p.role_id, query.resource_id)
535        .await?;
536    // D7 (REVOCATION — security critical): unassigning a role removes access
537    // for exactly this subject. Invalidate immediately so a cached allow cannot
538    // survive — a stale allow is the dangerous staleness direction, because a
539    // deny is the restrictive outcome and costs only a redundant re-check.
540    //
541    // (Pre-B1 this was justified by the engine being allow-wins. Under
542    // deny-override that premise no longer holds, but the conclusion does:
543    // see the decision_cache module docs for why, and for the addition that
544    // *can* narrow.)
545    authz
546        .get_ref()
547        .as_ref()
548        .invalidate_subject(user.tenant_id, p.user_id)
549        .await?;
550    Ok(HttpResponse::NoContent().finish())
551}

553// -----------------------------------------------------------------------
554// Handlers — Role ↔ Service account assignment
555//
556// A service account is a principal like any other, and the authorization engine
557// has always treated it as one: `RequirePermission::check_subject` applies RBAC
558// identically to a machine and a person, and `has_role` is declared
559// `User/ServiceAccount/Group -> Role`. What did not exist was any way to create
560// the edge — so a service account could authenticate and then do nothing, and
561// the only workaround was to give a machine a human's account.
562// -----------------------------------------------------------------------
563
564/// `POST /api/v1/roles/{role_id}/service-accounts`
565#[utoipa::path(
566    post,
567    path = "/api/v1/roles/{role_id}/service-accounts",
568    tag = "roles",
569    params(("role_id" = Uuid, Path, description = "Role ID")),
570    request_body = AssignRoleToServiceAccountRequest,
571    responses(
572        (status = 204, description = "Role assigned to service account"),
573        (status = 404, description = "Role or service account not found in this tenant"),
574    ),
575    security(("bearer" = []))
576)]
577pub async fn assign_to_service_account<C: Connection + Clone>(
578    user: AuthenticatedUser,
579    authz: AuthzData,
580    state: web::Data<AppState<C>>,
581    path: web::Path<Uuid>,
582    body: web::Json<AssignRoleToServiceAccountRequest>,
583) -> Result<HttpResponse, AxiamApiError> {
584    RequirePermission::new("roles:assign", Uuid::nil())
585        .check(&user, authz.get_ref().as_ref())
586        .await?;
587    let req = body.into_inner();
588    let role_id = path.into_inner();
589
590    // X1 `grant.pre_assign` — the same veto-only four-eyes hook the user and
591    // group paths fire, with `grantee_kind` naming which. A rule that reviews
592    // human grants and silently ignores machine ones reviews the easier half.
593    crate::reactor_hooks::grant_pre_assign(
594        &state.events.reactor_gate,
595        user.tenant_id,
596        serde_json::json!({
597            "grantee_kind": "service_account",
598            "service_account_id": req.service_account_id,
599            "role_id": role_id,
600            "resource_id": req.resource_id,
601            // How far the grant reaches, not just what it covers: a four-eyes
602            // rule that cannot see whether a grant spans every tenant of the
603            // organization or two of them is reviewing half of it.
604            "tenant_scope": req.tenant_scope,
605            "tenant_id": user.tenant_id,
606            "actor_id": user.user_id,
607        }),
608    )
609    .await?;
610
611    let scope = validate_tenant_scope(
612        state.get_ref(),
613        user.tenant_id,
614        req.resource_id,
615        req.tenant_scope,
616    )
617    .await?;
618    state
619        .role_repo
620        .assign_to_service_account(user.tenant_id, req.service_account_id, role_id, scope)
621        .await?;
622    // D7: the machine's own effective permissions changed — targeted flush, on
623    // the same terms as the user path. The engine caches by `(tenant, subject)`
624    // and does not care which table the subject came from.
625    authz
626        .get_ref()
627        .as_ref()
628        .invalidate_subject(user.tenant_id, req.service_account_id)
629        .await?;
630    Ok(HttpResponse::NoContent().finish())
631}

764/// `POST /api/v1/roles/{role_id}/groups`
765#[utoipa::path(
766    post,
767    path = "/api/v1/roles/{role_id}/groups",
768    tag = "roles",
769    params(("role_id" = Uuid, Path, description = "Role ID")),
770    request_body = AssignRoleToGroupRequest,
771    responses(
772        (status = 204, description = "Role assigned to group"),
773    ),
774    security(("bearer" = []))
775)]
776pub async fn assign_to_group<C: Connection + Clone>(
777    user: AuthenticatedUser,
778    authz: AuthzData,
779    state: web::Data<AppState<C>>,
780    path: web::Path<Uuid>,
781    body: web::Json<AssignRoleToGroupRequest>,
782) -> Result<HttpResponse, AxiamApiError> {
783    RequirePermission::new("roles:assign", Uuid::nil())
784        .check(&user, authz.get_ref().as_ref())
785        .await?;
786    let req = body.into_inner();
787    let role_id = path.into_inner();
788
789    // X1 `grant.pre_assign` — the group half. A role assigned to a group is
790    // inherited by every member, so it is the *wider* of the two assignment
791    // paths and the one a four-eyes rule most needs to see.
792    crate::reactor_hooks::grant_pre_assign(
793        &state.events.reactor_gate,
794        user.tenant_id,
795        serde_json::json!({
796            "grantee_kind": "group",
797            "group_id": req.group_id,
798            "role_id": role_id,
799            "resource_id": req.resource_id,
800            // How far the grant reaches, not just what it covers: a four-eyes
801            // rule that cannot see whether a grant spans every tenant of the
802            // organization or two of them is reviewing half of it.
803            "tenant_scope": req.tenant_scope,
804            "tenant_id": user.tenant_id,
805            "actor_id": user.user_id,
806        }),
807    )
808    .await?;
809
810    let scope = validate_tenant_scope(
811        state.get_ref(),
812        user.tenant_id,
813        req.resource_id,
814        req.tenant_scope,
815    )
816    .await?;
817    state
818        .role_repo
819        .assign_to_group(user.tenant_id, req.group_id, role_id, scope)
820        .await?;
821    // D7: the affected subjects are every member of the group (set unknown
822    // without a query) — conservative per-tenant flush.
823    authz
824        .get_ref()
825        .as_ref()
826        .invalidate_tenant(user.tenant_id)
827        .await?;
828    Ok(HttpResponse::NoContent().finish())
829}
