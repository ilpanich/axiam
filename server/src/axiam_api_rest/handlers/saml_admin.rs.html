1//! SAML service-provider registry and IdP signing-credential management
2//! (G-2, T23.2.5, contract §29).
3//!
4//! Eleven routes under `/api/v1/tenants/{tenant_id}/saml`, in the `saml` OpenAPI
5//! tag. **Compiled into every build** (D-42), with or without the `saml` Cargo
6//! feature: the registry is plain data, its validator is outside the feature and
7//! the credential service is `axiam-pki`'s, so an SDK can reach them from the
8//! committed `openapi.json`. Only [`parse_sp_metadata`] needs `samael` and
9//! answers `503` in a build without it. **They do not depend on the tenant's
10//! `saml_idp_enabled`**: an administrator registers SPs and issues the
11//! credential *before* switching the IdP on (D-20's own reasoning); the setting
12//! governs the browser routes only.
13//!
14//! The routes are thin. What a registration may be is decided by
15//! `axiam_federation::saml_sp` (the validator and the D-42 refusals), what a
16//! credential is by `axiam_pki::SamlIdpCredentialService`, and what a metadata
17//! document means by `axiam_federation::saml_idp::sp_metadata`; this module
18//! **calls** them on every write and turns the answers into the statuses §29.3
19//! pins.
20//!
21//! # What every write does, in this order
22//!
23//! 1. the **permission** — `saml_sp:read` for the reads, `saml_sp:write` for the
24//!    SP writes and `parse_sp_metadata`, `saml_idp:credential` for the three
25//!    credential writes, kept apart because one of them can stop sign-on at
26//!    every SP of the tenant (T-364) — and the **tenant fence**: the path tenant
27//!    must be the caller's (`403`, T-357). A service-account token never gets
28//!    this far: the extractor takes human principals only (`saml_sp` and
29//!    `saml_idp` are human-only permission families), so it is refused with the
30//!    same `401` as on every other human-only route;
31//! 2. the validator, then the four D-42 refusals (`400`, each naming its rule);
32//! 3. the write, with the datastore's uniqueness answers as `409 conflict`;
33//! 4. one audit row, with the actor, the ids and the **names** of what changed —
34//!    never a certificate, a document or a key (T-362).
35//!
36//! # No key, ever
37//!
38//! [`SamlIdpCredential`] is a response type of its own — certificate, serial,
39//! fingerprint, dates, status, issuer CA — built from
40//! `axiam_core::models::saml_idp_credential::SamlIdpCredential`, which carries no
41//! key field, and never from the sealed type (T-365). The core type derives no
42//! `Serialize`, so a route cannot expose the row by accident.
43

94/// Where a signing credential is in its life. An open set: an SDK decodes a
95/// value it does not know without failing.
96#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, utoipa::ToSchema)]
97#[serde(rename_all = "snake_case")]
98pub enum SamlIdpCredentialStatus {
99    /// Signs the assertions the tenant issues now.
100    Active,
101    /// Published in metadata, not yet signing.
102    Next,
103    /// Out of metadata and out of use; its key is destroyed.
104    Retired,
105}
106
107impl From<StoredStatus> for SamlIdpCredentialStatus {
108    fn from(status: StoredStatus) -> Self {
109        match status {
110            StoredStatus::Active => Self::Active,
111            StoredStatus::Next => Self::Next,
112            StoredStatus::Retired => Self::Retired,
113        }
114    }
115}

117/// The tenant's IdP signing credential, **public facts only**.
118///
119/// There is no key on it and no field a key could be put in: the private key is
120/// generated by the server, sealed at rest, never returned by any route and
121/// destroyed on retirement (D-21).
122#[derive(Debug, Serialize, utoipa::ToSchema)]
123pub struct SamlIdpCredential {
124    /// Credential id.
125    pub id: Uuid,
126    /// The tenant it signs for.
127    pub tenant_id: Uuid,
128    /// The signing CA that issued the leaf.
129    pub issuer_ca_id: Uuid,
130    /// The leaf certificate, PEM. Public: it is what the metadata publishes.
131    pub certificate_pem: String,
132    /// The certificate's serial, lower-case hex.
133    pub serial: String,
134    /// Lower-case hex SHA-256 of the certificate's DER — what an SP
135    /// administrator compares out of band.
136    pub fingerprint: String,
137    /// Start of the certificate's validity.
138    pub not_before: DateTime<Utc>,
139    /// End of the certificate's validity (at most 730 days after the start).
140    pub not_after: DateTime<Utc>,
141    /// `active`, `next` or `retired`. At most one `active` and one `next` per
142    /// tenant.
143    pub status: SamlIdpCredentialStatus,
144    /// When the credential was issued.
145    pub created_at: DateTime<Utc>,
146    /// When it was retired, or null.
147    pub retired_at: Option<DateTime<Utc>>,
148}
149
150impl From<StoredCredential> for SamlIdpCredential {
151    fn from(c: StoredCredential) -> Self {
152        // Field by field, so that a field added to the stored type is not on the
153        // wire until someone decides it should be.
154        Self {
155            id: c.id,
156            tenant_id: c.tenant_id,
157            issuer_ca_id: c.issuer_ca_id,
158            certificate_pem: c.certificate_pem,
159            serial: c.serial,
160            fingerprint: c.fingerprint,
161            not_before: c.not_before,
162            not_after: c.not_after,
163            status: c.status.into(),
164            created_at: c.created_at,
165            retired_at: c.retired_at,
166        }
167    }
168}

266/// The ordinary request-size limit of the registry routes. A registration with
267/// 32 ACS endpoints, 64 mappings, 256 groups and two 16 KiB certificates fits
268/// with room to spare.
269const REGISTRY_BODY_LIMIT: usize = 131_072;
270/// The limit of the metadata-parse route: a 512 KiB document, JSON-escaped.
271const PARSE_BODY_LIMIT: usize = 1_048_576;

273fn json_error(err: &JsonPayloadError, what: &str, scrub: bool) -> AxiamApiError {
274    let message = match err {
275        JsonPayloadError::Deserialize(inner) if scrub => format!(
276            "the request body is not a valid {what}: {}",
277            super::directory::scrub_serde_message(&inner.to_string())
278        ),
279        JsonPayloadError::Deserialize(_) => format!("the request body is not a valid {what}"),
280        JsonPayloadError::ContentType => "the request body must be application/json".to_string(),
281        JsonPayloadError::Overflow { .. } | JsonPayloadError::OverflowKnownLength { .. } => {
282            "the request body is too large".to_string()
283        }
284        _ => "the request body could not be read".to_string(),
285    };
286    AxiamApiError(AxiamError::Validation { message })
287}

289/// The JSON configuration of the registry and credential routes: a body limit
290/// and a `400 validation_error` for a body that cannot be read, never serde's

295        .error_handler(|err: JsonPayloadError, _req: &HttpRequest| {

314/// `{tenant_id}` must be the caller's tenant, as for `directory` and
315/// `email_config`.
316fn require_own_tenant(
317    user: &AuthenticatedUser,
318    tenant_id: Uuid,
319    what: &str,
320) -> Result<(), AxiamApiError> {
321    if tenant_id != user.tenant_id {
322        return Err(AxiamApiError(AxiamError::AuthorizationDenied {
323            reason: format!("cannot {what} for a different tenant"),
324            action: None,
325            resource_id: None,
326        }));
327    }
328    Ok(())
329}

343/// Lower-case hex SHA-256 of the DER of a PEM certificate, or `None`.
344fn certificate_fingerprint(pem: &str) -> Option<String> {
345    let der = axiam_federation::cert::pem_cert_to_der(pem).ok()?;
346    Some(hex::encode(Sha256::digest(der)))
347}

349/// One audit row, never failing the request: the write it describes has been
350/// done, or refused, already.
351#[allow(clippy::too_many_arguments)] // an audit row is this wide
352async fn audit<C: Connection + Clone>(
353    state: &AppState<C>,
354    http_req: &HttpRequest,
355    user: &AuthenticatedUser,
356    tenant_id: Uuid,
357    action: &str,
358    resource_id: Option<Uuid>,
359    outcome: AuditOutcome,
360    metadata: serde_json::Value,
361) {
362    if let Err(error) = state
363        .audit_repo
364        .append(CreateAuditLogEntry {
365            tenant_id,
366            actor_id: user.user_id,
367            actor_type: ActorType::User,
368            action: action.to_string(),
369            resource_id,
370            outcome,
371            ip_address: client_ip(http_req),
372            metadata: Some(metadata),
373        })
374        .await
375    {
376        tracing::error!(
377            target: "axiam::saml_admin",
378            %tenant_id,
379            action,
380            %error,
381            "a SAML registry audit row could not be written"
382        );
383    }
384}

386/// What the registry's write-time rules that need the datastore say: every
387/// `allowed_groups` entry must be a group of this tenant (D-42).
388async fn check_allowed_groups<C: Connection + Clone>(
389    state: &AppState<C>,
390    tenant_id: Uuid,
391    input: &SamlServiceProviderInput,
392) -> Result<(), AxiamApiError> {
393    if input.allowed_groups.is_empty() {
394        return Ok(());
395    }
396    let outside = state
397        .saml_idp
398        .sp_repo
399        .groups_outside_tenant(tenant_id, &input.allowed_groups)
400        .await?;
401    if outside.is_empty() {
402        return Ok(());
403    }
404    Err(validation(format!(
405        "allowed_groups names {} group(s) that are not groups of this tenant",
406        outside.len()
407    )))
408}

410/// The names of the registration's members that `new` changes relative to `old`.
411/// Names only: never a value, so no certificate, URL or mapping reaches the
412/// audit log through this list.
413fn changed_fields(old: &SamlServiceProvider, new: &SamlServiceProviderInput) -> Vec<&'static str> {
414    let mut out = Vec::new();
415    let mut note = |differs: bool, name: &'static str| {
416        if differs {
417            out.push(name);
418        }
419    };
420    note(old.enabled != new.enabled, "enabled");
421    note(old.display_name != new.display_name, "display_name");
422    note(old.entity_id != new.entity_id, "entity_id");
423    note(old.acs_urls != new.acs_urls, "acs_urls");
424    note(old.slo_url != new.slo_url, "slo_url");
425    note(old.slo_binding != new.slo_binding, "slo_binding");
426    note(old.name_id_format != new.name_id_format, "name_id_format");
427    note(old.sign_responses != new.sign_responses, "sign_responses");
428    note(
429        old.encrypt_assertions != new.encrypt_assertions,
430        "encrypt_assertions",
431    );
432    note(
433        old.sp_signing_cert_pem != new.sp_signing_cert_pem,
434        "sp_signing_cert_pem",
435    );
436    note(
437        old.sp_encryption_cert_pem != new.sp_encryption_cert_pem,
438        "sp_encryption_cert_pem",
439    );
440    note(
441        old.want_authn_requests_signed != new.want_authn_requests_signed,
442        "want_authn_requests_signed",
443    );
444    note(
445        old.allow_idp_initiated != new.allow_idp_initiated,
446        "allow_idp_initiated",
447    );
448    note(
449        old.attribute_mappings != new.attribute_mappings,
450        "attribute_mappings",
451    );
452    note(old.allowed_groups != new.allowed_groups, "allowed_groups");
453    out
454}

456/// The issuing scope this caller acts with: [`IssuingScope::Organization`] only
457/// for a principal whose own record lives in the organization's reserved scope
458/// (resolved from the caller's own tenant, never from the request); everyone
459/// else is confined to the signing CA of the tenant being acted on. This route
460/// takes human principals only, so the machine exclusion of
461/// `org_scope::is_organization_principal` is already the extractor's.
462async fn issuing_scope<C: Connection + Clone>(
463    user: &AuthenticatedUser,
464    state: &AppState<C>,
465) -> IssuingScope {
466    match state.tenant_repo.get_by_id(user.principal_tenant_id).await {
467        Ok(home) if home.is_organization_scope() => IssuingScope::Organization,
468        _ => IssuingScope::Tenant,
469    }
470}

485/// `GET /api/v1/tenants/{tenant_id}/saml/idp`
486///
487/// The tenant's IdP as an SP will meet it, and whether it answers yet:
488/// readiness is visible only here, never on the unauthenticated metadata route
489/// (T-368).
490#[utoipa::path(
491    get,
492    path = "/api/v1/tenants/{tenant_id}/saml/idp",
493    tag = "saml",
494    params(("tenant_id" = Uuid, Path, description = "Tenant ID")),
495    responses(
496        (status = 200, description = "The tenant's SAML IdP: its URLs, whether SAML is available \
497                                      in this build and enabled for the tenant, and its \
498                                      credential slots", body = SamlIdpInfo),
499        (status = 403, description = "Another tenant's IdP"),
500    ),
501    security(("bearer" = []))
502)]
503pub async fn get_idp<C: Connection + Clone>(
504    user: AuthenticatedUser,
505    authz: AuthzData,
506    state: web::Data<AppState<C>>,
507    path: web::Path<Uuid>,
508) -> Result<HttpResponse, AxiamApiError> {
509    RequirePermission::new("saml_sp:read", Uuid::nil())
510        .check(&user, authz.get_ref().as_ref())
511        .await?;
512    let tenant_id = path.into_inner();
513    require_own_tenant(&user, tenant_id, "read the SAML IdP")?;
514
515    let settings = state
516        .settings_repo
517        .get_effective_settings(user.org_id, tenant_id)
518        .await?;
519    let credentials = state.saml_idp.credential_service.list(tenant_id).await?;
520    let slot = |wanted: StoredStatus| {
521        credentials
522            .iter()
523            .find(|c| c.status == wanted)
524            .map(|c| c.id)
525    };
526    let active_credential_id = slot(StoredStatus::Active);
527    let next_credential_id = slot(StoredStatus::Next);
528
529    let base = state.auth_config.root_issuer();
530    let saml_available = cfg!(feature = "saml");
531    let saml_idp_enabled = settings.oidc.saml_idp_enabled;
532    let entity_id = idp_entity_id(base, tenant_id);
533    Ok(HttpResponse::Ok().json(SamlIdpInfo {
534        tenant_id,
535        saml_available,
536        saml_idp_enabled,
537        metadata_served: saml_available
538            && saml_idp_enabled
539            && (active_credential_id.is_some() || next_credential_id.is_some()),
540        metadata_url: entity_id.clone(),
541        entity_id,
542        sso_url: idp_sso_url(base, tenant_id),
543        slo_url: idp_slo_url(base, tenant_id),
544        active_credential_id,
545        next_credential_id,
546    }))
547}

931/// Why an import produced no draft. A refusal carries one of the three generic
932/// messages and its audit category; nothing else about the document or the
933/// response ever reaches the answer.
934enum ImportFailure {
935    // Only a build with `saml` has a parser to refuse a document.
936    #[cfg_attr(not(feature = "saml"), allow(dead_code))]
937    Refused {
938        message: &'static str,
939        category: &'static str,
940    },
941    /// The build has no SAML support.
942    #[cfg(not(feature = "saml"))]
943    Unavailable,
944}
945
946impl ImportFailure {
947    /// The answer: a `400` carrying the generic message, or the `503` of a build
948    /// without SAML.
949    fn into_api_error(self) -> AxiamApiError {
950        match self {
951            Self::Refused { message, .. } => validation(message),
952            #[cfg(not(feature = "saml"))]
953            Self::Unavailable => AxiamApiError(AxiamError::ServiceUnavailable(
954                "this server was built without SAML support".into(),
955            )),
956        }
957    }
958
959    const fn category(&self) -> &'static str {
960        match self {
961            Self::Refused { category, .. } => category,
962            #[cfg(not(feature = "saml"))]
963            Self::Unavailable => "saml_unavailable",
964        }
965    }
966}
967
968/// Fetch (for a URL) and parse the metadata. The parser is `samael`'s, so a build
969/// without `saml` has none and answers [`ImportFailure::Unavailable`].
970#[cfg(feature = "saml")]
971async fn import_metadata(
972    xml: Option<String>,
973    url: Option<String>,
974) -> Result<SamlSpMetadataDraft, ImportFailure> {
975    use axiam_federation::saml_idp::sp_metadata::{
976        MAX_SP_METADATA_BYTES, MetadataError, fetch_sp_metadata, parse_sp_metadata,
977    };
978    let refused = |error: MetadataError| ImportFailure::Refused {
979        message: error.message(),
980        category: error.category(),
981    };
982    let document = match (xml, url) {
983        (Some(xml), _) => {
984            if xml.len() > MAX_SP_METADATA_BYTES {
985                return Err(refused(MetadataError::NotSpMetadata));
986            }
987            xml.into_bytes()
988        }
989        // `false`: the guard's address and scheme rules in full. The `true` of its
990        // test seam is never passed by a route.
991        (None, Some(url)) => fetch_sp_metadata(&url, false).await.map_err(refused)?,
992        (None, None) => return Err(refused(MetadataError::NotSpMetadata)),
993    };
994    let draft = parse_sp_metadata(&document).map_err(refused)?;
995    Ok(SamlSpMetadataDraft {
996        service_provider: draft.service_provider,
997        signing_certificate_fingerprint: draft.signing_certificate_fingerprint,
998        encryption_certificate_fingerprint: draft.encryption_certificate_fingerprint,
999        warnings: draft.warnings,
1000    })
1001}

1006    _xml: Option<String>,
1007    _url: Option<String>,
1008) -> Result<SamlSpMetadataDraft, ImportFailure> {
1009    Err(ImportFailure::Unavailable)

1072pub async fn issue_idp_credential<C: Connection + Clone>(

1075    state: web::Data<AppState<C>>,
1076    http_req: HttpRequest,
1077    path: web::Path<Uuid>,
1078    body: web::Json<IssueSamlIdpCredential>,
1079) -> Result<HttpResponse, AxiamApiError> {
1080    RequirePermission::new("saml_idp:credential", Uuid::nil())
1081        .check(&user, authz.get_ref().as_ref())
1082        .await?;
1083    let tenant_id = path.into_inner();
1084    require_own_tenant(&user, tenant_id, "issue an IdP credential")?;
1085
1086    let request = body.into_inner();
1087    let validity_days = request.validity_days.unwrap_or(DEFAULT_VALIDITY_DAYS);
1088    if !(1..=MAX_VALIDITY_DAYS).contains(&validity_days) {
1089        return Err(validation(format!(
1090            "validity_days must be between 1 and {MAX_VALIDITY_DAYS}"
1091        )));
1092    }
1093    let status = match request.slot {
1094        SamlIdpSlot::Active => StoredStatus::Active,
1095        SamlIdpSlot::Next => StoredStatus::Next,
1096    };

1101    // no RSA-4096 key generation (T-363).
1102    let issued = state
1103        .saml_idp
1104        .credential_service
1105        .issue(
1106            user.org_id,
1107            tenant_id,
1108            scope,
1109            request.issuer_ca_id,
1110            u32::try_from(validity_days).unwrap_or(u32::MAX),
1111            status,
1112        )
1113        .await
1114        .map_err(|e| {
1115            conflict_from_already_exists(
1116                e,
1117                "the slot is occupied: retire the credential in it, or promote the next one",
1118            )
1119        })?;
1120
1121    audit(
1122        &state,
1123        &http_req,
1124        &user,
1125        tenant_id,
1126        AUDIT_CREDENTIAL_ISSUED,
1127        Some(issued.id),
1128        AuditOutcome::Success,
1129        serde_json::json!({
1130            "credential_id": issued.id,
1131            "issuer_ca_id": issued.issuer_ca_id,
1132            "slot": issued.status.as_str(),
1133            "fingerprint": issued.fingerprint,
1134            "validity_days": validity_days,
1135        }),
1136    )
1137    .await;
1138    Ok(HttpResponse::Created().json(SamlIdpCredential::from(issued)))
1139}

1160        (status = 429, description = "Rate limit"),

1170) -> Result<HttpResponse, AxiamApiError> {

1206/// `POST /api/v1/tenants/{tenant_id}/saml/idp-credentials/{credential_id}/retire`
1207///
1208/// Retires a `next` or an `active` credential and destroys its key. Retiring the
1209/// **`active`** one with no successor stops SAML sign-on for the whole tenant at
1210/// once — the incident response to a leaked key. Retiring a retired credential
1211/// returns it unchanged and writes no audit row.
1212#[utoipa::path(
1213    post,
1214    path = "/api/v1/tenants/{tenant_id}/saml/idp-credentials/{credential_id}/retire",
1215    tag = "saml",
1216    params(
1217        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
1218        ("credential_id" = Uuid, Path, description = "Credential ID"),
1219    ),
1220    responses(
1221        (status = 200, description = "The credential, retired (or as it already was)", body = SamlIdpCredential),
1222        (status = 403, description = "Another tenant's credentials"),
1223        (status = 404, description = "No such credential in this tenant"),
1224        (status = 429, description = "Rate limit"),
1225    ),
1226    security(("bearer" = []))
1227)]
1228pub async fn retire_idp_credential<C: Connection + Clone>(
1229    user: AuthenticatedUser,
1230    authz: AuthzData,
1231    state: web::Data<AppState<C>>,
1232    http_req: HttpRequest,
1233    path: web::Path<(Uuid, Uuid)>,
1234) -> Result<HttpResponse, AxiamApiError> {
1235    RequirePermission::new("saml_idp:credential", Uuid::nil())
1236        .check(&user, authz.get_ref().as_ref())
1237        .await?;
1238    let (tenant_id, credential_id) = path.into_inner();
1239    require_own_tenant(&user, tenant_id, "retire an IdP credential")?;
1240
1241    let service = &state.saml_idp.credential_service;
1242    // Read first: an unknown id is `404`, and a credential that is already retired
1243    // is answered as it is, with nothing written and nothing audited.
1244    let before = service
1245        .list(tenant_id)
1246        .await?
1247        .into_iter()
1248        .find(|c| c.id == credential_id)
1249        .ok_or_else(|| {
1250            AxiamApiError(AxiamError::NotFound {
1251                entity: "saml_idp_credential".into(),
1252                id: credential_id.to_string(),
1253            })
1254        })?;
1255    if before.status == StoredStatus::Retired {
1256        return Ok(HttpResponse::Ok().json(SamlIdpCredential::from(before)));
1257    }
1258
1259    let retired = service.retire(tenant_id, credential_id).await?;
1260    let still_active = service.get_active(tenant_id).await?.is_some();
1261    audit(
1262        &state,
1263        &http_req,
1264        &user,
1265        tenant_id,
1266        AUDIT_CREDENTIAL_RETIRED,
1267        Some(retired.id),
1268        AuditOutcome::Success,
1269        serde_json::json!({
1270            "credential_id": retired.id,
1271            "slot": before.status.as_str(),
1272            "fingerprint": retired.fingerprint,
1273            "tenant_has_active_credential": still_active,
1274        }),
1275    )
1276    .await;
1277    Ok(HttpResponse::Ok().json(SamlIdpCredential::from(retired)))
1278}
