1//! Admin endpoints for SCIM provisioning tokens.
2//!
3//! See `claude_dev/scim-provisioning-token-design.md`. These mint, list and
4//! revoke the long-lived credential an IdP pastes into its SCIM connector;
5//! the credential itself is consumed in `axiam_scim`, never here.
6//!
7//! Gated on `scim_tokens:*`, deliberately separate from `scim:provision`:
8//! minting a credential for a provisioner is an administrative act, and the
9//! provisioner should not be able to mint more of itself.
10

31#[derive(Debug, Deserialize, utoipa::ToSchema)]
32pub struct CreateScimTokenRequest {
33    /// Operator-facing label, e.g. `"okta-production"`.
34    pub name: String,
35    /// The tenant user the token authenticates as. Must already hold
36    /// `scim:provision` — see [`create`].
37    pub user_id: Uuid,
38    /// Lifetime in days. Defaults to the deployment maximum, and is refused
39    /// above it.
40    #[serde(default)]
41    pub expires_in_days: Option<i64>,
42}
43
44/// Metadata only. The handle is never in a list response — it exists in
45/// plaintext exactly once, in [`CreateScimTokenResponse`].
46#[derive(Debug, Serialize, utoipa::ToSchema)]
47pub struct ScimTokenResponse {
48    pub id: Uuid,
49    pub tenant_id: Uuid,
50    pub user_id: Uuid,
51    pub name: String,
52    pub created_by: Uuid,
53    pub status: ScimTokenStatus,
54    pub expires_at: DateTime<Utc>,
55    pub last_used_at: Option<DateTime<Utc>>,
56    pub revoked_at: Option<DateTime<Utc>>,
57    pub created_at: DateTime<Utc>,
58}
59
60impl ScimTokenResponse {
61    fn from_token(t: ScimToken, now: DateTime<Utc>) -> Self {
62        Self {
63            status: t.status(now),
64            id: t.id,
65            tenant_id: t.tenant_id,
66            user_id: t.user_id,
67            name: t.name,
68            created_by: t.created_by,
69            expires_at: t.expires_at,
70            last_used_at: t.last_used_at,
71            revoked_at: t.revoked_at,
72            created_at: t.created_at,
73        }
74    }
75}

77/// The one-time reveal. Same shape as service-account creation: the secret is
78/// returned once and only its hash is kept.
79#[derive(Debug, Serialize, utoipa::ToSchema)]
80pub struct CreateScimTokenResponse {
81    #[serde(flatten)]
82    pub token: ScimTokenResponse,
83    /// The plaintext handle — shown once, never retrievable again.
84    pub provisioning_token: String,
85}

91/// Deployment ceiling on a token's lifetime.
92///
93/// Read from `AXIAM__SCIM_TOKEN_MAX_LIFETIME_DAYS`, falling back to
94/// [`DEFAULT_MAX_LIFETIME_DAYS`]. An unparseable or non-positive value falls
95/// back rather than failing the request: a malformed env var should not take
96/// provisioning offline, and the fallback is the conservative direction.
97fn max_lifetime_days() -> i64 {
98    std::env::var("AXIAM__SCIM_TOKEN_MAX_LIFETIME_DAYS")
99        .ok()
100        .and_then(|v| v.trim().parse::<i64>().ok())
101        .filter(|d| *d > 0)
102        .unwrap_or(DEFAULT_MAX_LIFETIME_DAYS)
103}

105/// `POST /api/v1/scim-tokens`
106#[utoipa::path(
107    post,
108    path = "/api/v1/scim-tokens",
109    tag = "scim-tokens",
110    request_body = CreateScimTokenRequest,
111    responses(
112        (status = 201, description = "Token minted; the handle is returned once",
113         body = CreateScimTokenResponse),
114        (status = 400, description = "Invalid lifetime, or the named user cannot use SCIM"),
115    ),
116    security(("bearer" = []))
117)]
118pub async fn create<C: Connection + Clone>(
119    user: AuthenticatedUser,
120    authz: AuthzData,
121    state: web::Data<AppState<C>>,
122    body: web::Json<CreateScimTokenRequest>,
123) -> Result<HttpResponse, AxiamApiError> {
124    RequirePermission::new("scim_tokens:create", Uuid::nil())
125        .check(&user, authz.get_ref().as_ref())
126        .await?;
127
128    let input = body.into_inner();
129    let name = input.name.trim().to_string();
130    if name.is_empty() {
131        return Err(AxiamApiError(AxiamError::Validation {
132            message: "name must not be empty".into(),
133        }));
134    }
135
136    let max_days = max_lifetime_days();
137    let days = input.expires_in_days.unwrap_or(max_days);
138    if days <= 0 || days > max_days {
139        return Err(AxiamApiError(AxiamError::Validation {
140            message: format!("expires_in_days must be between 1 and {max_days}"),
141        }));
142    }
143
144    // The bound user must exist in this tenant. Checked before the RBAC check
145    // below so a typo'd UUID answers "no such user" rather than the more
146    // confusing "that user cannot use SCIM".
147    state
148        .user_repo
149        .get_by_id(user.tenant_id, input.user_id)
150        .await?;
151
152    // ...and must actually hold `scim:provision`. A token that authenticates
153    // as somebody who cannot use SCIM only ever produces 403s; minting one
154    // silently is a trap that costs an operator an afternoon.
155    RequirePermission::new(
156        axiam_core::models::scim_token::SCIM_PROVISION_ACTION,
157        Uuid::nil(),
158    )
159    .check_subject(user.tenant_id, input.user_id, authz.get_ref().as_ref())
160    .await
161    .map_err(|_| {
162        AxiamApiError(AxiamError::Validation {
163            message: "the named user does not hold scim:provision, so a token bound to \
164                          them could not provision anything. Grant the permission first."
165                .into(),
166        })
167    })?;
168
169    let raw = format!(
170        "{SCIM_TOKEN_PREFIX}{}",
171        axiam_auth::token::generate_refresh_token()
172    );
173    let token_hash = axiam_auth::token::hash_refresh_token(&raw);
174
175    let created = state
176        .scim_token_repo
177        .create(CreateScimToken {
178            tenant_id: user.tenant_id,
179            user_id: input.user_id,
180            name,
181            token_hash,
182            created_by: user.user_id,
183            expires_at: Utc::now() + Duration::days(days),
184        })
185        .await?;
186
187    tracing::info!(
188        target: "axiam::audit",
189        event = "scim.token_created",
190        tenant_id = %user.tenant_id,
191        token_id = %created.id,
192        bound_user_id = %created.user_id,
193        created_by = %user.user_id,
194        expires_at = %created.expires_at,
195        "SCIM provisioning token minted"
196    );
197
198    Ok(HttpResponse::Created().json(CreateScimTokenResponse {
199        token: ScimTokenResponse::from_token(created, Utc::now()),
200        provisioning_token: raw,
201    }))
202}

204/// `GET /api/v1/scim-tokens`
205#[utoipa::path(
206    get,
207    path = "/api/v1/scim-tokens",
208    tag = "scim-tokens",
209    responses((status = 200, description = "Tokens for this tenant (metadata only)",
210               body = Vec<ScimTokenResponse>)),
211    security(("bearer" = []))
212)]
213pub async fn list<C: Connection + Clone>(
214    user: AuthenticatedUser,
215    authz: AuthzData,
216    state: web::Data<AppState<C>>,
217) -> Result<HttpResponse, AxiamApiError> {
218    RequirePermission::new("scim_tokens:list", Uuid::nil())
219        .check(&user, authz.get_ref().as_ref())
220        .await?;
221
222    let now = Utc::now();
223    let tokens: Vec<ScimTokenResponse> = state
224        .scim_token_repo
225        .list_for_tenant(user.tenant_id)
226        .await?
227        .into_iter()
228        .map(|t| ScimTokenResponse::from_token(t, now))
229        .collect();
230
231    Ok(HttpResponse::Ok().json(tokens))
232}

234/// `DELETE /api/v1/scim-tokens/{id}`
235///
236/// Revokes rather than deletes: the row is what lets an operator later answer
237/// "was this credential still live on the day of the incident?", and a
238/// deleted row answers nothing.
239#[utoipa::path(
240    delete,
241    path = "/api/v1/scim-tokens/{id}",
242    tag = "scim-tokens",
243    params(("id" = Uuid, Path, description = "Token ID")),
244    responses((status = 204, description = "Token revoked")),
245    security(("bearer" = []))
246)]
247pub async fn revoke<C: Connection + Clone>(
248    user: AuthenticatedUser,
249    authz: AuthzData,
250    state: web::Data<AppState<C>>,
251    path: web::Path<Uuid>,
252) -> Result<HttpResponse, AxiamApiError> {
253    RequirePermission::new("scim_tokens:revoke", Uuid::nil())
254        .check(&user, authz.get_ref().as_ref())
255        .await?;
256
257    let id = path.into_inner();
258    // Read first so an unknown id answers 404 rather than a silent 204.
259    state.scim_token_repo.get_by_id(user.tenant_id, id).await?;
260    state.scim_token_repo.revoke(user.tenant_id, id).await?;
261
262    tracing::info!(
263        target: "axiam::audit",
264        event = "scim.token_revoked",
265        tenant_id = %user.tenant_id,
266        token_id = %id,
267        revoked_by = %user.user_id,
268        "SCIM provisioning token revoked"
269    );
270
271    Ok(HttpResponse::NoContent().finish())
272}
