19/// Refuse an OPAQUE mode the running server has no keys to serve.
20///
21/// `validate_org_settings` cannot make this check: it lives in `axiam-core`,
22/// which the crate layering keeps below `axiam-api-rest`, so it can never see
23/// an `AppState`. The invariant is nonetheless the same one
24/// `POST /api/v1/admin/bootstrap` enforces — writing `opaque_mode` above
25/// `disabled` onto a server holding no OPAQUE keys configures a tenant into a
26/// state where every `/api/v1/auth/opaque/*` route fails at runtime, and the
27/// only way back out is another settings write. Refuse at the write instead,
28/// with the same two env vars named, so the operator learns what is missing
29/// rather than discovering it from a login that stopped working.
30fn reject_opaque_without_keys<C: Connection + Clone>(
31    mode: OpaqueMode,
32    state: &AppState<C>,
33) -> Result<(), AxiamApiError> {
34    if mode != OpaqueMode::Disabled && state.opaque_server.is_none() {
35        return Err(AxiamApiError(axiam_core::error::AxiamError::Validation {
36            message: "opaque_mode is set but AXIAM__AUTH__OPAQUE_SESSION_KEY and \
37                      AXIAM__AUTH__OPAQUE_SETUP_KEY are not configured: saving \
38                      this would enable OPAQUE on a server that cannot serve it, \
39                      and every /api/v1/auth/opaque/* request would fail"
40                .into(),
41        }));
42    }
43    Ok(())
44}

46/// Refuse a `default_locale` this build does not ship (W7, plan §4.6).
47///
48/// `validate_org_settings` cannot make this check either, and for a sharper
49/// version of the same reason `reject_opaque_without_keys` gives: the shipped
50/// locale list lives in `axiam_oauth2::locale`, which is *above* `axiam-core`
51/// in the crate layering, so the model literally cannot see it.
52///
53/// Refusing at the write rather than at the read is what makes the setting
54/// honest. `tenant_default_locale` treats an unshipped tag as no preference
55/// and falls back to `en` — the only safe thing it can do at that point — but
56/// an operator who typed `fr-CA` into the settings page and saw it saved would
57/// reasonably believe their sign-in pages were now in French. They would not
58/// be, and nothing would say so except a `warn!` in a log they are not
59/// reading. The error names the tags that do work.
60fn reject_unshipped_locale(tag: Option<&str>) -> Result<(), AxiamApiError> {
61    let Some(tag) = tag.map(str::trim).filter(|t| !t.is_empty()) else {
62        return Ok(());
63    };
64    if axiam_oauth2::locale::Locale::from_tag(tag).is_some() {
65        return Ok(());
66    }
67    Err(AxiamApiError(axiam_core::error::AxiamError::Validation {
68        message: format!(
69            "default_locale {tag:?} is not a language this build ships; the sign-in page \
70             would silently fall back to the deployment default. Shipped tags: {}",
71            axiam_oauth2::locale::ALL_LOCALES
72                .iter()
73                .map(|l| l.as_tag())
74                .collect::<Vec<_>>()
75                .join(", ")
76        ),
77    }))
78}

80/// Mint a tenant's OPAQUE key material now that OPAQUE is switched on.
81///
82/// `opaque_server_setup` was created lazily, on the first `/auth/opaque/*`
83/// request. That is correct — `get_or_create` is idempotent and is still the
84/// only way key material comes into existence — but it made "did enabling
85/// OPAQUE do anything?" unanswerable: the table stayed empty until somebody
86/// tried to sign in, which is exactly when an operator wants to have already
87/// found out whether the configuration works.
88///
89/// Provisioning at the write moves the one operation that can fail for
90/// server-side reasons — the AES-GCM seal, a database write — to the request
91/// the operator is watching. It stays best-effort: a tenant whose row could
92/// not be written here gets it on first use as before, so a transient failure
93/// delays visibility rather than refusing a settings change.
94async fn provision_opaque_setup<C: Connection + Clone>(
95    state: &AppState<C>,
96    tenant_id: Uuid,
97    suite: OpaqueSuite,
98) {
99    match crate::handlers::opaque::server_setup(state, tenant_id, suite).await {
100        Ok(_) => tracing::info!(
101            %tenant_id,
102            %suite,
103            "provisioned OPAQUE server setup for tenant"
104        ),
105        Err(e) => tracing::warn!(
106            %tenant_id,
107            %suite,
108            error = %e.0,
109            "could not provision OPAQUE server setup; it will be minted on first use"
110        ),
111    }
112}

114/// Every tenant of an organization, including its own reserved scope.
115///
116/// Paged to the end rather than taking the default first page: a partial sweep
117/// would leave later tenants un-provisioned, unreconciled or unchecked for no
118/// reason anybody could see.
119async fn all_tenants_of<C: Connection + Clone>(
120    state: &AppState<C>,
121    org_id: Uuid,
122) -> Result<Vec<axiam_core::models::tenant::Tenant>, AxiamApiError> {
123    let mut tenants = Vec::new();
124    let mut offset = 0u64;
125    loop {
126        let page = state
127            .tenant_repo
128            .list_by_organization(
129                org_id,
130                Pagination {
131                    offset,
132                    limit: 200,
133                    search: None,
134                },
135            )
136            .await?;
137        let count = page.items.len() as u64;
138        tenants.extend(page.items);
139        offset += count;
140        if count == 0 || offset >= page.total {
141            break;
142        }
143    }
144    Ok(tenants)
145}

147/// Refuse `opaque_mode = required` while it would strand anybody.
148///
149/// [`OpaqueMode::Required`] refuses password login for the whole tenant, before
150/// any credential is examined — deliberately, because refusing only the enrolled
151/// users would turn `/auth/login` into an oracle for who is enrolled. The cost
152/// is that every account without a registration record becomes unable to
153/// authenticate at all, and **nobody can be enrolled retroactively**: a record
154/// can only be built by a client holding the plaintext password, which the
155/// server never has.
156///
157/// So the switch is a one-way door for everyone on the wrong side of it,
158/// including the only administrator of the deployment — which is exactly what
159/// happened in practice: OPAQUE was turned on, no record had ever been written
160/// because the enrolment path was silently omitting them, and the next sign-in
161/// simply failed with nothing to say why.
162///
163/// Refusing here turns that into an error at the moment of the change, naming
164/// the tenants and the number of people in each. The way through is the
165/// migration the mode's documentation describes: run `optional` until coverage
166/// is complete, then switch.
167async fn refuse_incomplete_opaque_coverage<C: Connection + Clone>(
168    state: &AppState<C>,
169    tenants: &[axiam_core::models::tenant::Tenant],
170) -> Result<(), AxiamApiError> {
171    use axiam_core::repository::OpaqueCredentialRepository as _;
172
173    let mut stranded = Vec::new();
174    for tenant in tenants {
175        let count = state
176            .opaque_credential_repo
177            .count_active_users_without_credential(tenant.id)
178            .await?;
179        if count > 0 {
180            stranded.push(format!("{} ({count})", tenant.slug));
181        }
182    }
183
184    if !stranded.is_empty() {
185        return Err(AxiamApiError(AxiamError::Validation {
186            message: format!(
187                "opaque_mode `required` would lock out active users who have no \
188                 OPAQUE registration record, and no record can be created for them \
189                 retroactively. Uncovered users by tenant: {}. Run `optional` until \
190                 every active user has set a password — creation, change-password \
191                 and reset completion each enrol one — then switch to `required`.",
192                stranded.join(", "),
193            ),
194        }));
195    }
196    Ok(())
197}

199/// Re-apply the tighten-only rule to every tenant after the baseline moves.
200///
201/// A tenant override may only ever be *more* restrictive than the organization's
202/// baseline, and `validate_tenant_override` enforces that when the override is
203/// written. Nothing re-checked it when the **baseline** changed, so raising the
204/// organization's floor left every tenant that had written an override for that
205/// field sitting below the new one — for good, and invisibly.
206///
207/// Overrides that are still stricter are untouched: a tenant that chose a
208/// 24-character minimum keeps it when the organization moves from 12 to 16.
209/// Fields that are now weaker are **cleared** rather than rewritten to the new
210/// value, so the tenant tracks the baseline from here on instead of freezing at
211/// today's level and needing the same repair after the next change.
212///
213/// Best-effort per tenant: the read path clamps identically
214/// (`clamp_overrides_to_org` in the settings repository), so a tenant this fails
215/// to rewrite still *behaves* correctly — only its stored row stays untidy.
216/// Failing the operator's settings change over that would be the wrong trade.
217async fn reconcile_tenant_overrides<C: Connection + Clone>(
218    state: &AppState<C>,
219    org: &SecuritySettings,
220    tenants: &[axiam_core::models::tenant::Tenant],
221) {
222    for tenant in tenants {
223        let Ok(Some(mut overrides)) = state.settings_repo.get_tenant_override(tenant.id).await
224        else {
225            continue;
226        };
227        let cleared = clamp_overrides_to_org(org, &mut overrides);
228        if cleared.is_empty() {
229            continue;
230        }
231        match state
232            .settings_repo
233            .set_tenant_override(tenant.id, overrides)
234            .await
235        {
236            Ok(_) => tracing::info!(
237                tenant_id = %tenant.id,
238                tenant = %tenant.slug,
239                fields = ?cleared,
240                "cleared tenant overrides that the new org baseline overtook"
241            ),
242            Err(e) => tracing::warn!(
243                tenant_id = %tenant.id,
244                tenant = %tenant.slug,
245                fields = ?cleared,
246                error = %e,
247                "could not rewrite tenant overrides; the read path still clamps them"
248            ),
249        }
250    }
251}

253/// `GET /api/v1/organizations/{org_id}/settings`
254#[utoipa::path(
255    get,
256    path = "/api/v1/organizations/{org_id}/settings",
257    tag = "settings",
258    params(
259        ("org_id" = Uuid, Path, description = "Organization ID"),

261    responses(
262        (status = 200, description = "Organization security settings",
263         body = SecuritySettings),
264    ),
265    security(("bearer" = []))

267pub async fn get_org_settings<C: Connection + Clone>(
268    user: AuthenticatedUser,
269    authz: AuthzData,
270    state: web::Data<AppState<C>>,
271    path: web::Path<Uuid>,
272) -> Result<HttpResponse, AxiamApiError> {
273    // SECURITY (NEW-2): org-baseline settings are a higher privilege than the
274    // per-tenant `settings:*` self-management permissions. Enforce the dedicated
275    // org-level permission declared in ROUTE_PERMISSION_MAP so a tenant-scoped
276    // `settings:get` grant cannot read the org baseline that cascades to every
277    // sibling tenant.
278    RequirePermission::new("organizations:get_settings", Uuid::nil())
279        .check(&user, authz.get_ref().as_ref())
280        .await?;
281    let org_id = path.into_inner();
282
283    // Authorization: only allow reads for the authenticated user's own org.
284    if org_id != user.org_id {
285        return Err(AxiamApiError(
286            axiam_core::error::AxiamError::AuthorizationDenied {
287                reason: "cannot read settings for a different organization".into(),
288                action: None,
289                resource_id: None,
290            },
291        ));
292    }
293
294    let settings = state.settings_repo.get_org_settings(org_id).await?;
295    Ok(HttpResponse::Ok().json(settings))
296}

298/// `PUT /api/v1/organizations/{org_id}/settings`
299#[utoipa::path(
300    put,
301    path = "/api/v1/organizations/{org_id}/settings",
302    tag = "settings",
303    params(
304        ("org_id" = Uuid, Path, description = "Organization ID"),
305    ),
306    request_body = SetOrgSettings,
307    responses(
308        (status = 200, description = "Organization settings updated",
309         body = SecuritySettings),
310        (status = 400,
311         description = "Settings are internally inconsistent, or enable OPAQUE \
312                        on a server holding no OPAQUE keys"),
313    ),
314    security(("bearer" = []))
315)]
316pub async fn set_org_settings<C: Connection + Clone>(
317    user: AuthenticatedUser,
318    authz: AuthzData,
319    state: web::Data<AppState<C>>,
320    path: web::Path<Uuid>,
321    body: web::Json<SetOrgSettings>,
322) -> Result<HttpResponse, AxiamApiError> {
323    // SECURITY (NEW-2): enforce the dedicated org-level permission (declared in
324    // ROUTE_PERMISSION_MAP) rather than the tenant-scoped `settings:update`, so a
325    // tenant admin cannot rewrite the org security baseline inherited by every
326    // sibling tenant.
327    RequirePermission::new("organizations:update_settings", Uuid::nil())
328        .check(&user, authz.get_ref().as_ref())
329        .await?;
330    let org_id = path.into_inner();
331
332    // Authorization: only allow writes to the authenticated user's own org.
333    if org_id != user.org_id {
334        return Err(AxiamApiError(
335            axiam_core::error::AxiamError::AuthorizationDenied {
336                reason: "cannot modify settings for a different organization".into(),
337                action: None,
338                resource_id: None,
339            },
340        ));
341    }
342
343    let input = body.into_inner();
344    validate_org_settings(&input)?;
345    reject_unshipped_locale(input.default_locale.as_deref())?;
346    reject_opaque_without_keys(input.opaque_mode, &state)?;
347    let opaque_mode = input.opaque_mode;
348    let opaque_suite = input.opaque_suite;
349
350    // An organization baseline reaches every tenant, so everything below is
351    // organization-wide. Read once and reuse: the coverage gate, the key-material
352    // provisioning and the override reconciliation all walk the same list.
353    let tenants = all_tenants_of(&state, org_id).await?;
354
355    // Checked BEFORE the write, so a refusal leaves the deployment exactly as it
356    // was. `required` at the organization makes every tenant `required` — the
357    // tighten-only rule means no tenant can hold itself below the baseline — so
358    // every tenant is in scope, the organization's own reserved scope included.
359    if opaque_mode == OpaqueMode::Required {
360        refuse_incomplete_opaque_coverage(&state, &tenants).await?;
361    }
362
363    let settings = state.settings_repo.set_org_settings(org_id, input).await?;
364
365    // Switching OPAQUE on at the organization switches it on for every tenant
366    // that has not tightened past it, so every one of them needs key material.
367    if opaque_mode != OpaqueMode::Disabled {
368        for tenant in &tenants {
369            provision_opaque_setup(&state, tenant.id, opaque_suite).await;
370        }
371    }
372
373    // And every tenant override the new baseline has overtaken is cleared, so
374    // the change actually reaches the tenants that had one — which is what
375    // "propagated to all the already existing ones" has to mean.
376    reconcile_tenant_overrides(&state, &settings, &tenants).await;
377
378    Ok(HttpResponse::Ok().json(settings))
379}

410/// `PUT /api/v1/settings`
411///
412/// Set tenant-level overrides. Only fields that are **more restrictive**
413/// than the org baseline are accepted. Omit a field (set to `null`) to
414/// inherit from the org.
415#[utoipa::path(
416    put,
417    path = "/api/v1/settings",
418    tag = "settings",
419    request_body = TenantSettingsOverride,
420    responses(
421        (status = 200, description = "Tenant settings updated",
422         body = SecuritySettings),
423        (status = 400, description = "Override violates org baseline, or \
424                                      enables OPAQUE on a server holding no \
425                                      OPAQUE keys"),
426    ),
427    security(("bearer" = []))
428)]
429pub async fn set_tenant_settings<C: Connection + Clone>(
430    user: AuthenticatedUser,
431    authz: AuthzData,
432    state: web::Data<AppState<C>>,
433    body: web::Json<TenantSettingsOverride>,
434) -> Result<HttpResponse, AxiamApiError> {
435    RequirePermission::new("settings:update", Uuid::nil())
436        .check(&user, authz.get_ref().as_ref())
437        .await?;
438    let org = state.settings_repo.get_org_settings(user.org_id).await?;
439    let overrides = body.into_inner();
440
441    // Validate: tenant can only be more restrictive than org
442    validate_tenant_override(&org, &overrides)?;
443    reject_unshipped_locale(overrides.default_locale.as_deref())?;
444
445    // The same runtime-serviceability guard as the org write, but keyed on the
446    // override's own value rather than the merged one. A tenant that leaves
447    // `opaque_mode` unset inherits whatever the org baseline says and changes
448    // nothing, so it has no business being refused here — only an override
449    // that itself raises the mode is asking for something the server cannot do.
450    if let Some(mode) = overrides.opaque_mode {
451        reject_opaque_without_keys(mode, &state)?;
452    }
453
454    // Merge org baseline + overrides into a complete settings row.
455    // The ID passed here is just a placeholder; the underlying
456    // SurrealSettingsRepository derives a deterministic UUID (v5)
457    // from (scope, scope_id) and uses that as the canonical ID.
458    let merged = effective_settings(&org, &overrides, user.tenant_id, Uuid::nil());
459
460    let result = state
461        .settings_repo
462        .store_effective_tenant_settings(user.tenant_id, merged)
463        .await?;
464
465    if result.opaque.opaque_mode != OpaqueMode::Disabled {
466        provision_opaque_setup(&state, user.tenant_id, result.opaque.opaque_suite).await;
467    }
468
469    Ok(HttpResponse::Ok().json(result))
470}

532#[utoipa::path(
533    put,
534    path = "/api/v1/tenants/{tenant_id}/settings",
535    tag = "settings",
536    params(
537        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
538    ),
539    request_body = TenantSettingsOverride,
540    responses(
541        (status = 200, description = "The stored overrides, sparse",
542         body = TenantSettingsOverride),
543        (status = 400, description = "An override is less restrictive than the org \
544                                      baseline, or enables OPAQUE on a server holding \
545                                      no OPAQUE keys"),
546    ),
547    security(("bearer" = []))
548)]
549pub async fn set_tenant_override<C: Connection + Clone>(
550    user: AuthenticatedUser,
551    authz: AuthzData,
552    state: web::Data<AppState<C>>,
553    path: web::Path<Uuid>,
554    body: web::Json<TenantSettingsOverride>,
555) -> Result<HttpResponse, AxiamApiError> {
556    RequirePermission::new("settings:update", Uuid::nil())
557        .check(&user, authz.get_ref().as_ref())
558        .await?;
559    let tenant_id = path.into_inner();
560
561    if tenant_id != user.tenant_id {
562        return Err(AxiamApiError(AxiamError::AuthorizationDenied {
563            reason: "cannot modify security settings for a different tenant".into(),
564            action: None,
565            resource_id: None,
566        }));
567    }
568
569    let overrides = body.into_inner();
570    let org = state.settings_repo.get_org_settings(user.org_id).await?;
571    validate_tenant_override(&org, &overrides)?;
572    reject_unshipped_locale(overrides.default_locale.as_deref())?;
573    if let Some(mode) = overrides.opaque_mode {
574        reject_opaque_without_keys(mode, &state)?;
575    }
576
577    // Same one-way door as at the organization, checked against this tenant
578    // alone. Read from the *effective* mode rather than the override so that a
579    // tenant inheriting `required` from a baseline it is only now writing an
580    // unrelated override against is not gated twice — the organization write
581    // already established coverage for it.
582    let would_be = effective_settings(&org, &overrides, tenant_id, Uuid::nil());
583    if would_be.opaque.opaque_mode == OpaqueMode::Required
584        && overrides.opaque_mode == Some(OpaqueMode::Required)
585    {
586        let tenant = state.tenant_repo.get_by_id(tenant_id).await?;
587        refuse_incomplete_opaque_coverage(&state, std::slice::from_ref(&tenant)).await?;
588    }
589
590    let stored = state
591        .settings_repo
592        .set_tenant_override(tenant_id, overrides)
593        .await?;
594
595    let merged = effective_settings(&org, &stored, tenant_id, Uuid::nil());
596    if merged.opaque.opaque_mode != OpaqueMode::Disabled {
597        provision_opaque_setup(&state, tenant_id, merged.opaque.opaque_suite).await;
598    }
599
600    Ok(HttpResponse::Ok().json(stored))
601}

603/// `DELETE /api/v1/tenants/{tenant_id}/settings`
604#[utoipa::path(
605    delete,
606    path = "/api/v1/tenants/{tenant_id}/settings",
607    tag = "settings",
608    params(
609        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
610    ),
611    responses(
612        (status = 204, description = "Overrides cleared; the tenant inherits the \
613                                      org baseline entirely"),
614    ),
615    security(("bearer" = []))
616)]
617pub async fn delete_tenant_override<C: Connection + Clone>(
618    user: AuthenticatedUser,
619    authz: AuthzData,
620    state: web::Data<AppState<C>>,
621    path: web::Path<Uuid>,
622) -> Result<HttpResponse, AxiamApiError> {
623    RequirePermission::new("settings:update", Uuid::nil())
624        .check(&user, authz.get_ref().as_ref())
625        .await?;
626    let tenant_id = path.into_inner();
627
628    if tenant_id != user.tenant_id {
629        return Err(AxiamApiError(AxiamError::AuthorizationDenied {
630            reason: "cannot modify security settings for a different tenant".into(),
631            action: None,
632            resource_id: None,
633        }));
634    }
635
636    state
637        .settings_repo
638        .delete_tenant_override(tenant_id)
639        .await?;
640    Ok(HttpResponse::NoContent().finish())
641}
