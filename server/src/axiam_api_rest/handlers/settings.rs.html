19/// Refuse an OPAQUE mode the running server has no keys to serve.
20///
21/// `validate_org_settings` cannot make this check: it lives in `axiam-core`,
22/// which the crate layering keeps below `axiam-api-rest`, so it can never see
23/// an `AppState`. The invariant is nonetheless the same one
24/// `POST /api/v1/admin/bootstrap` enforces — writing `opaque_mode` above
25/// `disabled` onto a server holding no OPAQUE keys configures a tenant into a
26/// state where every `/api/v1/auth/opaque/*` route fails at runtime, and the
27/// only way back out is another settings write. Refuse at the write instead,
28/// with the same two env vars named, so the operator learns what is missing
29/// rather than discovering it from a login that stopped working.
30fn reject_opaque_without_keys<C: Connection + Clone>(
31    mode: OpaqueMode,
32    state: &AppState<C>,
33) -> Result<(), AxiamApiError> {
34    if mode != OpaqueMode::Disabled && state.opaque_server.is_none() {
35        return Err(AxiamApiError(axiam_core::error::AxiamError::Validation {
36            message: "opaque_mode is set but AXIAM__AUTH__OPAQUE_SESSION_KEY and \
37                      AXIAM__AUTH__OPAQUE_SETUP_KEY are not configured: saving \
38                      this would enable OPAQUE on a server that cannot serve it, \
39                      and every /api/v1/auth/opaque/* request would fail"
40                .into(),
41        }));
42    }
43    Ok(())
44}

46/// Mint a tenant's OPAQUE key material now that OPAQUE is switched on.
47///
48/// `opaque_server_setup` was created lazily, on the first `/auth/opaque/*`
49/// request. That is correct — `get_or_create` is idempotent and is still the
50/// only way key material comes into existence — but it made "did enabling
51/// OPAQUE do anything?" unanswerable: the table stayed empty until somebody
52/// tried to sign in, which is exactly when an operator wants to have already
53/// found out whether the configuration works.
54///
55/// Provisioning at the write moves the one operation that can fail for
56/// server-side reasons — the AES-GCM seal, a database write — to the request
57/// the operator is watching. It stays best-effort: a tenant whose row could
58/// not be written here gets it on first use as before, so a transient failure
59/// delays visibility rather than refusing a settings change.
60async fn provision_opaque_setup<C: Connection + Clone>(
61    state: &AppState<C>,
62    tenant_id: Uuid,
63    suite: OpaqueSuite,
64) {
65    match crate::handlers::opaque::server_setup(state, tenant_id, suite).await {
66        Ok(_) => tracing::info!(
67            %tenant_id,
68            %suite,
69            "provisioned OPAQUE server setup for tenant"
70        ),
71        Err(e) => tracing::warn!(
72            %tenant_id,
73            %suite,
74            error = %e.0,
75            "could not provision OPAQUE server setup; it will be minted on first use"
76        ),
77    }
78}

80/// `GET /api/v1/organizations/{org_id}/settings`
81#[utoipa::path(
82    get,
83    path = "/api/v1/organizations/{org_id}/settings",
84    tag = "settings",
85    params(
86        ("org_id" = Uuid, Path, description = "Organization ID"),
87    ),
88    responses(
89        (status = 200, description = "Organization security settings",
90         body = SecuritySettings),
91    ),
92    security(("bearer" = []))
93)]
94pub async fn get_org_settings<C: Connection + Clone>(
95    user: AuthenticatedUser,
96    authz: AuthzData,
97    state: web::Data<AppState<C>>,
98    path: web::Path<Uuid>,
99) -> Result<HttpResponse, AxiamApiError> {
100    // SECURITY (NEW-2): org-baseline settings are a higher privilege than the
101    // per-tenant `settings:*` self-management permissions. Enforce the dedicated
102    // org-level permission declared in ROUTE_PERMISSION_MAP so a tenant-scoped
103    // `settings:get` grant cannot read the org baseline that cascades to every
104    // sibling tenant.
105    RequirePermission::new("organizations:get_settings", Uuid::nil())
106        .check(&user, authz.get_ref().as_ref())
107        .await?;
108    let org_id = path.into_inner();
109
110    // Authorization: only allow reads for the authenticated user's own org.
111    if org_id != user.org_id {
112        return Err(AxiamApiError(
113            axiam_core::error::AxiamError::AuthorizationDenied {
114                reason: "cannot read settings for a different organization".into(),
115                action: None,
116                resource_id: None,
117            },
118        ));
119    }
120
121    let settings = state.settings_repo.get_org_settings(org_id).await?;
122    Ok(HttpResponse::Ok().json(settings))
123}

125/// `PUT /api/v1/organizations/{org_id}/settings`
126#[utoipa::path(
127    put,
128    path = "/api/v1/organizations/{org_id}/settings",
129    tag = "settings",
130    params(
131        ("org_id" = Uuid, Path, description = "Organization ID"),
132    ),
133    request_body = SetOrgSettings,
134    responses(
135        (status = 200, description = "Organization settings updated",
136         body = SecuritySettings),
137        (status = 400,
138         description = "Settings are internally inconsistent, or enable OPAQUE \
139                        on a server holding no OPAQUE keys"),
140    ),
141    security(("bearer" = []))
142)]
143pub async fn set_org_settings<C: Connection + Clone>(
144    user: AuthenticatedUser,
145    authz: AuthzData,
146    state: web::Data<AppState<C>>,
147    path: web::Path<Uuid>,
148    body: web::Json<SetOrgSettings>,
149) -> Result<HttpResponse, AxiamApiError> {
150    // SECURITY (NEW-2): enforce the dedicated org-level permission (declared in
151    // ROUTE_PERMISSION_MAP) rather than the tenant-scoped `settings:update`, so a
152    // tenant admin cannot rewrite the org security baseline inherited by every
153    // sibling tenant.
154    RequirePermission::new("organizations:update_settings", Uuid::nil())
155        .check(&user, authz.get_ref().as_ref())
156        .await?;
157    let org_id = path.into_inner();
158
159    // Authorization: only allow writes to the authenticated user's own org.
160    if org_id != user.org_id {
161        return Err(AxiamApiError(
162            axiam_core::error::AxiamError::AuthorizationDenied {
163                reason: "cannot modify settings for a different organization".into(),
164                action: None,
165                resource_id: None,
166            },
167        ));
168    }
169
170    let input = body.into_inner();
171    validate_org_settings(&input)?;
172    reject_opaque_without_keys(input.opaque_mode, &state)?;
173    let opaque_mode = input.opaque_mode;
174    let opaque_suite = input.opaque_suite;
175    let settings = state.settings_repo.set_org_settings(org_id, input).await?;
176
177    // Switching OPAQUE on at the organization switches it on for every tenant
178    // that has not tightened past it, so every one of them needs key material.
179    // Paged to the end rather than taking the default first page: a partial
180    // sweep would leave later tenants looking un-provisioned for no reason
181    // anybody could see.
182    if opaque_mode != OpaqueMode::Disabled {
183        let mut offset = 0u64;
184        loop {
185            let page = state
186                .tenant_repo
187                .list_by_organization(org_id, Pagination { offset, limit: 200 })
188                .await?;
189            let count = page.items.len() as u64;
190            for tenant in page.items {
191                provision_opaque_setup(&state, tenant.id, opaque_suite).await;
192            }
193            offset += count;
194            if count == 0 || offset >= page.total {
195                break;
196            }
197        }
198    }
199
200    Ok(HttpResponse::Ok().json(settings))
201}

232/// `PUT /api/v1/settings`
233///
234/// Set tenant-level overrides. Only fields that are **more restrictive**
235/// than the org baseline are accepted. Omit a field (set to `null`) to
236/// inherit from the org.
237#[utoipa::path(
238    put,
239    path = "/api/v1/settings",
240    tag = "settings",
241    request_body = TenantSettingsOverride,
242    responses(
243        (status = 200, description = "Tenant settings updated",
244         body = SecuritySettings),
245        (status = 400, description = "Override violates org baseline, or \
246                                      enables OPAQUE on a server holding no \
247                                      OPAQUE keys"),
248    ),
249    security(("bearer" = []))
250)]
251pub async fn set_tenant_settings<C: Connection + Clone>(
252    user: AuthenticatedUser,
253    authz: AuthzData,
254    state: web::Data<AppState<C>>,
255    body: web::Json<TenantSettingsOverride>,
256) -> Result<HttpResponse, AxiamApiError> {
257    RequirePermission::new("settings:update", Uuid::nil())
258        .check(&user, authz.get_ref().as_ref())
259        .await?;
260    let org = state.settings_repo.get_org_settings(user.org_id).await?;
261    let overrides = body.into_inner();
262
263    // Validate: tenant can only be more restrictive than org
264    validate_tenant_override(&org, &overrides)?;
265
266    // The same runtime-serviceability guard as the org write, but keyed on the
267    // override's own value rather than the merged one. A tenant that leaves
268    // `opaque_mode` unset inherits whatever the org baseline says and changes
269    // nothing, so it has no business being refused here — only an override
270    // that itself raises the mode is asking for something the server cannot do.
271    if let Some(mode) = overrides.opaque_mode {
272        reject_opaque_without_keys(mode, &state)?;
273    }
274
275    // Merge org baseline + overrides into a complete settings row.
276    // The ID passed here is just a placeholder; the underlying
277    // SurrealSettingsRepository derives a deterministic UUID (v5)
278    // from (scope, scope_id) and uses that as the canonical ID.
279    let merged = effective_settings(&org, &overrides, user.tenant_id, Uuid::nil());
280
281    let result = state
282        .settings_repo
283        .store_effective_tenant_settings(user.tenant_id, merged)
284        .await?;
285
286    if result.opaque.opaque_mode != OpaqueMode::Disabled {
287        provision_opaque_setup(&state, user.tenant_id, result.opaque.opaque_suite).await;
288    }
289
290    Ok(HttpResponse::Ok().json(result))
291}

293// -----------------------------------------------------------------------
294// Tenant-scoped settings — the same overrides, addressed by tenant id
295// -----------------------------------------------------------------------
296//
297// `GET`/`PUT /api/v1/settings` act on the caller's own tenant implicitly and
298// exchange *merged* settings, which is what a tenant administrator editing
299// their own policy wants. It is not what the organization's tenant detail page
300// needs: that page shows one named tenant and has to distinguish an overridden
301// field from an inherited one, which a merged view cannot express.
302//
303// These three mirror the email-config trio exactly — same `{tenant_id}` path
304// segment, same `tenant_id == user.tenant_id` check on top of the permission,
305// same "raw own-scope row, never the merged view" rule — so a sibling tenant
306// answers `403` here for the same reason and in the same shape it does there.
307
308/// `GET /api/v1/tenants/{tenant_id}/settings`
309#[utoipa::path(
310    get,
311    path = "/api/v1/tenants/{tenant_id}/settings",
312    tag = "settings",
313    params(
314        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
315    ),
316    responses(
317        (status = 200, description = "The tenant's own overrides, sparse — an absent \
318                                      field is inherited from the org baseline",
319         body = TenantSettingsOverride),
320        (status = 404, description = "This tenant overrides nothing"),
321    ),
322    security(("bearer" = []))
323)]
324pub async fn get_tenant_override<C: Connection + Clone>(
325    user: AuthenticatedUser,
326    authz: AuthzData,
327    state: web::Data<AppState<C>>,
328    path: web::Path<Uuid>,
329) -> Result<HttpResponse, AxiamApiError> {
330    RequirePermission::new("settings:get", Uuid::nil())
331        .check(&user, authz.get_ref().as_ref())
332        .await?;
333    let tenant_id = path.into_inner();
334
335    if tenant_id != user.tenant_id {
336        return Err(AxiamApiError(AxiamError::AuthorizationDenied {
337            reason: "cannot read security settings for a different tenant".into(),
338            action: None,
339            resource_id: None,
340        }));
341    }
342
343    match state.settings_repo.get_tenant_override(tenant_id).await? {
344        Some(overrides) => Ok(HttpResponse::Ok().json(overrides)),
345        None => Err(AxiamApiError(AxiamError::NotFound {
346            entity: "tenant_settings_override".into(),
347            id: tenant_id.to_string(),
348        })),
349    }
350}

353#[utoipa::path(
354    put,

356    tag = "settings",
357    params(
358        ("tenant_id" = Uuid, Path, description = "Tenant ID"),

364        (status = 400, description = "An override is less restrictive than the org \

370pub async fn set_tenant_override<C: Connection + Clone>(
371    user: AuthenticatedUser,
372    authz: AuthzData,
373    state: web::Data<AppState<C>>,
374    path: web::Path<Uuid>,
375    body: web::Json<TenantSettingsOverride>,
376) -> Result<HttpResponse, AxiamApiError> {
377    RequirePermission::new("settings:update", Uuid::nil())
378        .check(&user, authz.get_ref().as_ref())
379        .await?;
380    let tenant_id = path.into_inner();
381
382    if tenant_id != user.tenant_id {
383        return Err(AxiamApiError(AxiamError::AuthorizationDenied {
384            reason: "cannot modify security settings for a different tenant".into(),
385            action: None,
386            resource_id: None,
387        }));
388    }
389
390    let overrides = body.into_inner();
391    let org = state.settings_repo.get_org_settings(user.org_id).await?;
392    validate_tenant_override(&org, &overrides)?;
393    if let Some(mode) = overrides.opaque_mode {
394        reject_opaque_without_keys(mode, &state)?;
395    }
396
397    let stored = state
398        .settings_repo
399        .set_tenant_override(tenant_id, overrides)
400        .await?;
401
402    let merged = effective_settings(&org, &stored, tenant_id, Uuid::nil());
403    if merged.opaque.opaque_mode != OpaqueMode::Disabled {
404        provision_opaque_setup(&state, tenant_id, merged.opaque.opaque_suite).await;
405    }
406
407    Ok(HttpResponse::Ok().json(stored))
408}
