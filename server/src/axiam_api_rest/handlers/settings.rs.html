19/// Refuse an OPAQUE mode the running server has no keys to serve.
20///
21/// `validate_org_settings` cannot make this check: it lives in `axiam-core`,
22/// which the crate layering keeps below `axiam-api-rest`, so it can never see
23/// an `AppState`. The invariant is nonetheless the same one
24/// `POST /api/v1/admin/bootstrap` enforces — writing `opaque_mode` above
25/// `disabled` onto a server holding no OPAQUE keys configures a tenant into a
26/// state where every `/api/v1/auth/opaque/*` route fails at runtime, and the
27/// only way back out is another settings write. Refuse at the write instead,
28/// with the same two env vars named, so the operator learns what is missing
29/// rather than discovering it from a login that stopped working.
30fn reject_opaque_without_keys<C: Connection + Clone>(
31    mode: OpaqueMode,
32    state: &AppState<C>,
33) -> Result<(), AxiamApiError> {
34    if mode != OpaqueMode::Disabled && state.opaque_server.is_none() {
35        return Err(AxiamApiError(axiam_core::error::AxiamError::Validation {
36            message: "opaque_mode is set but AXIAM__AUTH__OPAQUE_SESSION_KEY and \
37                      AXIAM__AUTH__OPAQUE_SETUP_KEY are not configured: saving \
38                      this would enable OPAQUE on a server that cannot serve it, \
39                      and every /api/v1/auth/opaque/* request would fail"
40                .into(),
41        }));
42    }
43    Ok(())
44}

46/// Mint a tenant's OPAQUE key material now that OPAQUE is switched on.
47///
48/// `opaque_server_setup` was created lazily, on the first `/auth/opaque/*`
49/// request. That is correct — `get_or_create` is idempotent and is still the
50/// only way key material comes into existence — but it made "did enabling
51/// OPAQUE do anything?" unanswerable: the table stayed empty until somebody
52/// tried to sign in, which is exactly when an operator wants to have already
53/// found out whether the configuration works.
54///
55/// Provisioning at the write moves the one operation that can fail for
56/// server-side reasons — the AES-GCM seal, a database write — to the request
57/// the operator is watching. It stays best-effort: a tenant whose row could
58/// not be written here gets it on first use as before, so a transient failure
59/// delays visibility rather than refusing a settings change.
60async fn provision_opaque_setup<C: Connection + Clone>(
61    state: &AppState<C>,
62    tenant_id: Uuid,
63    suite: OpaqueSuite,
64) {
65    match crate::handlers::opaque::server_setup(state, tenant_id, suite).await {
66        Ok(_) => tracing::info!(
67            %tenant_id,
68            %suite,
69            "provisioned OPAQUE server setup for tenant"
70        ),
71        Err(e) => tracing::warn!(
72            %tenant_id,
73            %suite,
74            error = %e.0,
75            "could not provision OPAQUE server setup; it will be minted on first use"
76        ),
77    }
78}

80/// Every tenant of an organization, including its own reserved scope.
81///
82/// Paged to the end rather than taking the default first page: a partial sweep
83/// would leave later tenants un-provisioned, unreconciled or unchecked for no
84/// reason anybody could see.
85async fn all_tenants_of<C: Connection + Clone>(
86    state: &AppState<C>,
87    org_id: Uuid,
88) -> Result<Vec<axiam_core::models::tenant::Tenant>, AxiamApiError> {
89    let mut tenants = Vec::new();
90    let mut offset = 0u64;
91    loop {
92        let page = state
93            .tenant_repo
94            .list_by_organization(
95                org_id,
96                Pagination {
97                    offset,
98                    limit: 200,
99                    search: None,
100                },
101            )
102            .await?;
103        let count = page.items.len() as u64;
104        tenants.extend(page.items);
105        offset += count;
106        if count == 0 || offset >= page.total {
107            break;
108        }
109    }
110    Ok(tenants)
111}

113/// Refuse `opaque_mode = required` while it would strand anybody.
114///
115/// [`OpaqueMode::Required`] refuses password login for the whole tenant, before
116/// any credential is examined — deliberately, because refusing only the enrolled
117/// users would turn `/auth/login` into an oracle for who is enrolled. The cost
118/// is that every account without a registration record becomes unable to
119/// authenticate at all, and **nobody can be enrolled retroactively**: a record
120/// can only be built by a client holding the plaintext password, which the
121/// server never has.
122///
123/// So the switch is a one-way door for everyone on the wrong side of it,
124/// including the only administrator of the deployment — which is exactly what
125/// happened in practice: OPAQUE was turned on, no record had ever been written
126/// because the enrolment path was silently omitting them, and the next sign-in
127/// simply failed with nothing to say why.
128///
129/// Refusing here turns that into an error at the moment of the change, naming
130/// the tenants and the number of people in each. The way through is the
131/// migration the mode's documentation describes: run `optional` until coverage
132/// is complete, then switch.
133async fn refuse_incomplete_opaque_coverage<C: Connection + Clone>(
134    state: &AppState<C>,
135    tenants: &[axiam_core::models::tenant::Tenant],
136) -> Result<(), AxiamApiError> {
137    use axiam_core::repository::OpaqueCredentialRepository as _;
138
139    let mut stranded = Vec::new();
140    for tenant in tenants {
141        let count = state
142            .opaque_credential_repo
143            .count_active_users_without_credential(tenant.id)
144            .await?;
145        if count > 0 {
146            stranded.push(format!("{} ({count})", tenant.slug));
147        }
148    }
149
150    if !stranded.is_empty() {
151        return Err(AxiamApiError(AxiamError::Validation {
152            message: format!(
153                "opaque_mode `required` would lock out active users who have no \
154                 OPAQUE registration record, and no record can be created for them \
155                 retroactively. Uncovered users by tenant: {}. Run `optional` until \
156                 every active user has set a password — creation, change-password \
157                 and reset completion each enrol one — then switch to `required`.",
158                stranded.join(", "),
159            ),
160        }));
161    }
162    Ok(())
163}

165/// Re-apply the tighten-only rule to every tenant after the baseline moves.
166///
167/// A tenant override may only ever be *more* restrictive than the organization's
168/// baseline, and `validate_tenant_override` enforces that when the override is
169/// written. Nothing re-checked it when the **baseline** changed, so raising the
170/// organization's floor left every tenant that had written an override for that
171/// field sitting below the new one — for good, and invisibly.
172///
173/// Overrides that are still stricter are untouched: a tenant that chose a
174/// 24-character minimum keeps it when the organization moves from 12 to 16.
175/// Fields that are now weaker are **cleared** rather than rewritten to the new
176/// value, so the tenant tracks the baseline from here on instead of freezing at
177/// today's level and needing the same repair after the next change.
178///
179/// Best-effort per tenant: the read path clamps identically
180/// (`clamp_overrides_to_org` in the settings repository), so a tenant this fails
181/// to rewrite still *behaves* correctly — only its stored row stays untidy.
182/// Failing the operator's settings change over that would be the wrong trade.
183async fn reconcile_tenant_overrides<C: Connection + Clone>(
184    state: &AppState<C>,
185    org: &SecuritySettings,
186    tenants: &[axiam_core::models::tenant::Tenant],
187) {
188    for tenant in tenants {
189        let Ok(Some(mut overrides)) = state.settings_repo.get_tenant_override(tenant.id).await
190        else {
191            continue;
192        };
193        let cleared = clamp_overrides_to_org(org, &mut overrides);
194        if cleared.is_empty() {
195            continue;
196        }
197        match state
198            .settings_repo
199            .set_tenant_override(tenant.id, overrides)
200            .await
201        {
202            Ok(_) => tracing::info!(
203                tenant_id = %tenant.id,
204                tenant = %tenant.slug,
205                fields = ?cleared,
206                "cleared tenant overrides that the new org baseline overtook"
207            ),
208            Err(e) => tracing::warn!(
209                tenant_id = %tenant.id,
210                tenant = %tenant.slug,
211                fields = ?cleared,
212                error = %e,
213                "could not rewrite tenant overrides; the read path still clamps them"
214            ),
215        }
216    }
217}

233pub async fn get_org_settings<C: Connection + Clone>(
234    user: AuthenticatedUser,
235    authz: AuthzData,
236    state: web::Data<AppState<C>>,
237    path: web::Path<Uuid>,
238) -> Result<HttpResponse, AxiamApiError> {
239    // SECURITY (NEW-2): org-baseline settings are a higher privilege than the
240    // per-tenant `settings:*` self-management permissions. Enforce the dedicated
241    // org-level permission declared in ROUTE_PERMISSION_MAP so a tenant-scoped
242    // `settings:get` grant cannot read the org baseline that cascades to every
243    // sibling tenant.
244    RequirePermission::new("organizations:get_settings", Uuid::nil())
245        .check(&user, authz.get_ref().as_ref())
246        .await?;
247    let org_id = path.into_inner();
248
249    // Authorization: only allow reads for the authenticated user's own org.
250    if org_id != user.org_id {
251        return Err(AxiamApiError(
252            axiam_core::error::AxiamError::AuthorizationDenied {
253                reason: "cannot read settings for a different organization".into(),
254                action: None,
255                resource_id: None,
256            },
257        ));
258    }
259
260    let settings = state.settings_repo.get_org_settings(org_id).await?;
261    Ok(HttpResponse::Ok().json(settings))
262}

273    responses(

275         body = SecuritySettings),

282pub async fn set_org_settings<C: Connection + Clone>(
283    user: AuthenticatedUser,
284    authz: AuthzData,
285    state: web::Data<AppState<C>>,
286    path: web::Path<Uuid>,
287    body: web::Json<SetOrgSettings>,
288) -> Result<HttpResponse, AxiamApiError> {
289    // SECURITY (NEW-2): enforce the dedicated org-level permission (declared in
290    // ROUTE_PERMISSION_MAP) rather than the tenant-scoped `settings:update`, so a
291    // tenant admin cannot rewrite the org security baseline inherited by every
292    // sibling tenant.
293    RequirePermission::new("organizations:update_settings", Uuid::nil())
294        .check(&user, authz.get_ref().as_ref())
295        .await?;
296    let org_id = path.into_inner();
297
298    // Authorization: only allow writes to the authenticated user's own org.
299    if org_id != user.org_id {
300        return Err(AxiamApiError(
301            axiam_core::error::AxiamError::AuthorizationDenied {
302                reason: "cannot modify settings for a different organization".into(),
303                action: None,
304                resource_id: None,
305            },
306        ));
307    }
308
309    let input = body.into_inner();
310    validate_org_settings(&input)?;
311    reject_opaque_without_keys(input.opaque_mode, &state)?;
312    let opaque_mode = input.opaque_mode;
313    let opaque_suite = input.opaque_suite;
314
315    // An organization baseline reaches every tenant, so everything below is
316    // organization-wide. Read once and reuse: the coverage gate, the key-material
317    // provisioning and the override reconciliation all walk the same list.
318    let tenants = all_tenants_of(&state, org_id).await?;
319
320    // Checked BEFORE the write, so a refusal leaves the deployment exactly as it
321    // was. `required` at the organization makes every tenant `required` — the
322    // tighten-only rule means no tenant can hold itself below the baseline — so
323    // every tenant is in scope, the organization's own reserved scope included.
324    if opaque_mode == OpaqueMode::Required {
325        refuse_incomplete_opaque_coverage(&state, &tenants).await?;
326    }
327
328    let settings = state.settings_repo.set_org_settings(org_id, input).await?;
329
330    // Switching OPAQUE on at the organization switches it on for every tenant
331    // that has not tightened past it, so every one of them needs key material.
332    if opaque_mode != OpaqueMode::Disabled {
333        for tenant in &tenants {
334            provision_opaque_setup(&state, tenant.id, opaque_suite).await;
335        }
336    }
337
338    // And every tenant override the new baseline has overtaken is cleared, so
339    // the change actually reaches the tenants that had one — which is what
340    // "propagated to all the already existing ones" has to mean.
341    reconcile_tenant_overrides(&state, &settings, &tenants).await;
342
343    Ok(HttpResponse::Ok().json(settings))
344}

375/// `PUT /api/v1/settings`
376///
377/// Set tenant-level overrides. Only fields that are **more restrictive**
378/// than the org baseline are accepted. Omit a field (set to `null`) to
379/// inherit from the org.
380#[utoipa::path(
381    put,
382    path = "/api/v1/settings",
383    tag = "settings",
384    request_body = TenantSettingsOverride,
385    responses(
386        (status = 200, description = "Tenant settings updated",
387         body = SecuritySettings),
388        (status = 400, description = "Override violates org baseline, or \
389                                      enables OPAQUE on a server holding no \
390                                      OPAQUE keys"),
391    ),
392    security(("bearer" = []))
393)]
394pub async fn set_tenant_settings<C: Connection + Clone>(
395    user: AuthenticatedUser,
396    authz: AuthzData,
397    state: web::Data<AppState<C>>,
398    body: web::Json<TenantSettingsOverride>,
399) -> Result<HttpResponse, AxiamApiError> {
400    RequirePermission::new("settings:update", Uuid::nil())
401        .check(&user, authz.get_ref().as_ref())
402        .await?;
403    let org = state.settings_repo.get_org_settings(user.org_id).await?;
404    let overrides = body.into_inner();
405
406    // Validate: tenant can only be more restrictive than org
407    validate_tenant_override(&org, &overrides)?;
408
409    // The same runtime-serviceability guard as the org write, but keyed on the
410    // override's own value rather than the merged one. A tenant that leaves
411    // `opaque_mode` unset inherits whatever the org baseline says and changes
412    // nothing, so it has no business being refused here — only an override
413    // that itself raises the mode is asking for something the server cannot do.
414    if let Some(mode) = overrides.opaque_mode {
415        reject_opaque_without_keys(mode, &state)?;
416    }
417
418    // Merge org baseline + overrides into a complete settings row.
419    // The ID passed here is just a placeholder; the underlying
420    // SurrealSettingsRepository derives a deterministic UUID (v5)
421    // from (scope, scope_id) and uses that as the canonical ID.
422    let merged = effective_settings(&org, &overrides, user.tenant_id, Uuid::nil());
423
424    let result = state
425        .settings_repo
426        .store_effective_tenant_settings(user.tenant_id, merged)
427        .await?;
428
429    if result.opaque.opaque_mode != OpaqueMode::Disabled {
430        provision_opaque_setup(&state, user.tenant_id, result.opaque.opaque_suite).await;
431    }
432
433    Ok(HttpResponse::Ok().json(result))
434}

437// Tenant-scoped settings — the same overrides, addressed by tenant id

439//
440// `GET`/`PUT /api/v1/settings` act on the caller's own tenant implicitly and

442// their own policy wants. It is not what the organization's tenant detail page
443// needs: that page shows one named tenant and has to distinguish an overridden

447// segment, same `tenant_id == user.tenant_id` check on top of the permission,
448// same "raw own-scope row, never the merged view" rule — so a sibling tenant
449// answers `403` here for the same reason and in the same shape it does there.

495/// `PUT /api/v1/tenants/{tenant_id}/settings`
496#[utoipa::path(
497    put,
498    path = "/api/v1/tenants/{tenant_id}/settings",
499    tag = "settings",
500    params(
501        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
502    ),
503    request_body = TenantSettingsOverride,
504    responses(
505        (status = 200, description = "The stored overrides, sparse",
506         body = TenantSettingsOverride),
507        (status = 400, description = "An override is less restrictive than the org \
508                                      baseline, or enables OPAQUE on a server holding \
509                                      no OPAQUE keys"),
510    ),
511    security(("bearer" = []))
512)]
513pub async fn set_tenant_override<C: Connection + Clone>(
514    user: AuthenticatedUser,
515    authz: AuthzData,
516    state: web::Data<AppState<C>>,
517    path: web::Path<Uuid>,
518    body: web::Json<TenantSettingsOverride>,
519) -> Result<HttpResponse, AxiamApiError> {
520    RequirePermission::new("settings:update", Uuid::nil())
521        .check(&user, authz.get_ref().as_ref())
522        .await?;
523    let tenant_id = path.into_inner();
524
525    if tenant_id != user.tenant_id {
526        return Err(AxiamApiError(AxiamError::AuthorizationDenied {
527            reason: "cannot modify security settings for a different tenant".into(),
528            action: None,
529            resource_id: None,
530        }));
531    }
532
533    let overrides = body.into_inner();
534    let org = state.settings_repo.get_org_settings(user.org_id).await?;
535    validate_tenant_override(&org, &overrides)?;
536    if let Some(mode) = overrides.opaque_mode {
537        reject_opaque_without_keys(mode, &state)?;
538    }
539
540    // Same one-way door as at the organization, checked against this tenant
541    // alone. Read from the *effective* mode rather than the override so that a
542    // tenant inheriting `required` from a baseline it is only now writing an
543    // unrelated override against is not gated twice — the organization write
544    // already established coverage for it.
545    let would_be = effective_settings(&org, &overrides, tenant_id, Uuid::nil());
546    if would_be.opaque.opaque_mode == OpaqueMode::Required
547        && overrides.opaque_mode == Some(OpaqueMode::Required)
548    {
549        let tenant = state.tenant_repo.get_by_id(tenant_id).await?;
550        refuse_incomplete_opaque_coverage(&state, std::slice::from_ref(&tenant)).await?;
551    }
552
553    let stored = state
554        .settings_repo
555        .set_tenant_override(tenant_id, overrides)
556        .await?;
557
558    let merged = effective_settings(&org, &stored, tenant_id, Uuid::nil());
559    if merged.opaque.opaque_mode != OpaqueMode::Disabled {
560        provision_opaque_setup(&state, tenant_id, merged.opaque.opaque_suite).await;
561    }
562
563    Ok(HttpResponse::Ok().json(stored))
564}
