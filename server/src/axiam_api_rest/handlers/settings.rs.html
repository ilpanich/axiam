18/// Refuse an OPAQUE mode the running server has no keys to serve.
19///
20/// `validate_org_settings` cannot make this check: it lives in `axiam-core`,
21/// which the crate layering keeps below `axiam-api-rest`, so it can never see
22/// an `AppState`. The invariant is nonetheless the same one
23/// `POST /api/v1/admin/bootstrap` enforces — writing `opaque_mode` above
24/// `disabled` onto a server holding no OPAQUE keys configures a tenant into a
25/// state where every `/api/v1/auth/opaque/*` route fails at runtime, and the
26/// only way back out is another settings write. Refuse at the write instead,
27/// with the same two env vars named, so the operator learns what is missing
28/// rather than discovering it from a login that stopped working.
29fn reject_opaque_without_keys<C: Connection + Clone>(
30    mode: OpaqueMode,
31    state: &AppState<C>,
32) -> Result<(), AxiamApiError> {
33    if mode != OpaqueMode::Disabled && state.opaque_server.is_none() {
34        return Err(AxiamApiError(axiam_core::error::AxiamError::Validation {
35            message: "opaque_mode is set but AXIAM__AUTH__OPAQUE_SESSION_KEY and \
36                      AXIAM__AUTH__OPAQUE_SETUP_KEY are not configured: saving \
37                      this would enable OPAQUE on a server that cannot serve it, \
38                      and every /api/v1/auth/opaque/* request would fail"
39                .into(),
40        }));
41    }
42    Ok(())
43}

45/// `GET /api/v1/organizations/{org_id}/settings`
46#[utoipa::path(
47    get,
48    path = "/api/v1/organizations/{org_id}/settings",
49    tag = "settings",
50    params(
51        ("org_id" = Uuid, Path, description = "Organization ID"),
52    ),
53    responses(
54        (status = 200, description = "Organization security settings",
55         body = SecuritySettings),
56    ),
57    security(("bearer" = []))
58)]
59pub async fn get_org_settings<C: Connection + Clone>(
60    user: AuthenticatedUser,
61    authz: AuthzData,
62    state: web::Data<AppState<C>>,
63    path: web::Path<Uuid>,
64) -> Result<HttpResponse, AxiamApiError> {
65    // SECURITY (NEW-2): org-baseline settings are a higher privilege than the
66    // per-tenant `settings:*` self-management permissions. Enforce the dedicated
67    // org-level permission declared in ROUTE_PERMISSION_MAP so a tenant-scoped
68    // `settings:get` grant cannot read the org baseline that cascades to every
69    // sibling tenant.
70    RequirePermission::new("organizations:get_settings", Uuid::nil())
71        .check(&user, authz.get_ref().as_ref())
72        .await?;
73    let org_id = path.into_inner();
74
75    // Authorization: only allow reads for the authenticated user's own org.
76    if org_id != user.org_id {
77        return Err(AxiamApiError(
78            axiam_core::error::AxiamError::AuthorizationDenied {
79                reason: "cannot read settings for a different organization".into(),
80                action: None,
81                resource_id: None,
82            },
83        ));
84    }
85
86    let settings = state.settings_repo.get_org_settings(org_id).await?;
87    Ok(HttpResponse::Ok().json(settings))
88}

90/// `PUT /api/v1/organizations/{org_id}/settings`
91#[utoipa::path(
92    put,
93    path = "/api/v1/organizations/{org_id}/settings",
94    tag = "settings",
95    params(
96        ("org_id" = Uuid, Path, description = "Organization ID"),
97    ),
98    request_body = SetOrgSettings,
99    responses(
100        (status = 200, description = "Organization settings updated",
101         body = SecuritySettings),
102        (status = 400,
103         description = "Settings are internally inconsistent, or enable OPAQUE \
104                        on a server holding no OPAQUE keys"),
105    ),
106    security(("bearer" = []))
107)]
108pub async fn set_org_settings<C: Connection + Clone>(
109    user: AuthenticatedUser,
110    authz: AuthzData,
111    state: web::Data<AppState<C>>,
112    path: web::Path<Uuid>,
113    body: web::Json<SetOrgSettings>,
114) -> Result<HttpResponse, AxiamApiError> {
115    // SECURITY (NEW-2): enforce the dedicated org-level permission (declared in
116    // ROUTE_PERMISSION_MAP) rather than the tenant-scoped `settings:update`, so a
117    // tenant admin cannot rewrite the org security baseline inherited by every
118    // sibling tenant.
119    RequirePermission::new("organizations:update_settings", Uuid::nil())
120        .check(&user, authz.get_ref().as_ref())
121        .await?;
122    let org_id = path.into_inner();
123
124    // Authorization: only allow writes to the authenticated user's own org.
125    if org_id != user.org_id {
126        return Err(AxiamApiError(
127            axiam_core::error::AxiamError::AuthorizationDenied {
128                reason: "cannot modify settings for a different organization".into(),
129                action: None,
130                resource_id: None,
131            },
132        ));
133    }
134
135    let input = body.into_inner();
136    validate_org_settings(&input)?;
137    reject_opaque_without_keys(input.opaque_mode, &state)?;
138    let settings = state.settings_repo.set_org_settings(org_id, input).await?;
139    Ok(HttpResponse::Ok().json(settings))
140}

156pub async fn get_tenant_settings<C: Connection + Clone>(
157    user: AuthenticatedUser,
158    authz: AuthzData,
159    state: web::Data<AppState<C>>,
160) -> Result<HttpResponse, AxiamApiError> {
161    RequirePermission::new("settings:get", Uuid::nil())
162        .check(&user, authz.get_ref().as_ref())
163        .await?;
164    let settings = state
165        .settings_repo
166        .get_effective_settings(user.org_id, user.tenant_id)
167        .await?;
168    Ok(HttpResponse::Ok().json(settings))
169}

171/// `PUT /api/v1/settings`
172///
173/// Set tenant-level overrides. Only fields that are **more restrictive**
174/// than the org baseline are accepted. Omit a field (set to `null`) to
175/// inherit from the org.
176#[utoipa::path(
177    put,
178    path = "/api/v1/settings",
179    tag = "settings",
180    request_body = TenantSettingsOverride,
181    responses(
182        (status = 200, description = "Tenant settings updated",
183         body = SecuritySettings),
184        (status = 400, description = "Override violates org baseline, or \
185                                      enables OPAQUE on a server holding no \
186                                      OPAQUE keys"),
187    ),
188    security(("bearer" = []))
189)]
190pub async fn set_tenant_settings<C: Connection + Clone>(
191    user: AuthenticatedUser,
192    authz: AuthzData,
193    state: web::Data<AppState<C>>,
194    body: web::Json<TenantSettingsOverride>,
195) -> Result<HttpResponse, AxiamApiError> {
196    RequirePermission::new("settings:update", Uuid::nil())
197        .check(&user, authz.get_ref().as_ref())
198        .await?;
199    let org = state.settings_repo.get_org_settings(user.org_id).await?;
200    let overrides = body.into_inner();
201
202    // Validate: tenant can only be more restrictive than org
203    validate_tenant_override(&org, &overrides)?;
204
205    // The same runtime-serviceability guard as the org write, but keyed on the
206    // override's own value rather than the merged one. A tenant that leaves
207    // `opaque_mode` unset inherits whatever the org baseline says and changes
208    // nothing, so it has no business being refused here — only an override
209    // that itself raises the mode is asking for something the server cannot do.
210    if let Some(mode) = overrides.opaque_mode {
211        reject_opaque_without_keys(mode, &state)?;
212    }
213
214    // Merge org baseline + overrides into a complete settings row.
215    // The ID passed here is just a placeholder; the underlying
216    // SurrealSettingsRepository derives a deterministic UUID (v5)
217    // from (scope, scope_id) and uses that as the canonical ID.
218    let merged = effective_settings(&org, &overrides, user.tenant_id, Uuid::nil());
219
220    let result = state
221        .settings_repo
222        .store_effective_tenant_settings(user.tenant_id, merged)
223        .await?;
224    Ok(HttpResponse::Ok().json(result))
225}
