1//! Recording an SRP verifier at the moments a plaintext password legitimately
2//! exists.
3//!
4//! There are exactly three such moments — user creation, an authenticated
5//! password change, and password-reset completion — and this module is the one
6//! place that handles all three, so the validation and the policy checks cannot
7//! drift between them.
8//!
9//! # Why enrolment has to be client-driven
10//!
11//! The server cannot compute a verifier. `v = g^x mod N` where
12//! `x = KDF(identity ":" password, salt)`, and the whole point of SRP is that
13//! the server never sees the password. So the client computes `v` and posts it;
14//! the server's job is to check that what arrived is well-formed, meets the
15//! tenant's policy, and is attributed to the right account — never to trust the
16//! client about *whose* account it is. That is why [`SrpEnrollment`] carries no
17//! user id: the tenant, user and canonical identity are all taken from the
18//! authenticated server-side context.
19//!
20//! # Why a missing verifier is sometimes an error
21//!
22//! Under [`SrpMode::Required`] a password set without a verifier produces an
23//! account that cannot authenticate at all: password login is refused
24//! tenant-wide, and there is nothing for `/auth/srp/challenge` to answer with.
25//! Accepting it silently would mean an admin creating a user, seeing `201`, and
26//! discovering only later that the account was born unusable. So it is rejected
27//! at the point of creation, where the message can say what to do about it.
28

31use axiam_core::models::srp::{CreateSrpCredential, SrpEnrollment, SrpMode};
32use axiam_core::repository::SrpCredentialRepository;
33use surrealdb::Connection;
34use uuid::Uuid;

38/// Validate and store a client-computed verifier for `user_id`.
39///
40/// `identity` must be the user's canonical `username` — the same string the
41/// challenge endpoint will hand back to clients, and the same one the client
42/// must have used inside the KDF. Passing anything else here produces a
43/// verifier that can never be satisfied.
44///
45/// A `None` enrolment is accepted under `disabled` and `optional`, and refused
46/// under `required`.
47pub(crate) async fn record_verifier<C: Connection + Clone>(
48    state: &AppState<C>,
49    policy: &SrpPolicy,
50    tenant_id: Uuid,
51    user_id: Uuid,
52    identity: &str,
53    enrollment: Option<&SrpEnrollment>,
54) -> AxiamResult<()> {
55    match (policy.srp_mode, enrollment) {
56        // Nothing offered, nothing expected.
57        (SrpMode::Disabled, None) => Ok(()),
58
59        // A verifier for a tenant that does not do SRP is refused rather than
60        // dropped on the floor: storing it would be dead weight, and silently
61        // discarding it would let a client believe SRP is active when the very
62        // next login will be a password login.
63        (SrpMode::Disabled, Some(_)) => Err(AxiamError::Validation {
64            message: "SRP is not enabled for this tenant".into(),
65        }),
66
67        // The migration state: clients that can do SRP enrol, clients that
68        // cannot keep working.
69        (SrpMode::Optional, None) => Ok(()),
70
71        (SrpMode::Required, None) => Err(AxiamError::Validation {
72            message: "this tenant requires Secure Remote Password: the request must \
73                      include an `srp` verifier, or the account will be unable to \
74                      authenticate"
75                .into(),
76        }),
77
78        (SrpMode::Optional | SrpMode::Required, Some(enrollment)) => {
79            let (group, params, salt, verifier) = enrollment.parse()?;
80
81            // Policy is a floor, not an exact match. A client that enrols with
82            // a *stronger* group or KDF than the tenant asks for is doing
83            // nothing wrong, and refusing it would block a client that had
84            // already moved ahead of a tenant's configuration.
85            if group.bits() < policy.srp_group.bits() {
86                return Err(AxiamError::Validation {
87                    message: format!(
88                        "srp group {} is weaker than the tenant minimum {}",
89                        group, policy.srp_group
90                    ),
91                });
92            }
93            if !srp_kdf_is_at_least(params.kdf(), policy.srp_kdf) {
94                return Err(AxiamError::Validation {
95                    message: format!(
96                        "srp kdf {} is weaker than the tenant minimum {}",
97                        params.kdf(),
98                        policy.srp_kdf
99                    ),
100                });
101            }
102
103            state
104                .srp_credential_repo
105                .upsert(CreateSrpCredential {
106                    tenant_id,
107                    user_id,
108                    identity: identity.to_string(),
109                    group,
110                    kdf_params: params,
111                    salt,
112                    verifier,
113                })
114                .await?;
115            Ok(())
116        }
117    }
118}

120/// Drop a user's verifier because the identity it was bound to no longer
121/// exists.
122///
123/// `x` is derived over `identity ":" password`, so renaming a user
124/// invalidates their verifier as surely as changing their password would — but
125/// silently, and only at their next login attempt, which is the worst possible
126/// time to discover it. Deleting it here converts a baffling "invalid
127/// credentials" into the honest state: this user has no verifier and must
128/// re-enrol via a password change.
129///
130/// Deliberately best-effort at the call site: a rename must not fail because
131/// of this.
132pub(crate) async fn invalidate_on_rename<C: Connection + Clone>(
133    state: &AppState<C>,
134    tenant_id: Uuid,
135    user_id: Uuid,
136) {
137    if let Err(e) = state
138        .srp_credential_repo
139        .delete_for_user(tenant_id, user_id)
140        .await
141    {
142        tracing::warn!(
143            target: "axiam::auth",
144            tenant_id = %tenant_id,
145            user_id = %user_id,
146            error = %e,
147            "failed to drop an SRP verifier after a username change; the user \
148             will see 'invalid credentials' on the SRP path until it is replaced"
149        );
150    }
151}
