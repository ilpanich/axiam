1//! The Shared Signals Framework transmitter's protocol surface (G-5, T23.5.2,
2//! SSF 1.0 §7 and §8, contract §32.6): transmitter metadata and the stream
3//! management API a receiver calls.
4//!
5//! # Discovery (D-45)
6//!
7//! `GET /.well-known/ssf-configuration?tenant_id={t}` everywhere, and
8//! `GET /.well-known/ssf-configuration/t/{t}` — SSF §7.2's insertion of the
9//! well-known segment into the issuer `{root}/t/{t}` — where the deployment
10//! serves tenant issuers (T21.6). Unauthenticated, as the specification
11//! requires. **An empty `404`** for an unknown tenant, a tenant whose effective
12//! `ssf_enabled` is off, a missing or malformed tenant id — one answer for every
13//! way of having nothing to say, so the route tells nobody which tenants exist
14//! or transmit (the D-20 shape).
15//!
16//! # The stream management API (D-50)
17//!
18//! `/ssf/v1/stream`, `/ssf/v1/status`, `/ssf/v1/verify`, authenticated by a
19//! **receiver token**: an access token issued to an OAuth2 client by the
20//! client-credentials grant with the `ssf.manage` scope ([`SsfReceiverToken`]).
21//! A user token, a service-account token or a client token without the scope is
22//! `403`. The tenant is the token's; a stream is the receiver's only when its
23//! `receiver_client_id` is the token's `client_id`, and any other stream —
24//! another receiver's, another tenant's, or one in a tenant whose transmitter
25//! is switched off — is the same `404` as one that does not exist (SSF §8.1.1:
26//! "no Event Stream with the given stream_id for this Event Receiver").
27//!
28//! Streams are registered and deleted by an administrator (`handlers::
29//! ssf_admin`); `POST` and `DELETE` on the configuration endpoint are `403`.
30//! What a receiver may change is `axiam_oauth2::ssf::apply_receiver_update`'s
31//! rule: its requested events (never beyond the administrator's ceiling), its
32//! description, its push endpoint under the outbound address policy, and the
33//! push `Authorization` header. The status it may set unless an administrator
34//! set something other than `enabled`. Verification is limited per stream by
35//! `min_verification_interval` (`429`) besides the route's bucket.
36

98/// Whether `tenant_id` exists, its effective `ssf_enabled` is on **and** the
99/// D-55 gate does not hold. An unknown tenant is `false`, never an error a
100/// caller could tell apart. Discovery, the receiver API, verification and the
101/// stream-updated announcement all ask this, so the gate makes SSF behave
102/// exactly as `ssf_enabled` off for every tenant.
103pub(crate) async fn ssf_enabled_for<C: Connection + Clone>(
104    state: &AppState<C>,
105    tenant_id: Uuid,
106) -> Result<bool, AxiamError> {
107    Ok(transmitter_status(state, tenant_id).await?.is_none())
108}

110/// Why the tenant's transmitter is inactive, or `None` when it is active: the
111/// tenant does not exist, its `ssf_enabled` is off, or the D-55 gate holds.
112pub(crate) async fn transmitter_status<C: Connection + Clone>(
113    state: &AppState<C>,
114    tenant_id: Uuid,
115) -> Result<Option<&'static str>, AxiamError> {
116    let tenant = match state.tenant_repo.get_by_id(tenant_id).await {
117        Ok(tenant) => tenant,
118        Err(AxiamError::NotFound { .. }) => return Ok(Some(NO_SUCH_TENANT)),
119        Err(other) => return Err(other),
120    };
121    let settings = state
122        .settings_repo
123        .get_effective_settings(tenant.organization_id, tenant_id)
124        .await?;
125    if !settings.oidc.ssf_enabled {
126        return Ok(Some(SWITCH_OFF));
127    }
128    if state.ssf.gate.check().await?.holds() {
129        return Ok(Some(SHARED_ISSUER_INACTIVE_REASON));
130    }
131    Ok(None)
132}
133
134/// [`transmitter_status`]'s answer for a tenant that does not exist.
135const NO_SUCH_TENANT: &str = "the tenant does not exist";
136/// [`transmitter_status`]'s answer for a tenant whose switch is off.
137pub(crate) const SWITCH_OFF: &str = "ssf_enabled is off for this tenant";

150async fn discovery_for<C: Connection + Clone>(
151    state: &AppState<C>,
152    tenant: Option<&str>,
153) -> HttpResponse {
154    let Some(tenant_id) = tenant.and_then(|raw| Uuid::parse_str(raw).ok()) else {
155        return HttpResponse::NotFound().finish();
156    };
157    match ssf_enabled_for(state, tenant_id).await {
158        Ok(true) => HttpResponse::Ok()
159            .append_header(("Cache-Control", "public, max-age=300"))
160            .json(build_ssf_configuration(&state.auth_config, tenant_id)),
161        Ok(false) => HttpResponse::NotFound().finish(),
162        Err(error) => {
163            tracing::error!(target: "axiam::ssf", %error, "SSF discovery could not read the tenant");
164            HttpResponse::ServiceUnavailable().finish()
165        }
166    }
167}

217/// A validated receiver token: an access token issued to an OAuth2 client by
218/// the client-credentials grant, carrying `ssf.manage`. Its own extractor for
219/// the reason `ProtectionApiToken` is: the subject is a `client_id`, not a
220/// principal in the RBAC graph, and the scope **is** the gate.
221pub struct SsfReceiverToken {
222    /// The receiver's `client_id` — the token's `sub`.
223    pub client_id: String,
224    /// The token's tenant.
225    pub tenant_id: Uuid,
226}

228fn forbidden(reason: &str) -> AxiamApiError {
229    AxiamApiError(AxiamError::AuthorizationDenied {
230        reason: reason.to_owned(),
231        action: None,
232        resource_id: None,
233    })
234}

236fn extract_receiver(req: &HttpRequest) -> Result<SsfReceiverToken, AxiamApiError> {
237    let claims = crate::extractors::auth::parse_validated_claims(req)?.0;
238    if claims.sub_kind != SubjectKind::OAuth2Client {
239        return Err(forbidden(
240            "the SSF stream management API takes an OAuth2 client's client-credentials token",
241        ));
242    }
243    if !claims
244        .scope
245        .as_deref()
246        .is_some_and(|s| s.split(' ').any(|granted| granted == SSF_MANAGE_SCOPE))
247    {
248        return Err(forbidden(&format!(
249            "the SSF stream management API requires the '{SSF_MANAGE_SCOPE}' scope"
250        )));
251    }
252    let tenant_id = Uuid::parse_str(&claims.tenant_id).map_err(|_| {
253        AxiamApiError(AxiamError::AuthenticationFailed {
254            reason: "invalid tenant_id claim".into(),
255        })
256    })?;
257    Ok(SsfReceiverToken {
258        client_id: claims.sub,
259        tenant_id,
260    })
261}

263impl actix_web::FromRequest for SsfReceiverToken {

267    fn from_request(

269        _payload: &mut actix_web::dev::Payload,

282/// The receiver's stream `raw_id`, or the one `404` for every other case.
283async fn owned_stream<C: Connection + Clone>(
284    state: &AppState<C>,
285    receiver: &SsfReceiverToken,
286    raw_id: Option<&str>,
287) -> Result<SsfStream, AxiamApiError> {
288    let Some(stream_id) = raw_id.and_then(|raw| Uuid::parse_str(raw).ok()) else {
289        return Err(no_such_stream());
290    };
291    if !ssf_enabled_for(state, receiver.tenant_id).await? {
292        return Err(no_such_stream());
293    }
294    let stream = match state
295        .ssf
296        .stream_repo
297        .get(receiver.tenant_id, stream_id)
298        .await
299    {
300        Ok(stream) => stream,
301        Err(AxiamError::NotFound { .. }) => return Err(no_such_stream()),
302        Err(other) => return Err(AxiamApiError(other)),
303    };
304    if stream.receiver_client_id != receiver.client_id {
305        return Err(no_such_stream());
306    }
307    Ok(stream)
308}

310async fn audit_receiver<C: Connection + Clone>(
311    state: &AppState<C>,
312    http_req: &HttpRequest,
313    receiver: &SsfReceiverToken,
314    action: &str,
315    stream_id: Uuid,
316    metadata: serde_json::Value,
317) {
318    let mut metadata = metadata;
319    if let Some(object) = metadata.as_object_mut() {
320        object.insert(
321            "receiver_client_id".into(),
322            serde_json::json!(receiver.client_id),
323        );
324    }
325    if let Err(error) = state
326        .audit_repo
327        .append(CreateAuditLogEntry {
328            tenant_id: receiver.tenant_id,
329            actor_id: Uuid::nil(),
330            actor_type: ActorType::System,
331            action: action.to_string(),
332            resource_id: Some(stream_id),
333            outcome: AuditOutcome::Success,
334            ip_address: client_ip(http_req),
335            metadata: Some(metadata),
336        })
337        .await
338    {
339        tracing::error!(target: "axiam::ssf", action, %error, "an SSF audit row could not be written");
340    }
341}

343/// A paused push stream was enabled again: enqueue what it held, oldest first
344/// (D-48). Best effort and bounded — the held events stay buffered if the
345/// broker is down, and the next resume (or a status write) takes them.
346pub(crate) async fn release_held<C: Connection + Clone>(state: &AppState<C>, stream: &SsfStream) {
347    let Some(outbox) = state.ssf.outbox.as_ref() else {
348        return;
349    };
350    match outbox.resume(stream).await {
351        Ok(0) => {}
352        Ok(released) => {
353            tracing::info!(target: "axiam::ssf", stream_id = %stream.id, released, "held SSF events released");
354        }
355        Err(error) => {
356            tracing::warn!(target: "axiam::ssf", stream_id = %stream.id, %error, "held SSF events could not be released");
357        }
358    }
359}

416    Ok(no_store(HttpResponse::Ok()).json(views))

439/// How many times a receiver's write is prepared again from a fresh read when
440/// another write overtook the read it was prepared from (F4 W4 P23W4-01). Past
441/// that the answer is `409`.
442const RECEIVER_WRITE_ATTEMPTS: usize = 3;
443
444/// Whether a failed write lost a race to another write of the same stream and
445/// should be prepared again from a fresh read.
446fn overtaken(error: &AxiamError, attempt: usize) -> bool {
447    matches!(error, AxiamError::Conflict { .. }) && attempt + 1 < RECEIVER_WRITE_ATTEMPTS
448}

450async fn update_with<C: Connection + Clone>(
451    receiver: SsfReceiverToken,
452    state: web::Data<AppState<C>>,
453    http_req: HttpRequest,
454    body: ReceiverStreamUpdate,
455    mode: ReceiverUpdateMode,
456) -> Result<HttpResponse, AxiamApiError> {
457    let Some(raw_id) = body.stream_id.clone() else {
458        return Err(AxiamApiError(AxiamError::Validation {
459            message: "stream_id is required".into(),
460        }));
461    };
462    // Read, decide, write — and if another write (an administrator's) landed
463    // between the read and the write, decide again against what it left: the
464    // update carries the version it was prepared from, so it never puts back
465    // what that write changed (D-51, T-406).
466    let mut attempt = 0;
467    let (stream, updated, changed) = loop {
468        let stream = owned_stream(&state, &receiver, Some(&raw_id)).await?;
469        let update = apply_receiver_update(&state.auth_config, &stream, &body, mode)
470            .map_err(|message| AxiamApiError(AxiamError::Validation { message }))?;
471        if matches!(update.authorization_header, SecretChange::Set(_))
472            && !state.ssf.stream_repo.has_encryption_key()
473        {
474            return Err(AxiamApiError(AxiamError::ServiceUnavailable(
475                "the push authorization header cannot be stored on this deployment".into(),
476            )));
477        }
478        let mut changed: Vec<&str> = Vec::new();
479        if update.events_requested != stream.events_requested {
480            changed.push("events_requested");
481        }
482        if update.description != stream.description {
483            changed.push("description");
484        }
485        if update.endpoint_url != stream.endpoint_url {
486            changed.push("endpoint_url");
487        }
488        if !matches!(update.authorization_header, SecretChange::Keep) {
489            changed.push("authorization_header");
490        }
491        match state
492            .ssf
493            .stream_repo
494            .update(receiver.tenant_id, stream.id, update)
495            .await
496        {
497            Ok(updated) => break (stream, updated, changed),
498            Err(error) if overtaken(&error, attempt) => attempt += 1,
499            Err(error) => return Err(AxiamApiError(error)),
500        }
501    };
502    audit_receiver(
503        &state,
504        &http_req,
505        &receiver,
506        AUDIT_RECEIVER_UPDATED,
507        stream.id,
508        serde_json::json!({ "changed": changed }),
509    )
510    .await;
511    Ok(no_store(HttpResponse::Ok()).json(stream_configuration(&state.auth_config, &updated)))
512}

673/// `POST /ssf/v1/status`
674#[utoipa::path(
675    post,
676    path = "/ssf/v1/status",
677    tag = "ssf-receiver",
678    request_body = SsfStatusUpdate,
679    responses(
680        (status = 200, description = "The stream's new status", body = SsfStreamStatusView),
681        (status = 400, description = "An unknown status or an over-long reason"),
682        (status = 401, description = "No valid token"),
683        (status = 403, description = "Not a receiver token, or an administrator set this status"),
684        (status = 404, description = "No such stream for this receiver"),
685        (status = 409, description = "The stream kept changing under this write; read it again and retry"),
686        (status = 429, description = "Rate limit"),
687    ),
688    security(("bearer" = []))
689)]
690pub async fn update_stream_status<C: Connection + Clone>(
691    receiver: SsfReceiverToken,
692    state: web::Data<AppState<C>>,
693    http_req: HttpRequest,
694    body: web::Json<SsfStatusUpdate>,
695) -> Result<HttpResponse, AxiamApiError> {
696    let body = body.into_inner();
697    let status = SsfStreamStatus::from_wire(&body.status).ok_or_else(|| {
698        AxiamApiError(AxiamError::Validation {
699            message: "status must be enabled, paused or disabled".into(),
700        })
701    })?;
702    if let Some(reason) = &body.reason {
703        validate_status_reason(reason)
704            .map_err(|message| AxiamApiError(AxiamError::Validation { message }))?;
705    }
706    let reason = body.reason.filter(|r| !r.is_empty());
707    // Read, decide, write; decided again from a fresh read when an
708    // administrator's write landed in between (F4 W4 P23W4-01, T-406), so the
709    // D-51 check below always judges the status the write would replace.
710    let mut attempt = 0;
711    let (stream, updated) = loop {
712        let stream = owned_stream(&state, &receiver, Some(&body.stream_id)).await?;
713        // D-51: what an administrator stopped, only an administrator restarts.
714        if stream.status_actor == SsfStatusActor::Admin && stream.status != SsfStreamStatus::Enabled
715        {
716            return Err(forbidden(
717                "an administrator set this stream's status; only an administrator can change it",
718            ));
719        }
720        let mut update = SsfStreamUpdate::from_stream(&stream);
721        update.status = status;
722        update.status_reason = reason.clone();
723        update.status_actor = SsfStatusActor::Receiver;
724        match state
725            .ssf
726            .stream_repo
727            .update(receiver.tenant_id, stream.id, update)
728            .await
729        {
730            Ok(updated) => break (stream, updated),
731            Err(error) if overtaken(&error, attempt) => attempt += 1,
732            Err(error) => return Err(AxiamApiError(error)),
733        }
734    };
735    let previous = stream.status;
736    audit_receiver(
737        &state,
738        &http_req,
739        &receiver,
740        AUDIT_RECEIVER_STATUS,
741        stream.id,
742        serde_json::json!({ "from": previous.as_str(), "to": updated.status.as_str() }),
743    )
744    .await;
745    if previous == SsfStreamStatus::Paused && updated.status == SsfStreamStatus::Enabled {
746        release_held(&state, &updated).await;
747    }
748    Ok(no_store(HttpResponse::Ok()).json(SsfStreamStatusView::from(&updated)))
749}

757pub struct SsfVerificationRequest {
758    /// The stream.
759    pub stream_id: String,
760    /// Echoed in the verification event; at most 1 024 bytes.
761    #[serde(default)]
762    pub state: Option<String>,
763}
764
765/// `POST /ssf/v1/verify` — have a verification event transmitted.
766#[utoipa::path(
767    post,
768    path = "/ssf/v1/verify",
769    tag = "ssf-receiver",
770    request_body = SsfVerificationRequest,
771    responses(
772        (status = 204, description = "The verification event was submitted for delivery"),
773        (status = 400, description = "An over-long state, or a disabled stream"),
774        (status = 401, description = "No valid token"),
775        (status = 403, description = "Not an OAuth2 client token with the ssf.manage scope"),
776        (status = 404, description = "No such stream for this receiver"),
777        (status = 429, description = "Sooner than min_verification_interval, or the route's limit"),
778        (status = 503, description = "Event delivery is not available on this deployment"),
779    ),
780    security(("bearer" = []))
781)]
782pub async fn request_verification<C: Connection + Clone>(
783    receiver: SsfReceiverToken,
784    state: web::Data<AppState<C>>,
785    http_req: HttpRequest,
786    body: web::Json<SsfVerificationRequest>,
787) -> Result<HttpResponse, AxiamApiError> {
788    let body = body.into_inner();
789    let stream = owned_stream(&state, &receiver, Some(&body.stream_id)).await?;
790    if body
791        .state
792        .as_ref()
793        .is_some_and(|s| s.len() > MAX_STATE_BYTES)
794    {
795        return Err(AxiamApiError(AxiamError::Validation {
796            message: format!("state must be at most {MAX_STATE_BYTES} bytes"),
797        }));
798    }
799    if stream.status == SsfStreamStatus::Disabled {
800        return Err(AxiamApiError(AxiamError::Validation {
801            message: "the stream is disabled: nothing is transmitted on it".into(),
802        }));
803    }
804    let Some(outbox) = state.ssf.outbox.clone() else {
805        return Err(AxiamApiError(AxiamError::ServiceUnavailable(
806            "SSF event delivery is not available on this deployment".into(),
807        )));
808    };
809    let now = Utc::now();
810    if !state
811        .ssf
812        .stream_repo
813        .claim_verification(
814            receiver.tenant_id,
815            stream.id,
816            now,
817            MIN_VERIFICATION_INTERVAL_SECS,
818        )
819        .await?
820    {
821        return Err(AxiamApiError(AxiamError::RateLimited));
822    }
823    let event = prepare_verification(&stream, body.state.as_deref(), now).map_err(|_| {
824        AxiamApiError(AxiamError::Validation {
825            message: "the stream is disabled: nothing is transmitted on it".into(),
826        })
827    })?;
828    outbox.submit(&stream, &event).await.map_err(|error| {
829        tracing::warn!(target: "axiam::ssf", stream_id = %stream.id, %error, "verification event not submitted");
830        AxiamApiError(AxiamError::ServiceUnavailable(
831            "the verification event could not be submitted".into(),
832        ))
833    })?;
834    audit_receiver(
835        &state,
836        &http_req,
837        &receiver,
838        AUDIT_VERIFICATION,
839        stream.id,
840        serde_json::json!({ "jti": event.jti }),
841    )
842    .await;
843    Ok(no_store(HttpResponse::NoContent()).finish())
844}

860/// `POST /ssf/v1/poll/{stream_id}` body (RFC 8936 §2.2). Every member is

863pub struct SsfPollRequest {

865    #[serde(rename = "maxEvents", default)]

876    #[serde(rename = "setErrs", default)]

880/// `POST /ssf/v1/poll/{stream_id}` answer (RFC 8936 §2.3).
881#[derive(Debug, Serialize, utoipa::ToSchema)]
882pub struct SsfPollResponse {
883    /// The compact SETs by `jti`: the oldest held events, up to `maxEvents`,
884    /// signed now, against the stream as it is now.
885    #[schema(value_type = std::collections::HashMap<String, String>)]
886    pub sets: serde_json::Map<String, serde_json::Value>,
887    /// Whether more are held than were returned.
888    #[serde(rename = "moreAvailable")]
889    pub more_available: bool,
890}

892/// The audit-safe reading of a receiver's `err`: one of RFC 8935 §2.4's codes,
893/// else a fixed word. Free text from a third party is not written to the audit
894/// log.
895fn audited_err_code(raw: &str) -> &'static str {
896    axiam_oauth2::ssf_delivery::RFC_8935_ERROR_CODES
897        .into_iter()
898        .find(|known| *known == raw)
899        .unwrap_or("unrecognized")
900}

902fn empty_poll_response() -> HttpResponse {
903    no_store(HttpResponse::Ok()).json(SsfPollResponse {
904        sets: serde_json::Map::new(),
905        more_available: false,
906    })
907}

909/// `POST /ssf/v1/poll/{stream_id}` — RFC 8936 poll delivery (D-48).
910///
911/// The receiver's token, as on the stream API, and the same single `404` for a
912/// stream that is not its own. In this order: the acknowledgements, then the
913/// reported errors, then the next SETs — so an event acknowledged here is not
914/// returned by this same call. An unacknowledged event comes back on the next
915/// poll (at-least-once); a SET is **signed now**, against the stream as it is
916/// now, so a stream paused or disabled meanwhile answers an empty `sets`.
917///
918/// Without `returnImmediately` the call long-polls for up to 30 seconds; at most
919/// one long poll waits per stream per server instance, and a second concurrent
920/// request on the same stream is answered at once as if `returnImmediately` were
921/// true (D-53).
922#[utoipa::path(
923    post,
924    path = "/ssf/v1/poll/{stream_id}",
925    tag = "ssf-receiver",
926    params(("stream_id" = String, Path, description = "The poll stream")),
927    request_body = Option<SsfPollRequest>,
928    responses(
929        (status = 200, description = "The oldest held SETs, up to maxEvents, and whether more are held", body = SsfPollResponse),
930        (status = 400, description = "A malformed body, a negative maxEvents, too many ack or \
931                                      setErrs entries, or a push stream"),
932        (status = 401, description = "No valid token"),
933        (status = 403, description = "Not an OAuth2 client token with the ssf.manage scope"),
934        (status = 404, description = "No such stream for this receiver"),
935        (status = 413, description = "The body is over 32 KiB"),
936        (status = 429, description = "Rate limit"),
937    ),
938    security(("bearer" = []))
939)]
940pub async fn poll_events<C: Connection + Clone>(
941    receiver: SsfReceiverToken,
942    state: web::Data<AppState<C>>,
943    http_req: HttpRequest,
944    path: web::Path<String>,
945    body: web::Bytes,
946) -> Result<HttpResponse, AxiamApiError> {
947    let raw_id = path.into_inner();
948    let mut stream = owned_stream(&state, &receiver, Some(&raw_id)).await?;
949    if stream.delivery_method != SsfDeliveryMethod::Poll {
950        return Err(AxiamApiError(AxiamError::Validation {
951            message: "this stream delivers by push; there is nothing to poll".into(),
952        }));
953    }
954    if body.len() > POLL_MAX_BODY_BYTES {
955        return Ok(HttpResponse::PayloadTooLarge().finish());
956    }
957    let request: SsfPollRequest = if body.iter().all(u8::is_ascii_whitespace) {
958        SsfPollRequest::default()
959    } else {
960        serde_json::from_slice(&body).map_err(|_| {
961            AxiamApiError(AxiamError::Validation {
962                message: "the poll request is not a valid RFC 8936 JSON body".into(),
963            })
964        })?
965    };
966    let max_events = match request.max_events {
967        None => POLL_MAX_EVENTS_PER_RESPONSE,
968        Some(n) if n < 0 => {
969            return Err(AxiamApiError(AxiamError::Validation {
970                message: "maxEvents must not be negative".into(),
971            }));
972        }
973        Some(n) => usize::try_from(n)
974            .unwrap_or(POLL_MAX_EVENTS_PER_RESPONSE)
975            .min(POLL_MAX_EVENTS_PER_RESPONSE),
976    };
977    if request.ack.len() > POLL_MAX_ACKS || request.set_errs.len() > POLL_MAX_SET_ERRS {
978        return Err(AxiamApiError(AxiamError::Validation {
979            message: format!(
980                "a poll request carries at most {POLL_MAX_ACKS} ack and {POLL_MAX_SET_ERRS} \
981                 setErrs entries"
982            ),
983        }));
984    }
985
986    // 1. Acknowledgements: exactly the named rows of this stream.
987    let acked: Vec<String> = request
988        .ack
989        .into_iter()
990        .filter(|jti| jti.len() <= MAX_JTI_BYTES)
991        .collect();
992    state
993        .ssf
994        .buffer_repo
995        .delete_by_jti(receiver.tenant_id, stream.id, &acked)
996        .await?;
997
998    // 2. Errors the receiver reports: each row is deleted (the receiver will not
999    //    accept that SET, so offering it again is pointless) and audited.
1000    for (jti, reported) in request.set_errs {
1001        if jti.len() > MAX_JTI_BYTES {
1002            continue;
1003        }
1004        let removed = state
1005            .ssf
1006            .buffer_repo
1007            .delete_by_jti(receiver.tenant_id, stream.id, std::slice::from_ref(&jti))
1008            .await?;
1009        audit_receiver(
1010            &state,
1011            &http_req,
1012            &receiver,
1013            AUDIT_POLL_SET_ERROR,
1014            stream.id,
1015            serde_json::json!({
1016                "jti": jti,
1017                "err": audited_err_code(&reported.err),
1018                "held": removed > 0,
1019            }),
1020        )
1021        .await;
1022    }
1023
1024    // 3. The next SETs. Nothing from a stream that is not enabled; max 0 asks
1025    //    for acknowledgements only.
1026    if stream.status != SsfStreamStatus::Enabled || max_events == 0 {
1027        return Ok(empty_poll_response());
1028    }
1029    let return_immediately = request.return_immediately.unwrap_or(false);
1030    let started = std::time::Instant::now();
1031    // D-53 (11): one waiting long poll per stream per instance. The slot is
1032    // taken the first time this request would wait, and released when it
1033    // returns or is dropped; a request that finds it taken answers at once.
1034    let mut wait_slot: Option<crate::state::bundles::PollWaitGuard> = None;
1035    // A key that cannot sign stays unusable for the rest of a long poll, which
1036    // looks at the buffer every half second: it is reported once per request,
1037    // not once per look (F4 W4 P23W4-03).
1038    let mut signing_failure_logged = false;
1039    loop {
1040        // D-55 at signing, on every look: a gate that started to hold while
1041        // this poll waited signs nothing.
1042        let issuer = state.ssf.gate.check().await?;
1043        let held = state
1044            .ssf
1045            .buffer_repo
1046            .list_oldest(receiver.tenant_id, stream.id, max_events + 1, Utc::now())
1047            .await?;
1048        if !held.is_empty() {
1049            let more_available = held.len() > max_events;
1050            let mut sets = serde_json::Map::new();
1051            let mut unsignable: Vec<String> = Vec::new();
1052            for pending in held.iter().take(max_events) {
1053                match sign_set(&state.auth_config, issuer, &stream, pending) {
1054                    Ok(set) => {
1055                        sets.insert(pending.jti.clone(), serde_json::Value::String(set));
1056                    }
1057                    // A key that cannot sign is the operator's to fix; the event
1058                    // stays for the next poll.
1059                    Err(SsfError::Signing(_)) => {
1060                        if !signing_failure_logged {
1061                            signing_failure_logged = true;
1062                            tracing::error!(target: "axiam::ssf", stream_id = %stream.id, "a held SSF event could not be signed");
1063                        }
1064                    }
1065                    // D-55 started to hold while this poll waited: nothing is
1066                    // signed, and the poll answers as a switched-off stream.
1067                    Err(SsfError::SharedIssuer) => return Err(no_such_stream()),
1068                    // The stream no longer carries it (narrowed meanwhile) or it
1069                    // can never be signed for this stream: drop it.
1070                    Err(_) => unsignable.push(pending.jti.clone()),
1071                }
1072            }
1073            if !unsignable.is_empty() {
1074                state
1075                    .ssf
1076                    .buffer_repo
1077                    .delete_by_jti(receiver.tenant_id, stream.id, &unsignable)
1078                    .await?;
1079            }
1080            if !sets.is_empty() || return_immediately {
1081                return Ok(no_store(HttpResponse::Ok()).json(SsfPollResponse {
1082                    sets,
1083                    more_available,
1084                }));
1085            }
1086        }
1087        if return_immediately || started.elapsed() >= POLL_LONG_POLL_MAX {
1088            return Ok(empty_poll_response());
1089        }
1090        if wait_slot.is_none() {
1091            wait_slot = state.ssf.poll_waiters.try_enter(stream.id);
1092            if wait_slot.is_none() {
1093                return Ok(empty_poll_response());
1094            }
1095        }
1096        // `saturating_sub`: the 30 s may have run out since the check above, and
1097        // a `Duration` subtraction that underflows panics.
1098        tokio::time::sleep(
1099            POLL_WAIT_STEP.min(POLL_LONG_POLL_MAX.saturating_sub(started.elapsed())),
1100        )
1101        .await;
1102        // The stream as it is now: paused, disabled or gone while waiting.
1103        stream = owned_stream(&state, &receiver, Some(&raw_id)).await?;
1104        if stream.status != SsfStreamStatus::Enabled {
1105            return Ok(empty_poll_response());
1106        }
1107    }
1108}
