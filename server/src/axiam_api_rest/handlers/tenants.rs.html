1//! Tenant management endpoints (nested under organizations).
2//!
3//! # Deleting a tenant destroys its audit trail (T-118)
4//!
5//! Tenant data is tenant-scoped all the way down, and `audit_log` is no
6//! exception: the rows that record what happened inside a tenant carry that
7//! tenant's id and go when it goes. That is the evidence disappearing at
8//! exactly the moment it matters, and until beta03 the threat model resolved
9//! it by telling operators to "export to a WORM sink first" — a rule nothing
10//! checked.
11//!
12//! It is now checked. [`export_audit`] streams a tenant's complete audit trail
13//! and records that it did so *in that tenant's own audit log*; [`delete`]
14//! refuses with `409` unless such a record exists and is less than
15//! [`AUDIT_EXPORT_MAX_AGE_HOURS`] hours old. The window is deliberately short
16//! and deliberately not configurable: the point is that the export is
17//! contemporaneous with the deletion, not that one was taken at some time in
18//! the tenant's history.
19//!
20//! ## What this does and does not prove
21//!
22//! It proves that a named principal asked this server for the whole trail, and
23//! that the server produced it, minutes before the deletion. It does **not**
24//! prove the bytes were kept — an operator can stream the export to
25//! `/dev/null`. Nothing in-product can prove custody of a file it handed to
26//! someone else; what changes is that destroying a tenant's audit trail is now
27//! a deliberate two-step act by an identified principal, recorded twice (the
28//! export receipt, and the deletion record written to the *system* audit log,
29//! which survives the tenant).
30//!
31//! ## GDPR Art. 17
32//!
33//! The gate delays an erasure by the length of one export, and does not block
34//! it: there is no `?force` and no configuration that turns the requirement
35//! off, because the export is the thing that makes erasure safe rather than an
36//! obstacle to it. Art. 17(3)(b)/(e) already permit retention where processing
37//! is necessary for legal obligations or the defence of legal claims, which is
38//! what an exported audit trail is for; and the export contains no more
39//! personal data than the tenant already held. A deployment whose lawful basis
40//! does not permit keeping the export simply deletes the file — the receipt
41//! records that an export happened, never its contents.
42

70/// Audit action recorded by [`export_audit`], and the one [`delete`] looks for.
71///
72/// Namespaced under `tenants.` like the rest of the management surface. It is
73/// written with `outcome = Success` only after the whole trail has been
74/// streamed, so a half-finished download leaves no receipt behind.
75pub const AUDIT_EXPORT_ACTION: &str = "tenants.audit_exported";
76
77/// Audit action recorded — in the **system** log, not the tenant's — when a
78/// tenant is deleted. The tenant's own entries are gone by then; this is what
79/// is left to say the deletion happened, who did it, and which export receipt
80/// authorised it.
81pub const TENANT_DELETED_ACTION: &str = "tenants.deleted";
82
83/// How recent the export receipt must be for [`delete`] to accept it.
84///
85/// Six hours: long enough that an operator can export, verify the file landed
86/// in their archive, and get an approval before deleting; short enough that
87/// the export is unmistakably *about this deletion* rather than a routine dump
88/// from last quarter. Not configurable — a deployment that could set this to a
89/// year would have the rule without the property the rule exists for.
90pub const AUDIT_EXPORT_MAX_AGE_HOURS: i64 = 6;
91
92/// Rows fetched from the datastore per round trip while streaming an export.
93///
94/// The export is streamed rather than assembled because a tenant's audit trail
95/// has no upper bound — the retention sweep (T-119) bounds its *age*, not its
96/// size — and materialising one in the response buffer would make a large
97/// tenant's export an out-of-memory condition on the server.
98const AUDIT_EXPORT_PAGE_SIZE: u64 = 500;
99
100/// One step of the export stream.
101enum ExportPhase {
102    /// Still draining `audit_log`, `offset` rows in.
103    Rows {
104        offset: u64,
105        count: u64,
106        hasher: Sha256,
107    },
108    /// Rows exhausted: write the receipt and emit the trailing manifest line.
109    Manifest { count: u64, digest: String },
110    /// Nothing further to emit.
111    Done,
112}
113
114/// The most recent successful export receipt for `tenant_id`, if one falls
115/// inside the freshness window.
116///
117/// Reads the tenant's own audit log — the receipt is an ordinary append-only
118/// entry, so this needs no new table, no new retention rule and no second
119/// source of truth that could disagree with the trail it is about.
120async fn recent_export_receipt<C: Connection + Clone>(
121    state: &AppState<C>,
122    tenant_id: Uuid,
123    now: DateTime<Utc>,
124) -> Result<Option<(Uuid, DateTime<Utc>)>, AxiamError> {
125    let cutoff = now - chrono::Duration::hours(AUDIT_EXPORT_MAX_AGE_HOURS);
126    let page = state
127        .audit_repo
128        .list(
129            tenant_id,
130            AuditLogFilter {
131                action: Some(AUDIT_EXPORT_ACTION.to_string()),
132                outcome: Some(AuditOutcome::Success),
133                from: Some(cutoff),
134                ..Default::default()
135            },
136            Pagination {
137                offset: 0,
138                // `list` orders by timestamp DESC, so one row is the newest.
139                limit: 1,
140                search: None,
141            },
142        )
143        .await?;
144    Ok(page
145        .items
146        .into_iter()
147        .next()
148        .map(|entry| (entry.id, entry.timestamp)))
149}

172/// `POST /api/v1/organizations/{org_id}/tenants`
173#[utoipa::path(
174    post,
175    path = "/api/v1/organizations/{org_id}/tenants",
176    tag = "tenants",
177    params(("org_id" = Uuid, Path, description = "Organization ID")),
178    request_body = CreateTenantRequest,
179    responses(
180        (status = 201, description = "Tenant created", body = Tenant),
181    ),
182    security(("bearer" = []))
183)]
184pub async fn create<C: Connection + Clone>(
185    user: AuthenticatedUser,
186    authz: AuthzData,
187    state: web::Data<AppState<C>>,
188    path: web::Path<OrgPath>,
189    body: web::Json<CreateTenantRequest>,
190) -> Result<HttpResponse, AxiamApiError> {
191    RequirePermission::new("tenants:create", Uuid::nil())
192        .check(&user, authz.get_ref().as_ref())
193        .await?;
194    // Organization-level action: the caller must live in the organization
195    // scope, not merely belong to the organization. See `handlers::org_scope`.
196    crate::handlers::org_scope::require_organization_principal(&user, state.get_ref()).await?;
197
198    // Authorization: only allow creating tenants under the caller's own org.
199    if path.org_id != user.org_id {
200        return Err(AxiamApiError(
201            axiam_core::error::AxiamError::AuthorizationDenied {
202                reason: "cannot access a different organization".into(),
203                action: None,
204                resource_id: None,
205            },
206        ));
207    }
208
209    let req = body.into_inner();
210
211    // The organization scope is created with its organization, never through
212    // this endpoint. Refusing the reserved slug here turns what would be an
213    // opaque unique-index violation into a message that says which name is
214    // taken and why.
215    if req.slug == axiam_core::models::tenant::ORGANIZATION_TENANT_SLUG {
216        return Err(AxiamApiError(AxiamError::Validation {
217            message: format!(
218                "`{}` is reserved for the organization's own scope, which is created \
219                 with the organization; choose another slug",
220                axiam_core::models::tenant::ORGANIZATION_TENANT_SLUG
221            ),
222        }));
223    }
224
225    let input = CreateTenant {
226        organization_id: path.org_id,
227        name: req.name,
228        slug: req.slug,
229        // Always an ordinary tenant: `CreateTenant::kind` is
230        // `skip_deserializing`, so a caller cannot ask for anything else, and
231        // saying so explicitly here keeps that visible at the one call site
232        // that matters.
233        kind: TenantKind::Standard,
234        metadata: req.metadata,
235    };
236    let tenant = state.tenant_repo.create(input).await?;
237
238    // Auto-seed permissions for the new tenant so RBAC works immediately.
239    seed_permissions(&state.db.current(), tenant.id, PERMISSION_REGISTRY)
240        .await
241        .map_err(|e| {
242            tracing::error!(
243                "Failed to seed permissions for new tenant {}: {}",
244                tenant.id,
245                e
246            );
247            AxiamApiError(AxiamError::Internal(
248                "Failed to seed permissions for tenant".into(),
249            ))
250        })?;
251
252    // …and its roles. Seeding permissions alone left the tenant holding a set
253    // of actions with nothing to attach them to: no roles, no grants, no
254    // assignments. The authorization engine filters every lookup by tenant, so
255    // the first question anyone asked about the new tenant — including the
256    // person who had just created it — resolved to `no roles assigned` and was
257    // denied. A tenant nobody can administer is not a tenant.
258    //
259    // Not best-effort, unlike the permission seed's sibling failure paths
260    // elsewhere: a tenant with permissions and no roles is exactly the
261    // unreachable state this repairs, so failing here has to be visible rather
262    // than leaving one behind for someone to find later.
263    seed_default_roles(&state.db.current(), tenant.id, PERMISSION_REGISTRY)
264        .await
265        .map_err(|e| {
266            tracing::error!(
267                tenant_id = %tenant.id,
268                error = %e,
269                "Failed to seed default roles for new tenant"
270            );
271            AxiamApiError(AxiamError::Internal(
272                "Failed to seed default roles for tenant".into(),
273            ))
274        })?;
275
276    // Nothing is assigned to anyone here, deliberately. Organization-level
277    // principals already reach this tenant — their global grants live in the
278    // organization tenant and carry across by the rule in
279    // `AuthorizationEngine::evaluate` — so fanning assignments out at creation
280    // time would write rows that grant nothing new, would miss every
281    // organization-level principal created afterwards, and would have to be
282    // undone in every tenant to revoke. See
283    // `claude_dev/organization-scope-design.md`.
284
285    Ok(HttpResponse::Created().json(tenant))
286}

288/// `GET /api/v1/organizations/{org_id}/tenants`
289#[utoipa::path(
290    get,
291    path = "/api/v1/organizations/{org_id}/tenants",
292    tag = "tenants",
293    params(
294        ("org_id" = Uuid, Path, description = "Organization ID"),
295        Pagination,
296    ),
297    responses(
298        (status = 200, description = "List of tenants", body = inline(PaginatedResult<Tenant>)),
299    ),
300    security(("bearer" = []))
301)]
302pub async fn list<C: Connection + Clone>(
303    user: AuthenticatedUser,
304    authz: AuthzData,
305    state: web::Data<AppState<C>>,
306    path: web::Path<OrgPath>,
307    query: web::Query<Pagination>,
308) -> Result<HttpResponse, AxiamApiError> {
309    // Authorization: only allow listing tenants under the caller's own org.
310    //
311    // Ahead of the permission check now, so a cross-organization caller still
312    // gets "cannot access a different organization" rather than a reach message
313    // about an organization that is not its own.
314    if path.org_id != user.org_id {
315        return Err(AxiamApiError(
316            axiam_core::error::AxiamError::AuthorizationDenied {
317                reason: "cannot access a different organization".into(),
318                action: None,
319                resource_id: None,
320            },
321        ));
322    }
323
324    // Reach first, because for a restricted principal it decides *where* the
325    // permission question can be answered.
326    //
327    // `RequirePermission::check` asks in `user.tenant_id`, which with no tenant
328    // selected is the organization's own reserved tenant. An organization-level
329    // account restricted to particular tenants holds nothing there —
330    // deliberately, and `axiam_authz`'s
331    // `a_tenant_scoped_assignment_does_not_apply_in_the_organization_scope`
332    // pins it — so the roster answered 403 and the filtering below, written for
333    // exactly this principal, was unreachable. The account could not read the
334    // one list that tells it which tenants it administers.
335    //
336    // Asked instead in a tenant it does reach, which is where it genuinely
337    // holds `tenants:list`. One tenant is enough: a `tenant_scope` applies to a
338    // whole assignment, so the same roles apply in every tenant it names and
339    // each returns the same answer.
340    let reachable = reachable_tenants(&user, state.get_ref(), path.org_id).await?;
341    let home = state
342        .tenant_repo
343        .get_by_id(user.principal_tenant_id)
344        .await?;
345    let gate = RequirePermission::new("tenants:list", Uuid::nil());
346    match reachable.as_deref() {
347        // The restricted organization principal, and only it. The subject scope
348        // is spelled out rather than taken from `user.subject_scope()`, which
349        // reports `Tenant` unless the request named another tenant with
350        // `X-Axiam-Tenant` — and this request names none. Taking it from there
351        // would look for the caller's grants in `first`, where they do not
352        // live, and deny for a second wrong reason.
353        Some([first, ..]) if home.is_organization_scope() => {
354            gate.check_subject_from(
355                *first,
356                SubjectScope::Organization { tenant_id: home.id },
357                user.user_id,
358                authz.get_ref().as_ref(),
359            )
360            .await?;
361        }
362        // Reaching nothing is nothing to list.
363        Some([]) if home.is_organization_scope() => {
364            return Err(AxiamApiError(
365                axiam_core::error::AxiamError::AuthorizationDenied {
366                    reason: "no roles assigned that reach any tenant of this organization".into(),
367                    action: Some("tenants:list".into()),
368                    resource_id: None,
369                },
370            ));
371        }
372        // Every other caller — an ordinary tenant principal, or an unrestricted
373        // organization one — keeps the check exactly where it has always been.
374        _ => gate.check(&user, authz.get_ref().as_ref()).await?,
375    }
376
377    // Only the tenants this caller can actually act on.
378    //
379    // `tenants:list` says the caller may see a tenant roster; it does not say
380    // *which* roster. Before this, every principal holding it saw every tenant
381    // of the organization — including a tenant administrator, whose reach is
382    // one tenant, and a deliberately narrowed organization-level account, whose
383    // reach is the few tenants it was given. Both were shown the whole list and
384    // then refused when they touched anything in it.
385    //
386    // That is worse than a usability problem. The roster names every workspace
387    // in the organization, their slugs and their creation dates; an
388    // administrator of one tenant learning the shape of its siblings is a small
389    // disclosure, but it is one the isolation boundary is supposed to prevent
390    // and one nothing needed.
391    //
392    // The filter is applied here rather than in the repository because it is a
393    // fact about the *caller*, not about the query — `list_by_organization` is
394    // also what the organization's own administrator uses, unfiltered.
395    let pagination = query.into_inner();
396    let result = match reachable {
397        None => {
398            state
399                .tenant_repo
400                .list_by_organization(path.org_id, pagination)
401                .await?
402        }
403        Some(ids) => page_of(collect_tenants(state.get_ref(), &ids).await?, &pagination),
404    };
405    Ok(HttpResponse::Ok().json(result))
406}

408/// The tenants `user` may act on within `org_id`, or `None` for "all of them".
409///
410/// Three answers, in the order they are decided:
411///
412/// * An **ordinary tenant principal** reaches exactly the tenant it lives in.
413///   `resolve_active_tenant` refuses it any other, so a longer list would be a
414///   list of tenants it cannot open.
415/// * An **organization-level principal with tenant-scoped roles** reaches the
416///   tenants those roles name.
417/// * An **organization-level principal with unrestricted roles** reaches every
418///   tenant of the organization — `None`, so the repository's own paginated
419///   query is used unchanged.
420async fn reachable_tenants<C: Connection + Clone>(
421    user: &AuthenticatedUser,
422    state: &AppState<C>,
423    org_id: Uuid,
424) -> Result<Option<Vec<Uuid>>, AxiamApiError> {
425    let home = state
426        .tenant_repo
427        .get_by_id(user.principal_tenant_id)
428        .await?;
429    if !home.is_organization_scope() {
430        // Its own tenant, and only if it is in the organization being listed —
431        // which `list`'s own `org_id` check already established, but repeating
432        // it here keeps this function correct on its own terms.
433        return Ok(Some(
434            (home.organization_id == org_id)
435                .then_some(vec![home.id])
436                .unwrap_or_default(),
437        ));
438    }
439
440    match crate::handlers::org_scope::principal_tenant_reach(user, state).await? {
441        axiam_core::models::role::TenantReach::Unrestricted => Ok(None),
442        axiam_core::models::role::TenantReach::Restricted(tenants) => {
443            Ok(Some(tenants.into_iter().collect()))
444        }
445    }
446}

448/// Load the named tenants, dropping any that no longer exist.
449///
450/// A tenant scope is a list of ids written when the assignment was made, and a
451/// tenant can be deleted afterwards. A dangling id is not an error to report to
452/// the caller — there is nothing they can do about it and the assignment is
453/// still valid for the rest — so it is simply not listed.
454async fn collect_tenants<C: Connection + Clone>(
455    state: &AppState<C>,
456    ids: &[Uuid],
457) -> Result<Vec<Tenant>, AxiamApiError> {
458    let mut tenants = Vec::with_capacity(ids.len());
459    for id in ids {
460        if let Ok(tenant) = state.tenant_repo.get_by_id(*id).await {
461            tenants.push(tenant);
462        }
463    }
464    // The repository lists by name; a filtered list that ordered differently
465    // would make the same tenants appear to move when a restriction is applied.
466    tenants.sort_by(|a, b| a.name.cmp(&b.name));
467    Ok(tenants)
468}

470/// Apply `pagination` to an already-materialised list.
471///
472/// `total` counts the whole reachable set rather than the page, so the pager
473/// belongs to the result set it is paging — the same contract
474/// [`Pagination`](axiam_core::repository::Pagination) states for the
475/// repository-side queries.
476fn page_of(tenants: Vec<Tenant>, pagination: &Pagination) -> PaginatedResult<Tenant> {
477    let filtered: Vec<Tenant> = match pagination.search.as_deref() {
478        Some(term) if !term.trim().is_empty() => {
479            let needle = term.trim().to_lowercase();
480            tenants
481                .into_iter()
482                .filter(|t| {
483                    t.name.to_lowercase().contains(&needle)
484                        || t.slug.to_lowercase().contains(&needle)
485                        || t.id.to_string().contains(&needle)
486                })
487                .collect()
488        }
489        _ => tenants,
490    };
491    let total = filtered.len() as u64;
492    let items = filtered
493        .into_iter()
494        .skip(pagination.offset as usize)
495        .take(pagination.limit as usize)
496        .collect();
497    PaginatedResult {
498        items,
499        total,
500        offset: pagination.offset,
501        limit: pagination.limit,
502    }
503}

610    delete,
611    path = "/api/v1/organizations/{org_id}/tenants/{tenant_id}",
612    tag = "tenants",
613    params(
614        ("org_id" = Uuid, Path, description = "Organization ID"),
615        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
616    ),
617    responses(
618        (status = 204, description = "Tenant deleted"),
619        (status = 404, description = "Tenant not found"),
620    ),
621    security(("bearer" = []))
622)]
623pub async fn delete<C: Connection + Clone>(
624    user: AuthenticatedUser,
625    authz: AuthzData,
626    state: web::Data<AppState<C>>,
627    path: web::Path<TenantPath>,
628) -> Result<HttpResponse, AxiamApiError> {
629    RequirePermission::new("tenants:delete", Uuid::nil())
630        .check(&user, authz.get_ref().as_ref())
631        .await?;
632    // Organization-level action: the caller must live in the organization
633    // scope, not merely belong to the organization. See `handlers::org_scope`.
634    crate::handlers::org_scope::require_organization_principal(&user, state.get_ref()).await?;
635
636    // Authorization: reject cross-org probing before touching the DB.
637    if path.org_id != user.org_id {
638        return Err(AxiamApiError(
639            axiam_core::error::AxiamError::AuthorizationDenied {
640                reason: "cannot access a different organization".into(),
641                action: None,
642                resource_id: None,
643            },
644        ));
645    }
646
647    let existing = state.tenant_repo.get_by_id(path.tenant_id).await?;
648    if existing.organization_id != path.org_id {
649        return Err(AxiamError::NotFound {
650            entity: "Tenant".into(),
651            id: path.tenant_id.to_string(),
652        }
653        .into());
654    }
655
656    // T-118: refuse unless the trail about to be destroyed was exported
657    // recently. Checked BEFORE the delete, so a refusal leaves the tenant
658    // exactly as it was.
659    let now = Utc::now();
660    let Some((receipt_id, exported_at)) =
661        recent_export_receipt(state.get_ref(), path.tenant_id, now).await?
662    else {
663        return Err(AxiamError::Conflict {
664            reason: format!(
665                "deleting this tenant destroys its audit trail. Export it first with \
666                 POST /api/v1/organizations/{}/tenants/{}/audit-export, then retry \
667                 within {AUDIT_EXPORT_MAX_AGE_HOURS} hours.",
668                path.org_id, path.tenant_id
669            ),
670        }
671        .into());
672    };
673
674    state.tenant_repo.delete(path.tenant_id).await?;
675
676    // The tenant's own audit entries — including the export receipt just
677    // checked — went with it, so this record goes to the SYSTEM log (nil
678    // tenant), which is the only part of the trail that outlives the tenant.
679    // Written through the dead-lettering sink for the same reason the erasure
680    // records are: this is the last chance to record it, and a transient
681    // datastore failure must not be how it gets lost.
682    write_erasure_audit_with_dlq(
683        &state.audit_repo,
684        CreateAuditLogEntry {
685            tenant_id: Uuid::nil(),
686            actor_id: user.user_id,
687            actor_type: ActorType::User,
688            action: TENANT_DELETED_ACTION.to_string(),
689            resource_id: Some(path.tenant_id),
690            outcome: AuditOutcome::Success,
691            ip_address: None,
692            metadata: Some(serde_json::json!({
693                "organization_id": path.org_id,
694                "tenant_slug": existing.slug,
695                "audit_export_receipt_id": receipt_id,
696                "audit_exported_at": exported_at,
697            })),
698        },
699    )
700    .await;
701
702    Ok(HttpResponse::NoContent().finish())
703}

705/// `POST /api/v1/organizations/{org_id}/tenants/{tenant_id}/audit-export`
706///
707/// Streams the tenant's complete audit trail as newline-delimited JSON, one
708/// [`axiam_core::models::audit::AuditLogEntry`] per line, newest first, and
709/// records the export in that tenant's audit log — which is what
710/// [`delete`] then requires (T-118).
711///
712/// The last line is not an entry but a manifest object carrying
713/// `record_count`, the SHA-256 `digest` over the entry lines that precede it,
714/// and `receipt_id` — the id of the audit entry this export wrote. An archive
715/// can therefore be tied back to the receipt that authorised the deletion, and
716/// re-hashing the file proves it is the export the receipt describes.
717///
718/// `POST` rather than `GET`: it is not safe and not idempotent — every call
719/// appends a receipt.
720#[utoipa::path(
721    post,
722    path = "/api/v1/organizations/{org_id}/tenants/{tenant_id}/audit-export",
723    tag = "tenants",
724    params(
725        ("org_id" = Uuid, Path, description = "Organization ID"),
726        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
727    ),
728    responses(
729        (status = 200, description = "Newline-delimited audit entries, then a \
730             manifest line", content_type = "application/x-ndjson"),
731        (status = 404, description = "Tenant not found"),
732    ),
733    security(("bearer" = []))
734)]
735pub async fn export_audit<C: Connection + Clone>(
736    user: AuthenticatedUser,
737    authz: AuthzData,
738    state: web::Data<AppState<C>>,
739    path: web::Path<TenantPath>,
740) -> Result<HttpResponse, AxiamApiError> {
741    RequirePermission::new("tenants:export_audit", Uuid::nil())
742        .check(&user, authz.get_ref().as_ref())
743        .await?;
744
745    // Authorization: reject cross-org probing before touching the DB.
746    if path.org_id != user.org_id {
747        return Err(AxiamApiError(AxiamError::AuthorizationDenied {
748            reason: "cannot access a different organization".into(),
749            action: None,
750            resource_id: None,
751        }));
752    }
753
754    let existing = state.tenant_repo.get_by_id(path.tenant_id).await?;
755    if existing.organization_id != path.org_id {
756        return Err(AxiamError::NotFound {
757            entity: "Tenant".into(),
758            id: path.tenant_id.to_string(),
759        }
760        .into());
761    }
762
763    let tenant_id = path.tenant_id;
764    let actor_id = user.user_id;
765    let audit_repo = state.audit_repo.clone();
766    // Fix the upper bound of the export at the moment it starts. Without it,
767    // rows appended while the stream is draining would shift the offsets of
768    // every page after them and the export would skip entries.
769    let started_at = Utc::now();
770
771    let stream = futures::stream::unfold(
772        ExportPhase::Rows {
773            offset: 0,
774            count: 0,
775            hasher: Sha256::new(),
776        },
777        move |phase| {
778            let audit_repo = audit_repo.clone();
779            async move {
780                match phase {
781                    ExportPhase::Rows {
782                        offset,
783                        count,
784                        mut hasher,
785                    } => {
786                        let page = audit_repo
787                            .list(
788                                tenant_id,
789                                AuditLogFilter {
790                                    to: Some(started_at),
791                                    ..Default::default()
792                                },
793                                Pagination {
794                                    offset,
795                                    limit: AUDIT_EXPORT_PAGE_SIZE,
796                                    search: None,
797                                },
798                            )
799                            .await;
800                        let page = match page {
801                            Ok(page) => page,
802                            Err(e) => {
803                                tracing::error!(
804                                    error = %e,
805                                    %tenant_id,
806                                    "tenants: audit export failed mid-stream"
807                                );
808                                // Ending the body with an error truncates the
809                                // transfer, so the caller sees a failure rather
810                                // than a short file that looks complete. No
811                                // receipt is written, so `delete` still refuses.
812                                return Some((
813                                    Err(actix_web::error::ErrorInternalServerError(
814                                        "audit export failed",
815                                    )),
816                                    ExportPhase::Done,
817                                ));
818                            }
819                        };
820
821                        if page.items.is_empty() {
822                            let digest = hex::encode(hasher.finalize());
823                            return Some((
824                                Ok(web::Bytes::new()),
825                                ExportPhase::Manifest { count, digest },
826                            ));
827                        }
828
829                        let fetched = page.items.len() as u64;
830                        let mut chunk = String::new();
831                        for entry in &page.items {
832                            match serde_json::to_string(entry) {
833                                Ok(line) => {
834                                    chunk.push_str(&line);
835                                    chunk.push('\n');
836                                }
837                                Err(e) => {
838                                    tracing::error!(
839                                        error = %e,
840                                        %tenant_id,
841                                        "tenants: audit entry could not be serialized"
842                                    );
843                                    return Some((
844                                        Err(actix_web::error::ErrorInternalServerError(
845                                            "audit export failed",
846                                        )),
847                                        ExportPhase::Done,
848                                    ));
849                                }
850                            }
851                        }
852                        hasher.update(chunk.as_bytes());
853                        Some((
854                            Ok(web::Bytes::from(chunk)),
855                            ExportPhase::Rows {
856                                offset: offset + fetched,
857                                count: count + fetched,
858                                hasher,
859                            },
860                        ))
861                    }
862
863                    ExportPhase::Manifest { count, digest } => {
864                        // The receipt is written HERE — after the last row has
865                        // been handed to the client — so that an export which
866                        // dies half way leaves nothing behind that would let a
867                        // tenant be deleted.
868                        let receipt = audit_repo
869                            .append(CreateAuditLogEntry {
870                                tenant_id,
871                                actor_id,
872                                actor_type: ActorType::User,
873                                action: AUDIT_EXPORT_ACTION.to_string(),
874                                resource_id: Some(tenant_id),
875                                outcome: AuditOutcome::Success,
876                                ip_address: None,
877                                metadata: Some(serde_json::json!({
878                                    "record_count": count,
879                                    "digest": format!("sha256:{digest}"),
880                                    "exported_through": started_at,
881                                })),
882                            })
883                            .await;
884                        let receipt = match receipt {
885                            Ok(entry) => entry,
886                            Err(e) => {
887                                tracing::error!(
888                                    error = %e,
889                                    %tenant_id,
890                                    "tenants: audit export receipt could not be written"
891                                );
892                                // Without a receipt the export did not happen as
893                                // far as `delete` is concerned, so the caller
894                                // must be told rather than handed a manifest
895                                // that claims otherwise.
896                                return Some((
897                                    Err(actix_web::error::ErrorInternalServerError(
898                                        "audit export receipt could not be written",
899                                    )),
900                                    ExportPhase::Done,
901                                ));
902                            }
903                        };
904
905                        let manifest = serde_json::json!({
906                            "axiam_export": "tenant_audit",
907                            "tenant_id": tenant_id,
908                            "exported_by": actor_id,
909                            "exported_through": started_at,
910                            "record_count": count,
911                            "digest": format!("sha256:{digest}"),
912                            "receipt_id": receipt.id,
913                            "receipt_valid_for_hours": AUDIT_EXPORT_MAX_AGE_HOURS,
914                        });
915                        Some((
916                            Ok(web::Bytes::from(format!("{manifest}\n"))),
917                            ExportPhase::Done,
918                        ))
919                    }
920
921                    ExportPhase::Done => None,
922                }
923            }
924        },
925    );
926
927    let filename = format!(
928        "axiam-audit-{tenant_id}-{}.ndjson",
929        started_at.format("%Y%m%dT%H%M%SZ")
930    );
931    Ok(HttpResponse::Ok()
932        .content_type("application/x-ndjson")
933        .insert_header((
934            header::CONTENT_DISPOSITION,
935            format!("attachment; filename=\"{filename}\""),
936        ))
937        .streaming(Box::pin(stream)))
938}
