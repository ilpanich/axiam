40/// `POST /api/v1/organizations/{org_id}/tenants`
41#[utoipa::path(
42    post,
43    path = "/api/v1/organizations/{org_id}/tenants",
44    tag = "tenants",
45    params(("org_id" = Uuid, Path, description = "Organization ID")),
46    request_body = CreateTenantRequest,
47    responses(
48        (status = 201, description = "Tenant created", body = Tenant),
49    ),
50    security(("bearer" = []))
51)]
52pub async fn create<C: Connection + Clone>(
53    user: AuthenticatedUser,
54    authz: AuthzData,
55    state: web::Data<AppState<C>>,
56    path: web::Path<OrgPath>,
57    body: web::Json<CreateTenantRequest>,
58) -> Result<HttpResponse, AxiamApiError> {
59    RequirePermission::new("tenants:create", Uuid::nil())
60        .check(&user, authz.get_ref().as_ref())
61        .await?;
62
63    // Authorization: only allow creating tenants under the caller's own org.
64    if path.org_id != user.org_id {
65        return Err(AxiamApiError(
66            axiam_core::error::AxiamError::AuthorizationDenied {
67                reason: "cannot access a different organization".into(),
68                action: None,
69                resource_id: None,
70            },
71        ));
72    }
73
74    let req = body.into_inner();
75
76    // The organization scope is created with its organization, never through
77    // this endpoint. Refusing the reserved slug here turns what would be an
78    // opaque unique-index violation into a message that says which name is
79    // taken and why.
80    if req.slug == axiam_core::models::tenant::ORGANIZATION_TENANT_SLUG {
81        return Err(AxiamApiError(AxiamError::Validation {
82            message: format!(
83                "`{}` is reserved for the organization's own scope, which is created \
84                 with the organization; choose another slug",
85                axiam_core::models::tenant::ORGANIZATION_TENANT_SLUG
86            ),
87        }));
88    }
89
90    let input = CreateTenant {
91        organization_id: path.org_id,
92        name: req.name,
93        slug: req.slug,
94        // Always an ordinary tenant: `CreateTenant::kind` is
95        // `skip_deserializing`, so a caller cannot ask for anything else, and
96        // saying so explicitly here keeps that visible at the one call site
97        // that matters.
98        kind: TenantKind::Standard,
99        metadata: req.metadata,
100    };
101    let tenant = state.tenant_repo.create(input).await?;
102
103    // Auto-seed permissions for the new tenant so RBAC works immediately.
104    seed_permissions(&state.db.current(), tenant.id, PERMISSION_REGISTRY)
105        .await
106        .map_err(|e| {
107            tracing::error!(
108                "Failed to seed permissions for new tenant {}: {}",
109                tenant.id,
110                e
111            );
112            AxiamApiError(AxiamError::Internal(
113                "Failed to seed permissions for tenant".into(),
114            ))
115        })?;
116
117    // …and its roles. Seeding permissions alone left the tenant holding a set
118    // of actions with nothing to attach them to: no roles, no grants, no
119    // assignments. The authorization engine filters every lookup by tenant, so
120    // the first question anyone asked about the new tenant — including the
121    // person who had just created it — resolved to `no roles assigned` and was
122    // denied. A tenant nobody can administer is not a tenant.
123    //
124    // Not best-effort, unlike the permission seed's sibling failure paths
125    // elsewhere: a tenant with permissions and no roles is exactly the
126    // unreachable state this repairs, so failing here has to be visible rather
127    // than leaving one behind for someone to find later.
128    seed_default_roles(&state.db.current(), tenant.id, PERMISSION_REGISTRY)
129        .await
130        .map_err(|e| {
131            tracing::error!(
132                tenant_id = %tenant.id,
133                error = %e,
134                "Failed to seed default roles for new tenant"
135            );
136            AxiamApiError(AxiamError::Internal(
137                "Failed to seed default roles for tenant".into(),
138            ))
139        })?;
140
141    // Nothing is assigned to anyone here, deliberately. Organization-level
142    // principals already reach this tenant — their global grants live in the
143    // organization tenant and carry across by the rule in
144    // `AuthorizationEngine::evaluate` — so fanning assignments out at creation
145    // time would write rows that grant nothing new, would miss every
146    // organization-level principal created afterwards, and would have to be
147    // undone in every tenant to revoke. See
148    // `claude_dev/organization-scope-design.md`.
149
150    Ok(HttpResponse::Created().json(tenant))
151}

153/// `GET /api/v1/organizations/{org_id}/tenants`
154#[utoipa::path(
155    get,
156    path = "/api/v1/organizations/{org_id}/tenants",
157    tag = "tenants",
158    params(
159        ("org_id" = Uuid, Path, description = "Organization ID"),
160        Pagination,
161    ),
162    responses(
163        (status = 200, description = "List of tenants", body = inline(PaginatedResult<Tenant>)),
164    ),
165    security(("bearer" = []))
166)]
167pub async fn list<C: Connection + Clone>(
168    user: AuthenticatedUser,
169    authz: AuthzData,
170    state: web::Data<AppState<C>>,
171    path: web::Path<OrgPath>,
172    query: web::Query<Pagination>,
173) -> Result<HttpResponse, AxiamApiError> {
174    RequirePermission::new("tenants:list", Uuid::nil())
175        .check(&user, authz.get_ref().as_ref())
176        .await?;
177
178    // Authorization: only allow listing tenants under the caller's own org.
179    if path.org_id != user.org_id {
180        return Err(AxiamApiError(
181            axiam_core::error::AxiamError::AuthorizationDenied {
182                reason: "cannot access a different organization".into(),
183                action: None,
184                resource_id: None,
185            },
186        ));
187    }
188
189    let result = state
190        .tenant_repo
191        .list_by_organization(path.org_id, query.into_inner())
192        .await?;
193    Ok(HttpResponse::Ok().json(result))
194}

196/// `GET /api/v1/organizations/{org_id}/tenants/{tenant_id}`
197#[utoipa::path(
198    get,
199    path = "/api/v1/organizations/{org_id}/tenants/{tenant_id}",
200    tag = "tenants",
201    params(
202        ("org_id" = Uuid, Path, description = "Organization ID"),
203        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
204    ),
205    responses(
206        (status = 200, description = "Tenant found", body = Tenant),
207        (status = 404, description = "Tenant not found"),
208    ),
209    security(("bearer" = []))
210)]
211pub async fn get<C: Connection + Clone>(
212    user: AuthenticatedUser,
213    authz: AuthzData,
214    state: web::Data<AppState<C>>,
215    path: web::Path<TenantPath>,
216) -> Result<HttpResponse, AxiamApiError> {
217    RequirePermission::new("tenants:get", Uuid::nil())
218        .check(&user, authz.get_ref().as_ref())
219        .await?;
220
221    // Authorization: reject cross-org probing before touching the DB.
222    if path.org_id != user.org_id {
223        return Err(AxiamApiError(
224            axiam_core::error::AxiamError::AuthorizationDenied {
225                reason: "cannot access a different organization".into(),
226                action: None,
227                resource_id: None,
228            },
229        ));
230    }
231
232    let tenant = state.tenant_repo.get_by_id(path.tenant_id).await?;
233    if tenant.organization_id != path.org_id {
234        return Err(AxiamError::NotFound {
235            entity: "Tenant".into(),
236            id: path.tenant_id.to_string(),
237        }
238        .into());
239    }
240    Ok(HttpResponse::Ok().json(tenant))
241}
