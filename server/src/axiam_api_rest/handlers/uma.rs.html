28/// A validated **Protection API Token** and the resource server it names.
29///
30/// Its own extractor rather than [`AuthenticatedUser`] or
31/// `AuthenticatedPrincipal`, because a PAT is a third principal shape and
32/// neither of those can carry it:
33///
34/// - `AuthenticatedUser` narrows the audience to `axiam:user`, and a PAT is a
35///   client-credentials token carrying `axiam:m2m`.
36/// - `AuthenticatedPrincipal` accepts `axiam:m2m` but parses `sub` as a UUID,
37///   which deliberately excludes OAuth2 clients — their `sub` is a `client_id`
38///   string, and they have no row in the role-assignment graph.
39///
40/// That exclusion is right for RBAC subjects and wrong for this endpoint: the
41/// Protection API does not evaluate the caller against grants at all. It only
42/// needs to know **which client is speaking**, because that is what the minted
43/// ticket is bound to. So this extractor keeps `sub` as the string it is.
44pub struct ProtectionApiToken {
45    /// The resource server's `client_id` — the token's `sub`, unparsed.
46    pub client_id: String,
47    pub tenant_id: Uuid,
48}

50impl actix_web::FromRequest for ProtectionApiToken {
51    type Error = AxiamApiError;
52    type Future = std::future::Ready<Result<Self, Self::Error>>;
53
54    fn from_request(
55        req: &actix_web::HttpRequest,
56        _payload: &mut actix_web::dev::Payload,
57    ) -> Self::Future {
58        std::future::ready(extract_pat(req))
59    }
60}

62fn extract_pat(req: &actix_web::HttpRequest) -> Result<ProtectionApiToken, AxiamApiError> {
63    let claims = crate::extractors::auth::parse_validated_claims(req)?.0;
64
65    // Checked before the scope, so a user token gets the answer that names the
66    // real problem rather than being told it is missing a scope it could add.
67    if claims.sub_kind != SubjectKind::OAuth2Client {
68        return Err(forbidden(
69            "a protection API token must be issued to an OAuth2 client, because \
70             a permission ticket is bound to the client that minted it"
71                .into(),
72        ));
73    }
74
75    // The `uma_protection` scope is the only thing that distinguishes a PAT
76    // from any other client-credentials token, so it is not optional.
77    if !claims
78        .scope
79        .as_deref()
80        .is_some_and(|s| s.split(' ').any(|granted| granted == UMA_PROTECTION_SCOPE))
81    {
82        return Err(forbidden(format!(
83            "the protection API requires the '{UMA_PROTECTION_SCOPE}' scope"
84        )));
85    }
86
87    let tenant_id = Uuid::parse_str(&claims.tenant_id).map_err(|_| {
88        AxiamApiError(AxiamError::AuthenticationFailed {
89            reason: "invalid tenant_id claim".into(),
90        })
91    })?;
92
93    Ok(ProtectionApiToken {
94        client_id: claims.sub,
95        tenant_id,
96    })
97}

130/// `POST /uma2/perm` — mint a permission ticket (UMA 2.0 §3.2).
131///
132/// # Who may call this
133///
134/// A **Protection API Token**: an ordinary AXIAM access token carrying the
135/// `uma_protection` scope. Two further conditions, both deliberate:
136///
137/// 1. **The token must be an OAuth2 client token** (`sub_kind` =
138///    `OAuth2Client`). A ticket is bound to the `client_id` that minted it, and
139///    for a client-credentials token `sub` *is* the `client_id`. A user token
140///    has no client identity in it, so binding one would mean inventing a
141///    value and calling it a binding.
142/// 2. **The scope is checked here, not by a route guard.** The `uma_protection`
143///    scope is what makes an access token a PAT; nothing else about the token
144///    distinguishes it.
145#[utoipa::path(
146    post,
147    path = "/uma2/perm",
148    tag = "uma",
149    request_body = PermissionRequestBody,
150    responses(
151        (status = 201, description = "Ticket minted", body = PermissionTicketResponse),
152        (status = 400, description = "Malformed request or undeclared scope"),
153        (status = 401, description = "Missing or invalid PAT"),
154        (status = 403, description = "Token lacks the uma_protection scope"),
155    ),
156    security(("bearer_auth" = []))
157)]
158pub async fn permission_request<C: Connection + Clone>(
159    pat: ProtectionApiToken,
160    authz: AuthzData,
161    state: web::Data<AppState<C>>,
162    body: web::Json<PermissionRequestBody>,
163) -> Result<HttpResponse, AxiamApiError> {
164    let minted = state
165        .uma_service(authz.get_ref().clone())
166        .request_ticket(pat.tenant_id, &pat.client_id, body.into_inner().0)
167        .await
168        .map_err(uma_error_to_api)?;
169
170    tracing::info!(
171        target: "axiam::audit",
172        event = "uma.ticket_minted",
173        tenant_id = %pat.tenant_id,
174        client_id = %pat.client_id,
175        ticket_id = %minted.ticket_id,
176        "permission ticket minted"
177    );
178
179    Ok(HttpResponse::Created()
180        .append_header(("Cache-Control", "no-store"))
181        .append_header(("Pragma", "no-cache"))
182        .json(PermissionTicketResponse {
183            ticket: minted.ticket,
184        }))
185}

187/// `GET /.well-known/uma2-configuration` — UMA 2.0 discovery (§2).
188///
189/// Advertises only what v1 actually serves. Claims-gathering
190/// (`claims_interaction_endpoint`) is absent because it is not implemented;
191/// advertising an endpoint that 404s is worse than not advertising it, since a
192/// conforming client would route to it on the strength of this document.
193#[utoipa::path(
194    get,
195    path = "/.well-known/uma2-configuration",
196    tag = "uma",
197    responses((status = 200, description = "UMA 2.0 configuration", body = Uma2Configuration))
198)]
199pub async fn uma2_configuration(auth_config: web::Data<AuthConfig>) -> HttpResponse {
200    let issuer = auth_config
201        .oauth2_issuer_url
202        .trim_end_matches('/')
203        .to_string();
204
205    HttpResponse::Ok().json(Uma2Configuration {
206        token_endpoint: format!("{issuer}/oauth2/token"),
207        introspection_endpoint: format!("{issuer}/oauth2/introspect"),
208        permission_endpoint: format!("{issuer}/uma2/perm"),
209        resource_registration_endpoint: format!("{issuer}/uma2/rreg/resource_set"),
210        jwks_uri: format!("{issuer}/.well-known/jwks.json"),
211        issuer,
212        grant_types_supported: vec![UMA_TICKET_GRANT_TYPE.to_string()],
213        uma_profiles_supported: vec![],
214        permission_ticket_lifetime: TICKET_TTL_SECS,
215    })
216}

222/// A resource set as UMA describes it, mapped onto an AXIAM resource.
223///
224/// # The mapping, in one place
225///
226/// | UMA | AXIAM |
227/// |---|---|
228/// | `_id` | the resource id — **not** a parallel identifier |
229/// | `name` | `resource.name` |
230/// | `type` | `resource.resource_type` |
231/// | `resource_scopes` | the `Scope` rows on that resource |
232///
233/// There is no second resource store. A resource registered here is an
234/// ordinary AXIAM resource that the admin UI lists, that role assignments
235/// cascade through, and that the authorization engine already understands —
236/// which is the entire reason UMA `_id` is the AXIAM id rather than a
237/// translation key.
238#[derive(Debug, Deserialize, Serialize, utoipa::ToSchema)]
239pub struct ResourceSet {
240    /// Present on responses, absent on registration — the server assigns it.
241    #[serde(rename = "_id", default, skip_serializing_if = "Option::is_none")]
242    pub id: Option<Uuid>,
243    pub name: String,
244    #[serde(rename = "type", default)]
245    pub resource_type: String,
246    #[serde(default)]
247    pub resource_scopes: Vec<String>,
248}
249
250/// `POST /uma2/rreg/resource_set` — register a resource set (FedAuthz §2.2.1).
251///
252/// Creates the resource **and** its declared scopes. The scope set is what the
253/// permission endpoint later validates ticket requests against, so registering
254/// a resource with no scopes produces one that can never appear in a ticket —
255/// legal, and worth knowing.
256#[utoipa::path(
257    post,
258    path = "/uma2/rreg/resource_set",
259    tag = "uma",
260    request_body = ResourceSet,
261    responses(
262        (status = 201, description = "Resource set registered", body = ResourceSet),
263        (status = 401, description = "Missing or invalid PAT"),
264        (status = 403, description = "Token lacks the uma_protection scope"),
265    ),
266    security(("bearer_auth" = []))
267)]
268pub async fn register_resource_set<C: Connection + Clone>(
269    pat: ProtectionApiToken,
270    state: web::Data<AppState<C>>,
271    body: web::Json<ResourceSet>,
272) -> Result<HttpResponse, AxiamApiError> {
273    let body = body.into_inner();
274
275    let resource = state
276        .resource_repo
277        .create_uma_registered(
278            CreateResource {
279                tenant_id: pat.tenant_id,
280                name: body.name,
281                // UMA's `type` is a free-form string and AXIAM's is too, but a
282                // resource server that omits it should not end up with an
283                // empty-string type that sorts oddly in the admin UI.
284                resource_type: if body.resource_type.is_empty() {
285                    "uma_resource".to_string()
286                } else {
287                    body.resource_type
288                },
289                parent_id: None,
290                metadata: None,
291            },
292            &pat.client_id,
293        )
294        .await?;
295
296    let scopes = sync_scopes(&state, pat.tenant_id, resource.id, &body.resource_scopes).await?;
297
298    tracing::info!(
299        target: "axiam::audit",
300        event = "uma.resource_registered",
301        tenant_id = %pat.tenant_id,
302        client_id = %pat.client_id,
303        resource_id = %resource.id,
304        "resource set registered"
305    );
306
307    Ok(HttpResponse::Created().json(ResourceSet {
308        id: Some(resource.id),
309        name: resource.name,
310        resource_type: resource.resource_type,
311        resource_scopes: scopes,
312    }))
313}

315/// `GET /uma2/rreg/resource_set/{id}` — read a resource set (FedAuthz §2.2.2).
316#[utoipa::path(

327pub async fn read_resource_set<C: Connection + Clone>(
328    pat: ProtectionApiToken,

336    Ok(HttpResponse::Ok().json(ResourceSet {

338        name: resource.name,
339        resource_type: resource.resource_type,
340        resource_scopes: scopes,
341    }))
342}

344/// `PUT /uma2/rreg/resource_set/{id}` — update a resource set (FedAuthz §2.2.3).
345///
346/// The scope list is **replaced**, not merged: FedAuthz defines the update as
347/// putting the resource set's new state, and a merge would make it impossible
348/// to remove a scope. Removing a scope narrows what tickets may be requested,
349/// so a silent merge would keep an authority the resource server tried to drop.
350#[utoipa::path(
351    put,
352    path = "/uma2/rreg/resource_set/{id}",
353    tag = "uma",
354    params(("id" = Uuid, Path, description = "Resource set id")),
355    request_body = ResourceSet,
356    responses(
357        (status = 200, description = "Resource set updated", body = ResourceSet),
358        (status = 404, description = "No such resource set"),
359    ),
360    security(("bearer_auth" = []))
361)]
362pub async fn update_resource_set<C: Connection + Clone>(
363    pat: ProtectionApiToken,
364    state: web::Data<AppState<C>>,
365    path: web::Path<Uuid>,
366    body: web::Json<ResourceSet>,
367) -> Result<HttpResponse, AxiamApiError> {
368    let id = path.into_inner();
369    let body = body.into_inner();
370
371    // Read first so an unknown id answers 404 before anything is written.
372    state.resource_repo.get_by_id(pat.tenant_id, id).await?;
373
374    let resource = state
375        .resource_repo
376        .update(
377            pat.tenant_id,
378            id,
379            UpdateResource {
380                name: Some(body.name),
381                resource_type: if body.resource_type.is_empty() {
382                    None
383                } else {
384                    Some(body.resource_type)
385                },
386                ..Default::default()
387            },
388        )
389        .await?;
390
391    let scopes = sync_scopes(&state, pat.tenant_id, id, &body.resource_scopes).await?;
392
393    Ok(HttpResponse::Ok().json(ResourceSet {
394        id: Some(resource.id),
395        name: resource.name,
396        resource_type: resource.resource_type,
397        resource_scopes: scopes,
398    }))
399}

434/// `GET /uma2/rreg/resource_set` — list registered ids (FedAuthz §2.2.5).
435///
436/// Lists only the resources **this client** registered. FedAuthz says the
437/// listing is of the resource sets the PAT's owner registered, and a resource
438/// server has no business enumerating the tenant's whole resource tree through
439/// an endpoint whose only credential is a protection scope.
440#[utoipa::path(
441    get,
442    path = "/uma2/rreg/resource_set",
443    tag = "uma",
444    responses((status = 200, description = "Registered resource set ids", body = Vec<Uuid>)),
445    security(("bearer_auth" = []))
446)]
447pub async fn list_resource_sets<C: Connection + Clone>(
448    pat: ProtectionApiToken,
449    state: web::Data<AppState<C>>,
450) -> Result<HttpResponse, AxiamApiError> {
451    let page = state
452        .resource_repo
453        .list(pat.tenant_id, Pagination::default())
454        .await?;
455
456    let ids: Vec<Uuid> = page
457        .items
458        .into_iter()
459        .filter(|r| r.uma_registered_by.as_deref() == Some(pat.client_id.as_str()))
460        .map(|r| r.id)
461        .collect();
462
463    Ok(HttpResponse::Ok().json(ids))
464}

466/// Replace a resource's declared scopes with `wanted`, returning the result.
467///
468/// Adds what is missing and removes what is no longer named. Scopes that are
469/// already correct are left alone rather than deleted and recreated, because
470/// their ids are referenced by permission grants — recreating a scope would
471/// silently detach every grant that named it.
472async fn sync_scopes<C: Connection + Clone>(
473    state: &web::Data<AppState<C>>,
474    tenant_id: Uuid,
475    resource_id: Uuid,
476    wanted: &[String],
477) -> Result<Vec<String>, AxiamApiError> {
478    let existing = state
479        .scope_repo
480        .list_by_resource(tenant_id, resource_id)
481        .await?;
482
483    for scope in &existing {
484        if !wanted.contains(&scope.name) {
485            state.scope_repo.delete(tenant_id, scope.id).await?;
486        }
487    }
488
489    for name in wanted {
490        if existing.iter().any(|s| &s.name == name) {
491            continue;
492        }
493        state
494            .scope_repo
495            .create(CreateScope {
496                tenant_id,
497                resource_id,
498                name: name.clone(),
499                description: String::new(),
500            })
501            .await?;
502    }
503
504    Ok(wanted.to_vec())
505}

507async fn declared_scope_names<C: Connection + Clone>(

509    tenant_id: Uuid,

530/// Map a [`UmaError`] onto the REST error shape.
531///
532/// `AccessDenied` becomes 403 and every ticket rejection becomes 400 with one
533/// message — the service already collapses unknown, consumed, expired and
534/// wrong-client into a single `InvalidGrant`, and re-separating them here would
535/// undo that.
536fn uma_error_to_api(e: UmaError) -> AxiamApiError {
537    let inner = match e {
538        UmaError::InvalidRequest(message) | UmaError::InvalidGrant(message) => {
539            AxiamError::Validation { message }
540        }
541        UmaError::AccessDenied => AxiamError::AuthorizationDenied {
542            reason: "the requesting party is not authorized for every requested permission".into(),
543            // Deliberately not naming the pair that failed. A resource server
544            // that could tell which of several pairs was refused could probe
545            // the grant table one scope at a time.
546            action: None,
547            resource_id: None,
548        },
549        UmaError::ExpiredSubjectToken => AxiamError::Validation {
550            message: "the presented token has expired".into(),
551        },
552        UmaError::Server(m) => AxiamError::Internal(m),
553    };
554    AxiamApiError(inner)
555}
