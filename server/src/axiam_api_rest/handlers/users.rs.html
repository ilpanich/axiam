24/// Validate email format: must contain exactly one '@' with non-empty local
25/// and domain parts, and the domain must contain at least one '.'.
26fn validate_email_format(email: &str) -> Result<(), AxiamError> {
27    let mut parts = email.splitn(2, '@');
28    let local = parts.next().unwrap_or("");
29    let domain = parts.next().unwrap_or("");
30    if local.is_empty() || domain.is_empty() || !domain.contains('.') {
31        return Err(AxiamError::Validation {
32            message: "email must be a valid email address".into(),
33        });
34    }
35    Ok(())
36}

38/// Minimum password policy applied at user-create time (CQ-B26).
39///
40/// The tenant's full policy (HIBP, history) is enforced by AuthService
41/// at login.  This guard ensures obviously weak passwords are rejected
42/// at the API boundary without requiring the full policy infrastructure.
43const MINIMUM_PASSWORD_POLICY: PasswordPolicy = PasswordPolicy {
44    min_length: 8,
45    require_uppercase: false,
46    require_lowercase: false,
47    require_digits: false,
48    require_symbols: false,
49    password_history_count: 0,
50    hibp_check_enabled: false,
51};

57#[derive(Debug, Deserialize, utoipa::ToSchema)]
58pub struct CreateUserRequest {
59    pub username: String,
60    pub email: String,
61    pub password: String,
62    pub metadata: Option<serde_json::Value>,
63    /// OPAQUE registration record for `password`, built client-side.
64    ///
65    /// The server cannot derive this — the envelope is sealed under a key only
66    /// the client can compute — so it has to arrive with the request or not at
67    /// all. Required under `opaque_mode = required`, where a user created
68    /// without one could never authenticate.
69    ///
70    /// Obtained by calling `POST /api/v1/auth/opaque/register/start` first and
71    /// carrying its `opaque_session` through, which is the one structural
72    /// difference from the SRP verifier this replaces: a verifier could be
73    /// computed offline from a self-chosen salt, whereas a record needs the
74    /// server's OPRF evaluation.
75    #[serde(default, skip_serializing_if = "Option::is_none")]
76    pub opaque: Option<axiam_core::models::opaque::OpaqueEnrollment>,
77}

87/// Public-safe user representation (no password_hash, no mfa_secret).
88#[derive(Debug, Serialize, utoipa::ToSchema)]
89pub struct UserResponse {
90    pub id: Uuid,
91    pub tenant_id: Uuid,
92    pub username: String,
93    pub email: String,
94    pub status: UserStatus,
95    pub mfa_enabled: bool,
96    /// True when the user's email address has been verified
97    /// (`email_verified_at` is set). Exposed for the admin UI / profile.
98    pub email_verified: bool,
99    pub metadata: serde_json::Value,
100    pub created_at: DateTime<Utc>,
101    pub updated_at: DateTime<Utc>,
102    /// Number of consecutive failed login attempts.
103    pub failed_login_attempts: u32,
104    /// Timestamp until which the account is locked, if any.
105    pub locked_until: Option<DateTime<Utc>>,
106    /// Whether the account is currently locked (locked_until is in the future).
107    pub is_locked: bool,
108}
109
110impl From<User> for UserResponse {
111    fn from(u: User) -> Self {
112        let is_locked = u
113            .locked_until
114            .map(|t| t > chrono::Utc::now())
115            .unwrap_or(false);
116        Self {
117            id: u.id,
118            tenant_id: u.tenant_id,
119            username: u.username,
120            email: u.email,
121            status: u.status,
122            mfa_enabled: u.mfa_enabled,
123            email_verified: u.email_verified_at.is_some(),
124            metadata: u.metadata,
125            created_at: u.created_at,
126            updated_at: u.updated_at,
127            failed_login_attempts: u.failed_login_attempts,
128            locked_until: u.locked_until,
129            is_locked,
130        }
131    }
132}

138/// `POST /api/v1/users`
139///
140/// Creates a user and atomically records a `terms_of_service` consent row
141/// (REQ-8 / Art. 7 proof of consent).  IP address and User-Agent are
142/// captured for the consent record.
143#[utoipa::path(
144    post,
145    path = "/api/v1/users",
146    tag = "users",
147    request_body = CreateUserRequest,
148    responses(
149        (status = 201, description = "User created", body = UserResponse),
150    ),
151    security(("bearer" = []))
152)]
153pub async fn create<C: Connection + Clone>(
154    http_req: HttpRequest,
155    user: AuthenticatedUser,
156    authz: AuthzData,
157    state: web::Data<AppState<C>>,
158    body: web::Json<CreateUserRequest>,
159) -> Result<HttpResponse, AxiamApiError> {
160    RequirePermission::new("users:create", Uuid::nil())
161        .check(&user, authz.get_ref().as_ref())
162        .await?;
163    let mut req = body.into_inner();
164
165    // CQ-B26: validate email format and password complexity before insert.
166    validate_email_format(&req.email)?;
167    let pw_violations = check_complexity(&req.password, &MINIMUM_PASSWORD_POLICY);
168    if !pw_violations.is_empty() {
169        let details: Vec<String> = pw_violations.iter().map(|v| v.to_string()).collect();
170        return Err(AxiamApiError(AxiamError::PasswordPolicy {
171            message: details.join("; "),
172        }));
173    }
174
175    use axiam_core::repository::{SettingsRepository as _, TenantRepository as _};
176
177    // Resolve the tenant's OPAQUE policy and reject a malformed record before
178    // the user row is written. Validating afterwards would leave a created
179    // user with no record under `required` — an account that exists and cannot
180    // log in — which is exactly the state this check prevents.
181    let opaque_enrollment = req.opaque.take();
182    let tenant = state.tenant_repo.get_by_id(user.tenant_id).await?;
183    let opaque_policy = state
184        .settings_repo
185        .get_effective_settings(tenant.organization_id, user.tenant_id)
186        .await?
187        .opaque;
188    let enrolled = crate::handlers::opaque_enrollment::validate_enrollment(
189        &state,
190        &opaque_policy,
191        user.tenant_id,
192        opaque_enrollment.as_ref(),
193    )?;
194
195    let mut input = CreateUser {
196        tenant_id: user.tenant_id,
197        username: req.username,
198        email: req.email,
199        password: req.password,
200        metadata: req.metadata,
201    };
202
203    // X1 `user.pre_create` — after the caller's permission and the format
204    // checks, before the write. Placed after validation so a reactor is never
205    // asked about an email that is not an email; placed before the transaction
206    // so a veto costs nothing to roll back and a normalization is what gets
207    // stored rather than a second write.
208    crate::reactor_hooks::user_pre_create(&state.events.reactor_gate, user.tenant_id, &mut input)
209        .await?;
210
211    // Capture IP and User-Agent for the Art. 7 proof-of-consent record.
212    let ip_address = client_ip(&http_req);
213    let user_agent = user_agent(&http_req);
214
215    // The user and its terms_of_service consent are written in one transaction:
216    // a consent failure rolls back user creation, so a user can never exist
217    // without proof-of-consent (REQ-8, threat T-5-consent-gap).
218    let created = state
219        .user_repo
220        .create_with_consent(input, "terms_of_service", "current", ip_address, user_agent)
221        .await?;
222
223    // Note that nothing here depends on `created.username`. The SRP version
224    // had to bind the verifier to the post-reactor-normalization username, and
225    // getting that wrong produced a verifier no login could ever satisfy. An
226    // OPAQUE record binds to a server-chosen credential identifier instead, so
227    // a `user.pre_create` reactor renaming the user cannot break it.
228    crate::handlers::opaque_enrollment::store_credential(
229        &state,
230        created.tenant_id,
231        created.id,
232        enrolled,
233    )
234    .await?;
235
236    // CQ-B22: dispatch the domain event to subscribed webhooks (best-effort;
237    // never blocks or fails the response).
238    state
239        .emit_webhook(
240            created.tenant_id,
241            "user.created",
242            serde_json::json!({ "id": created.id, "username": created.username }),
243        )
244        .await;
245
246    Ok(HttpResponse::Created().json(UserResponse::from(created)))
247}

310/// `PUT /api/v1/users/{user_id}`
311#[utoipa::path(
312    put,
313    path = "/api/v1/users/{user_id}",
314    tag = "users",
315    params(("user_id" = Uuid, Path, description = "User ID")),
316    request_body = UpdateUserRequest,
317    responses(
318        (status = 200, description = "User updated", body = UserResponse),
319        (status = 404, description = "User not found"),
320    ),
321    security(("bearer" = []))
322)]
323pub async fn update<C: Connection + Clone>(
324    user: AuthenticatedUser,
325    authz: AuthzData,
326    state: web::Data<AppState<C>>,
327    path: web::Path<Uuid>,
328    body: web::Json<UpdateUserRequest>,
329) -> Result<HttpResponse, AxiamApiError> {
330    let target_id = path.into_inner();
331    let self_update = is_own_resource(&user, target_id);
332    if !self_update {
333        RequirePermission::new("users:update", Uuid::nil())
334            .check(&user, authz.get_ref().as_ref())
335            .await?;
336    }
337    let req = body.into_inner();
338
339    // SEC-050: Self-update guards:
340    // 1. Strip `status` — a user cannot change their own account status.
341    // 2. Gate email change — if the new email differs from the stored one,
342    //    mark the address unverified so it cannot be used until re-verified.
343    let effective_status = if self_update { None } else { req.status };
344
345    // Load the current email to detect email changes on self-update.
346    let email_changed_for_self = if self_update {
347        if let Some(ref new_email) = req.email {
348            let current = state.user_repo.get_by_id(user.tenant_id, target_id).await?;
349            new_email != &current.email
350        } else {
351            false
352        }
353    } else {
354        false
355    };
356
357    let mut input = UpdateUser {
358        username: req.username,
359        email: req.email,
360        status: effective_status,
361        metadata: req.metadata,
362        // SEC-050: on email change by self, clear email_verified_at so the
363        // new address must be re-verified before it is considered valid.
364        email_verified_at: if email_changed_for_self {
365            Some(None) // explicitly set to None (requires the Option<Option<T>> pattern)
366        } else {
367            None // no change
368        },
369        ..Default::default()
370    };
371
372    // X1 `user.pre_update` — after the SEC-050 self-update guards, so a
373    // reactor cannot re-open what they closed. In particular it runs after
374    // `effective_status` has already stripped a self-service status change, and
375    // `status` is not in the event's allow-list, so neither the caller nor the
376    // reactor can set it here.
377    crate::reactor_hooks::user_pre_update(
378        &state.events.reactor_gate,
379        user.tenant_id,
380        target_id,
381        &mut input,
382    )
383    .await?;
384
385    let updated = state
386        .user_repo
387        .update(user.tenant_id, target_id, input)
388        .await?;
389
390    // A rename used to need work here: SRP derived `x` over
391    // `username ":" password`, so this handler had to read the pre-update
392    // username, detect a change, and drop the verifier — otherwise the user
393    // saw "invalid credentials" for an entirely correct password. OPAQUE binds
394    // to a random server-chosen credential identifier with no relationship to
395    // any name, so a rename is free and both the extra read and the deletion
396    // are gone rather than ported.
397
398    // D7 (REVOCATION — security critical): an update can NARROW access, most
399    // directly by setting `status` to a non-Active value. Nothing on the
400    // session-authenticated request path re-reads user status (`is_session_active`
401    // checks only the session row's `expires_at`), so a cached `Allow` for this
402    // subject would otherwise outlive the change by up to the cache TTL on every
403    // replica. Invalidate unconditionally rather than only on a status change:
404    // the flush is a single targeted per-subject removal, and gating it on which
405    // field moved is exactly the kind of reasoning that goes stale as the update
406    // surface grows.
407    authz
408        .get_ref()
409        .as_ref()
410        .invalidate_subject(user.tenant_id, target_id)
411        .await?;
412
413    // CQ-B22: dispatch the domain event to subscribed webhooks (best-effort).
414    state
415        .emit_webhook(
416            updated.tenant_id,
417            "user.updated",
418            serde_json::json!({ "id": updated.id, "username": updated.username }),
419        )
420        .await;
421
422    Ok(HttpResponse::Ok().json(UserResponse::from(updated)))
423}

425/// `DELETE /api/v1/users/{user_id}`
426#[utoipa::path(
427    delete,
428    path = "/api/v1/users/{user_id}",
429    tag = "users",
430    params(("user_id" = Uuid, Path, description = "User ID")),
431    responses(
432        (status = 204, description = "User deleted"),
433        (status = 404, description = "User not found"),
434    ),
435    security(("bearer" = []))
436)]
437pub async fn delete<C: Connection + Clone>(
438    user: AuthenticatedUser,
439    authz: AuthzData,
440    state: web::Data<AppState<C>>,
441    path: web::Path<Uuid>,
442) -> Result<HttpResponse, AxiamApiError> {
443    RequirePermission::new("users:delete", Uuid::nil())
444        .check(&user, authz.get_ref().as_ref())
445        .await?;
446    let target_id = path.into_inner();
447    state.user_repo.delete(user.tenant_id, target_id).await?;
448
449    // A SCIM provisioning token authenticates *as* this user, so deleting them
450    // without clearing their tokens would leave a year-long credential naming a
451    // principal that no longer exists. The resolver would refuse it on the
452    // bound-user check, but relying on that would make the credential's
453    // liveness depend on a lookup rather than on the row being gone.
454    // Best-effort: a delete that already succeeded must not be reported as a
455    // failure because the cleanup did not.
456    if let Err(e) = state
457        .scim_token_repo
458        .delete_for_user(user.tenant_id, target_id)
459        .await
460    {
461        tracing::warn!(
462            target: "axiam::audit",
463            event = "scim.token_cleanup_failed",
464            tenant_id = %user.tenant_id,
465            user_id = %target_id,
466            error = %e,
467            "could not remove SCIM provisioning tokens for a deleted user"
468        );
469    }
470
471    // D7 (REVOCATION — security critical): the subject no longer exists, so every
472    // cached `Allow` naming it is now a decision about a deleted principal. Flush
473    // this subject before returning; without it the deleted user's cached allows
474    // survive until the TTL expires.
475    authz
476        .get_ref()
477        .as_ref()
478        .invalidate_subject(user.tenant_id, target_id)
479        .await?;
480
481    // CQ-B22: dispatch the domain event to subscribed webhooks (best-effort).
482    state
483        .emit_webhook(
484            user.tenant_id,
485            "user.deleted",
486            serde_json::json!({ "id": target_id }),
487        )
488        .await;
489
490    Ok(HttpResponse::NoContent().finish())
491}

493/// `POST /api/v1/users/{user_id}/unlock`
494///
495/// Resets a locked user account: clears `locked_until`, resets
496/// `failed_login_attempts` to 0, and sets status back to `Active`.
497#[utoipa::path(
498    post,
499    path = "/api/v1/users/{user_id}/unlock",
500    tag = "users",
501    params(("user_id" = Uuid, Path, description = "User ID")),
502    responses(
503        (status = 200, description = "User unlocked", body = UserResponse),
504        (status = 404, description = "User not found"),
505    ),
506    security(("bearer" = []))
507)]
508pub async fn unlock<C: Connection + Clone>(
509    auth_user: AuthenticatedUser,
510    authz: AuthzData,
511    state: web::Data<AppState<C>>,
512    path: web::Path<Uuid>,
513) -> Result<HttpResponse, AxiamApiError> {
514    RequirePermission::new("users:admin", Uuid::nil())
515        .check(&auth_user, authz.get_ref().as_ref())
516        .await?;
517    let user_id = path.into_inner();
518    let tenant_id = auth_user.tenant_id;
519
520    let update = UpdateUser {
521        failed_login_attempts: Some(0),
522        locked_until: Some(None),
523        last_failed_login_at: Some(None),
524        status: Some(UserStatus::Active),
525        ..Default::default()
526    };
527
528    let user = state.user_repo.update(tenant_id, user_id, update).await?;
529    Ok(HttpResponse::Ok().json(UserResponse::from(user)))
530}

544    async fn setup_db() -> Surreal<surrealdb::engine::local::Db> {
545        let db = Surreal::new::<Mem>(()).await.unwrap();
546        db.use_ns("test").use_db("test").await.unwrap();
547        axiam_db::run_migrations(&db).await.unwrap();

549    }
550
551    /// The atomic registration path commits exactly one terms_of_service
552    /// consent row alongside the new user (threat T-5-consent-gap).
553    #[tokio::test]
554    async fn registration_creates_consent_row() {
555        let db = setup_db().await;
556        let user_repo = SurrealUserRepository::new(db.clone());
557        let consent_repo = SurrealConsentRepository::new(db.clone());
558        let tenant_id = Uuid::new_v4();
559
560        // Atomic user + consent creation — the handler's real path.
561        let user = user_repo
562            .create_with_consent(
563                axiam_core::models::user::CreateUser {
564                    tenant_id,
565                    username: "testuser".to_string(),
566                    email: "test@example.com".to_string(),
567                    password: "Test1234!".to_string(),
568                    metadata: None,
569                },
570                "terms_of_service",
571                "current",
572                Some("127.0.0.1".to_string()),
573                Some("test-agent/1.0".to_string()),
574            )
575            .await
576            .unwrap();
577
578        // The consent row must be committed with user_id == the new user's id.
579        let consents = consent_repo.list_by_user(tenant_id, user.id).await.unwrap();
580        assert_eq!(consents.len(), 1, "expected exactly one consent row");
581        assert_eq!(consents[0].consent_type, "terms_of_service");
582        assert_eq!(consents[0].version, "current");
583        assert_eq!(consents[0].user_id, user.id);
584        assert_eq!(consents[0].tenant_id, tenant_id);
585        assert_eq!(consents[0].ip_address.as_deref(), Some("127.0.0.1"));
586        assert_eq!(consents[0].user_agent.as_deref(), Some("test-agent/1.0"));
587    }
