24/// Validate email format: must contain exactly one '@' with non-empty local
25/// and domain parts, and the domain must contain at least one '.'.
26fn validate_email_format(email: &str) -> Result<(), AxiamError> {
27    let mut parts = email.splitn(2, '@');
28    let local = parts.next().unwrap_or("");
29    let domain = parts.next().unwrap_or("");
30    if local.is_empty() || domain.is_empty() || !domain.contains('.') {
31        return Err(AxiamError::Validation {
32            message: "email must be a valid email address".into(),
33        });
34    }
35    Ok(())
36}

38/// Minimum password policy applied at user-create time (CQ-B26).
39///
40/// The tenant's full policy (HIBP, history) is enforced by AuthService
41/// at login.  This guard ensures obviously weak passwords are rejected
42/// at the API boundary without requiring the full policy infrastructure.
43const MINIMUM_PASSWORD_POLICY: PasswordPolicy = PasswordPolicy {
44    min_length: 8,
45    require_uppercase: false,
46    require_lowercase: false,
47    require_digits: false,
48    require_symbols: false,
49    password_history_count: 0,
50    hibp_check_enabled: false,
51};

57#[derive(Debug, Deserialize, utoipa::ToSchema)]
58pub struct CreateUserRequest {
59    pub username: String,
60    pub email: String,
61    pub password: String,
62    pub metadata: Option<serde_json::Value>,
63    /// OPAQUE registration record for `password`, built client-side.
64    ///
65    /// The server cannot derive this — the envelope is sealed under a key only
66    /// the client can compute — so it has to arrive with the request or not at
67    /// all. Required under `opaque_mode = required`, where a user created
68    /// without one could never authenticate.
69    ///
70    /// Obtained by calling `POST /api/v1/auth/opaque/register/start` first and
71    /// carrying its `opaque_session` through, which is the one structural
72    /// difference from the SRP verifier this replaces: a verifier could be
73    /// computed offline from a self-chosen salt, whereas a record needs the
74    /// server's OPRF evaluation.
75    #[serde(default, skip_serializing_if = "Option::is_none")]
76    pub opaque: Option<axiam_core::models::opaque::OpaqueEnrollment>,
77}

87/// Public-safe user representation (no password_hash, no mfa_secret).
88#[derive(Debug, Clone, Serialize, utoipa::ToSchema)]
89pub struct UserResponse {
90    pub id: Uuid,
91    pub tenant_id: Uuid,
92    pub username: String,
93    pub email: String,
94    pub status: UserStatus,
95    pub mfa_enabled: bool,
96    /// True when the user's email address has been verified
97    /// (`email_verified_at` is set). Exposed for the admin UI / profile.
98    pub email_verified: bool,
99    pub metadata: serde_json::Value,
100    pub created_at: DateTime<Utc>,
101    pub updated_at: DateTime<Utc>,
102    /// Number of consecutive failed login attempts.
103    pub failed_login_attempts: u32,
104    /// Timestamp until which the account is locked, if any.
105    pub locked_until: Option<DateTime<Utc>>,
106    /// Whether the account is currently locked (locked_until is in the future).
107    pub is_locked: bool,
108}
109
110impl From<User> for UserResponse {
111    fn from(u: User) -> Self {
112        let is_locked = u
113            .locked_until
114            .map(|t| t > chrono::Utc::now())
115            .unwrap_or(false);
116        Self {
117            id: u.id,
118            tenant_id: u.tenant_id,
119            username: u.username,
120            email: u.email,
121            status: u.status,
122            mfa_enabled: u.mfa_enabled,
123            email_verified: u.email_verified_at.is_some(),
124            metadata: u.metadata,
125            created_at: u.created_at,
126            updated_at: u.updated_at,
127            failed_login_attempts: u.failed_login_attempts,
128            locked_until: u.locked_until,
129            is_locked,
130        }
131    }
132}

138/// `POST /api/v1/users`
139///
140/// Creates a user and atomically records a `terms_of_service` consent row
141/// (REQ-8 / Art. 7 proof of consent).  IP address and User-Agent are
142/// captured for the consent record.
143#[utoipa::path(
144    post,
145    path = "/api/v1/users",
146    tag = "users",
147    request_body = CreateUserRequest,
148    responses(
149        (status = 201, description = "User created", body = UserResponse),
150    ),
151    security(("bearer" = []))
152)]
153pub async fn create<C: Connection + Clone>(
154    http_req: HttpRequest,
155    user: AuthenticatedUser,
156    authz: AuthzData,
157    state: web::Data<AppState<C>>,
158    body: web::Json<CreateUserRequest>,
159) -> Result<HttpResponse, AxiamApiError> {
160    RequirePermission::new("users:create", Uuid::nil())
161        .check(&user, authz.get_ref().as_ref())
162        .await?;
163    let mut req = body.into_inner();
164
165    // CQ-B26: validate email format and password complexity before insert.
166    validate_email_format(&req.email)?;
167    let pw_violations = check_complexity(&req.password, &MINIMUM_PASSWORD_POLICY);
168    if !pw_violations.is_empty() {
169        let details: Vec<String> = pw_violations.iter().map(|v| v.to_string()).collect();
170        return Err(AxiamApiError(AxiamError::PasswordPolicy {
171            message: details.join("; "),
172        }));
173    }
174
175    use axiam_core::repository::{SettingsRepository as _, TenantRepository as _};
176
177    // Resolve the tenant's OPAQUE policy and reject a malformed record before
178    // the user row is written. Validating afterwards would leave a created
179    // user with no record under `required` — an account that exists and cannot
180    // log in — which is exactly the state this check prevents.
181    let opaque_enrollment = req.opaque.take();
182    let tenant = state.tenant_repo.get_by_id(user.tenant_id).await?;
183    let opaque_policy = state
184        .settings_repo
185        .get_effective_settings(tenant.organization_id, user.tenant_id)
186        .await?
187        .opaque;
188    let enrolled = crate::handlers::opaque_enrollment::validate_enrollment(
189        &state,
190        &opaque_policy,
191        user.tenant_id,
192        opaque_enrollment.as_ref(),
193    )?;
194
195    let mut input = CreateUser {
196        tenant_id: user.tenant_id,
197        username: req.username,
198        email: req.email,
199        password: req.password,
200        metadata: req.metadata,
201    };
202
203    // X1 `user.pre_create` — after the caller's permission and the format
204    // checks, before the write. Placed after validation so a reactor is never
205    // asked about an email that is not an email; placed before the transaction
206    // so a veto costs nothing to roll back and a normalization is what gets
207    // stored rather than a second write.
208    crate::reactor_hooks::user_pre_create(&state.events.reactor_gate, user.tenant_id, &mut input)
209        .await?;
210
211    // Capture IP and User-Agent for the Art. 7 proof-of-consent record.
212    let ip_address = client_ip(&http_req);
213    let user_agent = user_agent(&http_req);
214
215    // The user and its terms_of_service consent are written in one transaction:
216    // a consent failure rolls back user creation, so a user can never exist
217    // without proof-of-consent (REQ-8, threat T-5-consent-gap).
218    let created = state
219        .user_repo
220        .create_with_consent(input, "terms_of_service", "current", ip_address, user_agent)
221        .await?;
222
223    // Note that nothing here depends on `created.username`. The SRP version
224    // had to bind the verifier to the post-reactor-normalization username, and
225    // getting that wrong produced a verifier no login could ever satisfy. An
226    // OPAQUE record binds to a server-chosen credential identifier instead, so
227    // a `user.pre_create` reactor renaming the user cannot break it.
228    crate::handlers::opaque_enrollment::store_credential(
229        &state,
230        created.tenant_id,
231        created.id,
232        enrolled,
233    )
234    .await?;
235
236    // The account is written as `PendingVerification`, so without this the user
237    // has no way to activate it: nothing told them it exists, and the only path
238    // to a token was an administrator knowing to call
239    // `POST /auth/resend-verification` by hand. Creating an account and sending
240    // its activation link are one act from the operator's point of view, and
241    // this is where they are joined.
242    //
243    // Best-effort, deliberately: the user row and its consent record are already
244    // committed. A mail queue outage must not turn a successful creation into a
245    // 500 the caller retries — that would try to create the user a second time
246    // and fail on the unique index. The account stays activatable through
247    // `resend-verification`, which mints a fresh token.
248    crate::handlers::email_verification::enqueue_verification_email(
249        &state,
250        created.tenant_id,
251        created.id,
252        &created.email,
253    )
254    .await;
255
256    // CQ-B22: dispatch the domain event to subscribed webhooks (best-effort;
257    // never blocks or fails the response).
258    state
259        .emit_webhook(
260            created.tenant_id,
261            "user.created",
262            serde_json::json!({ "id": created.id, "username": created.username }),
263        )
264        .await;
265
266    Ok(HttpResponse::Created().json(UserResponse::from(created)))
267}

302/// `GET /api/v1/users/{user_id}`
303#[utoipa::path(
304    get,
305    path = "/api/v1/users/{user_id}",
306    tag = "users",
307    params(("user_id" = Uuid, Path, description = "User ID")),
308    responses(
309        (status = 200, description = "User found", body = UserResponse),
310        (status = 404, description = "User not found"),
311    ),
312    security(("bearer" = []))
313)]
314pub async fn get<C: Connection + Clone>(
315    user: AuthenticatedUser,
316    authz: AuthzData,
317    state: web::Data<AppState<C>>,
318    path: web::Path<Uuid>,
319) -> Result<HttpResponse, AxiamApiError> {
320    let target_id = path.into_inner();
321    if !is_own_resource(&user, target_id) {
322        RequirePermission::new("users:get", Uuid::nil())
323            .check(&user, authz.get_ref().as_ref())
324            .await?;
325    }
326    // The caller's own record lives in the tenant the caller belongs to, which
327    // is not the tenant an organization-level principal has selected. Reading
328    // the acting tenant here made "open my profile" a 404 for exactly that
329    // principal — see [`user_scope_tenant`].
330    let scope = user_scope_tenant(&user, target_id);
331    let target = state.user_repo.get_by_id(scope, target_id).await?;
332    Ok(HttpResponse::Ok().json(UserResponse::from(target)))
333}

336#[utoipa::path(
337    put,
338    path = "/api/v1/users/{user_id}",

340    params(("user_id" = Uuid, Path, description = "User ID")),
341    request_body = UpdateUserRequest,
342    responses(
343        (status = 200, description = "User updated", body = UserResponse),
344        (status = 404, description = "User not found"),
345    ),
346    security(("bearer" = []))
347)]
348pub async fn update<C: Connection + Clone>(
349    user: AuthenticatedUser,
350    authz: AuthzData,
351    state: web::Data<AppState<C>>,
352    path: web::Path<Uuid>,
353    body: web::Json<UpdateUserRequest>,
354) -> Result<HttpResponse, AxiamApiError> {
355    let target_id = path.into_inner();
356    let self_update = is_own_resource(&user, target_id);
357    if !self_update {
358        RequirePermission::new("users:update", Uuid::nil())
359            .check(&user, authz.get_ref().as_ref())
360            .await?;
361    }
362    // Same rule as `get` above, and it matters more here: writing a
363    // self-update into the *selected* tenant would not find the row to update
364    // at all. See [`user_scope_tenant`].
365    let scope = user_scope_tenant(&user, target_id);
366    let req = body.into_inner();
367
368    // SEC-050: Self-update guards:
369    // 1. Strip `status` — a user cannot change their own account status.
370    // 2. Gate email change — if the new email differs from the stored one,
371    //    mark the address unverified so it cannot be used until re-verified.
372    let effective_status = if self_update { None } else { req.status };
373
374    // Load the current email to detect email changes on self-update.
375    let email_changed_for_self = if self_update {
376        if let Some(ref new_email) = req.email {
377            let current = state.user_repo.get_by_id(scope, target_id).await?;
378            new_email != &current.email
379        } else {
380            false
381        }
382    } else {
383        false
384    };
385
386    let mut input = UpdateUser {
387        username: req.username,
388        email: req.email,
389        status: effective_status,
390        metadata: req.metadata,
391        // SEC-050: on email change by self, clear email_verified_at so the
392        // new address must be re-verified before it is considered valid.
393        email_verified_at: if email_changed_for_self {
394            Some(None) // explicitly set to None (requires the Option<Option<T>> pattern)
395        } else {
396            None // no change
397        },
398        ..Default::default()
399    };
400
401    // X1 `user.pre_update` — after the SEC-050 self-update guards, so a
402    // reactor cannot re-open what they closed. In particular it runs after
403    // `effective_status` has already stripped a self-service status change, and
404    // `status` is not in the event's allow-list, so neither the caller nor the
405    // reactor can set it here.
406    crate::reactor_hooks::user_pre_update(&state.events.reactor_gate, scope, target_id, &mut input)
407        .await?;
408
409    let updated = state.user_repo.update(scope, target_id, input).await?;
410
411    // A rename used to need work here: SRP derived `x` over
412    // `username ":" password`, so this handler had to read the pre-update
413    // username, detect a change, and drop the verifier — otherwise the user
414    // saw "invalid credentials" for an entirely correct password. OPAQUE binds
415    // to a random server-chosen credential identifier with no relationship to
416    // any name, so a rename is free and both the extra read and the deletion
417    // are gone rather than ported.
418
419    // D7 (REVOCATION — security critical): an update can NARROW access, most
420    // directly by setting `status` to a non-Active value. Nothing on the
421    // session-authenticated request path re-reads user status (`is_session_active`
422    // checks only the session row's `expires_at`), so a cached `Allow` for this
423    // subject would otherwise outlive the change by up to the cache TTL on every
424    // replica. Invalidate unconditionally rather than only on a status change:
425    // the flush is a single targeted per-subject removal, and gating it on which
426    // field moved is exactly the kind of reasoning that goes stale as the update
427    // surface grows.
428    authz
429        .get_ref()
430        .as_ref()
431        .invalidate_subject(scope, target_id)
432        .await?;
433
434    // CQ-B22: dispatch the domain event to subscribed webhooks (best-effort).
435    state
436        .emit_webhook(
437            updated.tenant_id,
438            "user.updated",
439            serde_json::json!({ "id": updated.id, "username": updated.username }),
440        )
441        .await;
442
443    Ok(HttpResponse::Ok().json(UserResponse::from(updated)))
444}

446/// `DELETE /api/v1/users/{user_id}`
447#[utoipa::path(
448    delete,
449    path = "/api/v1/users/{user_id}",
450    tag = "users",
451    params(("user_id" = Uuid, Path, description = "User ID")),
452    responses(
453        (status = 204, description = "User deleted"),
454        (status = 404, description = "User not found"),
455    ),
456    security(("bearer" = []))
457)]
458pub async fn delete<C: Connection + Clone>(
459    user: AuthenticatedUser,
460    authz: AuthzData,
461    state: web::Data<AppState<C>>,
462    path: web::Path<Uuid>,
463) -> Result<HttpResponse, AxiamApiError> {
464    RequirePermission::new("users:delete", Uuid::nil())
465        .check(&user, authz.get_ref().as_ref())
466        .await?;
467    let target_id = path.into_inner();
468
469    // Kill every live session BEFORE the row is tombstoned.
470    //
471    // `check_user_status` runs at login, and only at login: an already-issued
472    // access token keeps working on its own claims until it expires. Deleting
473    // the row without this left a "deleted" administrator holding a valid
474    // session for the rest of the token's lifetime — the single most important
475    // thing a delete is supposed to stop.
476    //
477    // Before the tombstone, because a failure here must abort the delete rather
478    // than leave a removed account with live sessions. `revoke_all_sessions`
479    // covers session-flow refresh tokens and OAuth2 refresh tokens alike.
480    state
481        .auth_service
482        .revoke_all_sessions(user.tenant_id, target_id)
483        .await?;
484
485    state.user_repo.delete(user.tenant_id, target_id).await?;
486
487    // Erase the personal data held OUTSIDE the user row.
488    //
489    // The row itself is anonymised by `user_repo.delete` above, but a user's
490    // personal data is not all in one table. A passkey is a registered
491    // authenticator bound to a person; a federation link is their identifier at
492    // an external provider; password history is a chain of hashes of secrets
493    // they chose. Leaving any of those behind means the account is "deleted"
494    // and the data is not, which is the distinction erasure is entirely about.
495    //
496    // These are the same tables `axiam-server`'s Art. 17 purge clears, and
497    // deliberately so: an administrator's Delete and a data subject's erasure
498    // request must not leave different residue.
499    //
500    // Best-effort with a warning each, like the graph cleanup below: the account
501    // can no longer authenticate by this point, so a row that outlives it grants
502    // nothing and is a cleanup task rather than an open door. Failing the
503    // response instead would invite a retry that 404s on the already-erased row.
504    {
505        use axiam_core::repository::{
506            FederationLinkRepository as _, PasswordHistoryRepository as _,
507            WebauthnCredentialRepository as _,
508        };
509
510        match state
511            .webauthn
512            .webauthn_credential_repo
513            .list_by_user(user.tenant_id, target_id)
514            .await
515        {
516            Ok(creds) => {
517                for cred in creds {
518                    if let Err(e) = state
519                        .webauthn
520                        .webauthn_credential_repo
521                        .delete(user.tenant_id, cred.id)
522                        .await
523                    {
524                        tracing::warn!(
525                            error = %e, user_id = %target_id, credential_id = %cred.id,
526                            "could not delete a deleted user's passkey"
527                        );
528                    }
529                }
530            }
531            Err(e) => tracing::warn!(
532                error = %e, user_id = %target_id,
533                "could not list a deleted user's passkeys"
534            ),
535        }
536
537        match state
538            .federation
539            .federation_link_repo
540            .get_by_user_id(user.tenant_id, target_id)
541            .await
542        {
543            Ok(links) => {
544                for link in links {
545                    if let Err(e) = state
546                        .federation
547                        .federation_link_repo
548                        .delete(user.tenant_id, link.id)
549                        .await
550                    {
551                        tracing::warn!(
552                            error = %e, user_id = %target_id, link_id = %link.id,
553                            "could not delete a deleted user's federation link"
554                        );
555                    }
556                }
557            }
558            Err(e) => tracing::warn!(
559                error = %e, user_id = %target_id,
560                "could not list a deleted user's federation links"
561            ),
562        }
563
564        // `keep_count = 0` prunes the whole chain.
565        if let Err(e) = state
566            .password_history_repo
567            .prune(user.tenant_id, target_id, 0)
568            .await
569        {
570            tracing::warn!(
571                error = %e, user_id = %target_id,
572                "could not prune a deleted user's password history"
573            );
574        }
575    }
576
577    // Strip the authorization graph: group memberships first, then the direct
578    // role assignments that remain once inherited ones have gone with the
579    // membership. A tombstone still reachable through `member_of` would keep
580    // appearing in group-member listings and, worse, keep contributing live
581    // authorization paths.
582    //
583    // Best-effort, unlike the session revocation above: the account can no
584    // longer authenticate at all by this point, so a leftover edge grants
585    // nothing. Reporting a completed delete as a failure would invite a retry
586    // that 404s on the already-tombstoned row.
587    use axiam_core::repository::{GroupRepository as _, RoleRepository as _};
588    match state
589        .group_repo
590        .get_user_groups(user.tenant_id, target_id)
591        .await
592    {
593        Ok(groups) => {
594            for group in groups {
595                if let Err(e) = state
596                    .group_repo
597                    .remove_member(user.tenant_id, target_id, group.id)
598                    .await
599                {
600                    tracing::warn!(
601                        error = %e,
602                        tenant_id = %user.tenant_id,
603                        user_id = %target_id,
604                        group_id = %group.id,
605                        "could not remove a deleted user from a group"
606                    );
607                }
608            }
609        }
610        Err(e) => tracing::warn!(
611            error = %e,
612            user_id = %target_id,
613            "could not list the deleted user's groups"
614        ),
615    }
616    match state
617        .role_repo
618        .get_user_role_assignments(user.tenant_id, target_id)
619        .await
620    {
621        Ok(assignments) => {
622            for a in assignments {
623                if let Err(e) = state
624                    .role_repo
625                    .unassign_from_user(user.tenant_id, target_id, a.role.id, a.resource_id)
626                    .await
627                {
628                    tracing::warn!(
629                        error = %e,
630                        tenant_id = %user.tenant_id,
631                        user_id = %target_id,
632                        role_id = %a.role.id,
633                        "could not unassign a role from a deleted user"
634                    );
635                }
636            }
637        }
638        Err(e) => tracing::warn!(
639            error = %e,
640            user_id = %target_id,
641            "could not list the deleted user's role assignments"
642        ),
643    }
644
645    // A SCIM provisioning token authenticates *as* this user, so deleting them
646    // without clearing their tokens would leave a year-long credential naming a
647    // principal that no longer exists. The resolver would refuse it on the
648    // bound-user check, but relying on that would make the credential's
649    // liveness depend on a lookup rather than on the row being gone.
650    // Best-effort: a delete that already succeeded must not be reported as a
651    // failure because the cleanup did not.
652    if let Err(e) = state
653        .scim_token_repo
654        .delete_for_user(user.tenant_id, target_id)
655        .await
656    {
657        tracing::warn!(
658            target: "axiam::audit",
659            event = "scim.token_cleanup_failed",
660            tenant_id = %user.tenant_id,
661            user_id = %target_id,
662            error = %e,
663            "could not remove SCIM provisioning tokens for a deleted user"
664        );
665    }
666
667    // D7 (REVOCATION — security critical): the subject no longer exists, so every
668    // cached `Allow` naming it is now a decision about a deleted principal. Flush
669    // this subject before returning; without it the deleted user's cached allows
670    // survive until the TTL expires.
671    authz
672        .get_ref()
673        .as_ref()
674        .invalidate_subject(user.tenant_id, target_id)
675        .await?;
676
677    // CQ-B22: dispatch the domain event to subscribed webhooks (best-effort).
678    state
679        .emit_webhook(
680            user.tenant_id,
681            "user.deleted",
682            serde_json::json!({ "id": target_id }),
683        )
684        .await;
685
686    Ok(HttpResponse::NoContent().finish())
687}

689/// `POST /api/v1/users/{user_id}/unlock`
690///
691/// Resets a locked user account: clears `locked_until`, resets
692/// `failed_login_attempts` to 0, and sets status back to `Active`.
693#[utoipa::path(
694    post,
695    path = "/api/v1/users/{user_id}/unlock",
696    tag = "users",
697    params(("user_id" = Uuid, Path, description = "User ID")),
698    responses(
699        (status = 200, description = "User unlocked", body = UserResponse),
700        (status = 404, description = "User not found"),
701    ),
702    security(("bearer" = []))
703)]
704pub async fn unlock<C: Connection + Clone>(
705    auth_user: AuthenticatedUser,
706    authz: AuthzData,
707    state: web::Data<AppState<C>>,
708    path: web::Path<Uuid>,
709) -> Result<HttpResponse, AxiamApiError> {
710    RequirePermission::new("users:admin", Uuid::nil())
711        .check(&auth_user, authz.get_ref().as_ref())
712        .await?;
713    let user_id = path.into_inner();
714    let tenant_id = auth_user.tenant_id;
715
716    let update = UpdateUser {
717        failed_login_attempts: Some(0),
718        locked_until: Some(None),
719        last_failed_login_at: Some(None),
720        status: Some(UserStatus::Active),
721        ..Default::default()
722    };
723
724    let user = state.user_repo.update(tenant_id, user_id, update).await?;
725    Ok(HttpResponse::Ok().json(UserResponse::from(user)))
726}
