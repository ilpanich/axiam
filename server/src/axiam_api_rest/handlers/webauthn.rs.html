105/// Build the response for a completed WebAuthn sign-in.
106///
107/// Both passkey ceremonies end here, and both are **primary authentication** —
108/// the same thing `POST /api/v1/auth/login` is — so they must leave the caller
109/// in the same state that endpoint does.
110///
111/// They did not. Until this helper existed, the two `authenticate/*/finish`
112/// handlers answered with the token pair in the JSON body and set no cookies at
113/// all, which made a browser passkey sign-in impossible to complete: the admin
114/// UI runs the ceremony, then calls `GET /api/v1/auth/me` to hydrate the
115/// session, and that endpoint reads `axiam_access` from a cookie that had never
116/// been set. The ceremony succeeded, the session existed server-side, and the
117/// user was bounced back to the login page. `POST /api/v1/auth/refresh` was
118/// unreachable for the same reason (it reads the refresh token from
119/// `axiam_refresh`, never from a body), and every state-changing call after it
120/// would have failed CSRF, there being no `axiam_csrf` to echo.
121///
122/// So this emits the same `Set-Cookie` triple and the same `X-CSRF-Token`
123/// header as `cookie_response_from_output`, the password path's builder.
124///
125/// **The body keeps its tokens.** They are what a non-browser client uses —
126/// CONTRACT.md §24 has the SDKs adopt them directly rather than digging a value
127/// back out of a cookie jar — and dropping them to "match login exactly" would
128/// break every such caller for a symmetry nobody asked for. Adding cookies
129/// beside an unchanged body is additive on the wire: a client that reads only
130/// the body sees no difference.
131fn webauthn_session_response(
132    config: &axiam_auth::config::AuthConfig,
133    out: axiam_auth::LoginOutput,
134) -> HttpResponse {
135    let csrf_token = generate_csrf_token();
136
137    HttpResponse::Ok()
138        .cookie(access_cookie(
139            &out.access_token,
140            config.access_token_lifetime_secs,
141            config.cookie_secure,
142        ))
143        .cookie(refresh_cookie(
144            &out.refresh_token,
145            config.refresh_token_lifetime_secs,
146            config.cookie_secure,
147        ))
148        .cookie(csrf_cookie(
149            &csrf_token,
150            // Session lifetime, not access-token lifetime — see the note in
151            // `handlers::auth`'s login response.
152            config.refresh_token_lifetime_secs,
153            config.cookie_secure,
154        ))
155        // CONTRACT.md §3 "Non-browser SDKs" — same reason as the password
156        // path: a cookie jar is awkward to read one value out of, so the
157        // freshly-minted CSRF token is echoed as a header too. No new
158        // disclosure; `axiam_csrf` is not httpOnly and carries the same value.
159        .insert_header((HEADER_CSRF, csrf_token))
160        .json(WebauthnLoginResponse {
161            access_token: out.access_token,
162            refresh_token: out.refresh_token,
163            session_id: out.session_id,
164            expires_in: out.expires_in,
165        })
166}

168/// Extract tenant_id from an unverified JWT state token by
169/// base64-decoding the payload segment.  This is safe because the
170/// token will be fully verified by `WebauthnService::finish_authentication`;
171/// we only peek to route the request to the correct tenant scope.
172fn peek_tenant_id(state_token: &str) -> Result<Uuid, AxiamApiError> {
173    use base64::Engine;
174    use base64::engine::general_purpose::URL_SAFE_NO_PAD;
175
176    let parts: Vec<&str> = state_token.split('.').collect();
177    if parts.len() != 3 {
178        return Err(AxiamApiError(
179            axiam_core::error::AxiamError::AuthenticationFailed {
180                reason: "invalid state token".into(),
181            },
182        ));
183    }
184
185    let payload = URL_SAFE_NO_PAD.decode(parts[1]).map_err(|_| {
186        AxiamApiError(axiam_core::error::AxiamError::AuthenticationFailed {
187            reason: "invalid state token payload".into(),
188        })
189    })?;
190
191    #[derive(Deserialize)]
192    struct Peek {
193        tenant_id: String,
194    }
195
196    let peek: Peek = serde_json::from_slice(&payload).map_err(|_| {
197        AxiamApiError(axiam_core::error::AxiamError::AuthenticationFailed {
198            reason: "invalid state token claims".into(),
199        })
200    })?;
201
202    peek.tenant_id.parse().map_err(|_| {
203        AxiamApiError(axiam_core::error::AxiamError::AuthenticationFailed {
204            reason: "invalid tenant_id in state token".into(),
205        })
206    })
207}

213/// `POST /api/v1/auth/webauthn/register/start`
214///
215/// Begin a WebAuthn passkey registration ceremony for the
216/// authenticated user.
217///
218/// X3 wave 3: routes through
219/// [`WebauthnService::start_registration_for_policy`], resolving the
220/// tenant's [`WebauthnAttestationPolicy`](axiam_core::models::webauthn_policy::WebauthnAttestationPolicy)
221/// first (an absent row is `WebauthnAttestationPolicy::default()` — `mode:
222/// none`, today's behavior unchanged). This is the single call-site switch
223/// that makes attestation enforcement live: calling
224/// [`WebauthnService::start_registration`] directly here would silently
225/// downgrade every tenant to the unattested ceremony regardless of policy.
226#[utoipa::path(
227    post,
228    path = "/api/v1/auth/webauthn/register/start",
229    tag = "webauthn",
230    responses(
231        (status = 200, description = "Registration challenge",
232         body = StartRegistrationResponse),
233        (status = 401, description = "Unauthorized"),
234        (status = 503, description = "Attestation policy requires attestation but no \
235            FIDO metadata is available (W2-D3 fail-closed)"),
236    ),
237    security(("bearer" = []))
238)]
239pub async fn start_registration<C: Connection + Clone>(
240    user: AuthenticatedUser,
241    state: web::Data<AppState<C>>,
242) -> Result<HttpResponse, AxiamApiError> {
243    let user_name = user.user_id.to_string();
244    let policy = state
245        .webauthn
246        .webauthn_attestation_policy_repo
247        .get_by_tenant(user.tenant_id)
248        .await?
249        .unwrap_or_default();
250
251    let (challenge, state_token) = state
252        .webauthn
253        .webauthn_service
254        .start_registration_for_policy(
255            user.tenant_id,
256            user.org_id,
257            user.user_id,
258            &user_name,
259            &policy,
260            &state.webauthn.attestation_metadata_source,
261            &state.webauthn.attestation_ca_cache,
262        )
263        .await?;
264
265    Ok(HttpResponse::Ok().json(StartRegistrationResponse {
266        challenge,
267        state_token,
268    }))
269}

271/// T-153: refuse an attested registration when the ingested MDS BLOB is too
272/// far past its `nextUpdate` for the operator's taste.
273///
274/// Runs BEFORE the ceremony is finished, not after, so a registration that is
275/// going to be refused is refused without first writing a credential.
276///
277/// Three conditions must all hold, and each omission is deliberate:
278///
279/// - **The tenant's policy requires attestation.** Under
280///   `AttestationMode::None` no metadata is consulted, so stale metadata
281///   cannot have misled the decision and refusing would deny a registration
282///   for a reason that does not apply to it.
283/// - **`mds_max_stale_days > 0`.** Opt-in. The default keeps the documented
284///   fail-open behaviour: a FIDO Alliance outage must not brick registration.
285/// - **A BLOB has actually been ingested.** "Never ingested" is a different
286///   condition from "ingested and old", and it is already handled by the
287///   policy's `unknown_aaguid` setting — an empty metadata source makes every
288///   AAGUID unknown. Treating never-ingested as stale here would make
289///   `mds_max_stale_days` silently disable WebAuthn on a deployment that has
290///   not enabled MDS at all.
291async fn enforce_mds_freshness<C: Connection + Clone>(
292    state: &web::Data<AppState<C>>,
293    policy: &axiam_core::models::webauthn_policy::WebauthnAttestationPolicy,
294    tenant_id: Uuid,
295    user_id: Uuid,
296) -> Result<(), AxiamApiError> {
297    use axiam_core::models::webauthn_policy::{AttestationDenyReason, AttestationMode};
298
299    let max_days = state.webauthn.pki_config.mds_max_stale_days;
300    if max_days == 0 || policy.mode == AttestationMode::None {
301        return Ok(());
302    }
303
304    let Some(meta) = state.webauthn.mds_repo.get_meta().await? else {
305        return Ok(());
306    };
307
308    let days_past = chrono::Utc::now()
309        .date_naive()
310        .signed_duration_since(meta.next_update)
311        .num_days();
312    if days_past <= i64::from(max_days) {
313        return Ok(());
314    }
315
316    tracing::warn!(
317        action = "webauthn.attestation_denied",
318        tenant_id = %tenant_id,
319        user_id = %user_id,
320        reason = ?AttestationDenyReason::MetadataStale,
321        days_past_next_update = days_past,
322        max_stale_days = max_days,
323        mds_no = meta.no,
324        "WebAuthn registration denied: the ingested FIDO MDS BLOB is further past \
325         its nextUpdate than AXIAM__PKI__MDS_MAX_STALE_DAYS permits, so an \
326         authenticator revoked since the last refresh could not be detected"
327    );
328    Err(AxiamApiError(
329        axiam_auth::error::AuthError::WebauthnAttestationDenied {
330            reason: AttestationDenyReason::MetadataStale,
331        }
332        .into(),
333    ))
334}

336/// `POST /api/v1/auth/webauthn/register/finish`
337///
338/// Complete a WebAuthn passkey registration ceremony.
339///
340/// X3 wave 3: routes through
341/// [`WebauthnService::finish_registration_for_policy`], which dispatches on
342/// which ceremony the state token was minted for and re-checks it against
343/// the policy **currently** in force (see that method's docs for why a
344/// policy tightened mid-ceremony still denies).
345#[utoipa::path(
346    post,
347    path = "/api/v1/auth/webauthn/register/finish",
348    tag = "webauthn",
349    request_body = FinishRegistrationRequest,
350    responses(
351        (status = 201, description = "Credential registered",
352         body = CredentialResponse),
353        (status = 401, description = "Unauthorized"),
354        (status = 403, description = "Denied by the tenant's attestation policy \
355            (this security key model is not permitted by your organization)"),
356    ),
357    security(("bearer" = []))
358)]
359pub async fn finish_registration<C: Connection + Clone>(
360    user: AuthenticatedUser,
361    state: web::Data<AppState<C>>,
362    body: web::Json<FinishRegistrationRequest>,
363) -> Result<HttpResponse, AxiamApiError> {
364    let b = body.into_inner();
365    let policy = state
366        .webauthn
367        .webauthn_attestation_policy_repo
368        .get_by_tenant(user.tenant_id)
369        .await?
370        .unwrap_or_default();
371
372    enforce_mds_freshness(&state, &policy, user.tenant_id, user.user_id).await?;
373
374    let cred = state
375        .webauthn
376        .webauthn_service
377        .finish_registration_for_policy(
378            user.tenant_id,
379            user.user_id,
380            &b.state_token,
381            &b.credential_name,
382            &b.response,
383            &policy,
384            &state.webauthn.attestation_metadata_source,
385            &state.webauthn.attestation_ca_cache,
386        )
387        .await?;
388
389    // A registered factor is a *required* factor. Without this the credential
390    // was listed on the profile page and accepted at
391    // `/auth/webauthn/authenticate`, but `AuthService::login` gates its MFA
392    // challenge on `mfa_enabled` — which only `confirm_mfa` ever set — so a
393    // user whose sole factor was a passkey or a security key signed in with a
394    // password and nothing else. Enrolling one now means the same thing
395    // enrolling TOTP means.
396    //
397    // Deliberately after the credential is persisted, and deliberately not
398    // failing the request if it goes wrong: the credential exists either way,
399    // and reporting the registration as failed would invite the user to
400    // register a second one. A failure here leaves MFA off, which the next
401    // enrollment or the profile page's own state will correct — and it is
402    // logged loudly rather than swallowed.
403    if let Err(e) = state
404        .mfa_method_service
405        .enable_after_enrollment(user.tenant_id, user.user_id)
406        .await
407    {
408        tracing::error!(
409            action = "webauthn.enable_mfa_failed",
410            tenant_id = %user.tenant_id,
411            user_id = %user.user_id,
412            credential_id = %cred.id,
413            error = %e,
414            "registered a WebAuthn credential but could not mark MFA as required \
415             for this account — sign-in will not challenge for it until the flag \
416             is set"
417        );
418    }
419
420    Ok(HttpResponse::Created().json(CredentialResponse {
421        id: cred.id,
422        credential_id: cred.credential_id,
423        name: cred.name,
424        credential_type: cred.credential_type,
425        created_at: cred.created_at.to_rfc3339(),
426        last_used_at: cred.last_used_at.map(|t| t.to_rfc3339()),
427    }))
428}

503        state_token,

507/// Resolve `(tenant_id, org_id)` for a usernameless ceremony.
508///
509/// Mirrors `handlers::auth::login`, including the two properties that matter
510/// more here than they look:
511///
512/// * slug-resolution failures become 401 `AuthenticationFailed`, not 404, so
513///   this endpoint cannot be used to enumerate which organizations and tenants
514///   exist — it is reachable without any credential at all;
515/// * the org is derived from the **tenant record**, never from what the client
516///   sent. NEW-1 is the same bug in `login`: a caller who supplies a raw
517///   `org_id` alongside their own tenant would otherwise get a ceremony — and
518///   then a session token — scoped to a foreign organization.
519async fn resolve_workspace<C: Connection + Clone>(
520    state: &web::Data<AppState<C>>,
521    b: &StartDiscoverableAuthenticationRequest,
522) -> Result<(Uuid, Uuid), AxiamApiError> {
523    use axiam_core::error::AxiamError;
524    use axiam_core::repository::{OrganizationRepository, TenantRepository};
525
526    let unauthenticated = || AxiamError::AuthenticationFailed {
527        reason: "invalid credentials".into(),
528    };
529
530    // A blank slug is not a slug — see `handlers::auth::blank_is_unnamed`. The
531    // login page collects the workspace in one step and hands both fields to
532    // this ceremony, so an organization-level sign-in arrives here with an
533    // empty tenant.
534    let org_slug = crate::handlers::auth::blank_is_unnamed(b.org_slug.as_deref());
535    let tenant_slug = crate::handlers::auth::blank_is_unnamed(b.tenant_slug.as_deref());
536
537    let org_id = match (b.org_id, org_slug) {
538        (Some(id), _) => id,
539        (None, Some(slug)) => {
540            state
541                .org_repo
542                .get_by_slug(slug)
543                .await
544                .map_err(|_| unauthenticated())?
545                .id
546        }
547        (None, None) => {
548            return Err(AxiamApiError(AxiamError::Validation {
549                message: "must provide org_id or org_slug".into(),
550            }));
551        }
552    };
553
554    let tenant_id = match (b.tenant_id, tenant_slug) {
555        (Some(id), _) => id,
556        (None, Some(slug)) => {
557            state
558                .tenant_repo
559                .get_by_slug(org_id, slug)
560                .await
561                .map_err(|_| unauthenticated())?
562                .id
563        }
564        // No tenant named: the organization's own scope, as `login` does. An
565        // organization-level principal holds passkeys like any other, and
566        // demanding a tenant here made the one credential it could sign in with
567        // unusable to it.
568        (None, None) => {
569            state
570                .tenant_repo
571                .get_organization_tenant(org_id)
572                .await
573                .map_err(|_| unauthenticated())?
574                .id
575        }
576    };
577
578    // NEW-1: authoritative org comes from the tenant record, and a client that
579    // named a different one is refused rather than quietly corrected.
580    let tenant = state
581        .tenant_repo
582        .get_by_id(tenant_id)
583        .await
584        .map_err(|_| unauthenticated())?;
585    if tenant.organization_id != org_id {
586        return Err(AxiamApiError(unauthenticated()));
587    }
588
589    Ok((tenant_id, tenant.organization_id))
590}

592/// `POST /api/v1/auth/webauthn/authenticate/discoverable/finish`
593///
594/// Complete a usernameless ceremony. The user is identified by the assertion
595/// itself; on success a session is created and tokens are issued.
596///
597/// # Why this fires `login.post_auth` when `authenticate/finish` does not
598///
599/// `sdks/CONTRACT.md` §22.5 excludes the ordinary WebAuthn finish because it
600/// "continues a login that was already gated at its first step" — the password
601/// step fired the event before ever issuing the MFA challenge token. A
602/// usernameless sign-in has no such first step; it *is* the first step. Leaving
603/// it out would hand an operator's `login.post_auth` veto — the embargoed-region
604/// example in the feature's own docs — a bypass consisting of clicking "Sign in
605/// with a passkey". That is SEC-095 exactly, which is why this uses the shared
606/// `intercept_federated_login_post_auth` rather than a fourth private copy.
607///
608/// It takes the *federated* variant because, like SAML and OIDC, this path
609/// completes in one round trip and has no step-up branch to route a
610/// `require_mfa` verdict into. A reactor demanding step-up here is refused
611/// rather than silently dropped. The passkey has already provided user
612/// verification, so this is a UV-backed factor, not an unverified one.
613#[utoipa::path(
614    post,
615    path = "/api/v1/auth/webauthn/authenticate/discoverable/finish",
616    tag = "webauthn",
617    request_body = FinishAuthenticationRequest,
618    responses(
619        (status = 200,
620         description = "Authentication successful. Sets the axiam_access, \
621                        axiam_refresh and axiam_csrf cookies and echoes the \
622                        CSRF token in X-CSRF-Token, exactly as POST \
623                        /api/v1/auth/login does.",
624         body = WebauthnLoginResponse),
625        (status = 401, description = "Authentication failed"),
626    )
627)]
628pub async fn finish_discoverable_authentication<C: Connection + Clone>(
629    req: HttpRequest,
630    state: web::Data<AppState<C>>,
631    body: web::Json<FinishAuthenticationRequest>,
632) -> Result<HttpResponse, AxiamApiError> {
633    use axiam_core::repository::UserRepository;
634
635    let b = body.into_inner();
636    let tenant_id = peek_tenant_id(&b.state_token)?;
637
638    let (user_id, org_id) = state
639        .webauthn
640        .webauthn_service
641        .finish_discoverable_authentication(tenant_id, &b.state_token, &b.response)
642        .await?;
643
644    let user = state
645        .user_repo
646        .get_by_id(tenant_id, user_id)
647        .await
648        .map_err(AxiamApiError)?;
649
650    // A passkey proves possession; it does not say the account may sign in.
651    // The username-bound ceremony inherits that gate from the password step
652    // that minted its challenge token — this one has no such step, so a
653    // deactivated, locked or anonymised user would otherwise get a session.
654    state
655        .auth_service
656        .ensure_can_sign_in(&user)
657        .map_err(|e| AxiamApiError(e.into()))?;
658
659    // X1/SEC-095: after the assertion verifies, before any session exists.
660    state
661        .auth_service
662        .intercept_federated_login_post_auth(
663            tenant_id,
664            org_id,
665            &user,
666            client_ip(&req).as_deref(),
667            user_agent(&req).as_deref(),
668        )
669        .await
670        .map_err(AxiamApiError)?;
671
672    let out = state
673        .auth_service
674        .create_session_and_tokens(
675            user_id,
676            tenant_id,
677            org_id,
678            client_ip(&req),
679            user_agent(&req),
680        )
681        .await?;
682
683    Ok(webauthn_session_response(&state.auth_config, out))
684}

739#[cfg(test)]
740mod tests {
741    use super::*;
742    use actix_web::http::header::SET_COOKIE;
743    use axiam_auth::config::AuthConfig;
744
745    fn out() -> axiam_auth::LoginOutput {
746        axiam_auth::LoginOutput {
747            access_token: "access-token-value".into(),
748            refresh_token: "refresh-token-value".into(),
749            session_id: Uuid::nil(),
750            expires_in: 900,
751        }
752    }
753
754    fn cookies(res: &HttpResponse) -> Vec<String> {
755        res.headers()
756            .get_all(SET_COOKIE)
757            .map(|v| v.to_str().unwrap().to_owned())
758            .collect()
759    }
760
761    fn cookie_named<'a>(set: &'a [String], name: &str) -> &'a str {
762        set.iter()
763            .find(|c| c.starts_with(&format!("{name}=")))
764            .unwrap_or_else(|| panic!("no {name} cookie in {set:?}"))
765    }
766
767    /// The regression this helper exists for: a completed passkey ceremony
768    /// used to set no cookies at all, so the browser had no session and
769    /// `GET /api/v1/auth/me` answered 401 immediately afterwards.
770    #[test]
771    fn sets_the_same_cookie_triple_as_password_login() {
772        let config = AuthConfig {
773            cookie_secure: true,
774            ..AuthConfig::default()
775        };
776        let set = cookies(&webauthn_session_response(&config, out()));
777
778        let access = cookie_named(&set, "axiam_access");
779        assert!(access.contains("access-token-value"));
780        assert!(
781            access.contains("HttpOnly"),
782            "access cookie must be httpOnly"
783        );
784
785        let refresh = cookie_named(&set, "axiam_refresh");
786        assert!(refresh.contains("refresh-token-value"));
787        assert!(
788            refresh.contains("HttpOnly"),
789            "refresh cookie must be httpOnly"
790        );
791
792        // Readable by JavaScript on purpose — the SPA has to echo it back in
793        // `X-CSRF-Token`, which it cannot do with an httpOnly cookie.
794        let csrf = cookie_named(&set, "axiam_csrf");
795        assert!(!csrf.contains("HttpOnly"), "csrf cookie must be readable");
796    }
797
798    /// §3's non-browser rule: the same token, in the header and the cookie.
799    /// An SDK reads the header; a mismatch would make the very first
800    /// state-changing call after a passkey sign-in fail CSRF validation.
801    #[test]
802    fn echoes_the_csrf_token_in_the_header_and_the_cookie() {
803        let res = webauthn_session_response(&AuthConfig::default(), out());
804
805        let header = res
806            .headers()
807            .get(HEADER_CSRF)
808            .expect("X-CSRF-Token header")
809            .to_str()
810            .unwrap()
811            .to_owned();
812        assert!(!header.is_empty());
813
814        let set = cookies(&res);
815        let csrf = cookie_named(&set, "axiam_csrf");
816        assert!(
817            csrf.contains(&header),
818            "cookie {csrf} must carry the header value {header}"
819        );
820    }
821
822    /// A fresh token per sign-in, not a constant.
823    #[test]
824    fn mints_a_distinct_csrf_token_per_response() {
825        let config = AuthConfig::default();
826        let first = webauthn_session_response(&config, out());
827        let second = webauthn_session_response(&config, out());
828        assert_ne!(
829            first.headers().get(HEADER_CSRF).unwrap(),
830            second.headers().get(HEADER_CSRF).unwrap()
831        );
832    }
833
834    /// Adding cookies must not have taken the tokens out of the body: the
835    /// non-browser SDKs adopt them from there (CONTRACT.md §24.3).
836    #[test]
837    fn keeps_the_token_pair_in_the_body() {
838        let config = AuthConfig::default();
839        let res = webauthn_session_response(&config, out());
840        assert_eq!(res.status(), actix_web::http::StatusCode::OK);
841
842        let body = actix_web::body::to_bytes(res.into_body());
843        let body = futures::executor::block_on(body).expect("body");
844        let parsed: serde_json::Value = serde_json::from_slice(&body).expect("json body");
845
846        assert_eq!(parsed["access_token"], "access-token-value");
847        assert_eq!(parsed["refresh_token"], "refresh-token-value");
848        assert_eq!(parsed["expires_in"], 900);
849        assert!(parsed.get("session_id").is_some());
850    }
