1//! Tenant WebAuthn attestation policy admin endpoints (X3 wave 3).
2//!
3//! Three admin-gated, tenant-scoped endpoints:
4//!
5//! - `GET  /api/v1/tenants/{tenant_id}/webauthn/attestation-policy` — the
6//!   effective policy. An absent row returns
7//!   [`WebauthnAttestationPolicy::default`] (D5), never a 404: "no policy
8//!   configured" is a valid, meaningful answer (today's `mode: none`
9//!   behavior, unchanged) rather than a missing resource.
10//! - `PUT  /api/v1/tenants/{tenant_id}/webauthn/attestation-policy` —
11//!   replace the policy. Validates the D5 invariants
12//!   (`require_fido_certified`/`min_certification` with `mode: none` is
13//!   rejected as a config error), persists, and invalidates the
14//!   process-wide [`AttestationCaCache`](axiam_auth::AttestationCaCache) —
15//!   a cached CA list built under the OLD `allowed_aaguids` restriction
16//!   must not survive a policy change (W2-D3).
17//! - `GET  /api/v1/tenants/{tenant_id}/webauthn/compliance-report` — D9.
18//!   Evaluates the *current* policy against every credential's recorded
19//!   `aaguid`/`attestation_format`. Read-only: never mutates or revokes
20//!   anything (revocation stays the existing admin credential-delete path).
21//!
22//! ## Permission choice
23//!
24//! Gated by dedicated `webauthn_policy:read` (both `GET`s) /
25//! `webauthn_policy:write` (`PUT`) permissions, mirroring the
26//! `email_config:read`/`email_config:write` pair, rather than reusing the
27//! generic `settings:get`/`settings:update` pair `handlers::settings` uses.
28//! The attestation policy is deliberately NOT part of `SecuritySettings`'s
29//! org/tenant inheritance model (see `axiam_core::models::webauthn_policy`'s
30//! module docs for why — no total order on AAGUID allow/block lists), so it
31//! should not share that pair's permission surface either: an admin with
32//! `settings:update` should not automatically be able to change which
33//! security keys a tenant accepts.
34

51/// `GET` response: the stored policy plus the unknown-AAGUID action it
52/// currently *resolves to*.
53///
54/// `unknown_aaguid` is nullable, where `null` means "use this mode's
55/// default" (deny under `direct_required`, allow otherwise). A client that
56/// only saw the stored `null` would have to re-derive that rule itself to
57/// display what the policy actually does — and a security rule implemented
58/// twice is a security rule that will eventually disagree with itself. So
59/// the server resolves it once, here, and reports both: the admin's stored
60/// intent and its effect.
61#[derive(Debug, Serialize, utoipa::ToSchema)]
62pub struct PolicyResponse {
63    #[serde(flatten)]
64    pub policy: WebauthnAttestationPolicy,
65    /// The action actually applied to an AAGUID with no MDS metadata,
66    /// with `unknown_aaguid: null` resolved against `mode`. Read-only —
67    /// `PUT` ignores it.
68    pub effective_unknown_aaguid: UnknownAaguidAction,
69}
70
71impl From<WebauthnAttestationPolicy> for PolicyResponse {
72    fn from(policy: WebauthnAttestationPolicy) -> Self {
73        Self {
74            effective_unknown_aaguid: policy.effective_unknown_aaguid(),
75            policy,
76        }
77    }
78}

84/// One credential's compliance outcome (D9).
85#[derive(Debug, Serialize, utoipa::ToSchema)]
86pub struct ComplianceReportEntry {
87    pub credential_id: Uuid,
88    pub user_id: Uuid,
89    /// The credential's user-assigned name (`WebauthnCredential::name`),
90    /// not the owning user's account name.
91    pub name: String,
92    pub aaguid: Option<Uuid>,
93    pub authenticator_name: Option<String>,
94    /// `false` only for a genuine policy violation
95    /// ([`ComplianceStatus::NonCompliant`]) — a credential with no recorded
96    /// AAGUID ([`ComplianceStatus::Unknown`], D9's pre-X3 case) is always
97    /// `true` here, never reported as a violation.
98    pub compliant: bool,
99    /// `None` only when `compliant` and the credential has a recorded
100    /// AAGUID; set for every non-compliant *and* every "unknown" (pre-X3)
101    /// credential.
102    pub reason: Option<String>,
103}

109fn tenant_scope_check(user: &AuthenticatedUser, tenant_id: Uuid) -> Result<(), AxiamApiError> {
110    if tenant_id != user.tenant_id {
111        return Err(AxiamApiError(AxiamError::AuthorizationDenied {
112            reason: "cannot access WebAuthn attestation policy for a different tenant".into(),
113            action: None,
114            resource_id: None,
115        }));
116    }
117    Ok(())
118}
119
120/// `AttestationDenyReason` serializes `snake_case` (its own
121/// `#[serde(rename_all = "snake_case")]`) — reuse that exact wire
122/// representation for the compliance report's `reason` string instead of a
123/// separate `{reason:?}` Debug format, so the two surfaces agree.
124fn deny_reason_str(reason: axiam_core::models::webauthn_policy::AttestationDenyReason) -> String {
125    serde_json::to_value(reason)
126        .ok()
127        .and_then(|v| v.as_str().map(str::to_owned))
128        .unwrap_or_else(|| format!("{reason:?}"))
129}

135/// `GET /api/v1/tenants/{tenant_id}/webauthn/attestation-policy`
136#[utoipa::path(
137    get,
138    path = "/api/v1/tenants/{tenant_id}/webauthn/attestation-policy",
139    tag = "webauthn-policy",
140    params(
141        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
142    ),
143    responses(
144        (status = 200, description = "Effective WebAuthn attestation policy \
145            (defaults returned when no policy row exists — D5)",
146         body = PolicyResponse),
147    ),
148    security(("bearer" = []))
149)]
150pub async fn get_policy<C: Connection + Clone>(
151    user: AuthenticatedUser,
152    authz: AuthzData,
153    state: web::Data<AppState<C>>,
154    path: web::Path<Uuid>,
155) -> Result<HttpResponse, AxiamApiError> {
156    RequirePermission::new("webauthn_policy:read", Uuid::nil())
157        .check(&user, authz.get_ref().as_ref())
158        .await?;
159    let tenant_id = path.into_inner();
160    tenant_scope_check(&user, tenant_id)?;
161
162    let policy = state
163        .webauthn
164        .webauthn_attestation_policy_repo
165        .get_by_tenant(tenant_id)
166        .await?
167        .unwrap_or_default();
168    Ok(HttpResponse::Ok().json(PolicyResponse::from(policy)))
169}

171/// `PUT /api/v1/tenants/{tenant_id}/webauthn/attestation-policy`
172#[utoipa::path(
173    put,
174    path = "/api/v1/tenants/{tenant_id}/webauthn/attestation-policy",
175    tag = "webauthn-policy",
176    params(
177        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
178    ),
179    request_body = WebauthnAttestationPolicy,
180    responses(
181        (status = 200, description = "Policy updated", body = WebauthnAttestationPolicy),
182        (status = 400, description = "Config error — e.g. require_fido_certified/\
183            min_certification set with mode: none"),
184    ),
185    security(("bearer" = []))
186)]
187pub async fn set_policy<C: Connection + Clone>(
188    user: AuthenticatedUser,
189    authz: AuthzData,
190    state: web::Data<AppState<C>>,
191    path: web::Path<Uuid>,
192    body: web::Json<WebauthnAttestationPolicy>,
193) -> Result<HttpResponse, AxiamApiError> {
194    RequirePermission::new("webauthn_policy:write", Uuid::nil())
195        .check(&user, authz.get_ref().as_ref())
196        .await?;
197    let tenant_id = path.into_inner();
198    tenant_scope_check(&user, tenant_id)?;
199
200    let input = body.into_inner();
201    validate_attestation_policy(&input)?;
202
203    let saved = state
204        .webauthn
205        .webauthn_attestation_policy_repo
206        .set(tenant_id, input)
207        .await?;
208
209    // W2-D3: a stale cached AttestationCaList built under the OLD
210    // allowed_aaguids restriction must not survive a policy change. The
211    // repository's own `.set()` already emits the `webauthn.policy_updated`
212    // structured audit event (D11); this HTTP mutation is additionally
213    // captured by the generic per-request `AuditMiddleware` wired in
214    // `axiam-server/src/main.rs` (method + path + actor + outcome).
215    state.webauthn.attestation_ca_cache.invalidate();
216
217    Ok(HttpResponse::Ok().json(saved))
218}

220/// `GET /api/v1/tenants/{tenant_id}/webauthn/compliance-report`
221#[utoipa::path(
222    get,
223    path = "/api/v1/tenants/{tenant_id}/webauthn/compliance-report",
224    tag = "webauthn-policy",
225    params(
226        ("tenant_id" = Uuid, Path, description = "Tenant ID"),
227    ),
228    responses(
229        (status = 200, description = "Per-credential compliance report against the \
230            current policy — never mutates or revokes anything",
231         body = [ComplianceReportEntry]),
232    ),
233    security(("bearer" = []))
234)]
235pub async fn compliance_report<C: Connection + Clone>(
236    user: AuthenticatedUser,
237    authz: AuthzData,
238    state: web::Data<AppState<C>>,
239    path: web::Path<Uuid>,
240) -> Result<HttpResponse, AxiamApiError> {
241    RequirePermission::new("webauthn_policy:read", Uuid::nil())
242        .check(&user, authz.get_ref().as_ref())
243        .await?;
244    let tenant_id = path.into_inner();
245    tenant_scope_check(&user, tenant_id)?;
246
247    let policy = state
248        .webauthn
249        .webauthn_attestation_policy_repo
250        .get_by_tenant(tenant_id)
251        .await?
252        .unwrap_or_default();
253
254    let credentials = state
255        .webauthn
256        .webauthn_credential_repo
257        .list_by_tenant(tenant_id)
258        .await?;
259
260    let mut out = Vec::with_capacity(credentials.len());
261    for cred in credentials {
262        let status = axiam_auth::attestation::evaluate_credential_compliance(
263            &policy,
264            cred.aaguid,
265            cred.attestation_format.as_deref(),
266            &state.webauthn.attestation_metadata_source,
267        )
268        .await?;
269
270        let (compliant, reason) = match status {
271            ComplianceStatus::Compliant => (true, None),
272            ComplianceStatus::NonCompliant(r) => (false, Some(deny_reason_str(r))),
273            // D9: "never as a violation" — always compliant=true, with the
274            // fixed pre-X3 explanation as the reason.
275            ComplianceStatus::Unknown => (true, Some(UNKNOWN_CREDENTIAL_REASON.to_string())),
276        };
277
278        out.push(ComplianceReportEntry {
279            credential_id: cred.id,
280            user_id: cred.user_id,
281            name: cred.name,
282            aaguid: cred.aaguid,
283            authenticator_name: cred.authenticator_name,
284            compliant,
285            reason,
286        });
287    }
288
289    Ok(HttpResponse::Ok().json(out))
290}
