12/// Trait for checking backend health (DB, etc.).
13///
14/// Object-safe — stored as `web::Data<Arc<dyn HealthChecker>>`.
15pub trait HealthChecker: Send + Sync {
16    fn check(&self) -> Pin<Box<dyn Future<Output = Result<(), String>> + Send + '_>>;
17}
18
19impl HealthChecker for axiam_db::DbManager {
20    fn check(&self) -> Pin<Box<dyn Future<Output = Result<(), String>> + Send + '_>> {
21        Box::pin(async {
22            self.health_check()
23                .await
24                .map_err(|e| format!("db health check failed: {e}"))
25        })
26    }
27}
28
29impl HealthChecker for axiam_db::DbPool {
30    fn check(&self) -> Pin<Box<dyn Future<Output = Result<(), String>> + Send + '_>> {
31        Box::pin(async {
32            // Probes every pooled handle so readiness reflects the whole pool
33            // (an auth-expired or poisoned handle anywhere trips the gate).
34            self.health_check()
35                .await
36                .map_err(|e| format!("db health check failed: {e}"))
37        })
38    }
39}

57/// The last-known outcome of one background sweep.
58///
59/// T-129: a scheduled job that stops running is invisible. GDPR erasure and
60/// certificate expiry are exactly the jobs whose *absence* is the incident —
61/// nothing errors, nothing 500s, deletions simply stop happening and the first
62/// symptom is a regulator's question months later. Failures were already
63/// logged, but a log line nobody greps for is not a control.
64#[derive(Serialize, utoipa::ToSchema, Clone, Debug)]
65pub struct JobStatus {
66    /// Stable identifier for the sweep, e.g. `gdpr_purge`.
67    pub name: String,
68    /// RFC 3339 timestamp of the last run that completed without error.
69    pub last_success_at: Option<String>,
70    /// RFC 3339 timestamp of the last run that returned an error.
71    pub last_failure_at: Option<String>,
72    /// The last error text, for the operator who is now looking at this page
73    /// wondering what went wrong.
74    pub last_error: Option<String>,
75    /// Consecutive failures since the last success. Reset to 0 on success.
76    pub consecutive_failures: u32,
77    /// Whether the job has missed enough expected runs to be considered stuck.
78    ///
79    /// The single field to build an alert on. Computed server-side rather than
80    /// left to the caller because the sweep interval is configuration the
81    /// caller does not have, and "how long is too long" is not a judgement a
82    /// dashboard should be re-deriving from timestamps.
83    pub stalled: bool,
84}

86/// Source of [`JobStatus`] snapshots.
87///
88/// A trait for the same reason [`HealthChecker`] is one: the jobs live in
89/// `axiam-server`, which sits above this crate in the layering, so the
90/// dependency has to point inward.
91pub trait JobHealthReporter: Send + Sync {
92    /// Current status of every registered job.
93    fn snapshot(&self) -> Vec<JobStatus>;
94}

108/// Response body for `GET /health/jobs`.
109#[derive(Serialize, utoipa::ToSchema)]
110pub struct JobsHealthResponse {
111    /// `ok` when no job is stalled, `degraded` when at least one is.
112    pub status: &'static str,
113    pub jobs: Vec<JobStatus>,
114}
115
116/// `GET /health/jobs` — scheduled-job liveness (T-129).
117///
118/// Always 200, including when a job is stalled, and that is deliberate. This
119/// is not a readiness gate: a stuck cleanup sweep is an operational problem,
120/// not a reason to pull a healthy server out of the load balancer and send its
121/// traffic to replicas running the same stuck code. Alert on
122/// `status == "degraded"`, or on a specific job's `stalled`.
123#[utoipa::path(
124    get,
125    path = "/health/jobs",
126    tag = "health",
127    responses(
128        (status = 200, description = "Scheduled-job status", body = JobsHealthResponse),
129    )
130)]
131pub async fn jobs<C: Connection + Clone>(state: web::Data<AppState<C>>) -> HttpResponse {
132    let jobs = state.job_health.snapshot();
133    let status = if jobs.iter().any(|j| j.stalled) {
134        "degraded"
135    } else {
136        "ok"
137    };
138    HttpResponse::Ok().json(JobsHealthResponse { status, jobs })
139}

152/// `GET /health` — liveness probe. Always returns 200.
153#[utoipa::path(
154    get,
155    path = "/health",
156    tag = "health",
157    responses(
158        (status = 200, description = "Service is alive", body = HealthResponse),
159    )
160)]
161pub async fn health() -> HttpResponse {
162    HttpResponse::Ok().json(HealthResponse { status: "ok" })
163}

165/// `GET /ready` — readiness probe. Checks DB connectivity.
166#[utoipa::path(
167    get,
168    path = "/ready",
169    tag = "health",
170    responses(
171        (status = 200, description = "Service is ready", body = ReadyResponse),
172        (status = 503, description = "Service is not ready", body = ReadyResponse),
173    )
174)]
175pub async fn ready<C: Connection + Clone>(state: web::Data<AppState<C>>) -> HttpResponse {
176    match state.health_checker.check().await {
177        Ok(()) => HttpResponse::Ok().json(ReadyResponse {
178            status: "ok",
179            database: "connected",
180        }),
181        Err(_) => HttpResponse::ServiceUnavailable().json(ReadyResponse {
182            status: "unavailable",
183            database: "disconnected",
184        }),
185    }
186}
