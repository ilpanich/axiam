1//! CSRF double-submit cookie middleware and cookie builder helpers.
2//!
3//! The [`CsrfMiddleware`] rejects state-changing requests that lack a valid
4//! `X-CSRF-Token` header matching the `axiam_csrf` cookie value.
5//! Comparison uses constant-time equality to prevent timing attacks (D-01).
6//!
7//! Cookie helpers build the three auth cookies (`axiam_access`,
8//! `axiam_refresh`, `axiam_csrf`) with the security attributes specified
9//! in the UI-SPEC design decisions (D-05 through D-09).
10

36// Exempt path suffixes — no CSRF token needed on these endpoints
37// ---------------------------------------------------------------------------
38
39/// Path suffixes that are exempt from CSRF validation.
40///
41/// These are either unauthenticated endpoints (login, MFA flows) that do not
42/// yet have a CSRF cookie, or token-based OAuth2 flows that use their own
43/// security model.
44const CSRF_EXEMPT_SUFFIXES: &[&str] = &[
45    "/api/v1/auth/login",
46    // The OPAQUE endpoints are exempt on exactly the same grounds as /login:
47    // the caller is unauthenticated and has no `axiam_csrf` cookie to echo.
48    // This is a separate registry from `permissions::PUBLIC_PATHS` and both
49    // must cover a route for it to work unauthenticated — omitting it here
50    // gives a 403 before the handler ever runs.
51    "/api/v1/auth/opaque/register/start",
52    "/api/v1/auth/opaque/login/start",
53    "/api/v1/auth/opaque/login/finish",
54    "/api/v1/auth/mfa/verify",
55    "/api/v1/auth/mfa/setup/enroll",
56    "/api/v1/auth/mfa/setup/confirm",
57    // WebAuthn **authentication** — the passkey and security-key equivalents of
58    // /login, and unauthenticated for the same reason: these ceremonies *are*
59    // the authentication, so the caller holds no session and has no
60    // `axiam_csrf` cookie to echo. They were listed in `permissions::
61    // PUBLIC_PATHS` but not here, and both registries must cover a route: the
62    // result was a `403 CSRF validation failed` on
63    // `/webauthn/authenticate/start` before the handler ever ran, which made
64    // passkey sign-in impossible from a browser with no prior session.
65    //
66    // The **registration** pair is deliberately NOT exempt. Adding a passkey is
67    // done by a user who is already signed in and therefore does carry the
68    // cookie, and an exemption there would let any site silently enrol its own
69    // authenticator onto a logged-in victim's account — account takeover by
70    // exactly the request forgery this middleware exists to stop.
71    "/api/v1/auth/webauthn/authenticate/start",
72    "/api/v1/auth/webauthn/authenticate/finish",
73    "/api/v1/auth/webauthn/authenticate/discoverable/start",
74    "/api/v1/auth/webauthn/authenticate/discoverable/finish",
75    "/api/v1/auth/device",
76    // Password reset request + confirm are unauthenticated and token-based:
77    // the caller has no session and therefore no CSRF cookie yet (same model
78    // as /login). Without these, a forgotten-password reset is CSRF-blocked (403).
79    "/api/v1/auth/reset",
80    "/api/v1/auth/reset/confirm",
81    // Bootstrap is a one-time endpoint called before any session exists.
82    // It is protected by the AXIAM_BOOTSTRAP_ADMIN_EMAIL env gate instead (D-10).
83    "/api/v1/admin/bootstrap",
84    // 28-05/CQ-B40: the four first-time-SSO public endpoints (D-22,
85    // handlers/federation.rs) are unauthenticated by design — same model as
86    // /login and /reset/confirm above: the caller has no prior session and
87    // therefore no `axiam_csrf` cookie to echo back. Without these entries a
88    // first-time OIDC/SAML SSO login is CSRF-blocked (403) before the
89    // handler ever runs, even though the route is correctly listed in
90    // PUBLIC_PATHS (AuthzMiddleware bypass is a separate registry from this
91    // one — both must cover a route for it to work unauthenticated).
92    "/api/v1/auth/federation/oidc/start",
93    "/api/v1/auth/federation/oidc/callback",
94    "/api/v1/auth/federation/saml/login",
95    "/api/v1/auth/federation/saml/acs",
96    // The login-provider surface added with "Sign in with X", exempt for the
97    // same reason as the four above: none of these callers has a prior session,
98    // so none has an `axiam_csrf` cookie to echo.
99    //
100    // The two `…/form` entries additionally *cannot* carry one even in
101    // principle — the request is a cross-site form POST performed by the
102    // identity provider. What stands in for CSRF protection there is the
103    // provider's own signed assertion plus the single-use server-side state
104    // row, both of which an attacker would have to forge.
105    "/api/v1/auth/federation/oauth2/start",
106    "/api/v1/auth/federation/oauth2/callback",
107    "/api/v1/auth/federation/oidc/callback/form",
108    "/api/v1/auth/federation/saml/acs/form",
109    // Redeeming a handoff code is same-origin but pre-session: the response
110    // being requested is the one that *sets* the CSRF cookie.
111    "/api/v1/auth/federation/handoff",
112];
113
114/// Path prefixes that are exempt from CSRF validation (OAuth2).
115const CSRF_EXEMPT_PREFIXES: &[&str] = &["/oauth2/"];
116
117fn is_csrf_exempt(path: &str) -> bool {
118    for suffix in CSRF_EXEMPT_SUFFIXES {
119        if path.ends_with(suffix) {
120            return true;
121        }
122    }
123    for prefix in CSRF_EXEMPT_PREFIXES {
124        if path.starts_with(prefix) {
125            return true;
126        }
127    }
128    false
129}

135/// Whether a request authenticates by bearer token ALONE.
136///
137/// Split out as a pure function so the condition can be pinned by tests: it is
138/// the exemption that decides whether CSRF validation runs at all, and the case
139/// that matters is the one where BOTH are present.
140///
141/// * bearer + no session cookie → exempt. There is no ambient credential to
142///   abuse and no double-submit cookie the caller could echo.
143/// * bearer + a session cookie → **not** exempt. That is precisely the shape a
144///   cross-site attacker would craft to escape the exemption: the browser
145///   supplies the cookie, the attacker supplies the header.
146/// * no bearer → not exempt, whatever the cookies say.
147fn is_bearer_only(authorization: Option<&str>, has_session_cookie: bool) -> bool {
148    if has_session_cookie {
149        return false;
150    }
151    authorization.is_some_and(|v| v.trim_start().to_ascii_lowercase().starts_with("bearer "))
152}

154/// CSRF double-submit cookie middleware.
155///
156/// Safe methods (GET, HEAD, OPTIONS) pass through unconditionally.
157/// State-changing methods require `X-CSRF-Token` to match the `axiam_csrf`
158/// cookie value, compared with constant-time equality.
159pub struct CsrfMiddleware;
160
161impl<S, B> Transform<S, ServiceRequest> for CsrfMiddleware
162where
163    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
164    B: 'static,
165{
166    type Response = ServiceResponse<EitherBody<B>>;
167    type Error = Error;
168    type Transform = CsrfMiddlewareService<S>;
169    type InitError = ();
170    type Future = Ready<Result<Self::Transform, Self::InitError>>;
171
172    fn new_transform(&self, service: S) -> Self::Future {
173        ready(Ok(CsrfMiddlewareService { inner: service }))
174    }
175}
176
177pub struct CsrfMiddlewareService<S> {
178    inner: S,
179}
180
181impl<S, B> Service<ServiceRequest> for CsrfMiddlewareService<S>
182where
183    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
184    B: 'static,
185{
186    type Response = ServiceResponse<EitherBody<B>>;
187    type Error = Error;
188    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>>>>;
189
190    fn poll_ready(
191        &self,
192        cx: &mut std::task::Context<'_>,
193    ) -> std::task::Poll<Result<(), Self::Error>> {
194        self.inner.poll_ready(cx)
195    }
196
197    fn call(&self, req: ServiceRequest) -> Self::Future {
198        // Safe methods — always exempt.
199        let method = req.method().clone();
200        if method == Method::GET || method == Method::HEAD || method == Method::OPTIONS {
201            let fut = self.inner.call(req);
202            return Box::pin(async move {
203                let res = fut.await?;
204                Ok(res.map_into_left_body())
205            });
206        }
207
208        // Exempt paths (login, MFA flows, OAuth2).
209        let path = req.path().to_owned();
210        if is_csrf_exempt(&path) {
211            let fut = self.inner.call(req);
212            return Box::pin(async move {
213                let res = fut.await?;
214                Ok(res.map_into_left_body())
215            });
216        }
217
218        // Bearer-only callers carry no ambient credential, so there is nothing
219        // for CSRF to protect and nothing they could echo back.
220        //
221        // CSRF is an attack on credentials the browser attaches BY ITSELF — the
222        // session cookie. A request authenticated solely by an `Authorization`
223        // header has no such credential: a cross-site page cannot set that
224        // header on a victim's behalf, and the double-submit cookie it would
225        // have to echo does not exist. Demanding one is unsatisfiable rather
226        // than protective, and it made the machine-facing surface unreachable
227        // for exactly the callers it was built for — a service account's
228        // `client_credentials` token could not `POST /api/v1/authz/check`,
229        // whose `AuthenticatedPrincipal` extractor exists to accept the
230        // `axiam:m2m` audience.
231        //
232        // The condition is deliberately narrow: the session cookie must be
233        // ABSENT as well. A request that carries both a session cookie and a
234        // bearer header is exactly the shape a cross-site attacker would craft
235        // to escape this exemption — the browser supplies the cookie, the
236        // attacker supplies the header — so it stays subject to validation.
237        let has_session_cookie = req.cookie(COOKIE_ACCESS).is_some()
238            || req.cookie(COOKIE_REFRESH).is_some()
239            || req.cookie(COOKIE_CSRF).is_some();
240        let auth_header = req
241            .headers()
242            .get(actix_web::http::header::AUTHORIZATION)
243            .and_then(|v| v.to_str().ok());
244        if is_bearer_only(auth_header, has_session_cookie) {
245            let fut = self.inner.call(req);
246            return Box::pin(async move {
247                let res = fut.await?;
248                Ok(res.map_into_left_body())
249            });
250        }
251
252        // Extract CSRF cookie and header.
253        let cookie_value = req.cookie(COOKIE_CSRF).map(|c| c.value().to_owned());
254
255        let header_value = req
256            .headers()
257            .get(HEADER_CSRF)
258            .and_then(|v| v.to_str().ok())
259            .map(|s| s.to_owned());
260
261        // Validate — both must be present and equal (constant-time).
262        let valid = match (cookie_value, header_value) {
263            (Some(cookie), Some(header)) => cookie.as_bytes().ct_eq(header.as_bytes()).into(),
264            _ => false,
265        };
266
267        if !valid {
268            let error: actix_web::Error = AxiamApiError(AxiamError::AuthorizationDenied {
269                reason: "CSRF validation failed".into(),
270                action: None,
271                resource_id: None,
272            })
273            .into();
274            return Box::pin(async move {
275                let res = req.error_response(error);
276                Ok(res.map_into_right_body())
277            });
278        }
279
280        let fut = self.inner.call(req);
281        Box::pin(async move {
282            let res = fut.await?;
283            Ok(res.map_into_left_body())
284        })
285    }
286}

292/// Generate a cryptographically random CSRF token (32 bytes, hex-encoded).
293pub fn generate_csrf_token() -> String {
294    let bytes: [u8; 32] = rand::random();
295    hex::encode(bytes)
296}

298/// Build the `axiam_access` httpOnly cookie (per D-05).
299///
300/// - `httpOnly(true)` — not accessible from JavaScript
301/// - `Secure` — controlled by `cookie_secure`; must be `true` in production
302/// - `SameSite::Strict` — no cross-site sending
303/// - `path("/")` — all paths
304///
305/// `cookie_secure` should be read from `AuthConfig::cookie_secure` (D-18).
306pub fn access_cookie(token: &str, max_age_secs: u64, cookie_secure: bool) -> Cookie<'static> {
307    Cookie::build(COOKIE_ACCESS, token.to_owned())
308        .http_only(true)
309        .secure(cookie_secure)
310        .same_site(SameSite::Strict)
311        .path("/")
312        .max_age(Duration::seconds(max_age_secs as i64))
313        .finish()
314}
315
316/// Build the `axiam_refresh` httpOnly cookie (per D-06).
317///
318/// Path-scoped to the refresh endpoint to minimise exposure surface.
319///
320/// `cookie_secure` should be read from `AuthConfig::cookie_secure` (D-18).
321pub fn refresh_cookie(token: &str, max_age_secs: u64, cookie_secure: bool) -> Cookie<'static> {
322    Cookie::build(COOKIE_REFRESH, token.to_owned())
323        .http_only(true)
324        .secure(cookie_secure)
325        .same_site(SameSite::Strict)
326        .path("/api/v1/auth/refresh")
327        .max_age(Duration::seconds(max_age_secs as i64))
328        .finish()
329}

331/// Build the `axiam_csrf` JS-readable cookie (per D-07, D-09).
332///
333/// - `httpOnly(false)` — JavaScript must read this to send `X-CSRF-Token`
334/// - `Secure` — controlled by `cookie_secure`; must be `true` in production
335/// - `SameSite::Strict` — no cross-site sending
336/// - `path("/")` — all paths; lifetime matches the access token
337///
338/// `cookie_secure` should be read from `AuthConfig::cookie_secure` (D-18).
339pub fn csrf_cookie(token: &str, max_age_secs: u64, cookie_secure: bool) -> Cookie<'static> {
340    Cookie::build(COOKIE_CSRF, token.to_owned())
341        .http_only(false)
342        .secure(cookie_secure)
343        .same_site(SameSite::Strict)
344        .path("/")
345        .max_age(Duration::seconds(max_age_secs as i64))
346        .finish()
347}

349/// Build the `axiam_op_session` cookie — the OP browser session (W3, plan §4.0).
350///
351/// - `httpOnly(true)` — script must never read it; it buys authorization codes
352/// - `Secure` — **unconditionally `true`**, unlike the other three; see below
353/// - `SameSite::Lax` — **the load-bearing attribute**, see below
354/// - `path("/oauth2/authorize")` — the only endpoint that consults it
355/// - `Max-Age` = the session's, i.e. `AuthConfig::refresh_token_lifetime_secs`
356///
357/// # Why `Lax`, and why it must stay `Lax`
358///
359/// A browser sends a `Lax` cookie on **top-level GET navigations and nothing
360/// else**. That is precisely an OpenID Connect redirect from a relying party,
361/// and precisely *not* an `<iframe>`, an `<img>`, or a cross-site `fetch` —
362/// so the one thing this cookie exists to enable works, and the probing
363/// technique it would otherwise enable does not. An attacker's page cannot
364/// frame `/oauth2/authorize` and watch whether a session exists, because the
365/// cookie is not sent inside a frame at all.
366///
367/// The cost is stated rather than hidden, and it is recorded in plan §9: the
368/// cross-site hidden-iframe silent-renew pattern (`prompt=none` in an invisible
369/// frame) **is not supported and fails closed**. Relying parties renew with a
370/// top-level `prompt=none` navigation or with a refresh token. Changing this to
371/// `SameSite=None` would make that pattern work and would hand every site on
372/// the internet the same probe; it is not a fix and must not be made as one.
373///
374/// # Why a second cookie rather than relaxing `axiam_access`
375///
376/// `axiam_access` is `SameSite=Strict; Path=/` and opens the entire API. This
377/// one is scoped to a single endpoint, so a browser that sends it cross-site can
378/// obtain exactly one thing: an authorization code, for a client that is
379/// registered, at a `redirect_uri` that matched exactly, bound to the relying
380/// party's own PKCE and `state`. The API surface keeps its Strict cookies and
381/// its double-submit CSRF token untouched, and SEC-046's threat model for the
382/// API cookie is unchanged rather than re-argued.
383///
384/// # Why `Secure` is not `cookie_secure` here
385///
386/// The other three cookies take `AuthConfig::cookie_secure` (D-18), whose
387/// documented purpose is local HTTP development *"e.g. http://localhost"*. This
388/// one does not, for two reasons that only apply to it:
389///
390/// 1. It is the only `SameSite=Lax` cookie in the codebase, i.e. the only one a
391///    browser sends on a **cross-site** top-level navigation. That is the whole
392///    point of it — and it means a plaintext hop exposes it on a request the
393///    user never typed, in a context the three `Strict` cookies never reach.
394/// 2. The endpoint it is scoped to must be TLS-protected regardless: RFC 6749
395///    §3.1 requires TLS on the authorization endpoint, and this project's own
396///    standard is TLS 1.3 minimum for external communication. A cookie that
397///    refuses to exist over plaintext is enforcing a rule the endpoint already
398///    has, not adding one.
399///
400/// It costs nothing in the case D-18 exists for: browsers treat
401/// `http://localhost` and `http://127.0.0.1` as trustworthy origins and store
402/// `Secure` cookies set from them. What it does refuse is a browser login hop
403/// over plaintext to a *non-loopback* host — which is a deployment that should
404/// not be completing OpenID Connect authorization requests at all.
405pub fn op_session_cookie(token: &str, max_age_secs: u64) -> Cookie<'static> {
406    Cookie::build(COOKIE_OP_SESSION, token.to_owned())
407        .http_only(true)
408        .secure(true)
409        .same_site(SameSite::Lax)
410        .path("/oauth2/authorize")
411        .max_age(Duration::seconds(max_age_secs as i64))
412        .finish()
413}

415// A removal cookie is still a `Set-Cookie` the browser parses and stores until
416// it expires, so it must mirror **every** attribute of the cookie it clears —
417// not just `path`. Emitting a bare `Path=/` removal for a cookie that was set
418// `HttpOnly; Secure; SameSite=Strict` leaves the (empty-valued) cookie
419// JS-readable, cross-site-sendable and cleartext-transmissible for the life of
420// the response, and makes the removal itself depend on transport the setters
421// explicitly do not trust: browsers refuse to let a non-`Secure` cookie from an
422// insecure origin overwrite a `Secure` one ("Leave Secure Cookies Alone").
423//
424// Rather than restate each setter's attributes here — a second copy that can
425// drift from the first, which is precisely the bug being fixed — every removal
426// is built by calling its own setter and then expiring the result.
427// `Cookie::make_removal` rewrites only value, `Max-Age` and `Expires`, so
428// `HttpOnly`, `Secure`, `SameSite` and `Path` are mirrored by construction.
429
430/// Clear the `axiam_access` cookie (Max-Age=0, per D-08).
431///
432/// Built from [`access_cookie`], so it mirrors its attributes; `cookie_secure`
433/// must be the same `AuthConfig::cookie_secure` value used to set it (D-18).
434pub fn clear_access_cookie(cookie_secure: bool) -> Cookie<'static> {
435    let mut c = access_cookie("", 0, cookie_secure);
436    c.make_removal();
437    c
438}
439
440/// Clear the `axiam_refresh` cookie.
441///
442/// Built from [`refresh_cookie`], so it mirrors its attributes — including the
443/// `/api/v1/auth/refresh` path scope, since a removal on a different path would
444/// not match the cookie.
445pub fn clear_refresh_cookie(cookie_secure: bool) -> Cookie<'static> {
446    let mut c = refresh_cookie("", 0, cookie_secure);
447    c.make_removal();
448    c
449}
450
451/// Clear the `axiam_csrf` cookie.
452///
453/// Built from [`csrf_cookie`], so it mirrors its attributes — including
454/// `httpOnly(false)`, which is deliberate there (D-07: JavaScript reads this one
455/// to populate `X-CSRF-Token`) and is therefore kept here too.
456pub fn clear_csrf_cookie(cookie_secure: bool) -> Cookie<'static> {
457    let mut c = csrf_cookie("", 0, cookie_secure);
458    c.make_removal();
459    c
460}
461
462/// Clear the `axiam_op_session` cookie (W3).
463///
464/// Built from [`op_session_cookie`], so it mirrors its attributes — including
465/// the `/oauth2/authorize` path scope, without which the removal would not
466/// match the cookie and a logged-out browser would keep presenting an OP
467/// session at the authorization endpoint — and including its unconditional
468/// `Secure`, without which the removal could not overwrite it at all
469/// ("Leave Secure Cookies Alone").
470pub fn clear_op_session_cookie() -> Cookie<'static> {
471    let mut c = op_session_cookie("", 0);
472    c.make_removal();
473    c
474}

484    // The bearer exemption (B-05). Before it, a service account's
485    // `client_credentials` token could not POST to /api/v1/authz/check — the
486    // one REST surface `AuthenticatedPrincipal` was widened to accept the
487    // `axiam:m2m` audience for — because it has no cookie to echo and never

490    fn bearer_without_a_session_cookie_is_exempt() {
491        assert!(is_bearer_only(Some("Bearer eyJhbGciOi..."), false));
492        // Header names are case-insensitive and so is the scheme.
493        assert!(is_bearer_only(Some("bearer eyJhbGciOi..."), false));
494        assert!(is_bearer_only(Some("  Bearer eyJhbGciOi..."), false));
495    }

499        // The load-bearing case. A cross-site page cannot set the header on a
500        // victim's behalf, but it can cause a request that carries the victim's
501        // cookie — so a request with both must still prove it has the token.
502        assert!(!is_bearer_only(Some("Bearer eyJhbGciOi..."), true));
503    }

505    #[test]
506    fn a_request_with_no_bearer_is_never_exempt() {

617    /// **T0.6** — the OP browser-session cookie's attributes, pinned one by
618    /// one, because every one of them is doing a job:
619    ///
620    /// - `SameSite=Lax` is what defeats iframe probing and what makes the
621    ///   cross-site RP redirect work at all. Plan §9 records that the price is
622    ///   hidden-iframe silent renew, which fails closed. A future change to
623    ///   `None` would buy that pattern back and sell the probe with it.
624    /// - `Path=/oauth2/authorize` is what bounds a stolen cookie to producing
625    ///   authorization codes for registered clients rather than API calls.
626    /// - `HttpOnly` keeps it out of script.
627    /// - `Max-Age` is the session's, not the access token's: it names the same
628    ///   thing the session row does.
629    #[test]
630    fn t0_6_the_op_session_cookie_attributes_are_pinned() {
631        let c = op_session_cookie("browser-token", 86_400);
632        assert_eq!(c.name(), "axiam_op_session");
633        assert_eq!(
634            c.same_site(),
635            Some(SameSite::Lax),
636            "Lax is deliberate: it is sent on a top-level navigation and not \
637             inside a frame. Do not 'fix' this to None — see plan §9."
638        );
639        assert_eq!(
640            c.path(),
641            Some("/oauth2/authorize"),
642            "the cookie must reach exactly one endpoint"
643        );
644        assert!(
645            c.http_only().unwrap_or(false),
646            "must not be script-readable"
647        );
648        assert!(c.secure().unwrap_or(false));
649        assert_eq!(c.max_age(), Some(Duration::seconds(86_400)));
650    }

673    /// A removal cookie is a `Set-Cookie` in its own right: the browser stores
674    /// what it says until it expires. If it does not carry the same flags as
675    /// the cookie it clears, the empty-valued replacement is weaker than the
676    /// value it replaced — and a non-`Secure` removal cannot overwrite a
677    /// `Secure` cookie from an insecure origin at all.
678    #[test]
679    fn removal_cookies_mirror_the_attributes_of_the_cookies_they_clear() {
680        for secure in [true, false] {
681            let pairs = [
682                (
683                    access_cookie("tok", 900, secure),
684                    clear_access_cookie(secure),
685                ),
686                (
687                    refresh_cookie("tok", 86400, secure),
688                    clear_refresh_cookie(secure),
689                ),
690                (csrf_cookie("tok", 900, secure), clear_csrf_cookie(secure)),
691                (op_session_cookie("tok", 86400), clear_op_session_cookie()),
692            ];
693
694            for (set, clear) in pairs {
695                let name = set.name().to_owned();
696                assert_eq!(clear.name(), name, "removal must target the same cookie");
697                assert_eq!(
698                    clear.path(),
699                    set.path(),
700                    "{name}: a removal on a different path does not match the cookie"
701                );
702                assert_eq!(
703                    clear.secure(),
704                    set.secure(),
705                    "{name}: removal Secure must match the setter's (secure={secure})"
706                );
707                assert_eq!(
708                    clear.http_only(),
709                    set.http_only(),
710                    "{name}: removal HttpOnly must match the setter's"
711                );
712                assert_eq!(
713                    clear.same_site(),
714                    set.same_site(),
715                    "{name}: removal SameSite must match the setter's"
716                );
717            }
718        }
719    }
