1//! CSRF double-submit cookie middleware and cookie builder helpers.
2//!
3//! The [`CsrfMiddleware`] rejects state-changing requests that lack a valid
4//! `X-CSRF-Token` header matching the `axiam_csrf` cookie value.
5//! Comparison uses constant-time equality to prevent timing attacks (D-01).
6//!
7//! Cookie helpers build the three auth cookies (`axiam_access`,
8//! `axiam_refresh`, `axiam_csrf`) with the security attributes specified
9//! in the UI-SPEC design decisions (D-05 through D-09).
10

29pub const COOKIE_REFRESH: &str = "axiam_refresh";

33// ---------------------------------------------------------------------------
34// Exempt path suffixes — no CSRF token needed on these endpoints
35// ---------------------------------------------------------------------------
36
37/// Path suffixes that are exempt from CSRF validation.
38///
39/// These are either unauthenticated endpoints (login, MFA flows) that do not
40/// yet have a CSRF cookie, or token-based OAuth2 flows that use their own
41/// security model.
42const CSRF_EXEMPT_SUFFIXES: &[&str] = &[
43    "/api/v1/auth/login",
44    // The OPAQUE endpoints are exempt on exactly the same grounds as /login:
45    // the caller is unauthenticated and has no `axiam_csrf` cookie to echo.
46    // This is a separate registry from `permissions::PUBLIC_PATHS` and both
47    // must cover a route for it to work unauthenticated — omitting it here
48    // gives a 403 before the handler ever runs.
49    "/api/v1/auth/opaque/register/start",
50    "/api/v1/auth/opaque/login/start",
51    "/api/v1/auth/opaque/login/finish",
52    "/api/v1/auth/mfa/verify",
53    "/api/v1/auth/mfa/setup/enroll",
54    "/api/v1/auth/mfa/setup/confirm",
55    // WebAuthn **authentication** — the passkey and security-key equivalents of
56    // /login, and unauthenticated for the same reason: these ceremonies *are*
57    // the authentication, so the caller holds no session and has no
58    // `axiam_csrf` cookie to echo. They were listed in `permissions::
59    // PUBLIC_PATHS` but not here, and both registries must cover a route: the
60    // result was a `403 CSRF validation failed` on
61    // `/webauthn/authenticate/start` before the handler ever ran, which made
62    // passkey sign-in impossible from a browser with no prior session.
63    //
64    // The **registration** pair is deliberately NOT exempt. Adding a passkey is
65    // done by a user who is already signed in and therefore does carry the
66    // cookie, and an exemption there would let any site silently enrol its own
67    // authenticator onto a logged-in victim's account — account takeover by
68    // exactly the request forgery this middleware exists to stop.
69    "/api/v1/auth/webauthn/authenticate/start",
70    "/api/v1/auth/webauthn/authenticate/finish",
71    "/api/v1/auth/webauthn/authenticate/discoverable/start",
72    "/api/v1/auth/webauthn/authenticate/discoverable/finish",
73    "/api/v1/auth/device",
74    // Password reset request + confirm are unauthenticated and token-based:
75    // the caller has no session and therefore no CSRF cookie yet (same model
76    // as /login). Without these, a forgotten-password reset is CSRF-blocked (403).
77    "/api/v1/auth/reset",
78    "/api/v1/auth/reset/confirm",
79    // Bootstrap is a one-time endpoint called before any session exists.
80    // It is protected by the AXIAM_BOOTSTRAP_ADMIN_EMAIL env gate instead (D-10).
81    "/api/v1/admin/bootstrap",
82    // 28-05/CQ-B40: the four first-time-SSO public endpoints (D-22,
83    // handlers/federation.rs) are unauthenticated by design — same model as
84    // /login and /reset/confirm above: the caller has no prior session and
85    // therefore no `axiam_csrf` cookie to echo back. Without these entries a
86    // first-time OIDC/SAML SSO login is CSRF-blocked (403) before the
87    // handler ever runs, even though the route is correctly listed in
88    // PUBLIC_PATHS (AuthzMiddleware bypass is a separate registry from this
89    // one — both must cover a route for it to work unauthenticated).
90    "/api/v1/auth/federation/oidc/start",
91    "/api/v1/auth/federation/oidc/callback",
92    "/api/v1/auth/federation/saml/login",
93    "/api/v1/auth/federation/saml/acs",
94    // The login-provider surface added with "Sign in with X", exempt for the
95    // same reason as the four above: none of these callers has a prior session,
96    // so none has an `axiam_csrf` cookie to echo.
97    //
98    // The two `…/form` entries additionally *cannot* carry one even in
99    // principle — the request is a cross-site form POST performed by the
100    // identity provider. What stands in for CSRF protection there is the
101    // provider's own signed assertion plus the single-use server-side state
102    // row, both of which an attacker would have to forge.
103    "/api/v1/auth/federation/oauth2/start",
104    "/api/v1/auth/federation/oauth2/callback",
105    "/api/v1/auth/federation/oidc/callback/form",
106    "/api/v1/auth/federation/saml/acs/form",
107    // Redeeming a handoff code is same-origin but pre-session: the response
108    // being requested is the one that *sets* the CSRF cookie.
109    "/api/v1/auth/federation/handoff",
110];
111
112/// Path prefixes that are exempt from CSRF validation (OAuth2).
113const CSRF_EXEMPT_PREFIXES: &[&str] = &["/oauth2/"];
114
115fn is_csrf_exempt(path: &str) -> bool {
116    for suffix in CSRF_EXEMPT_SUFFIXES {
117        if path.ends_with(suffix) {
118            return true;
119        }
120    }
121    for prefix in CSRF_EXEMPT_PREFIXES {
122        if path.starts_with(prefix) {
123            return true;
124        }
125    }
126    false
127}

133/// Whether a request authenticates by bearer token ALONE.
134///
135/// Split out as a pure function so the condition can be pinned by tests: it is
136/// the exemption that decides whether CSRF validation runs at all, and the case
137/// that matters is the one where BOTH are present.
138///
139/// * bearer + no session cookie → exempt. There is no ambient credential to
140///   abuse and no double-submit cookie the caller could echo.
141/// * bearer + a session cookie → **not** exempt. That is precisely the shape a
142///   cross-site attacker would craft to escape the exemption: the browser
143///   supplies the cookie, the attacker supplies the header.
144/// * no bearer → not exempt, whatever the cookies say.
145fn is_bearer_only(authorization: Option<&str>, has_session_cookie: bool) -> bool {
146    if has_session_cookie {
147        return false;
148    }
149    authorization.is_some_and(|v| v.trim_start().to_ascii_lowercase().starts_with("bearer "))
150}

152/// CSRF double-submit cookie middleware.
153///
154/// Safe methods (GET, HEAD, OPTIONS) pass through unconditionally.
155/// State-changing methods require `X-CSRF-Token` to match the `axiam_csrf`
156/// cookie value, compared with constant-time equality.
157pub struct CsrfMiddleware;
158
159impl<S, B> Transform<S, ServiceRequest> for CsrfMiddleware
160where
161    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
162    B: 'static,
163{
164    type Response = ServiceResponse<EitherBody<B>>;
165    type Error = Error;
166    type Transform = CsrfMiddlewareService<S>;
167    type InitError = ();
168    type Future = Ready<Result<Self::Transform, Self::InitError>>;
169
170    fn new_transform(&self, service: S) -> Self::Future {
171        ready(Ok(CsrfMiddlewareService { inner: service }))
172    }
173}
174
175pub struct CsrfMiddlewareService<S> {
176    inner: S,
177}
178
179impl<S, B> Service<ServiceRequest> for CsrfMiddlewareService<S>
180where
181    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
182    B: 'static,
183{
184    type Response = ServiceResponse<EitherBody<B>>;
185    type Error = Error;
186    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>>>>;
187
188    fn poll_ready(
189        &self,
190        cx: &mut std::task::Context<'_>,
191    ) -> std::task::Poll<Result<(), Self::Error>> {
192        self.inner.poll_ready(cx)
193    }
194
195    fn call(&self, req: ServiceRequest) -> Self::Future {
196        // Safe methods — always exempt.
197        let method = req.method().clone();
198        if method == Method::GET || method == Method::HEAD || method == Method::OPTIONS {
199            let fut = self.inner.call(req);
200            return Box::pin(async move {
201                let res = fut.await?;
202                Ok(res.map_into_left_body())
203            });
204        }
205
206        // Exempt paths (login, MFA flows, OAuth2).
207        let path = req.path().to_owned();
208        if is_csrf_exempt(&path) {
209            let fut = self.inner.call(req);
210            return Box::pin(async move {
211                let res = fut.await?;
212                Ok(res.map_into_left_body())
213            });
214        }
215
216        // Bearer-only callers carry no ambient credential, so there is nothing
217        // for CSRF to protect and nothing they could echo back.
218        //
219        // CSRF is an attack on credentials the browser attaches BY ITSELF — the
220        // session cookie. A request authenticated solely by an `Authorization`
221        // header has no such credential: a cross-site page cannot set that
222        // header on a victim's behalf, and the double-submit cookie it would
223        // have to echo does not exist. Demanding one is unsatisfiable rather
224        // than protective, and it made the machine-facing surface unreachable
225        // for exactly the callers it was built for — a service account's
226        // `client_credentials` token could not `POST /api/v1/authz/check`,
227        // whose `AuthenticatedPrincipal` extractor exists to accept the
228        // `axiam:m2m` audience.
229        //
230        // The condition is deliberately narrow: the session cookie must be
231        // ABSENT as well. A request that carries both a session cookie and a
232        // bearer header is exactly the shape a cross-site attacker would craft
233        // to escape this exemption — the browser supplies the cookie, the
234        // attacker supplies the header — so it stays subject to validation.
235        let has_session_cookie = req.cookie(COOKIE_ACCESS).is_some()
236            || req.cookie(COOKIE_REFRESH).is_some()
237            || req.cookie(COOKIE_CSRF).is_some();
238        let auth_header = req
239            .headers()
240            .get(actix_web::http::header::AUTHORIZATION)
241            .and_then(|v| v.to_str().ok());
242        if is_bearer_only(auth_header, has_session_cookie) {
243            let fut = self.inner.call(req);
244            return Box::pin(async move {
245                let res = fut.await?;
246                Ok(res.map_into_left_body())
247            });
248        }
249
250        // Extract CSRF cookie and header.
251        let cookie_value = req.cookie(COOKIE_CSRF).map(|c| c.value().to_owned());
252
253        let header_value = req
254            .headers()
255            .get(HEADER_CSRF)
256            .and_then(|v| v.to_str().ok())
257            .map(|s| s.to_owned());
258
259        // Validate — both must be present and equal (constant-time).
260        let valid = match (cookie_value, header_value) {
261            (Some(cookie), Some(header)) => cookie.as_bytes().ct_eq(header.as_bytes()).into(),
262            _ => false,
263        };
264
265        if !valid {
266            let error: actix_web::Error = AxiamApiError(AxiamError::AuthorizationDenied {
267                reason: "CSRF validation failed".into(),
268                action: None,
269                resource_id: None,
270            })
271            .into();
272            return Box::pin(async move {
273                let res = req.error_response(error);
274                Ok(res.map_into_right_body())
275            });
276        }
277
278        let fut = self.inner.call(req);
279        Box::pin(async move {
280            let res = fut.await?;
281            Ok(res.map_into_left_body())
282        })
283    }
284}

290/// Generate a cryptographically random CSRF token (32 bytes, hex-encoded).
291pub fn generate_csrf_token() -> String {
292    let bytes: [u8; 32] = rand::random();
293    hex::encode(bytes)
294}

296/// Build the `axiam_access` httpOnly cookie (per D-05).
297///
298/// - `httpOnly(true)` — not accessible from JavaScript
299/// - `Secure` — controlled by `cookie_secure`; must be `true` in production
300/// - `SameSite::Strict` — no cross-site sending
301/// - `path("/")` — all paths
302///
303/// `cookie_secure` should be read from `AuthConfig::cookie_secure` (D-18).
304pub fn access_cookie(token: &str, max_age_secs: u64, cookie_secure: bool) -> Cookie<'static> {
305    Cookie::build(COOKIE_ACCESS, token.to_owned())
306        .http_only(true)
307        .secure(cookie_secure)
308        .same_site(SameSite::Strict)
309        .path("/")
310        .max_age(Duration::seconds(max_age_secs as i64))
311        .finish()
312}
313
314/// Build the `axiam_refresh` httpOnly cookie (per D-06).
315///
316/// Path-scoped to the refresh endpoint to minimise exposure surface.
317///
318/// `cookie_secure` should be read from `AuthConfig::cookie_secure` (D-18).
319pub fn refresh_cookie(token: &str, max_age_secs: u64, cookie_secure: bool) -> Cookie<'static> {
320    Cookie::build(COOKIE_REFRESH, token.to_owned())
321        .http_only(true)
322        .secure(cookie_secure)
323        .same_site(SameSite::Strict)
324        .path("/api/v1/auth/refresh")
325        .max_age(Duration::seconds(max_age_secs as i64))
326        .finish()
327}
328
329/// Build the `axiam_csrf` JS-readable cookie (per D-07, D-09).
330///
331/// - `httpOnly(false)` — JavaScript must read this to send `X-CSRF-Token`
332/// - `Secure` — controlled by `cookie_secure`; must be `true` in production
333/// - `SameSite::Strict` — no cross-site sending
334/// - `path("/")` — all paths; lifetime matches the access token
335///
336/// `cookie_secure` should be read from `AuthConfig::cookie_secure` (D-18).
337pub fn csrf_cookie(token: &str, max_age_secs: u64, cookie_secure: bool) -> Cookie<'static> {
338    Cookie::build(COOKIE_CSRF, token.to_owned())
339        .http_only(false)
340        .secure(cookie_secure)
341        .same_site(SameSite::Strict)
342        .path("/")
343        .max_age(Duration::seconds(max_age_secs as i64))
344        .finish()
345}

347// A removal cookie is still a `Set-Cookie` the browser parses and stores until
348// it expires, so it must mirror **every** attribute of the cookie it clears —
349// not just `path`. Emitting a bare `Path=/` removal for a cookie that was set
350// `HttpOnly; Secure; SameSite=Strict` leaves the (empty-valued) cookie
351// JS-readable, cross-site-sendable and cleartext-transmissible for the life of
352// the response, and makes the removal itself depend on transport the setters
353// explicitly do not trust: browsers refuse to let a non-`Secure` cookie from an
354// insecure origin overwrite a `Secure` one ("Leave Secure Cookies Alone").
355//
356// Rather than restate each setter's attributes here — a second copy that can
357// drift from the first, which is precisely the bug being fixed — every removal
358// is built by calling its own setter and then expiring the result.
359// `Cookie::make_removal` rewrites only value, `Max-Age` and `Expires`, so
360// `HttpOnly`, `Secure`, `SameSite` and `Path` are mirrored by construction.
361
362/// Clear the `axiam_access` cookie (Max-Age=0, per D-08).
363///
364/// Built from [`access_cookie`], so it mirrors its attributes; `cookie_secure`
365/// must be the same `AuthConfig::cookie_secure` value used to set it (D-18).
366pub fn clear_access_cookie(cookie_secure: bool) -> Cookie<'static> {
367    let mut c = access_cookie("", 0, cookie_secure);
368    c.make_removal();
369    c
370}
371
372/// Clear the `axiam_refresh` cookie.
373///
374/// Built from [`refresh_cookie`], so it mirrors its attributes — including the
375/// `/api/v1/auth/refresh` path scope, since a removal on a different path would
376/// not match the cookie.
377pub fn clear_refresh_cookie(cookie_secure: bool) -> Cookie<'static> {
378    let mut c = refresh_cookie("", 0, cookie_secure);
379    c.make_removal();
380    c
381}
382
383/// Clear the `axiam_csrf` cookie.
384///
385/// Built from [`csrf_cookie`], so it mirrors its attributes — including
386/// `httpOnly(false)`, which is deliberate there (D-07: JavaScript reads this one
387/// to populate `X-CSRF-Token`) and is therefore kept here too.
388pub fn clear_csrf_cookie(cookie_secure: bool) -> Cookie<'static> {
389    let mut c = csrf_cookie("", 0, cookie_secure);
390    c.make_removal();
391    c
392}

512    /// A removal cookie is a `Set-Cookie` in its own right: the browser stores
513    /// what it says until it expires. If it does not carry the same flags as

515    /// value it replaced — and a non-`Secure` removal cannot overwrite a
516    /// `Secure` cookie from an insecure origin at all.

520            let pairs = [
521                (

523                    clear_access_cookie(secure),
524                ),
525                (
526                    refresh_cookie("tok", 86400, secure),
527                    clear_refresh_cookie(secure),
528                ),
529                (csrf_cookie("tok", 900, secure), clear_csrf_cookie(secure)),
530            ];
531
532            for (set, clear) in pairs {
533                let name = set.name().to_owned();
534                assert_eq!(clear.name(), name, "removal must target the same cookie");
535                assert_eq!(
536                    clear.path(),
537                    set.path(),

540                assert_eq!(
