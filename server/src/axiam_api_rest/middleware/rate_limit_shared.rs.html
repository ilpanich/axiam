1//! Shared (cross-replica) rate-limit pre-check middleware
2//! (SECHRD-03 / D-01a, D-01b, D-01d).
3//!
4//! [`RateLimitShared`] consults the process-wide write-behind
5//! [`SharedRateLimitCounter`] (`axiam-db::rate_limit_counter`) BEFORE the
6//! existing per-replica in-memory `Governor`/`GovernorLayer`
7//! (`server.rs::build_governor`, kept byte-for-byte unchanged as the
8//! fail-open fallback). This closes the multi-replica HPA gap where
9//! per-replica in-memory buckets otherwise multiply the effective rate
10//! limit by the replica count.
11//!
12//! # No synchronous datastore write on the request path (H2 fix)
13//!
14//! This middleware used to `await` ONE SurrealDB `UPSERT`
15//! (`SurrealRateLimitBucketRepository::increment`) per request, before the
16//! handler ran. That single write capped every wrapped endpoint at **16–21
17//! ops/s at any concurrency from 1 to 40 VUs** (p50 ≈ 55–65 ms at 1 VU,
18//! ≈ 1 s at 20 VUs) while structurally identical unwrapped endpoints ran at
19//! 68–4 248 ops/s — the datastore attributed ~39–46 ms of each request to
20//! that one RPC, and the same UPSERT issued directly ran in 6–9 ms and
21//! scaled fine, so the ordering (not the datastore) was the ceiling. Full
22//! measurements: `claude_dev/postseed-transient-investigation.md` §2, §3, §7.
23//!
24//! The counting mechanism — and ONLY the counting mechanism — changed: the
25//! key derivation (`{endpoint}:{key_part}`, the D8 `client_id` modes, the
26//! body peek/restore), the 429 response shape and every fail-open branch
27//! behave exactly as before. [`SharedRateLimitCounter::check`] is fully
28//! synchronous (one sharded-map increment, no datastore round trip, no task
29//! spawn) and a single background flusher coalesces the accumulated
30//! increments into one datastore write per bucket per
31//! `AXIAM__RATE_LIMIT__SHARED_SYNC_MS`. Cross-replica enforcement is
32//! therefore *eventual*, with the overshoot bound stated precisely in the
33//! [`axiam_db::rate_limit_counter`] module docs; the in-memory governor still
34//! makes a full per-replica decision on the very same endpoints.
35//!
36//! **Fail-open (D-01b, T-24-42 accepted risk):** when no client-IP key can be
37//! derived, no [`AppState`] (hence no counter) is registered, or the shared
38//! layer is disabled via `AXIAM__RATE_LIMIT__SHARED=off`, this middleware
39//! forwards the request unchanged so the existing in-memory governor makes
40//! the decision instead. Store outages are now surfaced by the counter's
41//! flusher (a `warn` alarm that never includes the raw key — T-24-43) while
42//! decisions keep being served from local counts. A counter-store outage must
43//! never hard-block auth traffic — this is the ONE deliberate fail-open
44//! exception; every other control fails closed.
45//!
46//! **CRITICAL (RESEARCH Pitfall 1):** `governor::StateStore::measure_and_replace`
47//! is a *synchronous* trait method. This middleware is deliberately NOT a
48//! `StateStore` implementation and never calls `block_on`/`futures::executor::
49//! block_on` — it is a separate Actix `Transform`/`Service` (mirroring
50//! [`crate::middleware::authz::AuthzMiddleware`]'s scaffold). It stays async
51//! only because the D8 body peek (`req.extract::<web::Bytes>()`) is async;
52//! the rate-limit decision itself no longer awaits anything.
53

73/// Fixed-window duration (seconds) for the shared bucket. A simple
74/// fixed-window counter is acceptable here (unlike the in-memory GCRA
75/// governor) because this layer only needs to be *approximately* right —
76/// it fails open by design (RESEARCH.md "Don't Hand-Roll").
77const WINDOW_SECS: i64 = 60;
78
79/// Reads the SAME `AXIAM__RATE_LIMIT__TRUSTED_HOPS` env var
80/// `server.rs::build_governor` uses, so the shared-store bucket key and the
81/// in-memory governor's key are derived from the identical client IP
82/// (D-01d parity — a rotating XFF must not yield a fresh bucket in either
83/// layer).
84fn trusted_hops() -> usize {
85    std::env::var("AXIAM__RATE_LIMIT__TRUSTED_HOPS")
86        .ok()
87        .and_then(|v| v.parse().ok())
88        .unwrap_or(0)
89}
90
91/// Truncates `now` down to the start of the current fixed
92/// [`WINDOW_SECS`]-second window.
93///
94/// The request path no longer calls this — it hands the raw `now` to
95/// [`axiam_db::SharedRateLimitCounter::check_at`], which derives the window
96/// itself so it can also see the *offset* into it (run-5 J1, the
97/// sliding-window bound). Kept, and asserted by a test, because
98/// [`WINDOW_SECS`] must keep agreeing with the counter's configured window:
99/// the two are one contract split across two crates.
100#[cfg_attr(not(test), allow(dead_code))]
101fn window_start(now: DateTime<Utc>) -> DateTime<Utc> {
102    let epoch = now.timestamp();
103    let start_epoch = epoch - epoch.rem_euclid(WINDOW_SECS);
104    DateTime::<Utc>::from_timestamp(start_epoch, 0).unwrap_or(now)
105}
106
107/// Builds the same-shaped 429 response the in-memory governor returns
108/// (`extractors::rate_limit::XForwardedForKeyExtractor::exceed_rate_limit_response`),
109/// so clients see one consistent rate-limit contract regardless of which
110/// layer rejected the request.
111fn too_many_requests_response() -> HttpResponse {
112    HttpResponse::TooManyRequests()
113        .content_type(ContentType::json())
114        .insert_header((RETRY_AFTER, WINDOW_SECS.to_string()))
115        .body(r#"{"error":"rate_limit_exceeded"}"#)
116}

122/// Shared SurrealDB-backed rate-limit pre-check middleware.
123///
124/// Generic over the SurrealDB connection type `C` so it can be wired
125/// against both the production `DbClient` and the in-memory `Mem` engine
126/// used in tests. `endpoint` MUST be unique per rate-limited resource so
127/// the shared bucket key (`"{endpoint}:{ip}"`) preserves per-endpoint
128/// granularity — never collapse distinct endpoints into one global bucket.
129///
130/// Wire this BEFORE (i.e. `.wrap()` AFTER, since the last `.wrap()` call is
131/// the outermost layer and therefore executes first) `build_governor(...)`
132/// on the same resource, e.g.:
133///
134/// ```ignore
135/// web::resource("/login")
136///     .wrap(build_governor(rate_limit_cfg.login_per_min))
137///     .wrap(RateLimitShared::<C>::new("login", rate_limit_cfg.login_per_min))
138///     .route(web::post().to(handlers::auth::login::<C>))
139/// ```
140pub struct RateLimitShared<C: Connection + Clone> {
141    endpoint: &'static str,
142    limit: u32,
143    /// D8: bucket-key derivation mode. Always [`RateLimitKeyMode::Ip`] for
144    /// instances built via [`RateLimitShared::new`] — only
145    /// [`RateLimitShared::new_client_identity_aware`] ever sets this to
146    /// something else.
147    key_mode: RateLimitKeyMode,
148    /// D8: whether this resource has a form-encoded OAuth2 `client_id` to
149    /// peek at (`/oauth2/token`, `/oauth2/revoke`, `/oauth2/introspect`
150    /// ONLY — see [`RateLimitShared::new_client_identity_aware`]). `false`
151    /// for every other resource (including `/auth/login`), which never
152    /// touches the request body and always keys per-IP.
153    client_identity_aware: bool,
154    _marker: PhantomData<C>,
155}
156
157impl<C: Connection + Clone> RateLimitShared<C> {
158    /// Plain IP-keyed constructor — unchanged since before D8. Use this for
159    /// every endpoint EXCEPT `/oauth2/token`, `/oauth2/revoke`, and
160    /// `/oauth2/introspect`. In particular, `/auth/login` MUST keep using
161    /// this constructor: it authenticates a user via username/password, so
162    /// there is no OAuth2 `client_id` in the request to key on, and this
163    /// type never reads `AXIAM__RATE_LIMIT__KEY` for it.
164    pub fn new(endpoint: &'static str, limit: u32) -> Self {
165        Self {
166            endpoint,
167            limit,
168            key_mode: RateLimitKeyMode::Ip,
169            client_identity_aware: false,
170            _marker: PhantomData,
171        }
172    }
173
174    /// D8 constructor for the three endpoints where an OAuth2 client
175    /// authenticates itself via a form-encoded `client_id`
176    /// (`client_secret_post`, RFC 6749 §2.3.1): `/oauth2/token`,
177    /// `/oauth2/revoke`, `/oauth2/introspect`.
178    ///
179    /// When `key_mode` is [`RateLimitKeyMode::ClientId`] or
180    /// [`RateLimitKeyMode::IpClientId`], this middleware peeks (and
181    /// restores) the request body to read `client_id`, and stashes it into
182    /// the request extensions (as [`RateLimitClientId`]) so the downstream
183    /// in-memory `Governor`'s `ClientAwareKeyExtractor` on the SAME
184    /// resource can reuse it without a second body read. When `key_mode` is
185    /// [`RateLimitKeyMode::Ip`] this is exactly as if `new()` had been
186    /// called — the body is never touched.
187    ///
188    /// **Do not use this constructor for `/auth/login` or any other
189    /// endpoint without a client identity** — see
190    /// [`crate::config::rate_limit::RateLimitKeyMode`] docs.
191    pub fn new_client_identity_aware(
192        endpoint: &'static str,
193        limit: u32,
194        key_mode: RateLimitKeyMode,
195    ) -> Self {
196        Self {
197            endpoint,
198            limit,
199            key_mode,
200            client_identity_aware: true,
201            _marker: PhantomData,
202        }
203    }
204}

246pub struct RateLimitSharedService<S, C: Connection + Clone> {
247    inner: Rc<S>,
248    endpoint: &'static str,
249    limit: u32,
250    key_mode: RateLimitKeyMode,
251    client_identity_aware: bool,
252    _marker: PhantomData<C>,
253}
254
255impl<S, B, C> Service<ServiceRequest> for RateLimitSharedService<S, C>
256where
257    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
258    B: 'static,
259    C: Connection + Clone + 'static,
260{
261    type Response = ServiceResponse<EitherBody<B>>;
262    type Error = Error;
263    type Future = std::pin::Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>>>>;
264
265    fn poll_ready(
266        &self,
267        cx: &mut std::task::Context<'_>,
268    ) -> std::task::Poll<Result<(), Self::Error>> {
269        self.inner.poll_ready(cx)
270    }
271
272    fn call(&self, mut req: ServiceRequest) -> Self::Future {
273        let endpoint = self.endpoint;
274        let limit = self.limit;
275        let key_mode = self.key_mode;
276        let client_identity_aware = self.client_identity_aware;
277        let inner = Rc::clone(&self.inner);
278
279        // Reuse the EXACT fixed IP-key extraction logic (D-01d) so the
280        // shared bucket key and the in-memory governor's key agree.
281        let extractor = XForwardedForKeyExtractor::with_trusted_hops(trusted_hops());
282        let ip = extractor.extract(&req).ok();
283
284        // The process-wide write-behind counter is read from
285        // `web::Data<AppState<C>>` (QUAL-01 — was a standalone
286        // `web::Data<Surreal<C>>` registration used to build a repository per
287        // request). Its absence is treated exactly like the old missing-DB-
288        // handle case below — fail open to the in-memory governor.
289        //
290        // This is a cheap `Arc` clone of the ONE counter registered in
291        // `main.rs`, never a new counter: a per-request (or per-worker)
292        // instance would fragment the local count.
293        let counter = req
294            .app_data::<web::Data<AppState<C>>>()
295            .map(|d| d.shared_rate_limit.clone());
296
297        Box::pin(async move {
298            // D8: for the client-identity-aware endpoints only
299            // (`/oauth2/token`, `/oauth2/revoke`, `/oauth2/introspect` —
300            // never `/auth/login`), peek the form-encoded body for
301            // `client_id` and restore the payload so the handler's
302            // `web::Form<..>` extractor still sees the full body
303            // untouched. This ALSO stashes the result into request
304            // extensions for the downstream in-memory `Governor`'s
305            // `ClientAwareKeyExtractor` to reuse (see
306            // `extractors::rate_limit::RateLimitClientId`).
307            //
308            // Skipped entirely in `ip` mode (the default) — no body read,
309            // byte-for-byte the pre-D8 code path.
310            let client_id = if client_identity_aware {
311                let client_id = if key_mode != RateLimitKeyMode::Ip {
312                    let bytes = req.extract::<web::Bytes>().await.unwrap_or_default();
313                    let client_id = extract_form_client_id(&bytes);
314                    // Restore the body EXACTLY as read so the handler's
315                    // `web::Form<..>` extraction downstream is unaffected —
316                    // this middleware must be transparent to the request.
317                    req.set_payload(Payload::from(bytes));
318                    client_id
319                } else {
320                    None
321                };
322                req.extensions_mut()
323                    .insert(RateLimitClientId(client_id.clone()));
324                client_id
325            } else {
326                None
327            };
328
329            // The bucket-key "identity" part: IP alone (`ip` mode, or
330            // fail-safe fallback when no client_id was found), the
331            // client_id alone, or the `(ip, client_id)` pair. Mirrors
332            // `extractors::rate_limit::ClientAwareKeyExtractor` exactly so
333            // this shared-store pre-check and the in-memory governor never
334            // disagree about which bucket a request belongs to.
335            let key_part = match (key_mode, &client_id) {
336                (RateLimitKeyMode::Ip, _) | (_, None) => ip.map(|ip| ip.to_string()),
337                (RateLimitKeyMode::ClientId, Some(cid)) => Some(format!("client:{cid}")),
338                (RateLimitKeyMode::IpClientId, Some(cid)) => {
339                    ip.map(|ip| format!("{ip}:client:{cid}"))
340                }
341            };
342
343            let now = Utc::now();
344            // Carried past the decision so a downstream rejection can be
345            // refunded against the SAME key and window (run-5 J1).
346            let charged = match (key_part, counter) {
347                (Some(key_part), Some(counter)) => {
348                    // Same bucket key as before — `{endpoint}:{key_part}` —
349                    // so an in-flight upgrade keeps counting against the same
350                    // `rate_limit_bucket` records.
351                    let key = format!("{endpoint}:{key_part}");
352                    // Fully synchronous: no datastore round trip on the
353                    // request path (H2 fix). Store outages are reported by
354                    // the counter's background flusher, which keeps serving
355                    // decisions from local counts (fail open, D-01b).
356                    //
357                    // `check_at` (not `check`) so the counter sees where
358                    // inside the window the request landed and can apply the
359                    // sliding-window bound that closes the run-5 J1
360                    // over-admission (`axiam_db::rate_limit_counter::WindowMode`).
361                    let allow = counter.check_at(&key, now, limit);
362                    Some((counter, key, allow))
363                }
364                // No key part (IP unavailable) or no counter registered —
365                // fail open; the in-memory governor still makes the real
366                // decision. (A counter built with
367                // `AXIAM__RATE_LIMIT__SHARED=off` also always allows, from
368                // inside `check_at`.)
369                _ => None,
370            };
371
372            let allow = charged.as_ref().is_none_or(|(_, _, allow)| *allow);
373
374            if allow {
375                let res = inner.call(req).await?;
376                // Run-5 J1: the in-memory governor sits INSIDE this
377                // middleware and may reject with 429 a request this layer
378                // already charged. Charging capacity to a request that was
379                // never served is what starved the gRPC families down to
380                // 1/33 of their ceiling; refund it so the cross-replica
381                // counter tracks *served* traffic. See
382                // `SharedRateLimitCounter::refund`.
383                if res.status() == StatusCode::TOO_MANY_REQUESTS
384                    && let Some((counter, key, true)) = &charged
385                {
386                    counter.refund(key, now);
387                }
388                Ok(res.map_into_left_body())
389            } else {
390                let res = req.into_response(too_many_requests_response());
391                Ok(res.map_into_right_body())
392            }
393        })
394    }
395}

397#[cfg(test)]
398mod window_contract_tests {
399    use super::*;
400
401    /// [`WINDOW_SECS`] and the shared counter's configured window are one
402    /// contract split across two crates: this middleware hands
403    /// `SharedRateLimitCounter::check_at` a raw `now` and lets the counter
404    /// derive the window, so if the two ever disagreed the `Retry-After` this
405    /// middleware advertises would be a lie about a window nobody uses.
406    #[test]
407    fn window_secs_matches_the_shared_counters_window() {
408        let counter_window = axiam_db::SharedRateLimitConfig::default().window;
409        assert_eq!(
410            counter_window.as_secs() as i64,
411            WINDOW_SECS,
412            "REST WINDOW_SECS must equal the shared counter's window"
413        );
414    }
415
416    /// The local truncation helper must agree with the counter's, so a
417    /// `Retry-After` computed here points at the same boundary the counter
418    /// will roll over on.
419    #[test]
420    fn window_start_truncates_to_the_window_boundary() {
421        let now = DateTime::<Utc>::from_timestamp(1_800_000_045, 0).expect("valid epoch");
422        let start = window_start(now);
423
424        assert_eq!(start.timestamp() % WINDOW_SECS, 0, "aligned to the window");
425        assert!(start <= now && now.timestamp() - start.timestamp() < WINDOW_SECS);
426    }
