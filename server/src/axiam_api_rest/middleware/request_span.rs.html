1//! The request tracer's root span, with credentials kept out of the target
2//! (F4 P23W3-03, closing T-325).
3//!
4//! `tracing-actix-web`'s [`DefaultRootSpanBuilder`] records `http.target` as
5//! the request's path **and query string**, verbatim, on every route — and the
6//! JSON log formatter repeats a span's fields on every event logged inside it.
7//! Several query parameters are bearer values or personal data: the SAML SSO
8//! endpoint's pending-sign-on `handle`, `RelayState` and `SAMLRequest` (T23.2.3),
9//! `/oauth2/authorize`'s `state` and `login_hint`, `end_session`'s
10//! `id_token_hint`, a password-reset or GDPR-cancellation `token`, an
11//! administrator's `search` term. A log reader — or a log pipeline — held all
12//! of them.
13//!
14//! [`RedactingRootSpanBuilder`] records the same field set as the default
15//! builder, under the same span name and target (so `RUST_LOG` filters select
16//! it exactly as before), with one difference: `http.target` is
17//! [`redacted_target`] — every query **value** is replaced by `[redacted]`
18//! unless its parameter is on the short list of structural ones in
19//! [`KEPT_QUERY_PARAMETERS`], and a path segment matched by a `{token}` route
20//! parameter is replaced too. Parameter **names** stay, so a log still says
21//! which parameters a request carried. It records no header but `User-Agent`,
22//! as the default builder does (W8 / T9.4).
23//!
24//! An allow-list, not a deny-list: a parameter added tomorrow is redacted until
25//! somebody decides its value is harmless, rather than logged until somebody
26//! notices it is not.
27

35/// What a redacted value is replaced with.
36pub const REDACTED: &str = "[redacted]";

38/// The query parameters whose values are recorded as sent: identifiers of a

43    "org_id",

64/// Route parameters whose path segment is a bearer value (`/account/export/{token}`).
65const REDACTED_PATH_PARAMETERS: &[&str] = &["token"];

67/// The span target the default builder's spans carry. Kept, so a deployment's
68/// `RUST_LOG` directives select the root span exactly as before (with the
69/// shipped `axiam=info` filter it is not recorded at all unless an operator
70/// enables `tracing_actix_web`).
71const SPAN_TARGET: &str = "tracing_actix_web::root_span_builder";
72
73/// `path` and `query` as they may be recorded: query values redacted except
74/// for [`KEPT_QUERY_PARAMETERS`], and every path segment that `pattern` (the
75/// matched route, `{name}` placeholders included) marks as a token replaced by
76/// [`REDACTED`].
77#[must_use]
78pub fn redacted_target(path: &str, query: &str, pattern: Option<&str>) -> String {
79    let mut out = redacted_path(path, pattern);
80    if query.is_empty() {
81        return out;
82    }
83    out.push('?');
84    for (index, pair) in query.split('&').enumerate() {
85        if index > 0 {
86            out.push('&');
87        }
88        match pair.split_once('=') {
89            Some((name, value)) => {
90                out.push_str(name);
91                out.push('=');
92                let decoded = percent_decoded_name(name);
93                if KEPT_QUERY_PARAMETERS.contains(&decoded.as_str()) {
94                    out.push_str(value);
95                } else {
96                    out.push_str(REDACTED);
97                }
98            }
99            None => out.push_str(pair),
100        }
101    }
102    out
103}

105fn percent_decoded_name(name: &str) -> String {
106    url::form_urlencoded::parse(name.as_bytes())
107        .next()
108        .map(|(decoded, _)| decoded.into_owned())
109        .unwrap_or_default()
110}
111
112fn redacted_path(path: &str, pattern: Option<&str>) -> String {
113    let Some(pattern) = pattern else {
114        return path.to_owned();
115    };
116    let segments: Vec<&str> = path.split('/').collect();
117    let placeholders: Vec<&str> = pattern.split('/').collect();
118    if segments.len() != placeholders.len() {
119        return path.to_owned();
120    }
121    segments
122        .iter()
123        .zip(placeholders)
124        .map(|(segment, placeholder)| {
125            let name = placeholder
126                .strip_prefix('{')
127                .and_then(|p| p.strip_suffix('}'))
128                .map(|p| p.split(':').next().unwrap_or_default());
129            match name {
130                Some(name) if REDACTED_PATH_PARAMETERS.contains(&name) => REDACTED,
131                _ => segment,
132            }
133        })
134        .collect::<Vec<_>>()
135        .join("/")
136}

138fn http_flavor(version: actix_web::http::Version) -> &'static str {
139    use actix_web::http::Version;
140    match version {
141        Version::HTTP_09 => "0.9",
142        Version::HTTP_10 => "1.0",
143        Version::HTTP_11 => "1.1",
144        Version::HTTP_2 => "2.0",
145        Version::HTTP_3 => "3.0",
146        _ => "other",
147    }
148}

150/// The root span builder `axiam-server` wraps the application in. See the
151/// module documentation.
152pub struct RedactingRootSpanBuilder;
153
154impl RootSpanBuilder for RedactingRootSpanBuilder {
155    fn on_request_start(request: &ServiceRequest) -> Span {
156        let user_agent = request
157            .headers()
158            .get("User-Agent")
159            .and_then(|h| h.to_str().ok())
160            .unwrap_or("");
161        let pattern = request.match_pattern();
162        let route = pattern.as_deref().unwrap_or("default");
163        let method = request.method().as_str();
164        let target = redacted_target(request.path(), request.query_string(), pattern.as_deref());
165        let connection_info = request.connection_info();
166        let request_id = request
167            .extensions()
168            .get::<RequestId>()
169            .map(ToString::to_string)
170            .unwrap_or_default();
171        tracing::info_span!(
172            target: SPAN_TARGET,
173            "HTTP request",
174            http.method = %method,
175            http.route = %route,
176            http.flavor = %http_flavor(request.version()),
177            http.scheme = %connection_info.scheme(),
178            http.host = %connection_info.host(),
179            http.client_ip = %connection_info.realip_remote_addr().unwrap_or(""),
180            http.user_agent = %user_agent,
181            http.target = %target,
182            http.status_code = tracing::field::Empty,
183            otel.name = %format!("{method} {route}"),
184            otel.kind = "server",
185            otel.status_code = tracing::field::Empty,
186            trace_id = tracing::field::Empty,
187            request_id = %request_id,
188            exception.message = tracing::field::Empty,
189            exception.details = tracing::field::Empty,
190        )
191    }
192
193    fn on_request_end<B: MessageBody>(span: Span, outcome: &Result<ServiceResponse<B>, Error>) {
194        DefaultRootSpanBuilder::on_request_end(span, outcome);
195    }
196}

207    /// T-325's values and the pre-existing bearer and personal values, each
208    /// redacted; structural values kept; names kept.
209    #[test]
210    fn query_values_are_redacted_unless_structural() {
211        let target = redacted_target(
212            "/saml/v2/0b5f1d6e-6f0a-4c39-9d1e-2b8f2a1c7e10/sso/continue",
213            "handle=AbC-_09&axiam_login_hop=1",
214            None,
215        );
216        assert_eq!(
217            target,
218            "/saml/v2/0b5f1d6e-6f0a-4c39-9d1e-2b8f2a1c7e10/sso/continue?handle=[redacted]&axiam_login_hop=1"
219        );
220        for sensitive in [
221            "RelayState",
222            "SAMLRequest",
223            // T23.2.4: the single-logout endpoint's second message parameter.
224            "SAMLResponse",
225            "Signature",
226            "state",
227            "login_hint",
228            "id_token_hint",
229            "code",
230            "code_challenge",
231            "nonce",
232            "redirect_uri",
233            "token",
234            "user_code",
235            "search",
236            "request_uri",
237            "handle",
238        ] {
239            let query = format!("{sensitive}=probe-value-123&tenant_id=t-1");
240            let recorded = redacted_target("/x", &query, None);
241            assert!(!recorded.contains("probe-value-123"), "{sensitive} leaked");
242            assert!(recorded.contains(&format!("{sensitive}={REDACTED}")));
243            assert!(recorded.ends_with("&tenant_id=t-1"));
244        }
245        // A percent-encoded name is judged by its decoded spelling, and still
246        // recorded as sent.
247        assert_eq!(
248            redacted_target("/x", "tenant%5Fid=t-1&st%61te=v", None),
249            "/x?tenant%5Fid=t-1&st%61te=[redacted]"
250        );
251        assert_eq!(redacted_target("/x", "", None), "/x");
252        assert_eq!(redacted_target("/x", "flag", None), "/x?flag");
253    }

295    /// The span the application actually records: run a request through
296    /// `TracingLogger` with this builder and read the `http.target` it wrote.
297    #[actix_rt::test]
298    async fn the_recorded_span_carries_the_redacted_target() {
299        #[derive(Clone, Default)]
300        struct Capture(Arc<Mutex<Vec<u8>>>);
301        impl std::io::Write for Capture {
302            fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
303                self.0.lock().unwrap().extend_from_slice(buf);
304                Ok(buf.len())
305            }
306            fn flush(&mut self) -> std::io::Result<()> {
307                Ok(())
308            }
309        }
310        let capture = Capture::default();
311        let writer = capture.clone();
312        let subscriber = tracing_subscriber::fmt()
313            .with_writer(move || writer.clone())
314            .with_ansi(false)
315            .finish();
316        let _guard = tracing::subscriber::set_default(subscriber);
317
318        let app = actix_web::test::init_service(
319            App::new()
320                .wrap(TracingLogger::<RedactingRootSpanBuilder>::new())
321                .route(
322                    "/saml/v2/{tenant_id}/sso/continue",
323                    web::get().to(|| async {
324                        tracing::info!("inside the request");
325                        HttpResponse::Ok().finish()
326                    }),
327                ),
328        )
329        .await;
330        let request = actix_web::test::TestRequest::get()
331            .uri("/saml/v2/t-1/sso/continue?handle=AbC-handle-value&axiam_login_hop=1")
332            .to_request();
333        let response = actix_web::test::call_service(&app, request).await;
334        assert!(response.status().is_success());
335
336        let log = String::from_utf8(capture.0.lock().unwrap().clone()).unwrap();
337        assert!(log.contains("inside the request"), "the event was recorded");
338        assert!(
339            log.contains("handle=[redacted]&axiam_login_hop=1"),
340            "the span records the redacted target"
341        );
342        assert!(
343            !log.contains("AbC-handle-value"),
344            "the handle never reaches the log"
345        );
346    }
