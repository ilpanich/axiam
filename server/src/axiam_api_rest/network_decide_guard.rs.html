1//! The decision route's two pieces of per-process state (RADIUS D4, W6.4;
2//! `claude_dev/radius-reply-profile-design.md` §3.4, §6.3).
3//!
4//! # The caller's bucket
5//!
6//! `POST /api/v1/network/decide` is metered **per authenticated caller** — the
7//! token's `sub`, a FreeRADIUS server's service account — at
8//! `network_decide_per_min` (maintainer decision D-6). The caller exists only
9//! once the handler has authenticated its certificate-bound token, so the bucket
10//! cannot be route middleware: the handler spends
11//! `network_decide:{tenant}:{sub}` from the shared counter, and when that counter
12//! is switched off (`AXIAM__RATE_LIMIT__SHARED=off`, which admits everything)
13//! from the in-process keyed limiter held here, exactly as the CIBA pending list
14//! does (R101W3-04; [`crate::ciba_list_limiter`], whose limiter is reused: it is
15//! a keyed `(tenant, subject)` limiter, and nothing in it is about CIBA). A key
16//! reaches it only after authentication, so the map is bounded by the number of
17//! callers, never by what a caller can invent.
18//!
19//! # One `reply_profile.invalid` row per profile per window
20//!
21//! A stored profile that no longer validates (one edited in the datastore behind
22//! the API) makes its segment no candidate, and the design note asks for **one
23//! audit row per profile per window** (§3.4), not one per decision: a site's
24//! power-restore storm would otherwise write a hundred identical rows a second.
25//! The window is [`INVALID_PROFILE_AUDIT_WINDOW`]; the key is the tenant, the
26//! segment, the profile's version and the rule it fails, so a profile rewritten
27//! (a new version) or failing a different rule is reported again at once. It is
28//! per process, as every in-process limiter is: N replicas write at most N rows
29//! a window.
30
31use std::collections::HashMap;
32use std::sync::Mutex;
33use std::time::{Duration, Instant};
34
35use uuid::Uuid;
36
37use crate::ciba_list_limiter::CibaListLimiter;
38
39/// How long one `reply_profile.invalid` row covers a defective profile.
40pub const INVALID_PROFILE_AUDIT_WINDOW: Duration = Duration::from_secs(3600);
41
42/// The key of one defective profile's window.
43type DefectKey = (Uuid, Uuid, u64, &'static str);
44
45/// The decision route's per-process state. One per process, in `AppState`.
46#[derive(Debug, Default)]
47pub struct NetworkDecideGuards {
48    callers: CibaListLimiter,
49    defects: Mutex<HashMap<DefectKey, Instant>>,
50}
51
52impl NetworkDecideGuards {
53    /// Spend one decision of `(tenant_id, caller)`'s `per_min` allowance in the
54    /// in-process limiter; `false` means it is spent and the route answers `429`.
55    pub fn admit(&self, tenant_id: Uuid, caller: Uuid, per_min: u32) -> bool {
56        self.callers.check(tenant_id, caller, per_min)
57    }
58
59    /// Whether a defective profile should be audited now: `true` the first time
60    /// `(tenant, segment, version, rule)` is seen in a window, `false` for the
61    /// rest of it.
62    pub fn first_in_window(
63        &self,
64        tenant_id: Uuid,
65        segment_id: Uuid,
66        version: u64,
67        rule: &'static str,
68    ) -> bool {
69        self.first_in_window_at(tenant_id, segment_id, version, rule, Instant::now())
70    }
71
72    fn first_in_window_at(
73        &self,
74        tenant_id: Uuid,
75        segment_id: Uuid,
76        version: u64,
77        rule: &'static str,
78        now: Instant,
79    ) -> bool {
80        // A poisoned lock means a panic elsewhere mid-update; audit rather than
81        // stay silent (one extra row is the safe failure).
82        let Ok(mut seen) = self.defects.lock() else {
83            return true;
84        };
85        // Expired windows go before the lookup, so the map holds only the
86        // defects of the last window: bounded by the tenant's profiles.
87        seen.retain(|_, at| now.duration_since(*at) < INVALID_PROFILE_AUDIT_WINDOW);
88        let key = (tenant_id, segment_id, version, rule);
89        if seen.contains_key(&key) {
90            return false;
91        }
92        seen.insert(key, now);
93        true
94    }
95}

97#[cfg(test)]
98mod tests {
99    use super::*;
100
101    #[test]
102    fn a_defect_is_audited_once_per_window_and_again_after_it() {
103        let g = NetworkDecideGuards::default();
104        let (t, s) = (Uuid::new_v4(), Uuid::new_v4());
105        let start = Instant::now();
106        assert!(g.first_in_window_at(t, s, 3, "3.4:value", start));
107        assert!(!g.first_in_window_at(t, s, 3, "3.4:value", start + Duration::from_secs(5)));
108        // A new version, or another rule, is another defect.
109        assert!(g.first_in_window_at(t, s, 4, "3.4:value", start));
110        assert!(g.first_in_window_at(t, s, 3, "3.4:dictionary", start));
111        // After the window, the same defect is reported again.
112        assert!(g.first_in_window_at(
113            t,
114            s,
115            3,
116            "3.4:value",
117            start + INVALID_PROFILE_AUDIT_WINDOW + Duration::from_secs(1)
118        ));
119    }
120
121    #[test]
122    fn each_caller_has_a_bucket_of_its_own() {
123        let g = NetworkDecideGuards::default();
124        let (t, a, b) = (Uuid::new_v4(), Uuid::new_v4(), Uuid::new_v4());
125        assert!(g.admit(t, a, 1));
126        assert!(!g.admit(t, a, 1));
127        assert!(g.admit(t, b, 1));
128    }
129}
