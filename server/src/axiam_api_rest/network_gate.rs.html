1//! The REST side of the network write gate (RADIUS Live profile, W6.2).
2//!
3//! Every handler that can create a reply profile, a segment, a
4//! network-sensitive grant or assignment, or a certificate binding — and every
5//! generic handler that could reach into the reserved `network:` namespace or
6//! `/network` subtree — calls [`authorize`] **before** its repository write.
7//! The decision itself is `axiam_authz::network_gate::authorize_network_write`
8//! (design note §5.1: one domain function, not handler logic); this module
9//! only supplies its reads from [`AppState`], writes the audit row of a
10//! refusal (§5.4) and turns the refusal into the response.
11//!
12//! `tests/network_gate_enumeration_test.rs` reads the sources and fails if a
13//! write of that kind reaches its repository from a function that does not call
14//! [`authorize`] first (the T-429 lesson); segment and profile writes cannot
15//! even compile without the clearance [`authorize`] returns.
16

69/// Run the gate for `write` as `principal`; on a refusal, audit it and answer
70/// `403` or `400`.
71///
72/// # Errors
73///
74/// The refusal as an [`AxiamApiError`] (`403` naming the missing permission
75/// and node, or `400`), or whatever the gate's reads return (`404` for a role,
76/// permission or segment that is not the tenant's).
77pub async fn authorize<C: Connection + Clone>(
78    principal: &AuthenticatedPrincipal,
79    authz: &(dyn AuthzChecker + 'static),
80    state: &AppState<C>,
81    write: NetworkWrite<'_>,
82) -> Result<GateClearance, AxiamApiError> {
83    authorize_as(&GateActor::principal(principal), authz, state, write).await
84}

86/// [`authorize`] for a writer that is not an [`AuthenticatedPrincipal`].
87///
88/// # Errors
89///
90/// As [`authorize`].
91pub async fn authorize_as<C: Connection + Clone>(
92    actor: &GateActor,
93    authz: &(dyn AuthzChecker + 'static),
94    state: &AppState<C>,
95    write: NetworkWrite<'_>,
96) -> Result<GateClearance, AxiamApiError> {
97    let segments = SurrealNetworkSegmentRepository::new(state.db.clone());
98    let profiles = SurrealReplyProfileRepository::new(state.db.clone());
99    let policy = SurrealNetworkPolicyRepository::new(state.db.clone());
100    let store = RepositoryGateStore {
101        roles: &state.role_repo,
102        groups: &state.group_repo,
103        permissions: &state.permission_repo,
104        resources: &state.resource_repo,
105        segments: &segments,
106        profiles: &profiles,
107        policy: &policy,
108        certificates: &state.pki.cert_repo,
109    };
110    match authorize_network_write(authz, &store, actor.writer, write).await? {
111        Ok(clearance) => Ok(clearance),
112        Err(refusal) => {
113            audit_refusal(state, actor, write_kind(&write), &refusal).await;
114            Err(into_error(refusal))
115        }
116    }
117}

119/// Answer a role write that the gate cleared and **its own transaction then
120/// refused** because what the gate read had changed by the time it ran
121/// (R110W6-04, #701).
122///
123/// The gate decides a role write — a grant or revoke, an assignment or
124/// unassignment, an `is_global` update, a delete — on the role's grants,
125/// holders and scope, read before the write. Two writes to one role can each
126/// pass it on a state the other is about to change (a `network:join` grant and
127/// the writer assigning itself the role; a grant and a group assignment; …).
128/// So the gate hands the write a claim on the role's epochs
129/// ([`GateClearance::role_claim`]), the `*_claimed` repository writes compare
130/// it inside their transaction, and a write that lost comes back as
131/// `AxiamError::Network(NetworkConflict::StaleRole { .. })` ("the role changed
132/// since it was read …"). That
133/// answers `409` with `reason_code` `stale_version` — the network routes'
134/// sub-code for "read again, decide again" (contract §36.3 rule 2): sent again,
135/// the request goes back through the gate, which decides on the role as it now
136/// is. It is audited as the gate audits its own refusals: one
137/// `network_role.*_refused` row, outcome denied, with the internal rule
138/// (`version:stale`, as the network routes record a stale write), the reason
139/// and the values.
140///
141/// Any other error is returned unchanged, unaudited, as it was before.
142///
143/// # Errors
144///
145/// `error` itself, when it is not that conflict or `write` is not a role
146/// write.
147pub async fn refused_after_clearance<C: Connection + Clone>(
148    principal: &AuthenticatedPrincipal,
149    state: &AppState<C>,
150    write: NetworkWrite<'_>,
151    error: AxiamError,
152) -> Result<HttpResponse, AxiamApiError> {
153    if NetworkConflictReason::of(&error) != Some(NetworkConflictReason::StaleVersion) {
154        return Err(AxiamApiError(error));
155    }
156    let Some(refusal) = StaleRefusal::of(&GateActor::principal(principal), write) else {
157        return Err(AxiamApiError(error));
158    };
159    refusal.append(&state.audit_repo, &error).await;
160    Ok(AxiamApiError(error)
161        .error_response_with_reason_code(Some(NetworkConflictReason::StaleVersion.as_str())))
162}

164/// The answer to a [`write_and_record`] error: a stale claim is `409` with
165/// `reason_code` `stale_version`, as [`refused_after_clearance`] answers it —
166/// its `network_role.*_refused` row is already appended, by the write's own
167/// task (R112W4-01) — and any other error is returned unchanged.
168///
169/// # Errors
170///
171/// `error` itself, when it is not a stale claim.
172pub fn refused_in_task(error: AxiamError) -> Result<HttpResponse, AxiamApiError> {
173    if NetworkConflictReason::of(&error) != Some(NetworkConflictReason::StaleVersion) {
174        return Err(AxiamApiError(error));
175    }
176    Ok(AxiamApiError(error)
177        .error_response_with_reason_code(Some(NetworkConflictReason::StaleVersion.as_str())))
178}

180/// The `network_role.*_refused` row of a stale claim, described before the
181/// write runs: everything but the reason, which only the refusal carries.
182struct StaleRefusal {
183    tenant_id: uuid::Uuid,
184    actor_id: uuid::Uuid,
185    actor_type: ActorType,
186    action: &'static str,
187    write: &'static str,
188    before: serde_json::Value,
189    after: serde_json::Value,
190}
191
192impl StaleRefusal {
193    /// `None` for a write that is not a role write (it holds no claim).
194    fn of(actor: &GateActor, write: NetworkWrite<'_>) -> Option<Self> {
195        let grantee = |g: Grantee| match g {
196            Grantee::User(id) => ("user", id.to_string()),
197            Grantee::ServiceAccount(id) => ("service_account", id.to_string()),
198            Grantee::Group(id) => ("group", id.to_string()),
199        };
200        let node = |r: Option<uuid::Uuid>| r.map_or_else(|| "none".to_owned(), |r| r.to_string());
201        let (audit_action, before, after) = match write {
202            NetworkWrite::PermissionGrant {
203                role_id,
204                permission_id,
205                effect,
206                ..
207            } => (
208                AUDIT_ROLE_GRANT_REFUSED,
209                vec![("role_id", role_id.to_string())],
210                vec![
211                    ("permission_id", permission_id.to_string()),
212                    ("effect", effect.as_str().to_owned()),
213                ],
214            ),
215            NetworkWrite::PermissionRevoke {
216                role_id,
217                permission_id,
218            } => (
219                AUDIT_ROLE_GRANT_REFUSED,
220                vec![
221                    ("role_id", role_id.to_string()),
222                    ("permission_id", permission_id.to_string()),
223                ],
224                vec![("grant", "revoked".to_owned())],
225            ),
226            NetworkWrite::RoleAssign {
227                role_id,
228                grantee: g,
229                resource_id,
230                inherit,
231            } => {
232                let (kind, who) = grantee(g);
233                (
234                    AUDIT_ROLE_ASSIGN_REFUSED,
235                    vec![("role_id", role_id.to_string())],
236                    vec![
237                        (kind, who),
238                        ("resource_id", node(resource_id)),
239                        ("inherit", inherit.to_string()),
240                    ],
241                )
242            }
243            NetworkWrite::RoleUnassign {
244                role_id,
245                grantee: g,
246                resource_id,
247            } => {
248                let (kind, who) = grantee(g);
249                (
250                    AUDIT_ROLE_UNASSIGN_REFUSED,
251                    vec![
252                        ("role_id", role_id.to_string()),
253                        (kind, who),
254                        ("resource_id", node(resource_id)),
255                    ],
256                    vec![("assignment", "removed".to_owned())],
257                )
258            }
259            NetworkWrite::RoleUpdate { role_id, is_global } => (
260                AUDIT_ROLE_WRITE_REFUSED,
261                vec![("role_id", role_id.to_string())],
262                vec![(
263                    "is_global",
264                    is_global.map_or_else(|| "unchanged".to_owned(), |g| g.to_string()),
265                )],
266            ),
267            NetworkWrite::RoleDelete { role_id } => (
268                AUDIT_ROLE_WRITE_REFUSED,
269                vec![("role_id", role_id.to_string())],
270                vec![("role", "deleted".to_owned())],
271            ),
272            _ => return None,
273        };
274        Some(Self {
275            tenant_id: actor.writer.tenant_id,
276            actor_id: actor.writer.subject_id,
277            actor_type: actor.actor_type.clone(),
278            action: audit_action,
279            write: write_kind(&write),
280            before: values(&before),
281            after: values(&after),
282        })
283    }
284
285    /// Append the row for `error`; a failed append is logged, and the write
286    /// stays refused.
287    async fn append<A: AuditLogRepository>(self, audit: &A, error: &AxiamError) {
288        let (tenant_id, action) = (self.tenant_id, self.action);
289        let entry = CreateAuditLogEntry {
290            tenant_id,
291            actor_id: self.actor_id,
292            actor_type: self.actor_type,
293            action: action.to_owned(),
294            resource_id: None,
295            outcome: AuditOutcome::Denied,
296            ip_address: None,
297            metadata: Some(serde_json::json!({
298                "write": self.write,
299                "rule": "version:stale",
300                "reason": error.to_string(),
301                "status": 409,
302                "before": self.before,
303                "after": self.after,
304            })),
305        };
306        if let Err(audit_error) = audit.append(entry).await {
307            tracing::error!(
308                target: "axiam::network_gate",
309                %tenant_id,
310                action,
311                error = %audit_error,
312                "a network write refusal could not be audited; the write stays refused"
313            );
314        }
315    }
316}

318/// Run the repository write of a network-sensitive write that the gate
319/// cleared, and record it once it landed (R110W6-06, #703, T-484): one
320/// `network_role.assigned` / `.unassigned` / `.granted` / `.revoked` /
321/// `.deleted` or `certificate_binding.bound` row (one of the six), with the
322/// role, the grantee, the node and the `network:` permissions the write
323/// conferred (or removed); an assignment's row
324/// also carries `exemption`, the named exemption it cleared under
325/// (`organization-bootstrap`) or `null`. Without it a cleared write is visible
326/// only in the route's generic request row, which says neither that it touched
327/// the network namespace nor where a join landed.
328///
329/// **The row cannot be separated from the write (R111W4-02, #726).** It is
330/// built here, before the write, from what the gate read
331/// ([`GateClearance::role`], [`GateClearance::permission`]) — not from reads
332/// after the write, which a concurrent role delete made fail and a concurrent
333/// regrant made lie. Then `repository_write` and the append run as **one
334/// spawned task**, which this function awaits through its `JoinHandle`. Actix
335/// drops a handler's future when its client goes away; dropping the handle
336/// does not cancel the task, so a write that commits is appended whether or not
337/// anyone still waits for the answer. The task is spawned through
338/// [`AppState::network_writes`] — on the composition root's runtime, which an
339/// orderly stop keeps until the task is done or the stop's budget is spent
340/// ([`NetworkWriteTasks`], R112W4-02) — so the write must be `Send` and own
341/// what it uses (`'static`).
342///
343/// **What follows the commit rides the same task (R112W4-01, #775).** The D7
344/// decision-cache flush the write needs (`flush`: the grantee for a user or
345/// service-account assignment, the tenant for a group assignment, a role
346/// delete, a grant or a revoke; local **and** broadcast, through
347/// `AuthzChecker::invalidate_*`) runs after the write and **before** the row,
348/// so a revocation's row is never readable while the allow it revokes is
349/// still served from the cache of the replica that wrote it. A failed flush
350/// (the broadcast was not confirmed) does not drop the row of a write that
351/// landed: the row is appended and the flush's error returned (`503`, the
352/// handlers' convention for an unconfirmed revocation). On a stale claim the
353/// `network_role.*_refused` row is appended in the task too (the caller then
354/// answers with [`refused_in_task`]). Before this, both ran in the handler's
355/// future, after the task: a client gone after the spawn left a committed
356/// revocation whose cached allow outlived it, on every replica, until the TTL.
357///
358/// No success row is written when the clearance is not network-sensitive,
359/// when the write fails, or when the write is not one of those named above (a
360/// role update, a resource write, a segment or profile write — those keep
361/// their own rows).
362///
363/// A failure to describe the write or to append the row is logged and does not
364/// undo or fail the write that already landed — the convention of
365/// [`audit_refusal`], and of every other audit append in the handlers. The row
366/// carries ids, names and actions only; never a certificate, key or token.
367///
368/// # Errors
369///
370/// Whatever `repository_write` returns; the flush's error once the write
371/// landed; `AxiamError::Internal` if the runtime cancels the task (it shuts
372/// down). A panic in the task is resumed here.
373pub async fn write_and_record<C, T, W>(
374    state: &AppState<C>,
375    authz: &Arc<dyn AuthzChecker>,
376    actor: &GateActor,
377    write: NetworkWrite<'_>,
378    cleared: &GateClearance,
379    flush: CacheFlush,
380    repository_write: W,
381) -> Result<T, AxiamError>
382where
383    C: Connection + Clone,
384    T: Send + 'static,
385    W: Future<Output = Result<T, AxiamError>> + Send + 'static,
386{
387    let row = cleared_row(actor, write, cleared);
388    let stale = StaleRefusal::of(actor, write);
389    let audit = state.audit_repo.clone();
390    let authz = Arc::clone(authz);
391    let tenant_id = actor.writer.tenant_id;
392    let task = state.network_writes.spawn(async move {
393        let written = match repository_write.await {
394            Ok(written) => written,
395            Err(error) => {
396                if let Some(refusal) = stale.filter(|_| {
397                    NetworkConflictReason::of(&error) == Some(NetworkConflictReason::StaleVersion)
398                }) {
399                    refusal.append(&audit, &error).await;
400                }
401                return Err(error);
402            }
403        };
404        let flushed = match flush {
405            CacheFlush::None => Ok(()),
406            CacheFlush::Subject(subject_id) => {
407                authz.invalidate_subject(tenant_id, subject_id).await
408            }
409            CacheFlush::Tenant => authz.invalidate_tenant(tenant_id).await,
410        };
411        match row {
412            Ok(None) => {}
413            Ok(Some(entry)) => {
414                let action = entry.action.clone();
415                if let Err(error) = audit.append(entry).await {
416                    tracing::error!(
417                        target: "axiam::network_gate",
418                        %tenant_id,
419                        %action,
420                        %error,
421                        "a cleared network write could not be audited; the write stands"
422                    );
423                }
424            }
425            Err(missing) => tracing::error!(
426                target: "axiam::network_gate",
427                %tenant_id,
428                missing,
429                "a cleared network write could not be described for the audit; the write stands"
430            ),
431        }
432        flushed.map(|()| written)
433    });
434    match task.await {
435        Ok(result) => result,
436        Err(join) if join.is_panic() => std::panic::resume_unwind(join.into_panic()),
437        Err(join) => Err(AxiamError::Internal(format!(
438            "a cleared network write was cancelled by the runtime: {join}"
439        ))),
440    }
441}

443/// Where the task of a cleared network write runs, and the tracker an orderly
444/// stop waits on (R112W4-02, #776, T-444).
445///
446/// Actix runs each worker on a runtime of its own and, on a stop, waits only
447/// for its **connections**; once the last one is gone the worker's runtime
448/// stops and every task spawned on it is dropped. A write whose client had
449/// gone no longer counted, so a stop could drop its task between the commit
450/// and the append. The composition root therefore hands the handlers its own
451/// runtime ([`Self::on`]), which outlives the workers until `serve` returns,
452/// and the teardown waits for every task this tracks ([`Self::wait_within`])
453/// inside the stop's budget, as it waits for the cleanup units.
454///
455/// The default (no runtime, [`AppState::for_test`]) spawns on the worker, as
456/// 1.1.2 W4.1 did, and still tracks the task.
457#[derive(Debug, Clone, Default)]
458pub struct NetworkWriteTasks {
459    runtime: Option<tokio::runtime::Handle>,
460    tracker: tokio_util::task::TaskTracker,
461}
462
463impl NetworkWriteTasks {
464    /// Tasks spawned on `runtime`, the composition root's.
465    #[must_use]
466    pub fn on(runtime: tokio::runtime::Handle) -> Self {
467        Self {
468            runtime: Some(runtime),
469            tracker: tokio_util::task::TaskTracker::new(),
470        }
471    }
472
473    fn spawn<T, F>(&self, task: F) -> tokio::task::JoinHandle<T>
474    where
475        T: Send + 'static,
476        F: Future<Output = T> + Send + 'static,
477    {
478        let task = self.tracker.track_future(task);
479        match &self.runtime {
480            Some(runtime) => runtime.spawn(task),
481            None => actix_web::rt::spawn(task),
482        }
483    }
484
485    /// The tasks still running: what a stop would wait for.
486    #[must_use]
487    pub fn in_flight(&self) -> usize {
488        self.tracker.len()
489    }
490
491    /// Wait, at most `bound`, for every task spawned so far; `false` when some
492    /// were still running at the bound. A task spawned after the call is still
493    /// run and tracked, and waited for too while the bound lasts (nothing is
494    /// refused: after the REST listener has stopped, no handler spawns one).
495    pub async fn wait_within(&self, bound: std::time::Duration) -> bool {
496        self.tracker.close();
497        tokio::time::timeout(bound, self.tracker.wait())
498            .await
499            .is_ok()
500    }
501}

503/// The D7 decision-cache flush a cleared write needs once it landed
504/// ([`write_and_record`], R112W4-01), in the writer's tenant.
505#[derive(Debug, Clone, Copy, PartialEq, Eq)]
506pub enum CacheFlush {
507    /// No cached decision reads what the write changes (a certificate
508    /// binding).
509    None,
510    /// Exactly this subject's decisions changed (a user or service-account
511    /// assignment or unassignment).
512    Subject(uuid::Uuid),
513    /// The affected subjects are not known without a query (a group
514    /// assignment, a role delete, a grant or a revoke): every decision of the
515    /// tenant.
516    Tenant,
517}

519/// The success row of a cleared write, built from the clearance alone
520/// (R111W4-02): `Ok(None)` when the write gets no row, `Err` naming what the
521/// clearance lacks when it cannot be described.
522fn cleared_row(
523    actor: &GateActor,
524    write: NetworkWrite<'_>,
525    cleared: &GateClearance,
526) -> Result<Option<CreateAuditLogEntry>, &'static str> {
527    if !cleared.is_network_sensitive() {
528        return Ok(None);
529    }
530    // The node the gate decided the assignment at — `/network`'s id for a
531    // tenant-wide assignment and for every assignment of a global role — not
532    // the resource the caller named (#730 g).
533    let node = || cleared.node().ok_or("the node the gate checked");
534    let grantee = |g: Grantee| match g {
535        Grantee::User(id) => serde_json::json!({ "kind": "user", "id": id }),
536        Grantee::ServiceAccount(id) => serde_json::json!({ "kind": "service_account", "id": id }),
537        Grantee::Group(id) => serde_json::json!({ "kind": "group", "id": id }),
538    };
539    let role = || cleared.role().ok_or("the role the gate read");
540    let permission = || cleared.permission().ok_or("the permission the gate read");
541    let network_permissions = |role: &ClearedRole| {
542        role.network_grants
543            .iter()
544            .map(|g| {
545                serde_json::json!({
546                    "action": g.permission.action,
547                    "effect": g.effect.as_str(),
548                    "scope_ids": g.scope_ids,
549                })
550            })
551            .collect::<Vec<_>>()
552    };
553    let (audit_action, resource_id, mut metadata) = match write {
554        NetworkWrite::RoleAssign {
555            role_id,
556            grantee: g,
557            resource_id,
558            inherit,
559        } => {
560            let role = role()?;
561            (
562                AUDIT_ROLE_ASSIGNED,
563                resource_id,
564                serde_json::json!({
565                    "role_id": role_id,
566                    "role": role.name,
567                    "grantee": grantee(g),
568                    "node": node()?,
569                    "inherit": inherit,
570                    "permissions": network_permissions(role),
571                    "exemption": cleared.exemption(),
572                }),
573            )
574        }
575        NetworkWrite::RoleUnassign {
576            role_id,
577            grantee: g,
578            resource_id,
579        } => {
580            let role = role()?;
581            (
582                AUDIT_ROLE_UNASSIGNED,
583                resource_id,
584                serde_json::json!({
585                    "role_id": role_id,
586                    "role": role.name,
587                    "grantee": grantee(g),
588                    "node": node()?,
589                    "permissions": network_permissions(role),
590                }),
591            )
592        }
593        NetworkWrite::PermissionGrant {
594            role_id,
595            permission_id,
596            effect,
597            scope_ids,
598        } => (
599            AUDIT_ROLE_GRANTED,
600            None,
601            serde_json::json!({
602                "role_id": role_id,
603                "role": role()?.name,
604                "permissions": [{
605                    "action": permission()?.action,
606                    "permission_id": permission_id,
607                    "effect": effect.as_str(),
608                    "scope_ids": scope_ids,
609                }],
610            }),
611        ),
612        NetworkWrite::PermissionRevoke {
613            role_id,
614            permission_id,
615        } => {
616            let effect = cleared
617                .revoked_effect()
618                .flatten()
619                .map_or("none", |e| e.as_str());
620            (
621                AUDIT_ROLE_REVOKED,
622                None,
623                serde_json::json!({
624                    "role_id": role_id,
625                    "role": role()?.name,
626                    "permissions": [{
627                        "action": permission()?.action,
628                        "permission_id": permission_id,
629                        "effect": effect,
630                    }],
631                }),
632            )
633        }
634        NetworkWrite::RoleDelete { role_id } => {
635            let role = role()?;
636            (
637                AUDIT_ROLE_DELETED,
638                None,
639                serde_json::json!({
640                    "role_id": role_id,
641                    "role": role.name,
642                    "permissions": network_permissions(role),
643                }),
644            )
645        }
646        NetworkWrite::CertificateBind {
647            certificate_id,
648            service_account_id,
649        } => {
650            let binding = cleared
651                .cleared_binding()
652                .ok_or("the binding the gate read")?;
653            (
654                AUDIT_BINDING_BOUND,
655                None,
656                serde_json::json!({
657                    "certificate_id": certificate_id,
658                    "grantee": { "kind": "service_account", "id": service_account_id },
659                    // The nodes at which the gate required the binder to hold
660                    // `network:grant`. No `before`: the route refuses a bound
661                    // certificate, so a success row never takes one from an
662                    // account (R112W4-05).
663                    "nodes": binding.nodes,
664                }),
665            )
666        }
667        _ => return Ok(None),
668    };
669    metadata["write"] = serde_json::json!(write_kind(&write));
670    Ok(Some(CreateAuditLogEntry {
671        tenant_id: actor.writer.tenant_id,
672        actor_id: actor.writer.subject_id,
673        actor_type: actor.actor_type.clone(),
674        action: audit_action.to_owned(),
675        resource_id,
676        outcome: AuditOutcome::Success,
677        ip_address: None,
678        metadata: Some(metadata),
679    }))
680}

682/// A short name for the write, for the audit row.
683fn write_kind(write: &NetworkWrite<'_>) -> &'static str {
684    match write {
685        NetworkWrite::ProfileWrite { .. } => "profile_write",
686        NetworkWrite::ProfileDelete { .. } => "profile_delete",
687        NetworkWrite::SegmentCreate { .. } => "segment_create",
688        NetworkWrite::SegmentUpdate { .. } => "segment_update",
689        NetworkWrite::SegmentDelete { .. } => "segment_delete",
690        NetworkWrite::RoleAssign { .. } => "role_assign",
691        NetworkWrite::RoleUnassign { .. } => "role_unassign",
692        NetworkWrite::RoleUpdate { .. } => "role_update",
693        NetworkWrite::RoleDelete { .. } => "role_delete",
694        NetworkWrite::PermissionGrant { .. } => "permission_grant",
695        NetworkWrite::PermissionRevoke { .. } => "permission_revoke",
696        NetworkWrite::PermissionCreate { .. } => "permission_create",
697        NetworkWrite::PermissionUpdate { .. } => "permission_update",
698        NetworkWrite::PermissionDelete { .. } => "permission_delete",
699        NetworkWrite::PolicyWrite { .. } => "policy_write",
700        NetworkWrite::CertificateBind { .. } => "certificate_bind",
701        NetworkWrite::ResourceCreate { .. } => "resource_create",
702        NetworkWrite::ResourceUpdate { .. } => "resource_update",
703        NetworkWrite::ResourceDelete { .. } => "resource_delete",
704    }
705}

707fn values(pairs: &[(&'static str, String)]) -> serde_json::Value {
708    serde_json::Value::Object(
709        pairs
710            .iter()
711            .map(|(k, v)| ((*k).to_owned(), serde_json::Value::String(v.clone())))
712            .collect(),
713    )
714}

716/// One append-only row per refusal (§5.4): the actor, the tenant, the write,
717/// the values before and after, and the **internal** reason. Refusals are
718/// audited because an escalation attempt is exactly what a reviewer looks for.
719///
720/// A failure to write the row is logged and does not turn the refusal into a
721/// success: the write stays refused either way.
722async fn audit_refusal<C: Connection + Clone>(
723    state: &AppState<C>,
724    actor: &GateActor,
725    write: &'static str,
726    refusal: &GateRefusal,
727) {
728    let mut metadata = serde_json::json!({
729        "write": write,
730        "rule": refusal.rule,
731        "reason": refusal.message,
732        "status": match refusal.status {
733            RefusalStatus::Forbidden => 403,
734            RefusalStatus::Invalid => 400,
735        },
736        "before": values(&refusal.before),
737        "after": values(&refusal.after),
738    });
739    if let Some(missing) = &refusal.missing {
740        metadata["missing_permission"] = serde_json::json!(missing.action);
741        metadata["node"] = serde_json::json!(missing.node);
742    }
743    if let Some(label) = &actor.label {
744        metadata["actor_label"] = serde_json::json!(label);
745    }
746    let entry = CreateAuditLogEntry {
747        tenant_id: actor.writer.tenant_id,
748        actor_id: actor.writer.subject_id,
749        actor_type: actor.actor_type.clone(),
750        action: refusal.audit_action.to_owned(),
751        resource_id: refusal.missing.as_ref().map(|m| m.node),
752        outcome: AuditOutcome::Denied,
753        ip_address: None,
754        metadata: Some(metadata),
755    };
756    if let Err(error) = state.audit_repo.append(entry).await {
757        tracing::error!(
758            target: "axiam::network_gate",
759            tenant_id = %actor.writer.tenant_id,
760            action = refusal.audit_action,
761            %error,
762            "a network write refusal could not be audited; the write stays refused"
763        );
764    }
765}
766
767fn into_error(refusal: GateRefusal) -> AxiamApiError {
768    match refusal.status {
769        RefusalStatus::Forbidden => {
770            let (action, resource_id) = refusal
771                .missing
772                .map_or((None, None), |m| (Some(m.action), Some(m.node.to_string())));
773            AxiamApiError(AxiamError::AuthorizationDenied {
774                reason: refusal.message,
775                action,
776                resource_id,
777            })
778        }
779        RefusalStatus::Invalid => AxiamApiError(AxiamError::Validation {
780            message: refusal.message,
781        }),
782    }
783}
