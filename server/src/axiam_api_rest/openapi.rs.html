535    ),
536    modifiers(&SecurityAddon),
537)]
538pub struct ApiDoc;

540/// SAML SP paths and schemas, compiled only with the `saml` feature.
541///
542/// Kept separate from [`ApiDoc`] because utoipa's `paths()`/`schemas()` macro
543/// arguments can't be `#[cfg]`-gated individually. [`api_doc`] merges this into
544/// the main document when the feature is on.
545#[cfg(feature = "saml")]
546#[derive(OpenApi)]
547#[openapi(
548    paths(
549        handlers::federation::saml_authn_request,
550        handlers::federation::saml_acs,
551        handlers::federation::saml_metadata,
552        // First-time SSO — public SAML login/ACS (FUNC-01 / D-12).
553        handlers::federation::saml_login_public,
554        handlers::federation::saml_acs_public,
555    ),
556    components(schemas(
557        handlers::federation::SamlAuthnRequestRequest,
558        handlers::federation::SamlAuthnRequestResponse,
559        handlers::federation::SamlAcsRequest,
560        handlers::federation::SamlMetadataQuery,
561        handlers::federation::SamlLoginRequest,
562        handlers::federation::SamlLoginResponse,
563        handlers::federation::SamlAcsPublicRequest,
564    ))
565)]
566struct SamlApiDoc;
567
568/// Build the full OpenAPI document, merging in the SAML SP endpoints when the
569/// `saml` feature is enabled. Use this instead of `ApiDoc::openapi()` directly.
570pub fn api_doc() -> utoipa::openapi::OpenApi {
571    let doc = ApiDoc::openapi();
572    #[cfg(feature = "saml")]
573    let doc = doc.merge_from(SamlApiDoc::openapi());
574    doc
575}

577/// Adds Bearer JWT security scheme to the OpenAPI spec.
578struct SecurityAddon;
579
580impl utoipa::Modify for SecurityAddon {
581    fn modify(&self, openapi: &mut utoipa::openapi::OpenApi) {
582        let components = openapi.components.get_or_insert_with(Default::default);
583        components.add_security_scheme(
584            "bearer",
585            utoipa::openapi::security::SecurityScheme::Http(
586                utoipa::openapi::security::HttpBuilder::new()
587                    .scheme(utoipa::openapi::security::HttpAuthScheme::Bearer)
588                    .bearer_format("JWT")
589                    .build(),
590            ),
591        );
592    }
593}
