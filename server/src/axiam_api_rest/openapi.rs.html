627/// The extension key carrying the specification's content digest.
628///
629/// `x-`-prefixed, which is what the OpenAPI specification reserves for extensions, so a
630/// validator and a code generator both ignore it.
631pub const SPEC_DIGEST_KEY: &str = "x-axiam-spec-digest";
632
633/// Stamp `info.x-axiam-spec-digest` with a SHA-256 over this document's own content.
634///
635/// **Why `info.version` is not enough.** It tracks `CARGO_PKG_VERSION`, which is the
636/// release version — it moves when a release is cut, not when a path is added. Two builds
637/// can therefore describe genuinely different APIs under the same string, and they have:
638/// `main` and a release branch both reported `1.0.0-alpha44` while differing by two paths
639/// (`ca-certificates/{id}/migrate-custody` and `.../mtls-trust-anchor`). A consumer pinning
640/// on `info.version` — an SDK deciding whether to regenerate, a gateway caching a spec, a
641/// contract test asserting it is current — could not tell those exports apart, and nothing
642/// in the document let it.
643///
644/// The version keeps its meaning: it is the API's semantic version, and consumers should
645/// keep reading it as one. This is the other question — "is this the same document?" —
646/// answered separately and exactly.
647///
648/// **`info.version` is inside the digest**, so a release bump changes it even when no path
649/// did. Deliberate: "same document" is checkable, where "same API" would need an argument
650/// about whether a description, a tag or an example counts. A consumer wanting the narrower
651/// question can compare the members they care about.
652///
653/// Computed over the document with this field ABSENT, so the digest is a function of the
654/// spec's content rather than of itself, and so re-stamping an already-stamped document is
655/// idempotent.
656fn stamp_spec_digest(mut doc: utoipa::openapi::OpenApi) -> utoipa::openapi::OpenApi {
657    use sha2::{Digest, Sha256};
658
659    // Absent, not empty: an empty `extensions` map serializes as no key at all here (the
660    // field is `Option` and flattened), but clearing it explicitly is what makes the digest
661    // computed below independent of whatever was there before.
662    doc.info.extensions = None;
663
664    let canonical = serde_json::to_vec(&doc).expect("OpenAPI serialization failed");
665    let digest = hex::encode(Sha256::digest(&canonical));
666
667    // Exactly one extension, deliberately. `Extensions` is a flattened `HashMap`, whose
668    // iteration order is not stable across runs, so two or more entries here would make the
669    // serialized bytes vary between builds — and the SDK OpenAPI drift gate compares them
670    // byte for byte. A second extension needs an ordered map first.
671    doc.info.extensions = Some(
672        utoipa::openapi::extensions::ExtensionsBuilder::new()
673            .add(SPEC_DIGEST_KEY, format!("sha256:{digest}"))
674            .build(),
675    );
676    doc
677}

679/// Adds Bearer JWT security scheme to the OpenAPI spec.
680struct SecurityAddon;
681
682impl utoipa::Modify for SecurityAddon {
683    fn modify(&self, openapi: &mut utoipa::openapi::OpenApi) {
684        let components = openapi.components.get_or_insert_with(Default::default);
685        components.add_security_scheme(
686            "bearer",
687            utoipa::openapi::security::SecurityScheme::Http(
688                utoipa::openapi::security::HttpBuilder::new()
689                    .scheme(utoipa::openapi::security::HttpAuthScheme::Bearer)
690                    .bearer_format("JWT")
691                    .build(),
692            ),
693        );
694    }
695}

697#[cfg(test)]
698mod spec_digest_tests {
699    use super::{SPEC_DIGEST_KEY, api_doc};
700
701    /// The digest is present, and says which algorithm produced it.
702    #[test]
703    fn the_document_carries_a_digest() {
704        let doc = api_doc();
705        let extensions = doc
706            .info
707            .extensions
708            .as_ref()
709            .expect("info carries no extensions");
710        let value = extensions
711            .get(SPEC_DIGEST_KEY)
712            .and_then(|v| v.as_str())
713            .expect("info carries no spec digest");
714
715        // `sha256:` + 64 hex characters. Naming the algorithm inline is what lets this
716        // change one day without every consumer having to guess from the length.
717        assert!(
718            value.starts_with("sha256:"),
719            "unexpected digest form: {value}"
720        );
721        assert_eq!(value.len(), "sha256:".len() + 64);
722        assert!(
723            value["sha256:".len()..]
724                .chars()
725                .all(|c| c.is_ascii_hexdigit())
726        );
727    }
728
729    /// Two exports of the same build agree.
730    ///
731    /// Not a tautology: the digest is computed over a serialization of the whole document,
732    /// and if any part of that serialization were order-unstable — a `HashMap` somewhere in
733    /// the tree — this would flake. It is the same property the SDK OpenAPI drift gate
734    /// depends on, asserted here where the failure names its cause.
735    #[test]
736    fn the_digest_is_stable_across_exports() {
737        let first = api_doc();
738        let second = api_doc();
739        assert_eq!(
740            first.info.extensions.as_ref().unwrap().get(SPEC_DIGEST_KEY),
741            second
742                .info
743                .extensions
744                .as_ref()
745                .unwrap()
746                .get(SPEC_DIGEST_KEY),
747        );
748        assert_eq!(
749            serde_json::to_string(&first).unwrap(),
750            serde_json::to_string(&second).unwrap(),
751        );
752    }
753
754    /// A document that differs by one path gets a different digest.
755    ///
756    /// This is the whole point of the field, and the reason `info.version` could not serve:
757    /// the version is the release version and moves when a release is cut, so two builds
758    /// describing genuinely different APIs can and do report the same string. `main` and a
759    /// release branch both said `1.0.0-alpha44` while differing by two paths.
760    #[test]
761    fn removing_a_path_changes_the_digest() {
762        let full = api_doc();
763        let before = full
764            .info
765            .extensions
766            .as_ref()
767            .unwrap()
768            .get(SPEC_DIGEST_KEY)
769            .cloned()
770            .unwrap();
771
772        let mut trimmed = api_doc();
773        let victim = trimmed
774            .paths
775            .paths
776            .keys()
777            .next()
778            .expect("the spec has no paths")
779            .clone();
780        // `remove`, not `shift_remove`: `preserve_path_order` is off in this build, so
781        // `PathsMap` is a `BTreeMap`. That is also why the digest is stable — a BTreeMap
782        // serializes in key order rather than in insertion order.
783        trimmed.paths.paths.remove(&victim);
784        let after = super::stamp_spec_digest(trimmed)
785            .info
786            .extensions
787            .as_ref()
788            .unwrap()
789            .get(SPEC_DIGEST_KEY)
790            .cloned()
791            .unwrap();
792
793        assert_ne!(
794            before, after,
795            "dropping {victim} left the digest unchanged, so it does not distinguish content"
796        );
797    }
798
799    /// Re-stamping an already-stamped document is a no-op.
800    ///
801    /// The digest is computed with the field cleared, so it is a function of the content
802    /// rather than of itself. Were it not, stamping twice would produce a different value
803    /// the second time and the field would be unusable for comparison.
804    #[test]
805    fn stamping_is_idempotent() {
806        let once = api_doc();
807        let expected = once
808            .info
809            .extensions
810            .as_ref()
811            .unwrap()
812            .get(SPEC_DIGEST_KEY)
813            .cloned()
814            .unwrap();
815
816        let twice = super::stamp_spec_digest(once);
817        assert_eq!(
818            twice.info.extensions.as_ref().unwrap().get(SPEC_DIGEST_KEY),
819            Some(&expected)
820        );
821    }
