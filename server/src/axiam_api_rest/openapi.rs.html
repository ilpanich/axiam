847/// Build the full OpenAPI document, merging in the SAML SP endpoints when the
848/// `saml` feature is enabled. Use this instead of `ApiDoc::openapi()` directly.
849pub fn api_doc() -> utoipa::openapi::OpenApi {
850    let doc = ApiDoc::openapi();
851    #[cfg(feature = "saml")]
852    let doc = doc.merge_from(SamlApiDoc::openapi());
853    let mut doc = doc;
854    add_oauth_authorization_server_alias(&mut doc);
855    stamp_spec_digest(doc)

858/// Documents `GET /.well-known/oauth-authorization-server` (T21.1) by cloning
859/// the already-generated `PathItem` for `/.well-known/openid-configuration`
860/// rather than hand-writing a second `#[utoipa::path]` annotation on a second
861/// handler. `server.rs` routes both paths to the literal same
862/// `handlers::oauth2::discovery` function, so the two are one operation with
863/// two conventional entry points — a second annotation could drift from the
864/// first (a changed summary, an added parameter) in a way the compiler would
865/// never catch, where cloning the generated item cannot.
866///
867/// Missing the source entry is a bug in `ApiDoc::paths(..)`, not something
868/// this function can recover from, so it is deliberately silent rather than
869/// panicking: `docs-ci.yml`'s OpenAPI parse-check and the digest test in this
870/// module already fail loudly if `ApiDoc` stops documenting discovery at all.
871fn add_oauth_authorization_server_alias(doc: &mut utoipa::openapi::OpenApi) {
872    if let Some(discovery_item) = doc
873        .paths
874        .paths
875        .get("/.well-known/openid-configuration")
876        .cloned()
877    {
878        doc.paths.paths.insert(
879            "/.well-known/oauth-authorization-server".to_string(),
880            discovery_item,
881        );
882    }
883}

885/// The extension key carrying the specification's content digest.
886///
887/// `x-`-prefixed, which is what the OpenAPI specification reserves for extensions, so a
888/// validator and a code generator both ignore it.
889pub const SPEC_DIGEST_KEY: &str = "x-axiam-spec-digest";
890
891/// Stamp `info.x-axiam-spec-digest` with a SHA-256 over this document's own content.
892///
893/// **Why `info.version` is not enough.** It tracks `CARGO_PKG_VERSION`, which is the
894/// release version — it moves when a release is cut, not when a path is added. Two builds
895/// can therefore describe genuinely different APIs under the same string, and they have:
896/// `main` and a release branch both reported `1.0.0-alpha44` while differing by two paths
897/// (`ca-certificates/{id}/migrate-custody` and `.../mtls-trust-anchor`). A consumer pinning
898/// on `info.version` — an SDK deciding whether to regenerate, a gateway caching a spec, a
899/// contract test asserting it is current — could not tell those exports apart, and nothing
900/// in the document let it.
901///
902/// The version keeps its meaning: it is the API's semantic version, and consumers should
903/// keep reading it as one. This is the other question — "is this the same document?" —
904/// answered separately and exactly.
905///
906/// **`info.version` is inside the digest**, so a release bump changes it even when no path
907/// did. Deliberate: "same document" is checkable, where "same API" would need an argument
908/// about whether a description, a tag or an example counts. A consumer wanting the narrower
909/// question can compare the members they care about.
910///
911/// Computed over the document with this field ABSENT, so the digest is a function of the
912/// spec's content rather than of itself, and so re-stamping an already-stamped document is
913/// idempotent.
914fn stamp_spec_digest(mut doc: utoipa::openapi::OpenApi) -> utoipa::openapi::OpenApi {
915    use sha2::{Digest, Sha256};
916
917    // Absent, not empty: an empty `extensions` map serializes as no key at all here (the
918    // field is `Option` and flattened), but clearing it explicitly is what makes the digest
919    // computed below independent of whatever was there before.
920    doc.info.extensions = None;
921
922    let canonical = serde_json::to_vec(&doc).expect("OpenAPI serialization failed");
923    let digest = hex::encode(Sha256::digest(&canonical));
924
925    // Exactly one extension, deliberately. `Extensions` is a flattened `HashMap`, whose
926    // iteration order is not stable across runs, so two or more entries here would make the
927    // serialized bytes vary between builds — and the SDK OpenAPI drift gate compares them
928    // byte for byte. A second extension needs an ordered map first.
929    doc.info.extensions = Some(
930        utoipa::openapi::extensions::ExtensionsBuilder::new()
931            .add(SPEC_DIGEST_KEY, format!("sha256:{digest}"))
932            .build(),
933    );
934    doc
935}

1023    /// The digest is present, and says which algorithm produced it.
1024    #[test]
1025    fn the_document_carries_a_digest() {
1026        let doc = api_doc();
1027        let extensions = doc
1028            .info
1029            .extensions
1030            .as_ref()
1031            .expect("info carries no extensions");
1032        let value = extensions
1033            .get(SPEC_DIGEST_KEY)
1034            .and_then(|v| v.as_str())
1035            .expect("info carries no spec digest");
1036
1037        // `sha256:` + 64 hex characters. Naming the algorithm inline is what lets this
1038        // change one day without every consumer having to guess from the length.
1039        assert!(
1040            value.starts_with("sha256:"),
1041            "unexpected digest form: {value}"
1042        );
1043        assert_eq!(value.len(), "sha256:".len() + 64);
1044        assert!(
1045            value["sha256:".len()..]
1046                .chars()
1047                .all(|c| c.is_ascii_hexdigit())
1048        );
1049    }
1050
1051    /// Two exports of the same build agree.
1052    ///
1053    /// Not a tautology: the digest is computed over a serialization of the whole document,
1054    /// and if any part of that serialization were order-unstable — a `HashMap` somewhere in
1055    /// the tree — this would flake. It is the same property the SDK OpenAPI drift gate
1056    /// depends on, asserted here where the failure names its cause.
1057    #[test]
1058    fn the_digest_is_stable_across_exports() {
1059        let first = api_doc();
1060        let second = api_doc();
1061        assert_eq!(
1062            first.info.extensions.as_ref().unwrap().get(SPEC_DIGEST_KEY),
1063            second
1064                .info
1065                .extensions
1066                .as_ref()
1067                .unwrap()
1068                .get(SPEC_DIGEST_KEY),
1069        );
1070        assert_eq!(
1071            serde_json::to_string(&first).unwrap(),
1072            serde_json::to_string(&second).unwrap(),
1073        );
1074    }
1075
1076    /// A document that differs by one path gets a different digest.
1077    ///
1078    /// This is the whole point of the field, and the reason `info.version` could not serve:
1079    /// the version is the release version and moves when a release is cut, so two builds
1080    /// describing genuinely different APIs can and do report the same string. `main` and a
1081    /// release branch both said `1.0.0-alpha44` while differing by two paths.
1082    #[test]
1083    fn removing_a_path_changes_the_digest() {
1084        let full = api_doc();
1085        let before = full
1086            .info
1087            .extensions
1088            .as_ref()
1089            .unwrap()
1090            .get(SPEC_DIGEST_KEY)
1091            .cloned()
1092            .unwrap();
1093
1094        let mut trimmed = api_doc();
1095        let victim = trimmed
1096            .paths
1097            .paths
1098            .keys()
1099            .next()
1100            .expect("the spec has no paths")
1101            .clone();
1102        // `remove`, not `shift_remove`: `preserve_path_order` is off in this build, so
1103        // `PathsMap` is a `BTreeMap`. That is also why the digest is stable — a BTreeMap
1104        // serializes in key order rather than in insertion order.
1105        trimmed.paths.paths.remove(&victim);
1106        let after = super::stamp_spec_digest(trimmed)
1107            .info
1108            .extensions
1109            .as_ref()
1110            .unwrap()
1111            .get(SPEC_DIGEST_KEY)
1112            .cloned()
1113            .unwrap();
1114
1115        assert_ne!(
1116            before, after,
1117            "dropping {victim} left the digest unchanged, so it does not distinguish content"
1118        );
1119    }
1120
1121    /// Re-stamping an already-stamped document is a no-op.
1122    ///
1123    /// The digest is computed with the field cleared, so it is a function of the content
1124    /// rather than of itself. Were it not, stamping twice would produce a different value
1125    /// the second time and the field would be unusable for comparison.
1126    #[test]
1127    fn stamping_is_idempotent() {
1128        let once = api_doc();
1129        let expected = once
1130            .info
1131            .extensions
1132            .as_ref()
1133            .unwrap()
1134            .get(SPEC_DIGEST_KEY)
1135            .cloned()
1136            .unwrap();
1137
1138        let twice = super::stamp_spec_digest(once);
1139        assert_eq!(
1140            twice.info.extensions.as_ref().unwrap().get(SPEC_DIGEST_KEY),
1141            Some(&expected)
1142        );
1143    }
