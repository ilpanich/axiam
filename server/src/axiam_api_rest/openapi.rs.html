716/// Build the full OpenAPI document, merging in the SAML SP endpoints when the
717/// `saml` feature is enabled. Use this instead of `ApiDoc::openapi()` directly.
718pub fn api_doc() -> utoipa::openapi::OpenApi {
719    let doc = ApiDoc::openapi();
720    #[cfg(feature = "saml")]
721    let doc = doc.merge_from(SamlApiDoc::openapi());
722    let mut doc = doc;
723    add_oauth_authorization_server_alias(&mut doc);
724    stamp_spec_digest(doc)
725}

727/// Documents `GET /.well-known/oauth-authorization-server` (T21.1) by cloning
728/// the already-generated `PathItem` for `/.well-known/openid-configuration`
729/// rather than hand-writing a second `#[utoipa::path]` annotation on a second
730/// handler. `server.rs` routes both paths to the literal same
731/// `handlers::oauth2::discovery` function, so the two are one operation with
732/// two conventional entry points — a second annotation could drift from the
733/// first (a changed summary, an added parameter) in a way the compiler would
734/// never catch, where cloning the generated item cannot.
735///
736/// Missing the source entry is a bug in `ApiDoc::paths(..)`, not something
737/// this function can recover from, so it is deliberately silent rather than
738/// panicking: `docs-ci.yml`'s OpenAPI parse-check and the digest test in this
739/// module already fail loudly if `ApiDoc` stops documenting discovery at all.
740fn add_oauth_authorization_server_alias(doc: &mut utoipa::openapi::OpenApi) {
741    if let Some(discovery_item) = doc
742        .paths
743        .paths
744        .get("/.well-known/openid-configuration")
745        .cloned()
746    {
747        doc.paths.paths.insert(
748            "/.well-known/oauth-authorization-server".to_string(),
749            discovery_item,
750        );
751    }
752}

754/// The extension key carrying the specification's content digest.
755///
756/// `x-`-prefixed, which is what the OpenAPI specification reserves for extensions, so a
757/// validator and a code generator both ignore it.
758pub const SPEC_DIGEST_KEY: &str = "x-axiam-spec-digest";
759
760/// Stamp `info.x-axiam-spec-digest` with a SHA-256 over this document's own content.
761///
762/// **Why `info.version` is not enough.** It tracks `CARGO_PKG_VERSION`, which is the
763/// release version — it moves when a release is cut, not when a path is added. Two builds
764/// can therefore describe genuinely different APIs under the same string, and they have:
765/// `main` and a release branch both reported `1.0.0-alpha44` while differing by two paths
766/// (`ca-certificates/{id}/migrate-custody` and `.../mtls-trust-anchor`). A consumer pinning
767/// on `info.version` — an SDK deciding whether to regenerate, a gateway caching a spec, a
768/// contract test asserting it is current — could not tell those exports apart, and nothing
769/// in the document let it.
770///
771/// The version keeps its meaning: it is the API's semantic version, and consumers should
772/// keep reading it as one. This is the other question — "is this the same document?" —
773/// answered separately and exactly.
774///
775/// **`info.version` is inside the digest**, so a release bump changes it even when no path
776/// did. Deliberate: "same document" is checkable, where "same API" would need an argument
777/// about whether a description, a tag or an example counts. A consumer wanting the narrower
778/// question can compare the members they care about.
779///
780/// Computed over the document with this field ABSENT, so the digest is a function of the
781/// spec's content rather than of itself, and so re-stamping an already-stamped document is
782/// idempotent.
783fn stamp_spec_digest(mut doc: utoipa::openapi::OpenApi) -> utoipa::openapi::OpenApi {
784    use sha2::{Digest, Sha256};
785
786    // Absent, not empty: an empty `extensions` map serializes as no key at all here (the
787    // field is `Option` and flattened), but clearing it explicitly is what makes the digest
788    // computed below independent of whatever was there before.
789    doc.info.extensions = None;
790
791    let canonical = serde_json::to_vec(&doc).expect("OpenAPI serialization failed");
792    let digest = hex::encode(Sha256::digest(&canonical));
793
794    // Exactly one extension, deliberately. `Extensions` is a flattened `HashMap`, whose
795    // iteration order is not stable across runs, so two or more entries here would make the
796    // serialized bytes vary between builds — and the SDK OpenAPI drift gate compares them
797    // byte for byte. A second extension needs an ordered map first.
798    doc.info.extensions = Some(
799        utoipa::openapi::extensions::ExtensionsBuilder::new()
800            .add(SPEC_DIGEST_KEY, format!("sha256:{digest}"))
801            .build(),
802    );
803    doc
804}

806/// Adds the two Bearer JWT security schemes to the OpenAPI spec.
807///
808/// Both are the same HTTP mechanism; they differ in who may hold the token,
809/// which is the `aud` claim. An operation that admits a service account lists
810/// both as alternatives, so the document says which routes a machine may call
811/// rather than leaving it to the prose (S-9, D-5). `bearer` alone means a
812/// person.
813struct SecurityAddon;
814
815impl utoipa::Modify for SecurityAddon {
816    fn modify(&self, openapi: &mut utoipa::openapi::OpenApi) {
817        use utoipa::openapi::security::{HttpAuthScheme, HttpBuilder, SecurityScheme};
818
819        let components = openapi.components.get_or_insert_with(Default::default);
820        components.add_security_scheme(
821            "bearer",
822            SecurityScheme::Http(
823                HttpBuilder::new()
824                    .scheme(HttpAuthScheme::Bearer)
825                    .bearer_format("JWT")
826                    .description(Some(
827                        "An AXIAM access token issued to a user (`aud` = `axiam:user`), \
828                         from the `axiam_access` cookie or the `Authorization` header.",
829                    ))
830                    .build(),
831            ),
832        );
833        components.add_security_scheme(
834            SERVICE_ACCOUNT_SECURITY_SCHEME,
835            SecurityScheme::Http(
836                HttpBuilder::new()
837                    .scheme(HttpAuthScheme::Bearer)
838                    .bearer_format("JWT")
839                    .description(Some(
840                        "An AXIAM access token issued to a service account \
841                         (`aud` = `axiam:m2m`, `sub_kind` = `service_account`), from \
842                         client credentials or the mTLS device login. Accepted only \
843                         by the operations that list this scheme — the management \
844                         families resources, scopes, permissions, roles, groups, \
845                         service accounts, certificates and webhooks, and the \
846                         authorization checks — and authorized there by the roles \
847                         assigned to the account, exactly as a user is. \
848                         `X-Axiam-Tenant` is honoured only for an account living in \
849                         the organization scope, as for a user; a service account \
850                         never issues under the organization CA. A certificate-bound token \
851                         (`cnf.x5t#S256`) must be presented over mTLS with that \
852                         certificate.",
853                    ))
854                    .build(),
855            ),
856        );
857    }
858}

864#[cfg(test)]
865mod spec_digest_tests {
866    use super::{SPEC_DIGEST_KEY, api_doc};
867
868    /// The digest is present, and says which algorithm produced it.
869    #[test]
870    fn the_document_carries_a_digest() {
871        let doc = api_doc();
872        let extensions = doc
873            .info
874            .extensions
875            .as_ref()
876            .expect("info carries no extensions");
877        let value = extensions
878            .get(SPEC_DIGEST_KEY)
879            .and_then(|v| v.as_str())
880            .expect("info carries no spec digest");
881
882        // `sha256:` + 64 hex characters. Naming the algorithm inline is what lets this
883        // change one day without every consumer having to guess from the length.
884        assert!(
885            value.starts_with("sha256:"),
886            "unexpected digest form: {value}"
887        );
888        assert_eq!(value.len(), "sha256:".len() + 64);
889        assert!(
890            value["sha256:".len()..]
891                .chars()
892                .all(|c| c.is_ascii_hexdigit())
893        );
894    }
895
896    /// Two exports of the same build agree.
897    ///
898    /// Not a tautology: the digest is computed over a serialization of the whole document,
899    /// and if any part of that serialization were order-unstable — a `HashMap` somewhere in
900    /// the tree — this would flake. It is the same property the SDK OpenAPI drift gate
901    /// depends on, asserted here where the failure names its cause.
902    #[test]
903    fn the_digest_is_stable_across_exports() {
904        let first = api_doc();
905        let second = api_doc();
906        assert_eq!(
907            first.info.extensions.as_ref().unwrap().get(SPEC_DIGEST_KEY),
908            second
909                .info
910                .extensions
911                .as_ref()
912                .unwrap()
913                .get(SPEC_DIGEST_KEY),
914        );
915        assert_eq!(
916            serde_json::to_string(&first).unwrap(),
917            serde_json::to_string(&second).unwrap(),
918        );
919    }
920
921    /// A document that differs by one path gets a different digest.
922    ///
923    /// This is the whole point of the field, and the reason `info.version` could not serve:
924    /// the version is the release version and moves when a release is cut, so two builds
925    /// describing genuinely different APIs can and do report the same string. `main` and a
926    /// release branch both said `1.0.0-alpha44` while differing by two paths.
927    #[test]
928    fn removing_a_path_changes_the_digest() {
929        let full = api_doc();
930        let before = full
931            .info
932            .extensions
933            .as_ref()
934            .unwrap()
935            .get(SPEC_DIGEST_KEY)
936            .cloned()
937            .unwrap();
938
939        let mut trimmed = api_doc();
940        let victim = trimmed
941            .paths
942            .paths
943            .keys()
944            .next()
945            .expect("the spec has no paths")
946            .clone();
947        // `remove`, not `shift_remove`: `preserve_path_order` is off in this build, so
948        // `PathsMap` is a `BTreeMap`. That is also why the digest is stable — a BTreeMap
949        // serializes in key order rather than in insertion order.
950        trimmed.paths.paths.remove(&victim);
951        let after = super::stamp_spec_digest(trimmed)
952            .info
953            .extensions
954            .as_ref()
955            .unwrap()
956            .get(SPEC_DIGEST_KEY)
957            .cloned()
958            .unwrap();
959
960        assert_ne!(
961            before, after,
962            "dropping {victim} left the digest unchanged, so it does not distinguish content"
963        );
964    }
965
966    /// Re-stamping an already-stamped document is a no-op.
967    ///
968    /// The digest is computed with the field cleared, so it is a function of the content
969    /// rather than of itself. Were it not, stamping twice would produce a different value
970    /// the second time and the field would be unusable for comparison.
971    #[test]
972    fn stamping_is_idempotent() {
973        let once = api_doc();
974        let expected = once
975            .info
976            .extensions
977            .as_ref()
978            .unwrap()
979            .get(SPEC_DIGEST_KEY)
980            .cloned()
981            .unwrap();
982
983        let twice = super::stamp_spec_digest(once);
984        assert_eq!(
985            twice.info.extensions.as_ref().unwrap().get(SPEC_DIGEST_KEY),
986            Some(&expected)
987        );
988    }
989}
