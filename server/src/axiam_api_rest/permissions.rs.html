1//! Compile-time permission registry and public-path allowlist.
2//!
3//! [`PERMISSION_REGISTRY`] lists every `(action, description)` pair that the
4//! permission seeder will UPSERT into SurrealDB on startup.
5//!
6//! [`PUBLIC_PATHS`] enumerates every path that the [`AuthzMiddleware`] must
7//! allow through without requiring a JWT (D-04).
8//!
9//! [`ROUTE_PERMISSION_MAP`] maps `(HTTP_METHOD, path_pattern, permission)`
10//! and is used by the integration test in Plan 05 to verify that every
11//! registered route has an authorisation check (D-08).
12

247// Public-path allowlist (D-04)
248// ---------------------------------------------------------------------------
249
250/// Paths that do NOT require authentication.
251///
252/// Matching rules applied by `AuthzMiddleware`:
253/// - Entries ending with `*` are prefix-matched (strip the `*`, use `starts_with`).
254/// - All other entries are exact-matched against `req.path()`.
255pub const PUBLIC_PATHS: &[&str] = &[
256    // Authentication flows (under /api/v1/auth scope)
257    "/api/v1/auth/login",
258    // Refresh authenticates via its own opaque `axiam_refresh` cookie, not the
259    // access token. It MUST be public: the `axiam_access` cookie expires with
260    // the 15-min access token, so a client refreshing after expiry carries no
261    // access credential — `AuthzMiddleware` would otherwise 401 it before the
262    // handler runs, making token rotation impossible.
263    "/api/v1/auth/refresh",
264    // SEC-047: /api/v1/auth/register removed from public paths — registration is now gated.
265    "/api/v1/auth/device",
266    "/api/v1/auth/mfa/verify",
267    "/api/v1/auth/mfa/setup/enroll",
268    "/api/v1/auth/mfa/setup/confirm",
269    "/api/v1/auth/reset",
270    "/api/v1/auth/reset/confirm",
271    "/api/v1/auth/verify-email",
272    "/api/v1/auth/resend-verification",
273    // WebAuthn — public registration/authentication initiation
274    "/api/v1/auth/webauthn/register/start",
275    "/api/v1/auth/webauthn/register/finish",
276    "/api/v1/auth/webauthn/authenticate/start",
277    "/api/v1/auth/webauthn/authenticate/finish",
278    // Health probes
279    "/health",
280    "/ready",
281    // OIDC discovery and token endpoints
282    "/.well-known/openid-configuration",
283    // UMA 2.0 discovery (X2). Public for the same reason as OIDC discovery:
284    // §2 makes it the document a resource server fetches *before* it holds any
285    // credential, and it carries only endpoint URLs the deployment publishes.
286    "/.well-known/uma2-configuration",
287    "/oauth2/jwks",
288    "/oauth2/authorize",
289    "/oauth2/token",
290    "/oauth2/userinfo",
291    "/oauth2/revoke",
292    "/oauth2/introspect",
293    // B2 / RFC 8628 §3.1. Necessarily public: the grant exists precisely for
294    // clients that cannot hold a secret (a television, a headless CLI), so
295    // there is no credential the device could present here. `client_id` is
296    // still checked against the tenant's registered clients inside the
297    // handler — the endpoint is unauthenticated, not unvalidated.
298    "/oauth2/device_authorization",
299    // B5 / RFC 9126. Public to `AuthzMiddleware` in the same sense the token
300    // endpoint is: it takes no AXIAM session or bearer token, because the
301    // caller is a client authenticating with its own credentials in the form
302    // body — which the handler verifies through the same
303    // `authenticate_client` path the token endpoint uses. Unauthenticated by
304    // the middleware's definition, not by the endpoint's.
305    "/oauth2/par",
306    // B5 / RP-Initiated Logout 1.0 §2. Necessarily public: a user whose
307    // session has ALREADY expired must still be able to complete a logout,
308    // and requiring a live session to end a session is a contradiction. The
309    // endpoint identifies what to terminate from a *signed* `id_token_hint`,
310    // so it is unauthenticated, not unauthenticated-and-unbounded.
311    "/oauth2/end_session",
312    // Federation callback endpoints (unauthenticated — IdP redirects here)
313    "/api/v1/federation/oidc/callback",
314    "/api/v1/federation/saml/acs",
315    // NOTE: /api/v1/federation/saml/metadata is intentionally NOT public — its
316    // handler requires an authenticated admin (AuthenticatedUser/JWT). It was
317    // previously listed here, which contradicted the handler and returned 401
318    // to unauthenticated callers (Phase 28 FUNC-01 verification, D-15 scope
319    // decision: keep metadata JWT-gated rather than making it public).
320    // First-time SSO (Phase 4 D-22) — unauthenticated, distinct from
321    // /api/v1/federation/* link-account endpoints (which require auth).
322    "/api/v1/auth/federation/oidc/start",
323    "/api/v1/auth/federation/oidc/callback",
324    "/api/v1/auth/federation/saml/login",
325    "/api/v1/auth/federation/saml/acs",
326    // Admin bootstrap (public until first admin is created; handler enforces one-shot logic)
327    "/api/v1/admin/bootstrap",
328    // OpenAPI docs
329    "/api/docs/*",
330    // GDPR delete-cancel link (emailed, public — single-use token-gated, D-09)
331    "/api/v1/auth/account/delete/cancel",
