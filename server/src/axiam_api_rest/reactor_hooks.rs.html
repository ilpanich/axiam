1//! The three REST-side reactor hooks (X1 / R2.2).
2//!
3//! `login.post_auth` lives in `axiam-auth` and `token.pre_issue` in
4//! `axiam-oauth2`, because those are the crates that own the operation. The
5//! other three — `user.pre_create`, `user.pre_update` and `grant.pre_assign` —
6//! have no service layer between the handler and the repository, so the handler
7//! *is* the operation and this module is where they run.
8//!
9//! # What these functions are, and what they are deliberately not
10//!
11//! Each is "call the gate, then apply the answer". They are **not** a second
12//! allow-list. A patch key is checked against the event's `mutable_fields`
13//! three times before it reaches [`apply_user_patch`] — by the reply validator
14//! (`ReactorReply::into_outcome`), by the gate's own re-check
15//! (`DispatchingReactorGate::enforce_allow_list`), and by the exhaustive
16//! `match` here that has no arm for anything else. The third is the one that
17//! makes the property structural: adding a field to the allow-list without
18//! deciding what it *means* here does not silently start writing it, because
19//! there is nowhere for it to go.
20

31/// The metadata namespace a `user.pre_create` / `user.pre_update` patch may
32/// write into, matching the registry's `metadata.` allow-list entry.
33const METADATA_PREFIX: &str = "metadata.";
34
35/// Turn a reactor veto into the error the handler returns.
36///
37/// `403`, not `400`: the request was well-formed and the caller was
38/// authenticated and permitted — an external policy refused it. The reactor's
39/// own text **is** surfaced here, unlike on the login and token paths, because
40/// this caller has already proven a `users:create`-class permission and is an
41/// administrator who needs to know why their write was refused. The audit
42/// record carries the same string plus the reactor's id.
43fn veto(action: &str, reason: String) -> AxiamApiError {
44    AxiamApiError(AxiamError::AuthorizationDenied {
45        reason: format!("refused by a reactor: {reason}"),
46        action: Some(action.to_string()),
47        resource_id: None,
48    })
49}
50
51/// A mutation on a veto-only event, or an outcome the site cannot honour.
52fn impossible(action: &str, what: &str) -> AxiamApiError {
53    tracing::error!(
54        target: "axiam::reactor",
55        action,
56        outcome = what,
57        "a reactor outcome reached a call site that cannot honour it — refusing \
58         the operation rather than proceeding as if it had not happened"
59    );
60    AxiamApiError(AxiamError::AuthorizationDenied {
61        reason: "refused by a reactor".into(),
62        action: Some(action.to_string()),
63        resource_id: None,
64    })
65}

67/// Apply an allow-listed `user.pre_*` patch.
68///
69/// Returns the merged metadata object, or `None` when the patch touched no
70/// metadata key. The caller decides whether a `None` means "leave it alone"
71/// (update) or "use what the request supplied" (create), because those differ.
72fn apply_user_patch(
73    patch: &BTreeMap<String, String>,
74    username: &mut Option<String>,
75    email: &mut Option<String>,
76    metadata: &mut Option<serde_json::Value>,
77) {
78    for (key, value) in patch {
79        match key.as_str() {
80            "username" => *username = Some(value.clone()),
81            "email" => *email = Some(value.clone()),
82            other => {
83                let Some(field) = other
84                    .strip_prefix(METADATA_PREFIX)
85                    .filter(|f| !f.is_empty())
86                else {
87                    // Unreachable: three checks upstream refuse it. Logged at
88                    // ERROR rather than ignored, because reaching this line
89                    // means one of them has a hole.
90                    tracing::error!(
91                        target: "axiam::reactor",
92                        field = %other,
93                        "a user patch field outside the allow-list reached the handler; \
94                         dropping it"
95                    );
96                    continue;
97                };
98                let object =
99                    metadata.get_or_insert_with(|| serde_json::Value::Object(Default::default()));
100                if !object.is_object() {
101                    // The request supplied a non-object `metadata`. The patch
102                    // namespace is defined in terms of object keys, so replace
103                    // rather than guess at a merge.
104                    *object = serde_json::Value::Object(Default::default());
105                }
106                if let Some(map) = object.as_object_mut() {
107                    map.insert(field.to_string(), serde_json::Value::String(value.clone()));
108                }
109            }
110        }
111    }
112}

114/// Run `user.pre_create`, applying any allow-listed mutation to `input`.
115pub async fn user_pre_create(
116    gate: &SharedReactorGate,
117    tenant_id: Uuid,
118    input: &mut CreateUser,
119) -> Result<(), AxiamApiError> {
120    let outcome = gate
121        .intercept(
122            tenant_id,
123            reactor_events::USER_PRE_CREATE,
124            serde_json::json!({
125                // Never the password (§22.3): a reactor validates a profile,
126                // it is not handed a credential.
127                "username": input.username,
128                "email": input.email,
129                "metadata": input.metadata,
130                "tenant_id": tenant_id,
131            }),
132        )
133        .await;
134
135    match outcome {
136        ReactorOutcome::Allow => Ok(()),
137        ReactorOutcome::Deny { reason } => Err(veto("users:create", reason)),
138        ReactorOutcome::Mutate { patch } => {
139            let mut username = None;
140            let mut email = None;
141            let mut metadata = input.metadata.clone();
142            apply_user_patch(&patch, &mut username, &mut email, &mut metadata);
143            if let Some(username) = username {
144                input.username = username;
145            }
146            if let Some(email) = email {
147                input.email = email;
148            }
149            input.metadata = metadata;
150            Ok(())
151        }
152        ReactorOutcome::RequireMfa => Err(impossible("users:create", "require_mfa")),
153    }
154}

156/// Run `user.pre_update`, applying any allow-listed mutation to `input`.
157pub async fn user_pre_update(
158    gate: &SharedReactorGate,
159    tenant_id: Uuid,
160    target_id: Uuid,
161    input: &mut UpdateUser,
162) -> Result<(), AxiamApiError> {
163    let outcome = gate
164        .intercept(
165            tenant_id,
166            reactor_events::USER_PRE_UPDATE,
167            serde_json::json!({
168                "user_id": target_id,
169                // The *proposed* change, which is what is being decided. Fields
170                // the request did not touch are absent rather than null, so a
171                // reactor can tell "unchanged" from "cleared".
172                "username": input.username,
173                "email": input.email,
174                "metadata": input.metadata,
175                "tenant_id": tenant_id,
176            }),
177        )
178        .await;
179
180    match outcome {
181        ReactorOutcome::Allow => Ok(()),
182        ReactorOutcome::Deny { reason } => Err(veto("users:update", reason)),
183        ReactorOutcome::Mutate { patch } => {
184            let mut metadata = input.metadata.clone();
185            apply_user_patch(&patch, &mut input.username, &mut input.email, &mut metadata);
186            input.metadata = metadata;
187            Ok(())
188        }
189        ReactorOutcome::RequireMfa => Err(impossible("users:update", "require_mfa")),
190    }
191}

193/// Run `grant.pre_assign` — the four-eyes hook. Veto only.
194pub async fn grant_pre_assign(
195    gate: &SharedReactorGate,
196    tenant_id: Uuid,
197    payload: serde_json::Value,
198) -> Result<(), AxiamApiError> {
199    match gate
200        .intercept(tenant_id, reactor_events::GRANT_PRE_ASSIGN, payload)
201        .await
202    {
203        ReactorOutcome::Allow => Ok(()),
204        ReactorOutcome::Deny { reason } => Err(veto("roles:assign", reason)),
205        // The registry marks this event `mutable: false` and carries no
206        // step-up notion; both are refused by the reply validator and again by
207        // the gate. Refusing here too means the assignment cannot proceed on an
208        // answer nobody can apply.
209        ReactorOutcome::Mutate { .. } => Err(impossible("roles:assign", "mutate")),
210        ReactorOutcome::RequireMfa => Err(impossible("roles:assign", "require_mfa")),
211    }
212}

220    /// A gate that always answers the same thing — enough to prove each site
221    /// reads the outcome it is given. The per-site allow/deny/mutate/timeout
222    /// matrix against a *real* chain lives in `axiam-amqp::reactor::gate`.
223    struct Fixed(ReactorOutcome);
224
225    impl DynReactorGate for Fixed {
226        fn intercept_dyn<'a>(
227            &'a self,
228            _tenant_id: Uuid,
229            _event: &'static str,
230            _payload: serde_json::Value,
231        ) -> std::pin::Pin<Box<dyn std::future::Future<Output = ReactorOutcome> + Send + 'a>>
232        {
233            let outcome = self.0.clone();
234            Box::pin(std::future::ready(outcome))
235        }
236
237        // SEC-101: a scripted test double stands in for a working
238        // transport, so registrations are acceptable under it.
239        fn can_dispatch_dyn(&self) -> bool {
240            true
241        }
242    }
243
244    fn gate(outcome: ReactorOutcome) -> SharedReactorGate {
245        Arc::new(Fixed(outcome))
246    }
247
248    fn patch(pairs: &[(&str, &str)]) -> ReactorOutcome {
249        ReactorOutcome::Mutate {
250            patch: pairs
251                .iter()
252                .map(|(k, v)| (k.to_string(), v.to_string()))
253                .collect(),
254        }
255    }

257    fn create_input() -> CreateUser {
258        CreateUser {
259            tenant_id: Uuid::new_v4(),
260            username: "alice".into(),
261            email: "alice@example.com".into(),
262            // Generated, never a literal: a username/password pair sitting in
263            // source trips secret scanners, and this test only cares that the
264            // value survives untouched — not what it is.
265            password: format!("pw-{}", Uuid::new_v4()),
266            metadata: None,
267        }
268    }

324    /// The wiring-layer half of the allow-list guarantee: a field outside the
325    /// event's list has nowhere to go, so it cannot be written even if it
326    /// arrives.
327    #[tokio::test]
328    async fn a_field_outside_the_allow_list_cannot_be_written_by_the_handler() {
329        let mut input = create_input();
330        let original_password = input.password.clone();
331        let original_tenant = input.tenant_id;
332
333        // These are exactly the keys `EVENT_REGISTRY` refuses for
334        // `user.pre_create`. Reaching `apply_user_patch` with them is
335        // impossible through the wire path; the point of the test is that the
336        // handler is not the layer that would let them through.
337        user_pre_create(
338            &gate(patch(&[
339                ("password", "hunter2"),
340                ("password_hash", "$argon2id$xxx"),
341                ("tenant_id", &Uuid::new_v4().to_string()),
342                ("roles", "admin"),
343                ("is_admin", "true"),
344                ("metadata", "not-a-namespace"),
345            ])),
346            input.tenant_id,
347            &mut input,
348        )
349        .await
350        .expect("the outcome is applied, minus everything it may not touch");
351
352        assert_eq!(input.password, original_password);
353        assert_eq!(input.tenant_id, original_tenant);
354        assert_eq!(input.username, "alice");
355        assert_eq!(input.email, "alice@example.com");
356        assert!(
357            input.metadata.is_none(),
358            "bare `metadata` is not the `metadata.` namespace"
359        );
360    }
