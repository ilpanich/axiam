21/// Build a per-endpoint Governor middleware instance from a requests-per-minute
22/// limit.
23///
24/// Each call creates an independent in-memory store — never share configs
25/// between endpoints with different limits (that would merge their counters).
26///
27/// **Public for cross-crate route modules only** (R5.2): `axiam-scim` mounts
28/// `/scim/v2` from its own crate but must be limited by the SAME two layers
29/// as every family wired here — this per-replica governor plus
30/// [`crate::middleware::rate_limit_shared::RateLimitShared`] as the outer
31/// wrap. Exporting the constructor is what lets it reuse the wiring instead
32/// of inventing a parallel one. It stays IP-keyed; use
33/// `build_client_aware_governor` only for the three OAuth2 endpoints that
34/// carry a form-encoded `client_id`.
35pub fn build_governor(
36    requests_per_min: u32,
37) -> Governor<XForwardedForKeyExtractor, NoOpMiddleware> {
38    // SEC-048: trusted_hops=0 default; override via AXIAM__RATE_LIMIT__TRUSTED_HOPS
39    // when running behind a single ingress/nginx layer (set to 1).
40    let trusted_hops: usize = std::env::var("AXIAM__RATE_LIMIT__TRUSTED_HOPS")
41        .ok()
42        .and_then(|v| v.parse().ok())
43        .unwrap_or(0);
44    let config = GovernorConfigBuilder::default()
45        .requests_per_minute(requests_per_min as u64)
46        .burst_size(requests_per_min)
47        .key_extractor(XForwardedForKeyExtractor::with_trusted_hops(trusted_hops))
48        .finish()
49        .expect("valid governor config");
50    Governor::new(&config)
51}

53/// D8: build a per-endpoint Governor middleware instance for the three
54/// endpoints where an OAuth2 client identity (`client_id`) is available —
55/// `/oauth2/token`, `/oauth2/revoke`, `/oauth2/introspect` ONLY.
56///
57/// `key_mode` (`AXIAM__RATE_LIMIT__KEY`, from [`RateLimitConfig::key`])
58/// controls whether the bucket key is IP alone (default, unchanged
59/// behavior), the OAuth2 `client_id` alone, or the `(ip, client_id)` pair —
60/// see [`crate::config::rate_limit::RateLimitKeyMode`] for the full
61/// NAT'd-fleet rationale. The `client_id` itself is read from request
62/// extensions populated by `middleware::rate_limit_shared::RateLimitShared`
63/// (wired as the OUTER `.wrap()` on the same resource, so it always runs
64/// first — see that middleware's docs).
65///
66/// **Never call this for `/auth/login`** (or any other endpoint without a
67/// client identity) — use [`build_governor`] there instead, unconditionally.
68fn build_client_aware_governor(
69    requests_per_min: u32,
70    key_mode: RateLimitKeyMode,
71) -> Governor<ClientAwareKeyExtractor, NoOpMiddleware> {
72    let trusted_hops: usize = std::env::var("AXIAM__RATE_LIMIT__TRUSTED_HOPS")
73        .ok()
74        .and_then(|v| v.parse().ok())
75        .unwrap_or(0);
76    let config = GovernorConfigBuilder::default()
77        .requests_per_minute(requests_per_min as u64)
78        .burst_size(requests_per_min)
79        .key_extractor(ClientAwareKeyExtractor::new(
80            XForwardedForKeyExtractor::with_trusted_hops(trusted_hops),
81            key_mode,
82        ))
83        .finish()
84        .expect("valid governor config");
85    Governor::new(&config)
86}

88/// Register health and readiness routes.
89///
90/// Generic over `C` (QUAL-01) since `/ready` now extracts
91/// `web::Data<AppState<C>>` for its `HealthChecker`.
92pub fn health_routes<C: surrealdb::Connection + Clone>(cfg: &mut web::ServiceConfig) {
93    cfg.route("/health", web::get().to(crate::health::health))
94        .route("/ready", web::get().to(crate::health::ready::<C>))
95        // T-129. Unauthenticated like the other two probes: it reports
96        // whether background sweeps are running and when they last
97        // succeeded, which is operational metadata about this server, not
98        // tenant data — the same category as `/ready` disclosing that the
99        // database is reachable.
100        .route("/health/jobs", web::get().to(crate::health::jobs::<C>));
101}

1222/// Build CORS middleware from configuration.
1223///
1224/// An empty `allowed_origins` slice yields a restrictive default policy
1225/// (denying cross-origin requests). A populated slice restricts CORS to the
1226/// specified origins.
1227pub fn build_cors(allowed_origins: &[String]) -> Cors {
1228    let mut cors = Cors::default()
1229        .allowed_methods(vec!["GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"])
1230        .allowed_headers(vec![
1231            header::AUTHORIZATION,
1232            header::CONTENT_TYPE,
1233            header::ACCEPT,
1234        ])
1235        .allowed_header("X-CSRF-Token")
1236        .max_age(3600);
1237
1238    for origin in allowed_origins {
1239        cors = cors.allowed_origin(origin);
1240    }
1241    cors
1242}
