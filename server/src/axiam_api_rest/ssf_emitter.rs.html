1//! The Shared Signals Framework event emitter (G-5, T23.5.3, D-52).
2//!
3//! Neither source G-5 first named can be tailed — the revocation feed
4//! publishes unlinkable hashes by design (T-39) and most audit actions are the
5//! middleware's `{METHOD} {path}` — so **events are emitted where the change
6//! happens**, through this one emitter:
7//!
8//! ```text
9//!   the change site ──► SsfEmitter ──► SsfStreamRepository::list_for_event
10//!                                  ──► prepare_event (one per stream)
11//!                                  ──► SsfOutbox::submit (push: dispatcher, poll/paused: buffer)
12//! ```
13//!
14//! # Best effort, like `emit_webhook`
15//!
16//! An emission is a side effect of some other operation — a logout, a password
17//! change, a disabled account — and never fails it: every error is logged
18//! (never with a subject, a SET or a credential) and swallowed. It is a no-op
19//! when no outbox is wired (a harness, a deployment without AMQP) and when the
20//! tenant's `ssf_enabled` is off, and it costs one indexed read — the streams
21//! that would carry the event — when nothing is registered, before it reads a
22//! user or a setting.
23//!
24//! # One `txn` per originating operation
25//!
26//! A password reset produces a `credential-change` and a `session-revoked`; they
27//! share one `txn` (SSF §4.1.9) so a receiver can tell they are one cause. The
28//! operation says so by running inside [`with_cause`]; the emitter reads the
29//! cause from a task-local, so the code between the handler and the repository
30//! that actually removes a session needs no extra parameter. An emission outside
31//! any cause makes a `txn` of its own.
32//!
33//! # Two ports
34//!
35//! The emitter also implements the two core ports a layer that cannot reach
36//! this crate reports through: [`SessionRevocationSink`] (the session
37//! repository, D-52's only source of `session-revoked`) and
38//! [`SsfSystemAccountSink`] (the directory sync's deactivation). Both are bound
39//! to it through [`Late`] handles because the repositories that hold them are
40//! built before the outbox exists.
41

68// ---------------------------------------------------------------------------
69// The cause of an operation

72/// What one originating operation tells every event it produces.
73#[derive(Debug, Clone)]
74pub struct SsfCause {
75    /// The SET `txn`, shared by every SET this operation produces.
76    pub txn: String,
77    /// Who did it, for the events that name an initiator (the
78    /// `session-revoked` the session repository reports on its own).
79    pub initiating_entity: Option<InitiatingEntity>,
80}
81
82tokio::task_local! {
83    static CAUSE: SsfCause;
84}
85
86/// Run `operation` as one cause: every event emitted inside it shares one `txn`
87/// and, for a revocation the session repository reports, one initiator.
88///
89/// Wrap the whole operation — the service call that revokes sessions and the
90/// emission of the credential change — not one of them.
91pub async fn with_cause<F: Future>(
92    initiating_entity: Option<InitiatingEntity>,
93    operation: F,
94) -> F::Output {
95    CAUSE
96        .scope(
97            SsfCause {
98                txn: Uuid::new_v4().to_string(),
99                initiating_entity,
100            },
101            operation,
102        )
103        .await
104}
105
106/// The cause of the operation this code runs inside, if it runs inside one.
107#[must_use]
108pub fn current_cause() -> Option<SsfCause> {
109    CAUSE.try_with(Clone::clone).ok()
110}

123/// Emits CAEP and RISC events from the places a change happens.
124///
125/// Cheap to clone: the repositories are handles and the outbox is shared, so a
126/// clone bound before the outbox exists sees it when it is bound.
127#[derive(Clone)]
128pub struct SsfEmitter<C: Connection + Clone> {
129    stream_repo: SurrealSsfStreamRepository<C>,
130    tenant_repo: SurrealTenantRepository<C>,
131    settings_repo: SurrealSettingsRepository<C>,
132    user_repo: SurrealUserRepository<C>,
133    auth_config: AuthConfig,
134    outbox: Arc<Late<dyn SsfOutbox>>,
135    /// D-55: while the deployment's tenants would share one issuer nothing is
136    /// produced, for any tenant.
137    gate: Arc<SsfIssuerGate>,
138}

140impl<C: Connection + Clone> SsfEmitter<C> {
141    /// An emitter with no outbox: it does nothing until [`Self::bind_outbox`].
142    pub fn new(
143        stream_repo: SurrealSsfStreamRepository<C>,
144        tenant_repo: SurrealTenantRepository<C>,
145        settings_repo: SurrealSettingsRepository<C>,
146        user_repo: SurrealUserRepository<C>,
147        auth_config: AuthConfig,
148        gate: Arc<SsfIssuerGate>,
149    ) -> Self {
150        Self {
151            stream_repo,
152            tenant_repo,
153            settings_repo,
154            user_repo,
155            auth_config,
156            outbox: Arc::new(Late::default()),
157            gate,
158        }
159    }

161    /// Wire the outbox events go to. Bound once: a second call is `false` and
162    /// changes nothing.
163    pub fn bind_outbox(&self, outbox: Arc<dyn SsfOutbox>) -> bool {
164        self.outbox.bind(outbox)
165    }
166
167    /// Whether an outbox is wired. Without one nothing is emitted.
168    #[must_use]
169    pub fn is_wired(&self) -> bool {
170        self.outbox.get().is_some()
171    }

173    /// Whether the tenant exists and its effective `ssf_enabled` is on. Any
174    /// failure reads as off: an emission that cannot tell must not guess on.
175    async fn switched_on(&self, tenant_id: Uuid) -> bool {
176        let tenant = match self.tenant_repo.get_by_id(tenant_id).await {
177            Ok(tenant) => tenant,
178            Err(AxiamError::NotFound { .. }) => return false,
179            Err(error) => {
180                tracing::warn!(target: "axiam::ssf", %tenant_id, %error, "SSF emission could not read the tenant");
181                return false;
182            }
183        };
184        let switch_on = match self
185            .settings_repo
186            .get_effective_settings(tenant.organization_id, tenant_id)
187            .await
188        {
189            Ok(settings) => settings.oidc.ssf_enabled,
190            Err(error) => {
191                tracing::warn!(target: "axiam::ssf", %tenant_id, %error, "SSF emission could not read the settings");
192                false
193            }
194        };
195        if !switch_on {
196            return false;
197        }
198        // D-55: the deployment's tenants would share one issuer — nothing is
199        // produced, as with the switch off. A count that cannot be read is off.
200        match self.gate.check().await {
201            Ok(issuer) => !issuer.holds(),
202            Err(error) => {
203                tracing::warn!(target: "axiam::ssf", %tenant_id, %error, "SSF emission could not count the deployment's tenants");
204                false
205            }
206        }
207    }
208
209    /// The streams that would carry `event_type` now: none when no outbox is
210    /// wired, when nothing is registered for it, or when the tenant's switch is
211    /// off. One indexed read in the common case.
212    async fn streams_for(&self, tenant_id: Uuid, event_type: SsfEventType) -> Vec<SsfStream> {
213        if !self.is_wired() {
214            return Vec::new();
215        }
216        let streams = match self.stream_repo.list_for_event(tenant_id, event_type).await {
217            Ok(streams) => streams,
218            Err(error) => {
219                tracing::warn!(target: "axiam::ssf", %tenant_id, %error, "SSF streams could not be listed");
220                return Vec::new();
221            }
222        };
223        if streams.is_empty() || !self.switched_on(tenant_id).await {
224            return Vec::new();
225        }
226        streams
227    }
228
229    /// One event, resolved and submitted per stream.
230    async fn fan_out(
231        &self,
232        streams: &[SsfStream],
233        event: &SsfEvent,
234        subject: &SsfSubject,
235        txn: &str,
236    ) {
237        let Some(outbox) = self.outbox.get() else {
238            return;
239        };
240        let now = Utc::now();
241        for stream in streams {
242            let pending = match prepare_event(
243                &self.auth_config,
244                stream,
245                event,
246                subject,
247                Some(txn),
248                now,
249            ) {
250                Ok(pending) => pending,
251                // Not for this stream: it does not carry the event, it was
252                // disabled meanwhile, or it names users by an address nothing
253                // vouches for (D-46: the event is not sent there).
254                Err(
255                    SsfError::EventNotDelivered
256                    | SsfError::StreamDisabled
257                    | SsfError::EmailNotVouched,
258                ) => continue,
259                Err(error) => {
260                    tracing::warn!(target: "axiam::ssf", stream_id = %stream.id, %error, "an SSF event could not be prepared");
261                    continue;
262                }
263            };
264            match outbox.submit(stream, &pending).await {
265                Ok(()) | Err(SsfOutboxError::Disabled) => {}
266                Err(error) => {
267                    tracing::warn!(target: "axiam::ssf", stream_id = %stream.id, %error, "an SSF event could not be submitted");
268                }
269            }
270        }
271    }
272
273    fn txn() -> String {
274        current_cause().map_or_else(|| Uuid::new_v4().to_string(), |cause| cause.txn)
275    }
276
277    /// Emit `event` about `subject` (already resolved).
278    pub async fn emit(&self, tenant_id: Uuid, event: SsfEvent, subject: SsfSubject) {
279        let streams = self.streams_for(tenant_id, event.event_type()).await;
280        if streams.is_empty() {
281            return;
282        }
283        self.fan_out(&streams, &event, &subject, &Self::txn()).await;
284    }
285
286    /// Emit `event` about the user `user_id`, reading the account only when a
287    /// stream will carry it.
288    pub async fn emit_for_user(&self, tenant_id: Uuid, user_id: Uuid, event: SsfEvent) {
289        let streams = self.streams_for(tenant_id, event.event_type()).await;
290        if streams.is_empty() {
291            return;
292        }
293        let user = match self.user_repo.get_by_id(tenant_id, user_id).await {
294            Ok(user) => user,
295            Err(error) => {
296                tracing::warn!(target: "axiam::ssf", %tenant_id, %user_id, %error, "an SSF event's subject could not be read");
297                return;
298            }
299        };
300        self.fan_out(
301            &streams,
302            &event,
303            &SsfSubject::from_user(&user),
304            &Self::txn(),
305        )
306        .await;
307    }
308
309    /// Whether some stream would carry `event_type` for the tenant right now:
310    /// an outbox is wired, the tenant's `ssf_enabled` is on and a stream is
311    /// registered for it. For a producer that has to *prepare* an event (D-53 (1)
312    /// remembers a step-up before it knows it will be told of one) and would
313    /// otherwise do that work for nobody.
314    pub async fn carries(&self, tenant_id: Uuid, event_type: SsfEventType) -> bool {
315        !self.streams_for(tenant_id, event_type).await.is_empty()
316    }
317
318    /// CAEP `assurance-level-change` (D-53 (1)): the `acr` class `user_id`'s
319    /// session achieved moved from `previous` to `current` by the user's own
320    /// step-up. The caller has already compared them: equal levels are not a
321    /// change and nothing is emitted for them.
322    pub async fn assurance_level_changed(
323        &self,
324        tenant_id: Uuid,
325        user_id: Uuid,
326        previous: AssuranceLevel,
327        current: AssuranceLevel,
328    ) {
329        if previous == current {
330            return;
331        }
332        self.emit_for_user(
333            tenant_id,
334            user_id,
335            SsfEvent::AssuranceLevelChange {
336                current_level: current,
337                previous_level: Some(previous),
338                initiating_entity: Some(InitiatingEntity::User),
339                event_timestamp: Utc::now().timestamp(),
340            },
341        )
342        .await;
343    }
344
345    /// CAEP `credential-change` (D-52): a credential of `user_id` was created,
346    /// changed or removed.
347    pub async fn credential_changed(
348        &self,
349        tenant_id: Uuid,
350        user_id: Uuid,
351        credential_type: CredentialType,
352        change_type: ChangeType,
353        initiated_by: InitiatingEntity,
354        detail: CredentialDetail,
355    ) {
356        self.emit_for_user(
357            tenant_id,
358            user_id,
359            SsfEvent::CredentialChange {
360                credential_type,
361                change_type,
362                initiating_entity: Some(initiated_by),
363                event_timestamp: Utc::now().timestamp(),
364                x509_issuer: None,
365                x509_serial: None,
366                fido2_aaguid: detail.fido2_aaguid,
367            },
368        )
369        .await;
370    }
371
372    /// RISC `account-disabled`: `user`'s account was set `Inactive`. A lockout
373    /// is not one.
374    pub async fn account_disabled(&self, tenant_id: Uuid, user: &User) {
375        self.emit(
376            tenant_id,
377            SsfEvent::AccountDisabled { reason: None },
378            SsfSubject::from_user(user),
379        )
380        .await;
381    }
382
383    /// RISC `account-enabled`: `user`'s account went from `Inactive` to `Active`.
384    pub async fn account_enabled(&self, tenant_id: Uuid, user: &User) {
385        self.emit(
386            tenant_id,
387            SsfEvent::AccountEnabled,
388            SsfSubject::from_user(user),
389        )
390        .await;
391    }
392
393    /// RISC `account-purged`. The subject is the account **as it was before**
394    /// the write that destroyed it (its address is gone afterwards), so the
395    /// caller captures it first.
396    pub async fn account_purged(&self, tenant_id: Uuid, subject: SsfSubject) {
397        self.emit(tenant_id, SsfEvent::AccountPurged, subject).await;
398    }

401impl<C: Connection + Clone> SessionRevocationSink for SsfEmitter<C> {
402    fn is_active(&self) -> bool {
403        self.is_wired()
404    }
405
406    fn sessions_revoked<'a>(
407        &'a self,
408        tenant_id: Uuid,
409        user_id: Uuid,
410        session_ids: &'a [Uuid],
411    ) -> SsfFuture<'a, ()> {
412        Box::pin(async move {
413            let streams = self
414                .streams_for(tenant_id, SsfEventType::SessionRevoked)
415                .await;
416            if streams.is_empty() || session_ids.is_empty() {
417                return;
418            }
419            let user = match self.user_repo.get_by_id(tenant_id, user_id).await {
420                Ok(user) => user,
421                Err(error) => {
422                    tracing::warn!(target: "axiam::ssf", %tenant_id, %user_id, %error, "a revoked session's user could not be read");
423                    return;
424                }
425            };
426            let subject = SsfSubject::from_user(&user);
427            let initiating_entity = current_cause().and_then(|cause| cause.initiating_entity);
428            let txn = Self::txn();
429            for session_id in session_ids {
430                let event = SsfEvent::SessionRevoked {
431                    initiating_entity,
432                    event_timestamp: Utc::now().timestamp(),
433                };
434                self.fan_out(
435                    &streams,
436                    &event,
437                    &subject.clone().with_session(*session_id),
438                    &txn,
439                )
440                .await;
441            }
442        })
443    }
444}

468/// Writes `ssf.inactive_shared_issuer` once per tenant whose `ssf_enabled` is
469/// on, when the D-55 gate starts to hold (the deployment's tenants would share
470/// one issuer, so SSF is inactive for every one of them). Bound to the gate,
471/// which calls it once per change per process — never per event.
472pub struct SharedIssuerAudit<C: Connection + Clone> {
473    org_repo: axiam_db::SurrealOrganizationRepository<C>,
474    tenant_repo: SurrealTenantRepository<C>,
475    settings_repo: SurrealSettingsRepository<C>,
476    audit_repo: axiam_db::SurrealAuditLogRepository<C>,
477}
478
479impl<C: Connection + Clone> SharedIssuerAudit<C> {
480    /// The audit writer over these repositories.
481    pub fn new(
482        org_repo: axiam_db::SurrealOrganizationRepository<C>,
483        tenant_repo: SurrealTenantRepository<C>,
484        settings_repo: SurrealSettingsRepository<C>,
485        audit_repo: axiam_db::SurrealAuditLogRepository<C>,
486    ) -> Self {
487        Self {
488            org_repo,
489            tenant_repo,
490            settings_repo,
491            audit_repo,
492        }
493    }
494
495    /// Every tenant of every organization, page by page.
496    async fn every_tenant(&self) -> Result<Vec<axiam_core::models::tenant::Tenant>, AxiamError> {
497        use axiam_core::repository::{OrganizationRepository, Pagination};
498        const PAGE: u64 = 100;
499        let mut tenants = Vec::new();
500        let mut offset = 0;
501        loop {
502            let orgs = self
503                .org_repo
504                .list(Pagination {
505                    offset,
506                    limit: PAGE,
507                    search: None,
508                })
509                .await?;
510            let got = orgs.items.len() as u64;
511            for org in orgs.items {
512                let mut tenant_offset = 0;
513                loop {
514                    let page = self
515                        .tenant_repo
516                        .list_by_organization(
517                            org.id,
518                            Pagination {
519                                offset: tenant_offset,
520                                limit: PAGE,
521                                search: None,
522                            },
523                        )
524                        .await?;
525                    let n = page.items.len() as u64;
526                    tenants.extend(page.items);
527                    tenant_offset += n;
528                    if n < PAGE {
529                        break;
530                    }
531                }
532            }
533            offset += got;
534            if got < PAGE {
535                return Ok(tenants);
536            }
537        }
538    }
539}

541impl<C: Connection + Clone> axiam_core::models::ssf::SharedIssuerObserver for SharedIssuerAudit<C> {
542    fn gate_changed(&self, holds: bool) -> SsfFuture<'_, ()> {
543        Box::pin(async move {
544            use axiam_core::models::audit::{ActorType, AuditOutcome, CreateAuditLogEntry};
545            use axiam_core::models::ssf::{
546                AUDIT_SSF_INACTIVE_SHARED_ISSUER, SHARED_ISSUER_INACTIVE_REASON,
547            };
548            use axiam_core::repository::AuditLogRepository;
549            if !holds {
550                return;
551            }
552            let tenants = match self.every_tenant().await {
553                Ok(tenants) => tenants,
554                Err(error) => {
555                    tracing::warn!(target: "axiam::ssf", %error, "the D-55 audit could not list the tenants");
556                    return;
557                }
558            };
559            for tenant in tenants {
560                let on = self
561                    .settings_repo
562                    .get_effective_settings(tenant.organization_id, tenant.id)
563                    .await
564                    .is_ok_and(|s| s.oidc.ssf_enabled);
565                if !on {
566                    continue;
567                }
568                if let Err(error) = self
569                    .audit_repo
570                    .append(CreateAuditLogEntry {
571                        tenant_id: tenant.id,
572                        actor_id: Uuid::nil(),
573                        actor_type: ActorType::System,
574                        action: AUDIT_SSF_INACTIVE_SHARED_ISSUER.into(),
575                        resource_id: None,
576                        outcome: AuditOutcome::Failure,
577                        ip_address: None,
578                        metadata: Some(serde_json::json!({
579                            "reason": SHARED_ISSUER_INACTIVE_REASON,
580                        })),
581                    })
582                    .await
583                {
584                    tracing::warn!(target: "axiam::ssf", tenant_id = %tenant.id, %error, "the D-55 audit row could not be written");
585                }
586            }
587        })
588    }
