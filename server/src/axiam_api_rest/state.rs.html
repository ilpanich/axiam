1//! `AppState<C>` — the single composition root for every REST handler
2//! dependency (QUAL-01).
3//!
4//! Before this module, `axiam-server/src/main.rs` registered ~49 individual
5//! `.app_data(web::Data::new(...))` calls and every handler extracted its own
6//! subset of `web::Data<Repo<C>>` / `web::Data<Service<...>>` params. Now
7//! `main.rs` builds ONE `AppState<C>` and registers it once; handlers extract
8//! `web::Data<AppState<C>>` and access dependencies as fields
9//! (`state.user_repo`, `state.auth_service`, ...).
10//!
11//! ## Three dependencies stay registered outside `AppState<C>` (documented deviation)
12//!
13//! `AuthConfig`, `Arc<dyn SessionValidator>`, and `Arc<dyn AuthzChecker>` are
14//! deliberately KEPT as their own standalone `web::Data<T>` registrations in
15//! `main.rs`, in addition to being available as `AppState<C>` fields
16//! (`state.auth_config`, `state.session_validator`) for handler-level
17//! consistency where it is convenient. This is not an oversight:
18//!
19//! - `impl actix_web::FromRequest for AuthenticatedUser`
20//!   (`extractors/auth.rs`) is a fixed, non-generic impl. It cannot know the
21//!   concrete `C` used by `AppState<C>` at the call site, so it cannot do
22//!   `req.app_data::<web::Data<AppState<C>>>()` — there is no `C` in scope.
23//!   It must keep looking up the connection-agnostic `web::Data<AuthConfig>`
24//!   and `web::Data<Arc<dyn SessionValidator>>` types directly.
25//! - `axiam-audit::middleware::AuditMiddleware` (a different crate, wrapping
26//!   the *entire* `App` in `main.rs`, with no dependency on
27//!   `axiam-api-rest::AppState`) independently does
28//!   `req.app_data::<web::Data<AuthConfig>>()`. It has no way to resolve a
29//!   generic `AppState<C>` either.
30//! - 118 handler call sites across the crate use the `AuthzData` type alias
31//!   (`web::Data<Arc<dyn AuthzChecker>>`) directly via
32//!   `authz.get_ref().as_ref()` — none of that call-site shape changes in
33//!   this plan (only the 283 `web::Data<T>` sites named in 29-03-PLAN.md's
34//!   scope note are migrated), so `Arc<dyn AuthzChecker>` must remain
35//!   registered under its own type too.
36//!
37//! `SurrealPasswordResetTokenRepository<C>` and
38//! `SurrealEmailVerificationTokenRepository<C>` were NEVER registered in
39//! `main.rs` at all (a pre-existing bug — see 29-03-SUMMARY.md) despite being
40//! required by `password_reset.rs`/`email_verification.rs`; since both
41//! services are now hoisted singletons, these two repos are constructed
42//! once in `main.rs` purely to build `password_reset_service` /
43//! `email_verification_service` and do not need their own `AppState` field.
44

94// ---------------------------------------------------------------------------
95// DynMailPublisher — object-safe seam over axiam_core::repository::MailPublisher
96// ---------------------------------------------------------------------------
97//
98// `MailPublisher::publish` returns `impl Future` (RPITIT), which is not
99// dyn-safe on its own. `AppState<C>` needs ONE concrete, non-generic field
100// type that works for both the real AMQP-backed `MailOutboundPublisher`
101// (production — requires a live `lapin::Channel`) and test harnesses (no
102// broker available). This mirrors the exact boxed-future seam
103// `extractors::auth::SessionValidator` already uses for the identical
104// reason. Mechanical call sites are unaffected: `state.mail_outbound_publisher
105// .publish(msg).await` reads identically to the old `mail_publisher.publish(msg).await`.
106
107/// Object-safe wrapper trait over [`axiam_core::repository::MailPublisher`].
108pub trait DynMailPublisher: Send + Sync {
109    fn publish<'a>(
110        &'a self,
111        msg: axiam_core::models::mail::OutboundMailMessage,
112    ) -> Pin<Box<dyn Future<Output = axiam_core::error::AxiamResult<()>> + Send + 'a>>;
113}
114
115impl<T> DynMailPublisher for T
116where
117    T: axiam_core::repository::MailPublisher + Send + Sync,
118{
119    fn publish<'a>(
120        &'a self,
121        msg: axiam_core::models::mail::OutboundMailMessage,
122    ) -> Pin<Box<dyn Future<Output = axiam_core::error::AxiamResult<()>> + Send + 'a>> {
123        Box::pin(axiam_core::repository::MailPublisher::publish(self, msg))
124    }
125}

127/// Always-succeeds no-op mail publisher for test harnesses that don't
128/// exercise a mail-sending route (the common case — see 29-03-SUMMARY.md).
129/// Mirrors the `AllowAllAuthzChecker` test-fixture precedent already
130/// established in this crate (`crate::authz::AllowAllAuthzChecker`).
131pub struct NoopMailPublisher;
132
133impl DynMailPublisher for NoopMailPublisher {
134    fn publish<'a>(
135        &'a self,
136        _msg: axiam_core::models::mail::OutboundMailMessage,
137    ) -> Pin<Box<dyn Future<Output = axiam_core::error::AxiamResult<()>> + Send + 'a>> {
138        Box::pin(async { Ok(()) })
139    }
140}

459impl<C: Connection + Clone> AppState<C> {
460    /// Switch on X4's external token-exchange path.
461    ///
462    /// A method on the assembled state rather than a constructor parameter,
463    /// because the collaborator it needs — the OIDC federation service — is
464    /// itself conditional on `AXIAM__AUTH__FEDERATION_ENCRYPTION_KEY` and is
465    /// installed after this state is built.
466    ///
467    /// **No federation service ⇒ no external path**, and that is the correct
468    /// coupling rather than an accident of ordering: without the encryption
469    /// key there is no way to read a provider's configuration, so there is
470    /// nothing to trust an external issuer *with*. Returns whether it was
471    /// enabled, so a caller can log the posture instead of guessing at it.
472    #[must_use = "the return value says whether the external path is actually live"]
473    pub fn enable_external_token_exchange(&mut self) -> bool {
474        let Some(federation) = self.oidc_federation_service.clone() else {
475            return false;
476        };
477        let (resolver, authority) = crate::token_exchange::external_collaborators(
478            federation,
479            self.role_repo.clone(),
480            self.permission_repo.clone(),
481        );
482        // `with_external_subjects` consumes and returns, so the field is
483        // rebuilt rather than mutated in place — the service is a handful of
484        // cheap handles, so this costs nothing and keeps the "both or
485        // neither" pairing enforced by the type.
486        self.token_exchange_service = self
487            .token_exchange_service
488            .clone()
489            .with_external_subjects(resolver, authority);
490        true
491    }

495    /// Built per call rather than stored, because it needs the
496    /// [`crate::authz::AuthzChecker`] — which arrives as its own `web::Data`

511            crate::uma::EngineEvaluator::new(authz),

514        )
515    }

517    /// X1 — tell the reactor gate that this tenant's registrations changed.
518    ///
519    /// A no-op when no gate is composed (AMQP off, or a test harness), which is
520    /// the same shape [`Self::emit_webhook`] uses and for the same reason: a
521    /// side effect of an administrative write must not fail the write, and a
522    /// handler must not have to know whether the subsystem is configured.
523    ///
524    /// This is an **optimisation on top of the TTL, not a replacement for it**.
525    /// It only reaches the replica that served the mutation; every other
526    /// replica picks the change up when its own routing entry expires, which is
527    /// the bound the acceptance test asserts.
528    pub fn invalidate_reactor_routing(&self, tenant_id: uuid::Uuid) {
529        if let Some(invalidate) = &self.reactor_routing_invalidator {
530            invalidate(tenant_id);
531        }
532    }

534    /// Dispatch a domain event to any webhooks subscribed to `event_type` in
535    /// `tenant_id` (CQ-B22). Best-effort: if no AMQP publisher is wired
536    /// (`webhook_publisher` is `None`, e.g. tests or AMQP disabled) this is a
537    /// no-op, and publish failures inside `emit` are logged, never propagated —
538    /// a webhook side effect must not fail the originating API request.
539    pub async fn emit_webhook(
540        &self,
541        tenant_id: uuid::Uuid,
542        event_type: &str,
543        payload: serde_json::Value,
544    ) {
545        if let Some(publisher) = &self.webhook_publisher {
546            self.webhook_delivery
547                .emit(publisher, tenant_id, event_type.to_string(), payload)
548                .await;
549        }
550    }
