1//! `AppState`'s cohesive sub-states (F3).
2//!
3//! # Why these exist
4//!
5//! `AppState` carried **75 public fields**, nearly all concrete
6//! `Surreal*Repository<C>` values, and every handler received all of them
7//! regardless of what it used. That is a god object by the usual definition,
8//! and it had the usual consequences: `state.rs` referenced `axiam_db` in
9//! nineteen places, `C: Connection + Clone` propagated through every handler
10//! signature in the crate, and twenty-five `#[allow(clippy::too_many_arguments)]`
11//! sites sat downstream of the same pressure.
12//!
13//! Forty-six of those fields are used by **one** handler module each. Grouping
14//! them by what they are *for* takes the root from 75 members to 36 and gives
15//! each group a name, a docstring and a boundary.
16//!
17//! # Why not `Arc<dyn Repository>`
18//!
19//! The obvious "fix" for a struct full of concrete types is to box them behind
20//! trait objects. That would put vtable dispatch on the authorization hot path,
21//! which is the one place this system cannot afford it -- `check_access` is
22//! what a service mesh calls on every request.
23//!
24//! This is a **field-grouping change, not a dispatch change**. Every type is
25//! what it was, monomorphisation is what it was, and the generated code is what
26//! it was. What changes is who can see what.
27//!
28//! # `use super::*`
29//!
30//! Deliberate. These structs and [`super::AppState`] are two halves of one
31//! dependency-injection container and name the same forty-odd types; a
32//! duplicated import list would be a second place to update every time a
33//! repository is added, for no reader's benefit.

37/// Certificate authority, X.509 issuance, PGP and device certificate auth.
38///
39/// Everything behind `/api/v1/ca-certificates`, `/api/v1/certificates`,
40/// `/api/v1/pgp-keys`, the revocation lists under `/pki/v1` and the mTLS
41/// device-auth path. Grouped because they share
42/// one subject -- key material this deployment issues or verifies -- and because

53    /// The CA rows themselves, for the handful of operations that are about a

59    pub device_auth_service: DeviceAuthServiceT<C>,

62    /// `None` when nothing registered one: a plaintext deployment, or a test

64    /// restart is needed, which is exactly what it used to do.
65    ///

391/// The streams that have a long poll waiting **on this instance**.
392///
393/// A long poll holds a request open for up to thirty seconds; a receiver that
394/// opens many on one stream (a retry loop with no backoff, a bug) would hold as
395/// many connections for no benefit, since RFC 8936 has one receiver draining one
396/// stream in order. At most one waits per stream; a second answers at once.
397/// Per instance on purpose: an exact cross-instance count would need shared
398/// state for a bound that only has to be small.
399#[derive(Debug, Default)]
400pub struct PollWaiters {
401    waiting: std::sync::Mutex<std::collections::HashSet<uuid::Uuid>>,
402}
403
404/// Holds a stream's wait slot; releases it when dropped — including when the
405/// request is cancelled because the receiver hung up.
406#[derive(Debug)]
407pub struct PollWaitGuard {
408    owner: Arc<PollWaiters>,
409    stream_id: uuid::Uuid,
410}
411
412impl PollWaiters {
413    /// Take `stream_id`'s wait slot, or `None` when a long poll already holds it.
414    #[must_use]
415    pub fn try_enter(self: &Arc<Self>, stream_id: uuid::Uuid) -> Option<PollWaitGuard> {
416        let mut waiting = self.waiting.lock().unwrap_or_else(|e| e.into_inner());
417        waiting.insert(stream_id).then(|| PollWaitGuard {
418            owner: Arc::clone(self),
419            stream_id,
420        })
421    }
422
423    /// How many streams have a long poll waiting.
424    #[must_use]
425    pub fn waiting(&self) -> usize {
426        self.waiting.lock().unwrap_or_else(|e| e.into_inner()).len()
427    }
428}
429
430impl Drop for PollWaitGuard {
431    fn drop(&mut self) {
432        self.owner
433            .waiting
434            .lock()
435            .unwrap_or_else(|e| e.into_inner())
436            .remove(&self.stream_id);
437    }
438}

450#[cfg(test)]
451mod poll_waiter_tests {
452    use super::*;
453
454    #[test]
455    fn one_slot_per_stream_released_on_drop() {
456        let waiters = Arc::new(PollWaiters::default());
457        let (a, b) = (uuid::Uuid::new_v4(), uuid::Uuid::new_v4());
458        let first = waiters.try_enter(a).expect("the first takes the slot");
459        assert!(waiters.try_enter(a).is_none(), "the second finds it taken");
460        assert!(waiters.try_enter(b).is_some(), "another stream is free");
461        assert_eq!(waiters.waiting(), 1, "b's guard was a temporary");
462        drop(first);
463        assert!(waiters.try_enter(a).is_some(), "released on drop");
464        assert_eq!(waiters.waiting(), 0);
465    }
466}

468/// What starting an on-demand SCIM reconciliation came to (G-6, T23.6.4).
469#[derive(Debug, Clone, Copy, PartialEq, Eq)]
470pub enum ScimReconcileStart {
471    /// The caller took the claim and the run is under way (`202`).
472    Started,
473    /// A run holds the claim, or ran within the on-demand window (`409`).
474    AlreadyClaimed,
475    /// The target is disabled: it receives nothing, so there is nothing to
476    /// reconcile and no claim was taken (`409`).
477    TargetDisabled,
478}

480/// The management API's way to start a reconciliation. A port, because the
481/// deliverer that makes the run lives in `axiam-scim`, which sits **above**
482/// this crate (layer 7) and so cannot be named here; `axiam-scim` implements
483/// it (`ReconcileLauncher`) and the composition root binds it.
484pub trait ScimReconcileTrigger: Send + Sync {
485    /// Take the target's reconciliation claim and, when it is taken, make the
486    /// run in the background. Answers as soon as the claim is decided.
487    ///
488    /// # Errors
489    ///
490    /// `NotFound` when the target does not exist in the tenant; any other
491    /// failure of the datastore while claiming.
492    fn start<'a>(
493        &'a self,
494        tenant_id: uuid::Uuid,
495        target_id: uuid::Uuid,
496    ) -> std::pin::Pin<
497        Box<
498            dyn std::future::Future<
499                    Output = Result<ScimReconcileStart, axiam_core::error::AxiamError>,
500                > + Send
501                + 'a,
502        >,
503    >;
504}
