1//! `AppState`'s cohesive sub-states (F3).
2//!
3//! # Why these exist
4//!
5//! `AppState` carried **75 public fields**, nearly all concrete
6//! `Surreal*Repository<C>` values, and every handler received all of them
7//! regardless of what it used. That is a god object by the usual definition,
8//! and it had the usual consequences: `state.rs` referenced `axiam_db` in
9//! nineteen places, `C: Connection + Clone` propagated through every handler
10//! signature in the crate, and twenty-five `#[allow(clippy::too_many_arguments)]`
11//! sites sat downstream of the same pressure.
12//!
13//! Forty-six of those fields are used by **one** handler module each. Grouping
14//! them by what they are *for* takes the root from 75 members to 36 and gives
15//! each group a name, a docstring and a boundary.
16//!
17//! # Why not `Arc<dyn Repository>`
18//!
19//! The obvious "fix" for a struct full of concrete types is to box them behind
20//! trait objects. That would put vtable dispatch on the authorization hot path,
21//! which is the one place this system cannot afford it -- `check_access` is
22//! what a service mesh calls on every request.
23//!
24//! This is a **field-grouping change, not a dispatch change**. Every type is
25//! what it was, monomorphisation is what it was, and the generated code is what
26//! it was. What changes is who can see what.
27//!
28//! # `use super::*`
29//!
30//! Deliberate. These structs and [`super::AppState`] are two halves of one
31//! dependency-injection container and name the same forty-odd types; a
32//! duplicated import list would be a second place to update every time a
33//! repository is added, for no reader's benefit.
34

37/// Certificate authority, X.509 issuance, PGP and device certificate auth.
38///
39/// Everything behind `/api/v1/ca-certificates`, `/api/v1/certificates`,
40/// `/api/v1/pgp-keys` and the mTLS device-auth path. Grouped because they share
41/// one subject -- key material this deployment issues or verifies -- and because
42/// nothing outside those four handlers has any business reaching a signing
43/// service.
44#[derive(Clone)]
45pub struct PkiState<C: Connection + Clone> {
46    pub ca_service: CaServiceT<C>,
47    pub cert_service: CertServiceT<C>,
48    pub cert_repo: SurrealCertificateRepository<C>,
49    /// The CA rows themselves, for the handful of operations that are about a
50    /// CA record rather than about signing with it — today, toggling
51    /// `mtls_trust_anchor`. `CaService` owns issuance; a flag that changes what
52    /// the TLS listener trusts is not issuance and does not belong behind it.
53    pub ca_cert_repo: axiam_db::SurrealCaCertificateRepository<C>,
54    pub pgp_service: PgpServiceT<C>,
55    pub device_auth_service: DeviceAuthServiceT<C>,
56}

58/// WebAuthn ceremonies, attestation policy and FIDO MDS metadata.
59///
60/// The registration/authentication ceremonies, the tenant attestation policy
61/// they are evaluated against, and the FIDO MDS3 metadata that policy reads.
62/// `attestation_ca_cache` and `attestation_metadata_source` sit here rather than
63/// in [`PkiState`] because they exist to answer "is this authenticator's
64/// attestation acceptable", which is a WebAuthn question, not a PKI one.
65#[derive(Clone)]
66pub struct WebauthnState<C: Connection + Clone> {
67    pub webauthn_service: WebauthnServiceT<C>,
68    /// X3 wave 3: direct access to WebAuthn credentials for the tenant-wide
69    /// compliance report (D9) — `webauthn_service` only exposes per-user
70    /// ceremony operations, not a tenant-wide listing.
71    pub webauthn_credential_repo: SurrealWebauthnCredentialRepository<C>,
72    /// X3 (D5): tenant WebAuthn attestation policy. Resolved by REST
73    /// handlers on every registration ceremony start/finish (an absent row
74    /// means `WebauthnAttestationPolicy::default()`, i.e. today's `mode:
75    /// none` behavior) and by the policy admin/compliance-report endpoints.
76    pub webauthn_attestation_policy_repo: SurrealWebauthnAttestationPolicyRepository<C>,
77    /// X3 (D10): server-global FIDO MDS3 metadata storage, read by the
78    /// `GET /api/v1/mds/status` / `POST /api/v1/mds/refresh` admin endpoints.
79    pub mds_repo: SurrealMdsRepository<C>,
80    /// X3 (W2-D4): the small, data-only view over `mds_repo` that
81    /// `axiam-auth`'s attestation enforcement and compliance evaluation
82    /// consume, keeping `axiam-auth` free of a hard `axiam-db` dependency.
83    pub attestation_metadata_source: MdsAttestationMetadataSource<SurrealMdsRepository<C>>,
84    /// X3 (W2-D3): process-wide cache of built `AttestationCaList`s.
85    /// `Arc`-shared (mirrors `tenant_org_cache`) so every worker/request
86    /// sees the same cache, and so it can be invalidated from the MDS
87    /// refresh and attestation-policy-update endpoints (and the background
88    /// MDS refresh job in `axiam-server`) without threading a second
89    /// `web::Data` registration through every call site.
90    pub attestation_ca_cache: Arc<AttestationCaCache>,
91    /// X3 (D10): FIDO MDS3 ingestion configuration (`mds_enabled`,
92    /// `mds_blob_url`/`mds_blob_path`, `mds_leaf_dns`). `POST
93    /// /api/v1/mds/refresh` reads this to decide whether ingestion is
94    /// enabled at all and which source to fetch from — `encryption_key`
95    /// here is unused by the REST layer (PKI-service construction in
96    /// `axiam-server` keeps its own copy for that).
97    pub pki_config: PkiConfig,
98}

137/// Reactors, webhooks and notification rules — everything AXIAM emits outward.
138///
139/// The three outbound-notification mechanisms share a lifecycle: a domain
140/// event happens, a routing table decides who hears about it, and delivery is
141/// best-effort and must never fail the request that produced it. Keeping them
142/// together keeps that shared property in one place.
143#[derive(Clone)]
144pub struct EventsState<C: Connection + Clone> {
145    pub reactor_repo: SurrealReactorRepository<C>,
146    /// X1 — the interceptor chain the `user.pre_create`, `user.pre_update` and
147    /// `grant.pre_assign` handlers call.
148    ///
149    /// `axiam-auth` and `axiam-oauth2` hold their own clone of the same gate
150    /// (they own the `login.post_auth` and `token.pre_issue` sites), so all
151    /// five hooks share one routing table, one per-tenant concurrency bound and
152    /// one audit sink. A deployment without AMQP holds
153    /// [`axiam_core::models::reactor::NoopReactorGate`] here — never `None`,
154    /// because a call site that branches on the feature is a call site where
155    /// the two builds can drift.
156    pub reactor_gate: axiam_core::models::reactor::SharedReactorGate,
157    /// X1 — invalidates the gate's routing table when a reactor registration
158    /// changes, so the change is live on this replica immediately rather than
159    /// at the end of the TTL.
160    ///
161    /// A closure rather than the routing table itself because the table is
162    /// generic over its source, and `AppState<C>` is already generic over one
163    /// parameter too many. `None` when no gate is composed.
164    pub reactor_routing_invalidator: Option<Arc<dyn Fn(uuid::Uuid) + Send + Sync>>,
165    pub webhook_repo: SurrealWebhookRepository<C>,
166    pub webhook_delivery: WebhookDeliveryServiceT<C>,
167    /// AMQP publisher used by [`AppState::emit_webhook`] to dispatch domain
168    /// events onto the durable webhook queue (CQ-B22). `None` in tests and when
169    /// AMQP is unavailable — `emit_webhook` becomes a no-op rather than failing
170    /// the originating request (webhook delivery is a best-effort side effect).
171    pub webhook_publisher: Option<Arc<axiam_amqp::WebhookPublisher>>,
172    pub notification_rule_repo: SurrealNotificationRuleRepository<C>,
173}

175/// The OAuth2 authorization server and OIDC provider.
176///
177/// The widest bundle, and the one that most justifies the split: twelve
178/// fields that only `handlers/oauth2.rs`, `uma.rs`, `token_exchange.rs` and
179/// `backchannel_logout.rs` touch, previously visible to all twenty-nine handler
180/// modules.
181#[derive(Clone)]
182pub struct OAuth2State<C: Connection + Clone> {
183    pub authorize_service: AuthorizeServiceT<C>,
184    pub token_service: TokenServiceT<C>,
185    /// B2 — device authorization grant (RFC 8628).
186    pub device_authorization_service: DeviceAuthorizationServiceT<C>,
187    /// B3 — token exchange (RFC 8693).
188    pub token_exchange_service: TokenExchangeServiceT<C>,
189    /// B5 — pushed authorization requests (RFC 9126).
190    pub par_service: ParServiceT<C>,
191    /// X2 — UMA 2.0 permission tickets.
192    ///
193    /// The repository rather than an assembled `UmaService`, because the
194    /// service also needs the [`crate::authz::AuthzChecker`], which is a
195    /// separate `web::Data` and not part of this state. Handlers assemble the
196    /// service with [`AppState::uma_service`].
197    pub permission_ticket_repo: SurrealPermissionTicketRepository<C>,
198    /// X2 — deployment ceiling on RPT lifetime, in seconds. The effective
199    /// lifetime is the minimum of this, the protocol default (300 s), and the
200    /// subject token's own remaining life.
201    pub rpt_max_lifetime_secs: i64,
202    /// B5 — which clients joined which session, for the back-channel logout
203    /// fan-out.
204    pub session_client_repo: SurrealSessionClientRepository<C>,
205    pub device_grant_repo: SurrealDeviceGrantRepository<C>,
206    /// X5.1 — single-use `jti` store for RFC 7523 client assertions and
207    /// RFC 9449 DPoP proofs. Decides replay by a `UNIQUE` index violation
208    /// rather than by reading first; see the repository's module docs.
209    pub proof_replay_repo: SurrealProofReplayRepository<C>,
210    /// B3: in-process cache + ETag for AXIAM's OWN `GET /oauth2/jwks`
211    /// response (the signing keys AXIAM serves to relying parties).
212    /// Constructed once at startup and shared via this `Arc` so all workers
213    /// hit the same cache instead of each maintaining its own.
214    pub oauth2_jwks_cache: Arc<Oauth2JwksCache>,
215    /// B3: `Cache-Control` max-age (and header rendering) for the
216    /// `GET /oauth2/jwks` response. Configured via
217    /// `AXIAM__OAUTH2__JWKS_CACHE_MAX_AGE_SECS` (default 300s).
218    pub oauth2_jwks_cache_config: Oauth2JwksCacheConfig,
219}

221/// Inbound SAML and OIDC federation.
222///
223/// Where AXIAM is the *relying party* rather than the provider --
224/// `OAuth2State` is the other direction. `assertion_replay_repo` belongs here and
225/// not with the other replay guards because what it guards is a SAML assertion,
226/// and the handler that consumes one is the only caller.
227#[derive(Clone)]
228pub struct FederationState<C: Connection + Clone> {
229    pub federation_config_repo: SurrealFederationConfigRepository<C>,
230    pub federation_link_repo: SurrealFederationLinkRepository<C>,
231    pub federation_login_state_repo: SurrealFederationLoginStateRepository<C>,
232    pub assertion_replay_repo: SurrealAssertionReplayRepository<C>,
233    /// `None` when `AXIAM__AUTH__FEDERATION_ENCRYPTION_KEY` is unset — the
234    /// OIDC federation encryption key is baked into `OidcFederationService`
235    /// at construction, so absence is resolved once at startup rather than
236    /// per-request (identical fail-closed error at the 4 call sites).
237    pub oidc_federation_service: Option<OidcFederationServiceT<C>>,
238    /// Constructed unconditionally (SamlFederationService::new needs no
239    /// encryption key) — but the FIELD itself stays `#[cfg(feature = "saml")]`
240    /// gated. Why: `axiam_federation::saml` only exists when
241    /// axiam-federation's OWN `saml` Cargo feature is on, and
242    /// axiam-api-rest's `saml` feature intentionally forwards to it so
243    /// `cargo build -p axiam-server --no-default-features` can still build
244    /// without the `samael`/`libxml2` dependency chain (documented escape
245    /// hatch for hosts with an incompatible system libxml2 — see
246    /// `axiam-server/src/main.rs`'s `--dump-openapi` doc comment and
247    /// `axiam-api-rest/Cargo.toml`). Un-gating this field would force
248    /// `axiam-federation/saml` on unconditionally and break that hatch.
249    #[cfg(feature = "saml")]
250    pub saml_federation_service: SamlFederationServiceT<C>,
251}
