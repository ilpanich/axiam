1//! `AppState`'s cohesive sub-states (F3).
2//!
3//! # Why these exist
4//!
5//! `AppState` carried **75 public fields**, nearly all concrete
6//! `Surreal*Repository<C>` values, and every handler received all of them
7//! regardless of what it used. That is a god object by the usual definition,
8//! and it had the usual consequences: `state.rs` referenced `axiam_db` in
9//! nineteen places, `C: Connection + Clone` propagated through every handler
10//! signature in the crate, and twenty-five `#[allow(clippy::too_many_arguments)]`
11//! sites sat downstream of the same pressure.
12//!
13//! Forty-six of those fields are used by **one** handler module each. Grouping
14//! them by what they are *for* takes the root from 75 members to 36 and gives
15//! each group a name, a docstring and a boundary.
16//!
17//! # Why not `Arc<dyn Repository>`
18//!
19//! The obvious "fix" for a struct full of concrete types is to box them behind
20//! trait objects. That would put vtable dispatch on the authorization hot path,
21//! which is the one place this system cannot afford it -- `check_access` is
22//! what a service mesh calls on every request.
23//!
24//! This is a **field-grouping change, not a dispatch change**. Every type is
25//! what it was, monomorphisation is what it was, and the generated code is what
26//! it was. What changes is who can see what.
27//!
28//! # `use super::*`
29//!
30//! Deliberate. These structs and [`super::AppState`] are two halves of one
31//! dependency-injection container and name the same forty-odd types; a
32//! duplicated import list would be a second place to update every time a
33//! repository is added, for no reader's benefit.

37/// Certificate authority, X.509 issuance, PGP and device certificate auth.
38///
39/// Everything behind `/api/v1/ca-certificates`, `/api/v1/certificates`,
40/// `/api/v1/pgp-keys` and the mTLS device-auth path. Grouped because they share
41/// one subject -- key material this deployment issues or verifies -- and because
42/// nothing outside those four handlers has any business reaching a signing
43/// service.
44#[derive(Clone)]
45pub struct PkiState<C: Connection + Clone> {
46    pub ca_service: CaServiceT<C>,
47    pub cert_service: CertServiceT<C>,
48    pub cert_repo: SurrealCertificateRepository<C>,
49    /// The CA rows themselves, for the handful of operations that are about a
50    /// CA record rather than about signing with it — today, toggling
51    /// `mtls_trust_anchor`. `CaService` owns issuance; a flag that changes what
52    /// the TLS listener trusts is not issuance and does not belong behind it.
53    pub ca_cert_repo: axiam_db::SurrealCaCertificateRepository<C>,
54    pub pgp_service: PgpServiceT<C>,
55    pub device_auth_service: DeviceAuthServiceT<C>,
56    /// Applies the current mTLS trust anchor set to the live TLS listener.
57    ///
58    /// `None` when nothing registered one: a plaintext deployment, or a test
59    /// harness. Toggling an anchor then updates the row and reports that a
60    /// restart is needed, which is exactly what it used to do.
61    ///
62    /// The implementation lives in `axiam-server` because it is the crate that
63    /// owns the listener; this is the seam that lets a handler reach it without
64    /// the layering pointing outward.
65    pub trust_anchor_reloader: Option<std::sync::Arc<dyn crate::TrustAnchorReloader>>,
66}

68/// WebAuthn ceremonies, attestation policy and FIDO MDS metadata.
69///
70/// The registration/authentication ceremonies, the tenant attestation policy
71/// they are evaluated against, and the FIDO MDS3 metadata that policy reads.
72/// `attestation_ca_cache` and `attestation_metadata_source` sit here rather than
73/// in [`PkiState`] because they exist to answer "is this authenticator's
74/// attestation acceptable", which is a WebAuthn question, not a PKI one.
75#[derive(Clone)]
76pub struct WebauthnState<C: Connection + Clone> {
77    pub webauthn_service: WebauthnServiceT<C>,
78    /// X3 wave 3: direct access to WebAuthn credentials for the tenant-wide
79    /// compliance report (D9) — `webauthn_service` only exposes per-user
80    /// ceremony operations, not a tenant-wide listing.
81    pub webauthn_credential_repo: SurrealWebauthnCredentialRepository<C>,
82    /// X3 (D5): tenant WebAuthn attestation policy. Resolved by REST
83    /// handlers on every registration ceremony start/finish (an absent row
84    /// means `WebauthnAttestationPolicy::default()`, i.e. today's `mode:
85    /// none` behavior) and by the policy admin/compliance-report endpoints.
86    pub webauthn_attestation_policy_repo: SurrealWebauthnAttestationPolicyRepository<C>,
87    /// X3 (D10): server-global FIDO MDS3 metadata storage, read by the
88    /// `GET /api/v1/mds/status` / `POST /api/v1/mds/refresh` admin endpoints.
89    pub mds_repo: SurrealMdsRepository<C>,
90    /// X3 (W2-D4): the small, data-only view over `mds_repo` that
91    /// `axiam-auth`'s attestation enforcement and compliance evaluation
92    /// consume, keeping `axiam-auth` free of a hard `axiam-db` dependency.
93    pub attestation_metadata_source: MdsAttestationMetadataSource<SurrealMdsRepository<C>>,
94    /// X3 (W2-D3): process-wide cache of built `AttestationCaList`s.
95    /// `Arc`-shared (mirrors `tenant_org_cache`) so every worker/request
96    /// sees the same cache, and so it can be invalidated from the MDS
97    /// refresh and attestation-policy-update endpoints (and the background
98    /// MDS refresh job in `axiam-server`) without threading a second
99    /// `web::Data` registration through every call site.
100    pub attestation_ca_cache: Arc<AttestationCaCache>,
101    /// X3 (D10): FIDO MDS3 ingestion configuration (`mds_enabled`,
102    /// `mds_blob_url`/`mds_blob_path`, `mds_leaf_dns`). `POST
103    /// /api/v1/mds/refresh` reads this to decide whether ingestion is
104    /// enabled at all and which source to fetch from — `encryption_key`
105    /// here is unused by the REST layer (PKI-service construction in
106    /// `axiam-server` keeps its own copy for that).
107    pub pki_config: PkiConfig,
108}

147/// Reactors, webhooks and notification rules — everything AXIAM emits outward.
148///
149/// The three outbound-notification mechanisms share a lifecycle: a domain
150/// event happens, a routing table decides who hears about it, and delivery is
151/// best-effort and must never fail the request that produced it. Keeping them
152/// together keeps that shared property in one place.
153#[derive(Clone)]
154pub struct EventsState<C: Connection + Clone> {
155    pub reactor_repo: SurrealReactorRepository<C>,
156    /// X1 — the interceptor chain the `user.pre_create`, `user.pre_update` and
157    /// `grant.pre_assign` handlers call.
158    ///
159    /// `axiam-auth` and `axiam-oauth2` hold their own clone of the same gate
160    /// (they own the `login.post_auth` and `token.pre_issue` sites), so all
161    /// five hooks share one routing table, one per-tenant concurrency bound and
162    /// one audit sink. A deployment without AMQP holds
163    /// [`axiam_core::models::reactor::NoopReactorGate`] here — never `None`,
164    /// because a call site that branches on the feature is a call site where
165    /// the two builds can drift.
166    pub reactor_gate: axiam_core::models::reactor::SharedReactorGate,
167    /// X1 — invalidates the gate's routing table when a reactor registration
168    /// changes, so the change is live on this replica immediately rather than
169    /// at the end of the TTL.
170    ///
171    /// A closure rather than the routing table itself because the table is
172    /// generic over its source, and `AppState<C>` is already generic over one
173    /// parameter too many. `None` when no gate is composed.
174    pub reactor_routing_invalidator: Option<Arc<dyn Fn(uuid::Uuid) + Send + Sync>>,
175    pub webhook_repo: SurrealWebhookRepository<C>,
176    pub webhook_delivery: WebhookDeliveryServiceT<C>,
177    /// AMQP publisher used by [`AppState::emit_webhook`] to dispatch domain
178    /// events onto the durable webhook queue (CQ-B22). `None` in tests and when
179    /// AMQP is unavailable — `emit_webhook` becomes a no-op rather than failing
180    /// the originating request (webhook delivery is a best-effort side effect).
181    pub webhook_publisher: Option<Arc<axiam_amqp::WebhookPublisher>>,
182    pub notification_rule_repo: SurrealNotificationRuleRepository<C>,
183}

185/// The OAuth2 authorization server and OIDC provider.
186///
187/// The widest bundle, and the one that most justifies the split: twelve
188/// fields that only `handlers/oauth2.rs`, `uma.rs`, `token_exchange.rs` and
189/// `backchannel_logout.rs` touch, previously visible to all twenty-nine handler
190/// modules.
191#[derive(Clone)]
192pub struct OAuth2State<C: Connection + Clone> {
193    pub authorize_service: AuthorizeServiceT<C>,
194    pub token_service: TokenServiceT<C>,
195    /// B2 — device authorization grant (RFC 8628).
196    pub device_authorization_service: DeviceAuthorizationServiceT<C>,
197    /// B3 — token exchange (RFC 8693).
198    pub token_exchange_service: TokenExchangeServiceT<C>,
199    /// B5 — pushed authorization requests (RFC 9126).
200    pub par_service: ParServiceT<C>,
201    /// X2 — UMA 2.0 permission tickets.
202    ///
203    /// The repository rather than an assembled `UmaService`, because the
204    /// service also needs the [`crate::authz::AuthzChecker`], which is a
205    /// separate `web::Data` and not part of this state. Handlers assemble the
206    /// service with [`AppState::uma_service`].
207    pub permission_ticket_repo: SurrealPermissionTicketRepository<C>,
208    /// X2 — deployment ceiling on RPT lifetime, in seconds. The effective
209    /// lifetime is the minimum of this, the protocol default (300 s), and the
210    /// subject token's own remaining life.
211    pub rpt_max_lifetime_secs: i64,
212    /// B5 — which clients joined which session, for the back-channel logout
213    /// fan-out.
214    pub session_client_repo: SurrealSessionClientRepository<C>,
215    pub device_grant_repo: SurrealDeviceGrantRepository<C>,
216    /// X5.1 — single-use `jti` store for RFC 7523 client assertions and
217    /// RFC 9449 DPoP proofs. Decides replay by a `UNIQUE` index violation
218    /// rather than by reading first; see the repository's module docs.
219    pub proof_replay_repo: SurrealProofReplayRepository<C>,
220    /// B3: in-process cache + ETag for AXIAM's OWN `GET /oauth2/jwks`
221    /// response (the signing keys AXIAM serves to relying parties).
222    /// Constructed once at startup and shared via this `Arc` so all workers
223    /// hit the same cache instead of each maintaining its own.
224    pub oauth2_jwks_cache: Arc<Oauth2JwksCache>,
225    /// B3: `Cache-Control` max-age (and header rendering) for the
226    /// `GET /oauth2/jwks` response. Configured via
227    /// `AXIAM__OAUTH2__JWKS_CACHE_MAX_AGE_SECS` (default 300s).
228    pub oauth2_jwks_cache_config: Oauth2JwksCacheConfig,
229}

231/// Inbound SAML and OIDC federation.
232///
233/// Where AXIAM is the *relying party* rather than the provider --
234/// `OAuth2State` is the other direction. `assertion_replay_repo` belongs here and
235/// not with the other replay guards because what it guards is a SAML assertion,
236/// and the handler that consumes one is the only caller.
237#[derive(Clone)]
238pub struct FederationState<C: Connection + Clone> {
239    pub federation_config_repo: SurrealFederationConfigRepository<C>,
240    pub federation_link_repo: SurrealFederationLinkRepository<C>,
241    pub federation_login_state_repo: SurrealFederationLoginStateRepository<C>,
242    /// Single-use, 60-second codes that turn a cross-site SSO return (SAML,
243    /// Apple's `response_mode=form_post`) into a same-site session issuance
244    /// without weakening `SameSite=Strict` on the session cookies. Here rather
245    /// than with the session repositories because the only thing that mints or
246    /// redeems one is a federation callback.
247    pub sso_handoff_code_repo: SurrealSsoHandoffCodeRepository<C>,
248    pub assertion_replay_repo: SurrealAssertionReplayRepository<C>,
249    /// `None` when `AXIAM__AUTH__FEDERATION_ENCRYPTION_KEY` is unset — the
250    /// OIDC federation encryption key is baked into `OidcFederationService`
251    /// at construction, so absence is resolved once at startup rather than
252    /// per-request (identical fail-closed error at the 4 call sites).
253    pub oidc_federation_service: Option<OidcFederationServiceT<C>>,
254    /// Constructed unconditionally (SamlFederationService::new needs no
255    /// encryption key) — but the FIELD itself stays `#[cfg(feature = "saml")]`
256    /// gated. Why: `axiam_federation::saml` only exists when
257    /// axiam-federation's OWN `saml` Cargo feature is on, and
258    /// axiam-api-rest's `saml` feature intentionally forwards to it so
259    /// `cargo build -p axiam-server --no-default-features` can still build
260    /// without the `samael`/`libxml2` dependency chain (documented escape
261    /// hatch for hosts with an incompatible system libxml2 — see
262    /// `axiam-server/src/main.rs`'s `--dump-openapi` doc comment and
263    /// `axiam-api-rest/Cargo.toml`). Un-gating this field would force
264    /// `axiam-federation/saml` on unconditionally and break that hatch.
265    #[cfg(feature = "saml")]
266    pub saml_federation_service: SamlFederationServiceT<C>,
267}
