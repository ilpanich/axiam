1//! `AppState<C>` — the single composition root for every REST handler
2//! dependency (QUAL-01).
3//!
4//! Before this module, `axiam-server/src/main.rs` registered ~49 individual
5//! `.app_data(web::Data::new(...))` calls and every handler extracted its own
6//! subset of `web::Data<Repo<C>>` / `web::Data<Service<...>>` params. Now
7//! `main.rs` builds ONE `AppState<C>` and registers it once; handlers extract
8//! `web::Data<AppState<C>>` and access dependencies as fields
9//! (`state.user_repo`, `state.auth_service`, ...).
10//!
11//! ## Three dependencies stay registered outside `AppState<C>` (documented deviation)
12//!
13//! `AuthConfig`, `Arc<dyn SessionValidator>`, and `Arc<dyn AuthzChecker>` are
14//! deliberately KEPT as their own standalone `web::Data<T>` registrations in
15//! `main.rs`, in addition to being available as `AppState<C>` fields
16//! (`state.auth_config`, `state.session_validator`) for handler-level
17//! consistency where it is convenient. This is not an oversight:
18//!
19//! - `impl actix_web::FromRequest for AuthenticatedUser`
20//!   (`extractors/auth.rs`) is a fixed, non-generic impl. It cannot know the
21//!   concrete `C` used by `AppState<C>` at the call site, so it cannot do
22//!   `req.app_data::<web::Data<AppState<C>>>()` — there is no `C` in scope.
23//!   It must keep looking up the connection-agnostic `web::Data<AuthConfig>`
24//!   and `web::Data<Arc<dyn SessionValidator>>` types directly.
25//! - `axiam-audit::middleware::AuditMiddleware` (a different crate, wrapping
26//!   the *entire* `App` in `main.rs`, with no dependency on
27//!   `axiam-api-rest::AppState`) independently does
28//!   `req.app_data::<web::Data<AuthConfig>>()`. It has no way to resolve a
29//!   generic `AppState<C>` either.
30//! - 118 handler call sites across the crate use the `AuthzData` type alias
31//!   (`web::Data<Arc<dyn AuthzChecker>>`) directly via
32//!   `authz.get_ref().as_ref()` — none of that call-site shape changes in
33//!   this plan (only the 283 `web::Data<T>` sites named in 29-03-PLAN.md's
34//!   scope note are migrated), so `Arc<dyn AuthzChecker>` must remain
35//!   registered under its own type too.
36//!
37//! `SurrealPasswordResetTokenRepository<C>` and
38//! `SurrealEmailVerificationTokenRepository<C>` were NEVER registered in
39//! `main.rs` at all (a pre-existing bug — see 29-03-SUMMARY.md) despite being
40//! required by `password_reset.rs`/`email_verification.rs`; since both
41//! services are now hoisted singletons, these two repos are constructed
42//! once in `main.rs` purely to build `password_reset_service` /
43//! `email_verification_service` and do not need their own `AppState` field.

100// DynMailPublisher — object-safe seam over axiam_core::repository::MailPublisher
101// ---------------------------------------------------------------------------
102//
103// `MailPublisher::publish` returns `impl Future` (RPITIT), which is not
104// dyn-safe on its own. `AppState<C>` needs ONE concrete, non-generic field
105// type that works for both the real AMQP-backed `MailOutboundPublisher`
106// (production — requires a live `lapin::Channel`) and test harnesses (no
107// broker available). This mirrors the exact boxed-future seam
108// `extractors::auth::SessionValidator` already uses for the identical
109// reason. Mechanical call sites are unaffected: `state.mail_outbound_publisher
110// .publish(msg).await` reads identically to the old `mail_publisher.publish(msg).await`.
111
112/// Object-safe wrapper trait over [`axiam_core::repository::MailPublisher`].
113pub trait DynMailPublisher: Send + Sync {
114    fn publish<'a>(
115        &'a self,
116        msg: axiam_core::models::mail::OutboundMailMessage,
117    ) -> Pin<Box<dyn Future<Output = axiam_core::error::AxiamResult<()>> + Send + 'a>>;
118}
119
120impl<T> DynMailPublisher for T
121where
122    T: axiam_core::repository::MailPublisher + Send + Sync,
123{
124    fn publish<'a>(
125        &'a self,
126        msg: axiam_core::models::mail::OutboundMailMessage,
127    ) -> Pin<Box<dyn Future<Output = axiam_core::error::AxiamResult<()>> + Send + 'a>> {
128        Box::pin(axiam_core::repository::MailPublisher::publish(self, msg))
129    }
130}

266/// Single composition root for every REST handler dependency (QUAL-01).
267///
268/// Generic bound mirrors `server.rs::register_api_v1_routes<C>` exactly, so
269/// the same `AppState<C>` instantiation backs both production
270/// (`AppState<axiam_db::DbClient>`) and every test harness
271/// (`AppState<surrealdb::engine::local::Db>`).
272///
273/// All fields are `pub` — this struct is an internal DI container, not a
274/// public API surface with invariants to protect; tests need to override
275/// individual fields (e.g. swap `authz_checker`... see note above: that one
276/// stays outside `AppState`, but e.g. a test overriding `health_checker`)
277/// after calling [`AppState::for_test`].
278///
279/// # Shape (F3)
280///
281/// Twenty-nine fields that most of the crate reads — the tenant and user
282/// repositories, the auth config, the session machinery — plus **seven
283/// cohesive sub-states** for the forty-six that a single handler module each
284/// reads. Before that split this struct had seventy-five fields and every
285/// handler saw all of them.
286///
287/// The sub-states are a *field-grouping* change and nothing more: every type
288/// is what it was and monomorphisation is what it was, so the authorization
289/// hot path is untouched. See [`bundles`] for why boxing them behind trait
290/// objects would have been the wrong fix.

444/// Assemble the OPAQUE server keys, requiring **both** or neither.
445///
446/// A deployment with one key and not the other is a misconfiguration that
447/// would otherwise fail deep inside a login, so it resolves to `None` here and
448/// the endpoints answer `503` — the same posture as having neither key, which
449/// is the one an operator can actually diagnose.
450pub(crate) fn opaque_keys_from(auth_config: &AuthConfig) -> Option<OpaqueServerKeys> {
451    match (auth_config.opaque_session_key, auth_config.opaque_setup_key) {
452        (Some(session_key), Some(setup_key)) => Some(OpaqueServerKeys {
453            session_key,
454            setup_key,
455        }),
456        _ => None,
457    }
458}

460impl<C: Connection + Clone> AppState<C> {
461    /// Switch on X4's external token-exchange path.
462    ///
463    /// A method on the assembled state rather than a constructor parameter,
464    /// because the collaborator it needs — the OIDC federation service — is
465    /// itself conditional on `AXIAM__AUTH__FEDERATION_ENCRYPTION_KEY` and is
466    /// installed after this state is built.
467    ///
468    /// **No federation service ⇒ no external path**, and that is the correct
469    /// coupling rather than an accident of ordering: without the encryption
470    /// key there is no way to read a provider's configuration, so there is
471    /// nothing to trust an external issuer *with*. Returns whether it was
472    /// enabled, so a caller can log the posture instead of guessing at it.
473    #[must_use = "the return value says whether the external path is actually live"]
474    pub fn enable_external_token_exchange(&mut self) -> bool {
475        let Some(federation) = self.federation.oidc_federation_service.clone() else {
476            return false;
477        };
478        let (resolver, authority) = crate::token_exchange::external_collaborators(
479            federation,
480            self.role_repo.clone(),
481            self.permission_repo.clone(),
482        );
483        // `with_external_subjects` consumes and returns, so the field is
484        // rebuilt rather than mutated in place — the service is a handful of
485        // cheap handles, so this costs nothing and keeps the "both or
486        // neither" pairing enforced by the type.
487        self.oauth2.token_exchange_service = self
488            .oauth2
489            .token_exchange_service
490            .clone()
491            .with_external_subjects(resolver, authority);
492        true
493    }

495    /// Assemble the X2 UMA service for one request.
496    ///
497    /// Built per call rather than stored, because it needs the
498    /// [`crate::authz::AuthzChecker`] — which arrives as its own `web::Data`
499    /// so that deployments can swap the checker without rebuilding this state.
500    /// The three parts it composes are all cheap handles (two repository
501    /// structs wrapping a cloned connection, and an `Arc`), so this is a
502    /// handful of refcount bumps, not a connection or a query.
503    pub fn uma_service(
504        &self,
505        authz: std::sync::Arc<dyn crate::authz::AuthzChecker>,
506    ) -> axiam_oauth2::uma::UmaService<
507        SurrealPermissionTicketRepository<C>,
508        crate::uma::EngineEvaluator,
509        crate::uma::ResourceScopeCatalog<SurrealScopeRepository<C>>,
510    > {
511        axiam_oauth2::uma::UmaService::new(

519    /// X1 — tell the reactor gate that this tenant's registrations changed.
520    ///
521    /// A no-op when no gate is composed (AMQP off, or a test harness), which is
522    /// the same shape [`Self::emit_webhook`] uses and for the same reason: a
523    /// side effect of an administrative write must not fail the write, and a
524    /// handler must not have to know whether the subsystem is configured.
525    ///
526    /// This is an **optimisation on top of the TTL, not a replacement for it**.
527    /// It only reaches the replica that served the mutation; every other
528    /// replica picks the change up when its own routing entry expires, which is
529    /// the bound the acceptance test asserts.
530    pub fn invalidate_reactor_routing(&self, tenant_id: uuid::Uuid) {
531        if let Some(invalidate) = &self.events.reactor_routing_invalidator {
532            invalidate(tenant_id);
533        }
534    }

536    /// Dispatch a domain event to any webhooks subscribed to `event_type` in
537    /// `tenant_id` (CQ-B22). Best-effort: if no publisher is wired
538    /// (`webhook_publisher` is `None`, e.g. tests) this is a no-op, and
539    /// publish failures inside `emit` are logged, never propagated — a webhook
540    /// side effect must not fail the originating API request. The publisher is
541    /// the `OutboundPublisher` port: the AMQP one in the full profile, the
542    /// in-process dispatcher in the minimal one (G-8).
543    pub async fn emit_webhook(
544        &self,
545        tenant_id: uuid::Uuid,
546        event_type: &str,
547        payload: serde_json::Value,
548    ) {
549        if let Some(publisher) = &self.events.webhook_publisher {
550            self.events
551                .webhook_delivery
552                .emit(
553                    publisher.as_ref(),
554                    tenant_id,
555                    event_type.to_string(),
556                    payload,
557                )
558                .await;
559        }
560    }
