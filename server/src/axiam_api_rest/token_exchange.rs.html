1//! X4 — the two adapters that connect the external token-exchange path to the
2//! running system.
3//!
4//! Same seam, and the same reasoning, as [`crate::uma`]: `axiam-oauth2` defines
5//! the grant against narrow traits rather than against `axiam-federation` and
6//! `axiam-authz` directly, so its own tests can drive every branch without a
7//! datastore or a network. This is where those traits get production
8//! implementations, because `axiam-api-rest` is the first crate holding both.
9//!
10//! Both adapters are deliberately thin — logic that lands here is logic the
11//! exchange's unit tests cannot see. The one exception is
12//! [`RbacScopeAuthority`], whose deny-override reading is genuinely a decision
13//! and is argued in full on the type.

30/// Resolves an external IdP's subject token through `axiam-federation`.
31///
32/// Pure translation: every decision — which providers are trusted, how the
33/// signature is checked, which user an `(issuer, sub)` pair means — belongs to
34/// [`OidcFederationService::verify_external_subject_token`], where it sits
35/// beside the login path that answers the same questions.
36pub struct FederationSubjectResolver<FC, FL, UR> {
37    federation: OidcFederationService<FC, FL, UR>,
38}
39
40impl<FC, FL, UR> FederationSubjectResolver<FC, FL, UR> {
41    pub fn new(federation: OidcFederationService<FC, FL, UR>) -> Self {
42        Self { federation }
43    }
44}

46impl<FC, FL, UR> ExternalSubjectResolver for FederationSubjectResolver<FC, FL, UR>
47where
48    FC: axiam_core::repository::FederationConfigRepository,
49    FL: axiam_core::repository::FederationLinkRepository,
50    UR: axiam_core::repository::UserRepository,
51{
52    fn resolve<'a>(
53        &'a self,
54        tenant_id: Uuid,
55        subject_token: &'a str,
56    ) -> Pin<
57        Box<
58            dyn Future<Output = Result<ResolvedExternalSubject, ExternalSubjectRejection>>
59                + Send
60                + 'a,
61        >,
62    > {
63        Box::pin(async move {
64            let now = chrono::Utc::now().timestamp();
65            match self
66                .federation
67                .verify_external_subject_token(tenant_id, subject_token, now)
68                .await
69            {
70                Ok(s) => Ok(ResolvedExternalSubject {
71                    issuer: s.issuer,
72                    provider_id: s.provider_id,
73                    provider_name: s.provider_name,
74                    external_subject: s.external_subject,
75                    user_id: s.user_id,
76                    newly_provisioned: s.newly_provisioned,
77                    candidate_scopes: s.candidate_scopes,
78                    subject_exp: s.subject_exp,
79                    max_lifetime_secs: s.max_lifetime_secs,
80                }),
81                Err(ExternalSubjectError::IssuerNotTrusted) => {
82                    Err(ExternalSubjectRejection::IssuerNotTrusted)
83                }
84                Err(ExternalSubjectError::Server(m)) => Err(ExternalSubjectRejection::Server(m)),
85                // "Not linked", "not active" and every token-level failure
86                // collapse to one wire answer on purpose: they are all facts
87                // about the *presented token's subject*, and distinguishing
88                // them for the caller would turn the grant into a probe for
89                // which partner identities exist in this tenant and what state
90                // they are in. The distinction survives where it is useful —
91                // the reason code below goes to the audit record.
92                Err(other) => Err(ExternalSubjectRejection::Rejected(
93                    other.reason_code().to_string(),
94                )),
95            }
96        })
97    }
98}

100/// Answers "which of these AXIAM scope names does this user actually hold",
101/// from the RBAC engine's own data.
102///
103/// # The mapping
104///
105/// An AXIAM scope name is checked as an **`action`**, the same mapping X2
106/// pinned for UMA resource scopes (`claude_dev/uma-mapping-design.md`). Using
107/// one vocabulary across both features means an operator writes `read:orders`
108/// once and it means the same thing in a permission grant, a UMA ticket and a
109/// `scope_map` entry.
110///
111/// # Why it is not a `check_access` call
112///
113/// [`AccessRequest`](axiam_authz::types::AccessRequest) is *resource-scoped*,
114/// and an OAuth2 scope is not: `read:orders` in a token is a claim about the
115/// bearer, not about one row. There is no resource id to put in the request,
116/// and inventing one — a tenant root, a per-provider configured resource —
117/// would be inventing a policy surface X4 was not asked to add.
118///
119/// So this reads the same two tables the engine reads and answers the question
120/// the engine is never asked: **does this user hold this action anywhere in
121/// this tenant?**
122///
123/// # Deny-override, at its broadest reading
124///
125/// A scope survives only if the user has an `allow` grant for the action **and
126/// no `deny` grant for it anywhere in their applicable roles**. A deny scoped
127/// to one resource therefore withholds the *scope* entirely, even though the
128/// same deny would only veto that one resource in a live `check_access`.
129///
130/// That asymmetry is deliberate, and it is the conservative direction. A
131/// bearer scope cannot express "except on resource X" — it travels in a token
132/// with no resource attached — so the only two honest answers are "grant it
133/// everywhere" or "do not grant it". On a cross-domain path, taking the
134/// narrower answer costs an operator one extra grant and buys the property
135/// that **adding a deny rule can never widen what a partner token can reach**,
136/// which is exactly the property B1 chose deny-override to get.
137///
138/// The live check still governs every actual request: a token carrying
139/// `read:orders` does not thereby read anything, it merely presents a scope
140/// the engine then evaluates per resource.
141pub struct RbacScopeAuthority<RR, PR> {
142    role_repo: RR,
143    permission_repo: PR,
144}

155impl<RR, PR> SubjectScopeAuthority for RbacScopeAuthority<RR, PR>
156where
157    RR: RoleRepository + Send + Sync,
158    PR: PermissionRepository + Send + Sync,
159{
160    fn held_scopes<'a>(
161        &'a self,
162        tenant_id: Uuid,
163        subject_id: Uuid,
164        candidates: &'a [String],
165    ) -> Pin<Box<dyn Future<Output = Result<Vec<String>, String>> + Send + 'a>> {
166        Box::pin(async move {
167            if candidates.is_empty() {
168                return Ok(Vec::new());
169            }
170
171            let assignments = self
172                .role_repo
173                .get_user_role_assignments(tenant_id, subject_id)
174                .await
175                .map_err(|e| e.to_string())?;
176            if assignments.is_empty() {
177                return Ok(Vec::new());
178            }
179
180            // Deduplicated: a user assigned the same role globally and on a
181            // resource yields two assignments and one role's grants.
182            let mut role_ids: Vec<Uuid> = Vec::with_capacity(assignments.len());
183            for a in &assignments {
184                if !role_ids.contains(&a.role.id) {
185                    role_ids.push(a.role.id);
186                }
187            }
188
189            let grants_by_role = self
190                .permission_repo
191                .get_role_permission_grants_for_roles(tenant_id, &role_ids)
192                .await
193                .map_err(|e| e.to_string())?;
194
195            let wanted: HashSet<&str> = candidates.iter().map(String::as_str).collect();
196            let mut allowed: HashSet<String> = HashSet::new();
197            let mut denied: HashSet<String> = HashSet::new();
198            for grants in grants_by_role.values() {
199                for grant in grants {
200                    if !wanted.contains(grant.permission.action.as_str()) {
201                        continue;
202                    }
203                    match grant.effect {
204                        PermissionEffect::Allow => {
205                            allowed.insert(grant.permission.action.clone());
206                        }
207                        PermissionEffect::Deny => {
208                            denied.insert(grant.permission.action.clone());
209                        }
210                    }
211                }
212            }
213
214            // Order follows `candidates`, not the datastore's, so the result
215            // is deterministic and the issued token's `scope` string is stable
216            // across calls.
217            Ok(candidates
218                .iter()
219                .filter(|c| allowed.contains(*c) && !denied.contains(*c))
220                .cloned()
221                .collect())
222        })
223    }
224}

226/// Boxed pair, as the exchange service takes them.
227pub fn external_collaborators<FC, FL, UR, RR, PR>(
228    federation: OidcFederationService<FC, FL, UR>,
229    role_repo: RR,
230    permission_repo: PR,
231) -> (
232    Arc<dyn ExternalSubjectResolver>,
233    Arc<dyn SubjectScopeAuthority>,
234)
235where
236    FC: axiam_core::repository::FederationConfigRepository + 'static,
237    FL: axiam_core::repository::FederationLinkRepository + 'static,
238    UR: axiam_core::repository::UserRepository + 'static,
239    RR: RoleRepository + Send + Sync + 'static,
240    PR: PermissionRepository + Send + Sync + 'static,
241{
242    (
243        Arc::new(FederationSubjectResolver::new(federation)),
244        Arc::new(RbacScopeAuthority::new(role_repo, permission_repo)),
245    )

417    /// The decision this type exists to make explicit: a deny anywhere in the
418    /// user's applicable roles withholds the scope, even alongside an allow.
419    /// A bearer scope cannot say "except on resource X", so the only honest
420    /// answers are "everywhere" and "not at all".
421    #[tokio::test]
422    async fn a_deny_anywhere_withholds_the_scope_even_against_an_allow() {
423        let allower = role("reader");
424        let denier = role("embargoed");
425        let mut grants = HashMap::new();
426        grants.insert(
427            allower.id,
428            vec![grant("read:orders", PermissionEffect::Allow)],
429        );
430        grants.insert(
431            denier.id,
432            vec![grant("read:orders", PermissionEffect::Deny)],
433        );
434
435        let a = authority(
436            vec![
437                RoleAssignment {
438                    role: allower,
439                    resource_id: None,
440                    tenant_scope: None,
441                },
442                RoleAssignment {
443                    role: denier,
444                    // Scoped to one resource — and it still withholds the
445                    // whole scope. That is the conservative reading, argued on
446                    // `RbacScopeAuthority`.
447                    resource_id: Some(Uuid::new_v4()),
448                    tenant_scope: None,
449                },
450            ],
451            grants,
452        );
453
454        assert!(
455            a.held_scopes(Uuid::new_v4(), Uuid::new_v4(), &v(&["read:orders"]))
456                .await
457                .unwrap()
458                .is_empty()
459        );
