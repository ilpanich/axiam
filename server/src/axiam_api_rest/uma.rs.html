1//! X2 — the two adapters that connect [`UmaService`] to the running system.
2//!
3//! `axiam-oauth2` defines the UMA grant against two narrow traits
4//! ([`PermissionEvaluator`], [`ScopeCatalog`]) rather than against the RBAC
5//! engine and the scope store directly, so that its own tests can drive it
6//! without a datastore. Those traits need production implementations, and this
7//! is where they live: `axiam-api-rest` is the first crate that has both an
8//! [`AuthzChecker`] and the repositories in scope.
9//!
10//! Both adapters are deliberately thin. Any logic that lands here is logic the
11//! UMA unit tests cannot see.
12

24/// Evaluates a UMA `(resource, scope)` pair as an ordinary access check.
25///
26/// # The mapping, and why it is not configurable
27///
28/// A UMA resource scope is the AXIAM **`action`**, evaluated against the
29/// registered resource with no AXIAM scope named
30/// (`claude_dev/uma-mapping-design.md`):
31///
32/// ```text
33/// AccessRequest { tenant_id, subject_id, action: <uma scope>, resource_id, scope: None }
34/// ```
35///
36/// Riding UMA scopes on the AXIAM `scope` field instead would be unsafe, not
37/// merely different: `grant_applies` treats a grant with an empty `scope_ids`
38/// as matching *any* requested scope, so one unscoped grant of a fixed
39/// `uma:access` action would satisfy every scope on the resource at once —
40/// registering a new `delete` scope would silently widen every existing grant
41/// to include it.
42///
43/// # Deny-override comes for free
44///
45/// Because this **is** the same check a live request makes, a
46/// `PermissionEffect::Deny` grant vetoes an RPT exactly as it vetoes a live
47/// check (B1). That is X2's stated requirement, and it holds here without any
48/// UMA-specific code — which is the point of the mapping, and the reason
49/// `DeniedByRule` is carried through as its own outcome instead of being
50/// flattened into "not allowed".
51pub struct EngineEvaluator {
52    authz: Arc<dyn AuthzChecker>,
53}
54
55impl EngineEvaluator {
56    pub fn new(authz: Arc<dyn AuthzChecker>) -> Self {
57        Self { authz }
58    }
59}
60
61impl PermissionEvaluator for EngineEvaluator {
62    fn evaluate<'a>(
63        &'a self,
64        tenant_id: Uuid,
65        subject_id: Uuid,
66        resource_id: Uuid,
67        scope: &'a str,
68    ) -> Pin<Box<dyn Future<Output = Result<PairOutcome, String>> + Send + 'a>> {
69        Box::pin(async move {
70            let request = AccessRequest {
71                tenant_id,
72                subject_scope: SubjectScope::Tenant,
73                subject_id,
74                action: scope.to_string(),
75                resource_id,
76                scope: None,
77            };
78
79            match self.authz.check_access(&request).await {
80                Ok(AccessDecision::Allow) => Ok(PairOutcome::Allowed),
81                Ok(AccessDecision::Deny(_)) => Ok(PairOutcome::NoGrant),
82                Ok(AccessDecision::DeniedByRule(_)) => Ok(PairOutcome::DeniedByRule),
83                // A failed check is not a denial. Reporting it as one would
84                // turn a datastore blip into `access_denied`, which tells the
85                // resource server the requesting party lacks authority — a
86                // claim this adapter is in no position to make.
87                Err(e) => Err(e.to_string()),
88            }
89        })
90    }
91}

93/// Answers "which scope names has this resource declared", from the ordinary
94/// AXIAM scope rows on that resource.
95///
96/// UMA's `_id` **is** the AXIAM resource id — there is no parallel resource
97/// store to translate through — so a resource's declared scopes are just its
98/// `Scope` rows. That set is the allow-list of names a resource server may ask
99/// for, not an input to any decision: an undeclared scope is refused when the
100/// ticket is minted, so the resource server learns it named something that does
101/// not exist rather than being told the requesting party was denied.
102pub struct ResourceScopeCatalog<R> {
103    scopes: R,
104}
105
106impl<R> ResourceScopeCatalog<R> {
107    pub fn new(scopes: R) -> Self {
108        Self { scopes }
109    }
110}
111
112impl<R> ScopeCatalog for ResourceScopeCatalog<R>
113where
114    R: ScopeRepository + Send + Sync,
115{
116    fn declared_scopes<'a>(
117        &'a self,
118        tenant_id: Uuid,
119        resource_id: Uuid,
120    ) -> Pin<Box<dyn Future<Output = Result<Vec<String>, String>> + Send + 'a>> {
121        Box::pin(async move {
122            let scopes = self
123                .scopes
124                .list_by_resource(tenant_id, resource_id)
125                .await
126                .map_err(|e| e.to_string())?;
127            Ok(scopes.into_iter().map(|s| s.name).collect())
128        })
129    }
130}
