1//! Accounting for request-audit rows that did not reach the datastore (T-108).
2//!
3//! The request-audit middleware hands each row to a bounded channel and a
4//! worker appends it. A row is lost in two places: the channel is full when the
5//! request finishes (a slow datastore, a burst), or the append fails (the
6//! datastore is down or rejects the row). Both used to leave one log line each,
7//! the second at WARN, and nothing an operator could alert on.
8//!
9//! [`RequestAuditLoss`] is the one place both are recorded. Per loss it:
10//!
11//! 1. counts it (`dropped` or `failed`, monotonic since the process started),
12//! 2. hands the row to the [`DeadLetterWriter`] — a non-blocking queue, so the
13//!    request path does no file I/O — and
14//! 3. emits one ERROR line naming the counts, at most once per
15//!    [`REPORT_INTERVAL`], so a flood is one line a minute rather than one per
16//!    request.
17//!
18//! `GET /health/jobs` reads [`RequestAuditLoss::snapshot`].
19
20use std::sync::Arc;
21use std::sync::atomic::{AtomicI64, AtomicU64, Ordering};
22use std::time::{Duration, Instant};
23
24use axiam_core::models::audit::CreateAuditLogEntry;
25use chrono::{DateTime, Utc};
26
27use crate::dead_letter::DeadLetterWriter;
28
29/// The least time between two ERROR lines about lost rows.
30pub const REPORT_INTERVAL: Duration = Duration::from_secs(60);
31
32/// How long after the last loss `/health/jobs` still reports `recent_loss`.
33///
34/// The counters never go down, so they cannot say "it stopped". This window can:
35/// a loss within it turns the endpoint's `status` to `degraded`, and the status
36/// returns to `ok` by itself once the datastore has been taking rows again for
37/// this long.
38pub const RECENT_LOSS_WINDOW: Duration = Duration::from_secs(15 * 60);
39
40/// `last_report_ms` before any line has been logged.
41const NEVER: u64 = u64::MAX;
42
43/// A point-in-time reading of [`RequestAuditLoss`].
44#[derive(Debug, Clone, PartialEq, Eq)]
45pub struct RequestAuditLossSnapshot {
46    /// Rows refused because the channel was full (or closed) when the request
47    /// finished. Since process start.
48    pub dropped: u64,
49    /// Rows the worker took off the channel and could not append. Since process
50    /// start.
51    pub failed: u64,
52    /// Lost rows written to the dead-letter file.
53    pub dead_lettered: u64,
54    /// Lost rows kept nowhere: no file is configured, or the file's queue was
55    /// full, or the file had reached its budget, or it could not be written.
56    pub not_recoverable: u64,
57    /// Whether a dead-letter file is configured.
58    pub dead_letter_configured: bool,
59    /// Whether the dead-letter file has reached the request rows' share of its
60    /// budget, so lost rows are refused by it (R1W2-02).
61    pub dead_letter_full: bool,
62    /// When the most recent row was lost, if any was.
63    pub last_loss_at: Option<DateTime<Utc>>,
64    /// Whether a row was lost within [`RECENT_LOSS_WINDOW`].
65    pub recent_loss: bool,
66}
67
68struct Inner {
69    dropped: AtomicU64,
70    failed: AtomicU64,
71    /// Unix milliseconds of the latest loss; 0 when none.
72    last_loss_ms: AtomicI64,
73    /// Milliseconds since `base` of the latest ERROR line; [`NEVER`] when none.
74    last_report_ms: AtomicU64,
75    /// Lost rows (dropped + failed) at the latest ERROR line, for the delta.
76    reported_total: AtomicU64,
77    base: Instant,
78    dead_letter: DeadLetterWriter,
79}
80
81/// Shared handle to the request-audit loss counters. Cloning shares them.
82#[derive(Clone)]
83pub struct RequestAuditLoss {
84    inner: Arc<Inner>,
85}
86
87impl RequestAuditLoss {
88    /// Counters at zero, routing lost rows to `dead_letter`.
89    pub fn new(dead_letter: DeadLetterWriter) -> Self {
90        Self {
91            inner: Arc::new(Inner {
92                dropped: AtomicU64::new(0),
93                failed: AtomicU64::new(0),
94                last_loss_ms: AtomicI64::new(0),
95                last_report_ms: AtomicU64::new(NEVER),
96                reported_total: AtomicU64::new(0),
97                base: Instant::now(),
98                dead_letter,
99            }),
100        }
101    }
102
103    /// The dead-letter writer lost rows go to.
104    pub fn dead_letter(&self) -> &DeadLetterWriter {
105        &self.inner.dead_letter
106    }
107
108    /// A row could not be queued: the channel was full or closed. Called on the
109    /// request path, so nothing here waits.
110    pub fn record_dropped(&self, entry: CreateAuditLogEntry, reason: &'static str) {
111        self.inner.dropped.fetch_add(1, Ordering::Relaxed);
112        self.lose(entry, reason, &reason);
113    }
114
115    /// The worker took a row off the channel and the append failed.
116    pub fn record_failed(&self, entry: CreateAuditLogEntry, error: &dyn std::fmt::Display) {
117        self.inner.failed.fetch_add(1, Ordering::Relaxed);
118        self.lose(entry, "append failed", error);
119    }
120
121    fn lose(
122        &self,
123        entry: CreateAuditLogEntry,
124        reason: &'static str,
125        detail: &dyn std::fmt::Display,
126    ) {
127        self.inner
128            .last_loss_ms
129            .store(Utc::now().timestamp_millis(), Ordering::Relaxed);
130        self.inner.dead_letter.submit(entry);
131        let now_ms = self.inner.base.elapsed().as_millis() as u64;
132        if self.should_report(now_ms) {
133            self.report(reason, detail);
134        }
135    }
136
137    /// Claim the right to log at `now_ms`; at most one caller per interval wins.
138    fn should_report(&self, now_ms: u64) -> bool {
139        let last = self.inner.last_report_ms.load(Ordering::Relaxed);
140        if last != NEVER && now_ms.saturating_sub(last) < REPORT_INTERVAL.as_millis() as u64 {
141            return false;
142        }
143        self.inner
144            .last_report_ms
145            .compare_exchange(last, now_ms, Ordering::Relaxed, Ordering::Relaxed)
146            .is_ok()
147    }
148
149    fn report(&self, reason: &'static str, detail: &dyn std::fmt::Display) {
150        let s = self.snapshot();
151        let total = s.dropped + s.failed;
152        let since_last_report = total - self.inner.reported_total.swap(total, Ordering::Relaxed);
153        tracing::error!(
154            target: "axiam.audit.loss",
155            reason,
156            detail = %detail,
157            lost_since_last_report = since_last_report,
158            dropped_total = s.dropped,
159            failed_total = s.failed,
160            dead_lettered_total = s.dead_lettered,
161            not_recoverable_total = s.not_recoverable,
162            dead_letter_configured = s.dead_letter_configured,
163            dead_letter_full = s.dead_letter_full,
164            "request audit rows are being lost (this line is logged at most once a minute; \
165             GET /health/jobs `request_audit` carries the running counts)"
166        );
167    }
168
169    /// The counters now.
170    pub fn snapshot(&self) -> RequestAuditLossSnapshot {
171        let dropped = self.inner.dropped.load(Ordering::Relaxed);
172        let failed = self.inner.failed.load(Ordering::Relaxed);
173        let dead_letter = &self.inner.dead_letter;
174        let configured = dead_letter.is_configured();
175        let last_ms = self.inner.last_loss_ms.load(Ordering::Relaxed);
176        let last_loss_at = (last_ms != 0)
177            .then(|| DateTime::<Utc>::from_timestamp_millis(last_ms))
178            .flatten();
179        let recent_loss = last_loss_at.is_some_and(|t| {
180            Utc::now().signed_duration_since(t)
181                < chrono::Duration::from_std(RECENT_LOSS_WINDOW).unwrap_or(chrono::Duration::MAX)
182        });
183        RequestAuditLossSnapshot {
184            dropped,
185            failed,
186            dead_lettered: dead_letter.written(),
187            // With no file every lost row is unrecoverable; with one, only the
188            // rows the file's writer could not keep.
189            not_recoverable: if configured {
190                dead_letter.lost()
191            } else {
192                dropped + failed
193            },
194            dead_letter_configured: configured,
195            dead_letter_full: dead_letter.is_full(),
196            last_loss_at,
197            recent_loss,
198        }
199    }
200}

202#[cfg(test)]
203mod tests {
204    use super::*;
205
206    #[test]
207    fn the_report_is_rate_limited_to_one_per_interval() {
208        let loss = RequestAuditLoss::new(DeadLetterWriter::disabled());
209        let step = REPORT_INTERVAL.as_millis() as u64;
210        assert!(loss.should_report(5), "the first loss is reported at once");
211        assert!(!loss.should_report(5 + step - 1), "inside the interval");
212        assert!(loss.should_report(5 + step), "after the interval");
213        assert!(!loss.should_report(5 + step + 1));
214    }
