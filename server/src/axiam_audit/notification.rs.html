1//! Notification dispatcher — matches audit events to notification rules
2//! and enqueues outbound mail messages for each matched recipient (T19.13).
3//!
4//! The dispatcher calls `mail_publisher.publish(...)` once per matched
5//! (event, recipient) pair.  On publish error the error is logged and
6//! execution continues — fire-and-forget (D-14).
7
8use axiam_core::error::AxiamResult;
9use axiam_core::models::mail::{MailType, OutboundMailMessage};
10use axiam_core::models::notification_rule::NotificationEventType;
11use axiam_core::repository::{MailPublisher, NotificationRuleRepository};
12use chrono::Utc;
13use uuid::Uuid;
14
15/// Dispatches audit events to matching notification rules by enqueuing
16/// one `OutboundMailMessage(Notification)` per matched recipient.
17pub struct NotificationDispatcher<N: NotificationRuleRepository> {
18    rule_repo: N,
19}
20
21impl<N: NotificationRuleRepository> NotificationDispatcher<N> {
22    /// Create a new dispatcher backed by the given rule repository.
23    pub fn new(rule_repo: N) -> Self {
24        Self { rule_repo }
25    }
26
27    /// Match an audit event against notification rules and enqueue one
28    /// `OutboundMailMessage` per matched recipient.
29    ///
30    /// `tenant_id` and `org_id` are used to populate the mail message
31    /// context so the consumer can resolve the correct email config.
32    ///
33    /// Returns `Ok(enqueued_count)` where count is the number of messages
34    /// successfully handed to `mail_publisher`.  Publish errors are logged
35    /// and do **not** propagate — callers get a successful result even if
36    /// some (or all) enqueue calls fail.
37    ///
38    /// Returns `Ok(0)` if no rules match or the action/outcome does not map
39    /// to any known notification event type.
40    pub async fn dispatch(
41        &self,
42        tenant_id: Uuid,
43        org_id: Uuid,
44        action: &str,
45        outcome: &str,
46        actor_id: Option<Uuid>,
47        details: &str,
48        mail_publisher: &impl MailPublisher,
49    ) -> AxiamResult<usize> {
50        let event_types = NotificationEventType::from_audit_action(action, outcome);
51        if event_types.is_empty() {
52            return Ok(0);
53        }
54
55        // Collect all event type strings and query once (avoids N+1).
56        let event_strings: Vec<String> = event_types.iter().map(|e| e.to_db_string()).collect();
57        let rules = self
58            .rule_repo
59            .get_by_events(tenant_id, &event_strings)
60            .await?;
61
62        // Built once for the whole dispatch: it is the same for every rule and
63        // every recipient, and it is the one place `username` is decided.
64        //
65        // `username` is present ONLY when there is no actor to name. The mail
66        // consumer resolves it from `user_id` and its answer is the better one,
67        // but a message's own context is overlaid on top of what the consumer
68        // resolved — so supplying a placeholder unconditionally would mask every
69        // real actor. It cannot be a JSON `null` either: the consumer
70        // stringifies non-string values, so a null would arrive as the literal
71        // word "null". The key has to be genuinely absent, which is why this is
72        // assembled rather than written as one `json!` literal.
73        //
74        // Supplying nothing at all would instead leave the consumer's generic
75        // "unknown" on the events an administrator most wants alerting on — a
76        // failed login, a lockout — which have no authenticated actor by
77        // definition.
78        // `event` is not here: it is the one key that varies per rule, and is
79        // inserted into the clone below.
80        let mut context = serde_json::Map::new();
81        context.insert("details".into(), details.into());
82        context.insert("action".into(), action.into());
83        context.insert("outcome".into(), outcome.into());
84        if actor_id.is_none() {
85            // An event AXIAM itself raises (a dead-lettered delivery) has no
86            // caller at all, authenticated or not; saying so would send an
87            // administrator looking for an intruder.
88            let who = if event_types.iter().all(|e| e.is_system_event()) {
89                "AXIAM (an automated process)"
90            } else {
91                "an unauthenticated caller"
92            };
93            context.insert("username".into(), who.into());
94        }
95
96        let mut enqueued = 0usize;
97        for rule in rules {
98            if rule.recipient_emails.is_empty() {
99                continue;
100            }
101            // Find the first matching event type for this rule.
102            let matched_event = event_types
103                .iter()
104                .find(|et| rule.events.contains(*et))
105                .map(|et| et.to_db_string());
106
107            let event_name = match matched_event {
108                Some(name) => name,
109                None => continue,
110            };
111
112            let mut context = context.clone();
113            context.insert("event".into(), event_name.clone().into());
114            let template_context = serde_json::Value::Object(context);
115
116            // Enqueue one OutboundMailMessage per recipient.
117            for recipient in &rule.recipient_emails {
118                let msg = OutboundMailMessage {
119                    mail_type: MailType::Notification,
120                    tenant_id,
121                    org_id,
122                    user_id: actor_id.unwrap_or(Uuid::nil()),
123                    to_address: recipient.clone(),
124                    template_context: template_context.clone(),
125                    attempt_count: 0,
126                    enqueued_at: Utc::now(),
127                };
128
129                match mail_publisher.publish(msg).await {
130                    Ok(()) => {
131                        enqueued += 1;
132                        tracing::debug!(
133                            event = %event_name,
134                            recipient = %recipient,
135                            "notification mail enqueued"
136                        );
137                    }
138                    Err(e) => {
139                        // Fire-and-forget: log and continue (D-14).
140                        tracing::warn!(
141                            error = %e,
142                            event = %event_name,
143                            "failed to enqueue notification mail; skipping recipient"
144                        );
145                    }
146                }
147            }
148        }
149
150        Ok(enqueued)
151    }
152}

158/// Connects a [`NotificationDispatcher`] to the audit event stream.
159///
160/// This is the piece that was missing. `NotificationDispatcher` was complete,
161/// tested, and exported — and constructed nowhere outside its own crate, so no
162/// notification rule an administrator configured had any effect. The rules were
163/// stored, listed by the API and rendered in the admin UI; nothing consulted
164/// them.
165///
166/// Implements [`crate::middleware::AuditEventSink`] so
167/// `AuditMiddleware::spawn_with_sink` can drive it on the audit worker's task,
168/// off the request path.
169pub struct NotificationSink<N: NotificationRuleRepository, P: MailPublisher> {
170    dispatcher: NotificationDispatcher<N>,
171    mail_publisher: P,
172}
173
174impl<N: NotificationRuleRepository, P: MailPublisher> NotificationSink<N, P> {
175    /// Build a sink over a rule repository and a mail publisher.
176    pub fn new(rule_repo: N, mail_publisher: P) -> Self {
177        Self {
178            dispatcher: NotificationDispatcher::new(rule_repo),
179            mail_publisher,
180        }
181    }
182}

184impl<N, P> crate::middleware::AuditEventSink for NotificationSink<N, P>
185where
186    N: NotificationRuleRepository + 'static,
187    P: MailPublisher + 'static,
188{
189    fn on_event<'a>(
190        &'a self,
191        event: &'a crate::middleware::AuditEvent,
192    ) -> std::pin::Pin<Box<dyn std::future::Future<Output = ()> + Send + 'a>> {
193        Box::pin(async move {
194            let entry = &event.entry;
195
196            // Rules are looked up by tenant; a nil tenant is an event the
197            // middleware could attribute to nobody (an unauthenticated request
198            // to a handler that sets no `AuditAttribution`). Skipping it here
199            // avoids a repository round-trip per health-adjacent request.
200            if entry.tenant_id.is_nil() {
201                return;
202            }
203
204            let outcome = format!("{:?}", entry.outcome);
205            let actor_id = if entry.actor_id.is_nil() {
206                None
207            } else {
208                Some(entry.actor_id)
209            };
210
211            // The rendered body's `details`. Deliberately the action and the
212            // HTTP status only: an audit notification goes to whatever addresses
213            // an administrator typed into a rule, which are not necessarily
214            // addresses cleared to see request contents (D-16).
215            let details = format!("{} ({outcome})", entry.action);
216
217            match self
218                .dispatcher
219                .dispatch(
220                    entry.tenant_id,
221                    event.org_id,
222                    &entry.action,
223                    &outcome,
224                    actor_id,
225                    &details,
226                    &self.mail_publisher,
227                )
228                .await
229            {
230                Ok(0) => {}
231                Ok(n) => tracing::debug!(
232                    enqueued = n,
233                    action = %entry.action,
234                    "notification rule matched; mail enqueued"
235                ),
236                // Swallowed on purpose: the audit entry is already written, and
237                // `AuditEventSink` is best-effort by contract.
238                Err(e) => tracing::warn!(
239                    error = %e,
240                    action = %entry.action,
241                    "failed to dispatch notification rules for an audit event"
242                ),
243            }
244        })
245    }
246}

252/// An [`AuditLogRepository`] that lets the notification rules see the rows it
253/// appends.
254///
255/// [`NotificationSink`] is driven by [`crate::AuditMiddleware`]'s worker, which
256/// only ever sees HTTP requests. A row a background process writes straight to
257/// the repository — the outbound dispatcher's `scim_push.delivery_failed`, the
258/// record of a dead letter — never reaches it, so a rule an administrator
259/// configured for that event (T19.13's mechanism, D-58: no second channel)
260/// would match nothing in a running server. The consumer that writes it is
261/// given this wrapper instead of the bare repository.
262///
263/// Only a row that **maps to a notification event** costs anything beyond the
264/// append: the [`NotificationGate`] is asked whether it may notify now, the
265/// tenant's organization is looked up (mail needs it to resolve the email
266/// configuration) and the sink is called, after the append has succeeded and
267/// with its failures swallowed, exactly as the middleware's worker does. Every
268/// other method is the inner repository's.
269///
270/// **Every row is appended; only the notification is gated.** A background
271/// process can write the same notifiable row thousands of times in a minute —
272/// one dead letter per reference while a downstream is down — and a rule
273/// mails each recipient once per row it sees (W5 F4 review, T-418). The gate
274/// is how a wrapper's owner coalesces them; there is deliberately no
275/// constructor without one.
276pub struct NotifyingAuditLog<A, T> {
277    inner: A,
278    sink: std::sync::Arc<dyn crate::middleware::AuditEventSink>,
279    tenants: T,
280    gate: std::sync::Arc<dyn NotificationGate>,
281}
282
283impl<A, T> NotifyingAuditLog<A, T> {
284    /// Wrap `inner`; `tenants` resolves a row's organization, and `gate`
285    /// decides which notifiable rows reach `sink`.
286    pub fn new(
287        inner: A,
288        sink: std::sync::Arc<dyn crate::middleware::AuditEventSink>,
289        tenants: T,
290        gate: std::sync::Arc<dyn NotificationGate>,
291    ) -> Self {
292        Self {
293            inner,
294            sink,
295            tenants,
296            gate,
297        }
298    }
299}

301/// Decides whether a notifiable audit row a background process appended may
302/// reach the notification rules **now** (W5 F4 review, T-418, D-73).
303///
304/// Called only for a row that maps to a notification event, after it was
305/// appended. `false` drops the notification, never the row. An implementation
306/// must not fail open into a flood: when it cannot decide, it should say
307/// `false` and log once.
308pub trait NotificationGate: Send + Sync {
309    /// Whether `entry` may be handed to the notification sink.
310    fn admit<'a>(
311        &'a self,
312        entry: &'a axiam_core::models::audit::CreateAuditLogEntry,
313    ) -> std::pin::Pin<Box<dyn std::future::Future<Output = bool> + Send + 'a>>;
314}
315
316impl<A, T> axiam_core::repository::AuditLogRepository for NotifyingAuditLog<A, T>
317where
318    A: axiam_core::repository::AuditLogRepository,
319    T: axiam_core::repository::TenantRepository,
320{
321    async fn append(
322        &self,
323        input: axiam_core::models::audit::CreateAuditLogEntry,
324    ) -> AxiamResult<axiam_core::models::audit::AuditLogEntry> {
325        let appended = self.inner.append(input.clone()).await?;
326        let outcome = format!("{:?}", input.outcome);
327        let notifiable = !input.tenant_id.is_nil()
328            && !NotificationEventType::from_audit_action(&input.action, &outcome).is_empty();
329        if notifiable && self.gate.admit(&input).await {
330            let org_id = match self.tenants.get_by_id(input.tenant_id).await {
331                Ok(tenant) => tenant.organization_id,
332                Err(error) => {
333                    tracing::warn!(
334                        %error,
335                        tenant_id = %input.tenant_id,
336                        "a notifiable audit row's organization could not be resolved"
337                    );
338                    Uuid::nil()
339                }
340            };
341            self.sink
342                .on_event(&crate::middleware::AuditEvent {
343                    entry: input,
344                    org_id,
345                })
346                .await;
347        }
348        Ok(appended)
349    }
350
351    fn list(
352        &self,
353        tenant_id: Uuid,
354        filter: axiam_core::repository::AuditLogFilter,
355        pagination: axiam_core::repository::Pagination,
356    ) -> impl std::future::Future<

359        >,

362    }

374    }

746    /// audit event produces mail. `NotificationDispatcher` could always do this;

758
759        sink.on_event(&AuditEvent {

765        assert_eq!(publisher.count(), 2, "one message per configured recipient");
766        let msgs = publisher.messages();

914    #[tokio::test]
915    async fn a_known_actor_is_left_for_the_consumer_to_resolve() {
916        // The complement. The consumer resolves `username` from `user_id` and
917        // its value is the better one, but the message's own context is overlaid
918        // on top — so the placeholder has to be dropped when there is a real
919        // actor, or it would win and the alert would name nobody.
920        let repo = MockRuleRepo::new(vec![make_rule(vec!["soc@example.com"])]);
921        let publisher = RecordingPublisher::new();
922        let dispatcher = NotificationDispatcher::new(repo);
923
924        let actor = Uuid::new_v4();
925        dispatcher
926            .dispatch(
927                Uuid::new_v4(),
928                Uuid::new_v4(),
929                "POST /api/v1/auth/login",
930                "Failure",
931                Some(actor),
932                "details",
933                &publisher,
934            )
935            .await
936            .unwrap();
937
938        let msg = publisher.messages().into_iter().next().unwrap();
939        // Genuinely absent, not null. The consumer stringifies non-string
940        // values, so a null would arrive as the literal word "null" and mask the
941        // name it resolved from `user_id`.
942        assert!(
943            msg.template_context
944                .as_object()
945                .unwrap()
946                .get("username")
947                .is_none(),
948            "a resolvable actor must be left to the consumer, not overlaid"
949        );
950        assert_eq!(msg.user_id, actor);
951    }
