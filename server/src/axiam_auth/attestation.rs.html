1//! X3 wave 2 — attestation CA-list building/caching (spec D7, addendum
2//! W2-D3) and the pure per-credential compliance-evaluation helper (spec D9,
3//! the endpoint itself is later-wave REST work).
4//!
5//! ## Why this depends on `AttestationMetadataSource`, not `axiam-pki` (W2-D4)
6//!
7//! `axiam-auth` needs three things to enforce the attestation policy: the
8//! effective [`WebauthnAttestationPolicy`] (resolved by the caller from
9//! `WebauthnAttestationPolicyRepository` — an absent row means the default,
10//! D5 — before calling into [`crate::webauthn::WebauthnService`]), an AAGUID
11//! → [`MdsEntry`] lookup, and the DER attestation roots needed to build a
12//! `webauthn-rs` `AttestationCaList`. The last two are exactly
13//! [`AttestationMetadataSource`] (`axiam-core`, implemented over
14//! `MdsRepository` in `axiam-db`). Depending on that small, data-only trait
15//! rather than on `axiam-pki` directly turned out to be cheap — nothing in
16//! this module or `crate::webauthn` needs `axiam-pki`'s certificate/JWT
17//! verification types, only plain `MdsEntry`/DER-bytes data — so the W2-D4
18//! preferred option is what shipped; `axiam-auth`'s `Cargo.toml` gained no
19//! new dependency for X3.
20//!
21//! ## The CA-list cache (W2-D3)
22//!
23//! Building an `AttestationCaList` means an OpenSSL `X509::from_der` parse
24//! per attestation root (up to ~255 unique certs on the live MDS BLOB,
25//! addendum W2-D2) — cheap once, wasteful to repeat on every registration
26//! start. [`AttestationCaCache`] follows the same shape as the other
27//! process-wide caches in this codebase (`axiam_authz::decision_cache`,
28//! `axiam_api_rest::tenant_org_cache`): a plain `Mutex`-guarded map, built
29//! lazily, invalidated wholesale rather than per-key. Unlike those, it has no
30//! TTL — nothing here goes stale on its own; it is stale only when its two
31//! trigger events happen (an MDS refresh changes which roots exist, or a
32//! tenant's `allowed_aaguids` changes which subset applies), and both are
33//! discrete mutations the caller can hook directly rather than needing a
34//! time-based fallback. `axiam-server` wiring those two triggers to
35//! [`AttestationCaCache::invalidate`] is later-wave work; this module ships
36//! the cache with an `invalidate` a caller can already reach.
37

59/// Cache key: `None` = every MDS-known root (no `allowed_aaguids`
60/// restriction); `Some(sorted, deduped aaguids)` = the filtered subset.
61/// Sorting/deduping in [`cache_key`] makes two policies with the same
62/// allowlist in different orders share one cache entry.
63type CacheKey = Option<Vec<Uuid>>;
64
65fn cache_key(allowed_aaguids: Option<&[Uuid]>) -> CacheKey {
66    allowed_aaguids.map(|ids| {
67        let mut v = ids.to_vec();
68        v.sort_unstable();
69        v.dedup();
70        v
71    })
72}

74/// Process-wide cache of built `AttestationCaList`s, keyed by the
75/// `allowed_aaguids` restriction that produced them. See the module docs for
76/// the invalidation contract.
77#[derive(Default)]
78pub struct AttestationCaCache {
79    entries: Mutex<HashMap<CacheKey, Arc<AttestationCaList>>>,
80}
81
82impl AttestationCaCache {
83    pub fn new() -> Self {
84        Self::default()
85    }
86
87    /// Drop every cached list. Call after an MDS refresh (the set of known
88    /// roots changed) or an attestation-policy update (the
89    /// `allowed_aaguids` restriction changed) — both invalidate the trust
90    /// picture a previously-built list encoded (W2-D3).
91    pub fn invalidate(&self) {
92        self.entries
93            .lock()
94            .unwrap_or_else(|p| p.into_inner())
95            .clear();
96    }
97
98    /// Number of distinct filters currently cached (test/observability
99    /// helper).
100    pub fn len(&self) -> usize {
101        self.entries.lock().unwrap_or_else(|p| p.into_inner()).len()
102    }
103
104    pub fn is_empty(&self) -> bool {
105        self.len() == 0
106    }
107
108    /// Return the cached `AttestationCaList` for `allowed_aaguids`, building
109    /// and caching it on a miss.
110    ///
111    /// `allowed_aaguids: Some(list)` restricts the built list to those
112    /// AAGUIDs (W2-D3: "the cryptographic layer and the policy layer
113    /// agree"); `None` builds a list that accepts every AAGUID MDS has
114    /// metadata for.
115    pub async fn get_or_build<M: AttestationMetadataSource>(
116        &self,
117        metadata: &M,
118        allowed_aaguids: Option<&[Uuid]>,
119    ) -> AxiamResult<Arc<AttestationCaList>> {
120        let key = cache_key(allowed_aaguids);
121        if let Some(hit) = self
122            .entries
123            .lock()
124            .unwrap_or_else(|p| p.into_inner())
125            .get(&key)
126        {
127            return Ok(Arc::clone(hit));
128        }
129
130        let built = Arc::new(build_ca_list(metadata, key.as_deref()).await?);
131        self.entries
132            .lock()
133            .unwrap_or_else(|p| p.into_inner())
134            .insert(key, Arc::clone(&built));
135        Ok(built)
136    }
137}
138
139async fn build_ca_list<M: AttestationMetadataSource>(
140    metadata: &M,
141    allowed_aaguids: Option<&[Uuid]>,
142) -> AxiamResult<AttestationCaList> {
143    let roots = metadata.attestation_roots(allowed_aaguids).await?;
144    let mut builder = AttestationCaListBuilder::new();
145    for root in roots {
146        // A single malformed root (a corrupt DER blob, however it got that
147        // way) must not take down the whole list — skip and log it rather
148        // than failing every registration for every OTHER authenticator.
149        if let Err(e) = builder.insert_device_der(
150            &root.der,
151            root.aaguid,
152            root.description.clone(),
153            BTreeMap::new(),
154        ) {
155            tracing::warn!(
156                aaguid = %root.aaguid,
157                error = %e,
158                "skipping MDS attestation root that failed to parse as X.509 DER"
159            );
160        }
161    }
162    Ok(builder.build())
163}

169/// The outcome of evaluating one stored credential against the *current*
170/// policy (D9). Never causes a mutation — revocation stays the existing
171/// admin credential-delete path (D7/D9: "never auto-revokes").
172#[derive(Debug, Clone, Copy, PartialEq, Eq)]
173pub enum ComplianceStatus {
174    Compliant,
175    NonCompliant(AttestationDenyReason),
176    /// No recorded AAGUID — a credential registered before X3 shipped (D6:
177    /// `aaguid` is `None` for every pre-X3 row). Always reported this way,
178    /// **never** as a violation, regardless of what the current policy says.
179    Unknown,
180}
181
182impl ComplianceStatus {
183    /// `true` only for [`ComplianceStatus::NonCompliant`] — an `Unknown`
184    /// credential is explicitly not a violation (D9).
185    pub fn is_violation(self) -> bool {
186        matches!(self, ComplianceStatus::NonCompliant(_))
187    }
188}
189
190/// Evaluate one credential's stored `aaguid`/`attestation_format` (D6)
191/// against the tenant's *current* [`WebauthnAttestationPolicy`] (D9).
192///
193/// Pure with respect to the policy engine — this only adds the two lookups
194/// `axiam_core::models::webauthn_policy::evaluate` needs (the MDS entry) on
195/// top of it; it does not itself decide anything `evaluate` doesn't already
196/// decide, and it never mutates or deletes anything.
197///
198/// A credential with `aaguid: None` (every pre-X3 row, and any authenticator
199/// that legitimately reports no AAGUID) is always [`ComplianceStatus::Unknown`],
200/// checked **before** the `mode == None` short-circuit below so the ordering
201/// matches D9 literally regardless of the tenant's current mode. For a
202/// present AAGUID, `mode == None` still skips the MDS lookup entirely
203/// (mirroring `evaluate`'s own D8-step-1 guarantee: a `mode: none` tenant
204/// makes no MDS lookups on the registration path either, so the compliance
205/// report should not either).
206pub async fn evaluate_credential_compliance<M: AttestationMetadataSource>(
207    policy: &WebauthnAttestationPolicy,
208    aaguid: Option<Uuid>,
209    attestation_format: Option<&str>,
210    metadata: &M,
211) -> AxiamResult<ComplianceStatus> {
212    let Some(aaguid) = aaguid else {
213        return Ok(ComplianceStatus::Unknown);
214    };
215
216    if policy.mode == AttestationMode::None {
217        return Ok(ComplianceStatus::Compliant);
218    }
219
220    let entry = metadata.get_entry(aaguid).await?;
221    let candidate = AttestationCandidate {
222        aaguid: Some(aaguid),
223        attestation_format,
224        mds_entry: entry.as_ref(),
225    };
226
227    Ok(match evaluate(policy, &candidate) {
228        AttestationDecision::Allow => ComplianceStatus::Compliant,
229        AttestationDecision::Deny(reason) => ComplianceStatus::NonCompliant(reason),
230    })
231}

268    struct MockMetadata {
269        entries: HashMap<Uuid, MdsEntry>,

274    impl AttestationMetadataSource for MockMetadata {
275        async fn get_entry(&self, aaguid: Uuid) -> AxiamResult<Option<MdsEntry>> {
276            Ok(self.entries.get(&aaguid).cloned())
277        }
278
279        async fn attestation_roots(
280            &self,
281            allowed_aaguids: Option<&[Uuid]>,
282        ) -> AxiamResult<Vec<AttestationRootMaterial>> {
283            self.attestation_roots_calls.fetch_add(1, Ordering::SeqCst);
284            let roots = match allowed_aaguids {
285                Some(allowed) => self
286                    .roots
287                    .iter()
288                    .filter(|r| allowed.contains(&r.aaguid))
289                    .cloned()
290                    .collect(),
291                None => self.roots.clone(),
292            };
293            Ok(roots)
294        }
295    }
