1//! Where on a request an access token travels, decided once.
2//!
3//! Two readers look for the caller's access token: the audit middleware (to name
4//! the actor in the request row) and the REST extractors (to authenticate the
5//! request). They used to apply opposite precedence — the middleware read the
6//! `Authorization` header and ignored the cookie whenever a header existed, the
7//! extractors read the cookie and ignored the header — so a request carrying a
8//! valid `axiam_access` cookie and an unrelated header (`Authorization: Basic …`,
9//! a proxy's own credential) was admitted as the cookie's user and recorded as
10//! `System` with a nil tenant (#751, R111-01). Both now call
11//! [`access_token_carrier`].
12//!
13//! The helper takes the two raw header values rather than a request, because
14//! this crate does not depend on actix; each caller reads its own request type.
15
16/// The channel an access token was found in.
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub enum Carrier {
19    /// The `axiam_access` cookie (browser clients, the admin UI).
20    Cookie,
21    /// `Authorization: Bearer <token>`.
22    Bearer,
23    /// `Authorization: DPoP <token>` (RFC 9449 §7.1).
24    Dpop,
25}
26
27/// Name of the cookie that carries the access token.
28pub const ACCESS_COOKIE: &str = "axiam_access";
29
30/// Pick the token a request presents: the `axiam_access` cookie first, then an
31/// `Authorization` header with a `Bearer` or `DPoP` scheme.
32///
33/// Returns `None` when neither carries a candidate — an absent or empty cookie,
34/// and a header with another scheme (`Basic`), no scheme, or no credentials. A
35/// header that is not a token carrier therefore never shadows the cookie. The
36/// candidate is not verified here; the caller validates it.
37pub fn access_token_carrier<'a>(
38    cookie: Option<&'a str>,
39    authorization: Option<&'a str>,
40) -> Option<(Carrier, &'a str)> {
41    if let Some(value) = cookie.map(str::trim).filter(|v| !v.is_empty()) {
42        return Some((Carrier::Cookie, value));
43    }
44    let header = authorization?.trim();
45    let mut parts = header.splitn(2, char::is_whitespace);
46    let scheme = parts.next().unwrap_or("");
47    let credentials = parts.next().unwrap_or("").trim();
48    if credentials.is_empty() {
49        return None;
50    }
51    // Case-insensitive scheme, RFC 9110 §11.1.
52    if scheme.eq_ignore_ascii_case("bearer") {
53        Some((Carrier::Bearer, credentials))
54    } else if scheme.eq_ignore_ascii_case("dpop") {
55        Some((Carrier::Dpop, credentials))
56    } else {
57        None
58    }
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64
65    #[test]
66    fn cookie_wins_over_any_header() {
67        assert_eq!(
68            access_token_carrier(Some("c"), Some("Bearer h")),
69            Some((Carrier::Cookie, "c"))
70        );
71        assert_eq!(
72            access_token_carrier(Some("c"), Some("Basic eDp5")),
73            Some((Carrier::Cookie, "c"))
74        );
75    }
76
77    #[test]
78    fn header_schemes_are_case_insensitive() {
79        assert_eq!(
80            access_token_carrier(None, Some("bEaReR  t")),
81            Some((Carrier::Bearer, "t"))
82        );
83        assert_eq!(
84            access_token_carrier(None, Some("dpop t")),
85            Some((Carrier::Dpop, "t"))
86        );
87    }
88
89    #[test]
90    fn non_carriers_yield_nothing() {
91        assert_eq!(access_token_carrier(None, Some("Basic eDp5")), None);
92        assert_eq!(access_token_carrier(None, Some("Bearer   ")), None);
93        assert_eq!(access_token_carrier(None, Some("Bearer")), None);
94        assert_eq!(access_token_carrier(Some("  "), None), None);
95        assert_eq!(access_token_carrier(None, None), None);
96    }
97
98    #[test]
99    fn empty_cookie_falls_through_to_the_header() {
100        assert_eq!(
101            access_token_carrier(Some(""), Some("Bearer h")),
102            Some((Carrier::Bearer, "h"))
103        );
104    }
105}

107/// What the session extractor decided about a request's access token, left in
108/// the request's extensions for the audit middleware to read after the handler
109/// (#758 (c), D-12).
110///
111/// The middleware verifies a token's signature and expiry to name the actor in
112/// the request row; it does not apply the extractor's further checks (audience,
113/// session, `client_id`-minted refusal, DPoP). `authenticated` on the row used
114/// to mean only "the signature checked out", so a device-grant token refused
115/// `403` at `/api/v1/device/verify` was recorded `authenticated: true`. It now
116/// means the extractor's verdict wherever the extractor ran. Lives here, in the
117/// lowest crate the middleware and the extractors both reach.
118#[derive(Debug, Clone, Copy, PartialEq, Eq)]
119pub enum AuthVerdict {
120    /// The extractor admitted the credential (the active-tenant decision that
121    /// follows is about what the principal asked for, not about the token).
122    Admitted,
123    /// The extractor refused the credential itself (`401`/`403`).
124    Refused,
125}
