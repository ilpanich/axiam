13/// Configuration for the authentication service.
14#[derive(Debug, Clone, Deserialize)]
15#[serde(default)]
16pub struct AuthConfig {
17    /// PEM-encoded Ed25519 private key for JWT signing.
18    pub jwt_private_key_pem: String,
19    /// PEM-encoded Ed25519 public key for JWT verification.
20    pub jwt_public_key_pem: String,
21    /// Access token lifetime in seconds (default: 900 = 15 minutes).
22    pub access_token_lifetime_secs: u64,
23    /// Refresh token lifetime in seconds (default: 2_592_000 = 30 days).
24    pub refresh_token_lifetime_secs: u64,
25    /// Authorization code lifetime in seconds (default: 600 = 10 minutes).
26    pub auth_code_lifetime_secs: u64,
27    /// JWT issuer (`iss` claim).
28    pub jwt_issuer: String,
29    /// OIDC issuer base URL (e.g. "https://auth.example.com"). Used for
30    /// OIDC discovery endpoint URLs. Falls back to `jwt_issuer` if unset.
31    pub oauth2_issuer_url: String,
32    /// Optional pepper prepended to passwords before Argon2id verification.
33    /// Wrapped in `SecretString` so `Debug`/logging never leaks it by
34    /// accident (SECHRD-12) — exposed only via `.expose_secret()` at the
35    /// `&str` boundary where a pepper value is consumed.
36    pub pepper: Option<SecretString>,
37    /// Previous pepper, kept **verify-only** while a pepper rotation drains
38    /// (`AXIAM__AUTH__PEPPER_PREVIOUS`, §13.4 observation 3).
39    ///
40    /// Client-secret hashes are keyed by the pepper, and the `v2.hs256$` tag
41    /// versions the *algorithm*, not the *key* — so without this, rotating
42    /// `AXIAM__AUTH__PEPPER` permanently invalidates **every** client secret and
43    /// every service account and OAuth2 client has to be re-issued in lockstep
44    /// with the restart.
45    ///
46    /// Set this to the outgoing value for the duration of a rotation. Secrets
47    /// hashed under it still verify, and each one is transparently rewritten
48    /// under the new pepper the first time its owner authenticates, so the
49    /// rotation drains itself. Unset it once the fleet has drained; nothing is
50    /// ever *written* under this key.
51    ///
52    /// It does **not** apply to Argon2id password peppering, which has its own
53    /// migration story — only to client-secret hashing.
54    pub pepper_previous: Option<SecretString>,
55    /// Minimum password length for policy enforcement.
56    pub min_password_length: usize,
57    /// 256-bit AES-GCM key for encrypting TOTP secrets at rest.
58    /// `None` disables MFA enrollment. Set programmatically (not from config files).
59    #[serde(skip)]
60    pub mfa_encryption_key: Option<[u8; 32]>,
61    /// 256-bit AES-GCM key for encrypting federation client secrets at rest.
62    /// `None` means federation config create/update will fail at runtime.
63    /// Set programmatically from `AXIAM__AUTH__FEDERATION_ENCRYPTION_KEY` (not from
64    /// config files). Federation is optional — absence is warned, not fatal.
65    #[serde(skip)]
66    pub federation_encryption_key: Option<[u8; 32]>,
67    /// 256-bit AES-GCM key that seals an OPAQUE exchange's server state
68    /// between its two messages, on both the login and the registration path.
69    ///
70    /// `None` means the OPAQUE endpoints answer `503`, whatever the org or
71    /// tenant policy says. That is deliberate: silently falling back to
72    /// password login when the key is missing would turn a misconfiguration
73    /// into an undetectable downgrade of a security control an operator
74    /// believes is on.
75    ///
76    /// Deliberately its own key rather than a reuse of
77    /// [`Self::mfa_encryption_key`]: the two rotate on different schedules, and
78    /// sharing one would couple an OPAQUE outage to an unrelated TOTP
79    /// rotation. Set from `AXIAM__AUTH__OPAQUE_SESSION_KEY` (not from config
80    /// files).
81    ///
82    /// Rotating this is cheap — it invalidates exchanges in flight and nothing
83    /// else. Contrast [`Self::opaque_setup_key`], which is the expensive one.
84    #[serde(skip)]
85    pub opaque_session_key: Option<[u8; 32]>,
86    /// 256-bit AES-GCM key that encrypts each tenant's OPAQUE server key
87    /// material — the OPRF seed and the long-term AKE key pair — at rest.
88    ///
89    /// `None` means the OPAQUE endpoints answer `503`, for the same reason as
90    /// above.
91    ///
92    /// This is a **separate key from [`Self::opaque_session_key`] on purpose**,
93    /// and the reason is what rotating each one costs. The session key seals
94    /// 120 seconds of in-flight state and can be rotated on any schedule.
95    /// This key protects data at rest: rotating it means re-encrypting every
96    /// `opaque_server_setup` row, and **losing** it means every registration
97    /// record in every tenant becomes unopenable and the whole estate needs a
98    /// password reset. Sharing one key would put the cheap rotation and the
99    /// catastrophic one on the same schedule, which is how an operator ends up
100    /// doing neither. Set from `AXIAM__AUTH__OPAQUE_SETUP_KEY`.
101    #[serde(skip)]
102    pub opaque_setup_key: Option<[u8; 32]>,
103    /// When `true`, access tokens decoded without an `aud` claim are treated as
104    /// `axiam:user`. Enables a back-compat window during the Phase 4 rollout
105    /// while pre-Phase-4 tokens are still circulating. Default: `true`.
106    #[serde(default = "default_true")]
107    pub allow_missing_aud_as_user: bool,
108    /// When `true` (default), all auth cookies are marked `Secure` and are
109    /// therefore sent only over HTTPS. Set `AXIAM__AUTH__COOKIE_SECURE=false`
110    /// **only** in local HTTP development (e.g. http://localhost) — **never**
111    /// in production or staging (D-18).
112    #[serde(default = "default_true")]
113    pub cookie_secure: bool,
114    /// MFA challenge token lifetime in seconds (default: 300 = 5 minutes).
115    pub mfa_challenge_lifetime_secs: u64,
116    /// Issuer name shown in authenticator apps.
117    pub totp_issuer: String,
118    /// Max consecutive failed login attempts before lockout (default: 5).
119    pub max_failed_login_attempts: u32,
120    /// Initial lockout duration in seconds (default: 900 = 15 min).
121    pub lockout_duration_secs: u64,
122    /// Exponential backoff multiplier for repeated lockouts (default: 2.0).
123    pub lockout_backoff_multiplier: f64,
124    /// Maximum lockout duration in seconds (default: 3600 = 1 hour).
125    pub max_lockout_duration_secs: u64,
126    /// Grace period in hours during which PendingVerification users
127    /// can still log in (default: 24). Set to 0 to disable.
128    pub email_verification_grace_period_hours: u32,
129    /// Password reset token expiry in hours (default: 1).
130    pub password_reset_token_expiry_hours: u32,
131    /// WebAuthn Relying Party ID (typically the domain name,
132    /// e.g. "auth.example.com").
133    pub webauthn_rp_id: String,
134    /// WebAuthn Relying Party origin
135    /// (e.g. "https://auth.example.com").
136    pub webauthn_rp_origin: String,
137    /// WebAuthn Relying Party display name.
138    pub webauthn_rp_name: String,
139    /// CQ-B14: Pre-parsed Ed25519 signing key. Populated once at startup via
140    /// `resolve_keys()`. When `Some`, token-issue functions skip PEM re-parsing.
141    /// When `None`, they fall back to parsing from `jwt_private_key_pem`.
142    #[serde(skip)]
143    pub jwt_encoding_key: Option<Arc<EncodingKey>>,
144    /// CQ-B14: Pre-parsed Ed25519 verification key. Populated once at startup via
145    /// `resolve_keys()`. When `Some`, token-verify functions skip PEM re-parsing.
146    /// When `None`, they fall back to parsing from `jwt_public_key_pem`.
147    #[serde(skip)]
148    pub jwt_decoding_key: Option<Arc<DecodingKey>>,
149    /// PERF-01: Consecutive HIBP failures/timeouts before the process-wide
150    /// `HibpBreaker` trips open (default: 5). Overridable via
151    /// `AXIAM__AUTH__HIBP_BREAKER_THRESHOLD`.
152    pub hibp_breaker_threshold: u32,
153    /// PERF-01: Cooldown in seconds the `HibpBreaker` stays open before
154    /// allowing a half-open probe request (default: 30). Overridable via
155    /// `AXIAM__AUTH__HIBP_BREAKER_COOLDOWN_SECS`.
156    pub hibp_breaker_cooldown_secs: u64,
157    /// B1: Maximum number of Argon2id hash/verify operations allowed to run
158    /// concurrently, enforced by the process-wide `crypto_semaphore` shared
159    /// across the login, password-change, password-reset and PKI crypto paths.
160    ///
161    /// Each in-flight Argon2id operation allocates a ~19 MiB memory arena
162    /// (OWASP params m=19456), so unbounded concurrency is an unauthenticated
163    /// memory-DoS vector: the login benchmark pegged 2 cores and reached
164    /// ~970 MiB RSS ≈ 50 concurrent × 19 MiB arenas, against a 1024 MiB
165    /// container cap. This permit count bounds peak concurrent arenas.
166    ///
167    /// `0` means "auto" and resolves to `min(available_parallelism, 4)` at
168    /// semaphore-construction time (see [`AuthConfig::resolved_max_concurrent_hashes`]).
169    /// Override via `AXIAM__AUTH__MAX_CONCURRENT_HASHES`.
170    pub max_concurrent_hashes: usize,
171    /// B1: How long (in seconds) a request waits to acquire a crypto-hash
172    /// permit before giving up and returning a `ServiceUnavailable` (HTTP 503)
173    /// backpressure error, instead of queueing unboundedly and adding to
174    /// tail latency. Default: 5. Override via
175    /// `AXIAM__AUTH__HASH_ACQUIRE_TIMEOUT_SECS`.
176    pub hash_acquire_timeout_secs: u64,
177    /// I6: TTL, in seconds, of the process-local **session-validation cache**.
178    ///
179    /// `0` (the default) disables the cache entirely — every authenticated
180    /// request performs its own SurrealDB read to confirm the session behind the
181    /// access token's `jti` still exists (D-15 / REQ-7). A non-zero value lets
182    /// repeated requests inside the window reuse the last positive answer.
183    ///
184    /// Override via `AXIAM__AUTH__SESSION_VALIDATION_CACHE_TTL_SECS`.
185    ///
186    /// # What it buys and what it costs
187    ///
188    /// The read it removes is the reason the D7 decision cache helped gRPC
189    /// authorization checks 13.1× but REST checks only 5% in benchmark run 4:
190    /// the decision cache elides the *authorization* round-trips, this one
191    /// elides the *authentication* round-trip, and the gRPC surface never had
192    /// the latter.
193    ///
194    /// The cost is a bounded revocation window. Every session-deleting path in
195    /// `SurrealSessionRepository` invalidates the cache in the same call, so on
196    /// a **single replica** a logout takes effect immediately. Across **two or
197    /// more replicas** there is no invalidation channel, so a session revoked on
198    /// replica A stays acceptable on replicas B…N for at most this many
199    /// seconds — the same trade-off, and the same reasoning, as
200    /// `AXIAM__AUTHZ__DECISION_CACHE_TTL_SECS`. Session *expiry* is not
201    /// affected: cached entries carry the row's own `expires_at` and are
202    /// rejected exactly on time regardless of the TTL.
203    ///
204    /// Suggested starting point when enabling: `5`, matching the decision
205    /// cache's default.
206    pub session_validation_cache_ttl_secs: u64,
207}

209impl AuthConfig {
210    /// Effective issuer for JWT `iss` claims and OIDC discovery.
211    ///
212    /// Returns `oauth2_issuer_url` when set, falling back to
213    /// `jwt_issuer`. Trailing slashes are stripped so that the
214    /// OIDC discovery `issuer` exactly matches token `iss` claims
215    /// (OIDC Core §2 requires an exact string match).
216    pub fn effective_issuer(&self) -> &str {
217        if self.oauth2_issuer_url.is_empty() {
218            self.jwt_issuer.trim_end_matches('/')
219        } else {
220            self.oauth2_issuer_url.trim_end_matches('/')
221        }
222    }
223
224    /// CQ-B14: Parse Ed25519 keys from PEM once and cache in `Arc`.
225    ///
226    /// Call this once at startup after loading config from environment.
227    /// After this returns `Ok(())`, all token functions skip per-call PEM
228    /// parsing and use the cached keys instead.
229    pub fn resolve_keys(&mut self) -> Result<(), String> {
230        let enc = EncodingKey::from_ed_pem(self.jwt_private_key_pem.as_bytes())
231            .map_err(|e| format!("invalid JWT private key PEM: {e}"))?;
232        let dec = DecodingKey::from_ed_pem(self.jwt_public_key_pem.as_bytes())
233            .map_err(|e| format!("invalid JWT public key PEM: {e}"))?;
234        self.jwt_encoding_key = Some(Arc::new(enc));
235        self.jwt_decoding_key = Some(Arc::new(dec));
236        Ok(())
237    }
238
239    /// B1: Resolve `max_concurrent_hashes` (0 = auto) to a concrete permit
240    /// count for the crypto semaphore.
241    ///
242    /// Auto (`0`) resolves to `min(available_parallelism, 4)`: bound by the
243    /// core count so Argon2id (a CPU-bound operation) does not oversubscribe
244    /// cores, and capped at 4 so peak concurrent ~19 MiB arenas stay well
245    /// under typical container memory caps. A non-zero value is used verbatim
246    /// (a `0` after resolution is impossible — it would be a zero-permit
247    /// semaphore that deadlocks all hashing, so it is clamped to 1).
248    pub fn resolved_max_concurrent_hashes(&self) -> usize {
249        match self.max_concurrent_hashes {
250            0 => std::thread::available_parallelism()
251                .map(|n| n.get().min(4))
252                .unwrap_or(4),
253            n => n.max(1),
254        }
255    }
256}
