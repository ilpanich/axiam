13/// Configuration for the authentication service.
14#[derive(Debug, Clone, Deserialize)]
15#[serde(default)]
16pub struct AuthConfig {
17    /// PEM-encoded Ed25519 private key for JWT signing.
18    pub jwt_private_key_pem: String,
19    /// PEM-encoded Ed25519 public key for JWT verification.
20    pub jwt_public_key_pem: String,
21    /// Access token lifetime in seconds (default: 900 = 15 minutes).
22    pub access_token_lifetime_secs: u64,
23    /// Refresh token lifetime in seconds (default: 2_592_000 = 30 days).
24    pub refresh_token_lifetime_secs: u64,
25    /// Authorization code lifetime in seconds (default: 600 = 10 minutes).
26    pub auth_code_lifetime_secs: u64,
27    /// JWT issuer (`iss` claim).
28    pub jwt_issuer: String,
29    /// OIDC issuer base URL (e.g. "https://auth.example.com"). Used for
30    /// OIDC discovery endpoint URLs. Falls back to `jwt_issuer` if unset.
31    pub oauth2_issuer_url: String,
32    /// Optional pepper prepended to passwords before Argon2id verification.
33    /// Wrapped in `SecretString` so `Debug`/logging never leaks it by
34    /// accident (SECHRD-12) — exposed only via `.expose_secret()` at the
35    /// `&str` boundary where a pepper value is consumed.
36    pub pepper: Option<SecretString>,
37    /// Previous pepper, kept **verify-only** while a pepper rotation drains
38    /// (`AXIAM__AUTH__PEPPER_PREVIOUS`, §13.4 observation 3).
39    ///
40    /// Client-secret hashes are keyed by the pepper, and the `v2.hs256$` tag
41    /// versions the *algorithm*, not the *key* — so without this, rotating
42    /// `AXIAM__AUTH__PEPPER` permanently invalidates **every** client secret and
43    /// every service account and OAuth2 client has to be re-issued in lockstep
44    /// with the restart.
45    ///
46    /// Set this to the outgoing value for the duration of a rotation. Secrets
47    /// hashed under it still verify, and each one is transparently rewritten
48    /// under the new pepper the first time its owner authenticates, so the
49    /// rotation drains itself. Unset it once the fleet has drained; nothing is
50    /// ever *written* under this key.
51    ///
52    /// It does **not** apply to Argon2id password peppering, which has its own
53    /// migration story — only to client-secret hashing.
54    pub pepper_previous: Option<SecretString>,
55    /// Minimum password length for policy enforcement.
56    pub min_password_length: usize,
57    /// 256-bit AES-GCM key for encrypting TOTP secrets at rest.
58    /// `None` disables MFA enrollment. Set programmatically (not from config files).
59    #[serde(skip)]
60    pub mfa_encryption_key: Option<[u8; 32]>,
61    /// 256-bit AES-GCM key for encrypting federation client secrets at rest.
62    /// `None` means federation config create/update will fail at runtime.
63    /// Set programmatically from `AXIAM__AUTH__FEDERATION_ENCRYPTION_KEY` (not from
64    /// config files). Federation is optional — absence is warned, not fatal.
65    #[serde(skip)]
66    pub federation_encryption_key: Option<[u8; 32]>,
67    /// 256-bit AES-GCM key that seals the SRP exchange's server state between
68    /// `/auth/srp/challenge` and `/auth/srp/verify`.
69    ///
70    /// `None` means the SRP endpoints answer `503`, whatever the org or tenant
71    /// policy says. That is deliberate: silently falling back to password login
72    /// when the key is missing would turn a misconfiguration into an
73    /// undetectable downgrade of a security control an operator believes is on.
74    ///
75    /// Deliberately its own key rather than a reuse of
76    /// [`Self::mfa_encryption_key`]: the two rotate on different schedules, and
77    /// sharing one would couple an SRP outage to an unrelated TOTP rotation.
78    /// Set from `AXIAM__AUTH__SRP_SESSION_KEY` (not from config files).
79    #[serde(skip)]
80    pub srp_session_key: Option<[u8; 32]>,
81    /// When `true`, access tokens decoded without an `aud` claim are treated as
82    /// `axiam:user`. Enables a back-compat window during the Phase 4 rollout
83    /// while pre-Phase-4 tokens are still circulating. Default: `true`.
84    #[serde(default = "default_true")]
85    pub allow_missing_aud_as_user: bool,
86    /// When `true` (default), all auth cookies are marked `Secure` and are
87    /// therefore sent only over HTTPS. Set `AXIAM__AUTH__COOKIE_SECURE=false`
88    /// **only** in local HTTP development (e.g. http://localhost) — **never**
89    /// in production or staging (D-18).
90    #[serde(default = "default_true")]
91    pub cookie_secure: bool,
92    /// MFA challenge token lifetime in seconds (default: 300 = 5 minutes).
93    pub mfa_challenge_lifetime_secs: u64,
94    /// Issuer name shown in authenticator apps.
95    pub totp_issuer: String,
96    /// Max consecutive failed login attempts before lockout (default: 5).
97    pub max_failed_login_attempts: u32,
98    /// Initial lockout duration in seconds (default: 900 = 15 min).
99    pub lockout_duration_secs: u64,
100    /// Exponential backoff multiplier for repeated lockouts (default: 2.0).
101    pub lockout_backoff_multiplier: f64,
102    /// Maximum lockout duration in seconds (default: 3600 = 1 hour).
103    pub max_lockout_duration_secs: u64,
104    /// Grace period in hours during which PendingVerification users
105    /// can still log in (default: 24). Set to 0 to disable.
106    pub email_verification_grace_period_hours: u32,
107    /// Password reset token expiry in hours (default: 1).
108    pub password_reset_token_expiry_hours: u32,
109    /// WebAuthn Relying Party ID (typically the domain name,
110    /// e.g. "auth.example.com").
111    pub webauthn_rp_id: String,
112    /// WebAuthn Relying Party origin
113    /// (e.g. "https://auth.example.com").
114    pub webauthn_rp_origin: String,
115    /// WebAuthn Relying Party display name.
116    pub webauthn_rp_name: String,
117    /// CQ-B14: Pre-parsed Ed25519 signing key. Populated once at startup via
118    /// `resolve_keys()`. When `Some`, token-issue functions skip PEM re-parsing.
119    /// When `None`, they fall back to parsing from `jwt_private_key_pem`.
120    #[serde(skip)]
121    pub jwt_encoding_key: Option<Arc<EncodingKey>>,
122    /// CQ-B14: Pre-parsed Ed25519 verification key. Populated once at startup via
123    /// `resolve_keys()`. When `Some`, token-verify functions skip PEM re-parsing.
124    /// When `None`, they fall back to parsing from `jwt_public_key_pem`.
125    #[serde(skip)]
126    pub jwt_decoding_key: Option<Arc<DecodingKey>>,
127    /// PERF-01: Consecutive HIBP failures/timeouts before the process-wide
128    /// `HibpBreaker` trips open (default: 5). Overridable via
129    /// `AXIAM__AUTH__HIBP_BREAKER_THRESHOLD`.
130    pub hibp_breaker_threshold: u32,
131    /// PERF-01: Cooldown in seconds the `HibpBreaker` stays open before
132    /// allowing a half-open probe request (default: 30). Overridable via
133    /// `AXIAM__AUTH__HIBP_BREAKER_COOLDOWN_SECS`.
134    pub hibp_breaker_cooldown_secs: u64,
135    /// B1: Maximum number of Argon2id hash/verify operations allowed to run
136    /// concurrently, enforced by the process-wide `crypto_semaphore` shared
137    /// across the login, password-change, password-reset and PKI crypto paths.
138    ///
139    /// Each in-flight Argon2id operation allocates a ~19 MiB memory arena
140    /// (OWASP params m=19456), so unbounded concurrency is an unauthenticated
141    /// memory-DoS vector: the login benchmark pegged 2 cores and reached
142    /// ~970 MiB RSS ≈ 50 concurrent × 19 MiB arenas, against a 1024 MiB
143    /// container cap. This permit count bounds peak concurrent arenas.
144    ///
145    /// `0` means "auto" and resolves to `min(available_parallelism, 4)` at
146    /// semaphore-construction time (see [`AuthConfig::resolved_max_concurrent_hashes`]).
147    /// Override via `AXIAM__AUTH__MAX_CONCURRENT_HASHES`.
148    pub max_concurrent_hashes: usize,
149    /// B1: How long (in seconds) a request waits to acquire a crypto-hash
150    /// permit before giving up and returning a `ServiceUnavailable` (HTTP 503)
151    /// backpressure error, instead of queueing unboundedly and adding to
152    /// tail latency. Default: 5. Override via
153    /// `AXIAM__AUTH__HASH_ACQUIRE_TIMEOUT_SECS`.
154    pub hash_acquire_timeout_secs: u64,
155    /// I6: TTL, in seconds, of the process-local **session-validation cache**.
156    ///
157    /// `0` (the default) disables the cache entirely — every authenticated
158    /// request performs its own SurrealDB read to confirm the session behind the
159    /// access token's `jti` still exists (D-15 / REQ-7). A non-zero value lets
160    /// repeated requests inside the window reuse the last positive answer.
161    ///
162    /// Override via `AXIAM__AUTH__SESSION_VALIDATION_CACHE_TTL_SECS`.
163    ///
164    /// # What it buys and what it costs
165    ///
166    /// The read it removes is the reason the D7 decision cache helped gRPC
167    /// authorization checks 13.1× but REST checks only 5% in benchmark run 4:
168    /// the decision cache elides the *authorization* round-trips, this one
169    /// elides the *authentication* round-trip, and the gRPC surface never had
170    /// the latter.
171    ///
172    /// The cost is a bounded revocation window. Every session-deleting path in
173    /// `SurrealSessionRepository` invalidates the cache in the same call, so on
174    /// a **single replica** a logout takes effect immediately. Across **two or
175    /// more replicas** there is no invalidation channel, so a session revoked on
176    /// replica A stays acceptable on replicas B…N for at most this many
177    /// seconds — the same trade-off, and the same reasoning, as
178    /// `AXIAM__AUTHZ__DECISION_CACHE_TTL_SECS`. Session *expiry* is not
179    /// affected: cached entries carry the row's own `expires_at` and are
180    /// rejected exactly on time regardless of the TTL.
181    ///
182    /// Suggested starting point when enabling: `5`, matching the decision
183    /// cache's default.
184    pub session_validation_cache_ttl_secs: u64,
185}

187impl AuthConfig {
188    /// Effective issuer for JWT `iss` claims and OIDC discovery.
189    ///
190    /// Returns `oauth2_issuer_url` when set, falling back to
191    /// `jwt_issuer`. Trailing slashes are stripped so that the
192    /// OIDC discovery `issuer` exactly matches token `iss` claims
193    /// (OIDC Core §2 requires an exact string match).
194    pub fn effective_issuer(&self) -> &str {
195        if self.oauth2_issuer_url.is_empty() {
196            self.jwt_issuer.trim_end_matches('/')
197        } else {
198            self.oauth2_issuer_url.trim_end_matches('/')
199        }
200    }
201
202    /// CQ-B14: Parse Ed25519 keys from PEM once and cache in `Arc`.
203    ///
204    /// Call this once at startup after loading config from environment.
205    /// After this returns `Ok(())`, all token functions skip per-call PEM
206    /// parsing and use the cached keys instead.
207    pub fn resolve_keys(&mut self) -> Result<(), String> {
208        let enc = EncodingKey::from_ed_pem(self.jwt_private_key_pem.as_bytes())
209            .map_err(|e| format!("invalid JWT private key PEM: {e}"))?;
210        let dec = DecodingKey::from_ed_pem(self.jwt_public_key_pem.as_bytes())
211            .map_err(|e| format!("invalid JWT public key PEM: {e}"))?;
212        self.jwt_encoding_key = Some(Arc::new(enc));
213        self.jwt_decoding_key = Some(Arc::new(dec));
214        Ok(())
215    }
216
217    /// B1: Resolve `max_concurrent_hashes` (0 = auto) to a concrete permit
218    /// count for the crypto semaphore.
219    ///
220    /// Auto (`0`) resolves to `min(available_parallelism, 4)`: bound by the
221    /// core count so Argon2id (a CPU-bound operation) does not oversubscribe
222    /// cores, and capped at 4 so peak concurrent ~19 MiB arenas stay well
223    /// under typical container memory caps. A non-zero value is used verbatim
224    /// (a `0` after resolution is impossible — it would be a zero-permit
225    /// semaphore that deadlocks all hashing, so it is clamped to 1).
226    pub fn resolved_max_concurrent_hashes(&self) -> usize {
227        match self.max_concurrent_hashes {
228            0 => std::thread::available_parallelism()
229                .map(|n| n.get().min(4))
230                .unwrap_or(4),
231            n => n.max(1),
232        }
233    }
234}
