13/// What is wrong with `AXIAM__AUTH__OAUTH2_DEFAULT_TENANT_ID`, in terms that
14/// name no part of the value (T-244).
15///
16/// Rendered into one boot-time `WARN` by the composition root. It describes
17/// the value's **shape** — how long it is, and whether its characters could
18/// belong to a UUID at all — because those two facts distinguish the mistakes
19/// an operator actually makes (a truncated paste, a tenant *slug* where an id
20/// belongs, a different identifier entirely) without echoing anything.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub struct DefaultTenantProblem {
23    /// Character count of the trimmed value. A UUID is 36.
24    pub length: usize,
25    /// A short phrase describing the character class.
26    pub shape: &'static str,
27}
28
29impl std::fmt::Display for DefaultTenantProblem {
30    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
31        write!(f, "{} characters, {}", self.length, self.shape)
32    }
33}

35/// The path segment that separates a deployment's root issuer from a tenant
36/// identifier under the T21.6 per-tenant issuer form: `{root}/t/{tenant_id}`.
37///
38/// One constant rather than three string literals, because the value appears
39/// in an issuer this module builds, in the routes `axiam-api-rest` mounts and
40/// in the `PUBLIC_PATHS` entries that must agree with them. A deployment that
41/// built issuers with one spelling and routed the other would publish a
42/// discovery document naming endpoints it does not serve.
43///
44/// `/t/` rather than `/tenants/` or `/realms/`: it is short enough that the
45/// three RFC 8414 §3 discovery forms stay readable, and it collides with no
46/// path AXIAM already serves.
47pub const TENANT_PATH_PREFIX: &str = "/t/";

49/// Configuration for the authentication service.
50#[derive(Debug, Clone, Deserialize)]
51#[serde(default)]
52pub struct AuthConfig {
53    /// PEM-encoded Ed25519 private key for JWT signing.
54    pub jwt_private_key_pem: String,
55    /// PEM-encoded Ed25519 public key for JWT verification.
56    pub jwt_public_key_pem: String,
57    /// Trust an `X-Client-Certificate` request header as proof of device
58    /// identity when the connection carries no TLS-verified client
59    /// certificate. Default **`false`**.
60    ///
61    /// # Why this is off by default
62    ///
63    /// The header carries a PEM certificate, and a certificate is public data.
64    /// `DeviceAuthService::authenticate` checks the fingerprint, the status,
65    /// the expiry and the chain up to the tenant or organization CA — every one
66    /// of which a *copy* of an enrolled device's certificate also satisfies.
67    /// Nothing in that path proves the sender holds the private key, because
68    /// nothing can: possession is proven by the TLS handshake, and on this path
69    /// there was no handshake to prove it in.
70    ///
71    /// That is sound only while the header cannot originate with the client —
72    /// i.e. a reverse proxy performs the mTLS handshake itself and
73    /// unconditionally overwrites the header on every request it forwards.
74    /// It stops being sound the moment the server is reachable by anything but
75    /// that proxy, where anyone holding a copy of a device certificate
76    /// authenticates as that device.
77    ///
78    /// Native mTLS is unaffected and always preferred: when rustls verified a
79    /// client certificate on the connection, that certificate is authoritative
80    /// and this setting is not consulted.
81    ///
82    /// Set it to `true` only if all of the following hold, and say so in your
83    /// runbook:
84    ///
85    /// 1. TLS (and the client-certificate handshake) terminates at a proxy you
86    ///    operate;
87    /// 2. that proxy sets `X-Client-Certificate` from the certificate **it**
88    ///    verified, on every request, overwriting anything the client sent;
89    /// 3. nothing else can reach this server's listener.
90    pub trust_forwarded_client_cert: bool,
91    /// Access token lifetime in seconds (default: 900 = 15 minutes).
92    pub access_token_lifetime_secs: u64,
93    /// Refresh token lifetime in seconds (default: 2_592_000 = 30 days).
94    pub refresh_token_lifetime_secs: u64,
95    /// Authorization code lifetime in seconds (default: 600 = 10 minutes).
96    pub auth_code_lifetime_secs: u64,
97    /// JWT issuer (`iss` claim).
98    pub jwt_issuer: String,
99    /// OIDC issuer base URL (e.g. "https://auth.example.com"). Used for
100    /// OIDC discovery endpoint URLs. Falls back to `jwt_issuer` if unset.
101    pub oauth2_issuer_url: String,
102    /// Base URL of the listener that performs the mutual-TLS handshake, when
103    /// that is a different host or port from
104    /// [`effective_issuer`](Self::effective_issuer)
105    /// (`AXIAM__AUTH__OAUTH2_MTLS_BASE_URL`). Empty by default.
106    ///
107    /// Setting it makes the discovery document carry RFC 8705 §5
108    /// `mtls_endpoint_aliases`; leaving it empty omits the member entirely,
109    /// which is what the RFC asks of a server with nothing to alias.
110    ///
111    /// # Why a separate URL is needed at all
112    ///
113    /// A TLS listener decides whether to request a client certificate during
114    /// the handshake, before it has seen a single byte of HTTP. So "ask for a
115    /// certificate on `/oauth2/token` but not on `/oauth2/authorize`" is not a
116    /// thing a listener can do — the choice is per-listener, and a deployment
117    /// that wants both a browser-facing authorization endpoint and a
118    /// certificate-authenticated token endpoint has to run two of them.
119    ///
120    /// Which leaves the client with a question metadata alone could not answer
121    /// before RFC 8705 §5: the discovery document names one `token_endpoint`,
122    /// and a client doing mTLS needs the *other* one. The aliases are that
123    /// answer — the same endpoints, re-based on the host that will ask for a
124    /// certificate.
125    ///
126    /// # When to leave it empty
127    ///
128    /// Two deployments should, and neither is unusual:
129    ///
130    /// - **`client_auth = optional` on a single listener.** rustls requests a
131    ///   certificate and accepts a connection without one, so the conventional
132    ///   endpoints already serve both populations and there is no second URL
133    ///   to point at. This is the shape `scripts/e2e-mtls-native-check.sh`
134    ///   exercises.
135    /// - **`client_auth = required` on a single listener.** Every endpoint is
136    ///   already behind the handshake; an alias would name the URL the client
137    ///   is using.
138    ///
139    /// Set it only when mTLS genuinely terminates somewhere else — typically a
140    /// second proxy hostname such as `https://mtls.iam.example.com` in front of
141    /// the same server.
142    ///
143    /// # What it must be
144    ///
145    /// An absolute URL, validated where the document is built. A value that is
146    /// not parseable fails the discovery request rather than being dropped:
147    /// silently omitting the aliases would send an mTLS client to the
148    /// conventional endpoints, which is the one outcome this setting exists to
149    /// prevent.
150    pub oauth2_mtls_base_url: String,
151    /// The tenant a discovery document describes when the caller named none
152    /// (`AXIAM__AUTH__OAUTH2_DEFAULT_TENANT_ID`). Empty by default.
153    ///
154    /// # What it fixes
155    ///
156    /// Every OAuth2 endpoint that authenticates a *client* — token, PAR,
157    /// introspection, revocation, device authorization, end-session — takes a
158    /// **required** `tenant_id` query parameter, and `/oauth2/authorize` needs
159    /// one for any request without a principal, which is every browser. The
160    /// discovery document published none of them, so a relying party that did
161    /// exactly what OIDC Discovery tells it to do — read the document, use the
162    /// URLs — got `400 missing field tenant_id` at the token endpoint. The
163    /// first OpenID Foundation conformance run could not complete a single
164    /// authorization for this reason.
165    ///
166    /// Setting this makes the **bare** document publish endpoint URLs carrying
167    /// the tenant, which is the deployment shape an OP is certified as: one
168    /// issuer, one tenant. A document fetched with an explicit `?tenant_id=`
169    /// carries that tenant instead, and this value is not consulted.
170    ///
171    /// # What it does NOT do
172    ///
173    /// It changes no endpoint's behaviour. A request that arrives without
174    /// `tenant_id` is refused exactly as it is today — this is a statement in a
175    /// document, not a fallback in a handler. That distinction is deliberate:
176    /// a default applied at the endpoint would silently give an
177    /// unparameterised request a tenant, and on a multi-tenant authorization
178    /// server the tenant is the isolation boundary. Leaving it empty is
179    /// therefore safe and is the correct setting for a deployment that serves
180    /// many tenants from one issuer.
181    pub oauth2_default_tenant_id: String,
182    /// T21.6 — whether this deployment serves the per-tenant **path** issuer
183    /// form `{root}/t/{tenant_id}` (`AXIAM__AUTH__TENANT_ISSUER_PATHS`).
184    ///
185    /// `false` by default, and with it false nothing about this deployment
186    /// changes: the `/t/{tenant_id}` scope and the three tenant discovery
187    /// routes are not mounted, no token can carry a tenant issuer, and
188    /// [`Self::accepts_issuer`] answers exactly what the pinned-issuer check
189    /// answered before this field existed.
190    ///
191    /// # What it is for
192    ///
193    /// RFC 8414 §2 forbids a query component in an issuer identifier, so the
194    /// deployment-wide issuer plus `?tenant_id=` cannot be published as the
195    /// issuer of one tenant. An MCP server naming AXIAM in its RFC 9728
196    /// `authorization_servers` therefore has no way to point a client at
197    /// anything but the default tenant. The path form is an issuer a client
198    /// can turn into a discovery URL by the RFC 8414 §3 rule with no query at
199    /// all.
200    ///
201    /// The tenant path is **derived, never configured**: a deployment sets a
202    /// root issuer, and `{root}/t/{tenant_id}` follows from it.
203    pub tenant_issuer_paths: bool,
204    /// T21.6 — the issuer the **current request** arrived under, when it
205    /// arrived on a per-tenant path.
206    ///
207    /// `None` on every request that is not on a `/t/{tenant_id}` path, which
208    /// is every request on a deployment with [`Self::tenant_issuer_paths`]
209    /// off. It is set only by cloning a deployment config through
210    /// [`Self::for_tenant_path`], never by configuration — hence `serde(skip)`:
211    /// an operator cannot set a per-request value from the environment, and an
212    /// environment that appeared to offer one would be offering a way to
213    /// forge `iss`.
214    ///
215    /// [`Self::effective_issuer`] answers this when it is set, which is what
216    /// makes every `iss` a request mints — access token, ID token, logout
217    /// token, the RFC 9207 response parameter — the tenant issuer without any
218    /// of the four minting sites learning that tenant paths exist.
219    /// [`Self::root_issuer`] answers the deployment's issuer regardless, and
220    /// is what every **validation** decision reads.
221    #[serde(skip)]
222    pub request_issuer: Option<String>,
223    /// T-39/T-143 — whether this deployment publishes
224    /// `GET /oauth2/revocations`.
225    ///
226    /// `AXIAM__AUTH__REVOCATION_FEED_ENABLED`, default `false`. With it off the
227    /// route is not mounted and no `revoked_session` row is written, so the
228    /// deployment is byte-identical to one built before the feed existed.
229    ///
230    /// The feed narrows the window in which a revoked session's access token
231    /// still verifies — from one token lifetime to one poll interval — for an
232    /// SDK guard that opts into polling it. It is never a control: a guard
233    /// that cannot fetch it behaves exactly as it does today, and the token
234    /// itself still decides.
235    #[serde(default)]
236    pub revocation_feed_enabled: bool,
237    /// Extra browser origins this deployment will hand a **federation SSO
238    /// handoff code** to (`AXIAM__AUTH__SSO_SPA_ORIGINS`; a list, set the same
239    /// way as `AXIAM__SERVER__CORS_ALLOWED_ORIGINS`).
240    ///
241    /// # Why this exists
242    ///
243    /// SAML and Apple return by cross-site form POST, so the session cookies
244    /// cannot be set on that response. AXIAM instead mints a single-use handoff
245    /// code and redirects the browser to the SPA with it — and that code is a
246    /// bearer credential for a session. The redirect target therefore may not
247    /// be whatever the caller of the *unauthenticated* login-start endpoint
248    /// asked for: on those two protocols the identity provider never sees the
249    /// SPA `redirect_uri`, so its own registered-redirect allowlist is not
250    /// there to reject an attacker's host.
251    ///
252    /// The default is empty, which means **the origin of
253    /// [`effective_issuer`](Self::effective_issuer) only**. That is the right
254    /// default rather than a restrictive one: the endpoint a SAML IdP or Apple
255    /// posts to is itself built from `effective_issuer`, so a deployment whose
256    /// SPA is served from that origin — the shape `docker/nginx.conf` and the
257    /// deployment guides describe, and the only shape in which
258    /// `SameSite=Strict` session cookies work same-origin — needs no entry
259    /// here at all.
260    ///
261    /// Set it when the SPA is served from a different host than the API (for
262    /// example `https://app.example.com` against `https://api.example.com`).
263    /// Values are compared as **origins**: scheme, host and port, with the path
264    /// ignored — so `https://app.example.com` does not admit
265    /// `https://app.example.com:8443`.
266    ///
267    /// # Since R-3: all four flows, not two
268    ///
269    /// This used to have no effect on the OIDC and OAuth2 flows, whose
270    /// `redirect_uri` is registered at the provider and checked there. It now
271    /// governs those two as well. The provider's check is a real control, but
272    /// only as strict as each provider's registration hygiene — several accept
273    /// wildcard or prefix registrations — and it is not a control AXIAM owns or
274    /// can inspect; it stays as a second, independent layer behind this one.
275    ///
276    /// **This is the one thing R-3 can break.** A deployment whose SPA lives on
277    /// an origin other than the issuer's, signing in through OIDC or OAuth2
278    /// providers, worked without this variable and now needs it — the same
279    /// requirement the SAML and Apple flows have imposed since beta08. The
280    /// shipped same-origin topology needs nothing. The `400` names this
281    /// variable, so the failure says what to set.
282    pub sso_spa_origins: Vec<String>,
283    /// Optional pepper prepended to passwords before Argon2id verification.
284    /// Wrapped in `SecretString` so `Debug`/logging never leaks it by
285    /// accident (SECHRD-12) — exposed only via `.expose_secret()` at the
286    /// `&str` boundary where a pepper value is consumed.
287    pub pepper: Option<SecretString>,
288    /// Previous pepper, kept **verify-only** while a pepper rotation drains
289    /// (`AXIAM__AUTH__PEPPER_PREVIOUS`, §13.4 observation 3).
290    ///
291    /// Client-secret hashes are keyed by the pepper, and the `v2.hs256$` tag
292    /// versions the *algorithm*, not the *key* — so without this, rotating
293    /// `AXIAM__AUTH__PEPPER` permanently invalidates **every** client secret and
294    /// every service account and OAuth2 client has to be re-issued in lockstep
295    /// with the restart.
296    ///
297    /// Set this to the outgoing value for the duration of a rotation. Secrets
298    /// hashed under it still verify, and each one is transparently rewritten
299    /// under the new pepper the first time its owner authenticates, so the
300    /// rotation drains itself. Unset it once the fleet has drained; nothing is
301    /// ever *written* under this key.
302    ///
303    /// It does **not** apply to Argon2id password peppering, which has its own
304    /// migration story — only to client-secret hashing.
305    pub pepper_previous: Option<SecretString>,
306    /// Minimum password length for policy enforcement.
307    pub min_password_length: usize,
308    /// 256-bit AES-GCM key for encrypting TOTP secrets at rest.
309    /// `None` disables MFA enrollment. Set programmatically (not from config files).
310    #[serde(skip)]
311    pub mfa_encryption_key: Option<[u8; 32]>,
312    /// 256-bit AES-GCM key for encrypting federation client secrets at rest.
313    /// `None` means federation config create/update will fail at runtime.
314    /// Set programmatically from `AXIAM__AUTH__FEDERATION_ENCRYPTION_KEY` (not from
315    /// config files). Federation is optional — absence is warned, not fatal.
316    #[serde(skip)]
317    pub federation_encryption_key: Option<[u8; 32]>,
318    /// 256-bit AES-GCM key that seals an OPAQUE exchange's server state
319    /// between its two messages, on both the login and the registration path.
320    ///
321    /// `None` means the OPAQUE endpoints answer `503`, whatever the org or
322    /// tenant policy says. That is deliberate: silently falling back to
323    /// password login when the key is missing would turn a misconfiguration
324    /// into an undetectable downgrade of a security control an operator
325    /// believes is on.
326    ///
327    /// Deliberately its own key rather than a reuse of
328    /// [`Self::mfa_encryption_key`]: the two rotate on different schedules, and
329    /// sharing one would couple an OPAQUE outage to an unrelated TOTP
330    /// rotation. Set from `AXIAM__AUTH__OPAQUE_SESSION_KEY` (not from config
331    /// files).
332    ///
333    /// Rotating this is cheap — it invalidates exchanges in flight and nothing
334    /// else. Contrast [`Self::opaque_setup_key`], which is the expensive one.
335    #[serde(skip)]
336    pub opaque_session_key: Option<[u8; 32]>,
337    /// 256-bit AES-GCM key that encrypts each tenant's OPAQUE server key
338    /// material — the OPRF seed and the long-term AKE key pair — at rest.
339    ///
340    /// `None` means the OPAQUE endpoints answer `503`, for the same reason as
341    /// above.
342    ///
343    /// This is a **separate key from [`Self::opaque_session_key`] on purpose**,
344    /// and the reason is what rotating each one costs. The session key seals
345    /// 120 seconds of in-flight state and can be rotated on any schedule.
346    /// This key protects data at rest: rotating it means re-encrypting every
347    /// `opaque_server_setup` row, and **losing** it means every registration
348    /// record in every tenant becomes unopenable and the whole estate needs a
349    /// password reset. Sharing one key would put the cheap rotation and the
350    /// catastrophic one on the same schedule, which is how an operator ends up
351    /// doing neither. Set from `AXIAM__AUTH__OPAQUE_SETUP_KEY`.
352    #[serde(skip)]
353    pub opaque_setup_key: Option<[u8; 32]>,

398    #[serde(skip)]
399    pub jwt_decoding_key: Option<Arc<DecodingKey>>,
400    /// PERF-01: Consecutive HIBP failures/timeouts before the process-wide
401    /// `HibpBreaker` trips open (default: 5). Overridable via
402    /// `AXIAM__AUTH__HIBP_BREAKER_THRESHOLD`.
403    pub hibp_breaker_threshold: u32,
404    /// PERF-01: Cooldown in seconds the `HibpBreaker` stays open before
405    /// allowing a half-open probe request (default: 30). Overridable via
406    /// `AXIAM__AUTH__HIBP_BREAKER_COOLDOWN_SECS`.
407    pub hibp_breaker_cooldown_secs: u64,
408    /// B1: Maximum number of Argon2id hash/verify operations allowed to run
409    /// concurrently, enforced by the process-wide `crypto_semaphore` shared
410    /// across the login, password-change, password-reset and PKI crypto paths.
411    ///
412    /// Each in-flight Argon2id operation allocates a ~19 MiB memory arena
413    /// (OWASP params m=19456), so unbounded concurrency is an unauthenticated
414    /// memory-DoS vector: the login benchmark pegged 2 cores and reached
415    /// ~970 MiB RSS ≈ 50 concurrent × 19 MiB arenas, against a 1024 MiB
416    /// container cap. This permit count bounds peak concurrent arenas.
417    ///
418    /// `0` means "auto" and resolves to `min(available_parallelism, 4)` at
419    /// semaphore-construction time (see [`AuthConfig::resolved_max_concurrent_hashes`]).
420    /// Override via `AXIAM__AUTH__MAX_CONCURRENT_HASHES`.
421    pub max_concurrent_hashes: usize,
422    /// B1: How long (in seconds) a request waits to acquire a crypto-hash
423    /// permit before giving up and returning a `ServiceUnavailable` (HTTP 503)
424    /// backpressure error, instead of queueing unboundedly and adding to
425    /// tail latency. Default: 5. Override via
426    /// `AXIAM__AUTH__HASH_ACQUIRE_TIMEOUT_SECS`.
427    pub hash_acquire_timeout_secs: u64,
428    /// I6: TTL, in seconds, of the process-local **session-validation cache**.
429    ///
430    /// `0` (the default) disables the cache entirely — every authenticated
431    /// request performs its own SurrealDB read to confirm the session behind the
432    /// access token's `jti` still exists (D-15 / REQ-7). A non-zero value lets
433    /// repeated requests inside the window reuse the last positive answer.
434    ///
435    /// Override via `AXIAM__AUTH__SESSION_VALIDATION_CACHE_TTL_SECS`.
436    ///
437    /// # What it buys and what it costs
438    ///
439    /// The read it removes is the reason the D7 decision cache helped gRPC
440    /// authorization checks 13.1× but REST checks only 5% in benchmark run 4:
441    /// the decision cache elides the *authorization* round-trips, this one
442    /// elides the *authentication* round-trip, and the gRPC surface never had
443    /// the latter.
444    ///
445    /// The cost is a bounded revocation window. Every session-deleting path in
446    /// `SurrealSessionRepository` invalidates the cache in the same call, so on
447    /// a **single replica** a logout takes effect immediately. Across **two or
448    /// more replicas** there is no invalidation channel, so a session revoked on
449    /// replica A stays acceptable on replicas B…N for at most this many
450    /// seconds — the same trade-off, and the same reasoning, as
451    /// `AXIAM__AUTHZ__DECISION_CACHE_TTL_SECS`. Session *expiry* is not
452    /// affected: cached entries carry the row's own `expires_at` and are
453    /// rejected exactly on time regardless of the TTL.
454    ///
455    /// Suggested starting point when enabling: `5`, matching the decision
456    /// cache's default.
457    pub session_validation_cache_ttl_secs: u64,
458}

460impl AuthConfig {
461    /// Effective issuer for JWT `iss` claims and OIDC discovery.
462    ///
463    /// Returns `oauth2_issuer_url` when set, falling back to
464    /// `jwt_issuer`. Trailing slashes are stripped so that the
465    /// OIDC discovery `issuer` exactly matches token `iss` claims
466    /// (OIDC Core §2 requires an exact string match).
467    pub fn effective_issuer(&self) -> &str {
468        // T21.6 — the per-request tenant issuer wins when there is one, and
469        // there is one only on a `/t/{tenant_id}` request. Every caller that
470        // *stamps* an `iss` reads this and is therefore correct on both path
471        // forms without knowing either exists; every caller that *checks* one
472        // reads `root_issuer` instead.
473        if let Some(ref issuer) = self.request_issuer {
474            return issuer.trim_end_matches('/');
475        }
476        self.root_issuer()
477    }
478
479    /// The deployment's own issuer, ignoring any per-request tenant issuer
480    /// (T21.6).
481    ///
482    /// Identical to [`Self::effective_issuer`] on every request that is not on
483    /// a `/t/{tenant_id}` path, which is every request on a deployment with
484    /// [`Self::tenant_issuer_paths`] off.
485    ///
486    /// This is the value every **validation** decision is derived from. The
487    /// distinction matters exactly once and it is load-bearing: the set of
488    /// issuers AXIAM accepts must be a property of the deployment, so a
489    /// request that arrived under tenant `A` cannot narrow — or widen — what
490    /// counts as a valid `iss`.
491    #[must_use]
492    pub fn root_issuer(&self) -> &str {
493        if self.oauth2_issuer_url.is_empty() {
494            self.jwt_issuer.trim_end_matches('/')
495        } else {
496            self.oauth2_issuer_url.trim_end_matches('/')
497        }
498    }
499
500    /// The issuer identifier of one tenant under the T21.6 path form, or
501    /// `None` when this deployment does not serve it.
502    ///
503    /// `{root}/t/{tenant_id}`, with no query and no fragment — which is what
504    /// RFC 8414 §2 requires of an issuer and what the `?tenant_id=` form could
505    /// never be.
506    #[must_use]
507    pub fn tenant_issuer(&self, tenant_id: uuid::Uuid) -> Option<String> {
508        if !self.tenant_issuer_paths {
509            return None;
510        }
511        Some(format!(
512            "{}{}{}",
513            self.root_issuer(),
514            TENANT_PATH_PREFIX,
515            tenant_id
516        ))
517    }
518
519    /// This config, as seen by a request that arrived under tenant
520    /// `tenant_id`'s path (T21.6).
521    ///
522    /// A clone rather than a mutation, and a clone rather than a borrow of a
523    /// shared cell: the deployment config is shared by every concurrent
524    /// request, and the one thing that must never happen is a request on the
525    /// root path minting an `iss` a neighbouring tenant-path request put
526    /// there. The cost is a handful of small allocations on a path that is
527    /// about to sign a token.
528    ///
529    /// `None` when this deployment does not serve tenant paths, so a caller
530    /// cannot manufacture a tenant issuer on a deployment that has not opted
531    /// in.
532    #[must_use]
533    pub fn for_tenant_path(&self, tenant_id: uuid::Uuid) -> Option<Self> {
534        let issuer = self.tenant_issuer(tenant_id)?;
535        Some(Self {
536            request_issuer: Some(issuer),
537            ..self.clone()
538        })
539    }
540
541    /// Is `iss` an issuer identifier this deployment mints under?
542    ///
543    /// With [`Self::tenant_issuer_paths`] off this is `iss == root_issuer()`,
544    /// which is exactly the set `jsonwebtoken`'s pinned-issuer check accepted
545    /// before T21.6 — so the default deployment's answer is unchanged.
546    ///
547    /// With it on the set is the root issuer **and** every `{root}/t/{uuid}`.
548    /// Three properties of that widening are worth stating, because each one
549    /// is a way it could have gone wrong:
550    ///
551    /// * the tenant segment must parse as a UUID, so `{root}/t/../admin`,
552    ///   `{root}/t/` and `{root}/t/a%2Fb` are all refused rather than
553    ///   normalised;
554    /// * nothing is accepted **after** the tenant segment, so
555    ///   `{root}/t/{uuid}/anything` is not an issuer;
556    /// * the comparison is on the exact string, with no trailing-slash or
557    ///   case folding — OIDC Core §2 requires `iss` to match the discovery
558    ///   document's `issuer` exactly, and a comparison that is more forgiving
559    ///   than the specification is one an attacker gets to choose the input to.
560    ///
561    /// Widening the accepted `iss` set does **not** widen the accepted `aud`
562    /// set (I3): the audience check is a separate decision in
563    /// [`crate::token::decode_access_token`] and stays pinned to the two
564    /// built-in audiences.
565    #[must_use]
566    pub fn accepts_issuer(&self, iss: &str) -> bool {
567        let root = self.root_issuer();
568        if iss == root {
569            return true;
570        }
571        if !self.tenant_issuer_paths {
572            return false;
573        }
574        self.tenant_of_issuer(iss).is_some()
575    }
576
577    /// The tenant a `{root}/t/{uuid}` issuer names, or `None` when `iss` is
578    /// not one (T21.6).
579    ///
580    /// `None` for the root issuer too: the root issuer names no tenant, which
581    /// is the whole reason the path form exists.
582    #[must_use]
583    pub fn tenant_of_issuer(&self, iss: &str) -> Option<uuid::Uuid> {
584        if !self.tenant_issuer_paths {
585            return None;
586        }
587        let rest = iss.strip_prefix(self.root_issuer())?;
588        let tenant = rest.strip_prefix(TENANT_PATH_PREFIX)?;
589        uuid::Uuid::parse_str(tenant).ok()
590    }

592    /// The mTLS listener's base URL, or `None` when this deployment has no
593    /// separate one (RFC 8705 §5).
594    ///
595    /// Trailing slashes are stripped for the same reason
596    /// [`effective_issuer`](Self::effective_issuer) strips them: the value is
597    /// concatenated with endpoint paths, and `https://host//oauth2/token` is a
598    /// different URL from the one the operator meant.
599    ///
600    /// A value that is entirely whitespace answers `None` rather than `Some("")`
601    /// — an empty alias base would produce relative alias URLs, which RFC 8705
602    /// §5 does not permit and no client would resolve the way the operator
603    /// intended.
604    /// The configured default tenant, when one is set and parses as a UUID.
605    ///
606    /// An unparseable value is treated as unset rather than as an error: this
607    /// is consulted while building a *public, unauthenticated* document, and a
608    /// deployment whose operator fat-fingered the UUID should serve the
609    /// document it served before the setting existed rather than 500 for every
610    /// relying party. The mis-set value is visible in the document by its
611    /// absence, and `mtls_base_url` takes the opposite view for the opposite
612    /// reason — a bad alias actively misdirects a client, a missing tenant only
613    /// fails to help one.
614    #[must_use]
615    pub fn default_tenant_id(&self) -> Option<uuid::Uuid> {
616        let trimmed = self.oauth2_default_tenant_id.trim();
617        if trimmed.is_empty() {
618            return None;
619        }
620        uuid::Uuid::parse_str(trimmed).ok()
621    }
622
623    /// Whether [`Self::default_tenant_id`] is silently discarding a value the
624    /// operator set (T-244, R-1…R-8).
625    ///
626    /// Treating an unparseable value as unset is right and stays — a public,
627    /// unauthenticated document should not `500` for every relying party
628    /// because somebody fat-fingered a UUID — but the operator was never told,
629    /// so the deployment concluded the setting does not work. This answers
630    /// `Some` exactly when a non-empty value failed to parse, and the caller
631    /// logs it **once at boot**.
632    ///
633    /// # Why not on the request path
634    ///
635    /// [`Self::default_tenant_id`] is called per discovery request. A warning
636    /// there is a log flood any anonymous caller can drive by requesting the
637    /// document in a loop.
638    ///
639    /// # Why the shape and never the value
640    ///
641    /// A tenant id is not a secret, but a variable this code cannot prove *is*
642    /// a tenant id may hold anything an operator pasted — including the
643    /// contents of the wrong clipboard. The diagnostic therefore carries the
644    /// length and a character class and nothing else.
645    #[must_use]
646    pub fn default_tenant_id_diagnostic(&self) -> Option<DefaultTenantProblem> {
647        let trimmed = self.oauth2_default_tenant_id.trim();
648        // Empty and whitespace-only are "unset", which is the default and
649        // needs no warning.
650        if trimmed.is_empty() || uuid::Uuid::parse_str(trimmed).is_ok() {
651            return None;
652        }
653        Some(DefaultTenantProblem {
654            length: trimmed.chars().count(),
655            shape: if trimmed.chars().all(|c| c.is_ascii_hexdigit() || c == '-') {
656                "hexadecimal, but not a 36-character UUID"
657            } else {
658                "contains characters a UUID cannot"
659            },
660        })
661    }
662
663    pub fn mtls_base_url(&self) -> Option<&str> {
664        let trimmed = self.oauth2_mtls_base_url.trim().trim_end_matches('/');
665        (!trimmed.is_empty()).then_some(trimmed)
666    }

668    /// CQ-B14: Parse Ed25519 keys from PEM once and cache in `Arc`.
669    ///
670    /// Call this once at startup after loading config from environment.
671    /// After this returns `Ok(())`, all token functions skip per-call PEM
672    /// parsing and use the cached keys instead.
673    pub fn resolve_keys(&mut self) -> Result<(), String> {
674        let enc = EncodingKey::from_ed_pem(self.jwt_private_key_pem.as_bytes())
675            .map_err(|e| format!("invalid JWT private key PEM: {e}"))?;
676        let dec = DecodingKey::from_ed_pem(self.jwt_public_key_pem.as_bytes())
677            .map_err(|e| format!("invalid JWT public key PEM: {e}"))?;
678        self.jwt_encoding_key = Some(Arc::new(enc));
679        self.jwt_decoding_key = Some(Arc::new(dec));
680        Ok(())
681    }
682
683    /// B1: Resolve `max_concurrent_hashes` (0 = auto) to a concrete permit
684    /// count for the crypto semaphore.
685    ///
686    /// Auto (`0`) resolves to `min(available_parallelism, 4)`: bound by the
687    /// core count so Argon2id (a CPU-bound operation) does not oversubscribe
688    /// cores, and capped at 4 so peak concurrent ~19 MiB arenas stay well
689    /// under typical container memory caps. A non-zero value is used verbatim
690    /// (a `0` after resolution is impossible — it would be a zero-permit
691    /// semaphore that deadlocks all hashing, so it is clamped to 1).
692    pub fn resolved_max_concurrent_hashes(&self) -> usize {
693        match self.max_concurrent_hashes {
694            0 => std::thread::available_parallelism()
695                .map(|n| n.get().min(4))
696                .unwrap_or(4),
697            n => n.max(1),
698        }
699    }
700}
