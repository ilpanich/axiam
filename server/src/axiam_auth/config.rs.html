13/// What is wrong with `AXIAM__AUTH__OAUTH2_DEFAULT_TENANT_ID`, in terms that
14/// name no part of the value (T-244).
15///
16/// Rendered into one boot-time `WARN` by the composition root. It describes
17/// the value's **shape** — how long it is, and whether its characters could
18/// belong to a UUID at all — because those two facts distinguish the mistakes
19/// an operator actually makes (a truncated paste, a tenant *slug* where an id
20/// belongs, a different identifier entirely) without echoing anything.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub struct DefaultTenantProblem {
23    /// Character count of the trimmed value. A UUID is 36.
24    pub length: usize,
25    /// A short phrase describing the character class.
26    pub shape: &'static str,
27}
28
29impl std::fmt::Display for DefaultTenantProblem {
30    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
31        write!(f, "{} characters, {}", self.length, self.shape)
32    }
33}

35/// Configuration for the authentication service.
36#[derive(Debug, Clone, Deserialize)]
37#[serde(default)]
38pub struct AuthConfig {
39    /// PEM-encoded Ed25519 private key for JWT signing.
40    pub jwt_private_key_pem: String,
41    /// PEM-encoded Ed25519 public key for JWT verification.
42    pub jwt_public_key_pem: String,
43    /// Trust an `X-Client-Certificate` request header as proof of device
44    /// identity when the connection carries no TLS-verified client
45    /// certificate. Default **`false`**.
46    ///
47    /// # Why this is off by default
48    ///
49    /// The header carries a PEM certificate, and a certificate is public data.
50    /// `DeviceAuthService::authenticate` checks the fingerprint, the status,
51    /// the expiry and the chain up to the tenant or organization CA — every one
52    /// of which a *copy* of an enrolled device's certificate also satisfies.
53    /// Nothing in that path proves the sender holds the private key, because
54    /// nothing can: possession is proven by the TLS handshake, and on this path
55    /// there was no handshake to prove it in.
56    ///
57    /// That is sound only while the header cannot originate with the client —
58    /// i.e. a reverse proxy performs the mTLS handshake itself and
59    /// unconditionally overwrites the header on every request it forwards.
60    /// It stops being sound the moment the server is reachable by anything but
61    /// that proxy, where anyone holding a copy of a device certificate
62    /// authenticates as that device.
63    ///
64    /// Native mTLS is unaffected and always preferred: when rustls verified a
65    /// client certificate on the connection, that certificate is authoritative
66    /// and this setting is not consulted.
67    ///
68    /// Set it to `true` only if all of the following hold, and say so in your
69    /// runbook:
70    ///
71    /// 1. TLS (and the client-certificate handshake) terminates at a proxy you
72    ///    operate;
73    /// 2. that proxy sets `X-Client-Certificate` from the certificate **it**
74    ///    verified, on every request, overwriting anything the client sent;
75    /// 3. nothing else can reach this server's listener.
76    pub trust_forwarded_client_cert: bool,
77    /// Access token lifetime in seconds (default: 900 = 15 minutes).
78    pub access_token_lifetime_secs: u64,
79    /// Refresh token lifetime in seconds (default: 2_592_000 = 30 days).
80    pub refresh_token_lifetime_secs: u64,
81    /// Authorization code lifetime in seconds (default: 600 = 10 minutes).
82    pub auth_code_lifetime_secs: u64,
83    /// JWT issuer (`iss` claim).
84    pub jwt_issuer: String,
85    /// OIDC issuer base URL (e.g. "https://auth.example.com"). Used for
86    /// OIDC discovery endpoint URLs. Falls back to `jwt_issuer` if unset.
87    pub oauth2_issuer_url: String,
88    /// Base URL of the listener that performs the mutual-TLS handshake, when
89    /// that is a different host or port from
90    /// [`effective_issuer`](Self::effective_issuer)
91    /// (`AXIAM__AUTH__OAUTH2_MTLS_BASE_URL`). Empty by default.
92    ///
93    /// Setting it makes the discovery document carry RFC 8705 §5
94    /// `mtls_endpoint_aliases`; leaving it empty omits the member entirely,
95    /// which is what the RFC asks of a server with nothing to alias.
96    ///
97    /// # Why a separate URL is needed at all
98    ///
99    /// A TLS listener decides whether to request a client certificate during
100    /// the handshake, before it has seen a single byte of HTTP. So "ask for a
101    /// certificate on `/oauth2/token` but not on `/oauth2/authorize`" is not a
102    /// thing a listener can do — the choice is per-listener, and a deployment
103    /// that wants both a browser-facing authorization endpoint and a
104    /// certificate-authenticated token endpoint has to run two of them.
105    ///
106    /// Which leaves the client with a question metadata alone could not answer
107    /// before RFC 8705 §5: the discovery document names one `token_endpoint`,
108    /// and a client doing mTLS needs the *other* one. The aliases are that
109    /// answer — the same endpoints, re-based on the host that will ask for a
110    /// certificate.
111    ///
112    /// # When to leave it empty
113    ///
114    /// Two deployments should, and neither is unusual:
115    ///
116    /// - **`client_auth = optional` on a single listener.** rustls requests a
117    ///   certificate and accepts a connection without one, so the conventional
118    ///   endpoints already serve both populations and there is no second URL
119    ///   to point at. This is the shape `scripts/e2e-mtls-native-check.sh`
120    ///   exercises.
121    /// - **`client_auth = required` on a single listener.** Every endpoint is
122    ///   already behind the handshake; an alias would name the URL the client
123    ///   is using.
124    ///
125    /// Set it only when mTLS genuinely terminates somewhere else — typically a
126    /// second proxy hostname such as `https://mtls.iam.example.com` in front of
127    /// the same server.
128    ///
129    /// # What it must be
130    ///
131    /// An absolute URL, validated where the document is built. A value that is
132    /// not parseable fails the discovery request rather than being dropped:
133    /// silently omitting the aliases would send an mTLS client to the
134    /// conventional endpoints, which is the one outcome this setting exists to
135    /// prevent.
136    pub oauth2_mtls_base_url: String,
137    /// The tenant a discovery document describes when the caller named none
138    /// (`AXIAM__AUTH__OAUTH2_DEFAULT_TENANT_ID`). Empty by default.
139    ///
140    /// # What it fixes
141    ///
142    /// Every OAuth2 endpoint that authenticates a *client* — token, PAR,
143    /// introspection, revocation, device authorization, end-session — takes a
144    /// **required** `tenant_id` query parameter, and `/oauth2/authorize` needs
145    /// one for any request without a principal, which is every browser. The
146    /// discovery document published none of them, so a relying party that did
147    /// exactly what OIDC Discovery tells it to do — read the document, use the
148    /// URLs — got `400 missing field tenant_id` at the token endpoint. The
149    /// first OpenID Foundation conformance run could not complete a single
150    /// authorization for this reason.
151    ///
152    /// Setting this makes the **bare** document publish endpoint URLs carrying
153    /// the tenant, which is the deployment shape an OP is certified as: one
154    /// issuer, one tenant. A document fetched with an explicit `?tenant_id=`
155    /// carries that tenant instead, and this value is not consulted.
156    ///
157    /// # What it does NOT do
158    ///
159    /// It changes no endpoint's behaviour. A request that arrives without
160    /// `tenant_id` is refused exactly as it is today — this is a statement in a
161    /// document, not a fallback in a handler. That distinction is deliberate:
162    /// a default applied at the endpoint would silently give an
163    /// unparameterised request a tenant, and on a multi-tenant authorization
164    /// server the tenant is the isolation boundary. Leaving it empty is
165    /// therefore safe and is the correct setting for a deployment that serves
166    /// many tenants from one issuer.
167    pub oauth2_default_tenant_id: String,
168    /// T-39/T-143 — whether this deployment publishes
169    /// `GET /oauth2/revocations`.
170    ///
171    /// `AXIAM__AUTH__REVOCATION_FEED_ENABLED`, default `false`. With it off the
172    /// route is not mounted and no `revoked_session` row is written, so the
173    /// deployment is byte-identical to one built before the feed existed.
174    ///
175    /// The feed narrows the window in which a revoked session's access token
176    /// still verifies — from one token lifetime to one poll interval — for an
177    /// SDK guard that opts into polling it. It is never a control: a guard
178    /// that cannot fetch it behaves exactly as it does today, and the token
179    /// itself still decides.
180    #[serde(default)]
181    pub revocation_feed_enabled: bool,
182    /// Extra browser origins this deployment will hand a **federation SSO
183    /// handoff code** to (`AXIAM__AUTH__SSO_SPA_ORIGINS`; a list, set the same
184    /// way as `AXIAM__SERVER__CORS_ALLOWED_ORIGINS`).
185    ///
186    /// # Why this exists
187    ///
188    /// SAML and Apple return by cross-site form POST, so the session cookies
189    /// cannot be set on that response. AXIAM instead mints a single-use handoff
190    /// code and redirects the browser to the SPA with it — and that code is a
191    /// bearer credential for a session. The redirect target therefore may not
192    /// be whatever the caller of the *unauthenticated* login-start endpoint
193    /// asked for: on those two protocols the identity provider never sees the
194    /// SPA `redirect_uri`, so its own registered-redirect allowlist is not
195    /// there to reject an attacker's host.
196    ///
197    /// The default is empty, which means **the origin of
198    /// [`effective_issuer`](Self::effective_issuer) only**. That is the right
199    /// default rather than a restrictive one: the endpoint a SAML IdP or Apple
200    /// posts to is itself built from `effective_issuer`, so a deployment whose
201    /// SPA is served from that origin — the shape `docker/nginx.conf` and the
202    /// deployment guides describe, and the only shape in which
203    /// `SameSite=Strict` session cookies work same-origin — needs no entry
204    /// here at all.
205    ///
206    /// Set it when the SPA is served from a different host than the API (for
207    /// example `https://app.example.com` against `https://api.example.com`).
208    /// Values are compared as **origins**: scheme, host and port, with the path
209    /// ignored — so `https://app.example.com` does not admit
210    /// `https://app.example.com:8443`.
211    ///
212    /// # Since R-3: all four flows, not two
213    ///
214    /// This used to have no effect on the OIDC and OAuth2 flows, whose
215    /// `redirect_uri` is registered at the provider and checked there. It now
216    /// governs those two as well. The provider's check is a real control, but
217    /// only as strict as each provider's registration hygiene — several accept
218    /// wildcard or prefix registrations — and it is not a control AXIAM owns or
219    /// can inspect; it stays as a second, independent layer behind this one.
220    ///
221    /// **This is the one thing R-3 can break.** A deployment whose SPA lives on
222    /// an origin other than the issuer's, signing in through OIDC or OAuth2
223    /// providers, worked without this variable and now needs it — the same
224    /// requirement the SAML and Apple flows have imposed since beta08. The
225    /// shipped same-origin topology needs nothing. The `400` names this
226    /// variable, so the failure says what to set.
227    pub sso_spa_origins: Vec<String>,
228    /// Optional pepper prepended to passwords before Argon2id verification.
229    /// Wrapped in `SecretString` so `Debug`/logging never leaks it by
230    /// accident (SECHRD-12) — exposed only via `.expose_secret()` at the
231    /// `&str` boundary where a pepper value is consumed.
232    pub pepper: Option<SecretString>,
233    /// Previous pepper, kept **verify-only** while a pepper rotation drains
234    /// (`AXIAM__AUTH__PEPPER_PREVIOUS`, §13.4 observation 3).
235    ///
236    /// Client-secret hashes are keyed by the pepper, and the `v2.hs256$` tag
237    /// versions the *algorithm*, not the *key* — so without this, rotating
238    /// `AXIAM__AUTH__PEPPER` permanently invalidates **every** client secret and

240    /// with the restart.
241    ///
242    /// Set this to the outgoing value for the duration of a rotation. Secrets
243    /// hashed under it still verify, and each one is transparently rewritten
244    /// under the new pepper the first time its owner authenticates, so the
245    /// rotation drains itself. Unset it once the fleet has drained; nothing is
246    /// ever *written* under this key.
247    ///
248    /// It does **not** apply to Argon2id password peppering, which has its own
249    /// migration story — only to client-secret hashing.
250    pub pepper_previous: Option<SecretString>,
251    /// Minimum password length for policy enforcement.
252    pub min_password_length: usize,

353    /// B1: Maximum number of Argon2id hash/verify operations allowed to run
354    /// concurrently, enforced by the process-wide `crypto_semaphore` shared
355    /// across the login, password-change, password-reset and PKI crypto paths.
356    ///
357    /// Each in-flight Argon2id operation allocates a ~19 MiB memory arena
358    /// (OWASP params m=19456), so unbounded concurrency is an unauthenticated
359    /// memory-DoS vector: the login benchmark pegged 2 cores and reached
360    /// ~970 MiB RSS ≈ 50 concurrent × 19 MiB arenas, against a 1024 MiB
361    /// container cap. This permit count bounds peak concurrent arenas.
362    ///
363    /// `0` means "auto" and resolves to `min(available_parallelism, 4)` at
364    /// semaphore-construction time (see [`AuthConfig::resolved_max_concurrent_hashes`]).
365    /// Override via `AXIAM__AUTH__MAX_CONCURRENT_HASHES`.

367    /// B1: How long (in seconds) a request waits to acquire a crypto-hash
368    /// permit before giving up and returning a `ServiceUnavailable` (HTTP 503)
369    /// backpressure error, instead of queueing unboundedly and adding to
370    /// tail latency. Default: 5. Override via
371    /// `AXIAM__AUTH__HASH_ACQUIRE_TIMEOUT_SECS`.
372    pub hash_acquire_timeout_secs: u64,
373    /// I6: TTL, in seconds, of the process-local **session-validation cache**.
374    ///
375    /// `0` (the default) disables the cache entirely — every authenticated
376    /// request performs its own SurrealDB read to confirm the session behind the
377    /// access token's `jti` still exists (D-15 / REQ-7). A non-zero value lets
378    /// repeated requests inside the window reuse the last positive answer.
379    ///
380    /// Override via `AXIAM__AUTH__SESSION_VALIDATION_CACHE_TTL_SECS`.
381    ///
382    /// # What it buys and what it costs
383    ///
384    /// The read it removes is the reason the D7 decision cache helped gRPC
385    /// authorization checks 13.1× but REST checks only 5% in benchmark run 4:
386    /// the decision cache elides the *authorization* round-trips, this one
387    /// elides the *authentication* round-trip, and the gRPC surface never had
388    /// the latter.
389    ///
390    /// The cost is a bounded revocation window. Every session-deleting path in
391    /// `SurrealSessionRepository` invalidates the cache in the same call, so on
392    /// a **single replica** a logout takes effect immediately. Across **two or
393    /// more replicas** there is no invalidation channel, so a session revoked on
394    /// replica A stays acceptable on replicas B…N for at most this many
395    /// seconds — the same trade-off, and the same reasoning, as
396    /// `AXIAM__AUTHZ__DECISION_CACHE_TTL_SECS`. Session *expiry* is not
397    /// affected: cached entries carry the row's own `expires_at` and are
398    /// rejected exactly on time regardless of the TTL.
399    ///
400    /// Suggested starting point when enabling: `5`, matching the decision
401    /// cache's default.
402    pub session_validation_cache_ttl_secs: u64,
403}

405impl AuthConfig {
406    /// Effective issuer for JWT `iss` claims and OIDC discovery.
407    ///
408    /// Returns `oauth2_issuer_url` when set, falling back to
409    /// `jwt_issuer`. Trailing slashes are stripped so that the
410    /// OIDC discovery `issuer` exactly matches token `iss` claims
411    /// (OIDC Core §2 requires an exact string match).
412    pub fn effective_issuer(&self) -> &str {
413        if self.oauth2_issuer_url.is_empty() {
414            self.jwt_issuer.trim_end_matches('/')
415        } else {
416            self.oauth2_issuer_url.trim_end_matches('/')
417        }
418    }
419
420    /// The mTLS listener's base URL, or `None` when this deployment has no
421    /// separate one (RFC 8705 §5).
422    ///
423    /// Trailing slashes are stripped for the same reason
424    /// [`effective_issuer`](Self::effective_issuer) strips them: the value is
425    /// concatenated with endpoint paths, and `https://host//oauth2/token` is a
426    /// different URL from the one the operator meant.
427    ///
428    /// A value that is entirely whitespace answers `None` rather than `Some("")`
429    /// — an empty alias base would produce relative alias URLs, which RFC 8705
430    /// §5 does not permit and no client would resolve the way the operator
431    /// intended.
432    /// The configured default tenant, when one is set and parses as a UUID.
433    ///
434    /// An unparseable value is treated as unset rather than as an error: this
435    /// is consulted while building a *public, unauthenticated* document, and a
436    /// deployment whose operator fat-fingered the UUID should serve the
437    /// document it served before the setting existed rather than 500 for every
438    /// relying party. The mis-set value is visible in the document by its
439    /// absence, and `mtls_base_url` takes the opposite view for the opposite
440    /// reason — a bad alias actively misdirects a client, a missing tenant only
441    /// fails to help one.
442    #[must_use]
443    pub fn default_tenant_id(&self) -> Option<uuid::Uuid> {
444        let trimmed = self.oauth2_default_tenant_id.trim();
445        if trimmed.is_empty() {
446            return None;
447        }
448        uuid::Uuid::parse_str(trimmed).ok()
449    }
450
451    /// Whether [`Self::default_tenant_id`] is silently discarding a value the
452    /// operator set (T-244, R-1…R-8).
453    ///
454    /// Treating an unparseable value as unset is right and stays — a public,
455    /// unauthenticated document should not `500` for every relying party
456    /// because somebody fat-fingered a UUID — but the operator was never told,
457    /// so the deployment concluded the setting does not work. This answers
458    /// `Some` exactly when a non-empty value failed to parse, and the caller
459    /// logs it **once at boot**.
460    ///
461    /// # Why not on the request path
462    ///
463    /// [`Self::default_tenant_id`] is called per discovery request. A warning
464    /// there is a log flood any anonymous caller can drive by requesting the
465    /// document in a loop.
466    ///
467    /// # Why the shape and never the value
468    ///
469    /// A tenant id is not a secret, but a variable this code cannot prove *is*
470    /// a tenant id may hold anything an operator pasted — including the
471    /// contents of the wrong clipboard. The diagnostic therefore carries the
472    /// length and a character class and nothing else.
473    #[must_use]
474    pub fn default_tenant_id_diagnostic(&self) -> Option<DefaultTenantProblem> {
475        let trimmed = self.oauth2_default_tenant_id.trim();
476        // Empty and whitespace-only are "unset", which is the default and
477        // needs no warning.
478        if trimmed.is_empty() || uuid::Uuid::parse_str(trimmed).is_ok() {
479            return None;
480        }
481        Some(DefaultTenantProblem {
482            length: trimmed.chars().count(),
483            shape: if trimmed.chars().all(|c| c.is_ascii_hexdigit() || c == '-') {
484                "hexadecimal, but not a 36-character UUID"
485            } else {
486                "contains characters a UUID cannot"
487            },
488        })
489    }
490
491    pub fn mtls_base_url(&self) -> Option<&str> {
492        let trimmed = self.oauth2_mtls_base_url.trim().trim_end_matches('/');
493        (!trimmed.is_empty()).then_some(trimmed)
494    }
495
496    /// CQ-B14: Parse Ed25519 keys from PEM once and cache in `Arc`.
497    ///
498    /// Call this once at startup after loading config from environment.
499    /// After this returns `Ok(())`, all token functions skip per-call PEM
500    /// parsing and use the cached keys instead.
501    pub fn resolve_keys(&mut self) -> Result<(), String> {
502        let enc = EncodingKey::from_ed_pem(self.jwt_private_key_pem.as_bytes())
503            .map_err(|e| format!("invalid JWT private key PEM: {e}"))?;
504        let dec = DecodingKey::from_ed_pem(self.jwt_public_key_pem.as_bytes())
505            .map_err(|e| format!("invalid JWT public key PEM: {e}"))?;
506        self.jwt_encoding_key = Some(Arc::new(enc));
507        self.jwt_decoding_key = Some(Arc::new(dec));
508        Ok(())
509    }
510
511    /// B1: Resolve `max_concurrent_hashes` (0 = auto) to a concrete permit
512    /// count for the crypto semaphore.
513    ///
514    /// Auto (`0`) resolves to `min(available_parallelism, 4)`: bound by the
515    /// core count so Argon2id (a CPU-bound operation) does not oversubscribe
516    /// cores, and capped at 4 so peak concurrent ~19 MiB arenas stay well
517    /// under typical container memory caps. A non-zero value is used verbatim
518    /// (a `0` after resolution is impossible — it would be a zero-permit
519    /// semaphore that deadlocks all hashing, so it is clamped to 1).
520    pub fn resolved_max_concurrent_hashes(&self) -> usize {
521        match self.max_concurrent_hashes {
522            0 => std::thread::available_parallelism()
523                .map(|n| n.get().min(4))
524                .unwrap_or(4),
525            n => n.max(1),
526        }
527    }
528}

587#[cfg(test)]
588mod default_tenant_tests {
589    use super::*;
590
591    fn config_with(value: &str) -> AuthConfig {
592        AuthConfig {
593            oauth2_default_tenant_id: value.to_owned(),
594            ..AuthConfig::default()
595        }
596    }
597
598    /// Unset, whitespace-only and valid all mean "nothing to say". A warning
599    /// in any of these is a warning an operator learns to scroll past.
600    #[test]
601    fn a_valid_or_absent_default_tenant_says_nothing() {
602        for quiet in [
603            "",
604            "   ",
605            "\n",
606            "6f3e0a5c-1b2d-4e8f-9a7b-0c1d2e3f4a5b",
607            "  6f3e0a5c-1b2d-4e8f-9a7b-0c1d2e3f4a5b  ",
608        ] {
609            assert_eq!(
610                config_with(quiet).default_tenant_id_diagnostic(),
611                None,
612                "{quiet:?} must produce no diagnostic"
613            );
614        }
615    }
616
617    /// A value that is set and unusable is exactly the case the operator
618    /// never heard about.
619    #[test]
620    fn an_unparseable_default_tenant_is_reported() {
621        let problem = config_with("not-a-uuid")
622            .default_tenant_id_diagnostic()
623            .expect("an unparseable value must be reported");
624        assert_eq!(problem.length, 10);
625        assert_eq!(problem.shape, "contains characters a UUID cannot");
626    }
627
628    /// A truncated paste is the commonest way to get here, and it is worth
629    /// distinguishing from a value that was never a UUID at all: the first is
630    /// "you lost some characters", the second is "that is a different thing".
631    #[test]
632    fn a_truncated_uuid_is_reported_as_the_right_shape() {
633        let problem = config_with("6f3e0a5c-1b2d-4e8f")
634            .default_tenant_id_diagnostic()
635            .expect("a truncated UUID must be reported");
636        assert_eq!(problem.length, 18);
637        assert_eq!(problem.shape, "hexadecimal, but not a 36-character UUID");
638    }
639
640    /// The rendered line carries the length and the class, and **no substring
641    /// of the value**. The check is deliberately crude — every three-character
642    /// window of the value — because the failure mode is somebody adding the
643    /// value to the message to make it easier to debug.
644    #[test]
645    fn the_rendered_diagnostic_never_echoes_the_value() {
646        let value = "tenant-acme-production";
647        let rendered = config_with(value)
648            .default_tenant_id_diagnostic()
649            .unwrap()
650            .to_string();
651        for window in value
652            .as_bytes()
653            .windows(3)
654            .map(|w| std::str::from_utf8(w).unwrap())
655        {
656            assert!(
657                !rendered.contains(window),
658                "the diagnostic must not echo any part of the value; \
659                 found {window:?} in {rendered:?}"
660            );
661        }
662        assert!(rendered.contains("22 characters"));
663    }
664
665    /// **I4 twin, and the proof this item changed nothing.** The accessor the
666    /// discovery builder actually reads answers `None` for an unparseable
667    /// value exactly as it did before the diagnostic existed — so the document
668    /// a misconfigured deployment serves is the document an unconfigured one
669    /// serves, which is what `oidc.rs::a_document_that_names_no_tenant_carries_no_query_string`
670    /// pins on the other side.
671    #[test]
672    fn an_unparseable_default_tenant_is_still_treated_as_unset() {
673        assert_eq!(config_with("not-a-uuid").default_tenant_id(), None);
674        assert_eq!(config_with("").default_tenant_id(), None);
675        assert_eq!(
676            config_with("not-a-uuid").default_tenant_id(),
677            config_with("").default_tenant_id(),
678            "a bad value and no value must be indistinguishable to the builder"
679        );
680    }
