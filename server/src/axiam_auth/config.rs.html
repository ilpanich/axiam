13/// Configuration for the authentication service.
14#[derive(Debug, Clone, Deserialize)]
15#[serde(default)]
16pub struct AuthConfig {
17    /// PEM-encoded Ed25519 private key for JWT signing.
18    pub jwt_private_key_pem: String,
19    /// PEM-encoded Ed25519 public key for JWT verification.
20    pub jwt_public_key_pem: String,
21    /// Trust an `X-Client-Certificate` request header as proof of device
22    /// identity when the connection carries no TLS-verified client
23    /// certificate. Default **`false`**.
24    ///
25    /// # Why this is off by default
26    ///
27    /// The header carries a PEM certificate, and a certificate is public data.
28    /// `DeviceAuthService::authenticate` checks the fingerprint, the status,
29    /// the expiry and the chain up to the tenant or organization CA — every one
30    /// of which a *copy* of an enrolled device's certificate also satisfies.
31    /// Nothing in that path proves the sender holds the private key, because
32    /// nothing can: possession is proven by the TLS handshake, and on this path
33    /// there was no handshake to prove it in.
34    ///
35    /// That is sound only while the header cannot originate with the client —
36    /// i.e. a reverse proxy performs the mTLS handshake itself and
37    /// unconditionally overwrites the header on every request it forwards.
38    /// It stops being sound the moment the server is reachable by anything but
39    /// that proxy, where anyone holding a copy of a device certificate
40    /// authenticates as that device.
41    ///
42    /// Native mTLS is unaffected and always preferred: when rustls verified a
43    /// client certificate on the connection, that certificate is authoritative
44    /// and this setting is not consulted.
45    ///
46    /// Set it to `true` only if all of the following hold, and say so in your
47    /// runbook:
48    ///
49    /// 1. TLS (and the client-certificate handshake) terminates at a proxy you
50    ///    operate;
51    /// 2. that proxy sets `X-Client-Certificate` from the certificate **it**
52    ///    verified, on every request, overwriting anything the client sent;
53    /// 3. nothing else can reach this server's listener.
54    pub trust_forwarded_client_cert: bool,
55    /// Access token lifetime in seconds (default: 900 = 15 minutes).
56    pub access_token_lifetime_secs: u64,
57    /// Refresh token lifetime in seconds (default: 2_592_000 = 30 days).
58    pub refresh_token_lifetime_secs: u64,
59    /// Authorization code lifetime in seconds (default: 600 = 10 minutes).
60    pub auth_code_lifetime_secs: u64,
61    /// JWT issuer (`iss` claim).
62    pub jwt_issuer: String,
63    /// OIDC issuer base URL (e.g. "https://auth.example.com"). Used for
64    /// OIDC discovery endpoint URLs. Falls back to `jwt_issuer` if unset.
65    pub oauth2_issuer_url: String,
66    /// Base URL of the listener that performs the mutual-TLS handshake, when
67    /// that is a different host or port from
68    /// [`effective_issuer`](Self::effective_issuer)
69    /// (`AXIAM__AUTH__OAUTH2_MTLS_BASE_URL`). Empty by default.
70    ///
71    /// Setting it makes the discovery document carry RFC 8705 §5
72    /// `mtls_endpoint_aliases`; leaving it empty omits the member entirely,
73    /// which is what the RFC asks of a server with nothing to alias.
74    ///
75    /// # Why a separate URL is needed at all
76    ///
77    /// A TLS listener decides whether to request a client certificate during
78    /// the handshake, before it has seen a single byte of HTTP. So "ask for a
79    /// certificate on `/oauth2/token` but not on `/oauth2/authorize`" is not a
80    /// thing a listener can do — the choice is per-listener, and a deployment
81    /// that wants both a browser-facing authorization endpoint and a
82    /// certificate-authenticated token endpoint has to run two of them.
83    ///
84    /// Which leaves the client with a question metadata alone could not answer
85    /// before RFC 8705 §5: the discovery document names one `token_endpoint`,
86    /// and a client doing mTLS needs the *other* one. The aliases are that
87    /// answer — the same endpoints, re-based on the host that will ask for a
88    /// certificate.
89    ///
90    /// # When to leave it empty
91    ///
92    /// Two deployments should, and neither is unusual:
93    ///
94    /// - **`client_auth = optional` on a single listener.** rustls requests a
95    ///   certificate and accepts a connection without one, so the conventional
96    ///   endpoints already serve both populations and there is no second URL
97    ///   to point at. This is the shape `scripts/e2e-mtls-native-check.sh`
98    ///   exercises.
99    /// - **`client_auth = required` on a single listener.** Every endpoint is
100    ///   already behind the handshake; an alias would name the URL the client
101    ///   is using.
102    ///
103    /// Set it only when mTLS genuinely terminates somewhere else — typically a
104    /// second proxy hostname such as `https://mtls.iam.example.com` in front of
105    /// the same server.
106    ///
107    /// # What it must be
108    ///
109    /// An absolute URL, validated where the document is built. A value that is
110    /// not parseable fails the discovery request rather than being dropped:
111    /// silently omitting the aliases would send an mTLS client to the
112    /// conventional endpoints, which is the one outcome this setting exists to
113    /// prevent.
114    pub oauth2_mtls_base_url: String,
115    /// The tenant a discovery document describes when the caller named none
116    /// (`AXIAM__AUTH__OAUTH2_DEFAULT_TENANT_ID`). Empty by default.
117    ///
118    /// # What it fixes
119    ///
120    /// Every OAuth2 endpoint that authenticates a *client* — token, PAR,
121    /// introspection, revocation, device authorization, end-session — takes a
122    /// **required** `tenant_id` query parameter, and `/oauth2/authorize` needs
123    /// one for any request without a principal, which is every browser. The
124    /// discovery document published none of them, so a relying party that did
125    /// exactly what OIDC Discovery tells it to do — read the document, use the
126    /// URLs — got `400 missing field tenant_id` at the token endpoint. The
127    /// first OpenID Foundation conformance run could not complete a single
128    /// authorization for this reason.
129    ///
130    /// Setting this makes the **bare** document publish endpoint URLs carrying
131    /// the tenant, which is the deployment shape an OP is certified as: one
132    /// issuer, one tenant. A document fetched with an explicit `?tenant_id=`
133    /// carries that tenant instead, and this value is not consulted.
134    ///
135    /// # What it does NOT do
136    ///
137    /// It changes no endpoint's behaviour. A request that arrives without
138    /// `tenant_id` is refused exactly as it is today — this is a statement in a
139    /// document, not a fallback in a handler. That distinction is deliberate:
140    /// a default applied at the endpoint would silently give an
141    /// unparameterised request a tenant, and on a multi-tenant authorization
142    /// server the tenant is the isolation boundary. Leaving it empty is
143    /// therefore safe and is the correct setting for a deployment that serves
144    /// many tenants from one issuer.
145    pub oauth2_default_tenant_id: String,
146    /// Extra browser origins this deployment will hand a **federation SSO
147    /// handoff code** to (`AXIAM__AUTH__SSO_SPA_ORIGINS`; a list, set the same
148    /// way as `AXIAM__SERVER__CORS_ALLOWED_ORIGINS`).
149    ///
150    /// # Why this exists
151    ///
152    /// SAML and Apple return by cross-site form POST, so the session cookies
153    /// cannot be set on that response. AXIAM instead mints a single-use handoff
154    /// code and redirects the browser to the SPA with it — and that code is a
155    /// bearer credential for a session. The redirect target therefore may not
156    /// be whatever the caller of the *unauthenticated* login-start endpoint
157    /// asked for: on those two protocols the identity provider never sees the
158    /// SPA `redirect_uri`, so its own registered-redirect allowlist is not
159    /// there to reject an attacker's host.
160    ///
161    /// The default is empty, which means **the origin of
162    /// [`effective_issuer`](Self::effective_issuer) only**. That is the right
163    /// default rather than a restrictive one: the endpoint a SAML IdP or Apple
164    /// posts to is itself built from `effective_issuer`, so a deployment whose
165    /// SPA is served from that origin — the shape `docker/nginx.conf` and the
166    /// deployment guides describe, and the only shape in which
167    /// `SameSite=Strict` session cookies work same-origin — needs no entry
168    /// here at all.
169    ///
170    /// Set it when the SPA is served from a different host than the API (for
171    /// example `https://app.example.com` against `https://api.example.com`).
172    /// Values are compared as **origins**: scheme, host and port, with the path
173    /// ignored — so `https://app.example.com` does not admit
174    /// `https://app.example.com:8443`.
175    ///
176    /// # Since R-3: all four flows, not two
177    ///
178    /// This used to have no effect on the OIDC and OAuth2 flows, whose
179    /// `redirect_uri` is registered at the provider and checked there. It now
180    /// governs those two as well. The provider's check is a real control, but
181    /// only as strict as each provider's registration hygiene — several accept
182    /// wildcard or prefix registrations — and it is not a control AXIAM owns or
183    /// can inspect; it stays as a second, independent layer behind this one.
184    ///
185    /// **This is the one thing R-3 can break.** A deployment whose SPA lives on
186    /// an origin other than the issuer's, signing in through OIDC or OAuth2
187    /// providers, worked without this variable and now needs it — the same
188    /// requirement the SAML and Apple flows have imposed since beta08. The
189    /// shipped same-origin topology needs nothing. The `400` names this
190    /// variable, so the failure says what to set.
191    pub sso_spa_origins: Vec<String>,
192    /// Optional pepper prepended to passwords before Argon2id verification.
193    /// Wrapped in `SecretString` so `Debug`/logging never leaks it by
194    /// accident (SECHRD-12) — exposed only via `.expose_secret()` at the
195    /// `&str` boundary where a pepper value is consumed.
196    pub pepper: Option<SecretString>,
197    /// Previous pepper, kept **verify-only** while a pepper rotation drains
198    /// (`AXIAM__AUTH__PEPPER_PREVIOUS`, §13.4 observation 3).
199    ///
200    /// Client-secret hashes are keyed by the pepper, and the `v2.hs256$` tag
201    /// versions the *algorithm*, not the *key* — so without this, rotating
202    /// `AXIAM__AUTH__PEPPER` permanently invalidates **every** client secret and
203    /// every service account and OAuth2 client has to be re-issued in lockstep
204    /// with the restart.
205    ///
206    /// Set this to the outgoing value for the duration of a rotation. Secrets
207    /// hashed under it still verify, and each one is transparently rewritten
208    /// under the new pepper the first time its owner authenticates, so the
209    /// rotation drains itself. Unset it once the fleet has drained; nothing is
210    /// ever *written* under this key.
211    ///
212    /// It does **not** apply to Argon2id password peppering, which has its own
213    /// migration story — only to client-secret hashing.
214    pub pepper_previous: Option<SecretString>,
215    /// Minimum password length for policy enforcement.
216    pub min_password_length: usize,
217    /// 256-bit AES-GCM key for encrypting TOTP secrets at rest.
218    /// `None` disables MFA enrollment. Set programmatically (not from config files).
219    #[serde(skip)]
220    pub mfa_encryption_key: Option<[u8; 32]>,
221    /// 256-bit AES-GCM key for encrypting federation client secrets at rest.
222    /// `None` means federation config create/update will fail at runtime.
223    /// Set programmatically from `AXIAM__AUTH__FEDERATION_ENCRYPTION_KEY` (not from
224    /// config files). Federation is optional — absence is warned, not fatal.
225    #[serde(skip)]
226    pub federation_encryption_key: Option<[u8; 32]>,
227    /// 256-bit AES-GCM key that seals an OPAQUE exchange's server state
228    /// between its two messages, on both the login and the registration path.
229    ///
230    /// `None` means the OPAQUE endpoints answer `503`, whatever the org or
231    /// tenant policy says. That is deliberate: silently falling back to
232    /// password login when the key is missing would turn a misconfiguration
233    /// into an undetectable downgrade of a security control an operator
234    /// believes is on.
235    ///
236    /// Deliberately its own key rather than a reuse of
237    /// [`Self::mfa_encryption_key`]: the two rotate on different schedules, and
238    /// sharing one would couple an OPAQUE outage to an unrelated TOTP
239    /// rotation. Set from `AXIAM__AUTH__OPAQUE_SESSION_KEY` (not from config
240    /// files).
241    ///
242    /// Rotating this is cheap — it invalidates exchanges in flight and nothing
243    /// else. Contrast [`Self::opaque_setup_key`], which is the expensive one.
244    #[serde(skip)]
245    pub opaque_session_key: Option<[u8; 32]>,
246    /// 256-bit AES-GCM key that encrypts each tenant's OPAQUE server key
247    /// material — the OPRF seed and the long-term AKE key pair — at rest.
248    ///
249    /// `None` means the OPAQUE endpoints answer `503`, for the same reason as
250    /// above.
251    ///
252    /// This is a **separate key from [`Self::opaque_session_key`] on purpose**,
253    /// and the reason is what rotating each one costs. The session key seals
254    /// 120 seconds of in-flight state and can be rotated on any schedule.
255    /// This key protects data at rest: rotating it means re-encrypting every
256    /// `opaque_server_setup` row, and **losing** it means every registration
257    /// record in every tenant becomes unopenable and the whole estate needs a
258    /// password reset. Sharing one key would put the cheap rotation and the
259    /// catastrophic one on the same schedule, which is how an operator ends up
260    /// doing neither. Set from `AXIAM__AUTH__OPAQUE_SETUP_KEY`.
261    #[serde(skip)]
262    pub opaque_setup_key: Option<[u8; 32]>,
263    /// When `true`, access tokens decoded without an `aud` claim are treated as
264    /// `axiam:user`. Enables a back-compat window during the Phase 4 rollout
265    /// while pre-Phase-4 tokens are still circulating. Default: `true`.
266    #[serde(default = "default_true")]
267    pub allow_missing_aud_as_user: bool,
268    /// When `true` (default), all auth cookies are marked `Secure` and are
269    /// therefore sent only over HTTPS. Set `AXIAM__AUTH__COOKIE_SECURE=false`
270    /// **only** in local HTTP development (e.g. http://localhost) — **never**
271    /// in production or staging (D-18).
272    #[serde(default = "default_true")]
273    pub cookie_secure: bool,
274    /// MFA challenge token lifetime in seconds (default: 300 = 5 minutes).
275    pub mfa_challenge_lifetime_secs: u64,
276    /// Issuer name shown in authenticator apps.
277    pub totp_issuer: String,
278    /// Max consecutive failed login attempts before lockout (default: 5).

302    #[serde(skip)]

304    /// CQ-B14: Pre-parsed Ed25519 verification key. Populated once at startup via

306    /// When `None`, they fall back to parsing from `jwt_public_key_pem`.
307    #[serde(skip)]
308    pub jwt_decoding_key: Option<Arc<DecodingKey>>,
309    /// PERF-01: Consecutive HIBP failures/timeouts before the process-wide
310    /// `HibpBreaker` trips open (default: 5). Overridable via
311    /// `AXIAM__AUTH__HIBP_BREAKER_THRESHOLD`.
312    pub hibp_breaker_threshold: u32,
313    /// PERF-01: Cooldown in seconds the `HibpBreaker` stays open before
314    /// allowing a half-open probe request (default: 30). Overridable via
315    /// `AXIAM__AUTH__HIBP_BREAKER_COOLDOWN_SECS`.
316    pub hibp_breaker_cooldown_secs: u64,
317    /// B1: Maximum number of Argon2id hash/verify operations allowed to run
318    /// concurrently, enforced by the process-wide `crypto_semaphore` shared
319    /// across the login, password-change, password-reset and PKI crypto paths.
320    ///
321    /// Each in-flight Argon2id operation allocates a ~19 MiB memory arena
322    /// (OWASP params m=19456), so unbounded concurrency is an unauthenticated
323    /// memory-DoS vector: the login benchmark pegged 2 cores and reached
324    /// ~970 MiB RSS ≈ 50 concurrent × 19 MiB arenas, against a 1024 MiB
325    /// container cap. This permit count bounds peak concurrent arenas.
326    ///
327    /// `0` means "auto" and resolves to `min(available_parallelism, 4)` at
328    /// semaphore-construction time (see [`AuthConfig::resolved_max_concurrent_hashes`]).
329    /// Override via `AXIAM__AUTH__MAX_CONCURRENT_HASHES`.
330    pub max_concurrent_hashes: usize,
331    /// B1: How long (in seconds) a request waits to acquire a crypto-hash
332    /// permit before giving up and returning a `ServiceUnavailable` (HTTP 503)
333    /// backpressure error, instead of queueing unboundedly and adding to
334    /// tail latency. Default: 5. Override via
335    /// `AXIAM__AUTH__HASH_ACQUIRE_TIMEOUT_SECS`.
336    pub hash_acquire_timeout_secs: u64,
337    /// I6: TTL, in seconds, of the process-local **session-validation cache**.
338    ///
339    /// `0` (the default) disables the cache entirely — every authenticated
340    /// request performs its own SurrealDB read to confirm the session behind the
341    /// access token's `jti` still exists (D-15 / REQ-7). A non-zero value lets
342    /// repeated requests inside the window reuse the last positive answer.
343    ///
344    /// Override via `AXIAM__AUTH__SESSION_VALIDATION_CACHE_TTL_SECS`.
345    ///
346    /// # What it buys and what it costs
347    ///
348    /// The read it removes is the reason the D7 decision cache helped gRPC
349    /// authorization checks 13.1× but REST checks only 5% in benchmark run 4:
350    /// the decision cache elides the *authorization* round-trips, this one
351    /// elides the *authentication* round-trip, and the gRPC surface never had
352    /// the latter.
353    ///
354    /// The cost is a bounded revocation window. Every session-deleting path in
355    /// `SurrealSessionRepository` invalidates the cache in the same call, so on
356    /// a **single replica** a logout takes effect immediately. Across **two or
357    /// more replicas** there is no invalidation channel, so a session revoked on
358    /// replica A stays acceptable on replicas B…N for at most this many
359    /// seconds — the same trade-off, and the same reasoning, as
360    /// `AXIAM__AUTHZ__DECISION_CACHE_TTL_SECS`. Session *expiry* is not
361    /// affected: cached entries carry the row's own `expires_at` and are
362    /// rejected exactly on time regardless of the TTL.
363    ///
364    /// Suggested starting point when enabling: `5`, matching the decision
365    /// cache's default.
366    pub session_validation_cache_ttl_secs: u64,
367}

369impl AuthConfig {
370    /// Effective issuer for JWT `iss` claims and OIDC discovery.
371    ///
372    /// Returns `oauth2_issuer_url` when set, falling back to
373    /// `jwt_issuer`. Trailing slashes are stripped so that the
374    /// OIDC discovery `issuer` exactly matches token `iss` claims
375    /// (OIDC Core §2 requires an exact string match).
376    pub fn effective_issuer(&self) -> &str {
377        if self.oauth2_issuer_url.is_empty() {
378            self.jwt_issuer.trim_end_matches('/')
379        } else {
380            self.oauth2_issuer_url.trim_end_matches('/')
381        }
382    }
383
384    /// The mTLS listener's base URL, or `None` when this deployment has no
385    /// separate one (RFC 8705 §5).
386    ///
387    /// Trailing slashes are stripped for the same reason
388    /// [`effective_issuer`](Self::effective_issuer) strips them: the value is
389    /// concatenated with endpoint paths, and `https://host//oauth2/token` is a
390    /// different URL from the one the operator meant.
391    ///
392    /// A value that is entirely whitespace answers `None` rather than `Some("")`
393    /// — an empty alias base would produce relative alias URLs, which RFC 8705
394    /// §5 does not permit and no client would resolve the way the operator
395    /// intended.
396    /// The configured default tenant, when one is set and parses as a UUID.
397    ///
398    /// An unparseable value is treated as unset rather than as an error: this
399    /// is consulted while building a *public, unauthenticated* document, and a
400    /// deployment whose operator fat-fingered the UUID should serve the
401    /// document it served before the setting existed rather than 500 for every
402    /// relying party. The mis-set value is visible in the document by its
403    /// absence, and `mtls_base_url` takes the opposite view for the opposite
404    /// reason — a bad alias actively misdirects a client, a missing tenant only
405    /// fails to help one.
406    #[must_use]
407    pub fn default_tenant_id(&self) -> Option<uuid::Uuid> {
408        let trimmed = self.oauth2_default_tenant_id.trim();
409        if trimmed.is_empty() {
410            return None;
411        }
412        uuid::Uuid::parse_str(trimmed).ok()
413    }
414
415    pub fn mtls_base_url(&self) -> Option<&str> {
416        let trimmed = self.oauth2_mtls_base_url.trim().trim_end_matches('/');
417        (!trimmed.is_empty()).then_some(trimmed)
418    }
419
420    /// CQ-B14: Parse Ed25519 keys from PEM once and cache in `Arc`.
421    ///
422    /// Call this once at startup after loading config from environment.
423    /// After this returns `Ok(())`, all token functions skip per-call PEM
424    /// parsing and use the cached keys instead.
425    pub fn resolve_keys(&mut self) -> Result<(), String> {
426        let enc = EncodingKey::from_ed_pem(self.jwt_private_key_pem.as_bytes())
427            .map_err(|e| format!("invalid JWT private key PEM: {e}"))?;
428        let dec = DecodingKey::from_ed_pem(self.jwt_public_key_pem.as_bytes())
429            .map_err(|e| format!("invalid JWT public key PEM: {e}"))?;
430        self.jwt_encoding_key = Some(Arc::new(enc));
431        self.jwt_decoding_key = Some(Arc::new(dec));
432        Ok(())
433    }
434
435    /// B1: Resolve `max_concurrent_hashes` (0 = auto) to a concrete permit
436    /// count for the crypto semaphore.
437    ///
438    /// Auto (`0`) resolves to `min(available_parallelism, 4)`: bound by the
439    /// core count so Argon2id (a CPU-bound operation) does not oversubscribe
440    /// cores, and capped at 4 so peak concurrent ~19 MiB arenas stay well
441    /// under typical container memory caps. A non-zero value is used verbatim
442    /// (a `0` after resolution is impossible — it would be a zero-permit
443    /// semaphore that deadlocks all hashing, so it is clamped to 1).
444    pub fn resolved_max_concurrent_hashes(&self) -> usize {
445        match self.max_concurrent_hashes {
446            0 => std::thread::available_parallelism()
447                .map(|n| n.get().min(4))
448                .unwrap_or(4),
449            n => n.max(1),
450        }
451    }
452}
