3use axiam_core::error::AxiamError;
4use axiam_core::models::webauthn_policy::AttestationDenyReason;
5use thiserror::Error;
6
7#[derive(Debug, Error)]
8pub enum AuthError {
9    #[error("invalid credentials")]
10    InvalidCredentials,
11
12    #[error("account is locked")]
13    AccountLocked,
14
15    #[error("account is inactive")]
16    AccountInactive,
17
18    #[error("account is pending verification")]
19    AccountPendingVerification,
20
21    #[error("MFA is required")]
22    MfaRequired,
23
24    #[error("invalid MFA code")]
25    MfaInvalidCode,
26
27    #[error("MFA is not enrolled for this user")]
28    MfaNotEnrolled,
29
30    #[error("token has expired")]
31    TokenExpired,
32
33    #[error("invalid token: {0}")]
34    TokenInvalid(String),
35
36    #[error("email verification token expired or invalid")]
37    VerificationTokenInvalid,
38
39    #[error("email already verified")]
40    EmailAlreadyVerified,
41
42    #[error("password reset token expired or invalid")]
43    ResetTokenInvalid,
44
45    #[error("federated users cannot reset passwords")]
46    FederatedUserPasswordReset,
47
48    #[error("MFA setup token expired or invalid")]
49    MfaSetupTokenInvalid,
50
51    #[error("MFA is already configured for this user")]
52    MfaAlreadyConfigured,
53
54    #[error("WebAuthn registration failed: {0}")]
55    WebauthnRegistration(String),
56
57    #[error("WebAuthn authentication failed: {0}")]
58    WebauthnAuthentication(String),
59
60    #[error("WebAuthn state token invalid or expired")]
61    WebauthnStateInvalid,
62
63    #[error("no WebAuthn credentials registered for this user")]
64    WebauthnNoCredentials,
65
66    #[error("cannot remove the last MFA method while MFA is enabled")]
67    MfaCannotRemoveLastMethod,
68
69    /// Generic cryptography failure (use typed sub-variants for new call sites).
70    #[error("cryptography error: {0}")]
71    Crypto(String),
72    /// A client-supplied OPAQUE protocol message was not well formed.
73    ///
74    /// Distinct from [`Self::Crypto`] because the two have different owners
75    /// and therefore different HTTP statuses. A malformed `KE1` is the
76    /// caller's mistake and must be a `400`; a stored record that will not
77    /// parse is AXIAM's own corruption and must be a `500`. Collapsing them —
78    /// which this originally did — meant a client sending junk got a `500`,
79    /// which reads as "the server is broken" in every dashboard and pager rule
80    /// an operator has.
81    ///
82    /// It carries no detail about *why* beyond the field name, because a
83    /// caller that could distinguish "bad point encoding" from "wrong length"
84    /// learns something about the server's parser and nothing it needs.
85    #[error("malformed OPAQUE message: {0}")]
86    OpaqueMalformed(String),
87
88    /// Key material could not be parsed or decoded (e.g. PEM/DER parse failure).
89    #[error("key parse error: {0}")]
90    CryptoKeyParse(String),
91
92    /// AES-GCM decryption failure (authentication tag mismatch or wrong key).
93    #[error("AES decryption error: {0}")]
94    CryptoAesDecrypt(String),
95
96    /// HMAC signature verification failed.
97    #[error("HMAC verification failed: {0}")]
98    CryptoHmacInvalid(String),
99
100    #[error("new password must differ from the current password")]
101    PasswordReusedCurrent,
102
103    /// X3 D7/D11/W2-D1: the registration ceremony completed but the
104    /// tenant's attestation policy (`axiam_core::models::webauthn_policy::evaluate`)
105    /// denied it, or `webauthn-rs` itself rejected the attestation as
106    /// unverifiable (W2-D1 point 3 — a `none`/self-attested credential
107    /// presented to the attested ceremony). `reason` is the machine-readable
108    /// [`AttestationDenyReason`] for the audit record; the `Display` text is
109    /// deliberately the fixed, non-specific message a user is shown — never
110    /// a raw library error string (D11).
111    #[error("this security key model is not permitted by your organization")]
112    WebauthnAttestationDenied { reason: AttestationDenyReason },
113
114    /// X1: a registered reactor refused this login on `login.post_auth`, or a
115    /// `fail_closed` reactor could not be reached.
116    ///
117    /// `reason` is the reactor's own text (or the failure that stood in for
118    /// it), carried for the audit record; the `Display` text is deliberately
119    /// the fixed, non-specific message the end user is shown — the same D11
120    /// split [`AuthError::WebauthnAttestationDenied`] makes. A reactor is
121    /// third-party code, and echoing its string to an unauthenticated caller
122    /// would let an extension author turn the login endpoint into an oracle
123    /// (or an XSS sink) without touching AXIAM.
124    #[error("login refused by policy")]
125    ReactorDenied { reason: String },
126
127    /// W2-D3: the tenant's policy requires attestation (`mode != none`) but
128    /// no attestation CA list could be built — MDS has never been ingested,
129    /// or no allowed AAGUID has a known root. Distinct from
130    /// [`AuthError::WebauthnAttestationDenied`] (a per-credential policy
131    /// decision): this is an operator-facing configuration problem that
132    /// blocks *every* attested registration for the tenant, and registration
133    /// MUST fail closed here rather than silently falling back to the
134    /// unattested ceremony.
135    #[error("attestation policy is enabled but no FIDO metadata is available")]
136    WebauthnAttestationUnavailable,
137}

139impl From<AuthError> for AxiamError {
140    fn from(err: AuthError) -> Self {
141        match err {
142            AuthError::InvalidCredentials
143            | AuthError::AccountLocked
144            | AuthError::AccountInactive
145            | AuthError::AccountPendingVerification
146            | AuthError::MfaRequired
147            | AuthError::MfaInvalidCode
148            | AuthError::MfaNotEnrolled
149            | AuthError::WebauthnRegistration(_)
150            | AuthError::WebauthnAuthentication(_)
151            | AuthError::WebauthnStateInvalid
152            | AuthError::WebauthnNoCredentials => AxiamError::AuthenticationFailed {
153                reason: err.to_string(),
154            },
155            AuthError::TokenExpired | AuthError::TokenInvalid(_) => {
156                AxiamError::AuthenticationFailed {
157                    reason: err.to_string(),
158                }
159            }
160            AuthError::VerificationTokenInvalid
161            | AuthError::EmailAlreadyVerified
162            | AuthError::ResetTokenInvalid
163            | AuthError::FederatedUserPasswordReset
164            | AuthError::MfaAlreadyConfigured
165            | AuthError::MfaCannotRemoveLastMethod => AxiamError::Validation {
166                message: err.to_string(),
167            },
168            AuthError::MfaSetupTokenInvalid => AxiamError::AuthenticationFailed {
169                reason: err.to_string(),
170            },
171            AuthError::Crypto(msg)
172            | AuthError::CryptoKeyParse(msg)
173            | AuthError::CryptoAesDecrypt(msg)
174            | AuthError::CryptoHmacInvalid(msg) => AxiamError::Crypto(msg),
175            AuthError::OpaqueMalformed(msg) => AxiamError::Validation { message: msg },
176            AuthError::PasswordReusedCurrent => AxiamError::Validation {
177                message: err.to_string(),
178            },
179            // D11: the end user sees the fixed `Display` text (never the raw
180            // `reason`); the raw `AttestationDenyReason` is for the caller to
181            // log/audit *before* this conversion happens (see
182            // `axiam_auth::webauthn`'s attested-registration path, which logs
183            // it at the point of denial).
184            AuthError::WebauthnAttestationDenied { .. } => AxiamError::AuthorizationDenied {
185                reason: err.to_string(),
186                action: Some("webauthn:register".into()),
187                resource_id: None,
188            },
189            AuthError::WebauthnAttestationUnavailable => {
190                AxiamError::ServiceUnavailable(err.to_string())
191            }
192            // The user sees the fixed `Display` text; the reactor's own reason
193            // was already written to the audit trail by the gate.
194            AuthError::ReactorDenied { .. } => AxiamError::AuthenticationFailed {
195                reason: err.to_string(),
196            },
197        }
198    }
199}
