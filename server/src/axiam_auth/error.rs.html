1//! Authentication error types.

3use axiam_core::error::AxiamError;
4use axiam_core::models::webauthn_policy::AttestationDenyReason;
5use thiserror::Error;
6
7#[derive(Debug, Error)]
8pub enum AuthError {
9    #[error("invalid credentials")]
10    InvalidCredentials,
11
12    #[error("account is locked")]
13    AccountLocked,
14
15    #[error("account is inactive")]
16    AccountInactive,
17
18    #[error("account is pending verification")]
19    AccountPendingVerification,
20
21    #[error("MFA is required")]
22    MfaRequired,
23
24    #[error("invalid MFA code")]
25    MfaInvalidCode,
26
27    #[error("MFA is not enrolled for this user")]
28    MfaNotEnrolled,
29
30    #[error("token has expired")]
31    TokenExpired,
32
33    #[error("invalid token: {0}")]
34    TokenInvalid(String),
35
36    #[error("email verification token expired or invalid")]
37    VerificationTokenInvalid,
38
39    #[error("email already verified")]
40    EmailAlreadyVerified,
41
42    #[error("password reset token expired or invalid")]
43    ResetTokenInvalid,
44
45    #[error("federated users cannot reset passwords")]
46    FederatedUserPasswordReset,
47
48    /// G-3 (T23.3.2): a local password operation — change, reset, an OPAQUE
49    /// registration — was attempted on a directory account, whose password
50    /// belongs to the tenant's LDAP / Active Directory server.
51    ///
52    /// Maps onto the existing `Validation` (`400 validation_error`) like
53    /// [`Self::FederatedUserPasswordReset`], so no SDK sees a new code; the
54    /// message says what to do instead. Raised only where the caller already
55    /// holds the account (an authenticated change, a consumed reset token), so
56    /// it reveals nothing to an unauthenticated caller; the unauthenticated
57    /// reset *request* answers a directory account exactly as it answers an
58    /// unknown address.
59    #[error("this account's password is managed by the organization's directory; change it there")]
60    DirectoryAccountPassword,
61
62    #[error("MFA setup token expired or invalid")]
63    MfaSetupTokenInvalid,
64
65    #[error("MFA is already configured for this user")]
66    MfaAlreadyConfigured,
67
68    #[error("WebAuthn registration failed: {0}")]
69    WebauthnRegistration(String),
70
71    #[error("WebAuthn authentication failed: {0}")]
72    WebauthnAuthentication(String),
73
74    #[error("WebAuthn state token invalid or expired")]
75    WebauthnStateInvalid,
76
77    #[error("no WebAuthn credentials registered for this user")]
78    WebauthnNoCredentials,
79
80    #[error("cannot remove the last MFA method while MFA is enabled")]
81    MfaCannotRemoveLastMethod,
82
83    /// Generic cryptography failure (use typed sub-variants for new call sites).
84    #[error("cryptography error: {0}")]
85    Crypto(String),
86    /// A client-supplied OPAQUE protocol message was not well formed.
87    ///
88    /// Distinct from [`Self::Crypto`] because the two have different owners
89    /// and therefore different HTTP statuses. A malformed `KE1` is the
90    /// caller's mistake and must be a `400`; a stored record that will not
91    /// parse is AXIAM's own corruption and must be a `500`. Collapsing them —
92    /// which this originally did — meant a client sending junk got a `500`,
93    /// which reads as "the server is broken" in every dashboard and pager rule
94    /// an operator has.
95    ///
96    /// It carries no detail about *why* beyond the field name, because a
97    /// caller that could distinguish "bad point encoding" from "wrong length"
98    /// learns something about the server's parser and nothing it needs.
99    #[error("malformed OPAQUE message: {0}")]
100    OpaqueMalformed(String),
101
102    /// Key material could not be parsed or decoded (e.g. PEM/DER parse failure).
103    #[error("key parse error: {0}")]
104    CryptoKeyParse(String),
105
106    /// AES-GCM decryption failure (authentication tag mismatch or wrong key).
107    #[error("AES decryption error: {0}")]
108    CryptoAesDecrypt(String),
109
110    /// HMAC signature verification failed.
111    #[error("HMAC verification failed: {0}")]
112    CryptoHmacInvalid(String),
113
114    #[error("new password must differ from the current password")]
115    PasswordReusedCurrent,
116
117    /// X3 D7/D11/W2-D1: the registration ceremony completed but the
118    /// tenant's attestation policy (`axiam_core::models::webauthn_policy::evaluate`)
119    /// denied it, or `webauthn-rs` itself rejected the attestation as
120    /// unverifiable (W2-D1 point 3 — a `none`/self-attested credential
121    /// presented to the attested ceremony). `reason` is the machine-readable
122    /// [`AttestationDenyReason`] for the audit record; the `Display` text is
123    /// deliberately the fixed, non-specific message a user is shown — never
124    /// a raw library error string (D11).
125    #[error("this security key model is not permitted by your organization")]
126    WebauthnAttestationDenied { reason: AttestationDenyReason },
127
128    /// X1: a registered reactor refused this login on `login.post_auth`, or a
129    /// `fail_closed` reactor could not be reached.
130    ///
131    /// `reason` is the reactor's own text (or the failure that stood in for
132    /// it), carried for the audit record; the `Display` text is deliberately
133    /// the fixed, non-specific message the end user is shown — the same D11
134    /// split [`AuthError::WebauthnAttestationDenied`] makes. A reactor is
135    /// third-party code, and echoing its string to an unauthenticated caller
136    /// would let an extension author turn the login endpoint into an oracle
137    /// (or an XSS sink) without touching AXIAM.
138    #[error("login refused by policy")]
139    ReactorDenied { reason: String },
140
141    /// W2-D3: the tenant's policy requires attestation (`mode != none`) but
142    /// no attestation CA list could be built — MDS has never been ingested,
143    /// or no allowed AAGUID has a known root. Distinct from
144    /// [`AuthError::WebauthnAttestationDenied`] (a per-credential policy
145    /// decision): this is an operator-facing configuration problem that
146    /// blocks *every* attested registration for the tenant, and registration
147    /// MUST fail closed here rather than silently falling back to the
148    /// unattested ceremony.
149    #[error("attestation policy is enabled but no FIDO metadata is available")]
150    WebauthnAttestationUnavailable,
151}

153impl From<AuthError> for AxiamError {
154    fn from(err: AuthError) -> Self {
155        match err {
156            AuthError::InvalidCredentials
157            | AuthError::AccountLocked
158            | AuthError::AccountInactive
159            | AuthError::AccountPendingVerification
160            | AuthError::MfaRequired
161            | AuthError::MfaInvalidCode
162            | AuthError::MfaNotEnrolled
163            | AuthError::WebauthnRegistration(_)
164            | AuthError::WebauthnAuthentication(_)
165            | AuthError::WebauthnStateInvalid
166            | AuthError::WebauthnNoCredentials => AxiamError::AuthenticationFailed {
167                reason: err.to_string(),
168            },
169            AuthError::TokenExpired | AuthError::TokenInvalid(_) => {
170                AxiamError::AuthenticationFailed {
171                    reason: err.to_string(),
172                }
173            }
174            AuthError::VerificationTokenInvalid
175            | AuthError::EmailAlreadyVerified
176            | AuthError::ResetTokenInvalid
177            | AuthError::FederatedUserPasswordReset
178            | AuthError::DirectoryAccountPassword
179            | AuthError::MfaAlreadyConfigured
180            | AuthError::MfaCannotRemoveLastMethod => AxiamError::Validation {
181                message: err.to_string(),
182            },
183            AuthError::MfaSetupTokenInvalid => AxiamError::AuthenticationFailed {
184                reason: err.to_string(),
185            },
186            AuthError::Crypto(msg)
187            | AuthError::CryptoKeyParse(msg)
188            | AuthError::CryptoAesDecrypt(msg)
189            | AuthError::CryptoHmacInvalid(msg) => AxiamError::Crypto(msg),
190            AuthError::OpaqueMalformed(msg) => AxiamError::Validation { message: msg },
191            AuthError::PasswordReusedCurrent => AxiamError::Validation {
192                message: err.to_string(),
193            },
194            // D11: the end user sees the fixed `Display` text (never the raw
195            // `reason`); the raw `AttestationDenyReason` is for the caller to
196            // log/audit *before* this conversion happens (see
197            // `axiam_auth::webauthn`'s attested-registration path, which logs
198            // it at the point of denial).
199            AuthError::WebauthnAttestationDenied { .. } => AxiamError::AuthorizationDenied {
200                reason: err.to_string(),
201                action: Some("webauthn:register".into()),
202                resource_id: None,
203            },
204            AuthError::WebauthnAttestationUnavailable => {
205                AxiamError::ServiceUnavailable(err.to_string())
206            }
207            // The user sees the fixed `Display` text; the reactor's own reason
208            // was already written to the audit trail by the gate.
209            AuthError::ReactorDenied { .. } => AxiamError::AuthenticationFailed {
210                reason: err.to_string(),
211            },
212        }
213    }
214}
