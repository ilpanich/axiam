1//! MFA method listing and deletion service.
2//!
3//! Provides a unified view of all MFA methods (TOTP + WebAuthn)
4//! and enforces the invariant that the last method cannot be removed
5//! while MFA is enabled.
6
7use axiam_core::error::AxiamResult;
8use axiam_core::models::mfa_method::{MfaMethod, MfaMethodType};
9use axiam_core::models::user::UpdateUser;
10use axiam_core::models::webauthn_credential::WebauthnCredentialType;
11use axiam_core::repository::{
12    RefreshTokenRepository, SessionRepository, UserRepository, WebauthnCredentialRepository,
13};
14use uuid::Uuid;
15
16use crate::error::AuthError;

18/// Service for listing and managing a user's MFA methods.
19///
20/// `S` is the session repository, held for [`Self::reset_mfa`] alone. The
21/// reset used to live on `AuthService`, which holds sessions but not
22/// credentials, and that split was the defect T-34's residual describes: the
23/// reset cleared the TOTP secret and revoked the sessions and left every
24/// registered passkey in place, because from where it stood there was nothing
25/// to clear them with. Putting it here — where both halves of "this account
26/// has a second factor" already live, and where `delete_method` already owns
27/// "the last method goes" — makes the eviction and the revocation one call
28/// that no caller can half-perform.
29///
30/// `R` is the OAuth2 refresh-token repository, held for the same call (D-N40,
31/// #841): ending the sessions ends every access token at its next request but
32/// not an OAuth2 refresh token, which the refresh grant does not check against
33/// a live session, so the reset revokes the user's refresh tokens as a
34/// password reset does.
35#[derive(Clone)]
36pub struct MfaMethodService<
37    U: UserRepository,
38    W: WebauthnCredentialRepository,
39    S: SessionRepository,
40    R: RefreshTokenRepository,
41> {
42    user_repo: U,
43    credential_repo: W,
44    session_repo: S,
45    refresh_token_repo: R,
46}
47
48impl<
49    U: UserRepository,
50    W: WebauthnCredentialRepository,
51    S: SessionRepository,
52    R: RefreshTokenRepository,
53> MfaMethodService<U, W, S, R>
54{
55    pub fn new(user_repo: U, credential_repo: W, session_repo: S, refresh_token_repo: R) -> Self {
56        Self {
57            user_repo,
58            credential_repo,
59            session_repo,
60            refresh_token_repo,
61        }
62    }

64    /// List all MFA methods registered for a user.
65    ///
66    /// Returns TOTP (if configured) and all WebAuthn credentials,
67    /// without exposing any secrets.
68    pub async fn list_methods(
69        &self,
70        tenant_id: Uuid,
71        user_id: Uuid,
72    ) -> AxiamResult<Vec<MfaMethod>> {
73        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
74        let mut methods = Vec::new();
75
76        // TOTP method (if enrolled and enabled).
77        if user.mfa_enabled && user.mfa_secret.is_some() {
78            methods.push(MfaMethod {
79                method_id: "totp".into(),
80                method_type: MfaMethodType::Totp,
81                name: "TOTP Authenticator".into(),
82                created_at: user.created_at,
83                last_used_at: None,
84            });
85        }
86
87        // WebAuthn credentials.
88        let credentials = self
89            .credential_repo
90            .list_by_user(tenant_id, user_id)
91            .await?;
92        for cred in credentials {
93            let method_type = match cred.credential_type {
94                WebauthnCredentialType::Passkey => MfaMethodType::Passkey,
95                WebauthnCredentialType::SecurityKey => MfaMethodType::SecurityKey,
96            };
97            methods.push(MfaMethod {
98                method_id: cred.id.to_string(),
99                method_type,
100                name: cred.name,
101                created_at: cred.created_at,
102                last_used_at: cred.last_used_at,
103            });
104        }
105
106        Ok(methods)
107    }

109    /// Return deduplicated method type strings for available methods.
110    pub async fn available_method_types(
111        &self,
112        tenant_id: Uuid,
113        user_id: Uuid,
114    ) -> AxiamResult<Vec<String>> {
115        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
116        let mut types = Vec::new();
117
118        // `mfa_enabled` is half of this condition, not decoration.
119        // [`AuthService::enroll_mfa`] writes the secret and leaves the flag
120        // off; [`AuthService::confirm_mfa`] is what turns it on. So a secret
121        // present while the flag is off is an enrollment the user never
122        // completed, and offering `totp` for it would list a factor the user
123        // never proved they hold. Same condition [`Self::list_methods`] uses.
124        if user.mfa_enabled && user.mfa_secret.is_some() {
125            types.push("totp".into());
126        }
127
128        let count = self
129            .credential_repo
130            .count_by_user(tenant_id, user_id)
131            .await?;
132        if count > 0 {
133            types.push("webauthn".into());
134        }
135
136        Ok(types)
137    }

139    /// Make MFA active because a factor was just enrolled.
140    ///
141    /// The counterpart of the disable in [`Self::delete_method`]: that one
142    /// turns `mfa_enabled` off when the last method goes, this one turns it on
143    /// when the first one arrives. Between them the flag means what it says —
144    /// *this account has a second factor* — rather than *this account has a
145    /// confirmed TOTP secret*, which is all it meant while `confirm_mfa` was
146    /// the only writer.
147    ///
148    /// That gap was the defect: `AuthService::login` gates the MFA challenge on
149    /// `mfa_enabled` alone, so a user who enrolled a passkey or a security key
150    /// and no TOTP signed in with a password and nothing else. The factor was
151    /// listed on the profile page, accepted at `/auth/webauthn/authenticate`,
152    /// and never asked for.
153    ///
154    /// # Why an abandoned TOTP enrollment is cleared rather than adopted
155    ///
156    /// [`AuthService::enroll_mfa`] writes `mfa_secret` and deliberately leaves
157    /// `mfa_enabled` false; [`AuthService::confirm_mfa`] verifies a code and
158    /// only then sets the flag. A secret sitting under a false flag is
159    /// therefore an enrollment nobody finished — the user scanned nothing, or
160    /// scanned it and could not produce a code.
161    ///
162    /// Turning the flag on for a *different* factor would promote that secret
163    /// to a live second factor in the same write, because every reader
164    /// downstream ([`Self::list_methods`], [`Self::available_method_types`],
165    /// `AuthService::verify_mfa`) reads the pair and not a separate
166    /// "confirmed" bit. The account would then accept codes from an
167    /// authenticator the user may never have successfully registered.
168    ///
169    /// So the pending secret is dropped. The cost is that a half-finished TOTP
170    /// enrollment has to be restarted; the alternative is a second factor that
171    /// was never confirmed, and re-running enrollment is two clicks.
172    ///
173    /// Idempotent: enrolling a second credential on an account that already has
174    /// MFA on writes nothing and returns `false`.
175    pub async fn enable_after_enrollment(
176        &self,
177        tenant_id: Uuid,
178        user_id: Uuid,
179    ) -> AxiamResult<bool> {
180        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
181        if user.mfa_enabled {
182            return Ok(false);
183        }
184
185        self.user_repo
186            .update(
187                tenant_id,
188                user_id,
189                UpdateUser {
190                    mfa_enabled: Some(true),
191                    // See the doc comment: an unconfirmed enrollment is
192                    // abandoned, not adopted. `None` here would leave it in
193                    // place, which is the whole hazard.
194                    mfa_secret: user.mfa_secret.is_some().then_some(None),
195                    ..Default::default()
196                },
197            )
198            .await?;
199
200        Ok(true)
201    }

203    /// Delete an MFA method by ID.
204    ///
205    /// - `"totp"` removes the TOTP secret.
206    /// - A UUID string removes the corresponding WebAuthn credential.
207    ///
208    /// Refuses to remove the last method when `mfa_enabled` is true.
209    pub async fn delete_method(
210        &self,
211        tenant_id: Uuid,
212        user_id: Uuid,
213        method_id: &str,
214    ) -> AxiamResult<()> {
215        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
216
217        // Count total methods.
218        let has_totp = user.mfa_enabled && user.mfa_secret.is_some();
219        let webauthn_count = self
220            .credential_repo
221            .count_by_user(tenant_id, user_id)
222            .await?;
223        let total = (if has_totp { 1u64 } else { 0 }) + webauthn_count;
224
225        // Prevent removing the last method when MFA is active.
226        if total <= 1 && user.mfa_enabled {
227            return Err(AuthError::MfaCannotRemoveLastMethod.into());
228        }
229
230        if method_id == "totp" {
231            // Remove TOTP secret.
232            self.user_repo
233                .update(
234                    tenant_id,
235                    user_id,
236                    UpdateUser {
237                        mfa_secret: Some(None),
238                        ..Default::default()
239                    },
240                )
241                .await?;
242        } else {
243            // Parse as WebAuthn credential UUID.
244            let cred_id: Uuid =
245                method_id
246                    .parse()
247                    .map_err(|_| axiam_core::error::AxiamError::NotFound {
248                        entity: "mfa_method".into(),
249                        id: method_id.into(),
250                    })?;
251
252            // Verify it belongs to this user.
253            let cred = self.credential_repo.get_by_id(tenant_id, cred_id).await?;
254            if cred.user_id != user_id {
255                return Err(axiam_core::error::AxiamError::NotFound {
256                    entity: "mfa_method".into(),
257                    id: method_id.into(),
258                });
259            }
260
261            self.credential_repo.delete(tenant_id, cred_id).await?;
262        }
263
264        // Post-delete safety: re-count actual remaining methods to
265        // handle concurrent deletions (TOCTOU).  If nothing remains,
266        // disable MFA so the user is never locked out.
267        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
268        if user.mfa_enabled {
269            let totp_remaining = if user.mfa_secret.is_some() { 1u64 } else { 0 };
270            let webauthn_remaining = self
271                .credential_repo
272                .count_by_user(tenant_id, user_id)
273                .await?;
274            if totp_remaining + webauthn_remaining == 0 {
275                self.user_repo
276                    .update(
277                        tenant_id,
278                        user_id,
279                        UpdateUser {
280                            mfa_enabled: Some(false),
281                            ..Default::default()
282                        },
283                    )
284                    .await?;
285            }
286        }
287
288        Ok(())
289    }

291    /// Reset every MFA factor a user holds and revoke their sessions and OAuth2
292    /// refresh tokens — the administrative unlock behind
293    /// `POST /api/v1/users/{id}/reset-mfa`.
294    ///
295    /// Returns how many WebAuthn credentials were evicted, which is the part a
296    /// caller could not otherwise observe and the part the audit record wants.
297    ///
298    /// # Why the credentials go, and not just the flag
299    ///
300    /// Clearing `mfa_enabled` and the TOTP secret does not remove a factor; it
301    /// removes the *challenge*. The credential rows survive, and every reader
302    /// downstream keys on them rather than on the flag: at the next login the
303    /// gate sees `mfa_enabled == false` and hands out a setup token, the user
304    /// enrols TOTP, `enable_after_enrollment` flips the flag back on, and
305    /// [`Self::available_method_types`] lists `webauthn` again because
306    /// `count_by_user` was never zero. A passkey an administrator reset the
307    /// account *because of* is a live second factor once more, with nobody
308    /// having re-registered it. That is T-34's residual, and the eviction is
309    /// what closes it.
310    ///
311    /// # Order
312    ///
313    /// Credentials first, then the user row, then the sessions, then the
314    /// OAuth2 refresh tokens. Each step narrows what an attacker holding the
315    /// old state can do, so a failure part-way leaves the account *more*
316    /// locked down rather than less: credentials gone but sessions alive is a
317    /// user who must re-enrol; sessions gone but credentials alive would be
318    /// the hole this closes.
319    ///
320    /// # The refresh tokens (D-N40, #841)
321    ///
322    /// The session revocation stops every access token at its next request,
323    /// but an OAuth2 refresh token is not checked against a live session, so
324    /// one minted before the reset — by a code grant or a native sign-in —
325    /// would keep minting access tokens for the account the administrator
326    /// just locked down. They are revoked with
327    /// [`RefreshTokenRepository::revoke_all_for_user`], exactly as a password
328    /// reset revokes them, and with the password reset's failure handling: an
329    /// error fails the reset (the handler answers `500`) rather than
330    /// reporting an unlock that left a grant live. Every step is idempotent,
331    /// so the administrator may simply retry.
332    pub async fn reset_mfa(&self, tenant_id: Uuid, user_id: Uuid) -> AxiamResult<u64> {
333        let evicted = self
334            .credential_repo
335            .delete_by_user(tenant_id, user_id)
336            .await?;
337
338        self.user_repo
339            .update(
340                tenant_id,
341                user_id,
342                UpdateUser {
343                    mfa_enabled: Some(false),
344                    mfa_secret: Some(None),
345                    ..Default::default()
346                },
347            )
348            .await?;
349
350        self.session_repo
351            .invalidate_user_sessions(tenant_id, user_id)
352            .await?;
353
354        let oauth2_revoked = self
355            .refresh_token_repo
356            .revoke_all_for_user(tenant_id, user_id)
357            .await?;
358        tracing::debug!(
359            %tenant_id, %user_id, %oauth2_revoked,
360            "reset_mfa: factors evicted, sessions and OAuth2 refresh tokens revoked"
361        );
362
363        Ok(evicted)
364    }
