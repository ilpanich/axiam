1//! Implementations of [`SecretProvider`] — where a deployment's keys actually
2//! come from.
3//!
4//! Three ship: [`EnvSecretProvider`] (the default, and what AXIAM has always
5//! done), [`FileSecretProvider`] (what Docker secrets and Kubernetes `Secret`
6//! volumes hand you), and [`VaultSecretProvider`] (HashiCorp Vault KV v2).
7//!
8//! Adding a fourth — AWS KMS, GCP Secret Manager, PKCS#11 — is a new type
9//! implementing the same trait and one arm in the composition root. Nothing in
10//! `axiam-auth`'s authentication code, and nothing in `axiam-api-rest`, needs
11//! to know it happened.
12//!
13//! # Which one is actually better, honestly
14//!
15//! `file` is **not** a security improvement over `env`. Both put the key on the
16//! same side of the trust boundary as the process that reads it: an attacker
17//! who can read a mounted secret file can generally also read
18//! `/proc/<pid>/environ`. It ships because it is what secret managers mount,
19//! not because it hardens anything.
20//!
21//! `vault` is a real improvement, for reasons that have nothing to do with
22//! encryption strength: the key is never written into a container spec, a
23//! Compose file, a CI variable or a shell history; access is authenticated per
24//! workload, audited, and revocable after the fact; and rotation is a write to
25//! Vault rather than a redeploy. The key still lands in this process's memory —
26//! it has to, to decrypt anything — so this is defence against the ways keys
27//! actually leak, not against a compromised server.
28//!
29//! The version that would survive a compromised server is envelope encryption
30//! against a Transit-style API where the plaintext key never leaves the KMS.
31//! That is a strictly larger change — every seal and open becomes a network
32//! round trip — and it is the natural next implementation of this same trait.
33

41/// Parse a key that arrived as text.
42///
43/// Accepts 64 hex characters, tolerating surrounding whitespace because a file
44/// written with `echo` has a trailing newline and an operator should not lose
45/// an afternoon to that.
46fn parse_key(name: &str, raw: &str) -> AxiamResult<[u8; 32]> {
47    let trimmed = raw.trim();
48    let bytes = hex::decode(trimmed).map_err(|_| {
49        AxiamError::Internal(format!(
50            "{name} must be 64 hex characters (32 bytes); got {} characters that are not valid hex",
51            trimmed.len()
52        ))
53    })?;
54    bytes.try_into().map_err(|v: Vec<u8>| {
55        AxiamError::Internal(format!(
56            "{name} must be exactly 32 bytes (256 bits); got {}",
57            v.len()
58        ))
59    })
60}

66/// Reads keys from `AXIAM__AUTH__<NAME>`, uppercased.
67///
68/// The default, and what every existing deployment already does.
69#[derive(Debug, Default, Clone, Copy)]
70pub struct EnvSecretProvider;
71
72impl EnvSecretProvider {
73    /// The environment variable a logical key name maps to.
74    pub fn var_name(name: &str) -> String {
75        format!("AXIAM__AUTH__{}", name.to_uppercase())
76    }
77}
78
79impl SecretProvider for EnvSecretProvider {
80    fn get_key(&self, name: &str) -> AxiamResult<Option<[u8; 32]>> {
81        match std::env::var(Self::var_name(name)) {
82            Ok(raw) => parse_key(&Self::var_name(name), &raw).map(Some),
83            // Genuinely absent, which is a configuration choice rather than a
84            // failure — see the port's module docs.
85            Err(_) => Ok(None),
86        }
87    }
88
89    fn get_secret(&self, name: &str) -> AxiamResult<Option<Zeroizing<String>>> {
90        Ok(std::env::var(Self::var_name(name)).ok().map(Zeroizing::new))
91    }
92
93    fn describe(&self) -> &'static str {
94        "env"
95    }
96}

102/// Reads each key from `<dir>/<name>`, the layout Docker secrets and Kubernetes
103/// `Secret` volumes produce.
104///
105/// A missing file is `Ok(None)`; a present-but-unreadable or malformed one is
106/// an error, because an operator who mounted a file meant for it to be used.
107#[derive(Debug, Clone)]
108pub struct FileSecretProvider {
109    dir: PathBuf,
110}
111
112impl FileSecretProvider {
113    /// Read keys from `dir`.
114    pub fn new(dir: impl Into<PathBuf>) -> Self {
115        Self { dir: dir.into() }
116    }
117
118    /// The path a logical key name maps to.
119    pub fn path_for(&self, name: &str) -> PathBuf {
120        self.dir.join(name)
121    }
122}
123
124impl SecretProvider for FileSecretProvider {
125    fn get_key(&self, name: &str) -> AxiamResult<Option<[u8; 32]>> {
126        let path: &Path = &self.path_for(name);
127        match std::fs::read_to_string(path) {
128            Ok(raw) => {
129                // `Zeroizing` so the decoded text does not outlive this call in
130                // a freed heap page. Not a strong control — the key is about to
131                // live in the process anyway — but free.
132                let raw = Zeroizing::new(raw);
133                parse_key(&path.display().to_string(), &raw).map(Some)
134            }
135            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(None),
136            Err(e) => Err(AxiamError::Internal(format!(
137                "{} exists but could not be read: {e}",
138                path.display()
139            ))),
140        }
141    }
142
143    fn get_secret(&self, name: &str) -> AxiamResult<Option<Zeroizing<String>>> {
144        let path: &Path = &self.path_for(name);
145        match std::fs::read_to_string(path) {
146            // Only a trailing newline is stripped, not all whitespace: a PEM
147            // document is multi-line and meaningful, and a pepper could
148            // legitimately end in a space. `echo` adding one newline is the
149            // case worth forgiving; guessing beyond that would corrupt a
150            // secret an operator set deliberately.
151            Ok(raw) => Ok(Some(Zeroizing::new(
152                raw.strip_suffix('\n').unwrap_or(&raw).to_string(),
153            ))),
154            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(None),
155            Err(e) => Err(AxiamError::Internal(format!(
156                "{} exists but could not be read: {e}",
157                path.display()
158            ))),
159        }
160    }
161
162    fn describe(&self) -> &'static str {
163        "file"
164    }
165}

171/// Keys fetched from HashiCorp Vault's KV v2 engine at startup, then served
172/// from memory.
173///
174/// Built by [`VaultSecretProvider::fetch`], which is the only async part of
175/// this module and the reason the trait itself can stay synchronous: a KMS
176/// belongs at composition time, not on a login.
177///
178/// `Debug` prints only the key *names* it holds, never their bytes.
179#[derive(Clone)]
180pub struct VaultSecretProvider {
181    keys: HashMap<String, [u8; 32]>,
182    secrets: HashMap<String, String>,
183}
184
185impl std::fmt::Debug for VaultSecretProvider {
186    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
187        let mut names: Vec<&str> = self
188            .keys
189            .keys()
190            .chain(self.secrets.keys())
191            .map(String::as_str)
192            .collect();
193        names.sort_unstable();
194        f.debug_struct("VaultSecretProvider")
195            .field("names", &names)
196            .finish()
197    }
198}

200/// How to reach Vault.
201///
202/// `Debug` is hand-written to redact the token. A derived one would print it,
203/// and this struct is exactly the kind of thing that ends up in a startup log
204/// line or a panic message — §7's `Sensitive` reasoning applied to config.
205#[derive(Clone, PartialEq, Eq)]
206pub struct VaultConfig {
207    /// Base address, e.g. `https://vault.internal:8200`.
208    pub address: String,
209    /// A token with read access to `mount`/`path`.
210    pub token: String,
211    /// KV v2 mount point. Vault's default is `secret`.
212    pub mount: String,
213    /// Path within the mount holding AXIAM's keys as fields.
214    pub path: String,
215}
216
217impl std::fmt::Debug for VaultConfig {
218    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
219        f.debug_struct("VaultConfig")
220            .field("address", &self.address)
221            .field("token", &"<redacted>")
222            .field("mount", &self.mount)
223            .field("path", &self.path)
224            .finish()
225    }
226}

228impl VaultSecretProvider {
229    /// Read every key in `names` from Vault, once.
230    ///
231    /// A key absent from the secret is `None` and not an error — a deployment
232    /// may legitimately store only some of them here. Vault being unreachable,
233    /// refusing the token, or returning something unparseable **is** an error,
234    /// and the composition root is expected to let it stop startup. A key this
235    /// important failing open would be worse than not booting.
236    pub async fn fetch(
237        client: &reqwest::Client,
238        config: &VaultConfig,
239        key_names: &[&str],
240        secret_names: &[&str],
241    ) -> AxiamResult<Self> {
242        let url = format!(
243            "{}/v1/{}/data/{}",
244            config.address.trim_end_matches('/'),
245            config.mount.trim_matches('/'),
246            config.path.trim_matches('/')
247        );
248
249        let response = client
250            .get(&url)
251            .header("X-Vault-Token", &config.token)
252            .send()
253            .await
254            .map_err(|e| {
255                // Deliberately does not include the token or the response body.
256                AxiamError::Internal(format!("vault: request to {url} failed: {e}"))
257            })?;
258
259        if !response.status().is_success() {
260            return Err(AxiamError::Internal(format!(
261                "vault: {url} answered {} — check the token's policy and the mount path",
262                response.status()
263            )));
264        }
265
266        let body: serde_json::Value = response
267            .json()
268            .await
269            .map_err(|e| AxiamError::Internal(format!("vault: {url} returned non-JSON: {e}")))?;
270
271        // KV v2 nests the secret under data.data; KV v1 does not. Accept both
272        // so a deployment on the older engine is not silently told its keys are
273        // absent.
274        let fields = body
275            .get("data")
276            .and_then(|d| d.get("data").or(Some(d)))
277            .ok_or_else(|| {
278                AxiamError::Internal(format!("vault: {url} returned no `data` object"))
279            })?;
280
281        let mut keys = HashMap::new();
282        for name in key_names {
283            if let Some(raw) = fields.get(*name).and_then(|v| v.as_str()) {
284                keys.insert((*name).to_string(), parse_key(name, raw)?);
285            }
286        }
287
288        let mut secrets = HashMap::new();
289        for name in secret_names {
290            if let Some(raw) = fields.get(*name).and_then(|v| v.as_str()) {
291                secrets.insert((*name).to_string(), raw.to_string());
292            }
293        }
294
295        Ok(Self { keys, secrets })
296    }
297
298    /// Build directly from already-resolved values. For tests and for a caller
299    /// that fetched by some other means.
300    pub fn from_values(keys: HashMap<String, [u8; 32]>, secrets: HashMap<String, String>) -> Self {
301        Self { keys, secrets }
302    }
303}
304
305impl SecretProvider for VaultSecretProvider {
306    fn get_key(&self, name: &str) -> AxiamResult<Option<[u8; 32]>> {
307        Ok(self.keys.get(name).copied())
308    }
309
310    fn get_secret(&self, name: &str) -> AxiamResult<Option<Zeroizing<String>>> {
311        Ok(self.secrets.get(name).cloned().map(Zeroizing::new))
312    }
313
314    fn describe(&self) -> &'static str {
315        "vault"
316    }
317}

323/// Which provider a deployment has chosen.
324#[derive(Debug, Clone, PartialEq, Eq)]
325pub enum SecretProviderKind {
326    /// `AXIAM__AUTH__<NAME>` environment variables. The default.
327    Env,
328    /// One file per key under a directory.
329    File {
330        /// Directory holding the key files.
331        dir: PathBuf,
332    },
333    /// HashiCorp Vault KV v2.
334    Vault(Box<VaultConfig>),
335}
336
337impl SecretProviderKind {
338    /// Parse the `AXIAM__AUTH__SECRET_PROVIDER` selector and its companions.
339    ///
340    /// Unknown values are refused rather than falling back to `env`. Falling
341    /// back would mean a typo in the provider name silently reverts a
342    /// deployment to reading environment variables that are probably not set,
343    /// which presents as "OPAQUE stopped working" with nothing in the logs
344    /// pointing at the cause.
345    pub fn from_env() -> AxiamResult<Self> {
346        match std::env::var("AXIAM__AUTH__SECRET_PROVIDER")
347            .unwrap_or_else(|_| "env".into())
348            .trim()
349            .to_lowercase()
350            .as_str()
351        {
352            "env" => Ok(Self::Env),
353            "file" => {
354                let dir = std::env::var("AXIAM__AUTH__SECRET_DIR").map_err(|_| {
355                    AxiamError::Internal(
356                        "AXIAM__AUTH__SECRET_PROVIDER=file requires AXIAM__AUTH__SECRET_DIR".into(),
357                    )
358                })?;
359                Ok(Self::File { dir: dir.into() })
360            }
361            "vault" => {
362                let required = |var: &str| {
363                    std::env::var(var).map_err(|_| {
364                        AxiamError::Internal(format!(
365                            "AXIAM__AUTH__SECRET_PROVIDER=vault requires {var}"
366                        ))
367                    })
368                };
369                Ok(Self::Vault(Box::new(VaultConfig {
370                    address: required("AXIAM__AUTH__VAULT_ADDR")?,
371                    token: required("AXIAM__AUTH__VAULT_TOKEN")?,
372                    mount: std::env::var("AXIAM__AUTH__VAULT_MOUNT")
373                        .unwrap_or_else(|_| "secret".into()),
374                    path: std::env::var("AXIAM__AUTH__VAULT_PATH")
375                        .unwrap_or_else(|_| "axiam".into()),
376                })))
377            }
378            other => Err(AxiamError::Internal(format!(
379                "AXIAM__AUTH__SECRET_PROVIDER=`{other}` is not a known provider \
380                 (expected `env`, `file` or `vault`)"
381            ))),
382        }
383    }
384
385    /// Build the provider this selector names, doing any network work now.
386    pub async fn build(
387        &self,
388        client: &reqwest::Client,
389        key_names: &[&str],
390        secret_names: &[&str],
391    ) -> AxiamResult<Box<dyn SecretProvider>> {
392        Ok(match self {
393            Self::Env => Box::new(EnvSecretProvider),
394            Self::File { dir } => Box::new(FileSecretProvider::new(dir.clone())),
395            Self::Vault(config) => {
396                Box::new(VaultSecretProvider::fetch(client, config, key_names, secret_names).await?)
397            }
398        })
399    }
400}

413    /// CodeQL's `rust/hardcoded-cryptographic-value` is right about shipping

416        let mut key = [0u8; 32];
417        getrandom::fill(&mut key).expect("a CSPRNG");

553    /// Serializes the tests that mutate the process environment.
554    ///
555    /// `cargo test` runs a crate's tests as threads in one process, so
556    /// `set_var` in one is visible to every other. Without this the three tests
557    /// below fail each other intermittently — which is worse than failing
558    /// outright, because it reads as flakiness in CI.
559    fn env_lock() -> std::sync::MutexGuard<'static, ()> {
560        static LOCK: std::sync::Mutex<()> = std::sync::Mutex::new(());
561        LOCK.lock().unwrap_or_else(|poisoned| poisoned.into_inner())
562    }

564    #[test]
565    fn an_unknown_provider_name_is_refused_rather_than_defaulting() {
566        // Falling back to `env` on a typo would silently revert a deployment to
567        // reading variables that are probably not set, presenting as "OPAQUE
568        // stopped working" with nothing in the logs pointing at the cause.
569        //
570        let _guard = env_lock();
571        // SAFETY: serialized by `env_lock`, variable removed before returning.
572        unsafe { std::env::set_var("AXIAM__AUTH__SECRET_PROVIDER", "valut") };
573        let err = SecretProviderKind::from_env().unwrap_err().to_string();
574        unsafe { std::env::remove_var("AXIAM__AUTH__SECRET_PROVIDER") };
575
576        assert!(
577            err.contains("valut"),
578            "the error must quote what was set: {err}"
579        );
580        assert!(err.contains("env"), "and name the valid options: {err}");
581    }
