67/// Login result — full success, MFA challenge, or MFA setup required.

155/// Authentication service.
156///
157/// Generic over repository implementations so that the auth layer
158/// has no dependency on the database crate.
159///
160/// `T` is the OAuth2 refresh-token repository. It is used by
161/// `revoke_all_sessions` and `revoke_all_sessions_except` to ensure that
162/// both session-flow and OAuth2-flow refresh tokens are revoked whenever
163/// a credential change occurs (RESEARCH §4 — "two chokepoints").
164#[derive(Clone)]
165pub struct AuthService<
166    U: UserRepository,
167    S: SessionRepository,
168    F: FederationLinkRepository,
169    T: RefreshTokenRepository,
170> {
171    user_repo: U,
172    session_repo: S,
173    federation_repo: F,
174    refresh_token_repo: T,
175    config: AuthConfig,
176    /// Bounding semaphore (CQ-B02): limits concurrent Argon2 operations to prevent
177    /// CPU-bound crypto from starving the Tokio async runtime under login bursts.
178    crypto_semaphore: Arc<Semaphore>,
179    /// X1 — the `login.post_auth` interceptor chain.
180    ///
181    /// A field rather than a type parameter, and never an `Option`. `AuthService`
182    /// is already generic over four repositories and is named explicitly in
183    /// `AppState<C>`; a fifth parameter would push the broker choice into the
184    /// signature of a crate whose whole design is not to know a broker exists.
185    /// A deployment without reactors holds
186    /// [`axiam_core::models::reactor::NoopReactorGate`] here, so there is one
187    /// login path in every build rather than two that can drift.
188    reactor_gate: SharedReactorGate,
189}
190
191impl<
192    U: UserRepository,
193    S: SessionRepository,
194    F: FederationLinkRepository,
195    T: RefreshTokenRepository,
196> AuthService<U, S, F, T>

230    /// Authenticate a user with username/email + password.
231    ///
232    /// Returns `LoginResult::Success` if no MFA, or
233    /// `LoginResult::MfaRequired` with a challenge token if MFA is
234    /// enabled.
235    pub async fn login(&self, input: LoginInput) -> AxiamResult<LoginResult> {
236        // 1. Look up user — try username first, then email.
237        let user = match self
238            .user_repo
239            .get_by_username(input.tenant_id, &input.username_or_email)
240            .await
241        {
242            Ok(u) => u,
243            Err(AxiamError::NotFound { .. }) => {
244                match self
245                    .user_repo
246                    .get_by_email(input.tenant_id, &input.username_or_email)
247                    .await
248                {
249                    Ok(u) => u,
250                    Err(AxiamError::NotFound { .. }) => {
251                        // SEC-026: timing equalization — run a dummy Argon2 verify so
252                        // user-not-found takes the same time as wrong-password (ASVS V2).
253                        //
254                        // B1: acquire the crypto permit with the SAME
255                        // acquire-with-timeout used by the real wrong-password
256                        // verify below (step 3), so the two branches stay
257                        // constant-time in BOTH regimes:
258                        //   * normal load — both acquire immediately and run
259                        //     exactly one Argon2id verify → 401;
260                        //   * saturation  — both hit the same timeout and return
261                        //     the same 503 backpressure error *before* hashing.
262                        // Making only the real path shed load (while this dummy
263                        // path waited unboundedly and always hashed) would make
264                        // "existing + wrong password → 503" distinguishable from
265                        // "no such user → 401" under load — reintroducing the very
266                        // enumeration oracle SEC-026 closes. Propagating the error
267                        // here keeps them indistinguishable, and also subjects the
268                        // enumeration/dummy path to the memory-DoS bound (an
269                        // attacker spamming unknown usernames is throttled too).
270                        let _permit = acquire_hash_permit(
271                            &self.crypto_semaphore,
272                            std::time::Duration::from_secs(self.config.hash_acquire_timeout_secs),
273                        )
274                        .await?;
275                        let pepper_owned = self.config.pepper.clone();
276                        let _ = tokio::task::spawn_blocking(move || {
277                            password::verify_password(
278                                "dummy",
279                                password::DUMMY_HASH,
280                                pepper_owned.as_ref().map(|p| p.expose_secret()),
281                            )
282                        })
283                        .await;
284                        return Err(AuthError::InvalidCredentials.into());
285                    }
286                    // CQ-B12: propagate real DB errors instead of swallowing them.
287                    Err(e) => return Err(e),
288                }
289            }
290            Err(e) => return Err(e),
291        };
292
293        // 2. Check temporary lockout (brute force protection).
294        if let Some(locked_until) = user.locked_until
295            && locked_until > Utc::now()
296        {
297            return Err(AuthError::InvalidCredentials.into());
298        }
299
300        // 3. Verify password — CPU-bound Argon2id runs in spawn_blocking behind semaphore (CQ-B02).
301        //    B1: bounded acquire — 503 backpressure on timeout instead of unbounded queueing.
302        let _permit = acquire_hash_permit(
303            &self.crypto_semaphore,
304            std::time::Duration::from_secs(self.config.hash_acquire_timeout_secs),
305        )
306        .await?;
307        let pw_owned = input.password.clone();
308        let hash_owned = user.password_hash.clone();
309        let pepper_owned = self.config.pepper.clone();
310        let valid = tokio::task::spawn_blocking(move || {
311            password::verify_password(
312                &pw_owned,
313                &hash_owned,
314                pepper_owned.as_ref().map(|p| p.expose_secret()),
315            )
316        })
317        .await
318        .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))?
319        .map_err(|e| AxiamError::Crypto(e.to_string()))?;
320
321        if !valid {
322            self.record_failed_login(input.tenant_id, &user, input.lockout_policy.as_ref())
323                .await?;
324            return Err(AuthError::InvalidCredentials.into());
325        }
326
327        self.complete_authenticated_login(
328            user,
329            input.tenant_id,
330            input.org_id,
331            input.ip_address,
332            input.user_agent,
333            input.mfa_policy,
334        )
335        .await
336    }

338    /// Everything a login does **after** the credential itself has been
339    /// checked: reset the failed-attempt counter, gate on account status, fire
340    /// the `login.post_auth` reactor hook, apply MFA policy, and either issue
341    /// tokens or return the appropriate challenge.
342    ///
343    /// This is a method rather than the tail of [`Self::login`] because there
344    /// is now more than one way to prove a password. OPAQUE
345    /// ([`crate::opaque::OpaqueServer::login_finish`]) establishes exactly the
346    /// same fact — "this principal knows the password" — by a different route,
347    /// and it must land in exactly the same place afterwards. Duplicating this
348    /// sequence for the OPAQUE path is how a deployment ends up with lockout
349    /// counters that only move on one of the two paths, or an MFA policy that
350    /// one of them quietly skips.
351    ///
352    /// `mfa_policy` is the tenant-effective policy, or `None` to skip
353    /// enforcement (the caller could not resolve settings).
354    pub async fn complete_authenticated_login(
355        &self,
356        user: User,
357        tenant_id: Uuid,
358        org_id: Uuid,
359        ip_address: Option<String>,
360        user_agent: Option<String>,
361        mfa_policy: Option<MfaPolicy>,
362    ) -> AxiamResult<LoginResult> {
363        // 4. Reset failed login counter on success.
364        if user.failed_login_attempts > 0 {
365            self.reset_failed_logins(tenant_id, user.id).await?;
366        }
367
368        // 5. Check account status.
369        Self::check_user_status(
370            &user.status,
371            user.created_at,
372            self.config.email_verification_grace_period_hours,
373        )?;
374
375        // 5c. X1 `login.post_auth` — after the credentials verify, before any
376        //     session or challenge is issued.
377        //
378        //     This is the placement the event name promises and the only one
379        //     that is safe. Earlier, and a reactor would be consulted about
380        //     credentials that have not been checked, which turns the hook into
381        //     a credential-probing oracle for whoever wrote the extension.
382        //     Later — after `create_session_and_tokens` — and a veto would be
383        //     refusing a session that already exists.
384        let reactor_requires_mfa = self
385            .intercept_login_post_auth(
386                tenant_id,
387                org_id,
388                &user,
389                ip_address.as_deref(),
390                user_agent.as_deref(),
391            )
392            .await?;
393
394        // 5b. MFA enforcement — if policy requires MFA but user hasn't set it up,
395        //     return a setup token (unless the user is federated).
396        //
397        //     A reactor's `require_mfa` composes with the tenant policy by
398        //     joining it, never by replacing it: it can only *add* a step-up
399        //     demand, which is why it is OR-ed in below rather than consulted
400        //     as an alternative.
401        let policy_enforces_mfa = mfa_policy
402            .as_ref()
403            .is_some_and(|policy| policy.mfa_enforced);
404        if (policy_enforces_mfa || reactor_requires_mfa) && !user.mfa_enabled {
405            let links = self
406                .federation_repo
407                .get_by_user_id(tenant_id, user.id)
408                .await?;
409            // A reactor's demand is not waived for a federated user the way the
410            // tenant policy's is. The federated carve-out exists because the
411            // upstream IdP already performed the second factor; a reactor that
412            // asked for step-up *after* seeing this login has, by definition,
413            // seen that and asked anyway.
414            if links.is_empty() || reactor_requires_mfa {
415                let setup_token = self.issue_mfa_setup_token(user.id, tenant_id, org_id)?;
416                return Ok(LoginResult::MfaSetupRequired(MfaSetupOutput {
417                    setup_token,
418                }));
419            }
420        }
421
422        // 6. Check MFA — trigger for any user with mfa_enabled, regardless
423        //    of whether they use TOTP or WebAuthn.
424        if user.mfa_enabled {
425            let challenge_token = self.issue_mfa_challenge(user.id, tenant_id, org_id)?;
426            return Ok(LoginResult::MfaRequired(MfaChallengeOutput {
427                challenge_token,
428                available_methods: Vec::new(), // populated by REST handler
429            }));
430        }
431
432        // 7. No MFA — issue tokens directly.
433        let output = self
434            .create_session_and_tokens(user.id, tenant_id, org_id, ip_address, user_agent)
435            .await?;
436
437        Ok(LoginResult::Success(output))
438    }

440    /// Refuse a sign-in by a principal whose account is not in a state that
441    /// permits one — temporary lockout, then account status.
442    ///
443    /// Every other sign-in path reaches [`Self::check_user_status`] through a
444    /// step that owns a `User` already: [`Self::login`] after the password
445    /// verifies, [`Self::verify_mfa`] after it re-fetches the user, `refresh`
446    /// after it loads the session's owner. A ceremony that authenticates
447    /// *without* any of those steps — usernameless passkey sign-in, where a
448    /// credential alone both names the user and proves them — has no such
449    /// moment, so it needs this one.
450    ///
451    /// Possession of a valid passkey is not the same question as whether the
452    /// account may sign in. Deactivating a user, locking one out for repeated
453    /// failures, or anonymising one under a GDPR erasure must all take effect
454    /// on every path, and a credential enrolled while the account was healthy
455    /// keeps verifying perfectly well afterwards.
456    ///
457    /// The lockout check is deliberately first and deliberately returns
458    /// `InvalidCredentials` rather than `AccountLocked`, matching
459    /// [`Self::login`]: which of the two a caller sees is otherwise a fact
460    /// about the account disclosed to whoever asks.
461    pub fn ensure_can_sign_in(&self, user: &User) -> Result<(), AuthError> {
462        if let Some(locked_until) = user.locked_until
463            && locked_until > Utc::now()
464        {
465            return Err(AuthError::InvalidCredentials);
466        }
467        Self::check_user_status(
468            &user.status,
469            user.created_at,
470            self.config.email_verification_grace_period_hours,
471        )
472    }

474    /// Fire `login.post_auth` for a principal whose authentication has just
475    /// succeeded, and apply the verdict (X1 / R2.2, SEC-095).
476    ///
477    /// Returns `Ok(true)` when a reactor demanded step-up MFA, which the
478    /// caller must honour; `Err(ReactorDenied)` when the login is vetoed.
479    ///
480    /// # Why this is a method rather than an inline block in [`Self::login`]
481    ///
482    /// It used to be inline, and that is exactly how SEC-095 happened: the
483    /// event fired on the password path and nowhere else, so a federated
484    /// sign-in — SAML ACS, OIDC callback — created a full session plus
485    /// access/refresh tokens without the gate ever being consulted. Nothing
486    /// scoped the event to passwords; the registry says "After credentials
487    /// verify, before session issuance: veto or require step-up MFA", and
488    /// `sdks/CONTRACT.md` §22.5 repeats that with no carve-out. An operator
489    /// who registered the feature's own worked example — an embargoed-region
490    /// login veto — got a control that was bypassed by clicking "Sign in with
491    /// Okta".
492    ///
493    /// Having ONE payload builder and ONE verdict `match` is the property that
494    /// keeps the federated paths honest: a third sign-in path cannot get a
495    /// subtly different payload, and a new [`ReactorOutcome`] variant is a
496    /// compile error at one site instead of a silent no-op at three.
497    ///
498    /// Placement at every call site is the same and is load-bearing: **after**
499    /// the credentials verify — earlier turns the hook into a
500    /// credential-probing oracle for whoever wrote the extension — and
501    /// **before** any session or challenge is issued, because a veto that
502    /// arrives after `create_session_and_tokens` is refusing a session that
503    /// already exists.
504    pub async fn intercept_login_post_auth(
505        &self,
506        tenant_id: Uuid,
507        org_id: Uuid,
508        user: &User,
509        ip_address: Option<&str>,
510        user_agent: Option<&str>,
511    ) -> AxiamResult<bool> {
512        match self
513            .reactor_gate
514            .intercept(
515                tenant_id,
516                reactor_events::LOGIN_POST_AUTH,
517                serde_json::json!({
518                    // What is being decided, never the means to act on it
519                    // elsewhere (§22.3): no password, no token, no session id
520                    // — the session does not exist yet.
521                    "user_id": user.id,
522                    "username": user.username,
523                    "tenant_id": tenant_id,
524                    "org_id": org_id,
525                    "ip_address": ip_address,
526                    "user_agent": user_agent,
527                    "mfa_enabled": user.mfa_enabled,
528                }),
529            )
530            .await
531        {
532            ReactorOutcome::Allow => Ok(false),
533            ReactorOutcome::Deny { reason } => {
534                // The gate has already audited this with the reactor's id; the
535                // reason travels in the typed error for the caller's own audit
536                // record and never reaches the HTTP response body.
537                tracing::info!(
538                    target: "axiam::reactor",
539                    tenant_id = %tenant_id,
540                    user_id = %user.id,
541                    "login vetoed by a reactor on login.post_auth"
542                );
543                Err(AuthError::ReactorDenied { reason }.into())
544            }
545            ReactorOutcome::RequireMfa => Ok(true),
546            // `login.post_auth` is veto-only: the registry marks it
547            // `mutable: false`, the reply validator refuses a `mutate` on it,
548            // and the gate re-checks. Reaching this arm would mean all three
549            // failed, so it is treated as a refusal rather than ignored.
550            ReactorOutcome::Mutate { .. } => {
551                tracing::error!(
552                    target: "axiam::reactor",
553                    tenant_id = %tenant_id,
554                    "a mutation reached the login.post_auth call site, which is a \
555                     veto-only event — refusing the login"
556                );
557                Err(AuthError::ReactorDenied {
558                    reason: "reactor attempted to mutate a veto-only event".into(),
559                }
560                .into())
561            }
562        }
563    }

565    /// [`Self::intercept_login_post_auth`] for a sign-in path that has **no**
566    /// step-up branch to route a `require_mfa` into — the SAML ACS and OIDC
567    /// callback handlers (SEC-095).
568    ///
569    /// A federated sign-in completes in one round trip: there is no
570    /// `MfaRequired` / `MfaSetupRequired` result for the caller to act on, so
571    /// a reactor's step-up demand cannot be honoured. It is **refused**, not
572    /// dropped — the same rule `reactor_hooks::impossible` applies on the REST
573    /// side. Silently ignoring it would mean a reactor that asked for a second
574    /// factor got a session with one, which is worse than an outage for the
575    /// operator who configured it: they would never learn their control did
576    /// nothing.
577    pub async fn intercept_federated_login_post_auth(
578        &self,
579        tenant_id: Uuid,
580        org_id: Uuid,
581        user: &User,
582        ip_address: Option<&str>,
583        user_agent: Option<&str>,
584    ) -> AxiamResult<()> {
585        let requires_mfa = self
586            .intercept_login_post_auth(tenant_id, org_id, user, ip_address, user_agent)
587            .await?;
588        if requires_mfa {
589            tracing::error!(
590                target: "axiam::reactor",
591                tenant_id = %tenant_id,
592                user_id = %user.id,
593                "a reactor demanded step-up MFA on a federated login.post_auth, which \
594                 this sign-in path cannot honour — refusing the login rather than \
595                 issuing a session the reactor did not authorise"
596            );
597            return Err(AuthError::ReactorDenied {
598                reason: "reactor required step-up MFA on a sign-in path that cannot \
599                         perform it"
600                    .into(),
601            }
602            .into());
603        }
604        Ok(())
605    }

607    /// Complete MFA verification after a login challenge.
608    pub async fn verify_mfa(&self, input: VerifyMfaInput) -> AxiamResult<LoginOutput> {
609        // 1. Decode the challenge token.
610        let claims = self.decode_mfa_challenge(&input.challenge_token)?;
611        let user_id: Uuid = claims
612            .sub
613            .parse()
614            .map_err(|_| AuthError::TokenInvalid("bad sub".into()))?;
615        let tenant_id: Uuid = claims
616            .tenant_id
617            .parse()
618            .map_err(|_| AuthError::TokenInvalid("bad tenant_id".into()))?;
619        let org_id: Uuid = claims
620            .org_id
621            .parse()
622            .map_err(|_| AuthError::TokenInvalid("bad org_id".into()))?;
623
624        // 2. Fetch user and verify TOTP.
625        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
626        Self::check_user_status(
627            &user.status,
628            user.created_at,
629            self.config.email_verification_grace_period_hours,
630        )?;
631
632        let encrypted_secret = user
633            .mfa_secret
634            .as_deref()
635            .ok_or(AuthError::MfaNotEnrolled)?;
636        let encryption_key = self
637            .config
638            .mfa_encryption_key
639            .as_ref()
640            .ok_or_else(|| AuthError::Crypto("MFA encryption key not configured".into()))?;
641
642        let secret_bytes = totp::decrypt_secret(encryption_key, encrypted_secret)?;
643        let (valid, used_step) = totp::verify_code_with_replay_check(
644            &secret_bytes,
645            &input.totp_code,
646            &self.config.totp_issuer,
647            &user.email,
648            user.totp_last_used_step,
649        )?;
650
651        if !valid {
652            return Err(AuthError::MfaInvalidCode.into());
653        }
654
655        // Persist the used step to prevent replay within this window
656        // (SEC-008/SECHRD-01). `update_totp_step` is an atomic
657        // compare-and-set: `Ok(false)` means the CAS was lost — either this
658        // exact step was already consumed (replay) or a concurrent
659        // submission of the same code already won. Either way, surface the
660        // SAME rejection as an invalid code so a replay/CAS-miss is
661        // indistinguishable from a wrong code.
662        let advanced = self
663            .user_repo
664            .update_totp_step(tenant_id, user_id, used_step)
665            .await?;
666        if !advanced {
667            return Err(AuthError::MfaInvalidCode.into());
668        }
669
670        // 3. Create session and issue tokens.
671        self.create_session_and_tokens(
672            user_id,
673            tenant_id,
674            org_id,
675            input.ip_address,
676            input.user_agent,
677        )
678        .await
679    }

681    /// Decode an MFA challenge token and return the embedded IDs.
682    ///
683    /// This is used by the WebAuthn REST handler to extract
684    /// `(user_id, tenant_id, org_id)` from an MFA challenge token
685    /// so it can start a WebAuthn authentication ceremony.
686    pub fn decode_mfa_challenge_ids(&self, token: &str) -> Result<(Uuid, Uuid, Uuid), AuthError> {
687        let claims = self.decode_mfa_challenge(token)?;
688        let user_id: Uuid = claims
689            .sub
690            .parse()
691            .map_err(|_| AuthError::TokenInvalid("bad sub".into()))?;
692        let tenant_id: Uuid = claims
693            .tenant_id
694            .parse()
695            .map_err(|_| AuthError::TokenInvalid("bad tenant_id".into()))?;
696        let org_id: Uuid = claims
697            .org_id
698            .parse()
699            .map_err(|_| AuthError::TokenInvalid("bad org_id".into()))?;
700        Ok((user_id, tenant_id, org_id))

703    /// Start MFA enrollment for a user (step 1 of 2).
704    ///
705    /// Generates a TOTP secret, encrypts it, stores it on the user,
706    /// but does NOT enable MFA yet — call `confirm_mfa` with a valid
707    /// code to activate.
708    pub async fn enroll_mfa(&self, tenant_id: Uuid, user_id: Uuid) -> AxiamResult<EnrollMfaOutput> {
709        let encryption_key = self
710            .config
711            .mfa_encryption_key
712            .as_ref()
713            .ok_or_else(|| AuthError::Crypto("MFA encryption key not configured".into()))?;
714
715        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
716
717        let (base32_secret, totp_uri) =
718            totp::generate_enrollment(&self.config.totp_issuer, &user.email)?;
719
720        // Parse the base32 secret to raw bytes for encryption. Under totp-rs
721        // 6.0 the fallible step is the *parse* rather than the byte extraction:
722        // `try_from_base32` rejects a malformed encoding, and `as_bytes` on the
723        // result cannot fail.
724        let secret = totp_rs::Secret::try_from_base32(&base32_secret)
725            .map_err(|e| AuthError::Crypto(format!("secret decode: {e}")))?;
726        let secret_bytes = secret.as_bytes().to_vec();
727        let encrypted = totp::encrypt_secret(encryption_key, &secret_bytes)?;
728
729        // Store encrypted secret but leave mfa_enabled = false.
730        self.user_repo
731            .update(
732                tenant_id,
733                user_id,
734                UpdateUser {
735                    mfa_secret: Some(Some(encrypted)),
736                    ..Default::default()
737                },
738            )
739            .await?;
740
741        Ok(EnrollMfaOutput {
742            secret_base32: base32_secret,
743            totp_uri,
744        })
745    }

747    /// Confirm MFA enrollment (step 2 of 2).
748    ///
749    /// The user provides a TOTP code to prove they saved the secret.
750    /// On success, `mfa_enabled` is set to `true`.
751    pub async fn confirm_mfa(
752        &self,
753        tenant_id: Uuid,
754        user_id: Uuid,
755        totp_code: &str,
756    ) -> AxiamResult<()> {
757        let encryption_key = self
758            .config
759            .mfa_encryption_key
760            .as_ref()
761            .ok_or_else(|| AuthError::Crypto("MFA encryption key not configured".into()))?;
762
763        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
764        let encrypted_secret = user
765            .mfa_secret
766            .as_deref()
767            .ok_or(AuthError::MfaNotEnrolled)?;
768
769        let secret_bytes = totp::decrypt_secret(encryption_key, encrypted_secret)?;
770        // SECHRD-01/T-24-03: reuse the same matched-step verification and
771        // atomic CAS as `verify_mfa` so replay tracking begins at
772        // enrollment-confirm time rather than at first login. A
773        // freshly-enrolled user always has `totp_last_used_step = NONE`, so
774        // this always succeeds on first use — it only closes the gap where
775        // the enrollment code itself could otherwise be replayed before the
776        // user's first full login.
777        let (valid, used_step) = totp::verify_code_with_replay_check(
778            &secret_bytes,
779            totp_code,
780            &self.config.totp_issuer,
781            &user.email,
782            user.totp_last_used_step,
783        )?;
784
785        if !valid {
786            return Err(AuthError::MfaInvalidCode.into());
787        }
788
789        // Seed totp_last_used_step atomically. A lost CAS here means the
790        // enrollment code was already consumed (e.g. a concurrent duplicate
791        // submission) — reject with the same enrollment-failure error rather
792        // than introduce a new response shape.
793        let advanced = self
794            .user_repo
795            .update_totp_step(tenant_id, user_id, used_step)
796            .await?;
797        if !advanced {
798            return Err(AuthError::MfaInvalidCode.into());
799        }
800
801        // Activate MFA.
802        self.user_repo
803            .update(
804                tenant_id,
805                user_id,
806                UpdateUser {
807                    mfa_enabled: Some(true),
808                    ..Default::default()
809                },
810            )
811            .await?;
812
813        Ok(())
814    }

816    /// Rotate a refresh token: consume the old one, verify the user
817    /// is still active, and issue a new token pair.
818    ///
819    /// # Per-stage timing (A2/J2)
820    ///
821    /// Run 5 measured this endpoint at 545/s and p50 88.8 ms, down from 839/s
822    /// and 47.5 ms in run 4 — the whole distribution shifted right with tight
823    /// medians in both runs, which is the signature of *added serialized work*
824    /// rather than of contention or of a slow tail. Refresh is the one hot
825    /// endpoint that is round-trip-bound rather than CPU-bound, so it carries
826    /// a stage breakdown (`consume_us`, `user_lookup_us`, `session_create_us`,
827    /// `token_mint_us`, `handler_total_us`) on the `auth.refresh` span and as
828    /// one DEBUG event on `target: "axiam::perf"` — the same shape
829    /// `oauth2.client_credentials` uses, and for the same reason: the next
830    /// time this number moves, the stage that moved should be readable off a
831    /// log line instead of bisected for.
832    ///
833    /// Cost is four `Instant::now()` reads and four `Span::record` calls that
834    /// are no-ops when nothing is subscribed — immaterial against a path whose
835    /// cheapest stage is a datastore round trip.
836    ///
837    /// # Datastore round trips: three, not five
838    ///
839    /// The rotation used to issue five sequential statements (session read,
840    /// expiry check, consume, user read, session create) plus a tenant read in
841    /// the REST handler above it. Two are now gone:
842    ///
843    /// - the session read and the consume are one atomic
844    ///   `consume_by_token_hash` (see that method's docs — it also removes the
845    ///   read-then-delete window rather than tolerating it);
846    /// - the handler's per-refresh tenant read is served from a TTL cache.
847    ///
848    /// **No security semantics were weakened to get there.** Single-use
849    /// rotation is still enforced by `RETURN BEFORE` (and now with no window
850    /// at all), the user-status check still happens before a new session is
851    /// minted, and an expired token is still consumed rather than left
852    /// replayable.
853    #[tracing::instrument(
854        name = "auth.refresh",
855        skip(self, input),
856        fields(
857            consume_us = tracing::field::Empty,
858            user_lookup_us = tracing::field::Empty,
859            session_create_us = tracing::field::Empty,
860            token_mint_us = tracing::field::Empty,
861            handler_total_us = tracing::field::Empty,
862        )
863    )]
864    pub async fn refresh(&self, input: RefreshInput) -> AxiamResult<RefreshOutput> {
865        let started = std::time::Instant::now();
866
867        // 1. Atomically find AND consume the session named by this token
868        //    (single-use guarantee, NEW-3). A `None` means no live session
869        //    matched — either the token was never valid, or a concurrent
870        //    refresh already consumed it. Both are the same answer to the
871        //    caller, and both abort BEFORE any new session is minted, so one
872        //    refresh token can never yield two live session lineages.
873        let t_consume = std::time::Instant::now();
874        let token_hash = token::hash_refresh_token(&input.raw_refresh_token);
875        let session = self
876            .session_repo
877            .consume_by_token_hash(input.tenant_id, &token_hash)
878            .await?
879            .ok_or_else(|| -> AxiamError {
880                AuthError::TokenInvalid("refresh token not found or already used".into()).into()
881            })?;
882        let consume_us = t_consume.elapsed().as_micros() as u64;
883
884        // 2. Reject an expired session. It has already been consumed by the
885        //    statement above, which is deliberate: an expired refresh token
886        //    must not be left behind to be replayed until the cleanup job
887        //    happens to run.
888        if session.expires_at <= Utc::now() {
889            return Err(AuthError::TokenExpired.into());
890        }
891
892        // 3. Verify user is still active.
893        let t_user = std::time::Instant::now();
894        let user = self
895            .user_repo
896            .get_by_id(input.tenant_id, session.user_id)
897            .await?;
898        Self::check_user_status(
899            &user.status,
900            user.created_at,
901            self.config.email_verification_grace_period_hours,
902        )?;
903        let user_lookup_us = t_user.elapsed().as_micros() as u64;
904
905        // 4. Create new session with rotated refresh token.
906        let t_create = std::time::Instant::now();
907        let raw_refresh = token::generate_refresh_token();
908        let new_hash = token::hash_refresh_token(&raw_refresh);
909        let expires_at =
910            Utc::now() + Duration::seconds(self.config.refresh_token_lifetime_secs as i64);
911
912        let new_session = self
913            .session_repo
914            .create(CreateSession {
915                tenant_id: input.tenant_id,
916                user_id: user.id,
917                token_hash: new_hash,
918                ip_address: input.ip_address,
919                user_agent: input.user_agent,
920                expires_at,
921            })
922            .await?;
923        let session_create_us = t_create.elapsed().as_micros() as u64;
924
925        // 5. Issue new access token (jti = new session.id per D-15).
926        let t_mint = std::time::Instant::now();
927        let access_token = token::issue_access_token(
928            user.id,
929            input.tenant_id,
930            input.org_id,
931            &[],
932            &self.config,
933            new_session.id.to_string(),
934            AUD_USER,
935        )?;
936        let token_mint_us = t_mint.elapsed().as_micros() as u64;
937
938        let handler_total_us = started.elapsed().as_micros() as u64;
939        let span = tracing::Span::current();
940        span.record("consume_us", consume_us);
941        span.record("user_lookup_us", user_lookup_us);
942        span.record("session_create_us", session_create_us);
943        span.record("token_mint_us", token_mint_us);
944        span.record("handler_total_us", handler_total_us);
945        tracing::debug!(
946            target: "axiam::perf",
947            stage = "auth.refresh",
948            consume_us,
949            user_lookup_us,
950            session_create_us,
951            token_mint_us,
952            handler_total_us,
953            "refresh stage timings (A2/J2)"
954        );
955
956        Ok(RefreshOutput {
957            access_token,
958            refresh_token: raw_refresh,
959            session_id: new_session.id,
960            expires_in: self.config.access_token_lifetime_secs,
961        })
962    }

1034    /// Change a user's password with current-password verification.
1035    ///
1036    /// Security protocol (D-14, D-21):
1037    /// 1. Verify `current_password` against the stored Argon2id hash.
1038    ///    Mismatch → `AuthenticationFailed` (maps to 401 in the REST layer).
1039    /// 2. Evaluate `new_password` against the tenant password policy.
1040    ///    Failure → `Validation` (maps to 422 in the REST layer).
1041    /// 3. Hash and store the new password.
1042    /// 4. Store old hash in password history.
1043    /// 5. Revoke all sessions except the caller's current session.
1044    /// 6. Emit audit log entry.
1045    #[allow(clippy::too_many_arguments)] // CQ-B35: http_client param added for HIBP check
1046    pub async fn change_password<H: PasswordHistoryRepository>(
1047        &self,
1048        tenant_id: Uuid,
1049        user_id: Uuid,
1050        current_session_id: Uuid,
1051        current_password: &str,
1052        new_password: &str,
1053        policy: &PasswordPolicy,
1054        history_repo: &H,
1055        http_client: Option<&reqwest::Client>, // CQ-B35: pass through to HIBP check
1056    ) -> AxiamResult<()> {
1057        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
1058
1059        // 1. Verify current password — CPU-bound, run in spawn_blocking behind semaphore (CQ-B02).
1060        //    B1: bounded acquire — held for the whole change_password call so the SEC-028
1061        //    same-password check and the history verify reuse this single permit.
1062        let _permit = acquire_hash_permit(
1063            &self.crypto_semaphore,
1064            std::time::Duration::from_secs(self.config.hash_acquire_timeout_secs),
1065        )
1066        .await?;
1067        let pw_owned = current_password.to_string();
1068        let hash_owned = user.password_hash.clone();
1069        let pepper_owned = self.config.pepper.clone();
1070        let valid = tokio::task::spawn_blocking(move || {
1071            password::verify_password(
1072                &pw_owned,
1073                &hash_owned,
1074                pepper_owned.as_ref().map(|p| p.expose_secret()),
1075            )
1076        })
1077        .await
1078        .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))?
1079        .map_err(|e| AxiamError::Crypto(e.to_string()))?;
1080
1081        if !valid {
1082            return Err(AuthError::InvalidCredentials.into());
1083        }
1084
1085        // 1b. SEC-028: Block reset to current password — reject if new password
1086        //     matches the stored hash. Runs in spawn_blocking (CPU-bound Argon2).
1087        let new_pw_owned = new_password.to_string();
1088        let current_hash_owned = user.password_hash.clone();
1089        let pepper_owned2 = self.config.pepper.clone();
1090        let is_same = tokio::task::spawn_blocking(move || {
1091            password::verify_password(
1092                &new_pw_owned,
1093                &current_hash_owned,
1094                pepper_owned2.as_ref().map(|p| p.expose_secret()),
1095            )
1096        })
1097        .await
1098        .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))?
1099        .map_err(|e| AxiamError::Crypto(e.to_string()))?;
1100        if is_same {
1101            return Err(AuthError::PasswordReusedCurrent.into());
1102        }
1103
1104        // 2. Evaluate new password against policy.
1105        let check = crate::policy::evaluate_password(
1106            new_password,
1107            self.config.pepper.as_ref().map(|p| p.expose_secret()),
1108            policy,
1109            tenant_id,
1110            user_id,
1111            history_repo,
1112            http_client,
1113        )
1114        .await?;
1115
1116        if !check.is_ok() {
1117            return Err(AxiamError::PasswordPolicy {
1118                message: check.error_message(),
1119            });
1120        }
1121
1122        // 3. Hash new password.
1123        let new_hash = password::hash_password(
1124            new_password,
1125            self.config.pepper.as_ref().map(|p| p.expose_secret()),
1126        )?;
1127
1128        // 4. Store old password in history before overwriting.
1129        history_repo
1130            .create(CreatePasswordHistoryEntry {
1131                tenant_id,
1132                user_id,
1133                password_hash: user.password_hash.clone(),
1134            })
1135            .await?;
1136
1137        // 5. Update user with new password hash.
1138        self.user_repo
1139            .update(
1140                tenant_id,
1141                user_id,
1142                UpdateUser {
1143                    password_hash: Some(new_hash),
1144                    ..Default::default()
1145                },
1146            )
1147            .await?;
1148
1149        // 6. Revoke all other sessions + OAuth2 tokens; preserve current session.
1150        self.revoke_all_sessions_except(tenant_id, user_id, current_session_id)
1151            .await?;
1152
1153        tracing::info!(
1154            %tenant_id,
1155            actor = %user_id,
1156            target = %user_id,
1157            mode = "self_service",
1158            event_type = "password_changed",
1159            "user changed their password"
1160        );
1161
1162        Ok(())
1163    }

1232    /// Check user account status, with grace period support for
1233    /// pending verification.
1234    ///
1235    /// `created_at` and `grace_period_hours` are used to determine
1236    /// if a `PendingVerification` user is still within the login
1237    /// grace period. Pass `grace_period_hours = 0` to disable the
1238    /// grace period (always reject pending users).
1239    fn check_user_status(
1240        status: &UserStatus,
1241        created_at: chrono::DateTime<Utc>,
1242        grace_period_hours: u32,
1243    ) -> Result<(), AuthError> {
1244        match status {
1245            UserStatus::Active => Ok(()),
1246            UserStatus::Locked => Err(AuthError::AccountLocked),
1247            UserStatus::Inactive => Err(AuthError::AccountInactive),
1248            // Anonymized users cannot log in — treat as inactive.
1249            UserStatus::Anonymized => Err(AuthError::AccountInactive),
1250            // Administratively deleted. Deliberately the same error as
1251            // `Inactive`: whether an account was suspended or removed is not
1252            // something an unauthenticated caller gets to learn.
1253            UserStatus::Deleted => Err(AuthError::AccountInactive),
1254            UserStatus::PendingVerification => {
1255                if grace_period_hours > 0 {
1256                    let grace_end = created_at + Duration::hours(grace_period_hours as i64);
1257                    if Utc::now() <= grace_end {
1258                        return Ok(());
1259                    }
1260                }
1261                Err(AuthError::AccountPendingVerification)
1262            }
1263        }
1264    }

1314    fn issue_mfa_challenge(
1315        &self,
1316        user_id: Uuid,
1317        tenant_id: Uuid,
1318        org_id: Uuid,
1319    ) -> Result<String, AuthError> {
1320        use jsonwebtoken::{Algorithm, EncodingKey, Header};
1321
1322        let now = Utc::now().timestamp();
1323        let claims = MfaChallengeClaims {
1324            sub: user_id.to_string(),
1325            tenant_id: tenant_id.to_string(),
1326            org_id: org_id.to_string(),
1327            purpose: "mfa_challenge".into(),
1328            iss: self.config.jwt_issuer.clone(),
1329            iat: now,
1330            exp: now + self.config.mfa_challenge_lifetime_secs as i64,
1331        };
1332
1333        let key = EncodingKey::from_ed_pem(self.config.jwt_private_key_pem.as_bytes())
1334            .map_err(|e| AuthError::Crypto(format!("bad private key: {e}")))?;
1335        let header = Header::new(Algorithm::EdDSA);
1336        jsonwebtoken::encode(&header, &claims, &key)
1337            .map_err(|e| AuthError::Crypto(format!("JWT encode: {e}")))
1338    }

1340    fn decode_mfa_challenge(&self, token: &str) -> Result<MfaChallengeClaims, AuthError> {
1341        use jsonwebtoken::{Algorithm, DecodingKey, Validation};
1342
1343        let key = DecodingKey::from_ed_pem(self.config.jwt_public_key_pem.as_bytes())
1344            .map_err(|e| AuthError::Crypto(format!("bad public key: {e}")))?;
1345
1346        let mut validation = Validation::new(Algorithm::EdDSA);
1347        validation.set_issuer(&[&self.config.jwt_issuer]);
1348        validation.set_required_spec_claims(&["sub", "exp", "iat", "iss"]);
1349
1350        let data =
1351            jsonwebtoken::decode::<MfaChallengeClaims>(token, &key, &validation).map_err(|e| {
1352                match e.kind() {
1353                    jsonwebtoken::errors::ErrorKind::ExpiredSignature => AuthError::TokenExpired,
1354                    _ => AuthError::TokenInvalid(e.to_string()),
1355                }
1356            })?;
1357
1358        if data.claims.purpose != "mfa_challenge" {
1359            return Err(AuthError::TokenInvalid("not an MFA challenge token".into()));
1360        }
1361
1362        Ok(data.claims)
1363    }

1392    /// Decode and validate an MFA setup token, returning (user_id,
1393    /// tenant_id, org_id).
1394    fn decode_mfa_setup_token(&self, token: &str) -> Result<(Uuid, Uuid, Uuid), AuthError> {
1395        use jsonwebtoken::{Algorithm, DecodingKey, Validation};
1396
1397        let key = DecodingKey::from_ed_pem(self.config.jwt_public_key_pem.as_bytes())
1398            .map_err(|e| AuthError::Crypto(format!("bad public key: {e}")))?;
1399
1400        let mut validation = Validation::new(Algorithm::EdDSA);
1401        validation.set_issuer(&[&self.config.jwt_issuer]);
1402        validation.set_required_spec_claims(&["sub", "exp", "iat", "iss"]);
1403
1404        let data =
1405            jsonwebtoken::decode::<MfaChallengeClaims>(token, &key, &validation).map_err(|e| {
1406                match e.kind() {
1407                    jsonwebtoken::errors::ErrorKind::ExpiredSignature => {
1408                        AuthError::MfaSetupTokenInvalid
1409                    }
1410                    _ => AuthError::MfaSetupTokenInvalid,
1411                }
1412            })?;
1413
1414        if data.claims.purpose != "mfa_setup" {
1415            return Err(AuthError::MfaSetupTokenInvalid);
1416        }
1417
1418        let user_id: Uuid = data
1419            .claims
1420            .sub
1421            .parse()
1422            .map_err(|_| AuthError::MfaSetupTokenInvalid)?;
1423        let tenant_id: Uuid = data
1424            .claims
1425            .tenant_id
1426            .parse()
1427            .map_err(|_| AuthError::MfaSetupTokenInvalid)?;
1428        let org_id: Uuid = data
1429            .claims
1430            .org_id
1431            .parse()
1432            .map_err(|_| AuthError::MfaSetupTokenInvalid)?;
1433
1434        Ok((user_id, tenant_id, org_id))
1435    }

1437    /// Record one failed authentication attempt against `user_id`, for the
1438    /// OPAQUE path.
1439    ///
1440    /// OPAQUE proves the password without the server seeing it, but a failed
1441    /// `KE3` is still a wrong password and must accrue toward lockout exactly
1442    /// as a wrong Argon2id verify does. Without this, turning OPAQUE on would
1443    /// silently remove brute-force protection from the accounts that adopted
1444    /// it — the opposite of what enabling an extra security layer is supposed
1445    /// to do.
1446    ///
1447    /// This is why [`crate::opaque::OpaqueRejection`] has two variants rather
1448    /// than one: the caller cannot accrue an attempt it cannot attribute.
1449    pub async fn record_failed_opaque_attempt(
1450        &self,
1451        tenant_id: Uuid,
1452        user_id: Uuid,
1453        policy: Option<&LockoutPolicy>,
1454    ) -> AxiamResult<()> {
1455        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
1456        self.record_failed_login(tenant_id, &user, policy).await
1457    }

1459    async fn record_failed_login(
1460        &self,
1461        tenant_id: Uuid,
1462        user: &axiam_core::models::user::User,
1463        policy: Option<&LockoutPolicy>,
1464    ) -> AxiamResult<()> {
1465        // D-06: delegate to the shared lockout helper — the single source of
1466        // truth for failed-attempt accrual, also called by gRPC
1467        // `UserService::validate_credentials` (SEC-026b).
1468        //
1469        // The tenant's own policy when the caller resolved one; the deployment
1470        // default only when it could not. Never no accrual at all: an
1471        // unresolvable policy must not be a way to switch lockout off.
1472        let fallback;
1473        let policy = match policy {
1474            Some(p) => p,
1475            None => {
1476                fallback = crate::lockout::policy_from_config(&self.config);
1477                &fallback
1478            }
1479        };
1480        crate::lockout::record_failed_login(&self.user_repo, policy, tenant_id, user).await
1481    }
