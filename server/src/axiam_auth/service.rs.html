291/// Authentication service.
292///
293/// Generic over repository implementations so that the auth layer
294/// has no dependency on the database crate.
295///
296/// `T` is the OAuth2 refresh-token repository. It is used by
297/// `revoke_all_sessions` and `revoke_all_sessions_except` to ensure that
298/// both session-flow and OAuth2-flow refresh tokens are revoked whenever
299/// a credential change occurs (RESEARCH §4 — "two chokepoints").
300#[derive(Clone)]
301pub struct AuthService<
302    U: UserRepository,
303    S: SessionRepository,
304    F: FederationLinkRepository,
305    T: RefreshTokenRepository,
306> {
307    user_repo: U,
308    session_repo: S,
309    federation_repo: F,
310    refresh_token_repo: T,
311    config: AuthConfig,
312    /// Bounding semaphore (CQ-B02): limits concurrent Argon2 operations to prevent
313    /// CPU-bound crypto from starving the Tokio async runtime under login bursts.
314    crypto_semaphore: Arc<Semaphore>,
315    /// X1 — the `login.post_auth` interceptor chain.
316    ///
317    /// A field rather than a type parameter, and never an `Option`. `AuthService`
318    /// is already generic over four repositories and is named explicitly in
319    /// `AppState<C>`; a fifth parameter would push the broker choice into the
320    /// signature of a crate whose whole design is not to know a broker exists.
321    /// A deployment without reactors holds
322    /// [`axiam_core::models::reactor::NoopReactorGate`] here, so there is one
323    /// login path in every build rather than two that can drift.
324    reactor_gate: SharedReactorGate,
325    /// G-3 (T23.3.2) — how a directory account's password is checked.
326    ///
327    /// An `Option`, unlike the reactor gate, because absence has a meaning the
328    /// login path must act on: with no authenticator, **no tenant can use a
329    /// directory**, and a directory account (one whose
330    /// [`User::directory_external_id`] is set) fails sign-in closed with the
331    /// generic failure — it is never checked against its local hash. The
332    /// composition root always attaches one; test harnesses that do not
333    /// exercise directories leave it out.
334    directory_authenticator: Option<SharedDirectoryAuthenticator>,
335    /// G-3 (T23.3.3) — where just-in-time provisioning, its refusals and the
336    /// linking of an account write their audit rows. Optional for the reason
337    /// the authenticator is: a harness that does not exercise directories
338    /// leaves it out, and the rows are simply not written.
339    directory_audit: Option<SharedDirectoryAuditSink>,
340    /// G-3 (T23.3.4, D-30) — applies the tenant's directory group mapping on
341    /// every successful directory sign-in, before a session or an MFA challenge
342    /// is issued. Optional for the reason the authenticator is: a harness that
343    /// does not exercise directories leaves it out and no mapping is applied.
344    /// With one attached, a mapping that cannot be applied **refuses the
345    /// sign-in** (fail closed); see [`Self::apply_directory_group_mapping`].
346    directory_group_mapper: Option<SharedDirectoryGroupMapper>,
347    /// F4 P23W3-02 (T-332) — the failed-attempt counter for login names AXIAM
348    /// holds no account for, which with just-in-time provisioning on reach the
349    /// directory. Shared by every clone of the service (one per process), and
350    /// consulted before the directory is asked; see
351    /// [`crate::unknown_name_lockout`].
352    unknown_name_lockout: Arc<crate::unknown_name_lockout::UnknownNameLockout>,
353    /// SA-13 (N16, D-3) — whether the tenant still admits its principals,
354    /// asked where every browser sign-in ends ([`Self::create_session_and_tokens`]:
355    /// password, OPAQUE, WebAuthn, federation and social login, the MFA and
356    /// setup-token completions, directory accounts) and on every refresh.
357    ///
358    /// Optional for the reason the directory authenticator is: a harness that
359    /// does not exercise tenant lifecycle leaves it out. The composition root
360    /// and `AppState` always attach the tenant repository, and with it a
361    /// suspended tenant's sign-in or refresh is refused `tenant_suspended`
362    /// with nothing issued. One tenant read per sign-in or refresh, uncached.
363    tenant_admission: Option<axiam_core::models::tenant::SharedTenantAdmission>,
364}

417    /// QR5-1 / SR5-2: the read-only lock re-check that runs **before**
418    /// [`Self::admit_tenant`] once a credential has verified.
419    ///
420    /// On a suspended tenant the admission answers `tenant_suspended`, which
421    /// is only safe to say to a guess that was *wrong*-looking until now. A
422    /// lock committed by a concurrent wrong guess while this one verified
423    /// would otherwise lose the race to the admission and let the correct
424    /// password be told apart from a wrong one. So the lock is consulted
425    /// first, without writing (`clear_failures = false`: a refused sign-in on
426    /// a suspended tenant must not reset the counter), and a lock in force is
427    /// the uniform wrong-password answer. The writing check stays at step 4 of
428    /// [`Self::complete_authenticated_login`], after the admission.
429    async fn refuse_if_locked_now(&self, tenant_id: Uuid, user_id: Uuid) -> AxiamResult<()> {
430        if self.tenant_admission.is_none() {
431            // No admission to precede: step 4 is the only lock check needed.
432            return Ok(());
433        }
434        if self
435            .user_repo
436            .admit_unless_locked(tenant_id, user_id, false)
437            .await?
438        {
439            Ok(())
440        } else {
441            Err(AuthError::InvalidCredentials.into())
442        }
443    }

496    /// Authenticate a user with username/email + password.
497    ///
498    /// Returns `LoginResult::Success` if no MFA, or
499    /// `LoginResult::MfaRequired` with a challenge token if MFA is
500    /// enabled.
501    pub async fn login(&self, input: LoginInput) -> AxiamResult<LoginResult> {
502        // 1. Look up user — try username first, then email.
503        let user = match self
504            .user_repo
505            .get_by_username(input.tenant_id, &input.username_or_email)
506            .await
507        {
508            Ok(u) => u,
509            Err(AxiamError::NotFound { .. }) => {
510                match self
511                    .user_repo
512                    .get_by_email(input.tenant_id, &input.username_or_email)
513                    .await
514                {
515                    Ok(u) => u,
516                    Err(AxiamError::NotFound { .. }) => {
517                        return self.login_unknown_user(&input).await;
518                    }
519                    // CQ-B12: propagate real DB errors instead of swallowing them.
520                    Err(e) => return Err(e),
521                }
522            }
523            Err(e) => return Err(e),
524        };
525
526        // 2. Check temporary lockout (brute force protection).
527        //
528        //    Before the directory branch below, deliberately: a locked account
529        //    is refused without the directory hearing anything, so AXIAM cannot
530        //    be used to run up the failed-bind count that locks the account in
531        //    Active Directory (T-302).
532        //
533        //    It still costs one verify under the hash permit, against the dummy
534        //    hash and never the real one (T-469, SEC-026): a lockout the
535        //    attacker's own failures triggered must not answer faster than an
536        //    unknown name — an enumeration oracle — nor `401` where every
537        //    hashing branch answers `503` under saturation. A correct password
538        //    during the lockout neither succeeds nor shows.
539        if let Some(locked_until) = user.locked_until
540            && locked_until > Utc::now()
541        {
542            self.equalising_dummy_verify().await?;
543            return Err(AuthError::InvalidCredentials.into());
544        }
545
546        // 2b. A directory account (G-3) is authenticated by its directory and
547        //     by nothing else — never by the local hash, which is an unusable
548        //     random one in any case.
549        if user.is_directory_account() {
550            return self.login_directory_account(user, input).await;
551        }
552
553        // 3. Verify password — CPU-bound Argon2id runs in spawn_blocking behind semaphore (CQ-B02).
554        //    B1: bounded acquire — 503 backpressure on timeout instead of unbounded queueing.
555        let _permit = acquire_hash_permit(
556            &self.crypto_semaphore,
557            std::time::Duration::from_secs(self.config.hash_acquire_timeout_secs),
558        )
559        .await?;
560        let pw_owned = input.password.clone();
561        let hash_owned = user.password_hash.clone();
562        let pepper_owned = self.config.pepper.clone();
563        let valid = tokio::task::spawn_blocking(move || {
564            password::verify_password(
565                &pw_owned,
566                &hash_owned,
567                pepper_owned.as_ref().map(|p| p.expose_secret()),
568            )
569        })
570        .await
571        .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))?
572        .map_err(|e| AxiamError::Crypto(e.to_string()))?;
573
574        if !valid {
575            self.record_failed_login(input.tenant_id, &user, input.lockout_policy.as_ref())
576                .await?;
577            return Err(AuthError::InvalidCredentials.into());
578        }
579
580        self.complete_authenticated_login(
581            user,
582            input.tenant_id,
583            input.org_id,
584            input.ip_address,
585            input.user_agent,
586            input.mfa_policy,
587        )
588        .await
589    }

591    /// The answer to a login name that matches no local account.
592    ///
593    /// SEC-026: run one dummy Argon2id verify so "no such user" costs what
594    /// "wrong password" costs, then refuse with the generic failure.
595    ///
596    /// # The just-in-time provisioning seam (G-3, T23.3.3)
597    ///
598    /// This is the one place a directory account that does not exist locally
599    /// yet is created. When the tenant has an enabled directory with
600    /// `jit_provisioning`, the typed name and password are authenticated
601    /// against it — through the same `directory_authenticator` that
602    /// [`Self::login_directory_account`] uses, timed beside the same dummy
603    /// verify and under the same hash permit — and on success the account is
604    /// created `Active` and marked in one write, and the login continues
605    /// through [`Self::complete_authenticated_login`] (MFA policy as for any
606    /// user, `amr = [pwd]`). A second first login for the same entry finds the
607    /// first one's account.
608    ///
609    /// **Every other outcome is exactly the unknown-user answer**, with the
610    /// same error and the same dummy-verify cost: no directory, a tenant
611    /// without `jit_provisioning`, a disabled directory, a wrong password, an
612    /// entry not found, a directory that is down, and — D-28 — an entry that
613    /// would collide with a local account (which also writes an audit row).
614    /// A tenant without a directory observes nothing new. The one thing that
615    /// differs is the directory's own latency on a tenant that has one,
616    /// recorded in T-301.
617    async fn login_unknown_user(&self, input: &LoginInput) -> AxiamResult<LoginResult> {
618        match self
619            .authenticate_unknown_name_against_directory(input)
620            .await?
621        {
622            Some(identity) => self.provision_directory_account(input, identity).await,
623            None => Err(AuthError::InvalidCredentials.into()),
624        }
625    }

627    /// One Argon2id verify against the process-wide dummy hash ([`password::equalising_dummy_verify`]), to make a branch
628    /// that verifies no real hash cost what one that does costs (SEC-026).
629    ///
630    /// B1: the crypto permit is acquired with the SAME acquire-with-timeout the
631    /// real verify uses, so the branches stay indistinguishable in both regimes
632    /// — under normal load both run exactly one verify and answer `401`; under
633    /// saturation both hit the same timeout and answer the same `503`
634    /// backpressure error *before* hashing. Making only the real path shed
635    /// load would make "existing account, wrong password → 503" distinguishable
636    /// from "no such account → 401" under load — the oracle SEC-026 closes —
637    /// and it also subjects these branches to the memory bound, so an attacker
638    /// spamming unknown names is throttled too.
639    async fn equalising_dummy_verify(&self) -> AxiamResult<()> {
640        let permit = acquire_hash_permit(
641            &self.crypto_semaphore,
642            std::time::Duration::from_secs(self.config.hash_acquire_timeout_secs),
643        )
644        .await?;
645        self.dummy_verify_holding(permit).await;
646        Ok(())
647    }

649    /// The dummy verify itself, under a permit the caller already holds; the
650    /// permit is released when the verify finishes.
651    async fn dummy_verify_holding(&self, permit: tokio::sync::SemaphorePermit<'_>) {
652        let pepper_owned = self.config.pepper.clone();
653        let _ = tokio::task::spawn_blocking(move || {
654            password::equalising_dummy_verify(pepper_owned.as_ref().map(|p| p.expose_secret()))
655        })
656        .await;
657        drop(permit);
658    }

660    /// Sign-in for a directory account (G-3, T23.3.2).
661    ///
662    /// The directory decides: the presented password is checked by binding to
663    /// the tenant's directory as the account's entry, through the
664    /// [`DirectoryAuthenticator`](axiam_core::models::directory::DirectoryAuthenticator)
665    /// the composition root injected. Everything AXIAM puts in front of a
666    /// password still applies, in the same order as for a local account:
667    ///
668    /// * the temporary lockout was checked by the caller, **before** the
669    ///   directory is contacted, so a locked account cannot be used to lock the
670    ///   account in Active Directory (T-302);
671    /// * an account whose status refuses sign-in, an empty password (an
672    ///   RFC 4513 unauthenticated bind), and a deployment with no authenticator
673    ///   are all refused **without** contacting the directory;
674    /// * a failed bind increments the same counter a wrong local password does,
675    ///   and a success resets it (in [`Self::complete_authenticated_login`]);
676    /// * an unavailable, misconfigured or disabled directory fails closed with
677    ///   the generic failure and **never** falls back to the local hash, and it
678    ///   does not count against the account — the user did nothing wrong.
679    ///
680    /// The directory's answer must be **the entry this account is bound to**:
681    /// the identifier it returns is compared with the account's
682    /// `directory_external_id`, so a login name that resolves to another entry
683    /// in the directory (a renamed account, a colliding `uid`) cannot sign in as
684    /// this one even with that other entry's correct password.
685    ///
686    /// # Enumeration
687    ///
688    /// Every refusal here is `InvalidCredentials`, the answer an unknown name
689    /// and a wrong local password get, and every branch runs the same
690    /// equalising dummy Argon2id verify the unknown-name branch runs — on the
691    /// bind path concurrently with the bind, under the same permit and the same
692    /// `503` backpressure rule. What remains is the directory's own latency:
693    /// when a bind takes longer than one Argon2id verify (two TLS handshakes and
694    /// three round trips usually do), a directory account answers measurably
695    /// later than a local or unknown one. That is recorded, open to tuning and
696    /// bounded by the login rate limits, in T-301.
697    async fn login_directory_account(
698        &self,
699        user: User,
700        input: LoginInput,
701    ) -> AxiamResult<LoginResult> {
702        let status_permits = Self::check_user_status(
703            &user.status,
704            user.created_at,
705            self.config.email_verification_grace_period_hours,
706        )
707        .is_ok();
708        let authenticator = match (&self.directory_authenticator, status_permits) {
709            (Some(authenticator), true) if !input.password.is_empty() => Arc::clone(authenticator),
710            (authenticator, _) => {
711                // Refused before the directory hears anything.
712                self.equalising_dummy_verify().await?;
713                if authenticator.is_none() {
714                    tracing::warn!(
715                        target: "axiam::directory",
716                        tenant_id = %input.tenant_id,
717                        user_id = %user.id,
718                        "directory account sign-in refused: no directory authenticator is \
719                         configured in this deployment"
720                    );
721                } else if status_permits {
722                    // An empty password is a wrong password, and counts as one.
723                    self.record_failed_login(input.tenant_id, &user, input.lockout_policy.as_ref())
724                        .await?;
725                }
726                return Err(AuthError::InvalidCredentials.into());
727            }
728        };
729
730        // The equalising verify runs beside the bind, holding a hash permit
731        // acquired first — so saturation answers `503` exactly where the local
732        // path would, and before the directory is contacted.
733        let permit = acquire_hash_permit(
734            &self.crypto_semaphore,
735            std::time::Duration::from_secs(self.config.hash_acquire_timeout_secs),
736        )
737        .await?;
738        let ((), outcome) = tokio::join!(
739            self.dummy_verify_holding(permit),
740            authenticator.authenticate(input.tenant_id, &user.username, &input.password),
741        );
742
743        let marker = user.directory_external_id.as_deref().unwrap_or_default();
744        match outcome {
745            Ok(identity) if identity.external_id.eq_ignore_ascii_case(marker) => {
746                // QR3-1 (SA-13, N16): the credential has verified; a suspended
747                // tenant is refused now, before its memberships are rewritten.
748                //
749                // Deliberately repeated by `complete_authenticated_login`
750                // below: this pair must run before `apply_directory_group_mapping`
751                // writes, which that later call cannot guard. The cost is one
752                // lock read and one tenant read per directory sign-in.
753                self.refuse_if_locked_now(input.tenant_id, user.id).await?;
754                self.admit_tenant(input.tenant_id).await?;
755                // G-3 (T23.3.4): the directory's groups become AXIAM memberships
756                // *before* anything is issued — so a removal in the directory
757                // takes effect at this sign-in, and a mapping that cannot be
758                // applied is a refused sign-in, never a stale session.
759                self.apply_directory_group_mapping(&input, &user, &identity.dn)
760                    .await?;
761                self.complete_authenticated_login(
762                    user,
763                    input.tenant_id,
764                    input.org_id,
765                    input.ip_address,
766                    input.user_agent,
767                    input.mfa_policy,
768                )
769                .await
770            }
771            Ok(_) => {
772                tracing::warn!(
773                    target: "axiam::directory",
774                    tenant_id = %input.tenant_id,
775                    user_id = %user.id,
776                    "directory sign-in refused: the entry that answered is not the one \
777                     this account is bound to"
778                );
779                self.record_failed_login(input.tenant_id, &user, input.lockout_policy.as_ref())
780                    .await?;
781                Err(AuthError::InvalidCredentials.into())
782            }
783            Err(DirectoryAuthError::InvalidCredentials) => {
784                self.record_failed_login(input.tenant_id, &user, input.lockout_policy.as_ref())
785                    .await?;
786                Err(AuthError::InvalidCredentials.into())
787            }
788            Err(other) => {
789                tracing::info!(
790                    target: "axiam::directory",
791                    tenant_id = %input.tenant_id,
792                    user_id = %user.id,
793                    outcome = ?other,
794                    "directory sign-in refused without counting against the account"
795                );
796                Err(AuthError::InvalidCredentials.into())
797            }
798        }
799    }

801    /// Everything a login does **after** the credential itself has been
802    /// checked: reset the failed-attempt counter, gate on account status, fire
803    /// the `login.post_auth` reactor hook, apply MFA policy, and either issue
804    /// tokens or return the appropriate challenge.
805    ///
806    /// This is a method rather than the tail of [`Self::login`] because there
807    /// is now more than one way to prove a password. OPAQUE
808    /// ([`crate::opaque::OpaqueServer::login_finish`]) establishes exactly the
809    /// same fact — "this principal knows the password" — by a different route,
810    /// and it must land in exactly the same place afterwards. Duplicating this
811    /// sequence for the OPAQUE path is how a deployment ends up with lockout
812    /// counters that only move on one of the two paths, or an MFA policy that
813    /// one of them quietly skips.
814    ///
815    /// `mfa_policy` is the tenant-effective policy, or `None` to skip
816    /// enforcement (the caller could not resolve settings).
817    pub async fn complete_authenticated_login(
818        &self,
819        user: User,
820        tenant_id: Uuid,
821        org_id: Uuid,
822        ip_address: Option<String>,
823        user_agent: Option<String>,
824        mfa_policy: Option<MfaPolicy>,
825    ) -> AxiamResult<LoginResult> {
826        // 3b. QR3-1 (SA-13, N16, D-3): a tenant that does not admit its
827        //     principals is refused here, once the credential has verified
828        //     and before anything below writes or calls out — the lockout
829        //     bookkeeping, the `login.post_auth` reactor, a challenge or a
830        //     session. After the credential, so a wrong password on a
831        //     suspended tenant still answers as a wrong password (no oracle);
832        //     before the side effects, because a suspended tenant admits no
833        //     writes from its principals. The directory paths run the same
834        //     check before their own writes (JIT provisioning, group mapping).
835        self.refuse_if_locked_now(tenant_id, user.id).await?;
836        self.admit_tenant(tenant_id).await?;
837
838        // 4. Admit the sign-in unless a lockout is in force *now*, resetting
839        //    the failed-login counter in the same conditional write (item 5 of
840        //    the 1.1.0 atomicity audit). The lock was checked when the account
841        //    was read, before the credential was verified; guesses sent
842        //    together at threshold − 1 all pass that check, and a wrong one
843        //    may lock the account while this one verifies. The reset used to
844        //    clear that lock and issue a session. Now a lock committed first
845        //    stands, and the sign-in is refused as a wrong password is — the
846        //    same answer `login` gives a locked account (no oracle) — with no
847        //    session, challenge or token issued. Every path that proves a
848        //    credential lands here: password, OPAQUE, directory bind, JIT.
849        //
850        //    SA-43 (owner decision 2026-11-24): for an account with MFA on, the
851        //    password is only the first of two factors and does NOT reset the
852        //    counter — the completed second factor does (`verify_mfa`, the
853        //    WebAuthn second-factor finish). Otherwise whoever knows the
854        //    password could zero the counter between rounds of TOTP guesses.
855        //    The lock check itself is unchanged; accounts without MFA are
856        //    reset here exactly as before.
857        let clear_failures =
858            !user.mfa_enabled && (user.failed_login_attempts > 0 || user.locked_until.is_some());
859        if !self
860            .user_repo
861            .admit_unless_locked(tenant_id, user.id, clear_failures)
862            .await?
863        {
864            return Err(AuthError::InvalidCredentials.into());
865        }
866
867        // 5. Check account status.
868        Self::check_user_status(
869            &user.status,
870            user.created_at,
871            self.config.email_verification_grace_period_hours,
872        )?;
873
874        // 5c. X1 `login.post_auth` — after the credentials verify, before any
875        //     session or challenge is issued.
876        //
877        //     This is the placement the event name promises and the only one
878        //     that is safe. Earlier, and a reactor would be consulted about
879        //     credentials that have not been checked, which turns the hook into
880        //     a credential-probing oracle for whoever wrote the extension.
881        //     Later — after `create_session_and_tokens` — and a veto would be
882        //     refusing a session that already exists.
883        let reactor_requires_mfa = self
884            .intercept_login_post_auth(
885                tenant_id,
886                org_id,
887                &user,
888                ip_address.as_deref(),
889                user_agent.as_deref(),
890            )
891            .await?;
892
893        // 5b. MFA enforcement — if policy requires MFA but user hasn't set it up,
894        //     return a setup token (unless the user is federated).
895        //
896        //     A reactor's `require_mfa` composes with the tenant policy by
897        //     joining it, never by replacing it: it can only *add* a step-up
898        //     demand, which is why it is OR-ed in below rather than consulted
899        //     as an alternative.
900        let policy_enforces_mfa = mfa_policy
901            .as_ref()
902            .is_some_and(|policy| policy.mfa_enforced);
903        if (policy_enforces_mfa || reactor_requires_mfa) && !user.mfa_enabled {
904            let links = self
905                .federation_repo
906                .get_by_user_id(tenant_id, user.id)
907                .await?;
908            // A reactor's demand is not waived for a federated user the way the
909            // tenant policy's is. The federated carve-out exists because the
910            // upstream IdP already performed the second factor; a reactor that
911            // asked for step-up *after* seeing this login has, by definition,
912            // seen that and asked anyway.
913            if links.is_empty() || reactor_requires_mfa {
914                let setup_token = self.issue_mfa_setup_token(
915                    user.id,
916                    tenant_id,
917                    org_id,
918                    user.credential_version,
919                )?;
920                return Ok(LoginResult::MfaSetupRequired(MfaSetupOutput {
921                    setup_token,
922                }));
923            }
924        }
925
926        // 6. Check MFA — trigger for any user with mfa_enabled, regardless
927        //    of whether they use TOTP or WebAuthn.
928        if user.mfa_enabled {
929            let challenge_token =
930                self.issue_mfa_challenge(user.id, tenant_id, org_id, user.credential_version)?;
931            return Ok(LoginResult::MfaRequired(MfaChallengeOutput {
932                challenge_token,
933                available_methods: Vec::new(), // populated by REST handler
934            }));
935        }
936
937        // 7. No MFA — issue tokens directly.
938        let output = self
939            .create_session_and_tokens(
940                user.id,
941                tenant_id,
942                org_id,
943                ip_address,
944                user_agent,
945                // A password and nothing else: this branch is reached only
946                // when no second factor was required (RFC 8176 `pwd`).
947                AuthenticationEvidence::now(vec![Amr::Pwd]),
948                // SR10-1: the version read with the credential that was just
949                // verified (the `user` snapshot), checked by the session write.
950                Some(user.credential_version),
951            )
952            .await?;
953
954        Ok(LoginResult::Success(output))
955    }

957    /// Refuse a sign-in by a principal whose account is not in a state that
958    /// permits one — temporary lockout, then account status.
959    ///
960    /// Every other sign-in path reaches [`Self::check_user_status`] through a
961    /// step that owns a `User` already: [`Self::login`] after the password
962    /// verifies, [`Self::verify_mfa`] after it re-fetches the user, `refresh`
963    /// after it loads the session's owner. A ceremony that authenticates
964    /// *without* any of those steps — usernameless passkey sign-in, where a
965    /// credential alone both names the user and proves them — has no such
966    /// moment, so it needs this one.
967    ///
968    /// Possession of a valid passkey is not the same question as whether the
969    /// account may sign in. Deactivating a user, locking one out for repeated
970    /// failures, or anonymising one under a GDPR erasure must all take effect
971    /// on every path, and a credential enrolled while the account was healthy
972    /// keeps verifying perfectly well afterwards.
973    ///
974    /// The lockout check is deliberately first and deliberately returns
975    /// `InvalidCredentials` rather than `AccountLocked`, matching
976    /// [`Self::login`]: which of the two a caller sees is otherwise a fact
977    /// about the account disclosed to whoever asks.
978    pub fn ensure_can_sign_in(&self, user: &User) -> Result<(), AuthError> {
979        if let Some(locked_until) = user.locked_until
980            && locked_until > Utc::now()
981        {
982            return Err(AuthError::InvalidCredentials);
983        }
984        Self::check_user_status(
985            &user.status,
986            user.created_at,
987            self.config.email_verification_grace_period_hours,
988        )
989    }

991    /// Fire `login.post_auth` for a principal whose authentication has just
992    /// succeeded, and apply the verdict (X1 / R2.2, SEC-095).
993    ///
994    /// Returns `Ok(true)` when a reactor demanded step-up MFA, which the
995    /// caller must honour; `Err(ReactorDenied)` when the login is vetoed.
996    ///
997    /// # Why this is a method rather than an inline block in [`Self::login`]
998    ///
999    /// It used to be inline, and that is exactly how SEC-095 happened: the
1000    /// event fired on the password path and nowhere else, so a federated
1001    /// sign-in — SAML ACS, OIDC callback — created a full session plus
1002    /// access/refresh tokens without the gate ever being consulted. Nothing
1003    /// scoped the event to passwords; the registry says "After credentials
1004    /// verify, before session issuance: veto or require step-up MFA", and
1005    /// `sdks/CONTRACT.md` §22.5 repeats that with no carve-out. An operator
1006    /// who registered the feature's own worked example — an embargoed-region
1007    /// login veto — got a control that was bypassed by clicking "Sign in with
1008    /// Okta".
1009    ///
1010    /// Having ONE payload builder and ONE verdict `match` is the property that
1011    /// keeps the federated paths honest: a third sign-in path cannot get a
1012    /// subtly different payload, and a new [`ReactorOutcome`] variant is a
1013    /// compile error at one site instead of a silent no-op at three.
1014    ///
1015    /// Placement at every call site is the same and is load-bearing: **after**
1016    /// the credentials verify — earlier turns the hook into a
1017    /// credential-probing oracle for whoever wrote the extension — and
1018    /// **before** any session or challenge is issued, because a veto that
1019    /// arrives after `create_session_and_tokens` is refusing a session that
1020    /// already exists.
1021    pub async fn intercept_login_post_auth(
1022        &self,
1023        tenant_id: Uuid,
1024        org_id: Uuid,
1025        user: &User,
1026        ip_address: Option<&str>,
1027        user_agent: Option<&str>,
1028    ) -> AxiamResult<bool> {
1029        match self
1030            .reactor_gate
1031            .intercept(
1032                tenant_id,
1033                reactor_events::LOGIN_POST_AUTH,
1034                serde_json::json!({
1035                    // What is being decided, never the means to act on it
1036                    // elsewhere (§22.3): no password, no token, no session id
1037                    // — the session does not exist yet.
1038                    "user_id": user.id,
1039                    "username": user.username,
1040                    "tenant_id": tenant_id,
1041                    "org_id": org_id,
1042                    "ip_address": ip_address,
1043                    "user_agent": user_agent,
1044                    "mfa_enabled": user.mfa_enabled,
1045                }),
1046            )
1047            .await
1048        {
1049            ReactorOutcome::Allow => Ok(false),
1050            ReactorOutcome::Deny { reason } => {
1051                // The gate has already audited this with the reactor's id; the
1052                // reason travels in the typed error for the caller's own audit
1053                // record and never reaches the HTTP response body.
1054                tracing::info!(
1055                    target: "axiam::reactor",
1056                    tenant_id = %tenant_id,
1057                    user_id = %user.id,
1058                    "login vetoed by a reactor on login.post_auth"
1059                );
1060                Err(AuthError::ReactorDenied { reason }.into())
1061            }
1062            ReactorOutcome::RequireMfa => Ok(true),
1063            // `login.post_auth` is veto-only: the registry marks it
1064            // `mutable: false`, the reply validator refuses a `mutate` on it,
1065            // and the gate re-checks. Reaching this arm would mean all three
1066            // failed, so it is treated as a refusal rather than ignored.
1067            ReactorOutcome::Mutate { .. } => {
1068                tracing::error!(
1069                    target: "axiam::reactor",
1070                    tenant_id = %tenant_id,
1071                    "a mutation reached the login.post_auth call site, which is a \
1072                     veto-only event — refusing the login"
1073                );
1074                Err(AuthError::ReactorDenied {
1075                    reason: "reactor attempted to mutate a veto-only event".into(),
1076                }
1077                .into())
1078            }
1079        }
1080    }

1083    /// step-up branch to route a `require_mfa` into — the SAML ACS and OIDC
1084    /// callback handlers (SEC-095).
1085    ///
1086    /// A federated sign-in completes in one round trip: there is no
1087    /// `MfaRequired` / `MfaSetupRequired` result for the caller to act on, so

1090    /// side. Silently ignoring it would mean a reactor that asked for a second
1091    /// factor got a session with one, which is worse than an outage for the
1092    /// operator who configured it: they would never learn their control did
1093    /// nothing.

1096        tenant_id: Uuid,
1097        org_id: Uuid,
1098        user: &User,
1099        ip_address: Option<&str>,
1100        user_agent: Option<&str>,

1102        let requires_mfa = self
1103            .intercept_login_post_auth(tenant_id, org_id, user, ip_address, user_agent)
1104            .await?;
1105        if requires_mfa {
1106            tracing::error!(
1107                target: "axiam::reactor",
1108                tenant_id = %tenant_id,
1109                user_id = %user.id,
1110                "a reactor demanded step-up MFA on a federated login.post_auth, which \
1111                 this sign-in path cannot honour — refusing the login rather than \
1112                 issuing a session the reactor did not authorise"
1113            );
1114            return Err(AuthError::ReactorDenied {
1115                reason: "reactor required step-up MFA on a sign-in path that cannot \
1116                         perform it"
1117                    .into(),
1118            }
1119            .into());
1120        }
1121        Ok(())
1122    }

1124    /// Complete MFA verification after a login challenge.
1125    pub async fn verify_mfa(&self, input: VerifyMfaInput) -> AxiamResult<LoginOutput> {
1126        // 1. Decode the challenge token.
1127        let claims = self.decode_mfa_challenge(&input.challenge_token)?;
1128        let user_id: Uuid = claims
1129            .sub
1130            .parse()
1131            .map_err(|_| AuthError::TokenInvalid("bad sub".into()))?;
1132        let tenant_id: Uuid = claims
1133            .tenant_id
1134            .parse()
1135            .map_err(|_| AuthError::TokenInvalid("bad tenant_id".into()))?;
1136        let org_id: Uuid = claims
1137            .org_id
1138            .parse()
1139            .map_err(|_| AuthError::TokenInvalid("bad org_id".into()))?;
1140
1141        // 2. Fetch user and verify TOTP.
1142        let user = self
1143            .user_repo
1144            .get_by_id(tenant_id, user_id)
1145            .await
1146            .map_err(|_| {
1147                // The same answer as a forged challenge: a gone account must
1148                // not be told apart from one that never existed (SR3-2).
1149                tracing::debug!(target: "axiam::auth", "MFA challenge for an unreadable user");
1150                AuthError::TokenInvalid(NOT_AN_MFA_CHALLENGE.into())
1151            })?;
1152        // SA-06 (S-6): a challenge minted before the credential last changed
1153        // answers to the old password. Refused as any invalid challenge is.
1154        if user.credential_superseded(claims.iat, claims.iat_ns, claims.cv) {
1155            tracing::debug!(
1156                target: "axiam::auth",
1157                user_id = %user_id,
1158                "MFA challenge predates a credential change"
1159            );
1160            return Err(AuthError::TokenInvalid(NOT_AN_MFA_CHALLENGE.into()).into());
1161        }
1162        Self::check_user_status(
1163            &user.status,
1164            user.created_at,
1165            self.config.email_verification_grace_period_hours,
1166        )?;
1167        // SA-43: an account serving a lockout is refused before the code is
1168        // looked at — a correct code during the lockout neither succeeds nor
1169        // shows — with the answer a wrong code gets.
1170        if crate::lockout::is_locked_out(&user) {
1171            return Err(AuthError::MfaInvalidCode.into());
1172        }
1173
1174        let encrypted_secret = user
1175            .mfa_secret
1176            .as_deref()
1177            .ok_or(AuthError::MfaNotEnrolled)?;
1178        let encryption_key = self
1179            .config
1180            .mfa_encryption_key
1181            .as_ref()
1182            .ok_or_else(|| AuthError::Crypto("MFA encryption key not configured".into()))?;
1183
1184        let secret_bytes = totp::decrypt_secret(encryption_key, encrypted_secret)?;
1185        let (valid, used_step) = totp::verify_code_with_replay_check(
1186            &secret_bytes,
1187            &input.totp_code,
1188            &self.config.totp_issuer,
1189            &user.email,
1190            user.totp_last_used_step,
1191        )?;
1192
1193        if !valid {
1194            // SA-43: a wrong code accrues on the account, toward the same
1195            // lockout a wrong password does, so guessing one account's TOTP is
1196            // bounded per account and not only per source IP.
1197            self.record_failed_login(tenant_id, &user, input.lockout_policy.as_ref())
1198                .await?;
1199            return Err(AuthError::MfaInvalidCode.into());
1200        }
1201
1202        // Persist the used step to prevent replay within this window
1203        // (SEC-008/SECHRD-01). `update_totp_step` is an atomic
1204        // compare-and-set: `Ok(false)` means the CAS was lost — either this
1205        // exact step was already consumed (replay) or a concurrent
1206        // submission of the same code already won. Either way, surface the
1207        // SAME rejection as an invalid code so a replay/CAS-miss is
1208        // indistinguishable from a wrong code.
1209        let advanced = self
1210            .user_repo
1211            .update_totp_step(tenant_id, user_id, used_step)
1212            .await?;
1213        if !advanced {
1214            return Err(AuthError::MfaInvalidCode.into());
1215        }
1216
1217        // SA-43: a wrong code sent alongside this one may have locked the
1218        // account after the check above; a lock committed first stands. The
1219        // sign-in is complete, so this is also where the failed-attempt
1220        // counter resets — the password step no longer resets it for an
1221        // account with MFA on. The reset is conditional on the lock, inside
1222        // the write, as `login`'s is.
1223        if !self
1224            .user_repo
1225            .admit_unless_locked(tenant_id, user_id, true)
1226            .await?
1227        {
1228            return Err(AuthError::MfaInvalidCode.into());
1229        }
1230
1231        // 3. Create session and issue tokens.
1232        self.create_session_and_tokens(
1233            user_id,
1234            tenant_id,
1235            org_id,
1236            input.ip_address,
1237            input.user_agent,
1238            // A password *and* a one-time code: this path is only reachable
1239            // with an MFA challenge token, which only the password step mints.
1240            // `mfa` is recorded alongside the two factors, as RFC 8176 §2
1241            // intends, rather than instead of them.
1242            AuthenticationEvidence::now(vec![Amr::Pwd, Amr::Otp, Amr::Mfa]),
1243            // SR10-1: the version of the user row read above, which the
1244            // challenge was just checked against.
1245            Some(user.credential_version),
1246        )
1247        .await
1248    }

1347    /// The WebAuthn second-factor finish: refuse a ceremony whose challenge
1348    /// predates the user's last credential change, then verify the assertion
1349    /// ([`WebauthnService::finish_authentication`]). Returns the
1350    /// authenticated `(user_id, org_id, credential_version)`; the version is
1351    /// the one [`Self::create_session_and_tokens`] must be given (SR10-1).
1352    ///
1353    /// `start_authentication` checks the credential epoch when the ceremony
1354    /// starts; the state token it mints lives
1355    /// `mfa_challenge_lifetime_secs`, and a password reset or change between
1356    /// start and finish must not be survived by it (SA-06, SR2-1). The state
1357    /// token carries the issue time of the MFA challenge it descends from,
1358    /// compared here at full precision. The refusal comes before the assertion
1359    /// is looked at and is the answer an invalid state token gets, as is an
1360    /// unreadable user (fail closed).
1361    pub async fn finish_webauthn_authentication<W: WebauthnCredentialRepository>(
1362        &self,
1363        webauthn: &WebauthnService<W>,
1364        tenant_id: Uuid,
1365        state_token: &str,
1366        response: &PublicKeyCredential,
1367    ) -> AxiamResult<(Uuid, Uuid, i64)> {
1368        let (user_id, state_tenant_id, issued) =
1369            webauthn.authentication_state_issue(state_token)?;
1370        if state_tenant_id != tenant_id {
1371            return Err(AuthError::WebauthnStateInvalid.into());
1372        }
1373        let user = self
1374            .user_repo
1375            .get_by_id(tenant_id, user_id)
1376            .await
1377            .map_err(|_| AuthError::WebauthnStateInvalid)?;
1378        if user.credential_superseded(issued.iat, issued.iat_ns, issued.credential_version) {
1379            return Err(AuthError::WebauthnStateInvalid.into());
1380        }
1381        // SR10-1: the version the ceremony was bound to when the state token
1382        // carries one, else the one just read (equal when both exist: the
1383        // check above refused any other). Handed to the session write.
1384        let credential_version = issued.credential_version.unwrap_or(user.credential_version);
1385        let (user_id, org_id) = webauthn
1386            .finish_authentication(tenant_id, state_token, response)
1387            .await?;
1388        // SA-43: the assertion proves possession; whether the account may sign
1389        // in is a separate question, asked here as the discoverable path and
1390        // `verify_mfa` ask it. An account deactivated, locked out or
1391        // anonymised between the password step and the assertion gets no
1392        // session.
1393        self.ensure_can_sign_in(&user)?;
1394        // SA-43: the second factor completed the sign-in, so the
1395        // failed-attempt counter resets here (the password step no longer
1396        // resets it for an account with MFA on) — unless a lock committed in
1397        // the meantime, which stands and refuses the sign-in.
1398        if !self
1399            .user_repo
1400            .admit_unless_locked(tenant_id, user_id, true)
1401            .await?
1402        {
1403            return Err(AuthError::InvalidCredentials.into());
1404        }
1405        Ok((user_id, org_id, credential_version))
1406    }

1408    /// Start MFA enrollment for a user (step 1 of 2).
1409    ///
1410    /// Generates a TOTP secret, encrypts it, and stages it as the user's
1411    /// **pending** secret — not `mfa_secret` (SA-43). Nothing about the
1412    /// account's live factors changes until `confirm_mfa` verifies a code
1413    /// against it: a confirmed authenticator keeps working while a
1414    /// re-enrolment is in progress (or abandoned), and a passkey-only account
1415    /// does not acquire a TOTP factor nobody confirmed.
1416    pub async fn enroll_mfa(&self, tenant_id: Uuid, user_id: Uuid) -> AxiamResult<EnrollMfaOutput> {
1417        let encryption_key = self
1418            .config
1419            .mfa_encryption_key
1420            .as_ref()
1421            .ok_or_else(|| AuthError::Crypto("MFA encryption key not configured".into()))?;
1422
1423        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
1424
1425        let (base32_secret, totp_uri) =
1426            totp::generate_enrollment(&self.config.totp_issuer, &user.email)?;
1427
1428        // Parse the base32 secret to raw bytes for encryption. Under totp-rs
1429        // 6.0 the fallible step is the *parse* rather than the byte extraction:
1430        // `try_from_base32` rejects a malformed encoding, and `as_bytes` on the
1431        // result cannot fail.
1432        let secret = totp_rs::Secret::try_from_base32(&base32_secret)
1433            .map_err(|e| AuthError::Crypto(format!("secret decode: {e}")))?;
1434        let secret_bytes = secret.as_bytes().to_vec();
1435        let encrypted = totp::encrypt_secret(encryption_key, &secret_bytes)?;
1436
1437        // SA-43: the pending column only; `mfa_secret` and `mfa_enabled` are
1438        // swapped in by `confirm_mfa`.
1439        self.user_repo
1440            .stage_pending_mfa_secret(tenant_id, user_id, encrypted)
1441            .await?;
1442
1443        Ok(EnrollMfaOutput {
1444            secret_base32: base32_secret,
1445            totp_uri,
1446        })
1447    }

1449    /// Confirm MFA enrollment (step 2 of 2).
1450    ///
1451    /// The user provides a TOTP code to prove they saved the secret.
1452    /// On success, `mfa_enabled` is set to `true`.
1453    pub async fn confirm_mfa(
1454        &self,
1455        tenant_id: Uuid,
1456        user_id: Uuid,
1457        totp_code: &str,
1458    ) -> AxiamResult<()> {

1461            .await?;
1462        self.activate_enrolment(tenant_id, user_id, verified).await
1463    }

1465    /// The read-only half of [`Self::confirm_mfa`] (SR1-2): check `totp_code`
1466    /// against the enrolment in progress and write nothing.
1467    ///
1468    /// Split out so the setup-token confirmation can spend its token between
1469    /// this check and [`Self::activate_enrolment`]: a mistyped code still does
1470    /// not cost the token, and a token that cannot be spent stores nothing.
1471    async fn verify_enrolment_code(
1472        &self,
1473        tenant_id: Uuid,
1474        user_id: Uuid,
1475        totp_code: &str,
1476    ) -> AxiamResult<VerifiedEnrolment> {
1477        let encryption_key = self
1478            .config
1479            .mfa_encryption_key
1480            .as_ref()
1481            .ok_or_else(|| AuthError::Crypto("MFA encryption key not configured".into()))?;
1482
1483        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
1484        // SA-43: the secret being confirmed is the pending one. An enrolment
1485        // started before the pending column existed left its secret in
1486        // `mfa_secret` under `mfa_enabled = false`; that one is still
1487        // confirmable in place (a secret under a true flag is a confirmed
1488        // factor, never an enrolment to confirm).
1489        let pending = self
1490            .user_repo
1491            .pending_mfa_secret(tenant_id, user_id)
1492            .await?;
1493        let encrypted_secret = match (&pending, &user.mfa_secret) {
1494            (Some(pending), _) => pending.as_str(),
1495            (None, Some(legacy)) if !user.mfa_enabled => legacy.as_str(),
1496            _ => return Err(AuthError::MfaNotEnrolled.into()),
1497        };
1498
1499        let secret_bytes = totp::decrypt_secret(encryption_key, encrypted_secret)?;
1500        // SECHRD-01/T-24-03: reuse the same matched-step verification and
1501        // atomic CAS as `verify_mfa` so replay tracking begins at
1502        // enrollment-confirm time rather than at first login. A
1503        // freshly-enrolled user always has `totp_last_used_step = NONE`, so
1504        // this always succeeds on first use — it only closes the gap where
1505        // the enrollment code itself could otherwise be replayed before the
1506        // user's first full login.
1507        let (valid, used_step) = totp::verify_code_with_replay_check(
1508            &secret_bytes,
1509            totp_code,
1510            &self.config.totp_issuer,
1511            &user.email,
1512            user.totp_last_used_step,
1513        )?;
1514
1515        if !valid {
1516            return Err(AuthError::MfaInvalidCode.into());
1517        }
1518
1519        Ok(VerifiedEnrolment { pending, used_step })
1520    }

1522    /// The writing half of [`Self::confirm_mfa`]: record the step the code
1523    /// used and promote the secret it was verified against.
1524    async fn activate_enrolment(
1525        &self,
1526        tenant_id: Uuid,
1527        user_id: Uuid,
1528        verified: VerifiedEnrolment,
1529    ) -> AxiamResult<()> {
1530        let VerifiedEnrolment { pending, used_step } = verified;
1531
1532        // Seed totp_last_used_step atomically. A lost CAS here means the
1533        // enrollment code was already consumed (e.g. a concurrent duplicate
1534        // submission) — reject with the same enrollment-failure error rather
1535        // than introduce a new response shape.
1536        let advanced = self
1537            .user_repo
1538            .update_totp_step(tenant_id, user_id, used_step)
1539            .await?;
1540        if !advanced {
1541            return Err(AuthError::MfaInvalidCode.into());
1542        }
1543
1544        // Activate MFA. SA-43: the pending secret becomes `mfa_secret` in the
1545        // same conditional write that sets the flag, and only if it is still
1546        // the one the code was just verified against — a re-enrolment racing
1547        // this confirmation refuses it rather than promoting a secret nobody
1548        // proved.
1549        match pending {
1550            Some(pending) => {
1551                if !self
1552                    .user_repo
1553                    .promote_pending_mfa_secret(tenant_id, user_id, &pending)
1554                    .await?
1555                {
1556                    return Err(AuthError::MfaInvalidCode.into());
1557                }
1558            }
1559            None => {
1560                self.user_repo
1561                    .update(
1562                        tenant_id,
1563                        user_id,
1564                        UpdateUser {
1565                            mfa_enabled: Some(true),
1566                            ..Default::default()
1567                        },
1568                    )
1569                    .await?;
1570            }
1571        }
1572
1573        Ok(())
1574    }

1576    /// Rotate a refresh token: consume the old one, verify the user
1577    /// is still active, and issue a new token pair.
1578    ///
1579    /// # Per-stage timing (A2/J2)
1580    ///
1581    /// Run 5 measured this endpoint at 545/s and p50 88.8 ms, down from 839/s
1582    /// and 47.5 ms in run 4 — the whole distribution shifted right with tight
1583    /// medians in both runs, which is the signature of *added serialized work*
1584    /// rather than of contention or of a slow tail. Refresh is the one hot
1585    /// endpoint that is round-trip-bound rather than CPU-bound, so it carries
1586    /// a stage breakdown (`consume_us`, `user_lookup_us`, `session_create_us`,
1587    /// `token_mint_us`, `handler_total_us`) on the `auth.refresh` span and as
1588    /// one DEBUG event on `target: "axiam::perf"` — the same shape
1589    /// `oauth2.client_credentials` uses, and for the same reason: the next
1590    /// time this number moves, the stage that moved should be readable off a
1591    /// log line instead of bisected for.
1592    ///
1593    /// Cost is four `Instant::now()` reads and four `Span::record` calls that
1594    /// are no-ops when nothing is subscribed — immaterial against a path whose
1595    /// cheapest stage is a datastore round trip.
1596    ///
1597    /// # Datastore round trips: four, not five
1598    ///
1599    /// The rotation used to issue five sequential statements (session read,
1600    /// expiry check, consume, user read, session create) plus a tenant read in
1601    /// the REST handler above it. Two are now gone:
1602    ///
1603    /// - the session read and the consume are one atomic
1604    ///   `consume_by_token_hash` (see that method's docs — it also removes the
1605    ///   read-then-delete window rather than tolerating it);
1606    /// - the handler's per-refresh tenant read is served from a TTL cache.
1607    ///
1608    /// That left three (consume, user read, session create). SR16-1 added a
1609    /// fourth, the post-commit re-read of the user's credential version, which
1610    /// the owner mandated. It is a *user* round trip, so it lies outside the
1611    /// session-repository budget that `refresh_roundtrip_budget_test.rs`
1612    /// counts (two calls: consume and create), as the user read always has.
1613    ///
1614    /// **No security semantics were weakened to get there.** Single-use
1615    /// rotation is still enforced by `RETURN BEFORE` (and now with no window
1616    /// at all), the user-status check still happens before a new session is
1617    /// minted, and an expired token is still consumed rather than left
1618    /// replayable.
1619    #[tracing::instrument(
1620        name = "auth.refresh",
1621        skip(self, input),
1622        fields(
1623            consume_us = tracing::field::Empty,
1624            user_lookup_us = tracing::field::Empty,
1625            session_create_us = tracing::field::Empty,
1626            token_mint_us = tracing::field::Empty,
1627            handler_total_us = tracing::field::Empty,
1628        )
1629    )]
1630    pub async fn refresh(&self, input: RefreshInput) -> AxiamResult<RefreshOutput> {
1631        let started = std::time::Instant::now();
1632
1633        // 1. Atomically find AND consume the session named by this token
1634        //    (single-use guarantee, NEW-3). A `None` means no live session
1635        //    matched — either the token was never valid, or a concurrent
1636        //    refresh already consumed it. Both are the same answer to the
1637        //    caller, and both abort BEFORE any new session is minted, so one
1638        //    refresh token can never yield two live session lineages.
1639        let t_consume = std::time::Instant::now();
1640        let token_hash = token::hash_refresh_token(&input.raw_refresh_token);
1641        let session = self
1642            .session_repo
1643            .consume_by_token_hash(input.tenant_id, &token_hash)
1644            .await?
1645            .ok_or_else(|| -> AxiamError {
1646                AuthError::TokenInvalid("refresh token not found or already used".into()).into()
1647            })?;
1648        let consume_us = t_consume.elapsed().as_micros() as u64;
1649
1650        // 2. Reject an expired session. It has already been consumed by the
1651        //    statement above, which is deliberate: an expired refresh token
1652        //    must not be left behind to be replayed until the cleanup job
1653        //    happens to run.
1654        if session.expires_at <= Utc::now() {
1655            return Err(AuthError::TokenExpired.into());
1656        }
1657
1658        // 3. Verify the account may still act (#519, P23W1-07). Not the
1659        //    password login's rule: the email-verification grace period is
1660        //    about signing in with a password, and every federated account is
1661        //    `PendingVerification` for life, so applying it here ended every
1662        //    federated user's session a day after the account was provisioned.
1663        let t_user = std::time::Instant::now();
1664        let user = self
1665            .user_repo
1666            // A status check on rotation: through the record cache, ≤ TTL old
1667            // (`axiam_db::record_cache`); the new session is built from the
1668            // consumed one, never from credential material.
1669            .get_by_id_cached(input.tenant_id, session.user_id)
1670            .await?;
1671        account_may_act(&user)?;
1672        // SA-13 (N16, D-3): and its tenant still admits it. The consumed row
1673        // stays consumed: a suspension revoked it anyway, and un-suspending
1674        // restores sign-in through a new session, not this one.
1675        self.admit_tenant(input.tenant_id).await?;
1676        let user_lookup_us = t_user.elapsed().as_micros() as u64;
1677
1678        // 4. Create new session with rotated refresh token.
1679        let t_create = std::time::Instant::now();
1680        let raw_refresh = token::generate_refresh_token();
1681        let new_hash = token::hash_refresh_token(&raw_refresh);
1682        let expires_at =
1683            Utc::now() + Duration::seconds(self.config.refresh_token_lifetime_secs as i64);
1684
1685        // SR16-1: a rotation is a session mint and gets the protection a
1686        // sign-in has. The version the consumed row carries is the credential
1687        // it was minted under; the new row is written only if the user's is
1688        // still that, inside the write's own transaction. A row with no stored
1689        // version (written before schema v119, or by a path that checked no
1690        // credential) is trusted only if the credential has not changed since
1691        // it was created, and then takes the user's current version — so an
1692        // upgrade logs nobody out, and a legacy row that predates a reset is
1693        // refused like any other.
1694        let rotation_version = match session.credential_version {
1695            Some(version) => version,
1696            None => {
1697                // One tie rule (SR19-1): a change at or after the row's
1698                // creation refuses, as `User::credential_changed_since` does
1699                // everywhere else.
1700                if user.credential_changed_since(session.created_at) {
1701                    // The uniform refusal: a distinct reason would tell a holder
1702                    // of a stale token that it was genuine and the credential
1703                    // has since changed (QR17-2).
1704                    return Err(AuthError::TokenInvalid(
1705                        "refresh token not found or already used".into(),
1706                    )
1707                    .into());
1708                }
1709                user.credential_version
1710            }
1711        };
1712
1713        let new_session = CreateSession {
1714            tenant_id: input.tenant_id,
1715            user_id: user.id,
1716            token_hash: new_hash,
1717            ip_address: input.ip_address,
1718            user_agent: input.user_agent,
1719            expires_at,
1720            // X7.2 (plan §4.3): **copied**, never re-stamped. A refresh is
1721            // not an authentication event — nobody proved anything to
1722            // AXIAM here — and a session whose `authenticated_at` reset on
1723            // every refresh would report a freshness that grows younger
1724            // the longer the user stays signed in. OIDC Core §12.2 says
1725            // the same thing about the `auth_time` of a refreshed ID
1726            // token: it must equal the original.
1727            authenticated_at: session.authenticated_at,
1728            amr: session.amr,
1729            // W3: **copied**, for a reason adjacent to the one above. A
1730            // refresh response never reaches `/oauth2/authorize`'s cookie
1731            // path, so the `axiam_op_session` value in the browser is not
1732            // reissued here and cannot be. A rotation that dropped the
1733            // digest would leave a live cookie pointing at a row that no
1734            // longer exists, and the user would be signed out of the
1735            // authorization endpoint alone — invisibly, at a moment nothing
1736            // in the interface corresponds to.
1737            browser_token_hash: session.browser_token_hash,
1738            // #844 (D-N43): **copied**, and for the reason this whole
1739            // field exists. The new row has a new id; every OAuth2 grant
1740            // and session participant recorded under the old one still
1741            // names that id, and the lineage is how a logout of this row
1742            // reaches them. A rotation that started a new lineage would be
1743            // exactly the bug it fixes.
1744            lineage_id: Some(session.lineage_id),
1745        };
1746        let new_session = match self
1747            .session_repo
1748            .create_if_credential_version(new_session, rotation_version)
1749            .await
1750        {
1751            Ok(created) => created,
1752            // The version-checked write's refusal is a wrong-password answer;
1753            // for a rotation the uniform refusal is the one a consumed or
1754            // unknown token gets.
1755            Err(AxiamError::AuthenticationFailed { .. }) => {
1756                return Err(AuthError::TokenInvalid(
1757                    "refresh token not found or already used".into(),
1758                )
1759                .into());
1760            }
1761            Err(e) => return Err(e),
1762        };
1763
1764        // SR13-1 for rotation (SR16-1): the in-transaction check is not
1765        // sufficient under snapshot isolation. A reset commits its credential
1766        // BEFORE it revokes, so a rotation row committed before the revocation
1767        // is removed by it, and one committed after is caught by this re-read.
1768        // Removed through the logout path, like the sign-in re-check, so the
1769        // row's id reaches the revocation feed for a session no client held.
1770        let stored_version = self
1771            .user_repo
1772            .get_by_id(input.tenant_id, session.user_id)
1773            .await
1774            .map(|u| u.credential_version);
1775        if stored_version.as_ref().ok() != Some(&rotation_version) {
1776            if let Err(e) = self
1777                .session_repo
1778                .invalidate(input.tenant_id, new_session.id)
1779                .await
1780            {
1781                tracing::error!(
1782                    session_id = %new_session.id,
1783                    error = %e,
1784                    "could not delete a rotated session refused by the post-commit credential re-check"
1785                );
1786            }
1787            return Err(
1788                AuthError::TokenInvalid("refresh token not found or already used".into()).into(),
1789            );
1790        }
1791        let session_create_us = t_create.elapsed().as_micros() as u64;
1792
1793        // 5. Issue new access token (jti = new session.id per D-15).
1794        let t_mint = std::time::Instant::now();
1795        let access_token = token::issue_access_token(
1796            user.id,
1797            input.tenant_id,
1798            input.org_id,
1799            &[],
1800            &self.config,
1801            new_session.id.to_string(),
1802            AUD_USER,
1803        )?;
1804        let token_mint_us = t_mint.elapsed().as_micros() as u64;
1805
1806        let handler_total_us = started.elapsed().as_micros() as u64;
1807        let span = tracing::Span::current();
1808        span.record("consume_us", consume_us);
1809        span.record("user_lookup_us", user_lookup_us);
1810        span.record("session_create_us", session_create_us);
1811        span.record("token_mint_us", token_mint_us);
1812        span.record("handler_total_us", handler_total_us);
1813        tracing::debug!(
1814            target: "axiam::perf",
1815            stage = "auth.refresh",
1816            consume_us,
1817            user_lookup_us,
1818            session_create_us,
1819            token_mint_us,
1820            handler_total_us,
1821            "refresh stage timings (A2/J2)"
1822        );
1823
1824        Ok(RefreshOutput {
1825            access_token,
1826            refresh_token: raw_refresh,
1827            session_id: new_session.id,
1828            expires_in: self.config.access_token_lifetime_secs,
1829        })

1832    /// End a single session (logout) **and the OAuth2 refresh tokens issued
1833    /// under it** (D-N39, #839).
1834    ///
1835    /// Every single-session ending goes through here: `POST /auth/logout`,
1836    /// RP-initiated logout (`/oauth2/end_session` and its cookie hop) and SAML
1837    /// single logout. Deleting the session row alone ended its access tokens
1838    /// at the session check but left every refresh token naming it live — the
1839    /// refresh grant does not ask whether `session_id` still exists — so a
1840    /// native sign-in (whose only credential is its OAuth2 refresh token) kept
1841    /// minting access tokens for a session the user had ended (F4-R12-01).
1842    /// The scope is exactly the G-C one,
1843    /// [`RefreshTokenRepository::revoke_all_for_session`]: the tokens of this
1844    /// session, for every client and grant bound to it, and nothing wider —
1845    /// the user's other sessions and devices keep theirs.
1846    ///
1847    /// A token revoked here is not stamped `rotated_at`, so presenting it
1848    /// later is an ordinary stale credential (`invalid_grant`), never a
1849    /// refresh-token replay: it triggers no second revocation and no
1850    /// `oauth2.refresh_token_replayed` row.
1851    ///
1852    /// # Order, and what a failure leaves behind (F4-R12-07, D-N43)
1853    ///
1854    /// The refresh tokens are revoked **first**, and the session is deleted
1855    /// only once that has succeeded. A logout is the caller asking for exactly
1856    /// this effect, so answering "signed out" while the session or its refresh
1857    /// tokens are still live is the half-ended state this must not leave
1858    /// silently: a failure is logged at `error` with `failed_writes` naming the
1859    /// write that failed (G-C's vocabulary: `refresh_tokens_revoke`,
1860    /// `session_invalidate`) and returned as an error.
1861    ///
1862    /// The order is what makes that error retryable. Every failure leaves the
1863    /// session row in place — a failed revocation never reaches the delete, and
1864    /// a failed delete leaves it by definition — so the credential the caller
1865    /// logged out with (the bearer token, the OP-session cookie, the SAML
1866    /// session index) still resolves to the session, and the same request
1867    /// repeated completes the logout: the revocation is idempotent (a second
1868    /// pass revokes nothing new) and the delete then runs. The reverse order,
1869    /// used until D-N43, could leave the session gone and its refresh tokens
1870    /// live, a state the caller's own credential no longer reaches (its retry
1871    /// was a `401`, or a "signed out" page over a live refresh token).
1872    ///
1873    /// # The lineage, not the row (#844, D-N43)
1874    ///
1875    /// A console refresh replaces the session row, so a relying party's grant
1876    /// names an id that stopped being live at the first refresh after it was
1877    /// issued — fifteen minutes in, with the default access-token lifetime.
1878    /// Revoking by that id alone (D-N39 as first written) therefore missed
1879    /// the common case. The scope is now the session's **lineage**
1880    /// ([`Session::lineage_id`]): resolved from `session_id` first, whether
1881    /// or not that row is still live
1882    /// ([`SessionRepository::lineage_of`]), then
1883    ///
1884    /// - every refresh token of the lineage is revoked
1885    ///   ([`RefreshTokenRepository::revoke_all_for_lineage`]) — the grants
1886    ///   issued before the refresh as much as those issued after it;
1887    /// - `session_id` is deleted, and so is the lineage's live row when
1888    ///   `session_id` is an id it has rotated away from
1889    ///   ([`SessionRepository::invalidate_lineage`]) — which is what ends the
1890    ///   SSO session when a relying party logs out with an `id_token_hint`
1891    ///   minted before the refresh.
1892    ///
1893    /// The lineage's live rows are ended only when they belong to the user
1894    /// the lineage was learnt from, and a lineage is only ever learnt from a
1895    /// row the datastore wrote. An id nothing ever recorded resolves to no
1896    /// lineage and the logout keeps its pre-#844 scope, that id alone.
1897    ///
1898    /// The lineage read belongs to the revocation step: if it fails, nothing
1899    /// is revoked or deleted and the logout is reported failed
1900    /// (`refresh_tokens_revoke`), retryable as below.
1901    ///
1902    /// The window between the two writes is one statement long. A refresh
1903    /// token of this session presented in it is already revoked; a console
1904    /// refresh that consumes the session row in it creates a row of the same
1905    /// lineage, which the second write then ends too.
1906    ///
1907    /// [`Session::lineage_id`]: axiam_core::models::session::Session::lineage_id
1908    pub async fn logout(&self, tenant_id: Uuid, session_id: Uuid) -> AxiamResult<()> {
1909        let lineage = match self.session_repo.lineage_of(tenant_id, session_id).await {
1910            Ok(lineage) => lineage,
1911            // Nothing attempted: the session stays, so the caller can retry.
1912            Err(e) => return logout_outcome(tenant_id, session_id, Err(e), None),
1913        };
1914        let lineage_id = lineage.map_or(session_id, |l| l.lineage_id);
1915        let revoked = self
1916            .refresh_token_repo
1917            .revoke_all_for_lineage(tenant_id, lineage_id, session_id)
1918            .await;
1919        let invalidated = match &revoked {
1920            Ok(count) => {
1921                tracing::debug!(
1922                    %tenant_id, %session_id, %lineage_id, refresh_tokens_revoked = count,
1923                    "logout: the session lineage's OAuth2 refresh tokens are revoked"
1924                );
1925                let own = self.session_repo.invalidate(tenant_id, session_id).await;
1926                Some(match (own, lineage) {
1927                    (Ok(()), Some(l)) => self
1928                        .session_repo
1929                        .invalidate_lineage(tenant_id, l.lineage_id, l.user_id, Some(session_id))
1930                        .await
1931                        .map(|ended| {
1932                            if ended > 0 {
1933                                tracing::debug!(
1934                                    %tenant_id, %session_id, %lineage_id, sessions_ended = ended,
1935                                    "logout: the lineage's live session was ended too"
1936                                );
1937                            }
1938                        }),
1939                    (own, _) => own,
1940                })
1941            }
1942            // Not attempted: the session stays, so the caller can retry.
1943            Err(_) => None,
1944        };
1945        logout_outcome(tenant_id, session_id, revoked, invalidated)
1946    }

2491    /// complete the login flow after WebAuthn authentication succeeds.
2492    ///
2493    /// # The authentication event (X7.2, plan §4.3)
2494    ///
2495    /// `evidence` is what the caller actually verified, and this is the one
2496    /// place it is recorded — every browser sign-in funnels through here, so

2499    ///
2500    /// It is a parameter and not a clock read for the federated case: an
2501    /// upstream identity provider may be replaying an SSO session it

2503    /// relying party the user just authenticated when they did not.
2504    ///
2505    /// # The credential version (SA-06, SR10-1)
2506    ///
2507    /// `credential_version` is the [`User::credential_version`] read together
2508    /// with the credential (or second factor, or setup/challenge artefact) the
2509    /// caller just verified. `Some` makes the session write refuse, inside its
2510    /// own transaction, when the user's version has moved — answered
2511    /// `invalid credentials`, with no session and no refresh token created. It
2512    /// is `None` only where AXIAM verified no credential of its own: a
2513    /// federated sign-in, whose upstream assertion has no AXIAM password
2514    /// behind it to be reset.
2515    pub async fn create_session_and_tokens(
2516        &self,
2517        user_id: Uuid,
2518        tenant_id: Uuid,
2519        org_id: Uuid,
2520        ip_address: Option<String>,
2521        user_agent: Option<String>,
2522        evidence: AuthenticationEvidence,
2523        credential_version: Option<i64>,
2524    ) -> AxiamResult<LoginOutput> {
2525        // SA-13 (N16, D-3): every browser sign-in ends here, so this is where a
2526        // suspended tenant is refused — after the caller verified the
2527        // credential (no oracle for an unauthenticated caller beyond what the
2528        // door itself answers), before anything is written.
2529        self.admit_tenant(tenant_id).await?;
2530
2531        let raw_refresh = token::generate_refresh_token();
2532        let token_hash = token::hash_refresh_token(&raw_refresh);

2534            Utc::now() + Duration::seconds(self.config.refresh_token_lifetime_secs as i64);
2535

2537        // funnels through, for the same reason the authentication evidence is:
2538        // a login path that forgot it would be a browser that silently cannot
2539        // use the authorization endpoint's login hop — discovered by a relying
2540        // party rather than by the compiler.
2541        //
2542        // It is minted unconditionally. This layer does not know, and must not
2543        // need to know, whether any client in the deployment is registered

2588                .get_by_id(tenant_id, user_id)

2594                // The row is removed through the logout path

2602                    tracing::error!(

2604                        error = %e,
2605                        "could not delete a session refused by the post-commit credential re-check"
2606                    );
2607                }
2608                return Err(AxiamError::AuthenticationFailed {
2609                    reason: "invalid credentials".into(),
2610                });
2611            }

2627            refresh_token: raw_refresh,

2630            browser_session_token: raw_browser_session,
2631        })

2716        token::sign_purpose_token(&claims, token::TYP_MFA_SETUP, &self.config)
2717    }
2718
2719    /// Decode and validate an MFA setup token, returning (user_id,
2720    /// tenant_id, org_id).
2721    ///
2722    /// A token minted before the user's credential last changed is refused
2723    /// with the same [`AuthError::MfaSetupTokenInvalid`] an expired or forged
2724    /// one gets (SA-06, S-6).
2725    async fn decode_mfa_setup_token(&self, token: &str) -> Result<SetupTokenIds, AuthError> {
2726        use jsonwebtoken::{Algorithm, DecodingKey, Validation};
2727
2728        let key = DecodingKey::from_ed_pem(self.config.jwt_public_key_pem.as_bytes())
2729            .map_err(|e| AuthError::Crypto(format!("bad public key: {e}")))?;
2730
2731        let mut validation = Validation::new(Algorithm::EdDSA);
2732        validation.set_issuer(&[&self.config.jwt_issuer]);
2733        validation.set_required_spec_claims(&["sub", "exp", "iat", "iss"]);
2734
2735        let data =
2736            jsonwebtoken::decode::<MfaChallengeClaims>(token, &key, &validation).map_err(|e| {
2737                match e.kind() {
2738                    jsonwebtoken::errors::ErrorKind::ExpiredSignature => {
2739                        AuthError::MfaSetupTokenInvalid
2740                    }
2741                    _ => AuthError::MfaSetupTokenInvalid,
2742                }
2743            })?;
2744
2745        // SA-03 rule 6: an access token (or any other kind) is not a setup token.
2746        if !token::purpose_token_typ_accepted(
2747            data.header.typ.as_deref(),
2748            token::TYP_MFA_SETUP,
2749            &self.config,
2750        ) || data.claims.purpose != "mfa_setup"
2751        {
2752            return Err(AuthError::MfaSetupTokenInvalid);
2753        }
2754
2755        let user_id: Uuid = data
2756            .claims
2757            .sub
2758            .parse()
2759            .map_err(|_| AuthError::MfaSetupTokenInvalid)?;
2760        let tenant_id: Uuid = data
2761            .claims
2762            .tenant_id
2763            .parse()
2764            .map_err(|_| AuthError::MfaSetupTokenInvalid)?;
2765        let org_id: Uuid = data
2766            .claims
2767            .org_id
2768            .parse()
2769            .map_err(|_| AuthError::MfaSetupTokenInvalid)?;
2770
2771        // SA-06 (S-6): fail closed — an unreadable user is as unusable as a
2772        // forged token, and one whose credential changed after `iat` is stale.
2773        let user = self
2774            .user_repo
2775            .get_by_id(tenant_id, user_id)
2776            .await
2777            .map_err(|_| AuthError::MfaSetupTokenInvalid)?;
2778        if user.credential_superseded(data.claims.iat, data.claims.iat_ns, data.claims.cv) {
2779            return Err(AuthError::MfaSetupTokenInvalid);
2780        }
2781
2782        Ok(SetupTokenIds {
2783            user_id,
2784            tenant_id,
2785            org_id,
2786            credential_version: user.credential_version,
2787            jti: data.claims.jti,
2788            exp: data.claims.exp,
2789        })
2790    }
2791
2792    /// Record one failed authentication attempt against `user_id`, for the
2793    /// OPAQUE path.
2794    ///
2795    /// OPAQUE proves the password without the server seeing it, but a failed
2796    /// `KE3` is still a wrong password and must accrue toward lockout exactly
2797    /// as a wrong Argon2id verify does. Without this, turning OPAQUE on would
2798    /// silently remove brute-force protection from the accounts that adopted
2799    /// it — the opposite of what enabling an extra security layer is supposed
2800    /// to do.
2801    ///
2802    /// This is why [`crate::opaque::OpaqueRejection`] has two variants rather
2803    /// than one: the caller cannot accrue an attempt it cannot attribute.
2804    pub async fn record_failed_opaque_attempt(
2805        &self,
2806        tenant_id: Uuid,
2807        user_id: Uuid,
2808        policy: Option<&LockoutPolicy>,
2809    ) -> AxiamResult<()> {
2810        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
2811        self.record_failed_login(tenant_id, &user, policy).await
2812    }
2813
2814    async fn record_failed_login(
2815        &self,
2816        tenant_id: Uuid,
2817        user: &axiam_core::models::user::User,
2818        policy: Option<&LockoutPolicy>,
2819    ) -> AxiamResult<()> {
2820        // D-06: delegate to the shared lockout helper — the single source of
2821        // truth for failed-attempt accrual, also called by gRPC
2822        // `UserService::validate_credentials` (SEC-026b).
2823        //
2824        // The tenant's own policy when the caller resolved one; the deployment
2825        // default only when it could not. Never no accrual at all: an
2826        // unresolvable policy must not be a way to switch lockout off.
2827        let fallback;
2828        let policy = match policy {
2829            Some(p) => p,
2830            None => {
2831                fallback = crate::lockout::policy_from_config(&self.config);
2832                &fallback
2833            }
2834        };
2835        crate::lockout::record_failed_login(&self.user_repo, policy, tenant_id, user).await
2836    }
2837}

2839/// May an already-issued credential still act for `user`? (F4 P23W1-01, -03)
2840///
2841/// The question every place that turns a long-lived credential back into a
2842/// principal asks — `/oauth2/authorize` resolving the OP cookie
2843/// ([`AuthService::check_session_holder`]), the session refresh
2844/// ([`AuthService::refresh`], #519), the OAuth2 `authorization_code`
2845/// and `refresh_token` grants in `axiam-oauth2`, and a federated sign-in,
2846/// whose identity provider vouches for the person and not for the account
2847/// (F4 P23W1-04). An
2848/// account status change revokes no credential (only deletion, SCIM
2849/// deprovisioning and a credential reset do), so each of them must ask; one
2850/// function answers, so no two of them can disagree about who may act.
2851///
2852/// **Refused:** `Locked`, `Inactive`, `Anonymized`, `Deleted` — the states
2853/// that mean "this account must not be used".
2854///
2855/// **Not refused: `PendingVerification`, whatever its age.** The email
2856/// verification grace period is a rule about *signing in with a password*,
2857/// which [`AuthService::login`] still applies. It is not a suspension, and it
2858/// cannot be one here: `UserRepository::create` writes `PendingVerification`
2859/// for every new row and federation provisioning never moves a federated user
2860/// off it, so every federated account is pending for life. Refusing it here
2861/// ended browser sign-on and every OAuth2 grant for the whole federated
2862/// population a day after each account was provisioned. This is the rule T-160
2863/// already applies to the token-exchange path, for the same reason.
2864///
2865/// # Errors
2866///
2867/// The [`AuthError`] a sign-in by this user would be refused with.
2868pub fn account_may_act(user: &User) -> Result<(), AuthError> {
2869    match user.status {
2870        UserStatus::PendingVerification => Ok(()),
2871        ref status => account_status_permits(status, user.created_at, 0),
2872    }
2873}
2874
2875/// The status rule itself; see `AuthService::check_user_status`.
2876fn account_status_permits(
2877    status: &UserStatus,
2878    created_at: chrono::DateTime<Utc>,
2879    grace_period_hours: u32,
2880) -> Result<(), AuthError> {
2881    match status {
2882        UserStatus::Active => Ok(()),
2883        UserStatus::Locked => Err(AuthError::AccountLocked),
2884        UserStatus::Inactive => Err(AuthError::AccountInactive),
2885        // Anonymized users cannot log in — treat as inactive.
2886        UserStatus::Anonymized => Err(AuthError::AccountInactive),
2887        // Administratively deleted. Deliberately the same error as
2888        // `Inactive`: whether an account was suspended or removed is not
2889        // something an unauthenticated caller gets to learn.
2890        UserStatus::Deleted => Err(AuthError::AccountInactive),
2891        UserStatus::PendingVerification => {
2892            if grace_period_hours > 0 {
2893                let grace_end = created_at + Duration::hours(grace_period_hours as i64);
2894                if Utc::now() <= grace_end {
2895                    return Ok(());
2896                }
2897            }
2898            Err(AuthError::AccountPendingVerification)
2899        }
2900    }
2901}

2903/// The answer of [`AuthService::logout`] from its two independent writes
2904/// (D-N39, #839).
2905///
2906/// `Ok` only when both succeeded. Otherwise the failure is logged at `error`
2907/// with `failed_writes` naming which of `session_invalidate` and
2908/// `refresh_tokens_revoke` failed — the names G-C's `revocation_failed` audit
2909/// row uses — and the first error is returned, so no caller can answer
2910/// "signed out" over a session or a refresh token that is still live.
2911fn logout_outcome(
2912    tenant_id: Uuid,
2913    session_id: Uuid,
2914    revoked: AxiamResult<u64>,
2915    invalidated: Option<AxiamResult<()>>,
2916) -> AxiamResult<()> {
2917    let failed_writes = logout_failed_writes(&revoked, invalidated.as_ref());
2918    if failed_writes.is_empty() {
2919        return Ok(());
2920    }
2921    tracing::error!(
2922        %tenant_id,
2923        %session_id,
2924        ?failed_writes,
2925        refresh_tokens_error = revoked.as_ref().err().map(ToString::to_string),
2926        session_error = invalidated
2927            .as_ref()
2928            .and_then(|r| r.as_ref().err())
2929            .map(ToString::to_string),
2930        "logout: the session could not be fully ended and is still live; the logout \
2931         is reported as failed, and repeating it with the same credential completes it"
2932    );
2933    revoked?;
2934    invalidated.unwrap_or(Ok(()))
2935}
2936
2937/// Which of the logout's two writes failed, in G-C's vocabulary. The session
2938/// write is `None` when it was not attempted (the revocation failed first).
2939fn logout_failed_writes(
2940    revoked: &AxiamResult<u64>,
2941    invalidated: Option<&AxiamResult<()>>,
2942) -> Vec<&'static str> {
2943    [
2944        revoked.is_err().then_some("refresh_tokens_revoke"),
2945        invalidated
2946            .is_some_and(Result::is_err)
2947            .then_some("session_invalidate"),
2948    ]
2949    .into_iter()
2950    .flatten()
2951    .collect()
2952}
