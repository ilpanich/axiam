1//! The directory half of [`AuthService`]: just-in-time provisioning and the
2//! administrator's act of linking an existing account (G-3, T23.3.3).
3//!
4//! A child module of `service` so it can reach the service's private fields
5//! without widening them.
6//!
7//! # The rule both halves serve (D-28)
8//!
9//! **A directory never takes over a local account.** Provisioning only ever
10//! *creates*, and only for a login name that matches nothing local; an entry
11//! that would collide with an existing account's username or email — compared
12//! ignoring case, in both columns — is refused with the ordinary
13//! invalid-credentials failure and an audit row. Turning an existing account
14//! into a directory account is [`AuthService::link_local_account_to_directory`],
15//! an explicit act that resolves the entry through the directory, refuses one
16//! already linked elsewhere, and retires every credential the account held that
17//! the directory does not decide.
18

40/// Audit action: a directory account was created at first sign-in.
41pub const AUDIT_JIT_PROVISIONED: &str = "directory.jit_provisioned";
42/// Audit action: a first sign-in the directory vouched for was refused —
43/// because the entry collides with a local account, or because it lacks what a
44/// local account needs. `metadata.reason` says which.
45pub const AUDIT_JIT_REFUSED: &str = "directory.jit_refused";
46/// Audit action: an administrator linked an existing account to its directory
47/// entry.
48pub const AUDIT_ACCOUNT_LINKED: &str = "directory.account_linked";
49/// Audit action (T23.3.4): applying the directory group mapping changed a
50/// user's memberships. Identifiers and counts only: the AXIAM groups added and
51/// removed, and how many directory groups were resolved and mapped.
52pub const AUDIT_GROUPS_MAPPED: &str = axiam_core::models::directory_sync::AUDIT_GROUPS_MAPPED;
53/// Audit action (T23.3.4): a sign-in the directory vouched for was refused
54/// because the group mapping could not be applied — the directory could not be
55/// asked, or the user is in more groups than the cap allows. Nothing was
56/// changed.
57pub const AUDIT_GROUP_MAPPING_REFUSED: &str = "directory.group_mapping_refused";

59/// The [`DirectoryAuditSink`] that appends to an [`AuditLogRepository`].
60///
61/// Lives here, not in `axiam-core`, because it logs a failed write and core has
62/// no logging. It never fails its caller: the directory path has already done,
63/// or refused, what the row describes.
64pub struct RepositoryDirectoryAuditSink<A>(pub A);
65
66impl<A: AuditLogRepository> DirectoryAuditSink for RepositoryDirectoryAuditSink<A> {
67    fn record<'a>(&'a self, entry: CreateAuditLogEntry) -> DirectoryFuture<'a, ()> {
68        Box::pin(async move {
69            let tenant_id = entry.tenant_id;
70            if let Err(error) = self.0.append(entry).await {
71                tracing::error!(
72                    target: "axiam::directory",
73                    %tenant_id,
74                    %error,
75                    "a directory audit row could not be written"
76                );
77            }
78        })
79    }
80}

99/// The values a directory entry contributes to a new account, cleaned.
100///
101/// The directory is another party's system: whoever administers it chooses
102/// every attribute, so a value is bounded, stripped of control and
103/// bidirectional-override characters, and refused when it cannot be made into
104/// what a local account holds, before it is stored or shown to anyone.
105#[derive(Debug, PartialEq, Eq)]
106struct JitProfile {
107    username: String,
108    email: String,
109    display_name: Option<String>,
110}
111
112impl JitProfile {
113    /// `None` when the entry cannot become an account: no usable e-mail
114    /// address (a local account must have one) or no usable name.
115    fn from_identity(identity: &DirectoryIdentity, typed_login: &str) -> Option<Self> {
116        // The login name the entry answered to stands in when the mapped
117        // username attribute is absent: the filter matched this entry by it.
118        let username = identity
119            .username
120            .as_deref()
121            .and_then(|raw| clean_identifier(raw, MAX_NAME_CHARS))
122            .or_else(|| clean_identifier(typed_login, MAX_NAME_CHARS))?;
123        let email = identity
124            .email
125            .as_deref()
126            .and_then(|raw| clean_identifier(raw, MAX_EMAIL_CHARS))
127            .filter(|email| plausible_email(email))?;
128        let display_name = identity
129            .display_name
130            .as_deref()
131            .and_then(clean_display_name);
132        Some(Self {
133            username,
134            email,
135            display_name,
136        })
137    }
138}

147    /// The just-in-time provisioning seam, called by
148    /// [`Self::login_unknown_user`] with an entry the directory has just
149    /// vouched for (a bind as that entry succeeded).
150    ///
151    /// Every refusal is the unknown-user answer, [`AuthError::InvalidCredentials`];
152    /// the dummy verify has already run beside the bind, so nothing here
153    /// changes what the caller sees or, beyond the work itself, how long.
154    pub(super) async fn provision_directory_account(
155        &self,
156        input: &LoginInput,
157        identity: DirectoryIdentity,
158    ) -> AxiamResult<LoginResult> {
159        let tenant_id = input.tenant_id;
160        let Some(profile) = JitProfile::from_identity(&identity, &input.username_or_email) else {
161            tracing::warn!(
162                target: "axiam::directory",
163                %tenant_id,
164                "just-in-time provisioning refused: the directory entry has no usable \
165                 username and e-mail address"
166            );
167            self.audit_directory(
168                input,
169                AUDIT_JIT_REFUSED,
170                Uuid::nil(),
171                serde_json::json!({
172                    "reason": "unusable_attributes",
173                    "directory_external_id": identity.external_id,
174                }),
175                AuditOutcome::Denied,
176            )
177            .await;
178            return Err(AuthError::InvalidCredentials.into());
179        };
180
181        // D-28: a directory never takes over a local account. The names
182        // compared are the entry's username, its address and what was typed,
183        // against both columns of every account, ignoring case.
184        let names = vec![
185            profile.username.clone(),
186            profile.email.clone(),
187            input.username_or_email.clone(),
188        ];
189        if let Some(collision) = self
190            .user_repo
191            .find_identity_collision(tenant_id, &names)
192            .await?
193        {
194            tracing::warn!(
195                target: "axiam::directory",
196                %tenant_id,
197                existing_user_id = %collision.user_id,
198                attribute = collision.attribute.as_str(),
199                "just-in-time provisioning refused: the directory entry collides with a \
200                 local account"
201            );
202            self.audit_directory(
203                input,
204                AUDIT_JIT_REFUSED,
205                collision.user_id,
206                serde_json::json!({
207                    "reason": "collision",
208                    "attribute": collision.attribute.as_str(),
209                    "existing_user_id": collision.user_id.to_string(),
210                    "directory_external_id": identity.external_id,
211                }),
212                AuditOutcome::Denied,
213            )
214            .await;
215            return Err(AuthError::InvalidCredentials.into());
216        }
217
218        let metadata = match &profile.display_name {
219            Some(name) => serde_json::json!({ "oidc": { "name": name } }),
220            None => serde_json::json!({}),
221        };
222        let created = match self
223            .user_repo
224            .create_directory_account(CreateDirectoryAccount {
225                tenant_id,
226                username: profile.username.clone(),
227                email: profile.email.clone(),
228                external_id: identity.external_id.clone(),
229                metadata,
230            })
231            .await
232        {
233            Ok(user) => {
234                self.audit_directory_as(
235                    input,
236                    AUDIT_JIT_PROVISIONED,
237                    ActorType::User,
238                    user.id,
239                    user.id,
240                    serde_json::json!({
241                        "source": "directory",
242                        "directory_external_id": identity.external_id,
243                    }),
244                    AuditOutcome::Success,
245                )
246                .await;
247                user
248            }
249            // The datastore refused: the username, the address or the entry is
250            // taken. Either another first login for this same entry won a race
251            // a moment ago — then its account is this login's account — or
252            // something else holds the name, which is a collision after all.
253            Err(AxiamError::AlreadyExists { .. }) => {
254                match self
255                    .user_repo
256                    .get_by_username(tenant_id, &profile.username)
257                    .await
258                {
259                    Ok(winner)
260                        if winner
261                            .directory_external_id
262                            .as_deref()
263                            .is_some_and(|marker| {
264                                marker.eq_ignore_ascii_case(&identity.external_id)
265                            }) =>
266                    {
267                        winner
268                    }
269                    Ok(_) | Err(AxiamError::NotFound { .. }) => {
270                        tracing::warn!(
271                            target: "axiam::directory",
272                            %tenant_id,
273                            "just-in-time provisioning refused: the name, the address or the \
274                             entry is already held by another account"
275                        );
276                        self.audit_directory(
277                            input,
278                            AUDIT_JIT_REFUSED,
279                            Uuid::nil(),
280                            serde_json::json!({
281                                "reason": "collision",
282                                "attribute": "unique_constraint",
283                                "directory_external_id": identity.external_id,
284                            }),
285                            AuditOutcome::Denied,
286                        )
287                        .await;
288                        return Err(AuthError::InvalidCredentials.into());
289                    }
290                    Err(other) => return Err(other),
291                }
292            }
293            Err(other) => return Err(other),
294        };
295
296        // F4 P23W3-05: the account's standing before its memberships. In the
297        // lost-race branch above `created` is another login's account, which
298        // may since have been deactivated (the sync job) or suspended; such an
299        // account is refused here — the same answer `complete_authenticated_login`
300        // gives — before the mapping could re-add directory memberships to it.
301        Self::check_user_status(
302            &created.status,
303            created.created_at,
304            self.config.email_verification_grace_period_hours,
305        )?;
306
307        // T23.3.4: the new account holds no membership yet, so a lookup that
308        // fails here leaves it granting nothing; the sign-in is refused and the
309        // next one maps the groups.
310        self.apply_directory_group_mapping(input, &created, &identity.dn)
311            .await?;
312
313        self.complete_authenticated_login(
314            created,
315            tenant_id,
316            input.org_id,
317            input.ip_address.clone(),
318            input.user_agent.clone(),
319            input.mfa_policy.clone(),
320        )
321        .await
322    }

324    /// Apply the tenant's directory group mapping for `user`, whose directory
325    /// entry is `user_dn` (G-3, T23.3.4, D-30).
326    ///
327    /// Called on every successful directory sign-in, after the directory has
328    /// vouched for the password and before the session (or an MFA challenge) is
329    /// issued. **Fail closed:** when the mapping cannot be applied — the
330    /// directory cannot be asked, the cap is hit, a write fails — the answer is
331    /// the ordinary invalid-credentials failure, **not counted** against the
332    /// account (the user did nothing wrong), with an audit row; memberships the
333    /// directory may have revoked are never kept by letting the sign-in through.
334    ///
335    /// A changed membership set is audited (`directory.groups_mapped`) with
336    /// identifiers and counts only; an unchanged one writes nothing, so a
337    /// sign-in that maps nothing new leaves no row.
338    pub(super) async fn apply_directory_group_mapping(
339        &self,
340        input: &LoginInput,
341        user: &User,
342        user_dn: &str,
343    ) -> AxiamResult<()> {
344        let Some(mapper) = &self.directory_group_mapper else {
345            return Ok(());
346        };
347        match mapper
348            .apply_for_user(input.tenant_id, user.id, user_dn)
349            .await
350        {
351            Ok(outcome) => {
352                if outcome.changed() {
353                    self.audit_directory(
354                        input,
355                        AUDIT_GROUPS_MAPPED,
356                        user.id,
357                        serde_json::json!({
358                            "source": "directory",
359                            "groups_added": outcome.added,
360                            "groups_removed": outcome.removed,
361                            "added_count": outcome.added.len(),
362                            "removed_count": outcome.removed.len(),
363                            "manual_memberships_left": outcome.left_manual.len(),
364                            "directory_groups_resolved": outcome.directory_groups_resolved,
365                            "directory_groups_mapped": outcome.directory_groups_mapped,
366                        }),
367                        AuditOutcome::Success,
368                    )
369                    .await;
370                }
371                Ok(())
372            }
373            Err(error) => {
374                tracing::warn!(
375                    target: "axiam::directory",
376                    tenant_id = %input.tenant_id,
377                    user_id = %user.id,
378                    outcome = ?error,
379                    "directory sign-in refused: the group mapping could not be applied"
380                );
381                self.audit_directory(
382                    input,
383                    AUDIT_GROUP_MAPPING_REFUSED,
384                    user.id,
385                    serde_json::json!({
386                        "reason": "mapping_not_applied",
387                        "outcome": format!("{error:?}"),
388                    }),
389                    AuditOutcome::Failure,
390                )
391                .await;
392                Err(AuthError::InvalidCredentials.into())
393            }
394        }
395    }

397    /// Link an **existing local account** to the directory entry it names
398    /// (G-3, T23.3.3, D-28). An administrator's act: no route exists yet
399    /// (T23.3.8 adds it) and nothing in the sign-in path calls it.
400    ///
401    /// * The entry is **resolved by the directory** — through the tenant's
402    ///   configured filter with the account's username — never supplied by the
403    ///   caller. None, or more than one, is `NotFound`.
404    /// * An entry already linked to **another** account is refused
405    ///   (`Conflict`): the marker is unique per tenant.
406    /// * The account is marked ([`UserRepository::mark_directory_account`]:
407    ///   marker set, local hash replaced with an unusable one, OPAQUE record
408    ///   deleted), and then everything it holds that **authenticates without
409    ///   the directory deciding** is retired: its WebAuthn credentials are
410    ///   deleted, its federation links deleted (an upstream OIDC or SAML
411    ///   identity signs in through the link without the directory deciding,
412    ///   F4 P23W3-01), its `User`-type certificates revoked, and every session
413    ///   and OAuth2 refresh token revoked. TOTP is kept: it is a second factor
414    ///   *behind* the directory password.
415    ///
416    /// # One unit of work, by order rather than by transaction
417    ///
418    /// The revocations live behind four repositories, two of which keep a
419    /// validation cache and publish revocations to a feed; a single SQL
420    /// transaction across their tables would bypass both and leave revoked
421    /// sessions valid until the cache expired. So the unit of work is an
422    /// **order** that is safe to stop in and to retry: the mark comes first
423    /// (from then on no local password can open a new session), the passkeys
424    /// and certificates next, and the sessions and refresh tokens **last**, so
425    /// that anything issued before the mark is swept by the final step. A
426    /// failure partway returns the error (and writes a failure row); calling
427    /// again with the account already linked to the same entry re-runs the
428    /// revocations and nothing else.
429    ///
430    /// Audited (`directory.account_linked`) with counts and identifiers only.
431    ///
432    /// # Errors
433    ///
434    /// `NotFound` (no such account, or no single matching entry), `Conflict`
435    /// (the entry or the account is linked elsewhere, or the tenant has no
436    /// enabled directory), `Validation` (a deleted account) and
437    /// `ServiceUnavailable` (no authenticator, or the directory cannot be
438    /// reached or is misconfigured).
439    #[allow(clippy::too_many_arguments)] // the five collaborators are the unit of work
440    pub async fn link_local_account_to_directory<W, C>(
441        &self,
442        tenant_id: Uuid,
443        user_id: Uuid,
444        actor_id: Uuid,
445        ip_address: Option<String>,
446        webauthn_repo: &W,
447        certificate_repo: &C,
448    ) -> AxiamResult<DirectoryLinkOutcome>
449    where
450        W: WebauthnCredentialRepository,
451        C: CertificateRepository,
452    {
453        let authenticator = self.directory_authenticator.clone().ok_or_else(|| {
454            AxiamError::ServiceUnavailable("directory sign-in is not available".into())
455        })?;
456        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
457        if user.status == UserStatus::Deleted {
458            return Err(AxiamError::Validation {
459                message: "a deleted account cannot be linked to a directory".into(),
460            });
461        }
462
463        let identity = match authenticator.lookup_entry(tenant_id, &user.username).await {
464            Ok(identity) => identity,
465            Err(DirectoryAuthError::InvalidCredentials) => {
466                return Err(AxiamError::NotFound {
467                    entity: "directory entry".into(),
468                    id: format!("the entry for login name {}", user.username),
469                });
470            }
471            Err(DirectoryAuthError::NotConfigured) => {
472                return Err(AxiamError::Conflict {
473                    reason: "the tenant has no enabled directory".into(),
474                });
475            }
476            Err(other) => {
477                tracing::warn!(
478                    target: "axiam::directory",
479                    %tenant_id,
480                    outcome = ?other,
481                    "linking refused: the directory could not answer"
482                );
483                return Err(AxiamError::ServiceUnavailable(
484                    "the directory could not be queried".into(),
485                ));
486            }
487        };
488
489        let was_already_linked = match user.directory_external_id.as_deref() {
490            Some(marker) if marker.eq_ignore_ascii_case(&identity.external_id) => true,
491            Some(_) => {
492                return Err(AxiamError::Conflict {
493                    reason: "the account is already linked to a different directory entry".into(),
494                });
495            }
496            None => false,
497        };
498
499        // 1. Mark. From here no local password opens a new session.
500        let marked = if was_already_linked {
501            user.clone()
502        } else {
503            match self
504                .user_repo
505                .mark_directory_account(tenant_id, user.id, &identity.external_id)
506                .await
507            {
508                Ok(marked) => marked,
509                Err(AxiamError::AlreadyExists { .. }) => {
510                    return Err(AxiamError::Conflict {
511                        reason: "that directory entry is already linked to another account".into(),
512                    });
513                }
514                Err(other) => return Err(other),
515            }
516        };
517
518        // 2-4. Retire what authenticates without the directory.
519        let retired = self
520            .retire_local_credentials(tenant_id, &marked, webauthn_repo, certificate_repo)
521            .await;
522        let (webauthn_credentials_deleted, federation_links_deleted, certificates_revoked) =
523            match retired {
524                Ok(counts) => counts,
525                Err((stage, error)) => {
526                    tracing::error!(
527                        target: "axiam::directory",
528                        %tenant_id,
529                        user_id = %marked.id,
530                        stage,
531                        %error,
532                        "linking is incomplete: the account is marked but a revocation failed; \
533                         repeat the call"
534                    );
535                    self.audit_directory_raw(
536                        tenant_id,
537                        ip_address,
538                        AUDIT_ACCOUNT_LINKED,
539                        ActorType::User,
540                        actor_id,
541                        marked.id,
542                        serde_json::json!({
543                            "stage": stage,
544                            "directory_external_id": identity.external_id,
545                            "marked": true,
546                        }),
547                        AuditOutcome::Failure,
548                    )
549                    .await;
550                    return Err(error);
551                }
552            };
553
554        self.audit_directory_raw(
555            tenant_id,
556            ip_address,
557            AUDIT_ACCOUNT_LINKED,
558            ActorType::User,
559            actor_id,
560            marked.id,
561            serde_json::json!({
562                "directory_external_id": identity.external_id,
563                "webauthn_credentials_deleted": webauthn_credentials_deleted,
564                "federation_links_deleted": federation_links_deleted,
565                "certificates_revoked": certificates_revoked,
566                "sessions_and_refresh_tokens_revoked": true,
567                "retry_of_interrupted_link": was_already_linked,
568            }),
569            AuditOutcome::Success,
570        )
571        .await;
572
573        Ok(DirectoryLinkOutcome {
574            user: marked,
575            webauthn_credentials_deleted,
576            federation_links_deleted,
577            certificates_revoked,
578            was_already_linked,
579        })
580    }
581
582    /// Steps 2–4 of linking, in the order that matters: the credential kinds
583    /// that outlive a session — passkeys, federation links, `User` certificates
584    /// — then the sessions and refresh tokens last. The error names the stage
585    /// that failed. Returns `(passkeys, federation links, certificates)`.
586    async fn retire_local_credentials<W, C>(
587        &self,
588        tenant_id: Uuid,
589        user: &User,
590        webauthn_repo: &W,
591        certificate_repo: &C,
592    ) -> Result<(u64, u64, u64), (&'static str, AxiamError)>
593    where
594        W: WebauthnCredentialRepository,
595        C: CertificateRepository,
596    {
597        let passkeys = webauthn_repo
598            .delete_by_user(tenant_id, user.id)
599            .await
600            .map_err(|error| ("webauthn_credentials", error))?;
601        // F4 P23W3-01: a federation link is a sign-in the directory never
602        // sees, exactly as a passkey is. Federated provisioning never links by
603        // name or address, so a deleted link is not re-made at the next
604        // upstream sign-in: that sign-in would try to create a new account and
605        // collide with this one.
606        let links = self
607            .federation_repo
608            .get_by_user_id(tenant_id, user.id)
609            .await
610            .map_err(|error| ("federation_links", error))?;
611        for link in &links {
612            self.federation_repo
613                .delete(tenant_id, link.id)
614                .await
615                .map_err(|error| ("federation_links", error))?;
616        }
617        let certificates = certificate_repo
618            .revoke_user_certificates(tenant_id, user.id, &user.username, &user.email)
619            .await
620            .map_err(|error| ("user_certificates", error))?;
621        self.revoke_all_sessions(tenant_id, user.id)
622            .await
623            .map_err(|error| ("sessions_and_refresh_tokens", error))?;
624        Ok((passkeys, links.len() as u64, certificates))
625    }

627    /// One audit row, written through the injected sink, if there is one.
628    async fn audit_directory(
629        &self,
630        input: &LoginInput,
631        action: &str,
632        resource_id: Uuid,
633        metadata: serde_json::Value,
634        outcome: AuditOutcome,
635    ) {
636        self.audit_directory_as(
637            input,
638            action,

700    /// The directory half of [`Self::login_unknown_user`]: authenticate the
701    /// typed name against the tenant's directory for provisioning, beside the
702    /// equalising dummy verify. `None` is "answer as an unknown user"; `Some`
703    /// is an entry the directory vouched for.
704    pub(super) async fn authenticate_unknown_name_against_directory(
705        &self,
706        input: &LoginInput,
707    ) -> AxiamResult<Option<DirectoryIdentity>> {
708        let authenticator = match &self.directory_authenticator {
709            Some(authenticator) if !input.password.is_empty() => {
710                std::sync::Arc::clone(authenticator)
711            }
712            _ => {
713                // Nothing to ask, or nobody to ask: the plain unknown-user cost.
714                self.equalising_dummy_verify().await?;
715                return Ok(None);
716            }
717        };
718        // F4 P23W3-02 (T-332): a name that has failed the tenant's threshold
719        // is answered as an unknown user without asking the directory — the
720        // brake an account's own counter is for a name that has one.
721        if self.unknown_name_lockout.is_locked(
722            input.tenant_id,
723            &input.username_or_email,
724            std::time::Instant::now(),
725        ) {
726            tracing::info!(
727                target: "axiam::directory",
728                tenant_id = %input.tenant_id,
729                "unknown login name refused without asking the directory: too many failures"
730            );
731            self.equalising_dummy_verify().await?;
732            return Ok(None);
733        }
734        // The same shape `login_directory_account` uses: a hash permit first,
735        // so saturation answers `503` exactly where every other branch does and
736        // before the directory is contacted, then the dummy verify and the
737        // directory call side by side. A tenant with no directory (or without
738        // `jit_provisioning`) is answered by the gate before any socket opens,
739        // so for it the whole call costs one verify, as it always has.
740        let permit = acquire_hash_permit(
741            &self.crypto_semaphore,
742            std::time::Duration::from_secs(self.config.hash_acquire_timeout_secs),
743        )
744        .await?;
745        let ((), outcome) = tokio::join!(
746            self.dummy_verify_holding(permit),
747            authenticator.authenticate_for_provisioning(
748                input.tenant_id,
749                &input.username_or_email,
750                &input.password,
751            ),
752        );
753        match outcome {
754            Ok(identity) => {
755                self.unknown_name_lockout
756                    .clear(input.tenant_id, &input.username_or_email);
757                Ok(Some(identity))
758            }
759            Err(DirectoryAuthError::NotConfigured) => {
760                tracing::debug!(
761                    target: "axiam::directory",
762                    tenant_id = %input.tenant_id,
763                    "unknown login name: no directory provisioning for this tenant"
764                );
765                Ok(None)
766            }
767            Err(other) => {
768                tracing::info!(
769                    target: "axiam::directory",
770                    tenant_id = %input.tenant_id,
771                    outcome = ?other,
772                    "unknown login name refused by the directory"
773                );
774                // Only a failure the directory decided counts — a wrong
775                // password or no such entry; an unusable directory counts
776                // against nobody, as for accounts (T-302).
777                if other == DirectoryAuthError::InvalidCredentials {
778                    let fallback;
779                    let policy = match input.lockout_policy.as_ref() {
780                        Some(policy) => policy,
781                        None => {
782                            fallback = crate::lockout::policy_from_config(&self.config);
783                            &fallback
784                        }
785                    };
786                    self.unknown_name_lockout.record_failure(
787                        input.tenant_id,
788                        &input.username_or_email,
789                        policy,
790                        std::time::Instant::now(),
791                    );
792                }
793                Ok(None)
794            }
795        }
796    }
