8/// Encrypt a TOTP secret with AES-256-GCM.
9///
10/// Returns `base64(nonce || ciphertext || tag)`.
11///
12/// Delegates to [`crate::crypto::aes256gcm_encrypt`] — the bundled format
13/// is shared; changing the wire format here would break existing TOTP secrets
14/// stored in the database.
15pub fn encrypt_secret(key: &[u8; 32], plaintext: &[u8]) -> Result<String, AuthError> {
16    crypto::aes256gcm_encrypt(key, plaintext)
17}
18
19/// Decrypt an AES-256-GCM encrypted TOTP secret.
20///
21/// Delegates to [`crate::crypto::aes256gcm_decrypt`].
22pub fn decrypt_secret(key: &[u8; 32], encoded: &str) -> Result<Vec<u8>, AuthError> {
23    crypto::aes256gcm_decrypt(key, encoded)
24}

26/// TOTP step size in seconds (RFC 6238 default), shared by every [`build`]
27/// call in this module.
28const TOTP_STEP_SECS: u64 = 30;
29
30/// TOTP skew tolerance (±1 step), shared by every [`build`] call in this
31/// module.
32///
33/// `u16` since totp-rs 6.0, which widened `Builder::with_skew`. The tolerated
34/// window is unchanged.
35const TOTP_SKEW: u16 = 1;
36
37/// Digits in a generated code (RFC 6238 default).
38const TOTP_DIGITS: u8 = 6;
39
40/// Build a configured [`Totp`] over the given raw secret.
41///
42/// One constructor for the whole module. totp-rs 6.0 replaced `TOTP::new`'s
43/// seven positional arguments with a builder, and seven positional arguments --
44/// three of which were bare numbers -- repeated at four call sites was exactly
45/// the shape a silent parameter swap lives in. Centralising it means the
46/// algorithm, digit count, skew and step duration are stated once and every
47/// caller gets the same ones by construction.
48fn build(secret_bytes: Vec<u8>, issuer: &str, account: &str) -> Result<Totp, AuthError> {
49    Builder::new()
50        .with_algorithm(Algorithm::SHA1) // RFC 6238 default
51        .with_digits(TOTP_DIGITS)
52        .with_skew(TOTP_SKEW)
53        .with_step_duration(TOTP_STEP_SECS)
54        .with_secret(secret_bytes)
55        .with_issuer(Some(issuer))
56        .with_account_name(account)
57        .build()
58        .map_err(|e| AuthError::Crypto(format!("TOTP init: {e}")))
59}

61/// Generate a TOTP enrollment: secret + otpauth URI.
62///
63/// Returns `(base32_secret, otpauth_uri)`.
64pub fn generate_enrollment(issuer: &str, account: &str) -> Result<(String, String), AuthError> {
65    let secret = Secret::generate();
66    // 6.0's `as_bytes` is infallible; 5.x's `to_bytes` returned a
67    // `Result` whose error branch a freshly generated secret could not
68    // reach, so the `map_err` it forced is gone with it.
69    let secret_bytes = secret.as_bytes().to_vec();
70
71    let totp = build(secret_bytes, issuer, account)?;
72
73    let base32 = secret.to_base32();
74    // 6.0 renamed `get_url` to `to_url` and made it fallible: rendering an
75    // otpauth URI can fail on an issuer or account name that cannot be encoded.
76    // Propagating that beats 5.x's infallible signature, which could not report
77    // it at all.
78    let uri = totp
79        .to_url()
80        .map_err(|e| AuthError::Crypto(format!("TOTP otpauth URL: {e}")))?;
81
82    Ok((base32, uri))
83}

85/// Verify a TOTP code against a raw secret.
86pub fn verify_code(
87    secret_bytes: &[u8],
88    code: &str,
89    issuer: &str,
90    account: &str,
91) -> Result<bool, AuthError> {
92    let totp = build(secret_bytes.to_vec(), issuer, account)?;
93    // 6.0 changed `check_current` from `Result<bool, SystemTimeError>` to
94    // `Option<u64>`, where `Some(step)` names WHICH step matched. Only the
95    // pass/fail is wanted here; `verify_code_with_replay_check` is where the
96    // step itself becomes load-bearing.
97    Ok(totp.check_current(code).is_some())
98}

100/// Verify a TOTP code with replay protection.
101///
102/// Determines WHICH step in the tolerated ±1-step skew window actually matched
103/// (`current_step - 1`, `current_step`, or `current_step + 1`) rather than
104/// assuming it was always `current_step`. This matters because recording
105/// `current_step` unconditionally would let a code accepted via the -1-skew
106/// step be replayed again once the wall clock advances past it (T-24-02).
107///
108/// Under totp-rs 5.x, `check_current` reported only pass/fail for the whole
109/// window, so this function re-derived `generate()` across the three candidate
110/// steps to identify the match. **6.0 answers with the matched step directly**
111/// (`Option<u64>`), so that probe is gone — and with it the only place in this
112/// module that compared a generated code against user input outside the
113/// library's own constant-time path.
114///
115/// The code is rejected (even though the HMAC is correct) unless the matched
116/// step is strictly greater than `last_used_step.unwrap_or(0)`.
117///
118/// Returns `Ok((valid, matched_step))` on success. The caller MUST persist
119/// `matched_step` via the atomic `user_repo.update_totp_step` CAS when
120/// `valid` is `true`, and MUST treat a lost CAS (`Ok(false)`) as an invalid
121/// code (SECHRD-01) — this function cannot see concurrent submissions by
122/// itself, so the persisted-step compare-and-set is the actual replay guard.
123///
124/// Per SEC-008 / SECHRD-01 (REQ-14 AC-5).
125pub fn verify_code_with_replay_check(
126    secret_bytes: &[u8],
127    code: &str,
128    issuer: &str,
129    account: &str,
130    last_used_step: Option<u64>,
131) -> Result<(bool, u64), AuthError> {
132    let totp = build(secret_bytes.to_vec(), issuer, account)?;
133
134    // Computed independently of totp-rs internals, because it is the value
135    // reported alongside a REJECTION -- where there is no matched step to
136    // report.
137    let current_step = std::time::SystemTime::now()
138        .duration_since(std::time::UNIX_EPOCH)
139        .map_err(|e| AuthError::Crypto(format!("system time error: {e}")))?
140        .as_secs()
141        / TOTP_STEP_SECS;
142
143    // Check the HMAC across the tolerated ±1-step skew window. Since 6.0 this
144    // answers with the matched step rather than a bare bool, which is exactly
145    // the value T-24-02 needs and which 5.x forced this function to re-derive.
146    let Some(matched_step) = totp.check_current(code) else {
147        return Ok((false, current_step));
148    };
149
150    // Replay check: reject unless the ACTUAL matched step (incl. -1 skew) is
151    // strictly greater than the last-used step.
152    let last = last_used_step.unwrap_or(0);
153    if matched_step <= last {
154        return Ok((false, matched_step));
155    }
156
157    Ok((true, matched_step))
158}

165    fn encrypt_decrypt_roundtrip() {
166        let key = [42u8; 32];
167        let plaintext = b"totp-secret-bytes";
168        let encrypted = encrypt_secret(&key, plaintext).unwrap();
169        let decrypted = decrypt_secret(&key, &encrypted).unwrap();
170        assert_eq!(decrypted, plaintext);

210    /// Regression: a TOTP code produced by the SAME base32 secret returned to the
211    /// client (the live enroll path encrypts `Secret::try_from_base32(base32)?.as_bytes()`
212    /// and confirm verifies those bytes) must validate. Guards the enroll→confirm
213    /// round-trip end to end (RFC 6238 dynamic-truncation compliant).
214    #[test]
215    fn enroll_base32_roundtrip_confirms() {
216        let (base32, _uri) = generate_enrollment("AXIAM", "admin@axiam.dev").unwrap();
217        let secret_bytes = Secret::try_from_base32(base32).unwrap().as_bytes().to_vec();
218        let totp = build(secret_bytes.clone(), "AXIAM", "admin@axiam.dev").unwrap();
219        // 6.0's `generate_current` is infallible and answers a `Token`; render it
220        // to the string a client would actually submit.
221        let code = totp.generate_current().to_string();
222        assert!(verify_code(&secret_bytes, &code, "AXIAM", "admin@axiam.dev").unwrap());
223    }

225    /// T-24-02: a code that only validates against the -1 skew step must have
226    /// `current_step - 1` (not `current_step`) recorded as the matched step,
227    /// and once persisted, the SAME code must not be replayable — even though
228    /// it still falls inside the ±1 skew window relative to a later call.
229    #[test]
230    fn totp_skew_step_recorded() {
231        let secret = Secret::generate();
232        let secret_bytes = secret.as_bytes().to_vec();
233
234        let totp = build(secret_bytes.clone(), "AXIAM", "skew@test.com").unwrap();
235
236        let current_step = std::time::SystemTime::now()
237            .duration_since(std::time::UNIX_EPOCH)
238            .unwrap()
239            .as_secs()
240            / 30;
241        let prev_step = current_step - 1;
242
243        // Hand-generate a code for the PREVIOUS step (simulates a client
244        // that entered a code right at a step boundary) — this is the -1
245        // skew case `TOTP::check` tolerates.
246        // 6.0's `generate` answers a `Token` rather than a String.
247        let prev_code = totp.generate(prev_step * 30).to_string();
248
249        let (valid, matched_step) = verify_code_with_replay_check(
250            &secret_bytes,
251            &prev_code,
252            "AXIAM",
253            "skew@test.com",
254            None,
255        )
256        .unwrap();
257        assert!(valid, "code from the -1 skew step should be accepted");
258        assert_eq!(
259            matched_step, prev_step,
260            "matched step must be current_step - 1 (the step that actually \
261             matched), not always current_step"
262        );
263
264        // Persist `matched_step` (as the real caller does via
265        // `update_totp_step`) and resubmit the SAME code. It is still inside
266        // the ±1 skew window relative to `check()`, but must now be rejected
267        // because its matched step is not strictly greater than the
268        // persisted last-used step.
269        let (replay_valid, _) = verify_code_with_replay_check(
270            &secret_bytes,
271            &prev_code,
272            "AXIAM",
273            "skew@test.com",
274            Some(matched_step),
275        )
276        .unwrap();
277        assert!(
278            !replay_valid,
279            "a skew-accepted code must not be replayable at a later step"
280        );
281    }
