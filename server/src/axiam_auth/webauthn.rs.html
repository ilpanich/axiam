1//! WebAuthn registration and authentication ceremonies.
2//!
3//! Ceremony state (`PasskeyRegistration` / `PasskeyAuthentication`) is
4//! serialized to JSON, encrypted with AES-256-GCM via [`totp::encrypt_secret`],
5//! and embedded in a short-lived JWT. Passkey data is encrypted at rest before
6//! storage in the database.
7//!
8//! ## X3 wave 2 — the attested registration ceremony (D7, addendum W2-D1/W2-D2)
9//!
10//! [`WebauthnService::start_attested_registration`] /
11//! [`WebauthnService::finish_attested_registration`] are **additive** —
12//! neither existing public method above them changed signature, so every
13//! caller of the unattested ceremony (today's `mode: none` behavior) is
14//! unaffected. They exist for a tenant whose policy has `mode != none`; the
15//! caller (a later-wave REST handler) is responsible for choosing which pair
16//! to call based on the resolved `WebauthnAttestationPolicy` — this service
17//! does not hold a policy repository itself (kept free of a hard
18//! `axiam-pki`/`axiam-db` dependency, W2-D4; see `crate::attestation`'s
19//! module docs for the fuller rationale).
20//!
21//! **AXIAM's `mode` does not map onto `webauthn-rs`'s conveyance the way the
22//! plan first assumed (verified against the real 0.5.5 source, addendum
23//! W2-D1):** `Webauthn::start_attested_passkey_registration` always requests
24//! `AttestationConveyancePreference::Direct` — there is no library-level
25//! `indirect` attested ceremony. So the real mapping is:
26//!
27//! - `mode: none` → [`WebauthnService::start_registration`] /
28//!   [`WebauthnService::finish_registration`], unchanged;
29//! - `mode: indirect` → the attested ceremony below (wire conveyance is
30//!   `direct`, not `indirect`) with the *lax* policy defaults
31//!   (`unknown_aaguid: allow` is the caller's job to set on the resolved
32//!   policy, not this module's);
33//! - `mode: direct_required` → the same attested ceremony with the *strict*
34//!   defaults.
35//!
36//! `start_attested_passkey_registration` also unconditionally requires user
37//! verification and rejects synchronised authenticators
38//! (`reject_synchronised_authenticators(true)`) — so **every** non-`none`
39//! mode excludes iCloud Keychain / Google Password Manager passkeys and
40//! hybrid (phone-as-authenticator) flows, not just `direct_required`. State
41//! this plainly wherever it is documented for admins/users — do not describe
42//! the wire value as `indirect`.

69/// State-token `purpose` for the attested registration ceremony — distinct
70/// from `"webauthn_register"` so a token minted for one ceremony can never be
71/// replayed against the other's `finish_*` entry point.
72const PURPOSE_REGISTER_ATTESTED: &str = "webauthn_register_attested";
73
74/// State-token `purpose` for the usernameless (discoverable-credential)
75/// authentication ceremony.
76///
77/// Distinct from `"webauthn_authenticate"` for a reason that matters: this
78/// ceremony's token carries a nil `sub`, because the user is not known when it
79/// is minted. `finish_authentication` *does* trust `sub` as the authenticated
80/// identity, so if the two ceremonies shared a purpose, a discoverable token
81/// replayed there would authenticate the nil user.
82const PURPOSE_AUTHENTICATE_DISCOVERABLE: &str = "webauthn_authenticate_discoverable";

92    /// Tenant scope.

94    /// Organization scope (needed for token issuance after auth).
95    org_id: String,

101    iat: i64,

118    /// Build the service, constructing the inner `Webauthn` instance
119    /// from the relying-party configuration in [`AuthConfig`].
120    pub fn new(credential_repo: W, config: AuthConfig) -> Result<Self, AuthError> {
121        let rp_origin = Url::parse(&config.webauthn_rp_origin)
122            .map_err(|e| AuthError::Crypto(format!("invalid RP origin URL: {e}")))?;
123        let builder = WebauthnBuilder::new(&config.webauthn_rp_id, &rp_origin)
124            .map_err(|e| AuthError::Crypto(format!("WebAuthn builder: {e}")))?
125            .rp_name(&config.webauthn_rp_name);
126        let webauthn = builder
127            .build()
128            .map_err(|e| AuthError::Crypto(format!("WebAuthn build: {e}")))?;
129
130        Ok(Self {
131            webauthn: Arc::new(webauthn),
132            credential_repo,
133            config,
134        })
135    }

139    /// Start passkey registration.
140    ///
141    /// Returns the challenge JSON for the browser **and** a JWT state
142    /// token that must be forwarded to [`finish_registration`].
143    pub async fn start_registration(
144        &self,
145        tenant_id: Uuid,
146        org_id: Uuid,
147        user_id: Uuid,
148        username: &str,
149    ) -> AxiamResult<(CreationChallengeResponse, String)> {
150        // Fetch existing credentials to exclude from re-registration.
151        let existing = self
152            .credential_repo
153            .list_by_user(tenant_id, user_id)
154            .await?;
155
156        let encryption_key = self.require_encryption_key()?;
157        let exclude_creds: Vec<Passkey> = existing
158            .iter()
159            .filter_map(|c| self.decrypt_passkey(&encryption_key, &c.passkey_json).ok())
160            .collect();
161
162        // Extract credential IDs for the exclusion list so the
163        // authenticator skips already-registered credentials.
164        let exclude_ids: Vec<CredentialID> =
165            exclude_creds.iter().map(|p| p.cred_id().clone()).collect();
166
167        let (mut ccr, reg_state) = self
168            .webauthn
169            .start_passkey_registration(user_id, username, username, Some(exclude_ids))
170            .map_err(|e| AuthError::WebauthnRegistration(e.to_string()))?;
171
172        require_discoverable_credential(&mut ccr);
173
174        let state_token =
175            self.encode_state_token(user_id, tenant_id, org_id, "webauthn_register", &reg_state)?;
176
177        Ok((ccr, state_token))
178    }

180    /// Complete passkey registration.
181    ///
182    /// Verifies the authenticator response, encrypts the resulting
183    /// passkey, and persists it in the credential repository.
184    pub async fn finish_registration(
185        &self,
186        tenant_id: Uuid,
187        caller_user_id: Uuid,
188        state_token: &str,
189        credential_name: &str,
190        response: &RegisterPublicKeyCredential,
191    ) -> AxiamResult<WebauthnCredential> {
192        let (user_id, decoded_tenant_id, _org_id, reg_state) =
193            self.decode_state_token::<PasskeyRegistration>(state_token, "webauthn_register")?;
194
195        if decoded_tenant_id != tenant_id {
196            return Err(AuthError::WebauthnStateInvalid.into());
197        }
198
199        if user_id != caller_user_id {
200            return Err(AuthError::WebauthnStateInvalid.into());
201        }
202
203        let passkey = self
204            .webauthn
205            .finish_passkey_registration(response, &reg_state)
206            .map_err(|e| AuthError::WebauthnRegistration(e.to_string()))?;
207
208        let encryption_key = self.require_encryption_key()?;
209
210        // Default to SecurityKey; the API handler can override based
211        // on client-side attestation metadata if available.
212        let credential_type = WebauthnCredentialType::SecurityKey;
213
214        // Serialize and encrypt the passkey for at-rest storage.
215        let passkey_json_plain = serde_json::to_string(&passkey)
216            .map_err(|e| AuthError::Crypto(format!("serialize passkey: {e}")))?;
217        let passkey_json_enc =
218            totp::encrypt_secret(&encryption_key, passkey_json_plain.as_bytes())?;
219
220        // credential_id as base64url-no-pad for external correlation.
221        let credential_id_str = URL_SAFE_NO_PAD.encode(passkey.cred_id().as_ref());
222
223        let created = self
224            .credential_repo
225            .create(CreateWebauthnCredential {
226                tenant_id,
227                user_id,
228                credential_id: credential_id_str,
229                name: credential_name.to_string(),
230                credential_type,
231                passkey_json: passkey_json_enc,
232                // D6/D7 (X3): this is the unattested `mode: none` ceremony
233                // (today's behavior, unchanged) — no attestation metadata was
234                // requested or resolved. Wiring the attested ceremony's
235                // metadata capture into this service is wave 2 (X3
236                // enforcement) work.
237                aaguid: None,
238                attestation_format: None,
239                attested: false,
240                authenticator_name: None,
241            })
242            .await?;
243
244        Ok(created)
245    }

249    /// Start passkey authentication.
250    ///
251    /// Returns the challenge JSON for the browser **and** a JWT state
252    /// token that must be forwarded to [`finish_authentication`].
253    pub async fn start_authentication(
254        &self,
255        tenant_id: Uuid,
256        org_id: Uuid,
257        user_id: Uuid,
258    ) -> AxiamResult<(RequestChallengeResponse, String)> {
259        let existing = self
260            .credential_repo
261            .list_by_user(tenant_id, user_id)
262            .await?;
263
264        if existing.is_empty() {
265            return Err(AuthError::WebauthnNoCredentials.into());
266        }
267
268        let encryption_key = self.require_encryption_key()?;
269        let passkeys: Vec<Passkey> = existing
270            .iter()
271            .filter_map(|c| self.decrypt_passkey(&encryption_key, &c.passkey_json).ok())
272            .collect();
273
274        if passkeys.is_empty() {
275            return Err(AuthError::WebauthnNoCredentials.into());
276        }
277
278        let (rcr, auth_state) = self
279            .webauthn
280            .start_passkey_authentication(&passkeys)
281            .map_err(|e| AuthError::WebauthnAuthentication(e.to_string()))?;
282
283        let state_token = self.encode_state_token(
284            user_id,
285            tenant_id,
286            org_id,
287            "webauthn_authenticate",
288            &auth_state,
289        )?;
290
291        Ok((rcr, state_token))
292    }

294    /// Start **usernameless** passkey authentication for a tenant.
295    ///
296    /// Unlike [`Self::start_authentication`], no user is known yet: the
297    /// challenge carries an empty `allowCredentials`, the authenticator offers
298    /// whichever discoverable credential it holds for this relying party, and
299    /// the assertion is what names the user. Only the workspace has to be known
300    /// in advance, because the credential must be resolved within one tenant's
301    /// isolation boundary.
302    ///
303    /// # Why there is no "does this tenant have any passkeys?" pre-check
304    ///
305    /// [`Self::start_authentication`] returns [`AuthError::WebauthnNoCredentials`]
306    /// when the named user has none, which is fine there — the caller already
307    /// proved who the user is by holding a valid MFA challenge token. Here the
308    /// caller is anonymous, so the equivalent check would answer "does this
309    /// workspace have any passkey users?" to whoever asks. The ceremony is
310    /// therefore always issued, and an unknown credential fails at
311    /// [`Self::finish_discoverable_authentication`] with the same error any
312    /// other bad assertion produces.
313    pub async fn start_discoverable_authentication(
314        &self,
315        tenant_id: Uuid,
316        org_id: Uuid,
317    ) -> AxiamResult<(RequestChallengeResponse, String)> {
318        let (rcr, auth_state) = self
319            .webauthn
320            .start_discoverable_authentication()
321            .map_err(|e| AuthError::WebauthnAuthentication(e.to_string()))?;
322
323        // `sub` is the one claim this ceremony cannot fill in: the user is
324        // literally what it is trying to discover. The nil UUID is a sentinel,
325        // and `finish_discoverable_authentication` never reads it back — the
326        // authenticated identity comes from the assertion's user handle. The
327        // distinct `purpose` is what stops this token being replayed into
328        // `finish_authentication`, which *would* trust `sub`.
329        let state_token = self.encode_state_token(
330            Uuid::nil(),
331            tenant_id,
332            org_id,
333            PURPOSE_AUTHENTICATE_DISCOVERABLE,
334            &auth_state,
335        )?;
336
337        Ok((rcr, state_token))
338    }

340    /// Complete usernameless passkey authentication.
341    ///
342    /// Resolves the user from the assertion's user handle, verifies the
343    /// assertion against that user's stored passkeys, and returns the
344    /// authenticated `(user_id, org_id)`.
345    ///
346    /// The user handle is only a *claim* until the signature is checked. It
347    /// selects which credentials to verify against; it does not authenticate
348    /// anyone by itself. Because the lookup is scoped to `tenant_id` from the
349    /// state token, a handle naming a user in another tenant resolves to no
350    /// credentials and fails — cross-tenant assertions cannot land here.
351    pub async fn finish_discoverable_authentication(
352        &self,
353        tenant_id: Uuid,
354        state_token: &str,
355        response: &PublicKeyCredential,
356    ) -> AxiamResult<(Uuid, Uuid)> {
357        let (_sentinel_user_id, decoded_tenant_id, org_id, auth_state) =
358            self.decode_state_token::<DiscoverableAuthentication>(
359                state_token,
360                PURPOSE_AUTHENTICATE_DISCOVERABLE,
361            )?;
362
363        if decoded_tenant_id != tenant_id {
364            return Err(AuthError::WebauthnStateInvalid.into());
365        }
366
367        let (claimed_user_id, _cred_id) = self
368            .webauthn
369            .identify_discoverable_authentication(response)
370            .map_err(|e| AuthError::WebauthnAuthentication(e.to_string()))?;
371
372        let encryption_key = self.require_encryption_key()?;
373        let credentials = self
374            .credential_repo
375            .list_by_user(tenant_id, claimed_user_id)
376            .await?;
377
378        let keys: Vec<DiscoverableKey> = credentials
379            .iter()
380            .filter_map(|c| self.decrypt_passkey(&encryption_key, &c.passkey_json).ok())
381            .map(|p| DiscoverableKey::from(&p))
382            .collect();
383
384        if keys.is_empty() {
385            // Deliberately the generic assertion failure, not
386            // `WebauthnNoCredentials`: distinguishing "no such user in this
387            // tenant" from "assertion did not verify" would answer a question
388            // the caller has not authenticated to ask.
389            return Err(AuthError::WebauthnAuthentication("assertion rejected".into()).into());
390        }
391
392        let auth_result = self
393            .webauthn
394            .finish_discoverable_authentication(response, auth_state, &keys)
395            .map_err(|e| AuthError::WebauthnAuthentication(e.to_string()))?;
396
397        let cred_id_b64 = URL_SAFE_NO_PAD.encode(auth_result.cred_id().as_ref());
398        if let Some(cred) = credentials.iter().find(|c| c.credential_id == cred_id_b64)
399            && let Err(e) = self
400                .credential_repo
401                .update_last_used(tenant_id, cred.id)
402                .await
403        {
404            tracing::warn!(
405                credential_id = %cred.id,
406                error = %e,
407                "failed to update last_used_at for WebAuthn credential"
408            );
409        }
410
411        Ok((claimed_user_id, org_id))
412    }

414    /// Complete passkey authentication.
415    ///
416    /// Verifies the authenticator assertion and updates the
417    /// `last_used_at` timestamp on the matched credential. Returns the
418    /// authenticated `user_id` on success.
419    pub async fn finish_authentication(
420        &self,
421        tenant_id: Uuid,
422        state_token: &str,
423        response: &PublicKeyCredential,
424    ) -> AxiamResult<(Uuid, Uuid)> {
425        let (user_id, decoded_tenant_id, org_id, auth_state) =
426            self.decode_state_token::<PasskeyAuthentication>(state_token, "webauthn_authenticate")?;
427
428        if decoded_tenant_id != tenant_id {
429            return Err(AuthError::WebauthnStateInvalid.into());
430        }
431
432        let auth_result = self
433            .webauthn
434            .finish_passkey_authentication(response, &auth_state)
435            .map_err(|e| AuthError::WebauthnAuthentication(e.to_string()))?;
436
437        // Update last_used_at for the credential that was used.
438        let cred_id_b64 = URL_SAFE_NO_PAD.encode(auth_result.cred_id().as_ref());
439        let credentials = self
440            .credential_repo
441            .list_by_user(tenant_id, user_id)
442            .await?;
443        if let Some(cred) = credentials.iter().find(|c| c.credential_id == cred_id_b64)
444            && let Err(e) = self
445                .credential_repo
446                .update_last_used(tenant_id, cred.id)
447                .await
448        {
449            tracing::warn!(
450                credential_id = %cred.id,
451                error = %e,
452                "failed to update last_used_at for WebAuthn credential"
453            );
454        }
455
456        Ok((user_id, org_id))
457    }

467    fn decrypt_passkey(&self, key: &[u8; 32], encrypted: &str) -> Result<Passkey, AuthError> {
468        let json_bytes = totp::decrypt_secret(key, encrypted)?;
469        let json_str = String::from_utf8(json_bytes)
470            .map_err(|e| AuthError::Crypto(format!("passkey UTF-8: {e}")))?;
471        serde_json::from_str(&json_str)
472            .map_err(|e| AuthError::Crypto(format!("passkey deserialize: {e}")))
473    }

475    /// Encode a state token from an already-JSON-serialized state value.
476    ///
477    /// This is the low-level primitive [`Self::encode_state_token`] wraps for
478    /// the common case (a `T: Serialize` state, no pre-processing). It exists
479    /// separately so the attested-registration path (W2-D2) can strip
480    /// `ca_list` out of the serialized `AttestedPasskeyRegistration` *before*
481    /// encryption, without duplicating everything else this method does.
482    fn encode_state_token_json(
483        &self,
484        user_id: Uuid,
485        tenant_id: Uuid,
486        org_id: Uuid,
487        purpose: &str,
488        state_json_value: JsonValue,
489    ) -> Result<String, AuthError> {
490        use jsonwebtoken::{Algorithm, EncodingKey, Header};
491
492        let encryption_key = self.require_encryption_key()?;
493        let state_json = serde_json::to_string(&state_json_value)
494            .map_err(|e| AuthError::Crypto(format!("serialize state: {e}")))?;
495        let encrypted_state = totp::encrypt_secret(&encryption_key, state_json.as_bytes())?;
496
497        let now = chrono::Utc::now().timestamp();
498        let claims = WebauthnStateClaims {
499            sub: user_id.to_string(),
500            tenant_id: tenant_id.to_string(),
501            org_id: org_id.to_string(),
502            purpose: purpose.into(),
503            state: encrypted_state,
504            iss: self.config.jwt_issuer.clone(),
505            iat: now,
506            exp: now + self.config.mfa_challenge_lifetime_secs as i64,
507        };
508
509        let key = EncodingKey::from_ed_pem(self.config.jwt_private_key_pem.as_bytes())
510            .map_err(|e| AuthError::Crypto(format!("bad private key: {e}")))?;
511        let header = Header::new(Algorithm::EdDSA);
512        jsonwebtoken::encode(&header, &claims, &key)
513            .map_err(|e| AuthError::Crypto(format!("JWT encode: {e}")))
514    }

516    fn encode_state_token<T: Serialize>(
517        &self,
518        user_id: Uuid,
519        tenant_id: Uuid,
520        org_id: Uuid,
521        purpose: &str,
522        state: &T,
523    ) -> Result<String, AuthError> {
524        let state_json_value = serde_json::to_value(state)
525            .map_err(|e| AuthError::Crypto(format!("serialize state: {e}")))?;
526        self.encode_state_token_json(user_id, tenant_id, org_id, purpose, state_json_value)
527    }

529    /// Decode a state token down to its raw JSON state value, without
530    /// deserializing it into a concrete type yet.
531    ///
532    /// [`Self::decode_state_token`] wraps this for the common case; the
533    /// attested-registration path (W2-D2) uses this directly so it can
534    /// re-insert the *current* `ca_list` into the value before deserializing
535    /// it as `AttestedPasskeyRegistration`.
536    fn decode_state_token_json(
537        &self,
538        token: &str,
539        expected_purpose: &str,
540    ) -> Result<(Uuid, Uuid, Uuid, JsonValue), AuthError> {
541        use jsonwebtoken::{Algorithm, DecodingKey, Validation};
542
543        let key = DecodingKey::from_ed_pem(self.config.jwt_public_key_pem.as_bytes())
544            .map_err(|e| AuthError::Crypto(format!("bad public key: {e}")))?;
545
546        let mut validation = Validation::new(Algorithm::EdDSA);
547        validation.set_issuer(&[&self.config.jwt_issuer]);
548        validation.set_required_spec_claims(&["sub", "exp", "iat", "iss"]);
549
550        let data = jsonwebtoken::decode::<WebauthnStateClaims>(token, &key, &validation)
551            .map_err(|_| AuthError::WebauthnStateInvalid)?;
552
553        if data.claims.purpose != expected_purpose {
554            return Err(AuthError::WebauthnStateInvalid);
555        }
556
557        let user_id: Uuid = data
558            .claims
559            .sub
560            .parse()
561            .map_err(|_| AuthError::WebauthnStateInvalid)?;
562        let tenant_id: Uuid = data
563            .claims
564            .tenant_id
565            .parse()
566            .map_err(|_| AuthError::WebauthnStateInvalid)?;
567        let org_id: Uuid = data
568            .claims
569            .org_id
570            .parse()
571            .map_err(|_| AuthError::WebauthnStateInvalid)?;
572
573        let encryption_key = self.require_encryption_key()?;
574        let state_bytes = totp::decrypt_secret(&encryption_key, &data.claims.state)?;
575        let state_json = String::from_utf8(state_bytes)
576            .map_err(|e| AuthError::Crypto(format!("state UTF-8: {e}")))?;
577        let state_value: JsonValue = serde_json::from_str(&state_json)
578            .map_err(|e| AuthError::Crypto(format!("state deserialize: {e}")))?;
579
580        Ok((user_id, tenant_id, org_id, state_value))
581    }

583    fn decode_state_token<T: serde::de::DeserializeOwned>(
584        &self,
585        token: &str,
586        expected_purpose: &str,
587    ) -> Result<(Uuid, Uuid, Uuid, T), AuthError> {
588        let (user_id, tenant_id, org_id, state_value) =
589            self.decode_state_token_json(token, expected_purpose)?;
590        let state: T = serde_json::from_value(state_value)
591            .map_err(|e| AuthError::Crypto(format!("state deserialize: {e}")))?;
592        Ok((user_id, tenant_id, org_id, state))
593    }
594
595    /// Read a state token's `purpose` claim **after** verifying its
596    /// signature, issuer and expiry.
597    ///
598    /// Used only by [`Self::finish_registration_for_policy`] to learn which
599    /// ceremony a token belongs to before dispatching. The full validation
600    /// runs first — an unsigned or expired token is rejected here, not
601    /// classified — so this never turns the `purpose` claim into an
602    /// attacker-chosen routing input.
603    fn state_token_purpose(&self, token: &str) -> Result<String, AuthError> {
604        use jsonwebtoken::{Algorithm, DecodingKey, Validation};
605
606        let key = DecodingKey::from_ed_pem(self.config.jwt_public_key_pem.as_bytes())
607            .map_err(|e| AuthError::Crypto(format!("bad public key: {e}")))?;
608
609        let mut validation = Validation::new(Algorithm::EdDSA);
610        validation.set_issuer(&[&self.config.jwt_issuer]);
611        validation.set_required_spec_claims(&["sub", "exp", "iat", "iss"]);
612
613        let data = jsonwebtoken::decode::<WebauthnStateClaims>(token, &key, &validation)
614            .map_err(|_| AuthError::WebauthnStateInvalid)?;
615        Ok(data.claims.purpose)
616    }
617
618    // ---- Attested registration ceremony (X3 wave 2, D7 + W2-D1/W2-D2/W2-D3) ----
619
620    /// Start the attested passkey registration ceremony (`mode: indirect` or
621    /// `mode: direct_required` — see the module docs for why both use this
622    /// same call).
623    ///
624    /// `ca_list` must already be built for the tenant's policy (typically via
625    /// `crate::attestation::AttestationCaCache`, filtered to
626    /// `allowed_aaguids` when the policy sets one) — this method does not
627    /// build it itself, matching this service's W2-D4 "no policy/metadata
628    /// dependency" design.
629    ///
630    /// **Fails closed (W2-D3):** an empty `ca_list` is rejected with
631    /// [`AuthError::WebauthnAttestationUnavailable`] *before* calling into
632    /// `webauthn-rs` — never a silent fallback to the unattested ceremony.
633    pub async fn start_attested_registration(
634        &self,
635        tenant_id: Uuid,
636        org_id: Uuid,
637        user_id: Uuid,
638        username: &str,
639        ca_list: AttestationCaList,
640    ) -> AxiamResult<(CreationChallengeResponse, String)> {
641        if ca_list.is_empty() {
642            return Err(AuthError::WebauthnAttestationUnavailable.into());
643        }
644
645        let existing = self
646            .credential_repo
647            .list_by_user(tenant_id, user_id)
648            .await?;
649
650        let encryption_key = self.require_encryption_key()?;
651        let exclude_ids: Vec<CredentialID> = existing
652            .iter()
653            .filter_map(|c| self.decrypt_passkey(&encryption_key, &c.passkey_json).ok())
654            .map(|p| p.cred_id().clone())
655            .collect();
656
657        let (mut ccr, reg_state) = self
658            .webauthn
659            .start_attested_passkey_registration(
660                user_id,
661                username,
662                username,
663                Some(exclude_ids),
664                ca_list,
665                None,
666            )
667            .map_err(|e| AuthError::WebauthnRegistration(e.to_string()))?;
668
669        // Same `residentKey: "discouraged"` default as the unattested ceremony
670        // (`start_attested_passkey_registration` also hard-codes
671        // `require_resident_key(false)`), so it needs the same amendment or an
672        // attestation-enforcing tenant would be the one tenant whose users
673        // cannot sign in without a username.
674        require_discoverable_credential(&mut ccr);
675
676        // W2-D2: strip `ca_list` out of the serialized state before it is
677        // encrypted and embedded in the state token. On the live BLOB this
678        // list is ~800 KB (255 unique roots) — embedding it in every
679        // registration's state token is unacceptable, and it never
680        // influences `CreationChallengeResponse` (verified by reading the
681        // builder), so stripping it is invisible to the client.
682        let mut state_value = serde_json::to_value(&reg_state)
683            .map_err(|e| AuthError::Crypto(format!("serialize attested state: {e}")))?;
684        let removed = state_value
685            .as_object_mut()
686            .and_then(|obj| obj.remove("ca_list"));
687        if removed.is_none() {
688            // Fail loudly rather than silently embedding the (huge) ca_list
689            // unstripped — an upstream `webauthn-rs` field rename must break
690            // this loudly, never degrade security silently (W2-D2).
691            return Err(AuthError::Crypto(
692                "AttestedPasskeyRegistration serialization did not contain a ca_list \
693                 field to strip (W2-D2) — refusing to embed the state unstripped"
694                    .into(),
695            )
696            .into());
697        }
698
699        let state_token = self.encode_state_token_json(
700            user_id,
701            tenant_id,
702            org_id,
703            PURPOSE_REGISTER_ATTESTED,
704            state_value,
705        )?;
706
707        Ok((ccr, state_token))
708    }
709
710    /// Complete the attested passkey registration ceremony, enforce the
711    /// tenant's attestation policy (D8), and persist the resulting credential
712    /// with its attestation metadata (D6) on success.
713    ///
714    /// `current_ca_list` is re-inserted into the decoded state (W2-D2) —
715    /// deliberately the **current** list, not the one captured at
716    /// `start_attested_registration` time, so an MDS refresh between start
717    /// and finish takes effect. `policy` and `metadata` drive the D8 policy
718    /// decision; on denial this returns
719    /// [`AuthError::WebauthnAttestationDenied`] and **never** persists a
720    /// credential or touches any existing one (D7: "existing credentials are
721    /// never auto-revoked" — this method has no delete/disable code path at
722    /// all).
723    #[allow(clippy::too_many_arguments)] // mirrors finish_registration's own parameter shape plus the three D8 policy inputs (policy/metadata/ca_list); grouping them into a struct would only move the naming problem, not remove it.
724    pub async fn finish_attested_registration<M: AttestationMetadataSource>(
725        &self,
726        tenant_id: Uuid,
727        caller_user_id: Uuid,
728        state_token: &str,
729        credential_name: &str,
730        response: &RegisterPublicKeyCredential,
731        policy: &WebauthnAttestationPolicy,
732        metadata: &M,
733        current_ca_list: &AttestationCaList,
734    ) -> AxiamResult<WebauthnCredential> {
735        let (user_id, decoded_tenant_id, _org_id, mut state_value) =
736            self.decode_state_token_json(state_token, PURPOSE_REGISTER_ATTESTED)?;
737
738        if decoded_tenant_id != tenant_id {
739            return Err(AuthError::WebauthnStateInvalid.into());
740        }
741        if user_id != caller_user_id {
742            return Err(AuthError::WebauthnStateInvalid.into());
743        }
744
745        // W2-D2: re-insert the CURRENT ca_list before deserializing.
746        let ca_list_value = serde_json::to_value(current_ca_list)
747            .map_err(|e| AuthError::Crypto(format!("serialize ca_list: {e}")))?;
748        state_value
749            .as_object_mut()
750            .ok_or(AuthError::WebauthnStateInvalid)?
751            .insert("ca_list".to_string(), ca_list_value);
752        let reg_state: AttestedPasskeyRegistration = serde_json::from_value(state_value)
753            .map_err(|e| AuthError::Crypto(format!("deserialize attested state: {e}")))?;
754
755        let attested = match self
756            .webauthn
757            .finish_attested_passkey_registration(response, &reg_state)
758        {
759            Ok(a) => a,
760            // W2-D1 point 3: a `none`/self-attested credential cannot survive
761            // the attested ceremony — `finish_attested_passkey_registration`
762            // rejects it with `AttestationNotVerifiable` (verified by reading
763            // `webauthn-rs-core::attestation::verify_attestation_ca_chain`,
764            // which returns `Ok(None)` for `ParsedAttestationData::None`/
765            // `Self_`, which `register_credential_internal` then turns into
766            // this error because a ca_list was supplied). Map it to the same
767            // user-actionable denial D8 step 2 would have produced.
768            Err(WebauthnError::AttestationNotVerifiable) => {
769                tracing::warn!(
770                    action = "webauthn.attestation_denied",
771                    tenant_id = %tenant_id,
772                    user_id = %user_id,
773                    reason = ?AttestationDenyReason::AttestationRequired,
774                    "WebAuthn registration denied: attestation could not be verified \
775                     (no/self attestation presented to an attested ceremony)"
776                );
777                return Err(AuthError::WebauthnAttestationDenied {
778                    reason: AttestationDenyReason::AttestationRequired,
779                }
780                .into());
781            }
782            Err(e) => return Err(AuthError::WebauthnRegistration(e.to_string()).into()),
783        };
784
785        let (aaguid, attestation_format) = extract_attestation_metadata(&attested);
786
787        let mds_entry: Option<MdsEntry> = match aaguid {
788            Some(id) => metadata.get_entry(id).await?,
789            None => None,
790        };
791        let candidate = AttestationCandidate {
792            aaguid,
793            attestation_format: attestation_format.as_deref(),
794            mds_entry: mds_entry.as_ref(),
795        };
796
797        if let AttestationDecision::Deny(reason) = evaluate(policy, &candidate) {
798            tracing::warn!(
799                action = "webauthn.attestation_denied",
800                tenant_id = %tenant_id,
801                user_id = %user_id,
802                aaguid = ?aaguid,
803                reason = ?reason,
804                "WebAuthn registration denied by attestation policy"
805            );
806            return Err(AuthError::WebauthnAttestationDenied { reason }.into());
807        }
808
809        let authenticator_name = mds_entry.and_then(|e| e.description);
810
811        let passkey: Passkey = Passkey::from(attested);
812        let encryption_key = self.require_encryption_key()?;
813        let passkey_json_plain = serde_json::to_string(&passkey)
814            .map_err(|e| AuthError::Crypto(format!("serialize passkey: {e}")))?;
815        let passkey_json_enc =
816            totp::encrypt_secret(&encryption_key, passkey_json_plain.as_bytes())?;
817        let credential_id_str = URL_SAFE_NO_PAD.encode(passkey.cred_id().as_ref());
818
819        let created = self
820            .credential_repo
821            .create(CreateWebauthnCredential {
822                tenant_id,
823                user_id,
824                credential_id: credential_id_str,
825                name: credential_name.to_string(),
826                credential_type: WebauthnCredentialType::SecurityKey,
827                passkey_json: passkey_json_enc,
828                aaguid,
829                attestation_format,
830                attested: true,
831                authenticator_name,
832            })
833            .await?;
834
835        Ok(created)
836    }
837
838    // ---- Policy-routed entry points (the ones callers should use) ----
839
840    /// Start registration, choosing the ceremony from the tenant's policy.
841    ///
842    /// This exists so the choice between the attested and unattested ceremony
843    /// is made in **one** place, from the policy, rather than at every call
844    /// site. That choice is the whole feature: a caller that reaches for
845    /// [`Self::start_registration`] directly for a `direct_required` tenant
846    /// silently downgrades it to no attestation at all, and nothing else in
847    /// the system would notice — no error, no audit line, just a policy that
848    /// quietly stopped applying. Callers (REST/gRPC handlers) should use this
849    /// and [`Self::finish_registration_for_policy`]; the two ceremony-specific
850    /// methods remain public for tests and for callers that have already
851    /// resolved the ceremony themselves.
852    pub async fn start_registration_for_policy<M: AttestationMetadataSource>(
853        &self,
854        tenant_id: Uuid,
855        org_id: Uuid,
856        user_id: Uuid,
857        username: &str,
858        policy: &WebauthnAttestationPolicy,
859        metadata: &M,
860        ca_cache: &crate::attestation::AttestationCaCache,
861    ) -> AxiamResult<(CreationChallengeResponse, String)> {
862        if policy.mode == AttestationMode::None {
863            return self
864                .start_registration(tenant_id, org_id, user_id, username)
865                .await;
866        }
867
868        let ca_list = ca_cache
869            .get_or_build(metadata, policy.allowed_aaguids.as_deref())
870            .await?;
871        self.start_attested_registration(tenant_id, org_id, user_id, username, (*ca_list).clone())
872            .await
873    }
874
875    /// Complete registration, routing on the ceremony that actually *started*
876    /// and re-checking it against the policy in force **now**.
877    ///
878    /// Routing on the state token's `purpose` rather than on the current
879    /// policy is deliberate: the token encodes which ceremony produced the
880    /// challenge the authenticator answered, and that cannot be revised after
881    /// the fact. But a policy can be tightened while a ceremony is in flight,
882    /// so an unattested token arriving at a tenant that now requires
883    /// attestation is **denied** rather than completed — the alternative is a
884    /// window, however short, in which a newly-enforced policy can be
885    /// sidestepped by a ceremony started a moment before it was applied.
886    ///
887    /// The reverse direction needs no special case: an attested token
888    /// arriving at a now-`mode: none` tenant still goes through full
889    /// attestation verification and the (now permissive) policy, which is
890    /// strictly safer than the tenant's current setting requires.
891    #[allow(clippy::too_many_arguments)] // finish_registration's five parameters plus the three policy inputs; a wrapper struct would relocate the naming, not reduce it.
892    pub async fn finish_registration_for_policy<M: AttestationMetadataSource>(
893        &self,
894        tenant_id: Uuid,
895        caller_user_id: Uuid,
896        state_token: &str,
897        credential_name: &str,
898        response: &RegisterPublicKeyCredential,
899        policy: &WebauthnAttestationPolicy,
900        metadata: &M,
901        ca_cache: &crate::attestation::AttestationCaCache,
902    ) -> AxiamResult<WebauthnCredential> {
903        if self.state_token_purpose(state_token)? == PURPOSE_REGISTER_ATTESTED {
904            let ca_list = ca_cache
905                .get_or_build(metadata, policy.allowed_aaguids.as_deref())
906                .await?;
907            return self
908                .finish_attested_registration(
909                    tenant_id,
910                    caller_user_id,
911                    state_token,
912                    credential_name,
913                    response,
914                    policy,
915                    metadata,
916                    &ca_list,
917                )
918                .await;
919        }
920
921        if policy.mode != AttestationMode::None {
922            tracing::warn!(
923                action = "webauthn.attestation_denied",
924                tenant_id = %tenant_id,
925                user_id = %caller_user_id,
926                reason = ?AttestationDenyReason::AttestationRequired,
927                "WebAuthn registration denied: unattested ceremony finishing against a \
928                 tenant whose attestation policy now requires attestation"
929            );
930            return Err(AuthError::WebauthnAttestationDenied {
931                reason: AttestationDenyReason::AttestationRequired,
932            }
933            .into());
934        }
935
936        self.finish_registration(
937            tenant_id,
938            caller_user_id,
939            state_token,
940            credential_name,
941            response,
942        )
943        .await
944    }
945}

947/// Derive `(aaguid, attestation_format)` (D6) from an `AttestedPasskey`.
948///
949/// `AttestedPasskey` exposes no direct `attestation_format` getter (the
950/// underlying `Credential::attestation_format` field is crate-private), so
951/// this reads the AAGUID-bearing `AttestationMetadata` variant instead — the
952/// attested ceremony only ever produces `Packed` or `Tpm` (W2-D1: the request
953/// builder restricts `attestation_formats` to exactly those two), so matching
954/// on that pair covers every value this method can actually see; `_ => (None,
955/// None)` exists only as a fail-safe for a format this crate does not expect
956/// here, not a path exercised by `webauthn-rs` 0.5.5's attested ceremony.
957fn extract_attestation_metadata(attested: &AttestedPasskey) -> (Option<Uuid>, Option<String>) {
958    match &attested.attestation().metadata {
959        AttestationMetadata::Packed { aaguid } => (Some(*aaguid), Some("packed".to_string())),
960        AttestationMetadata::Tpm { aaguid, .. } => (Some(*aaguid), Some("tpm".to_string())),
961        _ => (None, None),
962    }
963}

965/// Upgrade a registration challenge to demand a **discoverable** credential.
966///
967/// `webauthn-rs` 0.5.5's `start_passkey_registration` hard-codes
968/// `require_resident_key(false)` internally, and its core builder turns that
969/// into `residentKey: "discouraged"` — not "unspecified", but an active request
970/// that the authenticator NOT store a discoverable credential. A credential
971/// created that way has no user handle to discover, so usernameless sign-in
972/// cannot find it no matter how the authentication ceremony is written. That is
973/// why this is a registration-side fix and not only an authentication one.
974///
975/// The library exposes no builder knob for this on the passkey path, so the
976/// challenge is amended after the fact. That is safe because residency is a
977/// property of the *request*: `finish_passkey_registration` does not re-derive
978/// or validate `authenticatorSelection` from the stored ceremony state, so the
979/// two cannot disagree.
980///
981/// `requireResidentKey` is set alongside `residentKey` because they are the
982/// WebAuthn L1 and L2 spellings of the same requirement, and an L1-era
983/// authenticator reads only the former.
984///
985/// Everything else on the criteria — notably `userVerification`, which
986/// `start_passkey_registration` sets to `Required` — is preserved rather than
987/// rebuilt, so this widens the request in exactly one dimension.
988///
989/// **Existing credentials are not retroactively discoverable.** Passkeys
990/// enrolled before this change were created under `discouraged`, and whether
991/// any given one is discoverable is the authenticator's decision, not ours —
992/// most platform authenticators (iCloud Keychain, Google Password Manager,
993/// Windows Hello) store one regardless, while a FIDO2 security key generally
994/// honours the hint and does not. Those users keep password sign-in and the
995/// passkey second factor unchanged; re-enrolling a passkey is what makes
996/// usernameless sign-in available to them.
997fn require_discoverable_credential(ccr: &mut CreationChallengeResponse) {
998    let selection = ccr
999        .public_key
1000        .authenticator_selection
1001        .get_or_insert_with(Default::default);
1002    selection.resident_key = Some(ResidentKeyRequirement::Required);
1003    selection.require_resident_key = true;
1004}

1659    /// panics if reached), so polling once is sufficient and avoids pulling a

1661    fn tokio_test_block_on<F: std::future::Future>(fut: F) -> F::Output {
1662        use std::task::{Context, Poll, Waker};

1664        match fut.as_mut().poll(&mut Context::from_waker(Waker::noop())) {
1665            Poll::Ready(v) => v,
