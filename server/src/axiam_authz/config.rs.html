10/// Default freshness window for an inbound cross-replica invalidation message
11/// (§4.2). See [`AuthzConfig::decision_cache_broadcast_skew_secs`].
12pub const DEFAULT_BROADCAST_SKEW_SECS: u64 = 30;

14/// Default interval between self-addressed liveness heartbeats (§13.4
15/// observation 1). See [`AuthzConfig::decision_cache_broadcast_heartbeat_secs`].
16pub const DEFAULT_BROADCAST_HEARTBEAT_SECS: u64 = 10;
17
18/// Hard ceiling on [`AuthzConfig::decision_cache_ttl_secs`] (§15.2).
19///
20/// The TTL is the only bound on how long a revoked grant can still be served by
21/// a replica that missed its invalidation — the heartbeat shortens the window in
22/// which invalidations go undelivered, but it cannot bound a window the operator
23/// set to hours. Five minutes is already far beyond anything the cache needs to
24/// be useful (the default is 5 **seconds**), so the ceiling costs nothing real
25/// and removes the foot-gun.
26pub const MAX_DECISION_CACHE_TTL_SECS: u64 = 300;
27
28/// Bounds on the heartbeat interval (§15.2).
29///
30/// A lower bound stops a misconfiguration turning the liveness probe into a
31/// broadcast flood; the upper bound keeps detection latency
32/// (`interval × HEARTBEAT_MISS_THRESHOLD`) inside the same order as the TTL
33/// ceiling it backstops.
34pub const MIN_BROADCAST_HEARTBEAT_SECS: u64 = 1;
35/// See [`MIN_BROADCAST_HEARTBEAT_SECS`].
36pub const MAX_BROADCAST_HEARTBEAT_SECS: u64 = 60;

38/// Strategy for evaluating a `BatchCheckAccess` call (REST + gRPC).
39///
40/// Both strategies produce **byte-identical decisions in the same order** —
41/// they differ only in how the DB work is scheduled, so switching is a pure
42/// performance choice with no authorization-semantics risk.
43#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize)]
44#[serde(rename_all = "snake_case")]
45pub enum BatchStrategy {
46    /// **D1 path (default since H3/G3).** Coalesce the shared role-assignment,
47    /// ancestor and grant lookups across same-subject/same-resource items into
48    /// a minimal set of DB round-trips (3 for the benchmark's 5-item shape).
49    /// The run-2 benchmark measured this serializing on the database under
50    /// contamination (DB pinned at ~1 core while idle, ~1 s p50); re-measured
51    /// on the G3 **settled** protocol (runs 2-3) it is the clear winner:
52    /// 744 batch ops/s = 3 721 checks/s = **4.98×** repeated single checks
53    /// (748/s) over REST, and 866-872 gRPC batch ops/s ≈ 4 330 checks/s with
54    /// p95 74 ms (well inside the ≤2 s gate). See
55    /// `claude_dev/authz-batch-investigation.md` for the closing verdict.
56    Coalesced,
57    /// **D10 path.** Evaluate each item as an independent `check_access`,
58    /// **concurrently**, bounded by `batch_max_concurrency`. Issues more DB
59    /// round-trips (one set per item) than `Coalesced`. On the G3 settled
60    /// protocol it also beats repeated single checks, but by a smaller margin
61    /// than `Coalesced`: 200 batch ops/s = 1 000 checks/s = 1.37× singles over
62    /// REST, gRPC batch 216 ops/s at p95 282 ms. Retained as an opt-in
63    /// A/B/fallback strategy. Decisions and result order are identical to the
64    /// coalesced path (the cache is consulted per item, exactly as for a plain
65    /// `check_access`).
66    Concurrent,
67}

69/// Configuration for the authorization engine.
70#[derive(Debug, Clone, Deserialize)]
71#[serde(default)]
72pub struct AuthzConfig {
73    /// Bound on concurrent `check_access` evaluations within a single
74    /// `BatchCheckAccess` call (gRPC and REST). Kept well under the
75    /// ~30-connection SurrealDB handle budget so a batch cannot self-DoS
76    /// the connection pool (D-07). Enforced by the `Concurrent`
77    /// [`BatchStrategy`]; ignored by `Coalesced` (which issues a fixed, small
78    /// number of round-trips regardless of batch size).
79    ///
80    /// Configure via `AXIAM__AUTHZ__BATCH_MAX_CONCURRENCY` env var.
81    pub batch_max_concurrency: usize,
82
83    /// How a `BatchCheckAccess` is evaluated (D10/G3/H3). Defaults to
84    /// [`BatchStrategy::Coalesced`] — round-trip-minimizing evaluation, which
85    /// the G3 re-measurement on the settled benchmark protocol showed winning
86    /// decisively over `concurrent` (4.98× vs 1.37× repeated single checks;
87    /// gRPC batch p95 74 ms vs 282 ms). Set to `concurrent` to restore the
88    /// D10 per-item-parallel behavior (still selectable for an
89    /// apples-to-apples A/B).
90    ///
91    /// Configure via `AXIAM__AUTHZ__BATCH_STRATEGY` (`coalesced` | `concurrent`).
92    pub batch_strategy: BatchStrategy,
93
94    /// Enable the per-tenant authorization **decision cache** (D7). When
95    /// `false` (the default) the engine issues its usual DB round-trips on
96    /// every check — behaviour is byte-for-byte identical to a build without
97    /// the cache. Only when `true` does `axiam-server` attach a
98    /// [`DecisionCache`] to the authorization engines.
99    ///
100    /// Configure via `AXIAM__AUTHZ__DECISION_CACHE_ENABLED` (default `false`).
101    ///
102    /// SECURITY: the cache is safe under AXIAM's additive allow-wins /
103    /// default-deny model *only because* every access-narrowing mutation
104    /// invalidates the affected entries immediately (see
105    /// `decision_cache` module docs). A stale allow can outlive a revocation
106    /// by at most `decision_cache_ttl_secs` even if an invalidation is missed.
107    ///
108    /// SECURITY — **multi-replica caveat, read before enabling**: on its own
109    /// the cache and its invalidation are **process-local**. "Revocation is
110    /// immediate" is then a *single-process* property. With two or more
111    /// replicas, a revocation handled by replica A does not reach replicas
112    /// B…N, which keep serving the pre-revocation decision until their own
113    /// entries expire — so the deployment's **worst-case revocation latency is
114    /// `decision_cache_ttl_secs` (default 5 s)**, on every read path including
115    /// the `RequirePermission` guard that protects the admin endpoints. Either
116    /// enable [`Self::decision_cache_broadcast_enabled`], run a single replica,
117    /// or accept a ≤ TTL revocation window.
118    pub decision_cache_enabled: bool,
119
120    /// Enable **cross-replica** decision-cache invalidation over RabbitMQ
121    /// (§4.2). Requires [`Self::decision_cache_enabled`]; ignored when the
122    /// cache is off.
123    ///
124    /// Configure via `AXIAM__AUTHZ__DECISION_CACHE_BROADCAST_ENABLED`
125    /// (default `false`).
126    ///
127    /// When `true`, every invalidation a mutation triggers is also published,
128    /// HMAC-signed, to a **fanout** exchange that every replica binds its own
129    /// exclusive auto-delete queue to — so a revocation propagates to all
130    /// replicas in broker-latency time instead of being bounded by the TTL.
131    /// Two consequences an operator must know about before flipping it:
132    ///
133    /// * **A mutation whose broadcast the broker does not confirm returns
134    ///   503.** The database write is durable, but the fan-out did not happen,
135    ///   so it is reported as a failure rather than silently falling back to
136    ///   the TTL window this switch exists to eliminate. These mutations are
137    ///   idempotent in the narrowing direction; retry is safe.
138    /// * **A replica whose invalidation consumer is not connected stops
139    ///   serving from its cache entirely** (falls back to full database
140    ///   evaluation — correct, slower) until it reconnects, rather than
141    ///   serving allows it can no longer invalidate. This is logged at ERROR
142    ///   and counted in `DecisionCacheStats::bypassed`.
143    ///
144    /// When `false` (the default) none of the above exists: no AMQP dependency
145    /// is added by the cache, `invalidate_*` cannot fail, and the documented
146    /// TTL-bounded behaviour above is unchanged.
147    pub decision_cache_broadcast_enabled: bool,
148
149    /// Freshness window, in seconds, for an inbound cross-replica invalidation
150    /// message (§4.2 / AMQP §8 `issued_at` gate). A broadcast whose `issued_at`
151    /// is outside ±this many seconds of the receiving replica's clock is
152    /// rejected and logged.
153    ///
154    /// Configure via `AXIAM__AUTHZ__DECISION_CACHE_BROADCAST_SKEW_SECS`
155    /// (default `30`).
156    ///
157    /// Deliberately far tighter than the 5-minute AMQP default
158    /// (`AXIAM__AMQP__REPLAY_SKEW_SECS`): an invalidation is only useful for
159    /// about as long as the cache TTL, so a short window both bounds how long a
160    /// captured message stays replay-eligible and bounds the memory of the
161    /// per-replica nonce guard. Raise it only if replica clocks are poorly
162    /// synchronised — a skew larger than the true clock spread buys nothing.
163    pub decision_cache_broadcast_skew_secs: u64,
164
165    /// Interval, in seconds, between the self-addressed liveness heartbeats a
166    /// replica publishes to prove its own queue is still bound to the fanout
167    /// exchange (§13.4 observation 1).
168    ///
169    /// Configure via `AXIAM__AUTHZ__DECISION_CACHE_BROADCAST_HEARTBEAT_SECS`
170    /// (default `10`). Only has any effect when
171    /// [`Self::decision_cache_broadcast_enabled`] is on.
172    ///
173    /// **What this buys.** The cache's trust flag otherwise follows *consumer*
174    /// liveness alone. A party with broker `configure` rights can `queue.unbind`
175    /// a replica's queue: the consumer stays subscribed to a queue nothing
176    /// routes to, trust stays on, and the replica serves cached allows it will
177    /// never be told to invalidate — while the publisher sees nothing, because
178    /// `mandatory` is off and the broker acks an unroutable message. A heartbeat
179    /// that has to come back to us tests the whole loop, so an unbound queue is
180    /// detected within
181    /// `decision_cache_broadcast_heartbeat_secs × HEARTBEAT_MISS_THRESHOLD`.
182    ///
183    /// **Cannot be disabled** while broadcast is on, and clamped to
184    /// `MIN_BROADCAST_HEARTBEAT_SECS..=MAX_BROADCAST_HEARTBEAT_SECS` (§15.2).
185    /// An earlier revision accepted `0` as "disabled"; that made the mitigation
186    /// removable with one environment variable and only a warning, leaving
187    /// `decision_cache_ttl_secs` as the sole bound — the exact state this
188    /// exists to escape.
189    ///
190    /// Raising it weakens detection linearly and saves almost nothing — one
191    /// transient 200-byte message per replica per interval.
192    pub decision_cache_broadcast_heartbeat_secs: u64,
193
194    /// TTL, in seconds, for a cached decision (D7). Bounds worst-case
195    /// revocation latency if an invalidation event is ever missed — and, on a
196    /// multi-replica deployment, bounds the *actual* revocation latency on
197    /// every replica that did not handle the mutation (the cache is
198    /// process-local; see `decision_cache_enabled`). Short by design.
199    ///
200    /// Configure via `AXIAM__AUTHZ__DECISION_CACHE_TTL_SECS` (default `5`).
201    ///
202    /// **Clamped to [`MAX_DECISION_CACHE_TTL_SECS`] at construction** (§15.2).
203    /// This value is the *only* bound on how long a revoked grant can still be
204    /// served on a replica that missed its invalidation, so an unbounded `u64`
205    /// let an operator configure a multi-hour stale-allow window — and the
206    /// cross-replica heartbeat, which exists to shorten that window, cannot
207    /// help once the window itself is arbitrary. `cleanup_interval_secs` has
208    /// been clamped for this reason since T-04-35; this one had not been.
209    pub decision_cache_ttl_secs: u64,
210
211    /// Maximum cached decisions retained **per tenant** before FIFO eviction
212    /// (D7). Bounds memory.
213    ///
214    /// Configure via `AXIAM__AUTHZ__DECISION_CACHE_MAX_ENTRIES` (default
215    /// `10000`).
216    pub decision_cache_max_entries: usize,
217}

219impl Default for AuthzConfig {
220    fn default() -> Self {
221        Self {
222            batch_max_concurrency: 16,
223            batch_strategy: BatchStrategy::Coalesced,
224            decision_cache_enabled: false,
225            decision_cache_broadcast_enabled: false,
226            decision_cache_broadcast_skew_secs: DEFAULT_BROADCAST_SKEW_SECS,
227            decision_cache_broadcast_heartbeat_secs: DEFAULT_BROADCAST_HEARTBEAT_SECS,
228            decision_cache_ttl_secs: 5,
229            decision_cache_max_entries: 10_000,
230        }
231    }

234impl AuthzConfig {
235    /// Build the shared [`DecisionCache`] iff caching is enabled, returning it
236    /// as an `Arc` ready to attach to every authorization engine
237    /// (`AuthorizationEngine::with_decision_cache`). Returns `None` when the
238    /// feature flag is off — the caller then constructs engines exactly as
239    /// before (no cache, no behaviour change).
240    ///
241    /// The *same* `Arc<DecisionCache>` must be shared across the REST, gRPC and
242    /// AMQP engines so that an invalidation triggered from a REST mutation
243    /// handler is observed by every read path. (All role/permission/resource
244    /// mutations are REST endpoints today.)
245    /// When cross-replica invalidation is enabled the returned cache starts
246    /// **untrusted** (`DecisionCache::set_trusted(false)`): a replica must not
247    /// serve a single cached decision before its invalidation consumer has
248    /// actually subscribed, or the startup window would be exactly the
249    /// stale-allow hole this feature closes. The consumer flips it to trusted
250    /// once it is receiving.
251    pub fn build_decision_cache(&self) -> Option<Arc<DecisionCache>> {
252        if !self.decision_cache_enabled {
253            return None;
254        }
255        let cache = DecisionCache::new(DecisionCacheConfig {
256            ttl: Duration::from_secs(self.decision_cache_ttl_secs()),
257            max_entries_per_tenant: self.decision_cache_max_entries,
258        });
259        if self.decision_cache_broadcast_enabled {
260            cache.set_trusted(false);
261        }
262        Some(Arc::new(cache))
263    }
264
265    /// Whether cross-replica invalidation is actually active — the broadcast
266    /// switch means nothing without the cache it invalidates.
267    pub fn cross_replica_invalidation_enabled(&self) -> bool {
268        self.decision_cache_enabled && self.decision_cache_broadcast_enabled
269    }
270
271    /// The configured inbound-broadcast freshness window as a
272    /// `chrono::Duration`, mirroring `AmqpConfig::replay_skew`.
273    pub fn decision_cache_broadcast_skew(&self) -> chrono::Duration {
274        chrono::Duration::seconds(self.decision_cache_broadcast_skew_secs as i64)
275    }
276
277    /// The effective decision-cache TTL, clamped to
278    /// [`MAX_DECISION_CACHE_TTL_SECS`] (§15.2).
279    ///
280    /// Clamped here rather than in `main.rs` — where `cleanup_interval_secs` is
281    /// clamped — so that *every* construction path is covered, including tests
282    /// and any future embedder that builds an `AuthzConfig` directly. A bound
283    /// that only applies when one particular binary happens to apply it is not
284    /// really a bound.
285    pub fn decision_cache_ttl_secs(&self) -> u64 {
286        let configured = self.decision_cache_ttl_secs;
287        if configured > MAX_DECISION_CACHE_TTL_SECS {
288            tracing::warn!(
289                configured,
290                clamped_to = MAX_DECISION_CACHE_TTL_SECS,
291                "AXIAM__AUTHZ__DECISION_CACHE_TTL_SECS exceeds the maximum and has been clamped. \
292                 This value bounds how long a REVOKED grant can still be served on a replica that \
293                 missed its invalidation, so it is a security bound, not a tuning knob."
294            );
295        }
296        configured.min(MAX_DECISION_CACHE_TTL_SECS)
297    }
298
299    /// The heartbeat interval, or `None` when cross-replica invalidation is not
300    /// active at all.
301    ///
302    /// **There is deliberately no way to disable heartbeats while broadcast is
303    /// on** (§15.2). An earlier revision treated `0` as "disabled", which meant
304    /// the mitigation for the queue-unbind suppression hole could be switched
305    /// off with a single environment variable and only a warning — leaving the
306    /// TTL as the sole bound, which is precisely the state the heartbeat exists
307    /// to escape. A heartbeat costs one transient ~200-byte message per replica
308    /// per interval, so there is no deployment for which disabling it is the
309    /// right trade. Out-of-range values are clamped, not honoured.
310    pub fn decision_cache_broadcast_heartbeat(&self) -> Option<chrono::Duration> {
311        if !self.cross_replica_invalidation_enabled() {
312            return None;
313        }
314        let configured = self.decision_cache_broadcast_heartbeat_secs;
315        let effective =
316            configured.clamp(MIN_BROADCAST_HEARTBEAT_SECS, MAX_BROADCAST_HEARTBEAT_SECS);
317        if effective != configured {
318            tracing::warn!(
319                configured,
320                clamped_to = effective,
321                "AXIAM__AUTHZ__DECISION_CACHE_BROADCAST_HEARTBEAT_SECS is out of range and has \
322                 been clamped. Heartbeats cannot be disabled while cross-replica invalidation is \
323                 enabled: they are what detects a replica whose queue has been unbound."
324            );
325        }
326        Some(chrono::Duration::seconds(effective as i64))
327    }
328}

530            "build_decision_cache must not hand an unclamped TTL to the cache"
531        );
532        assert_ne!(
533            cfg.decision_cache_ttl_secs(),
534            cfg.decision_cache_ttl_secs,
535            "precondition: the raw field really is out of range here"
536        );
537    }

539    /// Heartbeats cannot be switched off while broadcast is on. An earlier
540    /// revision treated `0` as "disabled", which let one environment variable
541    /// remove the mitigation for the queue-unbind hole.
542    #[test]
543    fn heartbeats_cannot_be_disabled_while_broadcast_is_on() {
544        let cfg = AuthzConfig {
545            decision_cache_enabled: true,
546            decision_cache_broadcast_enabled: true,
547            decision_cache_broadcast_heartbeat_secs: 0,
548            ..AuthzConfig::default()
549        };
550        let interval = cfg
551            .decision_cache_broadcast_heartbeat()
552            .expect("heartbeats must stay on");
553        assert_eq!(
554            interval.num_seconds() as u64,
555            MIN_BROADCAST_HEARTBEAT_SECS,
556            "0 must clamp to the minimum, not disable the watchdog"
557        );
558    }

560    /// An absurdly long interval is clamped too: detection latency is
561    /// `interval * HEARTBEAT_MISS_THRESHOLD`, so an unbounded interval is an
562    /// unbounded suppression window by another route.
563    #[test]
564    fn an_over_long_heartbeat_interval_is_clamped() {
565        let cfg = AuthzConfig {
566            decision_cache_enabled: true,
567            decision_cache_broadcast_enabled: true,

571        assert_eq!(
572            cfg.decision_cache_broadcast_heartbeat()
573                .unwrap()
574                .num_seconds() as u64,
575            MAX_BROADCAST_HEARTBEAT_SECS
576        );
577    }
