10/// Strategy for evaluating a `BatchCheckAccess` call (REST + gRPC).
11///
12/// Both strategies produce **byte-identical decisions in the same order** —
13/// they differ only in how the DB work is scheduled, so switching is a pure
14/// performance choice with no authorization-semantics risk.
15#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize)]
16#[serde(rename_all = "snake_case")]
17pub enum BatchStrategy {
18    /// **D1 path (default since H3/G3).** Coalesce the shared role-assignment,
19    /// ancestor and grant lookups across same-subject/same-resource items into
20    /// a minimal set of DB round-trips (3 for the benchmark's 5-item shape).
21    /// The run-2 benchmark measured this serializing on the database under
22    /// contamination (DB pinned at ~1 core while idle, ~1 s p50); re-measured
23    /// on the G3 **settled** protocol (runs 2-3) it is the clear winner:
24    /// 744 batch ops/s = 3 721 checks/s = **4.98×** repeated single checks
25    /// (748/s) over REST, and 866-872 gRPC batch ops/s ≈ 4 330 checks/s with
26    /// p95 74 ms (well inside the ≤2 s gate). See
27    /// `claude_dev/authz-batch-investigation.md` for the closing verdict.
28    Coalesced,
29    /// **D10 path.** Evaluate each item as an independent `check_access`,
30    /// **concurrently**, bounded by `batch_max_concurrency`. Issues more DB
31    /// round-trips (one set per item) than `Coalesced`. On the G3 settled
32    /// protocol it also beats repeated single checks, but by a smaller margin
33    /// than `Coalesced`: 200 batch ops/s = 1 000 checks/s = 1.37× singles over
34    /// REST, gRPC batch 216 ops/s at p95 282 ms. Retained as an opt-in
35    /// A/B/fallback strategy. Decisions and result order are identical to the
36    /// coalesced path (the cache is consulted per item, exactly as for a plain
37    /// `check_access`).
38    Concurrent,
39}

41/// Configuration for the authorization engine.
42#[derive(Debug, Clone, Deserialize)]
43#[serde(default)]
44pub struct AuthzConfig {
45    /// Bound on concurrent `check_access` evaluations within a single
46    /// `BatchCheckAccess` call (gRPC and REST). Kept well under the
47    /// ~30-connection SurrealDB handle budget so a batch cannot self-DoS
48    /// the connection pool (D-07). Enforced by the `Concurrent`
49    /// [`BatchStrategy`]; ignored by `Coalesced` (which issues a fixed, small
50    /// number of round-trips regardless of batch size).
51    ///
52    /// Configure via `AXIAM__AUTHZ__BATCH_MAX_CONCURRENCY` env var.
53    pub batch_max_concurrency: usize,
54
55    /// How a `BatchCheckAccess` is evaluated (D10/G3/H3). Defaults to
56    /// [`BatchStrategy::Coalesced`] — round-trip-minimizing evaluation, which
57    /// the G3 re-measurement on the settled benchmark protocol showed winning
58    /// decisively over `concurrent` (4.98× vs 1.37× repeated single checks;
59    /// gRPC batch p95 74 ms vs 282 ms). Set to `concurrent` to restore the
60    /// D10 per-item-parallel behavior (still selectable for an
61    /// apples-to-apples A/B).
62    ///
63    /// Configure via `AXIAM__AUTHZ__BATCH_STRATEGY` (`coalesced` | `concurrent`).
64    pub batch_strategy: BatchStrategy,
65
66    /// Enable the per-tenant authorization **decision cache** (D7). When
67    /// `false` (the default) the engine issues its usual DB round-trips on
68    /// every check — behaviour is byte-for-byte identical to a build without
69    /// the cache. Only when `true` does `axiam-server` attach a
70    /// [`DecisionCache`] to the authorization engines.
71    ///
72    /// Configure via `AXIAM__AUTHZ__DECISION_CACHE_ENABLED` (default `false`).
73    ///
74    /// SECURITY: the cache is safe under AXIAM's additive allow-wins /
75    /// default-deny model *only because* every access-narrowing mutation
76    /// invalidates the affected entries immediately (see
77    /// `decision_cache` module docs). A stale allow can outlive a revocation
78    /// by at most `decision_cache_ttl_secs` even if an invalidation is missed.
79    ///
80    /// SECURITY — **multi-replica caveat, read before enabling**: the cache and
81    /// its invalidation are **process-local**. "Revocation is immediate" is a
82    /// *single-process* property. With two or more replicas, a revocation
83    /// handled by replica A does not reach replicas B…N, which keep serving the
84    /// pre-revocation decision until their own entries expire — so the
85    /// deployment's **worst-case revocation latency is
86    /// `decision_cache_ttl_secs` (default 5 s)**, on every read path including
87    /// the `RequirePermission` guard that protects the admin endpoints. Enable
88    /// this only on a single replica, or where a ≤ TTL revocation window is
89    /// acceptable.
90    pub decision_cache_enabled: bool,
91
92    /// TTL, in seconds, for a cached decision (D7). Bounds worst-case
93    /// revocation latency if an invalidation event is ever missed — and, on a
94    /// multi-replica deployment, bounds the *actual* revocation latency on
95    /// every replica that did not handle the mutation (the cache is
96    /// process-local; see `decision_cache_enabled`). Short by design.
97    ///
98    /// Configure via `AXIAM__AUTHZ__DECISION_CACHE_TTL_SECS` (default `5`).
99    pub decision_cache_ttl_secs: u64,
100
101    /// Maximum cached decisions retained **per tenant** before FIFO eviction
102    /// (D7). Bounds memory.
103    ///
104    /// Configure via `AXIAM__AUTHZ__DECISION_CACHE_MAX_ENTRIES` (default
105    /// `10000`).
106    pub decision_cache_max_entries: usize,
107}
108
109impl Default for AuthzConfig {
110    fn default() -> Self {
111        Self {
112            batch_max_concurrency: 16,
113            batch_strategy: BatchStrategy::Coalesced,
114            decision_cache_enabled: false,
115            decision_cache_ttl_secs: 5,
116            decision_cache_max_entries: 10_000,
117        }
118    }
119}
120
121impl AuthzConfig {
122    /// Build the shared [`DecisionCache`] iff caching is enabled, returning it
123    /// as an `Arc` ready to attach to every authorization engine
124    /// (`AuthorizationEngine::with_decision_cache`). Returns `None` when the
125    /// feature flag is off — the caller then constructs engines exactly as
126    /// before (no cache, no behaviour change).
127    ///
128    /// The *same* `Arc<DecisionCache>` must be shared across the REST, gRPC and
129    /// AMQP engines so that an invalidation triggered from a REST mutation
130    /// handler is observed by every read path. (All role/permission/resource
131    /// mutations are REST endpoints today.)
132    pub fn build_decision_cache(&self) -> Option<Arc<DecisionCache>> {
133        if !self.decision_cache_enabled {
134            return None;
135        }
136        Some(Arc::new(DecisionCache::new(DecisionCacheConfig {
137            ttl: Duration::from_secs(self.decision_cache_ttl_secs),
138            max_entries_per_tenant: self.decision_cache_max_entries,
139        })))
140    }
141}

147    #[test]
148    fn default_batch_max_concurrency_is_16() {
149        let cfg = AuthzConfig::default();

153    #[test]

155        let cfg: AuthzConfig = serde_json::from_str("{}").expect("empty object must deserialize");
156        assert_eq!(cfg.batch_max_concurrency, 16);
157    }
158
159    #[test]
160    fn deserializes_explicit_override() {
161        let cfg: AuthzConfig = serde_json::from_str(r#"{"batch_max_concurrency": 4}"#)
162            .expect("explicit override must deserialize");
163        assert_eq!(cfg.batch_max_concurrency, 4);
164    }
165
166    #[test]
167    fn default_batch_strategy_is_coalesced() {
168        // H3/G3: coalesced round-trip-minimizing evaluation is the shipped
169        // default — the settled-protocol re-measurement showed it winning
170        // decisively (4.98x vs 1.37x repeated single checks); `concurrent`
171        // remains selectable as an opt-in A/B strategy.
172        let cfg = AuthzConfig::default();
173        assert_eq!(cfg.batch_strategy, BatchStrategy::Coalesced);
174    }
175
176    #[test]
177    fn deserializes_batch_strategy_override() {
178        let cfg: AuthzConfig = serde_json::from_str(r#"{"batch_strategy": "coalesced"}"#)
179            .expect("batch_strategy override must deserialize");
180        assert_eq!(cfg.batch_strategy, BatchStrategy::Coalesced);
181        let cfg: AuthzConfig = serde_json::from_str(r#"{"batch_strategy": "concurrent"}"#)
182            .expect("batch_strategy override must deserialize");

184    }
