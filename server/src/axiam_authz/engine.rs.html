20/// Permission evaluation engine.
21///
22/// Implements the design-doc algorithm:
23/// 1. Fetch subject's role assignments (direct + group, with resource scope)
24/// 2. Filter applicable roles (global, scoped to target resource, or ancestor)
25/// 3. Collect permissions from applicable roles
26/// 4. Check if requested action matches any permission
27/// 5. Validate scope if specified
28/// 6. Default deny
29pub struct AuthorizationEngine<R, P, Res, S, G>
30where
31    R: RoleRepository,
32    P: PermissionRepository,
33    Res: ResourceRepository,
34    S: ScopeRepository,
35    G: GroupRepository,
36{
37    role_repo: R,
38    permission_repo: P,
39    resource_repo: Res,
40    scope_repo: S,
41    #[allow(dead_code)]
42    group_repo: G,
43    /// Optional per-tenant decision cache (D7). `None` unless
44    /// `with_decision_cache` was called (feature-flagged in `axiam-server`).
45    /// When `None`, `check_access` / `check_access_batch` behave exactly as a
46    /// build without the cache.
47    decision_cache: Option<Arc<DecisionCache>>,
48    /// Optional cross-replica invalidation transport (§4.2). `None` unless
49    /// `with_invalidation_broadcaster` was called — which `axiam-server` does
50    /// only when the decision cache *and* the broadcast channel are both
51    /// enabled. When `None`, `invalidate_*` is local-only and infallible,
52    /// exactly as before.
53    invalidation_broadcaster: Option<Arc<dyn InvalidationBroadcaster>>,
54    /// How `check_access_batch` schedules its work (D10). Defaults to
55    /// [`BatchStrategy::Concurrent`] when an engine is constructed directly via
56    /// `new()`; `axiam-server` always overrides this from `AuthzConfig` via
57    /// `with_batch_config` (see `AXIAM__AUTHZ__BATCH_STRATEGY`), whose own
58    /// default is [`BatchStrategy::Coalesced`] as of H3/G3 — that is the
59    /// shipped default in production. Never affects decisions or their order.
60    batch_strategy: BatchStrategy,
61    /// Bound on in-flight per-item evaluations under
62    /// [`BatchStrategy::Concurrent`] (D-07 pool-safety). Ignored by
63    /// `Coalesced`. Defaults to 16.
64    batch_max_concurrency: usize,
65}

72/// Filter a subject's role assignments down to those applicable to
73/// `resource_id`, returning the deduplicated list of applicable role IDs in
74/// first-seen order.
75///
76/// A role applies when it is global, scoped directly to the target resource,
77/// or scoped to any ancestor of the target resource (hierarchy inheritance).
78fn applicable_role_ids(
79    assignments: &[RoleAssignment],
80    resource_id: Uuid,
81    ancestor_ids: &HashSet<Uuid>,
82) -> Vec<Uuid> {
83    let mut seen = HashSet::new();
84    assignments
85        .iter()
86        .filter(|a| {
87            a.role.is_global
88                || a.resource_id == Some(resource_id)
89                || a.resource_id
90                    .map(|rid| ancestor_ids.contains(&rid))
91                    .unwrap_or(false)
92        })
93        .map(|a| a.role.id)
94        .filter(|role_id| seen.insert(*role_id))
95        .collect()
96}

98/// Whether a single grant applies to `action` under `requested_scope_id`.
99///
100/// Wildcard grants (empty `scope_ids`) match any requested scope; otherwise the
101/// requested scope must be present in the grant's scope list. When no scope is
102/// requested, an action match is sufficient.
103///
104/// **The wildcard rule is what makes a resource-level deny stronger than a
105/// scope-level one** (B1 design §2.3): a deny with no scopes matches every
106/// request for that action regardless of the scope named, so it masks the
107/// action entirely. A scoped deny masks only the scopes it names.
108fn grant_applies(grant: &PermissionGrant, action: &str, requested_scope_id: Option<Uuid>) -> bool {
109    if grant.permission.action != action {
110        return false;
111    }
112    match requested_scope_id {
113        Some(scope_id) => grant.scope_ids.is_empty() || grant.scope_ids.contains(&scope_id),
114        None => true,
115    }
116}

118/// Outcome of evaluating the applicable grants (B1).
119#[derive(Debug, Clone, Copy, PartialEq, Eq)]
120enum GrantOutcome {
121    /// At least one allow matched and no deny did.
122    Allowed,
123    /// An explicit deny matched. Beats any allow, matched or not.
124    DeniedByRule,
125    /// Nothing matched — default deny.
126    NoGrant,
127}
128
129/// Given the applicable role IDs and the (shared) grants-by-role map, decide
130/// the outcome for `action` under the optional `requested_scope_id`.
131///
132/// # Deny-override, in one pass (B1)
133///
134/// A matched deny **short-circuits**; a matched allow is only remembered. Two
135/// consequences worth naming:
136///
137/// 1. **The common case pays one enum comparison per grant already being
138///    examined.** Denies live on the same `grants` edge as allows and arrive in
139///    the same batched fetch that already happens, so a tenant with no deny
140///    rules issues exactly the queries it issued before. There is no
141///    deny-specific round trip, and therefore no need for a per-tenant
142///    `has_denies` flag to gate one — which would have been a second source of
143///    truth able to go stale.
144/// 2. **The result does not depend on grant order.** Because deny
145///    short-circuits and allow only accumulates, a rule set has one answer, not
146///    one answer per query plan.
147fn evaluate_grants(
148    unique_role_ids: &[Uuid],
149    grants_by_role: &HashMap<Uuid, Vec<PermissionGrant>>,
150    action: &str,
151    requested_scope_id: Option<Uuid>,
152) -> GrantOutcome {
153    let mut saw_allow = false;
154    for role_id in unique_role_ids {
155        let Some(grants) = grants_by_role.get(role_id) else {
156            continue;
157        };
158        for grant in grants {
159            if !grant_applies(grant, action, requested_scope_id) {
160                continue;
161            }
162            if grant.effect.is_deny() {
163                // Deny wins over every allow, matched or not, at any depth of
164                // the hierarchy and at equal specificity. There is nothing
165                // later in the scan that could overturn this.
166                return GrantOutcome::DeniedByRule;
167            }
168            saw_allow = true;
169        }
170    }
171    if saw_allow {
172        GrantOutcome::Allowed
173    } else {
174        GrantOutcome::NoGrant
175    }
176}

178/// Turn a [`GrantOutcome`] into the engine's public decision, with the deny
179/// reason text the single-check and coalesced-batch paths must share.
180fn decision_for(outcome: GrantOutcome, action: &str) -> AccessDecision {
181    match outcome {
182        GrantOutcome::Allowed => AccessDecision::Allow,
183        GrantOutcome::DeniedByRule => {
184            AccessDecision::DeniedByRule(format!("an explicit deny rule refuses action '{action}'"))
185        }
186        GrantOutcome::NoGrant => {
187            AccessDecision::Deny(format!("no permission grants action '{action}'"))
188        }
189    }
190}

200    /// Build an engine over the five repositories it reads.
201    ///
202    /// The engine is generic over `axiam-core`'s repository *traits*, never
203    /// over a datastore: that is what lets the same evaluation logic serve the
204    /// REST, gRPC and AMQP entry points, and what keeps this crate at layer 1
205    /// with no dependency on `axiam-db`.
206    ///
207    /// The optional collaborators -- a decision cache and an invalidation
208    /// broadcaster -- are absent by default and added with
209    /// [`Self::with_decision_cache`] and
210    /// [`Self::with_invalidation_broadcaster`]. Absent means every check is
211    /// evaluated from the datastore, which is the conservative default: §11.2
212    /// rule 6 bans decision caching unless a deployment opts into it.
213    pub fn new(
214        role_repo: R,
215        permission_repo: P,
216        resource_repo: Res,
217        scope_repo: S,
218        group_repo: G,
219    ) -> Self {

221            role_repo,
222            permission_repo,
223            resource_repo,
224            scope_repo,
225            group_repo,

227            invalidation_broadcaster: None,

276    /// Immediately drop every cached decision for `tenant_id`, **locally and
277    /// on every other replica**. No local effect when no cache is attached.
278    /// Called by the REST mutation handlers for coarse, access-narrowing
279    /// changes whose affected-subject set isn't known without a query (grant
280    /// revoke, role/permission delete or update, group-role unassignment,
281    /// resource reparent/delete).
282    ///
283    /// # Errors
284    ///
285    /// Only when a broadcaster is attached and the broadcast could **not** be
286    /// confirmed by the broker — see [`Self::broadcast`]. With no broadcaster
287    /// (the default) this is infallible.
288    pub async fn invalidate_tenant(&self, tenant_id: Uuid) -> AxiamResult<()> {
289        self.invalidate(InvalidationEvent::Tenant { tenant_id })
290            .await
291    }

293    /// Immediately drop every cached decision for a single `subject_id` within
294    /// `tenant_id`, **locally and on every other replica**. No local effect
295    /// when no cache is attached. Called by the REST mutation handlers when
296    /// exactly one subject's effective permissions change (user role
297    /// unassign/assign, group membership add/remove).
298    ///
299    /// # Errors
300    ///
301    /// As [`Self::invalidate_tenant`].
302    pub async fn invalidate_subject(&self, tenant_id: Uuid, subject_id: Uuid) -> AxiamResult<()> {
303        self.invalidate(InvalidationEvent::Subject {
304            tenant_id,
305            subject_id,
306        })
307        .await
308    }

310    /// Local-then-remote invalidation.
311    ///
312    /// Order is deliberate: the **local** cache is dropped first and
313    /// unconditionally, so a broadcast failure can never leave the replica
314    /// that handled the mutation serving its own stale decision. Only then is
315    /// the fan-out attempted.
316    async fn invalidate(&self, event: InvalidationEvent) -> AxiamResult<()> {
317        if let Some(cache) = self.decision_cache.as_ref() {
318            event.apply(cache);
319        }
320        self.broadcast(event).await
321    }
322
323    /// Fan the event out to the other replicas, if a broadcaster is attached.
324    ///
325    /// # Errors
326    ///
327    /// Propagates the broadcaster's error, which the REST mutation handlers
328    /// turn into a **503**. That is the deliberate posture for §4.2: the
329    /// operator asked for a revocation, the database write is durable, but the
330    /// invalidation did **not** reach the other replicas — so it is reported as
331    /// a failure rather than silently degrading to the TTL-bounded window the
332    /// operator opted out of by enabling this channel. Retrying the mutation is
333    /// safe (all of these handlers are idempotent in the narrowing direction).
334    async fn broadcast(&self, event: InvalidationEvent) -> AxiamResult<()> {
335        match self.invalidation_broadcaster.as_ref() {
336            Some(b) => b.broadcast(event).await,
337            None => Ok(()),
338        }
339    }

341    /// Evaluate whether the subject can perform the requested action.
342    ///
343    /// Issues 3–4 sequential DB round-trips (role assignments, ancestors,
344    /// optional scope lookup, batched grants). For batches that repeat the same
345    /// subject/resource, prefer [`Self::check_access_batch`], which resolves the
346    /// shared lookups once.
347    #[tracing::instrument(
348        name = "authz.check_access",
349        skip(self, request),
350        fields(
351            tenant_id = %request.tenant_id,
352            subject_id = %request.subject_id,
353            resource_id = %request.resource_id,
354            action = %request.action,
355            scope = request.scope.as_deref().unwrap_or("")
356        )
357    )]
358    pub async fn check_access(&self, request: &AccessRequest) -> AxiamResult<AccessDecision> {
359        // D7 fast path: when a decision cache is attached (feature-flagged),
360        // a fresh cached decision skips the DB round-trips entirely. When no
361        // cache is attached this whole block is absent and the call is
362        // identical to a build without the cache.
363        if let Some(cache) = self.decision_cache.as_ref()
364            && let Some(decision) = cache.get(request)
365        {
366            return Ok(decision);
367        }
368        let decision = self.evaluate(request).await?;
369        if let Some(cache) = self.decision_cache.as_ref() {
370            // Cache the FULL decision (allow *or* deny-with-reason), so a hit
371            // is byte-identical to a miss.
372            cache.insert(request, decision.clone());
373        }
374        Ok(decision)
375    }

377    /// Uncached RBAC evaluation — the actual algorithm (3–4 sequential DB
378    /// round-trips). Kept as a separate method so the [`Self::check_access`]
379    /// cache wrapper can never alter decision or deny-reason semantics: a cache
380    /// hit returns exactly what this would have produced.
381    async fn evaluate(&self, request: &AccessRequest) -> AxiamResult<AccessDecision> {
382        // 1. Fetch all role assignments (direct + group) with resource scope.
383        let assignments = self
384            .role_repo
385            .get_user_role_assignments(request.tenant_id, request.subject_id)
386            .instrument(tracing::debug_span!("db.get_user_role_assignments"))
387            .await?;
388
389        if assignments.is_empty() {
390            return Ok(AccessDecision::Deny("no roles assigned".into()));
391        }
392
393        // 2. Build the set of ancestor resource IDs for hierarchy inheritance.
394        let ancestors = self
395            .resource_repo
396            .get_ancestors(request.tenant_id, request.resource_id)
397            .instrument(tracing::debug_span!("db.get_ancestors"))
398            .await?;
399        let ancestor_ids: HashSet<Uuid> = ancestors.iter().map(|r| r.id).collect();
400
401        // 3. Filter applicable roles (global / target-scoped / ancestor-scoped),
402        //    deduplicated in first-seen order.
403        let unique_role_ids = applicable_role_ids(&assignments, request.resource_id, &ancestor_ids);
404
405        if unique_role_ids.is_empty() {
406            return Ok(AccessDecision::Deny(
407                "no applicable roles for this resource".into(),
408            ));
409        }
410
411        // 4. Resolve the requested scope to an ID (if specified).
412        let requested_scope_id = match self.resolve_scope(request).await? {
413            ScopeResolution::None => None,
414            ScopeResolution::Resolved(id) => Some(id),
415            ScopeResolution::NotFound(deny) => return Ok(deny),
416        };
417
418        // 5. Fetch every applicable role's grants in a SINGLE batched query
419        //    (CQ-B13 N+1 fix), then evaluate action + scope constraints.
420        let grants_by_role = self
421            .permission_repo
422            .get_role_permission_grants_for_roles(request.tenant_id, &unique_role_ids)
423            .instrument(tracing::debug_span!(
424                "db.get_role_permission_grants_for_roles"
425            ))
426            .await?;
427
428        Ok(decision_for(
429            evaluate_grants(
430                &unique_role_ids,
431                &grants_by_role,
432                &request.action,
433                requested_scope_id,
434            ),
435            &request.action,
436        ))
437    }

439    /// Resolve the request's optional scope name to a scope ID on the target
440    /// resource. Shared by [`Self::check_access`] and the batch path.
441    async fn resolve_scope(&self, request: &AccessRequest) -> AxiamResult<ScopeResolution> {
442        let Some(ref scope_name) = request.scope else {
443            return Ok(ScopeResolution::None);
444        };
445        let scopes = self
446            .scope_repo
447            .list_by_resource(request.tenant_id, request.resource_id)
448            .instrument(tracing::debug_span!("db.list_by_resource"))
449            .await?;
450        match scopes.iter().find(|s| s.name == *scope_name) {
451            Some(s) => Ok(ScopeResolution::Resolved(s.id)),
452            None => Ok(ScopeResolution::NotFound(AccessDecision::Deny(format!(
453                "scope '{}' not found on resource",
454                scope_name
455            )))),
456        }
457    }

459    /// Evaluate an ordered batch of authorization checks.
460    ///
461    /// The returned `Vec` has the same length and order as `requests`; result
462    /// `i` is the decision for `requests[i]` and is byte-identical to what
463    /// [`Self::check_access`] would return for that request in isolation — this
464    /// holds for **both** [`BatchStrategy`] variants, which differ only in how
465    /// the DB work is scheduled:
466    ///
467    /// - [`BatchStrategy::Coalesced`] (default since H3/G3, D1) — shared
468    ///   role-assignment/ancestor/grant lookups resolved once per (tenant,
469    ///   subject|resource): 15 round-trips → 3 for the benchmark's 5-item
470    ///   shape. On the G3 settled-protocol re-measurement this wins decisively
471    ///   (4.98× repeated single checks over REST; gRPC batch p95 74 ms) —
472    ///   the run-2 serialization concern only applied to the contaminated
473    ///   cells (see `claude_dev/authz-batch-investigation.md`).
474    /// - [`BatchStrategy::Concurrent`] (D10) — each item is an independent
475    ///   cache-aware [`Self::check_access`], run concurrently (bounded by
476    ///   `batch_max_concurrency`). Recovers per-item DB parallelism; still
477    ///   beats repeated single checks (1.37×) but by a smaller margin than
478    ///   `Coalesced`. Retained as a selectable opt-in strategy.
479    pub async fn check_access_batch(
480        &self,
481        requests: &[AccessRequest],
482    ) -> AxiamResult<Vec<AccessDecision>> {
483        match self.batch_strategy {
484            BatchStrategy::Concurrent => self.evaluate_concurrent(requests).await,
485            BatchStrategy::Coalesced => self.evaluate_coalesced_cached(requests).await,
486        }
487    }
488
489    /// **D10 opt-in strategy** — evaluate each item as an independent,
490    /// cache-aware [`Self::check_access`], run **concurrently** with a bound of
491    /// `batch_max_concurrency` in-flight, preserving input order.
492    ///
493    /// This was the run-2-era default (run-2 evidence): the coalesced path
494    /// ([`Self::evaluate_batch`]) minimizes round-trips but resolves the whole
495    /// batch on a single task, which appeared to serialize on the database (DB
496    /// pinned at ~1 core, ~1 s p50, everything else idle) and did not beat
497    /// repeated single checks. Single `check_access` calls, by contrast,
498    /// saturated the DB (2 cores, 745 req/s) precisely because 50 of them ran
499    /// concurrently. Evaluating a batch's items the same way recovers that
500    /// parallelism. **G3 re-measured both strategies on the settled protocol**
501    /// and found the run-2 coalesced numbers were contaminated by the post-seed
502    /// clamp: `Coalesced` in fact wins decisively (4.98× vs this strategy's
503    /// 1.37×), so `Coalesced` is the default as of H3 and this strategy is now
504    /// opt-in. `check_access` is reused verbatim, so every item's decision,
505    /// deny reason, cache lookup and cache insert are identical to a standalone
506    /// call — `buffered` keeps results in request order.
507    async fn evaluate_concurrent(
508        &self,
509        requests: &[AccessRequest],
510    ) -> AxiamResult<Vec<AccessDecision>> {
511        use futures::stream::{StreamExt, TryStreamExt};
512
513        // Build the per-item futures EAGERLY into a Vec before handing them to
514        // the stream. Mapping `requests.iter()` lazily inside `stream::iter`
515        // would store the `|req| self.check_access(req)` closure in the stream
516        // adapter, and when the whole `check_access_batch` future is erased
517        // behind the `AuthzChecker` trait object (`Pin<Box<dyn Future + Send>>`
518        // in axiam-api-rest / the gRPC async-trait), the borrow checker cannot
519        // prove that closure is `for<'a> FnMut(&'a AccessRequest) -> …`
520        // ("implementation of `FnOnce` is not general enough"). Collecting
521        // first applies the closure in this concrete-lifetime scope, so the
522        // boxed future holds only already-built item futures — no HRTB closure.
523        let item_futures: Vec<_> = requests.iter().map(|req| self.check_access(req)).collect();
524
525        futures::stream::iter(item_futures)
526            .buffered(self.batch_max_concurrency.max(1))
527            .try_collect()
528            .await
529    }
530
531    /// **D1 path** (default since H3/G3; `AXIAM__AUTHZ__BATCH_STRATEGY=coalesced`,
532    /// or the unset default) — coalesced evaluation with the D7 decision cache
533    /// layered on top: serve hits from the cache, evaluate only the misses
534    /// through the shared-lookup batch path, then backfill. Input order is
535    /// preserved.
536    async fn evaluate_coalesced_cached(
537        &self,
538        requests: &[AccessRequest],
539    ) -> AxiamResult<Vec<AccessDecision>> {
540        // With no cache attached, this is exactly the D1 coalesced path —
541        // zero behaviour change.
542        let Some(cache) = self.decision_cache.as_ref() else {
543            return self.evaluate_batch(requests).await;
544        };
545
546        // Cache-enabled path: serve hits from the cache, evaluate only the
547        // misses (still via the coalesced batch path so shared lookups are
548        // resolved once), then backfill the cache. Input order is preserved.
549        let mut results: Vec<Option<AccessDecision>> =
550            requests.iter().map(|r| cache.get(r)).collect();
551
552        let miss_indices: Vec<usize> = results
553            .iter()
554            .enumerate()
555            .filter_map(|(i, slot)| slot.is_none().then_some(i))
556            .collect();
557
558        if !miss_indices.is_empty() {
559            let miss_requests: Vec<AccessRequest> =
560                miss_indices.iter().map(|&i| requests[i].clone()).collect();
561            let miss_decisions = self.evaluate_batch(&miss_requests).await?;
562            for (slot, &i) in miss_indices.iter().enumerate() {
563                let decision = miss_decisions[slot].clone();
564                cache.insert(&requests[i], decision.clone());
565                results[i] = Some(decision);
566            }
567        }
568
569        // Every slot is now filled (hit or freshly evaluated).
570        Ok(results.into_iter().map(|d| d.expect("filled")).collect())
571    }

573    /// Uncached coalesced batch evaluation (the D1 fast path). Separated from
574    /// [`Self::check_access_batch`] so the D7 cache wrapper cannot alter
575    /// decision/deny-reason semantics.
576    #[tracing::instrument(
577        name = "authz.check_access_batch",
578        skip(self, requests),
579        fields(batch_size = requests.len())
580    )]
581    async fn evaluate_batch(&self, requests: &[AccessRequest]) -> AxiamResult<Vec<AccessDecision>> {
582        if requests.is_empty() {
583            return Ok(Vec::new());
584        }
585
586        // --- Coalesce round-trip 1: role assignments, once per (tenant, subject).
587        let mut assignments_by_subject: HashMap<(Uuid, Uuid), Vec<RoleAssignment>> = HashMap::new();
588        for req in requests {
589            let key = (req.tenant_id, req.subject_id);
590            if let std::collections::hash_map::Entry::Vacant(slot) =
591                assignments_by_subject.entry(key)
592            {
593                let assignments = self
594                    .role_repo
595                    .get_user_role_assignments(req.tenant_id, req.subject_id)
596                    .instrument(tracing::debug_span!(
597                        "db.get_user_role_assignments",
598                        tenant_id = %req.tenant_id,
599                        subject_id = %req.subject_id
600                    ))
601                    .await?;
602                slot.insert(assignments);
603            }
604        }
605
606        // Only items whose subject actually has role assignments proceed past
607        // the first gate — so ancestors/scopes/grants are fetched ONLY for
608        // resources those items target. This keeps the batch's round-trip
609        // profile identical to per-item `check_access` (no wasted ancestor walk
610        // for a subject that would deny at "no roles assigned").
611        let has_roles = |req: &AccessRequest| {
612            assignments_by_subject
613                .get(&(req.tenant_id, req.subject_id))
614                .map(|a| !a.is_empty())
615                .unwrap_or(false)
616        };
617
618        // --- Coalesce round-trip 2: ancestors, once per (tenant, resource).
619        let mut ancestors_by_resource: HashMap<(Uuid, Uuid), HashSet<Uuid>> = HashMap::new();
620        for req in requests.iter().filter(|r| has_roles(r)) {
621            let key = (req.tenant_id, req.resource_id);
622            if let std::collections::hash_map::Entry::Vacant(slot) =
623                ancestors_by_resource.entry(key)
624            {
625                let ancestors = self
626                    .resource_repo
627                    .get_ancestors(req.tenant_id, req.resource_id)
628                    .instrument(tracing::debug_span!(
629                        "db.get_ancestors",
630                        tenant_id = %req.tenant_id,
631                        resource_id = %req.resource_id
632                    ))
633                    .await?;
634                slot.insert(ancestors.iter().map(|r| r.id).collect());
635            }
636        }
637
638        // --- Coalesce round-trip 3: scope lists, once per (tenant, resource)
639        //     that any scoped, role-bearing item targets.
640        let mut scopes_by_resource: HashMap<(Uuid, Uuid), Vec<axiam_core::models::scope::Scope>> =
641            HashMap::new();
642        for req in requests
643            .iter()
644            .filter(|r| r.scope.is_some() && has_roles(r))
645        {
646            let key = (req.tenant_id, req.resource_id);
647            if let std::collections::hash_map::Entry::Vacant(slot) = scopes_by_resource.entry(key) {
648                let scopes = self
649                    .scope_repo
650                    .list_by_resource(req.tenant_id, req.resource_id)
651                    .instrument(tracing::debug_span!(
652                        "db.list_by_resource",
653                        tenant_id = %req.tenant_id,
654                        resource_id = %req.resource_id
655                    ))
656                    .await?;
657                slot.insert(scopes);
658            }
659        }
660
661        // --- Per-item: compute applicable roles from the shared lookups, and
662        //     accumulate the union of role IDs so grants can be fetched ONCE.
663        //     `PreDecision` short-circuits (deny) are recorded here so the final
664        //     pass need not re-derive them.
665        enum PreDecision {
666            Decided(AccessDecision),
667            NeedsGrants {
668                unique_role_ids: Vec<Uuid>,
669                requested_scope_id: Option<Uuid>,
670            },
671        }
672
673        let mut pre: Vec<PreDecision> = Vec::with_capacity(requests.len());
674        // Union of (tenant, role_id) that we must fetch grants for, deduped.
675        let mut grant_role_ids: HashMap<Uuid, Vec<Uuid>> = HashMap::new();
676        let mut grant_role_seen: HashSet<(Uuid, Uuid)> = HashSet::new();
677
678        for (i, req) in requests.iter().enumerate() {
679            let _item_span = tracing::debug_span!(
680                "authz.batch.item",
681                index = i,
682                resource_id = %req.resource_id,
683                action = %req.action
684            )
685            .entered();
686
687            let assignments = assignments_by_subject
688                .get(&(req.tenant_id, req.subject_id))
689                .map(Vec::as_slice)
690                .unwrap_or(&[]);
691            if assignments.is_empty() {
692                pre.push(PreDecision::Decided(AccessDecision::Deny(
693                    "no roles assigned".into(),
694                )));
695                continue;
696            }
697
698            // WIDENING fallback: a miss here silently yields "no ancestors",
699            // which can only ever *shrink* `applicable_role_ids` below —
700            // dropping an ancestor-scoped grant. If that dropped grant was a
701            // DENY, this would silently widen access (an ancestor-scoped deny
702            // would fail to apply). It is unreachable today only because this
703            // lookup key, `(req.tenant_id, req.resource_id)`, is populated by
704            // the round-trip-2 loop above under the exact same gate that
705            // guards this call site: the round-trip-2 loop runs over
706            // `requests.iter().filter(|r| has_roles(r))` (the `has_roles`
707            // closure defined above, keyed off `assignments_by_subject`), and
708            // this call site is reached only when the `assignments.is_empty()`
709            // check just above did NOT `continue` — i.e. only when
710            // `assignments_by_subject.get(&(req.tenant_id, req.subject_id))`
711            // is non-empty for this same `req`, which is exactly `has_roles`'s
712            // condition. So every `req` that reaches this line already had its
713            // `(tenant_id, resource_id)` key inserted into `ancestors_by_resource`
714            // by round-trip 2, and `.unwrap_or(&empty_ancestors)` never actually
715            // fires. If that population gate (`has_roles`) and this consumption
716            // gate (`assignments.is_empty()`) are ever allowed to diverge —
717            // e.g. by filtering round-trip 2 on something other than
718            // `has_roles`, or by reordering this block above the
719            // `assignments.is_empty()` check — a missed lookup here would
720            // silently drop an ancestor-scoped DENY instead of failing loudly.
721            let empty_ancestors = HashSet::new();
722            let ancestor_ids = ancestors_by_resource
723                .get(&(req.tenant_id, req.resource_id))
724                .unwrap_or(&empty_ancestors);
725
726            let unique_role_ids = applicable_role_ids(assignments, req.resource_id, ancestor_ids);
727            if unique_role_ids.is_empty() {
728                pre.push(PreDecision::Decided(AccessDecision::Deny(
729                    "no applicable roles for this resource".into(),
730                )));
731                continue;
732            }
733
734            // Resolve scope against the coalesced scope list (no I/O here).
735            let requested_scope_id = if let Some(ref scope_name) = req.scope {
736                let scopes = scopes_by_resource
737                    .get(&(req.tenant_id, req.resource_id))
738                    .map(Vec::as_slice)
739                    .unwrap_or(&[]);
740                match scopes.iter().find(|s| s.name == *scope_name) {
741                    Some(s) => Some(s.id),
742                    None => {
743                        pre.push(PreDecision::Decided(AccessDecision::Deny(format!(
744                            "scope '{}' not found on resource",
745                            scope_name
746                        ))));
747                        continue;
748                    }
749                }
750            } else {
751                None
752            };
753
754            for rid in &unique_role_ids {
755                if grant_role_seen.insert((req.tenant_id, *rid)) {
756                    grant_role_ids.entry(req.tenant_id).or_default().push(*rid);
757                }
758            }
759
760            pre.push(PreDecision::NeedsGrants {
761                unique_role_ids,
762                requested_scope_id,
763            });
764        }
765
766        // --- Coalesce round-trip 4: fetch grants for every applicable role in
767        //     the batch, one batched query per tenant.
768        //
769        // Keyed per-tenant (outer map keyed by `tenant_id`, inner by
770        // `role_id`) rather than by bare `role_id` across the whole batch —
771        // isomorphic to keying the flat map by `(tenant_id, role_id)`, and
772        // mirrors the `(Uuid, Uuid)` key `grant_role_seen` already uses a few
773        // lines above. `get_role_permission_grants_for_roles` returns a
774        // `role_id`-keyed map for a *single* tenant per call (it is invoked
775        // once per `tenant_id` in `grant_role_ids`), so a bare
776        // `grants_by_role.extend(map)` here would merge those per-tenant maps
777        // into one flat `role_id`-keyed map, and a `role_id` collision across
778        // two tenants in the same batch would silently overwrite one tenant's
779        // grants with another's. Nesting under `tenant_id` removes that
780        // reliance on UUIDv4 role-id uniqueness for cross-tenant batch
781        // isolation — a same-named `role_id` in two tenants' `Vec<Uuid>` below
782        // simply lands in two different inner maps and can never collide.
783        let mut grants_by_role: HashMap<Uuid, HashMap<Uuid, Vec<PermissionGrant>>> = HashMap::new();
784        for (tenant_id, role_ids) in &grant_role_ids {
785            let map = self
786                .permission_repo
787                .get_role_permission_grants_for_roles(*tenant_id, role_ids)
788                .instrument(tracing::debug_span!(
789                    "db.get_role_permission_grants_for_roles",
790                    tenant_id = %tenant_id,
791                    role_count = role_ids.len()
792                ))
793                .await?;
794            grants_by_role.entry(*tenant_id).or_default().extend(map);
795        }
796        let empty_role_grants: HashMap<Uuid, Vec<PermissionGrant>> = HashMap::new();
797
798        // --- Final pass: evaluate each item against the shared grants map,
799        //     preserving input order.
800        let mut decisions = Vec::with_capacity(requests.len());
801        for (item, req) in pre.into_iter().zip(requests.iter()) {
802            match item {
803                PreDecision::Decided(d) => decisions.push(d),
804                PreDecision::NeedsGrants {
805                    unique_role_ids,
806                    requested_scope_id,
807                } => {
808                    // Same two helpers the single-check path uses, so the
809                    // batch path's deny-override semantics and reason codes
810                    // cannot diverge from it. Scoped to this item's tenant
811                    // before handing off, so `evaluate_grants` (shared with
812                    // the single-check path, which is inherently
813                    // single-tenant) never sees another tenant's role ids.
814                    let tenant_grants = grants_by_role
815                        .get(&req.tenant_id)
816                        .unwrap_or(&empty_role_grants);
817                    decisions.push(decision_for(
818                        evaluate_grants(
819                            &unique_role_ids,
820                            tenant_grants,
821                            &req.action,
822                            requested_scope_id,
823                        ),
824                        &req.action,
825                    ));
826                }
827            }
828        }
829
830        Ok(decisions)
831    }
832}

1075    // -----------------------------------------------------------------

1077    /// **Adding a deny rule can never widen access.**
1078    ///
1079    /// This is the one guarantee deny-override buys over most-specific-wins,
1080    /// and it is what a future "most-specific-wins" refactor would break. The
1081    /// search is exhaustive over a small rule space rather than randomised, so
1082    /// it fails deterministically.
1083    #[test]
1084    fn adding_a_deny_can_never_widen_access() {
1085        let scope_a = Uuid::new_v4();
1086        let scope_b = Uuid::new_v4();
1087
1088        let candidate_grants = [
1089            allow("read"),
1090            allow("write"),
1091            grant_with("read", PermissionEffect::Allow, vec![scope_a]),
1092            deny("read"),
1093            grant_with("read", PermissionEffect::Deny, vec![scope_a]),
1094            grant_with("write", PermissionEffect::Deny, vec![scope_b]),
1095        ];
1096        let requests = [
1097            ("read", None),
1098            ("read", Some(scope_a)),
1099            ("read", Some(scope_b)),
1100            ("write", None),
1101            ("write", Some(scope_b)),
1102        ];
1103
1104        // Every subset of the candidate grants, as a base rule set.
1105        for mask in 0u32..(1 << candidate_grants.len()) {
1106            let base: Vec<PermissionGrant> = candidate_grants
1107                .iter()
1108                .enumerate()
1109                .filter(|(i, _)| mask & (1 << i) != 0)
1110                .map(|(_, g)| g.clone())
1111                .collect();
1112
1113            // Every possible additional DENY rule.
1114            for extra in candidate_grants.iter().filter(|g| g.effect.is_deny()) {
1115                let mut widened = base.clone();
1116                widened.push(extra.clone());
1117
1118                let (base_ids, base_map) = roles(vec![base.clone()]);
1119                let (wide_ids, wide_map) = roles(vec![widened]);
1120
1121                for (action, scope) in requests {
1122                    let before = evaluate_grants(&base_ids, &base_map, action, scope);
1123                    let after = evaluate_grants(&wide_ids, &wide_map, action, scope);
1124
1125                    if before != GrantOutcome::Allowed {
1126                        assert_ne!(
1127                            after,
1128                            GrantOutcome::Allowed,
1129                            "adding a deny turned a non-allow into an allow for \
1130                             ({action}, {scope:?}) — deny-override is broken"
1131                        );
1132                    }
1133                }
1134            }
1135        }
1136    }
