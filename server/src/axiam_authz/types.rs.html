5/// The result of an authorization check.
6///
7/// # Why deny has two shapes (B1)
8///
9/// `Deny` and `DeniedByRule` are both refusals and both become a 403. They are
10/// distinguished because they mean opposite things to the person who hit them:
11///
12/// - `Deny` — nothing granted this. **Ask an admin for access.**
13/// - `DeniedByRule` — an explicit deny rule matched. **An admin has already
14///   decided.**
15///
16/// Collapsing the two into a bare `false`, which is what a client does without
17/// this distinction, throws away the only information that tells a user which
18/// of those two situations they are in. SDK contract §11 requires helpers to
19/// surface the reason rather than flatten it.
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub enum AccessDecision {
22    /// A grant matched and no explicit deny overrode it.
23    Allow,
24    /// Default deny — no grant matched.
25    Deny(String),
26    /// An explicit deny grant matched and overrode any allow (B1).
27    DeniedByRule(String),
28}
29
30impl AccessDecision {
31    /// Whether this decision permits the request.
32    ///
33    /// The only place a caller should collapse a decision to a boolean. Both
34    /// refusals answer `false` here, which is correct at an enforcement point
35    /// and lossy anywhere a human will read the result -- see the type's own
36    /// documentation for why the two are distinguished at all.
37    pub fn is_allowed(&self) -> bool {
38        matches!(self, AccessDecision::Allow)
39    }
40
41    /// Whether this refusal came from an explicit deny rule rather than from
42    /// the absence of a grant.
43    pub fn is_denied_by_rule(&self) -> bool {
44        matches!(self, AccessDecision::DeniedByRule(_))
45    }
46
47    /// The stable machine-readable reason code carried on the wire
48    /// (`allowed` | `no_grant` | `denied_by_rule`) — SDK contract §11.
49    pub const fn reason_code(&self) -> &'static str {
50        match self {
51            Self::Allow => "allowed",
52            Self::Deny(_) => "no_grant",
53            Self::DeniedByRule(_) => "denied_by_rule",
54        }
55    }
56
57    /// The human-readable reason, empty for an allow.
58    pub fn reason(&self) -> &str {
59        match self {
60            Self::Allow => "",
61            Self::Deny(r) | Self::DeniedByRule(r) => r,
62        }
63    }
64}

66/// Input for an authorization check.
67#[derive(Debug, Clone)]
68pub struct AccessRequest {
69    /// The tenant the check is scoped to — the tenant being **acted upon**.
70    /// Resources, scopes and their hierarchy are resolved here, so a request
71    /// can never see another tenant's resources.
72    pub tenant_id: Uuid,
73    /// Where the subject's grants live, and whether they reach beyond
74    /// [`Self::tenant_id`].
75    ///
76    /// See [`SubjectScope`]. Defaults to [`SubjectScope::Tenant`], which is
77    /// every ordinary principal and is exactly the behaviour that existed
78    /// before organization scope.
79    pub subject_scope: SubjectScope,
80    /// Who is asking -- a user id, or a service account's id. The engine
81    /// resolves roles by subject id, directly and through group membership.
82    pub subject_id: Uuid,
83    /// What they want to do, as a `resource:verb` permission name
84    /// (`"users:create"`).
85    pub action: String,
86    /// What they want to do it to. The nil UUID is the "global" sentinel,
87    /// meaning a permission not attached to any particular resource; anything
88    /// else is a resource whose ancestors are walked for inherited grants.
89    pub resource_id: Uuid,
90    /// Optional scope for sub-resource granularity.
91    pub scope: Option<String>,
92}
93
94/// Whose grants answer for a subject, and how far they reach.
95///
96/// # Why this is an enum and not an `Option<Uuid>`
97///
98/// It was an `Option<Uuid>` first — "the tenant the subject's grants live in,
99/// when that is not the tenant being acted upon" — and that encoding is unsafe
100/// in a way that took a test to notice. Two ids that merely *differ* is not the
101/// same claim as "this principal operates organization-wide", but it reads like
102/// it: any caller that built a request for a subject in tenant A about tenant B
103/// got cross-tenant reach for free, because differing was the whole condition.
104/// A tenant user's ordinary global role — `admin`, say — then applied in every
105/// tenant in the deployment, which is the exact opposite of what a tenant is.
106///
107/// Naming the claim fixes that. [`Self::Organization`] is a statement about the
108/// *principal*, and a caller has to make it deliberately; there is no
109/// combination of ordinary values that produces it by accident.
110///
111/// # The invariant, and who keeps it
112///
113/// [`Self::Organization`] must only ever be constructed for a principal whose
114/// home tenant really is the organization's reserved scope. The REST extractor
115/// is the sole production producer, and it resolves the tenant record and
116/// checks `kind` before making the claim — never from request input, since a
117/// caller that could assert its own scope could assert anyone's grants as its
118/// own.
119#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default)]
120pub enum SubjectScope {
121    /// The subject's grants live in the tenant being acted upon, and reach no
122    /// further. Every ordinary user and service account.
123    #[default]
124    Tenant,
125    /// The subject is **organization-level**: its user row, roles and group
126    /// memberships live in the organization's reserved tenant, named here,
127    /// while the resource it is reaching for may be in any tenant of that
128    /// organization.
129    ///
130    /// Crossing that boundary narrows what carries: only **global** grants
131    /// apply. A resource-scoped assignment names a resource in the organization
132    /// tenant and says nothing about a same-named resource elsewhere;
133    /// honouring it across the boundary would be a silent escalation between
134    /// isolated tenants.
135    Organization {
136        /// The organization's reserved tenant, where the grants are.
137        tenant_id: Uuid,
138    },
139}
140
141impl AccessRequest {
142    /// The tenant whose role assignments answer for this subject.
143    pub fn assignment_tenant_id(&self) -> Uuid {
144        match self.subject_scope {
145            SubjectScope::Tenant => self.tenant_id,
146            SubjectScope::Organization { tenant_id } => tenant_id,
147        }
148    }
149
150    /// Whether the subject's grants are being read across a tenant boundary.
151    ///
152    /// True only for an organization-level principal acting on a tenant that is
153    /// not its own. When true, the engine keeps only global grants — the one
154    /// rule organization scope adds to RBAC.
155    ///
156    /// An organization-level principal acting on the organization tenant itself
157    /// is *not* crossing anything, and gets ordinary resource-scoped evaluation
158    /// there.
159    pub fn crosses_tenant_boundary(&self) -> bool {
160        self.assignment_tenant_id() != self.tenant_id
161    }
162}

168    fn sample_request(scope: Option<&str>) -> AccessRequest {
169        AccessRequest {
170            tenant_id: Uuid::new_v4(),
171            subject_scope: SubjectScope::Tenant,
172            subject_id: Uuid::new_v4(),
173            action: "read".to_string(),
174            resource_id: Uuid::new_v4(),
175            scope: scope.map(|s| s.to_string()),
176        }
177    }

179    // -----------------------------------------------------------------------
180    // `SubjectScope` — the type exists to make one dangerous claim unwritable
181    // by accident, so these assert that it is.
182    // -----------------------------------------------------------------------
183
184    #[test]
185    fn a_tenant_subject_never_crosses_a_boundary() {
186        // Whatever tenant is named, an ordinary principal's grants are read
187        // from the tenant being acted upon. There is no value it could carry
188        // that says otherwise — which is the whole reason this is an enum and
189        // not an `Option<Uuid>` whose two ids merely differ.
190        let req = sample_request(None);
191        assert!(!req.crosses_tenant_boundary());
192        assert_eq!(req.assignment_tenant_id(), req.tenant_id);
193    }
194
195    #[test]
196    fn an_organization_subject_reads_its_grants_from_the_organization_tenant() {
197        let org_tenant = Uuid::new_v4();
198        let req = AccessRequest {
199            subject_scope: SubjectScope::Organization {
200                tenant_id: org_tenant,
201            },
202            ..sample_request(None)
203        };
204        assert!(req.crosses_tenant_boundary());
205        assert_eq!(req.assignment_tenant_id(), org_tenant);
206    }
207
208    #[test]
209    fn an_organization_subject_in_its_own_tenant_is_not_crossing() {
210        // Otherwise every check an organization administrator makes inside the
211        // organization scope would silently drop resource-scoped grants —
212        // which is where it does most of its work.
213        let same = Uuid::new_v4();
214        let req = AccessRequest {
215            tenant_id: same,
216            subject_scope: SubjectScope::Organization { tenant_id: same },
217            ..sample_request(None)
218        };
219        assert!(!req.crosses_tenant_boundary());
220        assert_eq!(req.assignment_tenant_id(), same);
221    }

223    #[test]
224    fn the_default_scope_is_the_pre_existing_behaviour() {
225        // `Tenant` is what every request meant before organization scope, so a
226        // `Default` that meant anything else would change decisions for code
227        // that never opted in.
