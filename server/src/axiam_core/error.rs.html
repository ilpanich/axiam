1//! Error types for the AXIAM system.
2
3use thiserror::Error;
4
5/// Every variant carries only owned text, so an error can be cloned: one
6/// outcome handed to several waiters (the CRL re-sign shared by every request
7/// for the same CA's list, #616) is one value, not one per waiter.
8#[derive(Debug, Clone, Error)]
9pub enum AxiamError {
10    #[error("Entity not found: {entity} with id {id}")]
11    NotFound { entity: String, id: String },
12
13    #[error("Entity already exists: {entity}")]
14    AlreadyExists { entity: String },
15
16    /// The request is well-formed and authorized, but the resource is not in a
17    /// state that permits it — a precondition the caller can satisfy and retry.
18    ///
19    /// Distinct from [`Self::Validation`] (400: the *request* is wrong, and
20    /// resending it unchanged will always fail) and from
21    /// [`Self::AlreadyExists`] (409, but specifically a uniqueness violation).
22    /// Maps to `409 Conflict`, which `sdks/CONTRACT.md` §27.4 already binds to
23    /// `ConflictError` and marks non-retriable — correct here: the caller must
24    /// *do something else first*, not resend.
25    #[error("{reason}")]
26    Conflict { reason: String },
27
28    #[error("Authentication failed: {reason}")]
29    AuthenticationFailed { reason: String },
30
31    #[error("Authorization denied: {reason}")]
32    AuthorizationDenied {
33        reason: String,
34        /// The action being checked (e.g. `"users:create"`), when known at
35        /// the denial site. Surfaced to clients so SDKs can parse it from
36        /// the 403 response body (SDK-Q02).
37        action: Option<String>,
38        /// The resource id being checked, when known and not the
39        /// "global" nil-UUID sentinel. Surfaced to clients alongside
40        /// `action` (SDK-Q02).
41        resource_id: Option<String>,
42    },
43
44    #[error("Validation error: {message}")]
45    Validation { message: String },
46
47    #[error("Password policy violation: {message}")]
48    PasswordPolicy { message: String },
49
50    #[error("Database error: {0}")]
51    Database(String),
52
53    #[error("Certificate error: {0}")]
54    Certificate(String),
55
56    #[error("Cryptography error: {0}")]
57    Crypto(String),
58
59    #[error("Email delivery failed: {0}")]
60    EmailDelivery(String),
61
62    #[error("Email configuration error: {0}")]
63    EmailConfig(String),
64
65    #[error("Webhook delivery failed: {0}")]
66    WebhookDelivery(String),
67
68    #[error("Tenant context missing or invalid")]
69    TenantContext,
70
71    #[error("Rate limit exceeded")]
72    RateLimited,
73
74    #[error("Service unavailable: {0}")]
75    ServiceUnavailable(String),
76
77    /// The tenant requires OPAQUE, so password login is refused
78    /// (`OpaqueMode::Required`).
79    ///
80    /// Its own variant rather than an `AuthenticationFailed` because it is the
81    /// opposite kind of answer: nothing about the caller's credentials was
82    /// wrong, they used the wrong *protocol*. A client that cannot distinguish
83    /// the two would show "invalid username or password" to a user whose
84    /// password is perfectly good, and an SDK could not auto-switch to the
85    /// OPAQUE endpoints. It reveals only a tenant-level policy, never anything
86    /// about which accounts exist — see `models::opaque::OpaqueMode::Required`.
87    #[error("this tenant requires OPAQUE authentication")]
88    OpaqueRequired,
89
90    /// A self-service action would take the caller's own account below the MFA
91    /// floor their tenant enforces (`MfaPolicy::mfa_enforced`, T-267).
92    ///
93    /// Its own variant rather than an [`Self::AuthorizationDenied`] for the
94    /// reason [`Self::OpaqueRequired`] above is its own: nothing about the
95    /// caller's permissions is wrong. They hold every permission the action
96    /// needs — it is their own account — and the refusal is a *policy* their
97    /// administrator set, which only their administrator can act against. A
98    /// client that could not tell the two apart would show "forbidden" where
99    /// the true sentence is "your organization requires MFA; ask an
100    /// administrator to reset it", and `authorization_denied` carries an
101    /// `action`/`resource_id` pair that would be meaningless here.
102    ///
103    /// It reveals only a tenant-level policy the caller is already subject to
104    /// and can observe from their own login flow, and nothing about any other
105    /// account.
106    #[error(
107        "your tenant requires multi-factor authentication; an administrator must reset it for you"
108    )]
109    MfaEnforced,
110
111    /// A write that lost an optimistic-concurrency race in the datastore and
112    /// stayed lost after every retry `retry_on_write_conflict` was willing to
113    /// spend on it (T-262).
114    ///
115    /// Maps to `503 Service Unavailable` with `Retry-After: 1`, and to gRPC
116    /// `UNAVAILABLE`. It is a statement about the **server** — come back in a
117    /// moment — which is what makes `503` right and the two nearby answers
118    /// wrong:
119    ///
120    /// - `500`, which is what this was, tells a client to stop when the
121    ///   correct advice is to retry. An IdP driving SCIM provisioning (Okta,
122    ///   Entra) reads it as a failed sync and re-sends the whole record.
123    /// - `409`, which [`Self::Conflict`] and [`Self::AlreadyExists`] keep,
124    ///   means "your request conflicts with the resource's state" — SCIM
125    ///   RFC 7644 §3.12 — and a caller who believes that will change the
126    ///   request rather than resend it, which cannot help here.
127    ///
128    /// **It carries no payload.** The engine's own words ("Transaction write
129    /// conflict…") belong in the log, where `DbError::Conflict` keeps them,
130    /// and nowhere near a response body.
131    #[error("the datastore is busy; retry this request")]
132    WriteContention,
133
134    /// A refusal of a network route (segment, reply profile, policy, or a
135    /// network role write), typed so the REST layer maps it to its `409`
136    /// `reason_code` without reading the sentence back.
137    ///
138    /// `Display` is the exact sentence the wire carries; `409` like
139    /// [`Self::Conflict`] and [`Self::AlreadyExists`].
140    #[error("{0}")]
141    Network(NetworkConflict),
142
143    #[error("SAML assertion replay detected")]
144    ReplayDetected,
145
146    #[error("Internal error: {0}")]
147    Internal(String),
148}

150pub type AxiamResult<T> = Result<T, AxiamError>;
151
152/// Why a network write was refused with `409`.
153///
154/// Raised at the repository boundary, from the markers the write transactions
155/// `THROW` and the checks the repositories make themselves; the REST layer maps
156/// each variant to its `reason_code` in one place. `Display` is the sentence
157/// the response carries.
158#[derive(Debug, Clone, PartialEq, Eq)]
159pub enum NetworkConflict {
160    /// The segment's version is not the one the writer read.
161    StaleSegment,
162    /// The reply profile's version is not the one the writer read.
163    StaleProfile,
164    /// The tenant network policy's version is not the one the writer read.
165    StalePolicy,
166    /// A reply profile was written on the segment between the gate's read and
167    /// the write.
168    PlacementRaced,
169    /// The role changed between the gate's read and the write; `what_landed`
170    /// says what.
171    StaleRole { what_landed: String },
172    /// The new `filter_ids` would leave the segment's reply profile outside its
173    /// envelope. `segment_id` and `filter_id` name the profile and the
174    /// `filter_id` it uses, once the caller has looked them up.
175    StrandsProfile {
176        segment_id: Option<uuid::Uuid>,
177        filter_id: Option<String>,
178    },
179    /// The segment still has child resources.
180    HasChildren,
181    /// The segment's site was removed while the segment was being written.
182    SiteGone,
183    /// The site already has a segment with this `vlan_id`.
184    DuplicateVlan,
185    /// The site already has a segment with this rank.
186    DuplicateRank,
187    /// The site already has a segment of this name.
188    DuplicateName,
189    /// The segment already has a reply profile.
190    ProfileExists,
191}
192
193impl NetworkConflict {
194    /// `true` for a uniqueness clash, which answers the `already_exists` slug;
195    /// every other variant answers `conflict`.
196    #[must_use]
197    pub fn is_uniqueness(&self) -> bool {
198        matches!(
199            self,
200            Self::DuplicateVlan | Self::DuplicateRank | Self::DuplicateName | Self::ProfileExists
201        )
202    }
203
204    /// `true` when the write lost to another one: read again and retry.
205    #[must_use]
206    pub fn is_stale(&self) -> bool {
207        matches!(
208            self,
209            Self::StaleSegment
210                | Self::StaleProfile
211                | Self::StalePolicy
212                | Self::PlacementRaced
213                | Self::StaleRole { .. }
214        )
215    }
216}
217
218impl std::fmt::Display for NetworkConflict {
219    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
220        match self {
221            Self::StaleSegment => f.write_str(
222                "the network segment changed since it was read; read it again and retry",
223            ),
224            Self::StaleProfile => {
225                f.write_str("the reply profile changed since it was read; read it again and retry")
226            }
227            Self::StalePolicy => {
228                f.write_str("the network policy changed since it was read; read it again and retry")
229            }
230            Self::PlacementRaced => f.write_str(
231                "the network segment changed since it was read (a reply profile was written on \
232                 it); read it again and retry",
233            ),
234            Self::StaleRole { what_landed } => write!(
235                f,
236                "the role changed since it was read ({what_landed}); read it again and retry"
237            ),
238            Self::StrandsProfile {
239                segment_id: Some(segment_id),
240                filter_id: Some(filter_id),
241            } => write!(
242                f,
243                "the segment's reply profile {segment_id} uses filter_id '{filter_id}', which \
244                 the new filter_ids remove; change or delete the profile first"
245            ),
246            Self::StrandsProfile {
247                segment_id: Some(segment_id),
248                filter_id: None,
249            } => write!(
250                f,
251                "the new filter_ids would leave the segment's reply profile {segment_id} \
252                 outside its envelope; change or delete the profile first"
253            ),
254            Self::StrandsProfile { .. } => f.write_str(
255                "the new filter_ids would leave the segment's reply profile outside its \
256                 envelope; change or delete the profile first",
257            ),
258            Self::HasChildren => f.write_str("the network segment has child resources"),
259            Self::SiteGone => f.write_str(
260                "the network site was removed while the segment was being written; retry",
261            ),
262            Self::DuplicateVlan => f.write_str(
263                "Entity already exists: network_segment: the site already has a segment with \
264                 this vlan_id",
265            ),
266            Self::DuplicateRank => f.write_str(
267                "Entity already exists: network_segment: the site already has a segment with \
268                 this rank",
269            ),
270            Self::DuplicateName => f.write_str(
271                "Entity already exists: network_segment: the site already has a segment of \
272                 this name",
273            ),
274            Self::ProfileExists => f.write_str("Entity already exists: reply_profile"),
275        }
276    }
277}

283    fn every_conflict() -> Vec<NetworkConflict> {
284        vec![
285            NetworkConflict::StaleSegment,
286            NetworkConflict::StaleProfile,
287            NetworkConflict::StalePolicy,
288            NetworkConflict::PlacementRaced,
289            NetworkConflict::StaleRole {
290                what_landed: "a revoke".into(),

292            NetworkConflict::StrandsProfile {
293                segment_id: None,
294                filter_id: None,
295            },
296            NetworkConflict::HasChildren,
297            NetworkConflict::SiteGone,
298            NetworkConflict::DuplicateVlan,
299            NetworkConflict::DuplicateRank,

301            NetworkConflict::ProfileExists,

305    /// `AxiamError::Network` shows the conflict's own sentence, which is what
306    /// the `409` body carries.
307    #[test]
308    fn the_error_displays_the_conflicts_own_sentence() {
309        for conflict in every_conflict() {
310            assert_eq!(
311                AxiamError::Network(conflict.clone()).to_string(),
312                conflict.to_string()
313            );
314        }
315    }
316
317    #[test]
318    fn a_stale_write_asks_the_caller_to_read_again_and_retry() {
319        for (conflict, subject) in [
320            (NetworkConflict::StaleSegment, "the network segment"),
321            (NetworkConflict::StaleProfile, "the reply profile"),
322            (NetworkConflict::StalePolicy, "the network policy"),
323        ] {
324            let sentence = conflict.to_string();
325            assert!(sentence.starts_with(subject), "{sentence}");
326            assert!(sentence.ends_with("read it again and retry"), "{sentence}");
327        }
328        assert_eq!(
329            NetworkConflict::PlacementRaced.to_string(),
330            "the network segment changed since it was read (a reply profile was written on it); \
331             read it again and retry"
332        );
333        assert_eq!(
334            NetworkConflict::StaleRole {
335                what_landed: "the role was revoked".into()
336            }
337            .to_string(),
338            "the role changed since it was read (the role was revoked); read it again and retry"
339        );
340    }
341
342    /// The sentence names as much of the stranded profile as the caller knew.
343    #[test]
344    fn a_stranded_profile_is_named_as_far_as_it_is_known() {
345        let segment_id = uuid::Uuid::nil();
346        let full = NetworkConflict::StrandsProfile {
347            segment_id: Some(segment_id),
348            filter_id: Some("f-1".into()),
349        };
350        assert_eq!(
351            full.to_string(),
352            format!(
353                "the segment's reply profile {segment_id} uses filter_id 'f-1', which the new \
354                 filter_ids remove; change or delete the profile first"
355            )
356        );
357        let without_filter = NetworkConflict::StrandsProfile {
358            segment_id: Some(segment_id),
359            filter_id: None,
360        };
361        assert_eq!(
362            without_filter.to_string(),
363            format!(
364                "the new filter_ids would leave the segment's reply profile {segment_id} outside \
365                 its envelope; change or delete the profile first"
366            )
367        );
368        let unnamed = NetworkConflict::StrandsProfile {
369            segment_id: None,
370            filter_id: Some("f-1".into()),
371        };
372        assert_eq!(
373            unnamed.to_string(),
374            "the new filter_ids would leave the segment's reply profile outside its envelope; \
375             change or delete the profile first"
376        );
377    }

411    /// The two predicates partition the variants the REST layer maps: a
412    /// uniqueness clash is never stale, and neither predicate holds for the
413    /// structural refusals.
414    #[test]
415    fn uniqueness_and_staleness_classify_every_conflict() {
416        let classified: Vec<(bool, bool)> = every_conflict()
417            .iter()
418            .map(|c| (c.is_uniqueness(), c.is_stale()))
419            .collect();
420        assert_eq!(
421            classified,
422            vec![
423                (false, true),  // StaleSegment
424                (false, true),  // StaleProfile
425                (false, true),  // StalePolicy
426                (false, true),  // PlacementRaced
427                (false, true),  // StaleRole
428                (false, false), // StrandsProfile
429                (false, false), // HasChildren
430                (false, false), // SiteGone
431                (true, false),  // DuplicateVlan
432                (true, false),  // DuplicateRank
433                (true, false),  // DuplicateName
434                (true, false),  // ProfileExists
435            ]
436        );
437    }
