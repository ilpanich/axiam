1//! Which transport peers may speak for the client in `X-Forwarded-For`
2//! (T-212, R110-01, #711, #686).
3//!
4//! Every per-address rate limit keys a request on the rightmost untrusted
5//! `X-Forwarded-For` hop (`AXIAM__RATE_LIMIT__TRUSTED_HOPS`). That is right
6//! behind an appending proxy, and wrong on a connection that reaches the
7//! listener with no proxy in front: there nothing appends to the header, the
8//! rightmost entry is whatever the caller wrote, and a caller who rotates it
9//! gets a fresh bucket per request.
10//!
11//! [`TrustedProxies`] (`AXIAM__RATE_LIMIT__TRUSTED_PROXIES`, a comma-separated
12//! list of CIDR blocks) names the peers whose header is read at all. A request
13//! from a listed peer is keyed exactly as before, under `TRUSTED_HOPS`; a
14//! request from any other peer is keyed on the peer, whatever it wrote. With
15//! the list empty — the default — every peer's header is read, which is what
16//! the server did before the list existed: the shipped topologies all put an
17//! appending proxy in front, and one of them (`k8s/`) re-encrypts to the
18//! server's native TLS listener, so "native TLS on" does not mean "no proxy".
19//! The boot `WARN` of [`TrustedProxies::boot_warning`] is the empty list's
20//! other half.
21//!
22//! The REST and gRPC extractors are siblings that may not depend on each
23//! other, so the variable and the rule live here, below both, and both read
24//! [`TrustedProxies::configured`] — one parse per process.
25
26use std::net::IpAddr;
27use std::sync::{Arc, OnceLock};
28
29use crate::ip_class::{IpNetwork, IpNetworkError};
30
31/// The environment variable that holds the list.
32pub const TRUSTED_PROXIES_VAR: &str = "AXIAM__RATE_LIMIT__TRUSTED_PROXIES";
33
34/// The peers whose `X-Forwarded-For` is honoured; empty means every peer's.
35#[derive(Debug, Clone, Default, PartialEq, Eq)]
36pub struct TrustedProxies(Arc<[IpNetwork]>);
37
38/// An entry of the list that is not a CIDR block or an address.
39#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
40#[error("{TRUSTED_PROXIES_VAR}: entry `{entry}` is not a CIDR block or an IP address: {reason}")]
41pub struct TrustedProxiesError {
42    /// The entry as written.
43    pub entry: String,
44    /// Why it did not parse.
45    pub reason: IpNetworkError,
46}
47
48impl TrustedProxies {
49    /// The list `raw` names: CIDR blocks or bare addresses (a host block),
50    /// separated by commas; blank entries are skipped.
51    ///
52    /// # Errors
53    ///
54    /// The first entry that does not parse, host bits below the prefix
55    /// included (`10.0.0.1/8` is refused rather than widened).
56    pub fn parse(raw: &str) -> Result<Self, TrustedProxiesError> {
57        raw.split(',')
58            .map(str::trim)
59            .filter(|entry| !entry.is_empty())
60            .map(|entry| {
61                entry
62                    .parse::<IpNetwork>()
63                    .map_err(|reason| TrustedProxiesError {
64                        entry: entry.to_owned(),
65                        reason,
66                    })
67            })
68            .collect::<Result<Vec<_>, _>>()
69            .map(|networks| Self(networks.into()))
70    }
71
72    /// The list `AXIAM__RATE_LIMIT__TRUSTED_PROXIES` names (empty when unset).
73    ///
74    /// # Errors
75    ///
76    /// As [`TrustedProxies::parse`]. The composition root calls this once at
77    /// boot and refuses to start on an error.
78    pub fn from_env() -> Result<Self, TrustedProxiesError> {
79        std::env::var(TRUSTED_PROXIES_VAR)
80            .map_or_else(|_| Ok(Self::default()), |raw| Self::parse(&raw))
81    }
82
83    /// The process-wide list every extractor reads, parsed once.
84    ///
85    /// A value that does not parse is read as the empty list here: the
86    /// composition root has already refused to start on it
87    /// ([`TrustedProxies::from_env`]), so this arm is reached only by an
88    /// embedding that skipped that check.
89    #[must_use]
90    pub fn configured() -> Self {
91        static CONFIGURED: OnceLock<TrustedProxies> = OnceLock::new();
92        CONFIGURED
93            .get_or_init(|| Self::from_env().unwrap_or_default())
94            .clone()
95    }
96
97    /// Whether no peer is named, i.e. every peer's header is read.
98    #[must_use]
99    pub fn is_empty(&self) -> bool {
100        self.0.is_empty()
101    }
102
103    /// The blocks, in the order written.
104    #[must_use]
105    pub fn networks(&self) -> &[IpNetwork] {
106        &self.0
107    }
108
109    /// Whether a request from `peer` may name its client in `X-Forwarded-For`.
110    ///
111    /// Always with the list empty; otherwise only when `peer` is inside one of
112    /// the blocks. A request with no known peer is never honoured under a
113    /// list: there is nothing to check it against.
114    #[must_use]
115    pub fn honours_forwarded_for(&self, peer: Option<IpAddr>) -> bool {
116        self.is_empty() || peer.is_some_and(|peer| self.0.iter().any(|block| block.contains(peer)))
117    }
118
119    /// The boot `WARN` for this list on a server whose native TLS listener is
120    /// `native_tls`, or `None` when there is nothing to warn about.
121    ///
122    /// The case is the empty list on a native TLS listener: a client that must
123    /// reach that listener directly (the FreeRADIUS recipe's mutual TLS ends
124    /// at AXIAM) writes the header nothing appends to, and so chooses its own
125    /// rate-limit key on every route — while the proxy that re-encrypts to the
126    /// same listener (`k8s/`) needs its header read. Only the operator knows
127    /// which peers are which; the list is how they say so.
128    #[must_use]
129    pub fn boot_warning(&self, native_tls: bool) -> Option<&'static str> {
130        (native_tls && self.is_empty()).then_some(
131            "native TLS is on and AXIAM__RATE_LIMIT__TRUSTED_PROXIES is empty: \
132             X-Forwarded-For is read from every peer, so a client that reaches this \
133             listener with no appending proxy in front chooses its own rate-limit key \
134             on every per-address limiter (T-212). List the proxies in front of the \
135             server (CIDR blocks); every other peer is then keyed on its own address. \
136             Both listeners read the list. See docs/deployment/README.md, \
137             \"Deriving TRUSTED_HOPS\"",
138        )
139    }
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    fn ip(literal: &str) -> IpAddr {
147        literal.parse().expect("literal parses")
148    }
149
150    #[test]
151    fn parses_blocks_and_bare_addresses_and_skips_blank_entries() {
152        let list = TrustedProxies::parse(" 10.0.0.0/8, ,192.0.2.7,fd00::/8 ,").expect("parses");
153        let written: Vec<String> = list.networks().iter().map(ToString::to_string).collect();
154        assert_eq!(written, ["10.0.0.0/8", "192.0.2.7/32", "fd00::/8"]);
155    }
156
157    #[test]
158    fn an_unparseable_entry_is_refused_by_name() {
159        for (raw, entry) in [
160            ("10.0.0.0/8,proxy.internal", "proxy.internal"),
161            ("10.0.0.1/8", "10.0.0.1/8"),
162            ("10.0.0.0/33", "10.0.0.0/33"),
163        ] {
164            let error = TrustedProxies::parse(raw).expect_err(raw);
165            assert_eq!(error.entry, entry, "{raw}");
166            assert!(error.to_string().contains(TRUSTED_PROXIES_VAR), "{error}");
167        }
168    }
169
170    #[test]
171    fn the_empty_list_honours_every_peer() {
172        let list = TrustedProxies::parse("").expect("parses");
173        assert!(list.is_empty());
174        assert!(list.honours_forwarded_for(Some(ip("203.0.113.9"))));
175        assert!(list.honours_forwarded_for(None));
176    }
177
178    #[test]
179    fn a_list_honours_only_the_peers_inside_it() {
180        let list = TrustedProxies::parse("10.42.0.0/16,2001:db8::1").expect("parses");
181        assert!(list.honours_forwarded_for(Some(ip("10.42.7.1"))));
182        assert!(list.honours_forwarded_for(Some(ip("::ffff:10.42.7.1"))));
183        assert!(list.honours_forwarded_for(Some(ip("2001:db8::1"))));
184        assert!(!list.honours_forwarded_for(Some(ip("10.43.0.1"))));
185        assert!(!list.honours_forwarded_for(Some(ip("2001:db8::2"))));
186        assert!(!list.honours_forwarded_for(None));
187    }
188
189    #[test]
190    fn the_boot_warning_is_for_the_empty_list_on_native_tls_only() {
191        let empty = TrustedProxies::default();
192        let listed = TrustedProxies::parse("10.0.0.0/8").expect("parses");
193        assert!(empty.boot_warning(true).is_some());
194        assert!(empty.boot_warning(false).is_none());
195        assert!(listed.boot_warning(true).is_none());
196        assert!(listed.boot_warning(false).is_none());
197    }
198
199    /// The variable is unset in the test environment, which reads as the empty
200    /// list (every peer's header is honoured); the process-wide list agrees and
201    /// is parsed once, so two reads are the same list.
202    #[test]
203    fn an_unset_variable_reads_as_the_empty_list_and_the_configured_list_agrees() {
204        if std::env::var_os(TRUSTED_PROXIES_VAR).is_some() {
205            return; // the environment names a list; the cases above cover parsing
206        }
207        assert!(
208            TrustedProxies::from_env()
209                .expect("unset is not an error")
210                .is_empty()
211        );
212        let configured = TrustedProxies::configured();
213        assert!(configured.is_empty());
214        assert_eq!(configured, TrustedProxies::configured());
215        assert!(
216            configured.honours_forwarded_for(None),
217            "no list: every request is honoured"
218        );
219    }
