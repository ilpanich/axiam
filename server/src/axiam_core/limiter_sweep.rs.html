1//! The registry of in-process rate-limiter maps that need sweeping (#673,
2//! R110W2-05).
3//!
4//! A keyed `governor` limiter keeps one cell per key it has seen, and neither
5//! `actix-governor` nor `tower_governor` ever evicts one. The key is an
6//! address (which an IPv6 sender rotates at will) or, on the five federation
7//! routes, a verified principal, so every map grows with the number of distinct
8//! keys seen since start. A cell replenishes within a minute, after which it
9//! carries no information a fresh cell would not.
10//!
11//! Each limiter's owner registers one closure here when it builds the limiter;
12//! one interval task calls [`sweep_all`]. The REST and gRPC crates are siblings
13//! that may not depend on each other, so the registry lives below both, and it
14//! holds closures rather than limiters so that this crate needs no `governor`.
15//!
16//! A closure holds its limiter by `Weak`, so the registry never keeps a map
17//! alive: when the server (or a test's app) that built it is dropped, the
18//! closure reports itself dead and the next sweep forgets it.
19
20use std::sync::Mutex;
21
22/// What one registered sweeper reports after it ran.
23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24pub enum SweepOutcome {
25    /// The limiter is gone; forget this sweeper.
26    Dropped,
27    /// The limiter was swept and holds this many cells afterwards.
28    Swept {
29        /// Cells that remain in the map.
30        remaining: usize,
31    },
32}
33
34/// What a [`sweep_all`] pass found.
35#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
36pub struct SweepReport {
37    /// Limiters still alive.
38    pub limiters: usize,
39    /// Cells still held across all of them.
40    pub cells: usize,
41}
42
43type Sweeper = Box<dyn Fn() -> SweepOutcome + Send + Sync>;
44
45static REGISTRY: Mutex<Vec<Sweeper>> = Mutex::new(Vec::new());
46
47fn registry() -> std::sync::MutexGuard<'static, Vec<Sweeper>> {
48    // A sweeper that panicked poisons nothing worth refusing the rest for.
49    REGISTRY.lock().unwrap_or_else(|e| e.into_inner())
50}
51
52/// Register one limiter's sweeper. The closure must hold the limiter weakly and
53/// return [`SweepOutcome::Dropped`] once it can no longer be upgraded.
54pub fn register(sweeper: impl Fn() -> SweepOutcome + Send + Sync + 'static) {
55    registry().push(Box::new(sweeper));
56}
57
58/// Sweep every registered limiter once and forget the dead ones.
59pub fn sweep_all() -> SweepReport {
60    let mut report = SweepReport::default();
61    registry().retain(|sweep| match sweep() {
62        SweepOutcome::Dropped => false,
63        SweepOutcome::Swept { remaining } => {
64            report.limiters += 1;
65            report.cells += remaining;
66            true
67        }
68    });
69    report
70}

72#[cfg(test)]
73mod tests {
74    use super::*;
75    use std::sync::{Arc, Weak};
76
77    #[test]
78    fn a_dead_sweeper_is_forgotten_and_a_live_one_is_counted() {
79        let live = Arc::new(7usize);
80        let dead = Arc::new(3usize);
81        for owner in [&live, &dead] {
82            let weak: Weak<usize> = Arc::downgrade(owner);
83            register(move || match weak.upgrade() {
84                Some(n) => SweepOutcome::Swept { remaining: *n },
85                None => SweepOutcome::Dropped,
86            });
87        }
88        drop(dead);
89        let first = sweep_all();
90        assert!(first.cells >= 7 && first.limiters >= 1);
91        // The dead one is gone: a second pass reports the same figures.
92        assert_eq!(sweep_all(), first);
93    }
94}
