49/// The action namespaces the server reserves for its own **control records**
50/// (R101W3-03, #637).
51///
52/// Rows in these namespaces are evidence the server writes and later reads
53/// back to decide what it may do: the `tenants.audit_exported` receipt that
54/// authorises a tenant's deletion (T-118), the `tenants.deleted` record whose
55/// export time bounds what the tenant purge deletes (T-472), `tenants.purged`,
56/// and the GDPR records (`gdpr.*`). Only the server writes them. The AMQP audit
57/// ingestion (`axiam_amqp::audit_consumer::process_audit_event`), whose
58/// messages a tenant's own producers sign, refuses any action in them; every
59/// action constant of a control record must fall in one of them (tests in
60/// `axiam-amqp` and `axiam-db` pin the ones the purge and the deletion read).
61pub const RESERVED_ACTION_NAMESPACES: &[&str] = &["tenants.", "gdpr."];
62
63/// The namespace of [`RESERVED_ACTION_NAMESPACES`] that `action` falls in, or
64/// `None` when the action is open to external producers.
65///
66/// The comparison ignores ASCII case, so `Tenants.audit_exported` is reserved
67/// as `tenants.audit_exported` is, although the server's own readers match the
68/// exact spelling.
69pub fn reserved_action_namespace(action: &str) -> Option<&'static str> {
70    RESERVED_ACTION_NAMESPACES.iter().copied().find(|ns| {
71        action
72            .get(..ns.len())
73            .is_some_and(|head| head.eq_ignore_ascii_case(ns))
74    })
75}

77#[cfg(test)]
78mod tests {
79    use super::*;
80
81    #[test]
82    fn the_control_namespaces_are_reserved_whatever_their_case() {
83        for action in [
84            "tenants.audit_exported",
85            "tenants.deleted",
86            "tenants.purged",
87            "TENANTS.Audit_Exported",
88            "gdpr.user_pseudonymized",
89            "Gdpr.erasure_requested",
90        ] {
91            assert!(reserved_action_namespace(action).is_some(), "{action}");
92        }
93        for action in [
94            "user.login",
95            "tenant.audit_exported",
96            "tenantsx.deleted",
97            "gdpr",
98            "",
99            "DELETE /api/v1/organizations/x/tenants/y",
100            "é",
101        ] {
102            assert_eq!(reserved_action_namespace(action), None, "{action}");
103        }
104    }
