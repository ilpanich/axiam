38/// A CA (Certificate Authority) certificate at the organization level.
39///
40/// CA certificates are the root of trust for all tenant certificates
41/// within the organization. Private keys for signing CAs are encrypted
42/// with AES-256-GCM and stored separately; non-signing CAs only store
43/// the public certificate.
44#[derive(Clone, Serialize, Deserialize, utoipa::ToSchema)]
45pub struct CaCertificate {
46    pub id: Uuid,
47    /// The organization this CA belongs to.
48    pub organization_id: Uuid,
49    /// The certificate subject (e.g., `CN=ACME Corp Root CA`).
50    pub subject: String,
51    /// PEM-encoded public certificate.
52    ///
53    /// The certificate that *signs*, which under `vault_pki` custody is the
54    /// intermediate rather than the root beneath which it was created.
55    pub public_cert_pem: String,
56    /// The issuers above [`Self::public_cert_pem`], concatenated PEM, nearest
57    /// issuer first and the root last.
58    ///
59    /// `None` for a CA that is its own root, which is every CA AXIAM generated
60    /// before Vault's PKI engine was an option. Present for a `vault_pki` CA,
61    /// where it is the only copy of the root certificate anything outside Vault
62    /// will ever see — a relying party cannot validate an AXIAM-issued leaf
63    /// without it, and `root/generate/internal` returns it exactly once.
64    #[serde(default, skip_serializing_if = "Option::is_none")]
65    pub chain_pem: Option<String>,
66    /// SHA-256 fingerprint of the certificate.
67    pub fingerprint: String,
68    pub key_algorithm: KeyAlgorithm,
69    /// Validity start.
70    pub not_before: DateTime<Utc>,
71    /// Validity end.
72    pub not_after: DateTime<Utc>,
73    pub status: CertificateStatus,
74    /// AES-256-GCM encrypted private key — only for CAs under
75    /// [`CaKeyCustody::Database`]. `None` under every other custodian, and for
76    /// an imported CA whose key AXIAM never held.
77    #[serde(skip_serializing)]
78    #[schema(read_only)]
79    pub encrypted_private_key: Option<Vec<u8>>,
80    /// Which custodian holds this CA's signing key.
81    ///
82    /// Recorded per CA rather than read from configuration, so adopting a new
83    /// custodian does not strand the CAs that already exist. Not secret — an
84    /// operator needs to see it, and it discloses only where a key is kept.
85    #[serde(default = "default_key_custody")]
86    #[schema(value_type = String, example = "database")]
87    pub key_custody: CaKeyCustody,
88    /// Where the custodian put the key. A Vault path under its mount; `None`
89    /// for database custody, whose locator is the row itself.
90    pub key_locator: Option<String>,
91    pub created_at: DateTime<Utc>,
92}
93
94/// What a CA row with no recorded custodian means.
95///
96/// Every row written before custody was a concept holds its key sealed into
97/// itself, so `Database` is the only reading that finds those keys. A row that
98/// genuinely has no key is `External`, and the v45 migration distinguishes the
99/// two by whether `encrypted_private_key` is present.
100fn default_key_custody() -> CaKeyCustody {
101    CaKeyCustody::Database
102}

104/// Manual `Debug` impl (SECHRD-09 / D-06): `#[serde(skip_serializing)]` only
105/// affects `Serialize`, not `{:?}` — this closes that residual leak by
106/// redacting `encrypted_private_key` while keeping other fields readable.
107impl std::fmt::Debug for CaCertificate {
108    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
109        f.debug_struct("CaCertificate")
110            .field("id", &self.id)
111            .field("organization_id", &self.organization_id)
112            .field("subject", &self.subject)
113            .field("public_cert_pem", &self.public_cert_pem)
114            .field("chain_pem", &self.chain_pem)
115            .field("fingerprint", &self.fingerprint)
116            .field("key_algorithm", &self.key_algorithm)
117            .field("not_before", &self.not_before)
118            .field("not_after", &self.not_after)
119            .field("status", &self.status)
120            .field(
121                "encrypted_private_key",
122                &self.encrypted_private_key.as_ref().map(|_| "[REDACTED]"),
123            )
124            .field("key_custody", &self.key_custody)
125            .field("key_locator", &self.key_locator)
126            .field("created_at", &self.created_at)
127            .finish()
128    }
129}

131/// Fields required to generate a new CA certificate (user-facing DTO).
132#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
133pub struct CreateCaCertificate {
134    #[serde(default)]
135    pub organization_id: Uuid,
136    pub subject: String,
137    pub key_algorithm: KeyAlgorithm,
138    /// Validity duration in days.
139    pub validity_days: u32,
140    /// Common name for the signing intermediate, under `vault_pki` custody.
141    ///
142    /// Ignored by every other custodian, which generates one self-signed CA and
143    /// has no second certificate to name. Defaults to the root's subject with
144    /// `Intermediate Authority` appended, which is what HashiCorp's own PKI
145    /// walkthrough calls it.
146    #[serde(default, skip_serializing_if = "Option::is_none")]
147    pub intermediate_subject: Option<String>,
148    /// Validity of the signing intermediate, in days. Defaults to the root's.
149    ///
150    /// Vault caps it to the root's own expiry regardless, so asking for more
151    /// than the root has is not an error — it is quietly shortened, and the
152    /// stored record describes what came back rather than what was asked for.
153    #[serde(default, skip_serializing_if = "Option::is_none")]
154    pub intermediate_validity_days: Option<u32>,
155    /// Issue directly from the generated root instead of creating an
156    /// intermediate. `vault_pki` custody only.
157    ///
158    /// Off by default, and the default is the safer one: a root that signs only
159    /// one intermediate can have that intermediate revoked and replaced without
160    /// redistributing the trust anchor, while a root that signs leaves directly
161    /// cannot be rotated without touching every relying party.
162    #[serde(default)]
163    pub issue_from_root: bool,
164}

166/// All fields required to store a CA certificate in the database.
167///
168/// Produced by the PKI service after generating the keypair and cert.
169#[derive(Debug, Clone)]
170pub struct StoreCaCertificate {
171    /// Chosen by the service, not the repository.
172    ///
173    /// A custodian outside the database addresses the key *by CA id*, so the id
174    /// has to exist before the key is stored — and the key has to be stored
175    /// before the row, or a failed write leaves a CA certificate whose key is
176    /// nowhere. Letting the repository mint the id made that ordering
177    /// impossible to express.
178    pub id: Uuid,
179    pub organization_id: Uuid,
180    pub subject: String,
181    pub public_cert_pem: String,
182    /// The issuers above `public_cert_pem`, concatenated PEM. See
183    /// [`CaCertificate::chain_pem`].
184    pub chain_pem: Option<String>,
185    pub fingerprint: String,
186    pub key_algorithm: KeyAlgorithm,
187    pub not_before: DateTime<Utc>,
188    pub not_after: DateTime<Utc>,
189    pub encrypted_private_key: Option<Vec<u8>>,
190    pub key_custody: CaKeyCustody,
191    pub key_locator: Option<String>,
192}

194/// Import an externally-generated CA (BYOK).
195///
196/// The other way a CA comes to exist. Generation makes a key AXIAM chose;
197/// this takes one an organization already has — from an offline root, an
198/// existing internal PKI, or an HSM ceremony — and puts AXIAM in the chain
199/// rather than at the top of it.
200///
201/// `private_key_pem` is optional, and the two cases are genuinely different
202/// rather than one being a degraded form of the other:
203///
204/// * **With a key**, AXIAM takes custody of it (Vault, or sealed into the row)
205///   and can issue certificates against this CA.
206/// * **Without one**, the certificate is a trust anchor and nothing more.
207///   `/api/v1/certificates` cannot sign against it, and says so rather than
208///   failing at the point of use.
209#[derive(Clone, Serialize, Deserialize, utoipa::ToSchema)]
210pub struct ImportCaCertificate {
211    #[serde(default)]
212    pub organization_id: Uuid,
213    /// PEM-encoded CA certificate.
214    pub public_cert_pem: String,
215    /// PEM-encoded private key, if AXIAM is to hold it.
216    ///
217    /// Write-only: it goes to the configured custodian on the way in and is
218    /// never returned by any endpoint.
219    #[serde(default, skip_serializing)]
220    pub private_key_pem: Option<String>,
221}

226impl std::fmt::Debug for ImportCaCertificate {

234            )
235            .finish()
236    }

239/// Response returned when a CA certificate is generated.
240///
241/// Includes the private key PEM, which is returned **once** and never
242/// stored or retrievable again — when the custodian produced one at all. Under
243/// `vault_pki` custody the key was born inside Vault and there is nothing to
244/// return, which is the point of that custodian rather than a shortcoming of
245/// this response.
246#[derive(Clone, Serialize, Deserialize, utoipa::ToSchema)]
247pub struct GeneratedCaCertificate {
248    #[serde(flatten)]
249    pub certificate: CaCertificate,
250    /// PEM-encoded private key — returned only on generation, and only when
251    /// there is one to return.
252    ///
253    /// Absent under `vault_pki` custody, where the key was generated inside
254    /// Vault and no API exports it. The field is omitted rather than sent as
255    /// `null` so a client that has always read it keeps working unchanged for
256    /// every custodian that does produce a key.
257    #[serde(default, skip_serializing_if = "Option::is_none")]
258    pub private_key_pem: Option<String>,
259}
260
261/// Manual `Debug` impl (SECHRD-09 / D-06): redacts `private_key_pem` (raw
262/// key material returned only once on generation); delegates to
263/// `CaCertificate`'s own redacting `Debug` for the nested `certificate` field.
264impl std::fmt::Debug for GeneratedCaCertificate {
265    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
266        f.debug_struct("GeneratedCaCertificate")
267            .field("certificate", &self.certificate)
268            // `Option`-aware rather than an unconditional `[REDACTED]`: a
269            // `vault_pki` CA has no key, and printing the redaction marker for
270            // one would say a key was withheld when none exists.
271            .field(
272                "private_key_pem",
273                &self.private_key_pem.as_ref().map(|_| "[REDACTED]"),
274            )
275            .finish()
276    }
277}

279/// A tenant-level certificate for users, services, or IoT devices.
280///
281/// Certificates are signed by the organization's CA. The private key is
282/// returned once on generation and never stored by AXIAM.
283#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
284pub struct Certificate {
285    pub id: Uuid,
286    /// The tenant this certificate belongs to.
287    pub tenant_id: Uuid,
288    /// The CA certificate that signed this certificate.
289    pub issuer_ca_id: Uuid,
290    /// The certificate subject (e.g., `CN=device-001`).
291    pub subject: String,
292    /// PEM-encoded public certificate.
293    pub public_cert_pem: String,
294    /// SHA-256 fingerprint of the certificate.
295    pub fingerprint: String,
296    pub cert_type: CertificateType,
297    pub key_algorithm: KeyAlgorithm,
298    /// Validity start.
299    pub not_before: DateTime<Utc>,
300    /// Validity end.
301    pub not_after: DateTime<Utc>,
302    pub status: CertificateStatus,
303    /// Arbitrary key-value metadata (e.g., device serial, user ID binding).
304    pub metadata: serde_json::Value,
305    pub created_at: DateTime<Utc>,
306}
