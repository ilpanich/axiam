38/// A CA (Certificate Authority) certificate at the organization level.
39///
40/// CA certificates are the root of trust for all tenant certificates
41/// within the organization. Private keys for signing CAs are encrypted
42/// with AES-256-GCM and stored separately; non-signing CAs only store
43/// the public certificate.
44#[derive(Clone, Serialize, Deserialize, utoipa::ToSchema)]
45pub struct CaCertificate {
46    pub id: Uuid,
47    /// The organization this CA belongs to.
48    pub organization_id: Uuid,
49    /// The tenant this CA signs for, when it is a tenant signing CA.
50    ///
51    /// `None` for an organization-level CA — the trust anchor, and the only
52    /// kind that existed before tenant signing CAs. `Some` for an intermediate
53    /// created under one, which exists so a tenant's user, service and device
54    /// certificates chain through a CA that can be revoked and replaced
55    /// without touching the anchor the rest of the estate trusts.
56    #[serde(default, skip_serializing_if = "Option::is_none")]
57    pub tenant_id: Option<Uuid>,
58    /// The CA in this organization that signed this one.
59    ///
60    /// `None` for an organization-level CA, which is either self-signed or
61    /// imported and has no parent inside AXIAM.
62    #[serde(default, skip_serializing_if = "Option::is_none")]
63    pub parent_ca_id: Option<Uuid>,
64    /// The certificate subject (e.g., `CN=ACME Corp Root CA`).
65    pub subject: String,
66    /// PEM-encoded public certificate.
67    ///
68    /// The certificate that *signs*, which under `vault_pki` custody is the
69    /// intermediate rather than the root beneath which it was created.
70    pub public_cert_pem: String,
71    /// The issuers above [`Self::public_cert_pem`], concatenated PEM, nearest
72    /// issuer first and the root last.
73    ///
74    /// `None` for a CA that is its own root, which is every CA AXIAM generated
75    /// before Vault's PKI engine was an option. Present for a `vault_pki` CA,
76    /// where it is the only copy of the root certificate anything outside Vault
77    /// will ever see — a relying party cannot validate an AXIAM-issued leaf
78    /// without it, and `root/generate/internal` returns it exactly once.
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    pub chain_pem: Option<String>,
81    /// SHA-256 fingerprint of the certificate.
82    pub fingerprint: String,
83    pub key_algorithm: KeyAlgorithm,
84    /// Validity start.
85    pub not_before: DateTime<Utc>,
86    /// Validity end.
87    pub not_after: DateTime<Utc>,
88    pub status: CertificateStatus,
89    /// AES-256-GCM encrypted private key — only for CAs under
90    /// [`CaKeyCustody::Database`]. `None` under every other custodian, and for
91    /// an imported CA whose key AXIAM never held.
92    #[serde(skip_serializing)]
93    #[schema(read_only)]
94    pub encrypted_private_key: Option<Vec<u8>>,
95    /// Which custodian holds this CA's signing key.
96    ///
97    /// Recorded per CA rather than read from configuration, so adopting a new
98    /// custodian does not strand the CAs that already exist. Not secret — an
99    /// operator needs to see it, and it discloses only where a key is kept.
100    #[serde(default = "default_key_custody")]
101    #[schema(value_type = String, example = "database")]
102    pub key_custody: CaKeyCustody,
103    /// Where the custodian put the key. A Vault path under its mount; `None`
104    /// for database custody, whose locator is the row itself.
105    pub key_locator: Option<String>,
106    pub created_at: DateTime<Utc>,
107}
108
109/// What a CA row with no recorded custodian means.
110///
111/// Every row written before custody was a concept holds its key sealed into
112/// itself, so `Database` is the only reading that finds those keys. A row that
113/// genuinely has no key is `External`, and the v45 migration distinguishes the
114/// two by whether `encrypted_private_key` is present.
115fn default_key_custody() -> CaKeyCustody {
116    CaKeyCustody::Database
117}

119/// Manual `Debug` impl (SECHRD-09 / D-06): `#[serde(skip_serializing)]` only
120/// affects `Serialize`, not `{:?}` — this closes that residual leak by
121/// redacting `encrypted_private_key` while keeping other fields readable.
122impl std::fmt::Debug for CaCertificate {
123    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
124        f.debug_struct("CaCertificate")
125            .field("id", &self.id)
126            .field("organization_id", &self.organization_id)
127            .field("tenant_id", &self.tenant_id)
128            .field("parent_ca_id", &self.parent_ca_id)
129            .field("subject", &self.subject)
130            .field("public_cert_pem", &self.public_cert_pem)
131            .field("chain_pem", &self.chain_pem)
132            .field("fingerprint", &self.fingerprint)
133            .field("key_algorithm", &self.key_algorithm)
134            .field("not_before", &self.not_before)
135            .field("not_after", &self.not_after)
136            .field("status", &self.status)
137            .field(
138                "encrypted_private_key",
139                &self.encrypted_private_key.as_ref().map(|_| "[REDACTED]"),
140            )
141            .field("key_custody", &self.key_custody)
142            .field("key_locator", &self.key_locator)
143            .field("created_at", &self.created_at)
144            .finish()
145    }
146}

148/// Fields required to generate a new CA certificate (user-facing DTO).
149#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
150pub struct CreateCaCertificate {
151    #[serde(default)]
152    pub organization_id: Uuid,
153    pub subject: String,
154    pub key_algorithm: KeyAlgorithm,
155    /// Validity duration in days.
156    pub validity_days: u32,
157    /// Common name for the signing intermediate, under `vault_pki` custody.
158    ///
159    /// Ignored by every other custodian, which generates one self-signed CA and
160    /// has no second certificate to name. Defaults to the root's subject with
161    /// `Intermediate Authority` appended, which is what HashiCorp's own PKI
162    /// walkthrough calls it.
163    #[serde(default, skip_serializing_if = "Option::is_none")]
164    pub intermediate_subject: Option<String>,
165    /// Validity of the signing intermediate, in days. Defaults to the root's.
166    ///
167    /// Vault caps it to the root's own expiry regardless, so asking for more
168    /// than the root has is not an error — it is quietly shortened, and the
169    /// stored record describes what came back rather than what was asked for.
170    #[serde(default, skip_serializing_if = "Option::is_none")]
171    pub intermediate_validity_days: Option<u32>,
172    /// Issue directly from the generated root instead of creating an
173    /// intermediate. `vault_pki` custody only.
174    ///
175    /// Off by default, and the default is the safer one: a root that signs only
176    /// one intermediate can have that intermediate revoked and replaced without
177    /// redistributing the trust anchor, while a root that signs leaves directly
178    /// cannot be rotated without touching every relying party.
179    #[serde(default)]
180    pub issue_from_root: bool,
181}

207    pub fingerprint: String,

216/// Import an externally-generated CA (BYOK).
217///
218/// The other way a CA comes to exist. Generation makes a key AXIAM chose;
219/// this takes one an organization already has — from an offline root, an
220/// existing internal PKI, or an HSM ceremony — and puts AXIAM in the chain
221/// rather than at the top of it.
222///
223/// `private_key_pem` is optional, and the two cases are genuinely different
224/// rather than one being a degraded form of the other:
225///
226/// * **With a key**, AXIAM takes custody of it (Vault, or sealed into the row)
227///   and can issue certificates against this CA.
228/// * **Without one**, the certificate is a trust anchor and nothing more.
229///   `/api/v1/certificates` cannot sign against it, and says so rather than
230///   failing at the point of use.
231#[derive(Clone, Serialize, Deserialize, utoipa::ToSchema)]
232pub struct ImportCaCertificate {
233    #[serde(default)]
234    pub organization_id: Uuid,
235    /// PEM-encoded CA certificate.
236    pub public_cert_pem: String,
237    /// PEM-encoded private key, if AXIAM is to hold it.
238    ///
239    /// Write-only: it goes to the configured custodian on the way in and is
240    /// never returned by any endpoint.
241    #[serde(default, skip_serializing)]
242    pub private_key_pem: Option<String>,
243}

256            )
257            .finish()
258    }

261/// Create a tenant signing CA, with AXIAM generating the key.
262///
263/// The tenant counterpart to [`CreateCaCertificate`]. The organization CA named
264/// by `parent_ca_id` signs it, and the resulting intermediate is what the
265/// tenant's user, service and device certificates are issued from — so a
266/// tenant's issuance can be revoked, rotated or handed to a different operator
267/// without redistributing the anchor every relying party trusts.
268///
269/// `key_algorithm` is the *intermediate's* own key, not the parent's; the two
270/// need not match, and an Ed25519 intermediate under an RSA root is a perfectly
271/// ordinary shape.
272#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
273pub struct CreateIntermediateCa {
274    /// Taken from the path, not the body.
275    #[serde(default)]
276    pub organization_id: Uuid,
277    /// Taken from the path, not the body.
278    #[serde(default)]
279    pub tenant_id: Uuid,
280    /// The organization CA that signs it. Must be Active, in its validity
281    /// window, and hold a key AXIAM can sign with.
282    pub parent_ca_id: Uuid,
283    /// The intermediate's subject (e.g., `CN=ACME R&D Signing CA`).
284    pub subject: String,
285    pub key_algorithm: KeyAlgorithm,
286    /// Validity duration in days. Capped to the parent's own expiry — an
287    /// intermediate that outlives its issuer is a certificate nothing accepts.
288    pub validity_days: u32,
289}

291/// Sign a certificate signing request the tenant produced elsewhere.
292///
293/// The BYOK counterpart to [`CreateIntermediateCa`]: the private key is
294/// generated by whoever made the CSR — an offline ceremony, an HSM, a
295/// tenant-side tool — and never reaches AXIAM at all. What comes back is a
296/// certificate and its chain, and the response carries no `private_key_pem`
297/// because there is none to carry.
298///
299/// The CSR's subject is what the certificate says. Its requested extensions are
300/// not: AXIAM decides that this is a CA, and constrains it to sign leaves only.
301#[derive(Clone, Serialize, Deserialize, utoipa::ToSchema)]
302pub struct SignIntermediateCsr {
303    /// Taken from the path, not the body.
304    #[serde(default)]
305    pub organization_id: Uuid,
306    /// Taken from the path, not the body.
307    #[serde(default)]
308    pub tenant_id: Uuid,
309    /// The organization CA that signs it.
310    pub parent_ca_id: Uuid,
311    /// PEM-encoded PKCS#10 certificate signing request.
312    pub csr_pem: String,
313    /// Validity duration in days. Capped to the parent's own expiry.
314    pub validity_days: u32,
315}
316
317/// A CSR is public by construction — it carries a public key and a signature
318/// proving possession of the private one — but it is bulky and appears in
319/// handler-level tracing spans, so it is elided rather than redacted.
320impl std::fmt::Debug for SignIntermediateCsr {
321    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
322        f.debug_struct("SignIntermediateCsr")
323            .field("organization_id", &self.organization_id)
324            .field("tenant_id", &self.tenant_id)
325            .field("parent_ca_id", &self.parent_ca_id)
326            .field("csr_pem", &format_args!("[{} bytes]", self.csr_pem.len()))
327            .field("validity_days", &self.validity_days)
328            .finish()
329    }
330}

332/// Response returned when a CA certificate is generated.
333///
334/// Includes the private key PEM, which is returned **once** and never
335/// stored or retrievable again — when the custodian produced one at all. Under
336/// `vault_pki` custody the key was born inside Vault and there is nothing to
337/// return, which is the point of that custodian rather than a shortcoming of
338/// this response.
339#[derive(Clone, Serialize, Deserialize, utoipa::ToSchema)]
340pub struct GeneratedCaCertificate {
341    #[serde(flatten)]
342    pub certificate: CaCertificate,
343    /// PEM-encoded private key — returned only on generation, and only when
344    /// there is one to return.
345    ///
346    /// Absent under `vault_pki` custody, where the key was generated inside
347    /// Vault and no API exports it. The field is omitted rather than sent as
348    /// `null` so a client that has always read it keeps working unchanged for
349    /// every custodian that does produce a key.
350    #[serde(default, skip_serializing_if = "Option::is_none")]
351    pub private_key_pem: Option<String>,
352}
353
354/// Manual `Debug` impl (SECHRD-09 / D-06): redacts `private_key_pem` (raw
355/// key material returned only once on generation); delegates to
356/// `CaCertificate`'s own redacting `Debug` for the nested `certificate` field.
357impl std::fmt::Debug for GeneratedCaCertificate {
358    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
359        f.debug_struct("GeneratedCaCertificate")
360            .field("certificate", &self.certificate)
361            // `Option`-aware rather than an unconditional `[REDACTED]`: a
362            // `vault_pki` CA has no key, and printing the redaction marker for
363            // one would say a key was withheld when none exists.
364            .field(
365                "private_key_pem",
366                &self.private_key_pem.as_ref().map(|_| "[REDACTED]"),
367            )
368            .finish()
369    }
370}

372/// A tenant-level certificate for users, services, or IoT devices.
373///
374/// Certificates are signed by the organization's CA. The private key is
375/// returned once on generation and never stored by AXIAM.
376#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
377pub struct Certificate {
378    pub id: Uuid,
379    /// The tenant this certificate belongs to.
380    pub tenant_id: Uuid,
381    /// The CA certificate that signed this certificate.
382    pub issuer_ca_id: Uuid,
383    /// The certificate subject (e.g., `CN=device-001`).
384    pub subject: String,
385    /// PEM-encoded public certificate.
386    pub public_cert_pem: String,
387    /// SHA-256 fingerprint of the certificate.
388    pub fingerprint: String,
389    pub cert_type: CertificateType,
390    pub key_algorithm: KeyAlgorithm,
391    /// Validity start.
392    pub not_before: DateTime<Utc>,
393    /// Validity end.
394    pub not_after: DateTime<Utc>,
395    pub status: CertificateStatus,
396    /// Arbitrary key-value metadata (e.g., device serial, user ID binding).
397    pub metadata: serde_json::Value,
398    pub created_at: DateTime<Utc>,
399}
