1//! Client-Initiated Backchannel Authentication — G-7 (OpenID Connect CIBA
2//! Core 1.0).
3//!
4//! A relying party that already knows *who* it wants to authenticate — a call
5//! centre that has the caller on the phone, a point-of-sale terminal, a
6//! back-office service acting for a customer — asks AXIAM over a direct,
7//! client-authenticated call (`POST /oauth2/bc-authorize`) to authenticate that
8//! user **on another device**. AXIAM notifies the user, the user approves on the
9//! identity pages after a full sign-in, and the client obtains tokens from the
10//! token endpoint with `grant_type=urn:openid:params:grant-type:ciba`.
11//!
12//! This module holds the domain types every layer shares: the client metadata
13//! (CIBA Core §4), the pending-request row and its state machine, the approval
14//! evidence, the ping-mode notification contract (§10.2) and the port the user
15//! notification goes through.
16//!
17//! # The state machine
18//!
19//! ```text
20//! pending ──approve──▶ approved ──redeem──▶ redeemed
21//!    │                    │
22//!    ├──deny──▶ denied    └──(expiry)──▶ expired
23//!    └──(expiry)──▶ expired
24//! ```
25//!
26//! Every arrow is a **conditional write** in the datastore (T-406's lesson):
27//! approval and denial are conditional on the version the approving page read
28//! and on the request's user being the approving user; redemption is the X6
29//! two-layer single-use arbiter, conditional on the client that started the
30//! request. Two parties write to one row — the user deciding and the client
31//! polling — and neither can put back what the other changed.
32

44/// How a CIBA client learns that a request has been decided (CIBA Core §5).
45///
46/// `push` is deliberately absent: AXIAM does not offer it, and the FAPI-CIBA
47/// profile forbids it — push delivers the tokens themselves to a client
48/// endpoint, which makes the notification endpoint a token sink.
49#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize, utoipa::ToSchema)]
50#[serde(rename_all = "lowercase")]
51pub enum CibaDeliveryMode {
52    /// The client polls the token endpoint at `interval` until the request is
53    /// decided.
54    Poll,
55    /// AXIAM calls the client's `backchannel_client_notification_endpoint`
56    /// once the request is decided, and the client then calls the token
57    /// endpoint once.
58    Ping,
59}
60
61impl CibaDeliveryMode {
62    /// The wire and storage spelling.
63    #[must_use]
64    pub const fn as_str(self) -> &'static str {
65        match self {
66            Self::Poll => "poll",
67            Self::Ping => "ping",
68        }
69    }
70
71    /// Parse a stored or registered value. `None` for anything else —
72    /// including `push`, which is not offered.
73    #[must_use]
74    pub fn from_wire(raw: &str) -> Option<Self> {
75        match raw.trim() {
76            "poll" => Some(Self::Poll),
77            "ping" => Some(Self::Ping),
78            _ => None,
79        }
80    }
81}

83/// The JWS algorithm a CIBA client signs its authentication requests with
84/// (CIBA Core §4 `backchannel_authentication_request_signing_alg`, §7.1.1).
85///
86/// Exactly the three algorithms AXIAM verifies on any client-signed JWT
87/// (`axiam_oauth2::jose::PERMITTED_ALGORITHMS`): FAPI 2.0 §5.3.1.1's list. A
88/// registration naming anything else — `RS256`, `HS256`, `none` — is refused
89/// rather than stored, so no row can hold an algorithm the verifier would not
90/// honour (D-61).
91#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize, utoipa::ToSchema)]
92pub enum CibaRequestSigningAlg {
93    /// RSASSA-PSS with SHA-256.
94    #[serde(rename = "PS256")]
95    Ps256,
96    /// ECDSA on P-256 with SHA-256.
97    #[serde(rename = "ES256")]
98    Es256,
99    /// Ed25519.
100    #[serde(rename = "EdDSA")]
101    EdDsa,
102}
103
104impl CibaRequestSigningAlg {
105    /// Every algorithm, in discovery order.
106    pub const ALL: [Self; 3] = [Self::Ps256, Self::Es256, Self::EdDsa];
107
108    /// The JOSE spelling, on the wire and in storage.
109    #[must_use]
110    pub const fn as_str(self) -> &'static str {
111        match self {
112            Self::Ps256 => "PS256",
113            Self::Es256 => "ES256",
114            Self::EdDsa => "EdDSA",
115        }
116    }
117
118    /// Parse the JOSE spelling, exactly (JOSE names are case-sensitive).
119    #[must_use]
120    pub fn from_wire(raw: &str) -> Option<Self> {
121        Self::ALL.into_iter().find(|a| a.as_str() == raw.trim())
122    }
123}

125/// The CIBA client metadata AXIAM stores (CIBA Core §4).
126///
127/// Three of the four members §4 defines. The fourth,
128/// `backchannel_user_code_parameter`, is refused at registration rather than
129/// stored, because AXIAM holds no user code to check (D-64). A stored value
130/// nothing reads would be a security switch that does nothing — the SEC-097
131/// shape.
132#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize, utoipa::ToSchema)]
133pub struct CibaClientMetadata {
134    /// `backchannel_token_delivery_mode` — required when the client holds the
135    /// CIBA grant, absent otherwise.
136    #[serde(default)]
137    pub backchannel_token_delivery_mode: Option<CibaDeliveryMode>,
138    /// `backchannel_client_notification_endpoint` — required in `ping` mode,
139    /// refused in `poll` mode. Held to the outbound URL policy webhooks use.
140    #[serde(default)]
141    pub backchannel_client_notification_endpoint: Option<String>,
142    /// `backchannel_authentication_request_signing_alg` — when set, **every**
143    /// backchannel authentication request from this client must be a signed
144    /// `request` JWT under exactly this algorithm (CIBA Core §7.1.1), and an
145    /// unsigned one is refused. Required for a `fapi2` client (FAPI-CIBA
146    /// §5.2.2). When absent, a `request` parameter is refused.
147    #[serde(default)]
148    pub backchannel_authentication_request_signing_alg: Option<CibaRequestSigningAlg>,
149}
150
151impl CibaClientMetadata {
152    /// Whether any member is set.
153    #[must_use]
154    pub fn is_empty(&self) -> bool {
155        self.backchannel_token_delivery_mode.is_none()
156            && self
157                .backchannel_authentication_request_signing_alg
158                .is_none()
159            && self
160                .backchannel_client_notification_endpoint
161                .as_deref()
162                .is_none_or(|e| e.trim().is_empty())
163    }
164}

166/// Where a backchannel authentication request has got to.
167#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize, utoipa::ToSchema)]
168#[serde(rename_all = "lowercase")]
169pub enum CibaRequestStatus {
170    /// Stored; the user has not decided. The token endpoint answers
171    /// `authorization_pending`.
172    Pending,
173    /// The user approved after authenticating. The next token request from
174    /// the client that started it redeems it.
175    Approved,
176    /// The user refused. The token endpoint answers `access_denied`.
177    Denied,
178    /// The request outlived its `expires_in` undecided or unredeemed. The
179    /// token endpoint answers `expired_token`.
180    Expired,
181    /// Exchanged for tokens. Terminal; a second redemption is `invalid_grant`.
182    Redeemed,
183}
184
185impl CibaRequestStatus {
186    /// The storage spelling.
187    #[must_use]
188    pub const fn as_str(self) -> &'static str {
189        match self {
190            Self::Pending => "pending",
191            Self::Approved => "approved",
192            Self::Denied => "denied",
193            Self::Expired => "expired",
194            Self::Redeemed => "redeemed",
195        }
196    }
197
198    /// Parse a stored value. `None` for anything unrecognised: the states
199    /// differ by whether they grant access, so the caller fails closed.
200    #[must_use]
201    pub fn from_wire(raw: &str) -> Option<Self> {
202        match raw.trim() {
203            "pending" => Some(Self::Pending),
204            "approved" => Some(Self::Approved),
205            "denied" => Some(Self::Denied),
206            "expired" => Some(Self::Expired),
207            "redeemed" => Some(Self::Redeemed),
208            _ => None,
209        }
210    }
211}

213/// The authentication the approving user performed — what the ID token minted
214/// from this request reports (`auth_time`, `acr`, `amr`), the same evidence an
215/// authorization code carries.
216#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
217pub struct CibaApprovalEvidence {
218    /// The session the user approved from. The minted access token names it in
219    /// `sid`, so ending that session ends the tokens (as for a code grant).
220    pub session_id: Uuid,
221    /// When the user authenticated (OIDC Core §2 `auth_time`).
222    pub auth_time: DateTime<Utc>,
223    /// The class the authentication achieved, derived from `amr` by the
224    /// service — never supplied by the caller.
225    pub acr: String,
226    /// RFC 8176 method references of that authentication.
227    pub amr: Vec<Amr>,
228}

230/// A backchannel authentication request, as stored.
231#[derive(Debug, Clone, PartialEq, Eq)]
232pub struct CibaRequest {
233    /// Record id. Also the handle the approval page addresses the request by —
234    /// not a secret: approval requires the request's own user to be signed in.
235    pub id: Uuid,
236    /// The tenant the request belongs to.
237    pub tenant_id: Uuid,
238    /// The client that started it; the only client that may redeem it.
239    pub client_id: String,
240    /// SHA-256 of the `auth_req_id`. The raw value is returned once.
241    pub auth_req_id_hash: String,
242    /// The user the hint resolved to. `None` for a request whose hint named
243    /// nobody who may sign in: such a request is stored and answered exactly
244    /// like a real one so that `bc-authorize` is not a user oracle (D-63), and
245    /// it can only ever expire.
246    pub user_id: Option<Uuid>,
247    /// The granted scopes (always including `openid`).
248    pub scopes: Vec<String>,
249    /// The `binding_message` shown to the user on both devices.
250    pub binding_message: Option<String>,
251    /// The requested `acr_values`, verbatim (unknown values are kept and can
252    /// never be satisfied).
253    pub acr_values: Vec<String>,
254    /// RFC 8707 resource this request was made for, if any.
255    pub resource: Option<String>,
256    /// The delivery mode the client was registered for when it asked.
257    pub delivery_mode: CibaDeliveryMode,
258    /// Where the request is.
259    pub status: CibaRequestStatus,
260    /// Bumped by every status transition. Approval and denial are conditional
261    /// on the value the approving page read.
262    pub version: u64,
263    /// Minimum seconds between token requests; raised by `slow_down`.
264    pub interval_secs: u64,
265    /// When the client last asked the token endpoint about this request.
266    pub last_polled_at: Option<DateTime<Utc>>,
267    /// When the request stops being redeemable.
268    pub expires_at: DateTime<Utc>,
269    /// Set at approval.
270    pub approval: Option<CibaApprovalEvidence>,
271    /// When the user decided.
272    pub decided_at: Option<DateTime<Utc>>,
273    /// When the request was stored.
274    pub created_at: DateTime<Utc>,
275}
276
277impl CibaRequest {
278    /// Whether `now` is past the request's expiry.
279    #[must_use]
280    pub fn is_expired_at(&self, now: DateTime<Utc>) -> bool {
281        self.expires_at <= now
282    }
283}

285/// The two secrets a ping-mode request needs at delivery time.
286///
287/// Sealed at rest by the repository (AES-256-GCM under `pki_encryption_key`):
288/// `client_notification_token` is the bearer credential AXIAM presents to the
289/// client's notification endpoint, and `auth_req_id` is the body of that
290/// notification — which is why, unlike a poll-mode request, a ping-mode request
291/// must keep the identifier recoverable rather than only its hash.
292#[derive(Clone, PartialEq, Eq)]
293pub struct CibaPingCredentials {
294    /// The `auth_req_id` the notification carries (CIBA Core §10.2).
295    pub auth_req_id: String,
296    /// The bearer token the client supplied at `bc-authorize` (§7.1).
297    pub client_notification_token: String,
298}
299
300/// Redacting `Debug`: both members are credentials.
301impl std::fmt::Debug for CibaPingCredentials {
302    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
303        f.debug_struct("CibaPingCredentials")
304            .finish_non_exhaustive()
305    }
306}

337/// The ping-mode notification body (CIBA Core §10.2) — the contract T23.7.2's
338/// deliverer sends, as `POST` to the client's
339/// `backchannel_client_notification_endpoint` with
340/// `Authorization: Bearer <client_notification_token>` and
341/// `Content-Type: application/json`.
342///
343/// Nothing else is in it: the notification says *that* a request was decided,
344/// never how; the client learns the outcome from the token endpoint, where it
345/// authenticates.

347pub struct CibaPingNotification {
348    /// The `auth_req_id` the client was given at `bc-authorize`.
349    pub auth_req_id: String,
350}

352/// What the user is told about a request (the port's payload).
353///

368    /// The `binding_message`, if the client sent one.
369    pub binding_message: Option<String>,
370    /// The scopes requested.
371    pub scopes: Vec<String>,
372    /// When the request expires.
373    pub expires_at: DateTime<Utc>,

376/// A boxed future returned by [`CibaUserNotifier`].
377pub type CibaNotifyFuture<'a> =
378    std::pin::Pin<Box<dyn std::future::Future<Output = Result<(), AxiamError>> + Send + 'a>>;
379
380/// The port through which a stored request reaches its user (G-7).
381///
382/// `bc-authorize` calls it **best effort, after the request is stored, and
383/// detached from the response**: a failed notification never fails the request
384/// (the user can still find it on the identity pages), and the response does
385/// not wait on it, so its timing says nothing about whether the hint named a
386/// real user. T23.7.2 implements it over `axiam-email` (later push); until a
387/// deployment wires one, [`NoopCibaUserNotifier`] is in place.
388pub trait CibaUserNotifier: Send + Sync {
389    /// Tell the user a request is waiting.
390    fn notify(&self, notification: CibaUserNotification) -> CibaNotifyFuture<'_>;
391}
392
393/// The notifier that notifies nobody.
394#[derive(Debug, Clone, Copy, Default)]
395pub struct NoopCibaUserNotifier;
396
397impl CibaUserNotifier for NoopCibaUserNotifier {
398    fn notify(&self, _notification: CibaUserNotification) -> CibaNotifyFuture<'_> {
399        // The request still waits on the identity pages; nobody is told.
400        Box::pin(async { Ok(()) })
401    }
402}
