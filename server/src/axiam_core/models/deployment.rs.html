1//! The deployment profile (G-8, D-59): whether this process runs with a message
2//! broker (`full`) or without one (`minimal`, `AXIAM__AMQP__ENABLED=false`).
3//!
4//! A value type in layer 0 so that the REST health endpoint, the REST and gRPC
5//! reactor-administration surfaces and the composition root all speak the same
6//! two words, and so the list of what the minimal profile gives up is written
7//! down once.
8
9use serde::{Deserialize, Serialize};
10
11/// Which messaging profile the process was composed with.
12#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default, Serialize, Deserialize)]
13#[serde(rename_all = "lowercase")]
14pub enum DeploymentProfile {
15    /// RabbitMQ is used: asynchronous authorization, external audit ingestion,
16    /// reactors, cache invalidation across replicas, durable outbound queues.
17    #[default]
18    Full,
19    /// No broker (`AXIAM__AMQP__ENABLED=false`). Single instance by definition;
20    /// outbound delivery and mail run on in-process queues.
21    Minimal,
22}
23
24/// What the minimal profile does not provide, as reported by `GET /health`
25/// (`unavailable`) — the closed set the contract and the docs name.
26pub const MINIMAL_PROFILE_UNAVAILABLE: [&str; 4] = [
27    "reactors",
28    "amqp_authz",
29    "amqp_audit_ingestion",
30    "decision_cache_broadcast",
31];
32
33impl DeploymentProfile {
34    /// The wire name: `"full"` or `"minimal"`.
35    pub const fn as_str(self) -> &'static str {
36        match self {
37            DeploymentProfile::Full => "full",
38            DeploymentProfile::Minimal => "minimal",
39        }
40    }
41
42    /// Whether this is the broker-less profile.
43    pub const fn is_minimal(self) -> bool {
44        matches!(self, DeploymentProfile::Minimal)
45    }
46
47    /// The capabilities this profile does not provide: empty for `full`,
48    /// [`MINIMAL_PROFILE_UNAVAILABLE`] for `minimal`.
49    pub const fn unavailable(self) -> &'static [&'static str] {
50        match self {
51            DeploymentProfile::Full => &[],
52            DeploymentProfile::Minimal => &MINIMAL_PROFILE_UNAVAILABLE,
53        }
54    }
55
56    /// The sentence the reactor-administration surfaces answer with when a
57    /// registration would be enabled in the minimal profile.
58    pub const fn reactor_refusal_message() -> &'static str {
59        "reactors are unavailable in the minimal profile (AXIAM__AMQP__ENABLED=false): \
60         there is no broker to deliver interception events through, and an enabled \
61         registration would apply its failure_policy to every dispatch — login.post_auth, \
62         user.pre_create, user.pre_update and grant.pre_assign default to fail_closed, so \
63         it would deny those operations for the tenant. Create or update the registration \
64         with enabled: false, or run the full profile."
65    }
66}

68#[cfg(test)]
69mod tests {
70    use super::*;
71
72    #[test]
73    fn wire_names_are_pinned() {
74        assert_eq!(DeploymentProfile::Full.as_str(), "full");
75        assert_eq!(DeploymentProfile::Minimal.as_str(), "minimal");
76        assert_eq!(
77            serde_json::to_string(&DeploymentProfile::Minimal).unwrap(),
78            "\"minimal\""
79        );
80        assert_eq!(DeploymentProfile::default(), DeploymentProfile::Full);
81    }
82
83    #[test]
84    fn only_the_minimal_profile_reports_unavailable_features() {
85        assert!(DeploymentProfile::Full.unavailable().is_empty());
86        assert_eq!(
87            DeploymentProfile::Minimal.unavailable(),
88            [
89                "reactors",
90                "amqp_authz",
91                "amqp_audit_ingestion",
92                "decision_cache_broadcast"
93            ]
94        );
95    }
96
97    #[test]
98    fn the_refusal_names_the_profile_and_the_fix() {
99        let message = DeploymentProfile::reactor_refusal_message();
100        assert!(message.contains("minimal profile"));
101        assert!(message.contains("AXIAM__AMQP__ENABLED=false"));
102        assert!(message.contains("enabled: false"));
103    }
104}
