1//! Directory (LDAP / Active Directory) identity-source configuration (G-3).
2//!
3//! One row per tenant. The model carries **no secret material**: the bind
4//! secret is encrypted at rest by the repository (decision D-15, the same way
5//! the per-tenant SMTP password is) and reaches the caller only through the
6//! dedicated [`DirectoryConfigRepository::decrypt_bind_secret`] method. That
7//! is why [`DirectoryConfig`], the type every read returns and every API
8//! response will be built from, has no secret field to redact in the first
9//! place; the only type that holds the plaintext is [`NewDirectoryConfig`], the
10//! write input, and its `Debug` redacts it.
11//!
12//! Validation of the values (the URL, the filter template, the trust anchors)
13//! lives in `axiam-directory::config`, which sits above this crate; this module
14//! only declares the shape and the per-[`DirectoryKind`] defaults.
15//!
16//! [`DirectoryConfigRepository::decrypt_bind_secret`]: crate::repository::DirectoryConfigRepository::decrypt_bind_secret
17

25/// Which kind of directory server a configuration points at.
26///
27/// It drives **defaults only**: the external-id attribute, the group-membership
28/// strategy and the change attribute the sync job reads. Every one of them is
29/// still an explicit, editable field of the configuration (or, for the
30/// strategy and change attribute, derived from this value at the point of use);
31/// nothing about the kind changes what is *allowed*.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize, utoipa::ToSchema)]
33#[serde(rename_all = "snake_case")]
34pub enum DirectoryKind {
35    /// OpenLDAP and other RFC 4519 directories with `entryUUID`.
36    OpenLdap,
37    /// Microsoft Active Directory (`objectGUID`, `memberOf`, `uSNChanged`).
38    ActiveDirectory,
39}
40
41/// How a user's directory groups are discovered.
42///
43/// Derived from [`DirectoryKind`]; see [`DirectoryKind::group_strategy`].
44#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
45pub enum GroupStrategy {
46    /// Read the groups straight off the user entry (`memberOf` on AD).
47    MemberOf,
48    /// Search the group subtree for groups whose `member` attribute names the
49    /// user's DN (OpenLDAP, which has no `memberOf` unless an overlay is on).
50    ReverseMember,
51}
52
53impl DirectoryKind {
54    /// The wire/storage spelling. One function so the repository, the schema
55    /// `ASSERT` and the REST layer cannot drift apart.
56    pub const fn as_str(self) -> &'static str {
57        match self {
58            Self::OpenLdap => "open_ldap",
59            Self::ActiveDirectory => "active_directory",
60        }
61    }
62
63    /// Parse a wire value. Unknown values yield `None` so a caller refuses
64    /// them rather than defaulting: a typo must not become OpenLDAP.
65    pub fn from_wire(raw: &str) -> Option<Self> {
66        match raw {
67            "open_ldap" => Some(Self::OpenLdap),
68            "active_directory" => Some(Self::ActiveDirectory),
69            _ => None,
70        }
71    }
72
73    /// The attribute that holds the directory's immutable identifier for an
74    /// entry: `entryUUID` (RFC 4530) or `objectGUID`. This is what AXIAM stores
75    /// as the user's `external_id`, because a DN or a username can change when
76    /// an account is renamed or moved and the identifier cannot.
77    pub const fn external_id_attribute(self) -> &'static str {
78        match self {
79            Self::OpenLdap => "entryUUID",
80            Self::ActiveDirectory => "objectGUID",
81        }
82    }
83
84    /// How group membership is discovered for this kind of directory.
85    pub const fn group_strategy(self) -> GroupStrategy {
86        match self {
87            Self::OpenLdap => GroupStrategy::ReverseMember,
88            Self::ActiveDirectory => GroupStrategy::MemberOf,
89        }
90    }
91
92    /// The attribute the incremental sync orders changes by: the operational
93    /// `modifyTimestamp` (OpenLDAP) or the per-server `uSNChanged` (AD).
94    pub const fn change_attribute(self) -> &'static str {
95        match self {
96            Self::OpenLdap => "modifyTimestamp",
97            Self::ActiveDirectory => "uSNChanged",
98        }
99    }
100
101    /// The default for [`DirectoryConfig::group_member_attribute`]: the
102    /// user-side `memberOf` on AD, the group-side `member` on OpenLDAP.
103    pub const fn default_group_member_attribute(self) -> &'static str {
104        match self.group_strategy() {
105            GroupStrategy::MemberOf => "memberOf",
106            GroupStrategy::ReverseMember => "member",
107        }
108    }
109
110    /// The default for [`DirectoryConfig::user_attribute_map`].
111    pub fn default_user_attribute_map(self) -> UserAttributeMap {
112        UserAttributeMap {
113            username: match self {
114                Self::OpenLdap => "uid",
115                Self::ActiveDirectory => "sAMAccountName",
116            }
117            .to_string(),
118            email: "mail".to_string(),
119            display_name: "displayName".to_string(),
120            external_id: self.external_id_attribute().to_string(),
121        }
122    }
123}

153/// One row of the group-mapping table (G-3, T23.3.4, D-30): a directory group,
154/// named by its distinguished name, and the AXIAM group a member of it is put
155/// into.
156///
157/// **The table is the only way a directory group reaches an AXIAM group.**
158/// There is no match by name, no prefix or wildcard, and no AXIAM group is ever
159/// created from a directory one: a directory administrator who names a group
160/// `admins` gains nothing unless a tenant administrator mapped it here.
161///
162/// The DN is stored as the administrator typed it and compared after RFC 4514
163/// normalisation (`axiam_directory::dn`), so `CN=Staff, OU=Groups` and
164/// `cn=staff,ou=groups` are the same row. One DN may map to several AXIAM
165/// groups; the same (DN, group) pair twice is refused as redundant.
166#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize, utoipa::ToSchema)]
167pub struct GroupMapping {
168    /// The directory group's distinguished name.
169    pub directory_group_dn: String,
170    /// The AXIAM group of the same tenant a member of that directory group is
171    /// put into. Checked to exist in the tenant when the configuration is
172    /// written.
173    pub group_id: Uuid,
174}

176/// A tenant's directory configuration, as stored and as read back.
177///
178/// Carries no secret: see the module documentation.
179#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, utoipa::ToSchema)]
180pub struct DirectoryConfig {
181    /// Row identifier.
182    pub id: Uuid,
183    /// The owning tenant. At most one configuration exists per tenant.
184    pub tenant_id: Uuid,
185    /// Whether the directory is used for sign-in and sync.
186    pub enabled: bool,
187    /// The kind of directory, which selects defaults.
188    pub kind: DirectoryKind,
189    /// `ldaps://host[:port]` or `ldap://host[:port]` together with
190    /// [`Self::start_tls`]. A plaintext URL is refused at configuration time.
191    pub url: String,
192    /// Upgrade an `ldap://` connection with StartTLS before any bind.
193    pub start_tls: bool,
194    /// The service account AXIAM binds as to search. It should hold read-only
195    /// rights: AXIAM never writes to a directory.
196    pub bind_dn: String,
197    /// Where users are searched for.
198    pub base_dn: String,
199    /// The user-lookup filter template. It contains exactly one `{username}`

204    pub user_attribute_map: UserAttributeMap,
205    /// Where groups are searched for (reverse-`member` lookups, group sync).
206    pub group_base_dn: Option<String>,
207    /// Restricts which entries under [`Self::group_base_dn`] are groups.
208    pub group_filter: Option<String>,
209    /// `memberOf` (user-side, AD) or `member` (group-side, OpenLDAP).
210    pub group_member_attribute: String,
211    /// How many levels of nested groups are followed, `0..=10`.
212    #[schema(minimum = 0, maximum = 10)]
213    pub group_nesting_depth: u8,
214    /// The group-mapping table (D-30): which directory groups put a user into
215    /// which AXIAM groups. Empty means no directory group maps to anything, and
216    /// a sign-in then removes every directory-sourced membership the user held.
217    #[schema(max_items = 500)]
218    pub group_mappings: Vec<GroupMapping>,
219    /// Seconds between incremental sync runs.
220    #[schema(minimum = 300, maximum = 86400)]
221    pub sync_interval_secs: u64,
222    /// Provision an AXIAM user on first successful directory sign-in.
223    pub jit_provisioning: bool,
224    /// PEM CA certificates that anchor trust in the directory's server
225    /// certificate. Empty means the platform roots used by the rest of the
226    /// workspace's outbound TLS. An organisation CA's PEM can be pasted here.
227    pub trust_anchors_pem: Vec<String>,
228    /// When the row was created.
229    pub created_at: DateTime<Utc>,
230    /// When the row was last written.
231    pub updated_at: DateTime<Utc>,
232}

285/// `Debug` names the secret's presence and nothing else: `Zeroizing<String>`
286/// delegates `Debug` to the inner string, so a derived impl would print the
287/// bind password the first time anyone logged an input with `{:?}`.

289    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
290        f.debug_struct("NewDirectoryConfig")
291            .field("tenant_id", &self.tenant_id)
292            .field("enabled", &self.enabled)
293            .field("kind", &self.kind)
294            .field("url", &self.url)
295            .field("start_tls", &self.start_tls)
296            .field("bind_dn", &self.bind_dn)
297            .field(
298                "bind_secret",
299                &self.bind_secret.as_ref().map(|_| "<redacted>"),
300            )
301            .field("base_dn", &self.base_dn)
302            .field("user_filter", &self.user_filter)
303            .field("user_attribute_map", &self.user_attribute_map)
304            .field("group_base_dn", &self.group_base_dn)
305            .field("group_filter", &self.group_filter)
306            .field("group_member_attribute", &self.group_member_attribute)
307            .field("group_nesting_depth", &self.group_nesting_depth)
308            .field("group_mappings", &self.group_mappings.len())
309            .field("sync_interval_secs", &self.sync_interval_secs)
310            .field("jit_provisioning", &self.jit_provisioning)
311            .field("trust_anchors_pem", &self.trust_anchors_pem.len())
312            .finish()
313    }
314}

320/// What a successful directory bind established about a user.
321///
322/// Returned by [`DirectoryAuthenticator::authenticate`] only after the
323/// directory accepted a bind **as that entry, with the presented password**.
324/// `external_id` is the entry's immutable identifier (`entryUUID`, or the AD
325/// `objectGUID` decoded to its canonical text form), which is what AXIAM keys
326/// a directory account on; the other fields are the mapped attributes, which
327/// just-in-time provisioning (T23.3.3) copies and group mapping (T23.3.4)
328/// starts from.
329///
330/// Every field but the identifier names a person, so `Debug` prints presence
331/// only.
332#[derive(Clone, PartialEq, Eq)]
333pub struct DirectoryIdentity {
334    /// The entry's immutable identifier, normalised: a lowercase hyphenated
335    /// UUID for `entryUUID` and `objectGUID`, the attribute's text otherwise.
336    pub external_id: String,
337    /// The entry's distinguished name, exactly as the directory returned it
338    /// from the user search (never constructed by AXIAM).
339    pub dn: String,
340    /// The mapped login-name attribute, when the entry carried one.
341    pub username: Option<String>,
342    /// The mapped e-mail attribute, when the entry carried one.
343    pub email: Option<String>,
344    /// The mapped display-name attribute, when the entry carried one.
345    pub display_name: Option<String>,
346}
347
348impl fmt::Debug for DirectoryIdentity {
349    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
350        f.debug_struct("DirectoryIdentity")
351            .field("external_id", &"<redacted>")
352            .field("dn", &"<redacted>")
353            .field("username", &self.username.as_ref().map(|_| "<redacted>"))
354            .field("email", &self.email.as_ref().map(|_| "<redacted>"))
355            .field(
356                "display_name",
357                &self.display_name.as_ref().map(|_| "<redacted>"),
358            )
359            .finish()
360    }
361}

363/// Why the directory accepted the password but refused the account.
364///
365/// Read from Active Directory's `data XXX` sub-code in the bind diagnostic
366/// (`533`, `701`, `775`, `532`, `773`, `530`/`531`), or from the LDAP result
367/// code where the directory uses one for a refusal (`unwillingToPerform`,
368/// `insufficientAccessRights`). The end user never learns which: every variant
369/// is answered with the generic sign-in failure.
370#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
371pub enum DirectoryAccountRestriction {
372    /// The account is disabled (AD `533`, or a directory that refuses the bind
373    /// outright with `unwillingToPerform` / `insufficientAccessRights`).
374    Disabled,
375    /// The directory has locked the account (AD `775`).
376    Locked,
377    /// The account has expired (AD `701`).
378    Expired,
379    /// The password has expired (AD `532`).
380    PasswordExpired,
381    /// The password must be changed before the next sign-in (AD `773`).
382    PasswordMustChange,
383    /// Sign-in is not permitted at this time or from this workstation
384    /// (AD `530`, `531`).
385    NotPermittedNow,
386}
387
388/// Why a directory authentication did not succeed.
389///
390/// A small, closed set. It is the vocabulary between the directory client and
391/// the login path, never a response: the login path answers **every** variant
392/// with the same generic failure, so an unauthenticated caller cannot tell a
393/// wrong password from an unknown entry, a disabled account or an unreachable
394/// directory. The variants exist so the login path can decide what to **count**
395/// (only [`Self::InvalidCredentials`] moves the brute-force counter) and what
396/// to tell the **operator** in a log line.
397#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, thiserror::Error)]
398pub enum DirectoryAuthError {
399    /// The tenant has no directory configuration, or it is disabled.
400    #[error("no enabled directory is configured for this tenant")]
401    NotConfigured,
402    /// The password is wrong, the login name matched no entry or more than
403    /// one, the matched entry is not the account AXIAM holds, or the password
404    /// was empty (refused before any network I/O).
405    #[error("invalid credentials")]
406    InvalidCredentials,
407    /// The directory accepted the credentials but refuses the account.
408    #[error("the directory refused the account")]
409    AccountRestricted(DirectoryAccountRestriction),
410    /// The directory could not be reached or did not answer in time: a
411    /// connection, TLS or timeout failure, a busy server, an exhausted
412    /// connection pool, or the encryption key for the bind secret missing.
413    /// A TLS verification failure is reported here too — it is
414    /// indistinguishable from an interception attempt, and is never retried in
415    /// a weaker form.
416    #[error("the directory is unavailable")]
417    Unavailable,
418    /// The directory answered, but the configuration cannot work: the service
419    /// bind was refused, the search base does not exist, the directory sent a
420    /// referral (never followed), the entry lacks its identifier attribute, or
421    /// the stored configuration is unusable.
422    #[error("the directory configuration is unusable")]
423    Misconfigured,
424}

426/// The port the login path authenticates a directory account through.
427///
428/// Declared here, in layer 0, because the login path lives in `axiam-auth`
429/// (layer 1) and the LDAP client in `axiam-directory` (layer 3): the client
430/// implements this trait and the composition root injects it, so no production
431/// dependency points outward. It is object-safe (a boxed future rather than
432/// `impl Future`) because `AuthService` holds it as an optional field rather
433/// than a fifth type parameter, the same seam [`super::reactor::DynReactorGate`]
434/// is for the reactor gate.
435///
436/// # Contract for implementations
437///
438/// * Resolve the entry by the tenant's user filter with `login_name`
439///   RFC 4515-escaped, require **exactly one** match, then bind **as that
440///   entry** with `password`. Success means that bind succeeded.
441/// * Refuse an empty `password` with [`DirectoryAuthError::InvalidCredentials`]
442///   before any network I/O: an empty password is an RFC 4513 unauthenticated
443///   bind, which many servers report as success.
444/// * Never send anything over an unencrypted connection, never follow a
445///   referral, and never return a connection that is bound as the user to a
446///   pool.
447/// * Never log `password`, and never surface the directory's diagnostic text
448///   above `debug`.
449pub trait DirectoryAuthenticator: Send + Sync {
450    /// Authenticate `login_name` / `password` against `tenant_id`'s directory.
451    fn authenticate<'a>(
452        &'a self,
453        tenant_id: Uuid,
454        login_name: &'a str,
455        password: &'a str,
456    ) -> std::pin::Pin<
457        Box<
458            dyn std::future::Future<Output = Result<DirectoryIdentity, DirectoryAuthError>>
459                + Send
460                + 'a,
461        >,
462    >;
463
464    /// Authenticate `login_name` / `password` for a login name **AXIAM holds no
465    /// account for** (just-in-time provisioning, G-3, T23.3.3).
466    ///
467    /// Same contract as [`Self::authenticate`], plus one gate: the
468    /// implementation answers [`DirectoryAuthError::NotConfigured`], **without
469    /// contacting the directory**, unless the tenant's directory is enabled
470    /// **and** has `jit_provisioning` on. The gate sits in front of the bind
471    /// because a bind for a name the tenant never asked to provision would run
472    /// up a failed-bind count in the directory that no AXIAM counter can see
473    /// (T-302).
474    ///
475    /// A provided method, defaulting to `NotConfigured`: an implementation that
476    /// does not override it provisions nobody, which is the safe reading of "I
477    /// was not asked to". It is a method of this port rather than a second
478    /// port so that one injected object is all `AuthService` knows about
479    /// directories.
480    fn authenticate_for_provisioning<'a>(
481        &'a self,
482        tenant_id: Uuid,
483        login_name: &'a str,
484        password: &'a str,
485    ) -> DirectoryFuture<'a, Result<DirectoryIdentity, DirectoryAuthError>> {
486        let _ = (tenant_id, login_name, password);
487        Box::pin(async { Err(DirectoryAuthError::NotConfigured) })
488    }
489
490    /// Resolve the entry `login_name` names, through the tenant's configured
491    /// user filter, **without binding as it** — the lookup an administrator's
492    /// act of linking an existing account needs, since the administrator does
493    /// not hold the user's directory password.
494    ///
495    /// The entry is found by the directory and returned as the directory
496    /// describes it; a caller never supplies an identifier of its own. Exactly
497    /// one match, as for [`Self::authenticate`]: none or several is
498    /// [`DirectoryAuthError::InvalidCredentials`] ("no such single entry").
499    /// Not gated on `jit_provisioning` — linking is an administrator's
500    /// decision, not provisioning — but it does require an enabled directory.
501    /// Defaults to `NotConfigured`, like the method above.
502    fn lookup_entry<'a>(
503        &'a self,
504        tenant_id: Uuid,
505        login_name: &'a str,
506    ) -> DirectoryFuture<'a, Result<DirectoryIdentity, DirectoryAuthError>> {
507        let _ = (tenant_id, login_name);
508        Box::pin(async { Err(DirectoryAuthError::NotConfigured) })
509    }
510}

524/// Where the directory path writes its audit rows (G-3, T23.3.3).
525///
526/// `AuthService` has four repository type parameters already and `axiam-audit`
527/// (layer 2) sits above `axiam-auth`, so the audit write is a type-erased port
528/// declared here and implemented in `axiam-auth` over the existing
529/// append-only [`crate::repository::AuditLogRepository`]. An implementation
530/// never fails the caller: the directory path has already done, or refused,
531/// what the row describes, and a row that cannot be written is logged, not

537    /// Append one row.
538    fn record<'a>(
539        &'a self,
540        entry: crate::models::audit::CreateAuditLogEntry,
541    ) -> DirectoryFuture<'a, ()>;
542}

551/// What one application of the group-mapping table did to one user.
552///
553/// Identifiers and counts only — it is what the audit row is built from, and
554/// no name, DN or attribute of the person appears in it.
555#[derive(Debug, Clone, Default, PartialEq, Eq)]
556pub struct GroupMappingOutcome {
557    /// How many directory groups the user belongs to, nesting included. The
558    /// number the hard cap of `1 000` is measured against.
559    pub directory_groups_resolved: usize,
560    /// How many of those matched a row of the mapping table.
561    pub directory_groups_mapped: usize,
562    /// AXIAM groups the user was added to by this application.
563    pub added: Vec<Uuid>,
564    /// AXIAM groups a directory-sourced membership was removed from, because
565    /// the directory no longer backs it.
566    pub removed: Vec<Uuid>,
567    /// Mapped AXIAM groups the user was already a **manual** member of: left
568    /// exactly as they were, no second edge written (D-30).
569    pub left_manual: Vec<Uuid>,
570}
571
572impl GroupMappingOutcome {
573    /// Whether the user's memberships changed at all.
574    #[must_use]
575    pub fn changed(&self) -> bool {
576        !self.added.is_empty() || !self.removed.is_empty()
577    }
578}

580/// The port the sign-in path applies a directory user's group mapping through
581/// (G-3, T23.3.4, D-30).
582///
583/// Declared here for the reason [`DirectoryAuthenticator`] is: the login path
584/// is `axiam-auth` (layer 1), the LDAP lookup and the mapping live in
585/// `axiam-directory` (layer 3), and the composition root injects one into the
586/// other. The sync job (T23.3.5) calls the same implementation.
587///
588/// # Contract for implementations
589///
590/// * Ask the directory which groups `user_dn` belongs to — nested to the
591///   tenant's `group_nesting_depth`, cycle-safe, at most `1 000` groups — over
592///   the service-bound pooled connection, **never** the user's own bind.
593/// * Map them through the tenant's table and make the user's
594///   **directory-sourced** memberships equal to the result: add what is
595///   missing, remove what is no longer backed. A membership an administrator
596///   made by hand is never added, changed or removed.
597/// * **Fail closed.** A lookup that fails, times out or hits the cap is
598///   `Err(`[`DirectoryAuthError::Unavailable`]`)` and changes **nothing**: the
599///   caller refuses the sign-in rather than keep memberships the directory may
600///   have revoked.
601pub trait DirectoryGroupMapper: Send + Sync {
602    /// Apply the mapping for `user_id`, whose directory entry is `user_dn`
603    /// (the DN the directory returned from the user search).
604    fn apply_for_user<'a>(
605        &'a self,
606        tenant_id: Uuid,
607        user_id: Uuid,
608        user_dn: &'a str,
609    ) -> DirectoryFuture<'a, Result<GroupMappingOutcome, DirectoryAuthError>>;
610
611    /// Remove **every directory-sourced membership** of `user_id`, asking the
612    /// directory nothing (T23.3.5, D-31): what the sync job does for an account
613    /// it deactivates, whose entry has vanished or been disabled and so backs
614    /// no group. Memberships an administrator made by hand are never touched,
615    /// and a changed membership set flushes the decision cache exactly as
616    /// [`Self::apply_for_user`] does.
617    ///
618    /// The default refuses (`Unavailable`), so an implementation that cannot do
619    /// it is never mistaken for one that did.
620    fn remove_directory_memberships<'a>(
621        &'a self,
622        _tenant_id: Uuid,
623        _user_id: Uuid,
624    ) -> DirectoryFuture<'a, Result<GroupMappingOutcome, DirectoryAuthError>> {
625        Box::pin(async { Err(DirectoryAuthError::Unavailable) })
626    }
627}

629/// What `AuthService` holds: a shared, type-erased group mapper.
630pub type SharedDirectoryGroupMapper = std::sync::Arc<dyn DirectoryGroupMapper>;
