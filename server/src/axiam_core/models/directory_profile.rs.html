1//! Cleaning of the values a directory entry contributes to an account
2//! (G-3, T23.3.3 and T23.3.5).
3//!
4//! The directory is another party's system: whoever administers it chooses
5//! every attribute, so a value is bounded, stripped of control and
6//! bidirectional-override characters, and refused when it cannot be made into
7//! what a local account holds, **before** it is stored or shown to anyone.
8//!
9//! Two callers share these rules and must not drift apart: just-in-time
10//! provisioning (`axiam-auth`), which builds a new account from an entry, and
11//! the sync job (`axiam-directory`), which keeps an existing account's
12//! username, email address and display name equal to the entry's. They live in
13//! this layer-0 crate because the second cannot reach the first (layering) and
14//! a copy would be a second definition of "an acceptable username".
15
16use crate::models::directory::DirectoryIdentity;
17
18/// Longest username or display name taken from a directory, in characters.
19pub const MAX_NAME_CHARS: usize = 255;
20/// Longest e-mail address taken from a directory (RFC 5321 §4.5.3.1.3).
21pub const MAX_EMAIL_CHARS: usize = 254;
22
23/// A username or an address: trimmed, non-empty, bounded, and with no control
24/// or whitespace characters at all. Refused rather than repaired — a repaired
25/// name is a different name.
26#[must_use]
27pub fn clean_identifier(raw: &str, max_chars: usize) -> Option<String> {
28    let trimmed = raw.trim();
29    let valid = !trimmed.is_empty()
30        && trimmed.chars().count() <= max_chars
31        && !trimmed
32            .chars()
33            .any(|c| c.is_control() || c.is_whitespace() || is_bidi_control(c));
34    valid.then(|| trimmed.to_string())
35}
36
37/// One `@`, something on each side, a dot in the domain.
38#[must_use]
39pub fn plausible_email(email: &str) -> bool {
40    let mut parts = email.splitn(2, '@');
41    let local = parts.next().unwrap_or("");
42    let domain = parts.next().unwrap_or("");
43    !local.is_empty() && domain.contains('.') && !domain.starts_with('.') && !domain.contains('@')
44}

46/// A display name: control and bidirectional-override characters removed (a
47/// right-to-left override makes a name render as something else), whitespace
48/// runs collapsed, bounded. Dropped when nothing is left.
49#[must_use]
50pub fn clean_display_name(raw: &str) -> Option<String> {
51    let cleaned: String = raw
52        .chars()
53        .filter(|c| !is_bidi_control(*c) && (!c.is_control() || c.is_whitespace()))
54        .map(|c| if c.is_whitespace() { ' ' } else { c })
55        .collect();
56    let collapsed = cleaned.split_whitespace().collect::<Vec<_>>().join(" ");
57    if collapsed.is_empty() {
58        return None;
59    }
60    Some(collapsed.chars().take(MAX_NAME_CHARS).collect())
61}
62
63/// The Unicode directional formatting characters (UAX #9): embeddings,
64/// overrides, isolates and the marks.
65#[must_use]
66pub fn is_bidi_control(c: char) -> bool {
67    matches!(
68        c,
69        '\u{200E}' | '\u{200F}' | '\u{202A}'..='\u{202E}' | '\u{2066}'..='\u{2069}'
70    )
71}

73/// The cleaned attributes of an entry, **each independently optional**: an
74/// attribute the entry does not carry, or carries in a form that cannot be
75/// cleaned, is `None`, and the sync job leaves the account's value alone
76/// (T23.3.5). Provisioning, which needs a username and an address, has its own
77/// stricter constructor in `axiam-auth`.
78#[derive(Debug, Clone, PartialEq, Eq, Default)]
79pub struct CleanedAttributes {
80    /// The entry's username attribute, cleaned.
81    pub username: Option<String>,
82    /// The entry's e-mail address, cleaned and plausible.
83    pub email: Option<String>,
84    /// The entry's display name, cleaned.
85    pub display_name: Option<String>,
86}
87
88impl CleanedAttributes {
89    /// Clean what `identity` carries.
90    #[must_use]
91    pub fn from_identity(identity: &DirectoryIdentity) -> Self {
92        Self {
93            username: identity
94                .username
95                .as_deref()
96                .and_then(|raw| clean_identifier(raw, MAX_NAME_CHARS)),
97            email: identity
98                .email
99                .as_deref()
100                .and_then(|raw| clean_identifier(raw, MAX_EMAIL_CHARS))
101                .filter(|email| plausible_email(email)),
102            display_name: identity
103                .display_name
104                .as_deref()
105                .and_then(clean_display_name),
106        }
107    }
108}

110#[cfg(test)]
111mod tests {
112    use super::*;
113
114    fn identity(username: &str, email: &str, display: &str) -> DirectoryIdentity {
115        DirectoryIdentity {
116            external_id: "id".into(),
117            dn: "uid=x,dc=example,dc=com".into(),
118            username: Some(username.into()),
119            email: Some(email.into()),
120            display_name: Some(display.into()),
121        }
122    }
123
124    #[test]
125    fn an_ordinary_entry_is_taken_as_it_is() {
126        let cleaned =
127            CleanedAttributes::from_identity(&identity("alice", "alice@example.com", "Alice A"));
128        assert_eq!(cleaned.username.as_deref(), Some("alice"));
129        assert_eq!(cleaned.email.as_deref(), Some("alice@example.com"));
130        assert_eq!(cleaned.display_name.as_deref(), Some("Alice A"));
131    }
132
133    #[test]
134    fn a_value_that_cannot_be_cleaned_is_absent_not_repaired() {
135        let cleaned = CleanedAttributes::from_identity(&identity(
136            "ali ce",
137            "not-an-address",
138            "\u{202E}\u{200F}",
139        ));
140        assert_eq!(cleaned, CleanedAttributes::default());
141    }
142
143    #[test]
144    fn a_display_name_loses_its_overrides_and_collapses_whitespace() {
145        assert_eq!(
146            clean_display_name("  Al\u{202E}ice \n  A\u{0007}  ").as_deref(),
147            Some("Alice A")
148        );
149        assert_eq!(
150            clean_display_name(&"x".repeat(400)).map(|s| s.len()),
151            Some(255)
152        );
153    }
154
155    #[test]
156    fn an_overlong_identifier_is_refused() {
157        assert!(clean_identifier(&"a".repeat(MAX_NAME_CHARS + 1), MAX_NAME_CHARS).is_none());
158        assert!(clean_identifier(&"a".repeat(MAX_NAME_CHARS), MAX_NAME_CHARS).is_some());
159    }
160}
