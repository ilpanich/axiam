1//! Mapping an external identity provider's claims onto an AXIAM user.
2//!
3//! This module is the fix for the defect where `attribute_map` was collected by
4//! the admin UI, stored by the database, and read by nothing: `provision_new_user`
5//! took the username and email straight off the `email` claim and the configured
6//! map was inert. An operator could write a mapping, save it, see it come back
7//! on the next page load, and have it change nothing.
8//!
9//! It is deliberately **one** mechanism for all three protocols. OIDC ID-token
10//! claims, SAML assertion attributes and an OAuth2 userinfo response all arrive
11//! here as a `serde_json::Value` object and leave as a [`MappedIdentity`]. A
12//! claim-mapping bug should be fixable in one place, and an operator should not
13//! have to learn two syntaxes because AXIAM happened to implement two protocols
14//! in different years.
15//!
16//! See `claude_dev/federation-sso-login-design.md` §8.
17
18use serde_json::Value;

20use super::federation::ProviderKind;

22/// The AXIAM-side fields an attribute map may set.
23///
24/// A key outside this list is a **validation error** at write time, not a
25/// silently ignored one. The whole defect being closed here is a field that
26/// looked configured and did nothing; accepting `"e-mail": "mail"` and quietly
27/// dropping it would recreate that failure one typo down.
28pub const ATTRIBUTE_MAP_FIELDS: &[&str] = &[
29    "external_subject",
30    "username",
31    "email",
32    "email_verified",
33    "display_name",
34];
35
36/// Keys AXIAM used to accept, and the field each one now means, **in
37/// precedence order**.
38///
39/// The SAML path shipped before this module existed and read `"name"` (falling
40/// back to `"displayName"`) as the display name. Refusing those keys outright
41/// would turn an existing, working SAML config into one that cannot be saved —
42/// an operator opening the edit dialog to change something unrelated would be
43/// told their attribute map is invalid. They are accepted, canonicalised, and
44/// deliberately absent from [`ATTRIBUTE_MAP_FIELDS`] so the admin UI offers only
45/// the current names to anyone writing a new one.
46///
47/// Order is load-bearing: the SAML path read `name` *before* `displayName`, and
48/// a config that sets both must keep resolving to the same attribute it did
49/// before this table existed.
50pub const ATTRIBUTE_MAP_ALIASES: &[(&str, &str)] =
51    &[("name", "display_name"), ("displayName", "display_name")];
52
53/// The AXIAM field a map key names, accepting the deprecated aliases.
54pub fn canonical_field(key: &str) -> Option<&'static str> {
55    if let Some(f) = ATTRIBUTE_MAP_FIELDS.iter().find(|f| **f == key) {
56        return Some(f);
57    }
58    ATTRIBUTE_MAP_ALIASES
59        .iter()
60        .find(|(alias, _)| *alias == key)
61        .map(|(_, field)| *field)
62}
63
64/// Prefix marking a mapping value as a literal rather than a claim path.
65///
66/// `{"email_verified": "@true"}` states "I accept this provider's unflagged
67/// email as verified" in a place that shows up in an audit diff, rather than
68/// AXIAM making that decision silently on the operator's behalf.
69pub const LITERAL_PREFIX: char = '@';
70
71/// Bound on the number of entries in an attribute map.
72///
73/// Not tidiness: the map is walked once per federated login, so an unbounded
74/// map is an unbounded per-login cost an admin can set. Five recognised fields
75/// makes anything past a handful a mistake anyway.
76pub const MAX_ATTRIBUTE_MAP_ENTRIES: usize = 32;
77
78/// Why a submitted `attribute_map` was refused.
79#[derive(Debug, Clone, PartialEq, Eq)]
80pub enum AttributeMapError {
81    /// The map was not a JSON object.
82    NotAnObject,
83    /// The map names an AXIAM field that does not exist.
84    UnknownField(String),
85    /// A value was not a string (paths and literals are both strings).
86    NonStringValue(String),
87    /// A value was empty, or a literal with nothing after the `@`.
88    EmptyValue(String),
89    /// Too many entries.
90    TooManyEntries(usize),
91}
92
93impl std::fmt::Display for AttributeMapError {
94    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
95        match self {
96            Self::NotAnObject => write!(f, "attribute_map must be a JSON object"),
97            Self::UnknownField(k) => write!(
98                f,
99                "attribute_map key '{k}' is not an AXIAM user field; the recognised \
100                 keys are {}",
101                ATTRIBUTE_MAP_FIELDS.join(", ")
102            ),
103            Self::NonStringValue(k) => write!(
104                f,
105                "attribute_map entry '{k}' must be a string — either a claim path \
106                 like 'user.email' or a literal like '@true'"
107            ),
108            Self::EmptyValue(k) => {
109                write!(f, "attribute_map entry '{k}' is empty; remove it instead")
110            }
111            Self::TooManyEntries(n) => write!(
112                f,
113                "attribute_map has {n} entries; the maximum is {MAX_ATTRIBUTE_MAP_ENTRIES}"
114            ),
115        }
116    }
117}

119/// Reject an attribute map that cannot be applied.
120///
121/// An absent map (`null`) and an empty object are both fine and both mean
122/// "use the per-kind defaults".
123pub fn validate_attribute_map(map: &Value) -> Result<(), AttributeMapError> {
124    if map.is_null() {
125        return Ok(());
126    }
127    let Some(obj) = map.as_object() else {
128        return Err(AttributeMapError::NotAnObject);
129    };
130    if obj.len() > MAX_ATTRIBUTE_MAP_ENTRIES {
131        return Err(AttributeMapError::TooManyEntries(obj.len()));
132    }
133    for (key, value) in obj {
134        if canonical_field(key).is_none() {
135            return Err(AttributeMapError::UnknownField(key.clone()));
136        }
137        let Some(raw) = value.as_str() else {
138            return Err(AttributeMapError::NonStringValue(key.clone()));
139        };
140        let trimmed = raw.trim();
141        if trimmed.is_empty() || trimmed == LITERAL_PREFIX.to_string() {
142            return Err(AttributeMapError::EmptyValue(key.clone()));
143        }
144    }
145    Ok(())
146}

148/// The per-kind default mapping, as `(axiam_field, claim_path)` pairs.
149///
150/// The OIDC rows reproduce `provision_new_user`'s pre-existing behaviour
151/// exactly — username from `email`, email from `email` — so an existing Google
152/// config behaves identically before and after this change. That is asserted by
153/// a test, not merely intended.
154pub fn default_attribute_map(kind: ProviderKind) -> &'static [(&'static str, &'static str)] {
155    match kind {
156        ProviderKind::Github => &[
157            // GitHub's numeric `id` is the stable subject; `login` is renameable
158            // and must never be the identity key.
159            ("external_subject", "id"),
160            ("username", "login"),
161            // Injected by the GitHub path from `GET /user/emails` — see
162            // `MappedIdentity`'s doc comment.
163            ("email", "email"),
164            ("email_verified", "email_verified"),
165            ("display_name", "name"),
166        ],
167        ProviderKind::Facebook => &[
168            ("external_subject", "id"),
169            ("username", "email"),
170            ("email", "email"),
171            // Deliberately no `email_verified` entry: the Graph API carries no
172            // verification flag, so absent reads as unverified and the login is
173            // refused unless the operator writes `"email_verified": "@true"`.
174            ("display_name", "name"),
175        ],
176        ProviderKind::GenericSaml => &[
177            ("external_subject", "sub"),
178            ("username", "email"),
179            ("email", "email"),
180            ("display_name", "displayName"),
181        ],
182        // Google, Microsoft, Apple, generic OIDC and generic OAuth2 all use the
183        // standard OIDC claim names.
184        _ => &[
185            ("external_subject", "sub"),
186            ("username", "email"),
187            ("email", "email"),
188            ("email_verified", "email_verified"),
189            ("display_name", "name"),
190        ],
191    }
192}

194/// What an external identity resolved to, before any AXIAM user exists.
195///
196/// `email` and `email_verified` travel together on purpose. An email the
197/// provider has not affirmatively marked verified is never adopted as an AXIAM
198/// identity: AXIAM keys account recovery, verification and administrative
199/// notification on the address, and adopting an unverified one is account
200/// takeover by whoever typed it into the provider first.
201///
202/// For GitHub, `GET /user` frequently returns a null or unverified `email`, so
203/// the GitHub path resolves the primary **verified** address from
204/// `GET /user/emails` and writes it into the userinfo object under the ordinary
205/// `email` / `email_verified` keys before mapping runs. That keeps one syntax:
206/// a default map and an operator's custom map both name `email`.
207#[derive(Debug, Clone, PartialEq, Eq)]
208pub struct MappedIdentity {
209    /// The external subject identifier. Without it there is no identity, which
210    /// is why [`map_identity`] returns `None` rather than a partial result.
211    pub external_subject: String,
212    /// Preferred username, if the provider supplied one.
213    pub username: Option<String>,
214    /// Email address, if the provider supplied one.
215    pub email: Option<String>,
216    /// Whether the provider affirmatively said the address is verified.
217    /// Absent, null or falsey all read as `false`.
218    pub email_verified: bool,
219    /// Human-readable name, if supplied. Stored in the user's `metadata` —
220    /// `User` has no column for it.
221    pub display_name: Option<String>,
222}
223
224/// Resolve one AXIAM field from the map (or the per-kind default), against a
225/// claims object.
226///
227/// Returns `None` when the field has no mapping, when the path is absent from
228/// the claims, or when the value present is not something a string can be made
229/// of (an array or an object).
230pub fn resolve_field(
231    kind: ProviderKind,
232    attribute_map: &Value,
233    claims: &Value,
234    field: &str,
235) -> Option<String> {
236    // The canonical name first, then each deprecated alias in the table's own
237    // order. Iterating the JSON object instead would make precedence depend on
238    // key ordering — which for `serde_json`'s default map is alphabetical, and
239    // would silently flip `name` and `displayName` for every SAML config that
240    // sets both.
241    let configured = std::iter::once(field)
242        .chain(
243            ATTRIBUTE_MAP_ALIASES
244                .iter()
245                .filter(|(_, f)| *f == field)
246                .map(|(alias, _)| *alias),
247        )
248        .find_map(|key| attribute_map.get(key).and_then(Value::as_str))
249        .map(str::trim)
250        .filter(|s| !s.is_empty());
251
252    let path = match configured {
253        Some(p) => p,
254        None => default_attribute_map(kind)
255            .iter()
256            .find(|(f, _)| *f == field)
257            .map(|(_, p)| *p)?,
258    };
259
260    if let Some(literal) = path.strip_prefix(LITERAL_PREFIX) {
261        let literal = literal.trim();
262        return (!literal.is_empty()).then(|| literal.to_string());
263    }
264
265    lookup_path(claims, path).and_then(stringify_claim)
266}

268/// Look a claim path up in a claims object.
269///
270/// The **whole path is tried as a single key first**, and that ordering is
271/// load-bearing rather than an optimisation: SAML attribute names are routinely
272/// OIDs like `urn:oid:0.9.2342.19200300.100.1.3`, which contain dots and are not
273/// nested paths. Only if no such key exists is the value split on `.` and walked
274/// as a path, which is what makes `user.email` work for a nested userinfo
275/// response.
276pub fn lookup_path<'a>(claims: &'a Value, path: &str) -> Option<&'a Value> {
277    if let Some(direct) = claims.get(path) {
278        return Some(direct);
279    }
280    if !path.contains('.') {
281        return None;
282    }
283    let mut cursor = claims;
284    for segment in path.split('.') {
285        cursor = cursor.get(segment)?;
286    }
287    Some(cursor)
288}

290/// Coerce a claim value to a string.
291///
292/// Strings, numbers and booleans convert; arrays, objects and null do not.
293/// GitHub's `id` is a JSON number and is a subject identifier, so numbers must
294/// convert or GitHub has no `external_subject`.
295fn stringify_claim(value: &Value) -> Option<String> {
296    match value {
297        Value::String(s) => {
298            let t = s.trim();
299            (!t.is_empty()).then(|| t.to_string())
300        }
301        Value::Number(n) => Some(n.to_string()),
302        Value::Bool(b) => Some(b.to_string()),
303        Value::Null | Value::Array(_) | Value::Object(_) => None,
304    }
305}
306
307/// Whether a resolved `email_verified` value means "verified".
308///
309/// Deliberately narrow. `true`, `"true"` and `1` are affirmative; everything
310/// else — including absent, `null`, `"unknown"` and `0` — is not. A provider
311/// that does not say yes has not said yes.
312fn is_affirmative(raw: &str) -> bool {
313    matches!(raw.trim().to_ascii_lowercase().as_str(), "true" | "1")
314}
315
316/// Map a provider's claims onto an AXIAM identity.
317///
318/// Returns `None` when no `external_subject` can be resolved. There is no
319/// partial success here: an identity with no subject is not an identity, and
320/// falling back to some other field would silently key the account on something
321/// the provider is free to let the user change.
322pub fn map_identity(
323    kind: ProviderKind,
324    attribute_map: &Value,
325    claims: &Value,
326) -> Option<MappedIdentity> {
327    let external_subject = resolve_field(kind, attribute_map, claims, "external_subject")?;
328    Some(MappedIdentity {
329        external_subject,
330        username: resolve_field(kind, attribute_map, claims, "username"),
331        email: resolve_field(kind, attribute_map, claims, "email"),
332        email_verified: resolve_field(kind, attribute_map, claims, "email_verified")
333            .as_deref()
334            .is_some_and(is_affirmative),
335        display_name: resolve_field(kind, attribute_map, claims, "display_name"),
336    })
337}
