1//! FIDO Alliance Metadata Service (MDS3) domain models (X3, D1/D5/D8).
2//!
3//! These types are the *parsed, storage-shaped* view of a verified MDS3
4//! BLOB — the cryptographic verification and JWT/JSON parsing that produces
5//! them live in `axiam-pki::mds` (PKI owns trust-anchor concerns), but the
6//! resulting domain types live here per the repo's convention that
7//! `axiam-core` holds every model a `Repository` trait moves in or out of
8//! storage.
9
10use serde::{Deserialize, Serialize};
11use uuid::Uuid;

13/// FIDO certification level, as recorded in an MDS `statusReports` entry's
14/// `FIDO_CERTIFIED*` status.
15///
16/// Variant order is significant: `derive(PartialOrd, Ord)` gives `L1 < L1Plus
17/// < L2 < L2Plus < L3 < L3Plus`, which `WebauthnAttestationPolicy::evaluate`
18/// (D8 step 9) relies on directly for the `min_certification` boundary
19/// check (`entry_level >= policy_min`).
20#[derive(
21    Debug,
22    Clone,
23    Copy,
24    PartialEq,
25    Eq,
26    PartialOrd,
27    Ord,
28    Hash,
29    Serialize,
30    Deserialize,
31    utoipa::ToSchema,
32)]
33pub enum CertificationLevel {
34    L1,
35    L1Plus,
36    L2,
37    L2Plus,
38    L3,
39    L3Plus,
40}
41
42/// The `status` field of a FIDO MDS `StatusReport`
43/// (<https://fidoalliance.org/metadata/metadata-schema/#authenticatorstatus-enum>).
44///
45/// `#[serde(rename_all = "SCREAMING_SNAKE_CASE")]` maps every unit variant
46/// onto the exact wire string the MDS schema uses (with three explicit
47/// renames for the `L1plus`/`L2plus`/`L3plus` variants, which are not
48/// SCREAMING_SNAKE — the schema itself is inconsistent here). `Other` is a
49/// deliberate, fail-safe catch-all (`#[serde(other)]`) for status values not
50/// in the schema at the time this was written: an entry whose status we
51/// don't recognize is treated as neither FIDO-certified nor
52/// compromised/revoked by every helper below, rather than erroring the whole
53/// ingestion run.
54#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, utoipa::ToSchema)]
55#[serde(rename_all = "SCREAMING_SNAKE_CASE")]
56pub enum MdsAuthenticatorStatus {
57    NotFidoCertified,
58    FidoCertified,
59    UserVerificationBypass,
60    AttestationKeyCompromise,
61    UserKeyRemoteCompromise,
62    UserKeyPhysicalCompromise,
63    UpdateAvailable,
64    Revoked,
65    SelfAssertionSubmitted,
66    #[serde(rename = "FIDO_CERTIFIED_L1")]
67    FidoCertifiedL1,
68    #[serde(rename = "FIDO_CERTIFIED_L1plus")]
69    FidoCertifiedL1Plus,
70    #[serde(rename = "FIDO_CERTIFIED_L2")]
71    FidoCertifiedL2,
72    #[serde(rename = "FIDO_CERTIFIED_L2plus")]
73    FidoCertifiedL2Plus,
74    #[serde(rename = "FIDO_CERTIFIED_L3")]
75    FidoCertifiedL3,
76    #[serde(rename = "FIDO_CERTIFIED_L3plus")]
77    FidoCertifiedL3Plus,
78    /// Fail-safe catch-all for any status string not covered above.
79    #[serde(other)]
80    Other,
81}
82
83impl MdsAuthenticatorStatus {
84    /// `true` for the compromise/revocation statuses D8 step 7 treats as
85    /// **sticky** — any occurrence anywhere in an entry's `statusReports`
86    /// history counts, regardless of a later benign report.
87    pub fn is_compromise_or_revoked(self) -> bool {
88        matches!(
89            self,
90            Self::Revoked
91                | Self::UserKeyPhysicalCompromise
92                | Self::UserKeyRemoteCompromise
93                | Self::AttestationKeyCompromise
94        )
95    }
96
97    /// The `CertificationLevel` this status represents, if it is one of the
98    /// `FIDO_CERTIFIED*` family. Bare `FIDO_CERTIFIED` (no level suffix) is
99    /// FIDO's own Level 1 designation, so it maps to `CertificationLevel::L1`.
100    pub fn certification_level(self) -> Option<CertificationLevel> {
101        match self {
102            Self::FidoCertified | Self::FidoCertifiedL1 => Some(CertificationLevel::L1),
103            Self::FidoCertifiedL1Plus => Some(CertificationLevel::L1Plus),
104            Self::FidoCertifiedL2 => Some(CertificationLevel::L2),
105            Self::FidoCertifiedL2Plus => Some(CertificationLevel::L2Plus),
106            Self::FidoCertifiedL3 => Some(CertificationLevel::L3),
107            Self::FidoCertifiedL3Plus => Some(CertificationLevel::L3Plus),
108            _ => None,
109        }
110    }
111
112    /// `true` for any `FIDO_CERTIFIED*` status (any level, including the
113    /// bare `FIDO_CERTIFIED`).
114    pub fn is_fido_certified(self) -> bool {
115        self.certification_level().is_some()
116    }
117}

119/// One entry in an MDS `StatusReport` array — the certification/compromise
120/// history of an authenticator model.
121///
122/// `rename_all = "camelCase"` makes this deserialize directly from the raw
123/// FIDO MDS3 wire JSON (`effectiveDate`, `certificationDescriptor`) — the
124/// same struct doubles as both the storage-shaped domain type and the raw
125/// wire type, since its shape happens to already match.
126#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
127#[serde(rename_all = "camelCase")]
128pub struct MdsStatusReport {
129    pub status: MdsAuthenticatorStatus,
130    /// ISO 8601 date (`YYYY-MM-DD`) the status took effect, kept as the raw
131    /// MDS string rather than parsed — it is display/audit metadata only,
132    /// never compared against `now`.
133    pub effective_date: Option<String>,
134    /// Free-text certification descriptor (e.g. a lab report reference),
135    /// present on some `FIDO_CERTIFIED*` reports.
136    pub certification_descriptor: Option<String>,
137}
138
139/// A parsed FIDO MDS3 metadata entry, keyed by AAGUID.
140///
141/// UAF/U2F entries (keyed by `aaid`/`attestationCertificateKeyIdentifiers`
142/// instead of `aaguid`) never reach this type — `axiam-pki::mds` skips them
143/// during payload parsing and logs the skipped count (D1).
144#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
145pub struct MdsEntry {
146    pub aaguid: Uuid,
147    /// `metadataStatement.description` — human-readable authenticator model
148    /// name, captured at registration time as `authenticator_name` (D6).
149    pub description: Option<String>,
150    /// `metadataStatement.attestationRootCertificates` (base64-encoded DER,
151    /// as MDS ships them) — the seed material for the per-tenant
152    /// `AttestationCaList` webauthn-rs verifies registrations against (D7).
153    #[serde(default)]
154    pub attestation_root_certificates: Vec<String>,
155    #[serde(default)]
156    pub status_reports: Vec<MdsStatusReport>,
157    /// `timeOfLastStatusChange`, kept as the raw MDS date string (display
158    /// metadata only, same rationale as `MdsStatusReport::effective_date`).
159    pub time_of_last_status_change: Option<String>,
160}
161
162impl MdsEntry {
163    /// D8 step 7: sticky compromise/revocation check across the *entire*
164    /// status history, not just the latest report.
165    pub fn is_compromised_or_revoked(&self) -> bool {
166        self.status_reports
167            .iter()
168            .any(|r| r.status.is_compromise_or_revoked())
169    }
170
171    /// D8 step 8: `true` if any status report is a `FIDO_CERTIFIED*` status
172    /// (any level).
173    pub fn is_fido_certified(&self) -> bool {
174        self.status_reports
175            .iter()
176            .any(|r| r.status.is_fido_certified())
177    }
178
179    /// D8 step 9: the **highest** `FIDO_CERTIFIED*` level appearing anywhere
180    /// in the status history, or `None` if the entry was never certified.
181    pub fn highest_certification_level(&self) -> Option<CertificationLevel> {
182        self.status_reports
183            .iter()
184            .filter_map(|r| r.status.certification_level())
185            .max()
186    }
187}

189/// Metadata about the last successfully verified MDS3 BLOB (server-global,
190/// single row — D10).
191#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
192pub struct MdsBlobMeta {
193    /// The BLOB's `no` (serial number) claim — monotonically increasing;
194    /// drives rollback protection (D4 step 8).
195    pub no: i64,
196    /// The BLOB's `nextUpdate` claim (`YYYY-MM-DD`).
197    pub next_update: chrono::NaiveDate,
198    pub entry_count: u64,
199    pub last_refreshed_at: chrono::DateTime<chrono::Utc>,
200    /// `true` when `next_update` is in the past as of the last refresh
201    /// (D4 step 9). Staleness never hard-fails ingestion — it is logged at
202    /// WARN and left for policy/ops to act on.
203    pub stale: bool,
204}
