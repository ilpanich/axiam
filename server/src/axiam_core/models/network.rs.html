1//! Network segments, their reply profiles and the tenant's session cap (RADIUS
2//! D4, `claude_dev/radius-reply-profile-design.md` §3.5 and §4).
3//!
4//! # Segments are resources
5//!
6//! A segment is a node of the existing resource tree, in a **reserved
7//! subtree**, so that RBAC, the hierarchy cascade and deny-override apply to it
8//! unchanged (§4.1, maintainer decision D-5):
9//!
10//! ```text
11//! /network                       network_root     (one per tenant, made on first use)
12//! ├── /network/zones             network_branch   (kind = access)
13//! │   └── /network/zones/hq      network_site     site_key = "hq"
14//! │       └── …/hq/corp          network_segment  vlan 10, rank 10
15//! └── /network/quarantine        network_branch   (kind = quarantine)
16//!     └── /network/quarantine/hq network_site
17//!         └── …/hq/q             network_segment  vlan 999
18//! ```
19//!
20//! Quarantine is a **sibling branch**, never a descendant of `zones`:
21//! deny-override cannot express "deny the subtree except this leaf", so a
22//! quarantine segment under `zones` would be denied by the very rule that
23//! quarantines. The fixed two-branch shape makes the correct placement the only
24//! placement, and [`NetworkSegmentPath`] refuses every other.
25//!
26//! The segment's own row (a *sidecar* of its `resource` row, sharing its id)
27//! holds what the tree cannot: the kind, the VLAN, the `Filter-Id` names the
28//! segment's profile may choose from (its **envelope**) and the `rank` that
29//! orders placement.
30//!
31//! # Two writers, one version
32//!
33//! Segments, profiles and the tenant policy are written conditionally on the
34//! `version` read (plan §10, the two-writer rule). A profile write is also
35//! conditional on the **owning segment's** version, so a concurrent change to
36//! its `filter_ids` cannot slip a profile past an envelope that no longer holds.
37//!
38//! # What this module is not
39//!
40//! Who may write any of this (the H1 to H6 rule, §5.2) is the authorization
41//! function's, not the repository's. The repository refuses only what it can
42//! decide from the data: a path outside `/network`, a value outside the
43//! dictionary, a `filter_id` outside the envelope.
44

126/// A segment's place in the reserved subtree, parsed from
127/// `/network/{zones|quarantine}/<site>/<segment>`.
128#[derive(Debug, Clone, PartialEq, Eq)]
129pub struct NetworkSegmentPath {
130    /// The kind the branch implies.
131    pub kind: NetworkSegmentKind,
132    /// The site key. A site is a key, not a location: VLAN numbers are
133    /// per-site facts. A single-site tenant uses `default`.
134    pub site_key: String,
135    /// The segment's name, unique within its site.
136    pub name: String,
137}
138
139impl NetworkSegmentPath {
140    /// Parse a segment path. Anything outside `/network`, anything but the
141    /// four-level shape, and any component that is not a plain identifier
142    /// (`^[A-Za-z0-9][A-Za-z0-9._-]{0,62}$`) is `Validation`.
143    pub fn parse(path: &str) -> AxiamResult<Self> {
144        let outside = || AxiamError::Validation {
145            message: "a network segment lives at /network/{zones|quarantine}/<site>/<segment>"
146                .into(),
147        };
148        let rest = path.strip_prefix('/').ok_or_else(outside)?;
149        let parts: Vec<&str> = rest.split('/').collect();
150        let [root, branch, site, name] = parts[..] else {
151            return Err(outside());
152        };
153        if root != NETWORK_ROOT_NAME {
154            return Err(outside());
155        }
156        let kind = match branch {
157            ZONES_BRANCH => NetworkSegmentKind::Access,
158            QUARANTINE_BRANCH => NetworkSegmentKind::Quarantine,
159            _ => return Err(outside()),
160        };
161        if !is_valid_identifier(site) || !is_valid_identifier(name) {
162            return Err(AxiamError::Validation {
163                message: "a site key and a segment name are letters, digits, '.', '_' and '-', \
164                          at most 63 characters, starting with a letter or digit"
165                    .into(),
166            });
167        }
168        Ok(Self {
169            kind,
170            site_key: site.to_owned(),
171            name: name.to_owned(),
172        })
173    }
174
175    /// The path in its written form.
176    #[must_use]
177    pub fn render(&self) -> String {
178        render_path(self.kind, &self.site_key, &self.name)
179    }
180}

182/// `/network/<branch of kind>/<site>/<name>`.
183#[must_use]
184pub fn render_path(kind: NetworkSegmentKind, site_key: &str, name: &str) -> String {
185    format!("/{NETWORK_ROOT_NAME}/{}/{site_key}/{name}", kind.branch())
186}

276/// Check the data of a segment write that does not depend on the datastore.
277///
278/// `kind` is the segment's kind: access needs a rank, quarantine none.
279pub fn validate_segment_fields(
280    kind: NetworkSegmentKind,
281    vlan_id: u16,
282    filter_ids: &[String],
283    rank: Option<u32>,
284) -> AxiamResult<()> {
285    let invalid = |message: &str| AxiamError::Validation {
286        message: message.into(),
287    };
288    if !is_valid_vlan(vlan_id) {
289        return Err(invalid("vlan_id must be between 1 and 4094"));
290    }
291    if filter_ids.len() > MAX_FILTER_IDS {
292        return Err(invalid("a segment declares at most 32 filter_ids"));
293    }
294    for (i, id) in filter_ids.iter().enumerate() {
295        if !is_valid_identifier(id) {
296            return Err(invalid("a filter_id is not a valid Filter-Id name"));
297        }
298        if filter_ids[..i].contains(id) {
299            return Err(invalid("filter_ids must not repeat a name"));
300        }
301    }
302    match (kind, rank) {
303        (NetworkSegmentKind::Access, None) => Err(invalid("an access segment needs a rank")),
304        (NetworkSegmentKind::Quarantine, Some(_)) => {
305            Err(invalid("a quarantine segment has no rank"))
306        }
307        _ => Ok(()),
308    }
309}

408/// Which way a write moves the `Session-Timeout` cap.
409///
410/// Lowering only narrows (reads clamp), so it is always allowed. Raising widens
411/// what a profile writer may set from then on, and is allowed and audited as
412/// such (plan section 13.6): the audit action says which it was.
413#[derive(Debug, Clone, Copy, PartialEq, Eq)]
414pub enum CapChange {
415    /// The new cap is above the current one.
416    Raised,
417    /// The new cap is below the current one.
418    Lowered,
419    /// The new cap equals the current one (a write that only bumps the version).
420    Unchanged,
421}
422
423impl CapChange {
424    /// The direction from `before` to `after`.
425    #[must_use]
426    pub const fn between(before: u32, after: u32) -> Self {
427        if after > before {
428            Self::Raised
429        } else if after < before {
430            Self::Lowered
431        } else {
432            Self::Unchanged
433        }
434    }
435
436    /// `raised`, `lowered` or `unchanged`: the word the audit row carries.
437    #[must_use]
438    pub const fn as_str(self) -> &'static str {
439        match self {
440            Self::Raised => "raised",
441            Self::Lowered => "lowered",
442            Self::Unchanged => "unchanged",
443        }
444    }
445}

461impl SetNetworkPolicy {
462    /// Refuse a cap outside the settable range.
463    pub fn validate(&self) -> AxiamResult<()> {

465            .contains(&self.session_timeout_max_secs)

471                ),
472            });
473        }
474        Ok(())
475    }

512/// Why a stored profile may no longer be rendered: the internal rule, for the
513/// `reply_profile.invalid` audit row (design note §5.4). Never on the wire.
514#[derive(Debug, Clone, PartialEq, Eq)]
515pub struct ProfileDefect {
516    /// The rule the row fails: `3.4:segment` (the owning segment's own row is
517    /// outside its bounds), `3.4:dictionary` (an unknown dictionary version),
518    /// `3.3:admin-class` (an administrative `Service-Type`), `5.2:H3-envelope`
519    /// (a `Filter-Id` the segment no longer declares) or `3.4:value` (any other
520    /// value outside the dictionary's bounds).
521    pub rule: &'static str,
522    /// What was wrong, in words.
523    pub reason: String,
524}
525
526impl ProfileDefect {
527    fn new(rule: &'static str, reason: impl Into<String>) -> Self {
528        Self {
529            rule,
530            reason: reason.into(),
531        }
532    }
533}

535impl StoredReplyProfile {
536    /// Re-validate the row against its dictionary, the owning `segment`'s
537    /// envelope and the tenant's current `cap`, and render the reply it would
538    /// produce (design note §3.4).
539    ///
540    /// A value above the cap is clamped down, which only narrows; every other
541    /// failure is a [`ProfileDefect`] and the segment is **not a candidate**.
542    /// The cap is applied by clamping, not by refusing, because lowering it is
543    /// allowed below stored profiles (§5.2); the absolute ceiling
544    /// ([`SESSION_TIMEOUT_CAP_CEILING_SECS`]) is a refusal.
545    pub fn revalidate(
546        &self,
547        segment: &NetworkSegment,
548        cap: u32,
549    ) -> Result<(EffectiveReplyProfile, RenderedReply), ProfileDefect> {
550        if segment.id != self.segment_id || segment.tenant_id != self.tenant_id {
551            return Err(ProfileDefect::new(
552                "3.4:segment",
553                "the profile is not the segment's",
554            ));
555        }
556        validate_segment_fields(
557            segment.kind,
558            segment.vlan_id,
559            &segment.filter_ids,
560            segment.rank,
561        )
562        .map_err(|e| ProfileDefect::new("3.4:segment", e.to_string()))?;
563        if dictionary(&self.dictionary).is_none() {
564            return Err(ProfileDefect::new(
565                "3.4:dictionary",
566                "unknown reply dictionary version",
567            ));
568        }
569        let value = |reason: &str| ProfileDefect::new("3.4:value", reason);
570        let session_timeout_secs = u32::try_from(self.session_timeout_secs)
571            .map_err(|_| value("session_timeout_secs is out of range"))?;
572        let termination_action = TerminationAction::parse(&self.termination_action)
573            .ok_or_else(|| value("unknown termination_action"))?;
574        let idle_timeout_secs = self
575            .idle_timeout_secs
576            .map(u32::try_from)
577            .transpose()
578            .map_err(|_| value("idle_timeout_secs is out of range"))?;
579        let service_type = self
580            .service_type
581            .as_deref()
582            .map(|s| ServiceType::parse(s).ok_or_else(|| value("unknown service_type")))
583            .transpose()?;
584        if service_type.is_some_and(ServiceType::is_admin_class) {
585            return Err(ProfileDefect::new(
586                "3.3:admin-class",
587                "an administrative Service-Type is disabled",
588            ));
589        }
590        if let Some(filter) = &self.filter_id
591            && !segment.filter_ids.iter().any(|f| f == filter)
592        {
593            return Err(ProfileDefect::new(
594                "5.2:H3-envelope",
595                "filter_id is outside the segment's declared filter_ids",
596            ));
597        }
598        let input = ReplyProfileInput {
599            dictionary: self.dictionary.clone(),
600            session_timeout_secs,
601            termination_action,
602            idle_timeout_secs,
603            filter_id: self.filter_id.clone(),
604            service_type,
605        };
606        input
607            .validate(&segment.filter_ids, SESSION_TIMEOUT_CAP_CEILING_SECS)
608            .map_err(|e| ProfileDefect::new("3.4:value", e.to_string()))?;
609
610        let effective_timeout = clamp_session_timeout(session_timeout_secs, cap);
611        let effective = EffectiveReplyProfile {
612            dictionary: input.dictionary,
613            session_timeout_secs: effective_timeout,
614            session_timeout_clamped: effective_timeout != session_timeout_secs,
615            termination_action,
616            idle_timeout_secs: idle_timeout_secs.map(|i| i.min(effective_timeout)),
617            filter_id: input.filter_id,
618            service_type,
619        };
620        let reply = render_reply(segment.vlan_id, &effective)
621            .map_err(|e| ProfileDefect::new("3.4:value", e.to_string()))?;
622        Ok((effective, reply))
623    }
624}

626/// Choose the segment a device is placed on (design note §4.3 steps 2 to 5).
627///
628/// `segments[i]` is a candidate when it is enabled, `joinable[i]` (the
629/// subject's `check_access(network:join, segment)` answered **Allow**, with
630/// deny-override in force) and `usable[i]` (its profile passed re-validation).
631/// Among `access` candidates the lowest `rank` wins (unique per site, so the
632/// choice is decided at write time, not by query order); a `quarantine`
633/// candidate is used only when no `access` candidate remains — which is what
634/// deny-override makes the case for a quarantined device (§4.4), so quarantine
635/// needs no precedence rule of its own. Two quarantine candidates at one site
636/// are ordered by name. `None` is "no candidate": an Access-Reject, with no
637/// fallback segment.
638///
639/// The three slices are read in step; a length mismatch is no candidate.
640#[must_use]
641pub fn select_placement(
642    segments: &[NetworkSegment],
643    joinable: &[bool],
644    usable: &[bool],
645) -> Option<usize> {
646    if segments.len() != joinable.len() || segments.len() != usable.len() {
647        return None;
648    }
649    let candidates = || {
650        segments
651            .iter()
652            .enumerate()
653            .filter(|(i, s)| s.enabled && joinable[*i] && usable[*i])
654    };
655    candidates()
656        .filter(|(_, s)| s.kind == NetworkSegmentKind::Access)
657        .min_by_key(|(_, s)| (s.rank.unwrap_or(u32::MAX), s.name.clone()))
658        .or_else(|| {
659            candidates()
660                .filter(|(_, s)| s.kind == NetworkSegmentKind::Quarantine)
661                .min_by_key(|(_, s)| s.name.clone())
662        })
663        .map(|(i, _)| i)
664}

711/// The reserved permission namespace. `permissions:create` and
712/// `permissions:update` refuse an action in it, so a tenant cannot mint a
713/// look-alike or rename one of the nine.
714pub const NETWORK_PERMISSION_PREFIX: &str = "network:";
715/// May be placed on a segment. Held by a device's service account.
716pub const NETWORK_JOIN: &str = "network:join";
717/// Administrative login to a segment's NAS devices. Defined, not grantable
718/// under option B.
719pub const NETWORK_ADMIN: &str = "network:admin";
720/// Read segments and profiles.
721pub const NETWORK_READ: &str = "network:read";
722/// Write the tree and each segment's envelope.
723///
724/// It also guards the tenant's network policy, the `Session-Timeout` cap
725/// (`network.session_timeout_max_secs`): the write gate asks for it at the
726/// `/network` root when `PUT /api/v1/network/policy` is called, and there is
727/// no separate constant for the cap.
728///
729/// The maintainer's correction of 2026-10-15 (plan section 13.6). The first
730/// reading guarded the cap with `network:admin`, which is administrative login
731/// to a segment's NAS devices and is not grantable under option B (it is in
732/// [`NETWORK_ADMIN_CLASS_ACTIONS`]; the grant gate refuses an allow of it and
733/// the seeder withholds it), so the cap was writable by nobody. The cap is the
734/// `Session-Timeout` envelope every reply profile is clamped to, envelopes are
735/// this permission's ("each segment's envelope"), and H3 keeps envelope writers
736/// apart from profile writers ([`NETWORK_PROFILES_WRITE`]). A holder of the
737/// profile permission alone therefore cannot write the cap.
738///
739/// Unlike `network:admin` it can be held: it is grantable and delegable under
740/// the section 5.3 bound ("you delegate only what you hold", it is in
741/// [`NETWORK_META_ACTIONS`]) and the seeded super-admin holds it
742/// ([`SEEDED_SUPER_ADMIN_NETWORK_ACTIONS`]). Only a grant **at the `/network`
743/// root** counts for the cap: one at a site below it does not, and a deny of it
744/// at the root refuses the write (deny-override).
745pub const NETWORK_SEGMENTS_WRITE: &str = "network:segments_write";
746/// Write a segment's reply profile.
747pub const NETWORK_PROFILES_WRITE: &str = "network:profiles_write";
748/// Confer `network:join` at a node.
749pub const NETWORK_GRANT: &str = "network:grant";
750/// Confer the administrative class. Option A only.
751pub const NETWORK_GRANT_ADMIN: &str = "network:grant_admin";
752/// Ask for placement decisions for a site (D4's caller).
753pub const NETWORK_DECIDE: &str = "network:decide";
754/// Read the rendered export for a site.
755pub const NETWORK_EXPORT: &str = "network:export";
756
757/// The meta permissions of design note §5.3: assigning a role that carries one
758/// of these at a node needs the assigner to hold it there ("you delegate only
759/// what you hold").
760pub const NETWORK_META_ACTIONS: &[&str] = &[
761    NETWORK_GRANT,
762    NETWORK_PROFILES_WRITE,
763    NETWORK_SEGMENTS_WRITE,
764    NETWORK_DECIDE,
765    NETWORK_EXPORT,
766];
767
768/// The administrative class (§3.3, H4): defined so that option A inherits the
769/// rule, and not grantable as an allow under option B.
770pub const NETWORK_ADMIN_CLASS_ACTIONS: &[&str] = &[NETWORK_ADMIN, NETWORK_GRANT_ADMIN];
771
772/// The `network:*` actions the seeded **super-admin** role holds: the meta
773/// permissions and `network:read`, never `network:join` or the administrative
774/// class.
775///
776/// # Why the bootstrap role is exempt from §4.2 rule 1, and only this far
777///
778/// Rule 1 forbids a network-sensitive role to be global. The seeded
779/// super-admin is global, and it is the **one** role that must be: with the
780/// §5.3 bound in force a network administrator can only be made by someone who
781/// already holds `network:grant` (and every meta permission conferred) at the
782/// node, so a tenant whose bootstrap role held none of them could never
783/// administer its network at all. The exemption is narrow:
784///
785/// * it carries no `network:join`, so no device lands on every segment by
786///   holding the super-admin role, which is what rule 1's "global defeats
787///   segment scoping" is about;
788/// * it carries neither `network:admin` nor `network:grant_admin`, which are
789///   not grantable under option B;
790/// * it does **not** bypass §5.3: assigning the super-admin role is a
791///   tenant-wide assignment of a role carrying meta permissions, so the
792///   assigner must hold every one of them at `/network`;
793/// * it is made only by the seeder (`seed_default_roles`, and
794///   `reconcile_default_role_grants` on boot). The API refuses a `network:`
795///   grant to any global role, this one included.
796///
797/// The seeded `admin` and `viewer` roles hold no `network:` permission.
798pub const SEEDED_SUPER_ADMIN_NETWORK_ACTIONS: &[&str] = &[
799    NETWORK_READ,
800    NETWORK_SEGMENTS_WRITE,
801    NETWORK_PROFILES_WRITE,
802    NETWORK_GRANT,
803    NETWORK_DECIDE,
804    NETWORK_EXPORT,
805];
806
807/// Whether `action` is in the reserved `network:` namespace.
808///
809/// Case-insensitive and blind to surrounding whitespace, so that a look-alike
810/// (`Network:join`, ` network:grant`) counts as reserved: the engine matches
811/// actions exactly and such a permission would grant nothing, but it would
812/// read as one of the nine in every listing an administrator looks at.
813#[must_use]
814pub fn is_network_action(action: &str) -> bool {
815    action
816        .trim_start()
817        .get(..NETWORK_PERMISSION_PREFIX.len())
818        .is_some_and(|p| p.eq_ignore_ascii_case(NETWORK_PERMISSION_PREFIX))
819}

825/// Which segment, profile or policy write a [`NetworkWriteClearance`] was issued for.
826#[derive(Debug, Clone, PartialEq, Eq)]
827pub enum NetworkWriteTarget {
828    /// Create the segment at this path.
829    SegmentCreate {
830        /// The path of the segment to create.
831        path: String,
832    },
833    /// Rewrite this segment's envelope.
834    SegmentUpdate {
835        /// The segment.
836        segment_id: Uuid,
837    },
838    /// Delete this segment.
839    SegmentDelete {
840        /// The segment.
841        segment_id: Uuid,
842    },
843    /// Create or replace this segment's reply profile.
844    ProfileWrite {
845        /// The owning segment.
846        segment_id: Uuid,
847    },
848    /// Delete this segment's reply profile.
849    ProfileDelete {
850        /// The owning segment.
851        segment_id: Uuid,
852    },
853    /// Write the tenant's network policy (the `Session-Timeout` cap).
854    PolicyWrite,
855}
856
857/// Proof that the authorization function of design note §5.1 cleared one
858/// segment or profile write.
859///
860/// Every write method of [`crate::repository::NetworkSegmentRepository`] and
861/// [`crate::repository::ReplyProfileRepository`] takes one, and refuses a
862/// clearance issued for another tenant, another actor or another write. That
863/// makes "a write reaches the repository without the function" a **compile
864/// error** rather than a review finding (the T-429 lesson).
865///
866/// The constructor is public because the function lives in `axiam-authz`,
867/// which this crate cannot name; it is hidden from the documentation and its
868/// call sites are pinned by `network_gate_enumeration_test` in `axiam-api-rest`
869/// to the function's own module and to tests. The type is neither `Clone` nor
870/// `Default`, so a clearance cannot be copied into a second write.
871///
872/// A clearance can be **narrowed** after it is minted, never widened:
873/// [`NetworkWriteClearance::freeze_profiled_placement`] hands a condition the
874/// gate could not settle on its own read to the write's transaction
875/// (R110W6-03, and the decision of 2026-10-16 for a segment delete).
876#[derive(Debug, PartialEq, Eq)]
877pub struct NetworkWriteClearance {
878    tenant_id: Uuid,
879    actor: Uuid,
880    target: NetworkWriteTarget,
881    profiled_placement_frozen: bool,
882}
883
884impl NetworkWriteClearance {
885    /// Mint a clearance. **Only** `axiam_authz::network_gate` calls this in
886    /// production code; anything else is caught by the enumeration test.
887    #[doc(hidden)]
888    #[must_use]
889    pub fn issued_by_the_network_gate(
890        tenant_id: Uuid,
891        actor: Uuid,
892        target: NetworkWriteTarget,
893    ) -> Self {
894        Self {
895            tenant_id,
896            actor,
897            target,
898            profiled_placement_frozen: false,
899        }
900    }
901
902    /// Narrow a segment replace's or delete's clearance: **if the segment has
903    /// a reply profile**, as the write's own transaction finds it, a replace
904    /// may not change the segment's `vlan_id`, `rank` or `enabled`
905    /// (design note §5.2, correction R110W6-03) and a delete may not happen at
906    /// all (the decision of 2026-10-16: deleting a profiled segment removes
907    /// every device placed there).
908    ///
909    /// The gate sets it when it cleared the write without checking
910    /// `network:grant` at the segment — because the replace moves no
911    /// placement, or because the segment had no profile when the gate read it.
912    /// The repository then re-checks both facts against the row it replaces
913    /// or deletes, so a profile created between the gate's read and the write
914    /// is not missed. It only ever adds a condition, so calling it cannot
915    /// widen what the gate cleared.
916    #[must_use]
917    pub fn freeze_profiled_placement(mut self) -> Self {
918        self.profiled_placement_frozen = true;
919        self
920    }
921
922    /// Whether [`Self::freeze_profiled_placement`] narrowed this clearance.
923    #[must_use]
924    pub fn profiled_placement_frozen(&self) -> bool {
925        self.profiled_placement_frozen
926    }
927
928    /// The tenant the write was cleared in.
929    #[must_use]
930    pub fn tenant_id(&self) -> Uuid {
931        self.tenant_id
932    }
933
934    /// The writer the write was cleared for.
935    #[must_use]
936    pub fn actor(&self) -> Uuid {
937        self.actor
938    }
939
940    /// The write that was cleared.
941    #[must_use]
942    pub fn target(&self) -> &NetworkWriteTarget {
943        &self.target
944    }
945
946    /// Refuse unless this clearance is for exactly this write, in this tenant,
947    /// by this actor. A mismatch is a programming error in the caller, so it is
948    /// reported as `AuthorizationDenied` and the write does not happen.
949    pub fn check(
950        &self,
951        tenant_id: Uuid,
952        actor: Uuid,
953        target: &NetworkWriteTarget,
954    ) -> AxiamResult<()> {
955        if self.tenant_id == tenant_id && self.actor == actor && &self.target == target {
956            Ok(())
957        } else {
958            Err(AxiamError::AuthorizationDenied {
959                reason: "the network write was not cleared by the authorization function \
960                         for this tenant, actor and target"
961                    .into(),
962                action: None,
963                resource_id: None,
964            })
965        }
966    }
967}
