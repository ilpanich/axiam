1//! Initial access tokens for RFC 7591 dynamic client registration (T21.4).
2//!
3//! RFC 7591 §1.2 names two profiles for the registration endpoint: **open**,
4//! where anybody may register, and **protected**, where the request carries an
5//! "initial access token" the authorization server issued out of band. This is
6//! that token, and it exists because the difference between the two profiles
7//! is the difference between a deployment that lets Claude Code register
8//! itself and a deployment that lets anybody on the internet write a row.
9//!
10//! Three properties, and each one is the answer to a question the SCIM
11//! provisioning token (`models::scim_token`) answers the other way — worth
12//! stating side by side, because the two are the same shape and not the same
13//! credential:
14//!
15//! 1. **Single use.** A SCIM token authenticates an ongoing integration; this
16//!    one authorises exactly one registration. It is consumed by the request
17//!    that succeeds, so a token that leaks after use is a token that does
18//!    nothing.
19//! 2. **Short lived.** Days, not a year. An administrator mints one when they
20//!    are about to hand it to somebody, not to leave in a console.
21//! 3. **It carries no identity.** A SCIM token resolves to a tenant user whose
22//!    RBAC decides everything. This one resolves to a *tenant* and nothing
23//!    else: what the registration may ask for is the tenant's
24//!    `dcr_allowed_scopes` and `external_client_allowed_resources`, never the
25//!    minting administrator's own authority. A stolen initial access token can
26//!    therefore create a client and cannot do anything an anonymous
27//!    registration on the same tenant could not.
28
29use chrono::{DateTime, Utc};
30use serde::{Deserialize, Serialize};
31use uuid::Uuid;
32
33/// Prefix on every issued handle.
34///
35/// Load-bearing for the reason `SCIM_TOKEN_PREFIX` is: a credential that gets
36/// pasted into a terminal, a chat message or an MCP client's configuration
37/// will eventually be pasted somewhere it is committed, and a fixed greppable
38/// prefix is what lets a secret scanner or an operator with `grep` find it.
39pub const REGISTRATION_TOKEN_PREFIX: &str = "axiam_dcr_";
40
41/// Entropy behind the handle. 256 bits, as every other opaque bearer handle in
42/// the system (refresh tokens, device codes, UMA tickets, SCIM tokens).
43pub const REGISTRATION_TOKEN_ENTROPY_BYTES: usize = 32;
44
45/// Default lifetime, in hours, when the minting request names none.
46///
47/// Twenty-four hours: an administrator mints one of these in the same sitting
48/// as they hand it over. A token that outlives the conversation it was created
49/// for is a token nobody remembers issuing.
50pub const DEFAULT_REGISTRATION_TOKEN_TTL_HOURS: u32 = 24;
51
52/// The longest lifetime the server will mint.
53///
54/// A week. Past that the credential is not an "initial access token" any more,
55/// it is a shared secret for the registration endpoint — and a deployment that
56/// wants one of those wants `dynamic_registration: anonymous` with a host
57/// glob, which is at least visible as the decision it is.
58pub const MAX_REGISTRATION_TOKEN_TTL_HOURS: u32 = 24 * 7;
59
60/// An initial access token as stored. The plaintext handle is **not** a field:
61/// only its hash is ever persisted, and the plaintext is returned exactly once
62/// at creation.
63#[derive(Debug, Clone, Serialize, Deserialize)]
64pub struct OAuth2RegistrationToken {
65    /// Row identity.
66    pub id: Uuid,
67    /// The tenant this token may register a client in. The only authority it
68    /// carries.
69    pub tenant_id: Uuid,
70    /// Operator-facing label, e.g. `"mcp-inspector-demo"`, so a tenant with
71    /// several outstanding tokens can tell them apart.
72    pub name: String,
73    /// SHA-256 of the handle, hex-encoded — the same treatment
74    /// `axiam_auth::token::hash_refresh_token` gives a refresh token.
75    pub token_hash: String,
76    /// The administrator who minted it. Recorded for the audit trail: a
77    /// registration that arrives on this token is attributable to the person
78    /// who handed it out.
79    pub created_by: Uuid,
80    /// When it stops being usable, whether or not it was ever spent.
81    pub expires_at: DateTime<Utc>,
82    /// When it was spent.
83    ///
84    /// A field rather than a delete, because a spent token is evidence: an
85    /// operator asking "was this handle ever used, and when" gets an answer,
86    /// and reaches the administrator who handed it out through
87    /// [`Self::created_by`]. A row that deleted itself on use would answer
88    /// neither question.
89    pub used_at: Option<DateTime<Utc>>,
90    /// The `client_id` the spending registration produced — **reserved, and
91    /// always `None` in this build.**
92    ///
93    /// The column exists because the shape is right and RFC 7592 (deferred)
94    /// will want it, and it is left unwritten because nothing can write it
95    /// truthfully: the token must be spent *before* the client is created (the
96    /// single-use guarantee depends on that order), so no `client_id` exists at
97    /// the moment there is a row to put it in. Writing a placeholder and
98    /// replacing it afterwards would leave the placeholder behind on any
99    /// failure, in a list an operator reads.
100    ///
101    /// The link a person actually needs is in the audit log, which records
102    /// `oauth2.client_registered` with the `client_id`, the source address and
103    /// the time, and is append-only.
104    pub used_by_client_id: Option<String>,
105    /// Row creation time.
106    pub created_at: DateTime<Utc>,
107}
108
109impl OAuth2RegistrationToken {
110    /// Whether this token may still authorise a registration, as of `now`.
111    ///
112    /// Spent and expired are checked together and answer the same way, because
113    /// the wire response does not distinguish them either: a caller probing
114    /// handles must not learn which arm rejected them.
115    pub fn is_usable(&self, now: DateTime<Utc>) -> bool {
116        self.used_at.is_none() && self.expires_at > now
117    }
118}
119
120/// Input for minting a token. `token_hash` is computed by the caller so this
121/// type never holds the plaintext.
122#[derive(Debug, Clone)]
123pub struct CreateOAuth2RegistrationToken {
124    /// The tenant the token may register into.
125    pub tenant_id: Uuid,
126    /// Operator-facing label.
127    pub name: String,
128    /// SHA-256 of the handle, hex-encoded.
129    pub token_hash: String,
130    /// The minting administrator.
131    pub created_by: Uuid,
132    /// When it stops being usable.
133    pub expires_at: DateTime<Utc>,
134}
