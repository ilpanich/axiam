31/// Whether a grant permits an action or refuses it (B1, deny-override).
32///
33/// # Precedence
34///
35/// Default deny -> an [`PermissionEffect::Allow`] grant permits ->
36/// a [`PermissionEffect::Deny`] grant refuses, **and beats every allow**,
37/// wherever either sits in the resource hierarchy. Deny wins; there is no
38/// most-specific-wins tie-break.
39///
40/// That choice is deliberate and is argued in full in
41/// `claude_dev/deny-override-design.md` §2.1. The short version: deny-override
42/// buys one checkable property — **adding a deny rule can never widen access,
43/// and can never be undone by adding allows** — and most-specific-wins buys
44/// expressiveness at the cost of making "is X denied?" unanswerable without
45/// enumerating every other rule that might out-specify it.
46///
47/// [`PermissionEffect::Allow`] is the default, so data written before this
48/// existed, and clients that send no `effect`, both mean "allow". No migration.
49#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize, utoipa::ToSchema)]
50#[serde(rename_all = "lowercase")]
51pub enum PermissionEffect {
52    /// Permits the action (the default, and the pre-B1 behaviour).
53    #[default]
54    Allow,
55    /// Refuses the action, overriding any allow.
56    Deny,
57}
58
59impl PermissionEffect {
60    /// Stable, log-safe and wire-safe name.
61    pub const fn as_str(self) -> &'static str {
62        match self {
63            Self::Allow => "allow",
64            Self::Deny => "deny",
65        }
66    }
67
68    /// Parse a wire value. Unknown values map to `None` so a caller can reject
69    /// them rather than silently defaulting — a typo'd `"denyy"` must not
70    /// become an allow.
71    pub fn from_wire(raw: &str) -> Option<Self> {
72        match raw.trim().to_ascii_lowercase().as_str() {
73            "allow" => Some(Self::Allow),
74            "deny" => Some(Self::Deny),
75            _ => None,
76        }
77    }
78
79    /// Whether this effect refuses the action.
80    pub const fn is_deny(self) -> bool {
81        matches!(self, Self::Deny)
82    }
83}

85/// A permission grant with optional scope constraints.
86/// Empty `scope_ids` means the grant covers all scopes (wildcard).
87#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
88pub struct PermissionGrant {
89    pub permission: Permission,
90    pub scope_ids: Vec<Uuid>,
91    /// B1: whether this grant permits or refuses. Defaults to
92    /// [`PermissionEffect::Allow`], which is what every pre-B1 row and every
93    /// `effect`-less request means.
94    ///
95    /// Scope interaction: an empty `scope_ids` is a **wildcard**, so a deny
96    /// with no scopes masks the action entirely on this node and its
97    /// descendants — every scope, and unscoped checks too. A scoped deny masks
98    /// only the scopes it names.
99    #[serde(default)]
100    pub effect: PermissionEffect,
101}

103#[cfg(test)]
104mod tests {
105    use super::*;
106
107    #[test]
108    fn allow_is_the_default_so_pre_b1_rows_and_effectless_requests_mean_allow() {
109        // The no-migration property: absent `effect` must read as allow. If
110        // this ever flipped, every existing grant would silently become a deny
111        // and the whole authorization model would invert.
112        assert_eq!(PermissionEffect::default(), PermissionEffect::Allow);
113        let grant: PermissionEffect = serde_json::from_str("\"allow\"").unwrap();
114        assert_eq!(grant, PermissionEffect::Allow);
115    }
116
117    #[test]
118    fn wire_names_round_trip_through_serde_and_as_str() {
119        for effect in [PermissionEffect::Allow, PermissionEffect::Deny] {
120            let json = serde_json::to_string(&effect).unwrap();
121            assert_eq!(json, format!("\"{}\"", effect.as_str()));
122            let back: PermissionEffect = serde_json::from_str(&json).unwrap();
123            assert_eq!(back, effect);
124        }
125    }
126
127    #[test]
128    fn from_wire_accepts_case_and_whitespace_variation() {
129        assert_eq!(
130            PermissionEffect::from_wire("deny"),
131            Some(PermissionEffect::Deny)
132        );
133        assert_eq!(
134            PermissionEffect::from_wire("  DENY "),
135            Some(PermissionEffect::Deny)
136        );
137        assert_eq!(
138            PermissionEffect::from_wire("Allow"),
139            Some(PermissionEffect::Allow)
140        );
141    }
142
143    #[test]
144    fn from_wire_refuses_anything_it_does_not_recognise() {
145        // The security-relevant half. A typo must NOT fall back to allow:
146        // `denyy` becoming an allow is how a deny rule silently stops working.
147        for raw in ["denyy", "DENY!", "", "permit", "true", "0"] {
148            assert_eq!(
149                PermissionEffect::from_wire(raw),
150                None,
151                "{raw:?} must not parse"
152            );
153        }
154    }
155
156    #[test]
157    fn is_deny_answers_only_for_deny() {
158        assert!(PermissionEffect::Deny.is_deny());
159        assert!(!PermissionEffect::Allow.is_deny());
160    }
161
162    #[test]
163    fn a_grant_without_an_effect_field_deserializes_as_allow() {
164        // `PermissionGrant.effect` carries `#[serde(default)]` for exactly the
165        // rows written before B1. Asserting on the struct, not just the enum,
166        // because the default lives on the field.
167        let json = serde_json::json!({
168            "permission": {
169                "id": Uuid::nil(),
170                "tenant_id": Uuid::nil(),
171                "action": "read",
172                "description": "",
173                "created_at": "2026-01-01T00:00:00Z",
174                "updated_at": "2026-01-01T00:00:00Z"
175            },
176            "scope_ids": []
177        });
178        let grant: PermissionGrant = serde_json::from_value(json).unwrap();
179        assert_eq!(grant.effect, PermissionEffect::Allow);
180        assert!(
181            grant.scope_ids.is_empty(),
182            "empty scope_ids is the wildcard"
183        );
184    }
