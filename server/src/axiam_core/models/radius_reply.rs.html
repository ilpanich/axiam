1//! The RADIUS reply-profile dictionary `axiam-radius-reply/1` and the typed
2//! profile built from it (RADIUS Live profile, `claude_dev/radius-reply-profile-design.md`
3//! §3).
4//!
5//! # A typed record, not an attribute list
6//!
7//! A reply profile is the set of RADIUS attributes AXIAM would put in an
8//! Access-Accept for a device placed on one network segment. There is **no
9//! field in which a writer names an attribute**: every attribute a profile can
10//! produce is a named, bounded field of [`ReplyProfileInput`], and the
11//! dictionary below ([`DICTIONARY_V1`]) is the closed set of them. A vendor
12//! attribute, an unknown attribute number or a free-form string attribute
13//! cannot be expressed, so it cannot be validated wrongly (§3.1).
14//!
15//! **The VLAN is not a profile field.** The three tunnel attributes are
16//! derived from the owning segment's `vlan_id`
17//! ([`crate::models::network::NetworkSegment::vlan_id`]); a profile cannot name
18//! one, so the only way to put a device on VLAN 99 is to write the profile of
19//! the segment that owns VLAN 99 (§3.1).
20//!
21//! # The dictionary is code
22//!
23//! It is a compile-time table in this layer-0 crate, each entry citing its RFC
24//! section. A dictionary change is a code change with review; no API adds an
25//! entry. Every stored profile records the version it was validated against
26//! and an unknown version is refused, never ignored (§3.4).
27//!
28//! # Bounds
29//!
30//! The tenant's `Session-Timeout` cap is read from
31//! [`crate::models::network::NetworkPolicy`]. A value above the cap is refused
32//! at write time ([`ReplyProfileInput::validate`]) and clamped down at read
33//! time ([`clamp_session_timeout`]): lowering the cap afterwards only narrows
34//! (§3.4, §5.2).
35

78/// One row of the dictionary.
79#[derive(Debug, Clone, Copy, PartialEq, Eq)]
80pub struct DictionaryEntry {
81    /// The attribute's name in FreeRADIUS's dictionary.
82    pub name: &'static str,
83    /// The attribute number.
84    pub number: u8,
85    /// `integer` or `string`, as FreeRADIUS's dictionary spells it.
86    pub data_type: &'static str,
87    /// Whether FreeRADIUS's dictionary marks the attribute `has_tag` (RFC 2868).
88    ///
89    /// **Not** the same as "a tag octet goes on the wire" (correction of
90    /// 2026-10-15, W5.4's finding on FreeRADIUS 3.2.8): the two *integer* tunnel
91    /// attributes always carry the tag octet, `0x00` here; the *string*
92    /// `Tunnel-Private-Group-Id` tagged `0` is encoded with **no** tag octet
93    /// (RFC 2868 §3.6 allows it: a first octet above `0x1F` starts the string,
94    /// and a decimal VLAN's first octet is `0x31`…`0x39`). Only a non-zero tag
95    /// would add an octet, and AXIAM never renders one. [`ReplyAttribute::tag`]
96    /// is what a rendered reply says goes on the wire.
97    pub tagged: bool,
98    /// The RFC sections that define it.
99    pub rfc: &'static str,
100    /// Where its value comes from.
101    pub source: AttributeSource,
102}

304/// Whether `name` is an acceptable `Filter-Id` / segment-name identifier:
305/// `^[A-Za-z0-9][A-Za-z0-9._-]{0,62}$` (design note §3.2).
306///
307/// Written by hand rather than with a regular expression: the alphabet is the
308/// whole point, and it keeps this layer-0 crate free of a dependency.
309#[must_use]
310pub fn is_valid_identifier(name: &str) -> bool {
311    let bytes = name.as_bytes();
312    if bytes.is_empty() || bytes.len() > 63 {
313        return false;
314    }
315    bytes[0].is_ascii_alphanumeric()
316        && bytes[1..]
317            .iter()
318            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-'))
319}

338/// What a writer supplies for a profile. Closed: an unknown key is a refusal
339/// naming it, never ignored (design note §3.4).
340#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, utoipa::ToSchema)]
341#[schema(as = NetworkReplyProfileInput)]
342#[serde(deny_unknown_fields)]
343pub struct ReplyProfileInput {
344    /// The dictionary version the profile is written against; defaults to
345    /// [`REPLY_DICTIONARY_V1`]. An unknown version is refused.
346    #[serde(default = "default_dictionary")]
347    pub dictionary: String,
348    /// `Session-Timeout`, seconds: [`SESSION_TIMEOUT_FLOOR_SECS`] up to the
349    /// tenant cap. Never zero.
350    pub session_timeout_secs: u32,
351    /// `Termination-Action`; defaults to `RADIUS-Request`.
352    #[serde(default)]
353    pub termination_action: TerminationAction,
354    /// `Idle-Timeout`, seconds: [`IDLE_TIMEOUT_FLOOR_SECS`] up to
355    /// `session_timeout_secs`.
356    #[serde(default)]
357    pub idle_timeout_secs: Option<u32>,
358    /// `Filter-Id`: one of the owning segment's declared `filter_ids`.
359    #[serde(default)]
360    pub filter_id: Option<String>,
361    /// `Service-Type`: `framed_user` or absent under option B.
362    #[serde(default)]
363    pub service_type: Option<ServiceType>,
364}
365
366impl ReplyProfileInput {
367    /// Check the profile against its dictionary and the owning segment's
368    /// envelope, at write time (design note §3.4 and §5.2 H3).
369    ///
370    /// * `segment_filter_ids` is the owning segment's declared `Filter-Id`
371    ///   names; `filter_id` must be one of them.
372    /// * `cap` is the tenant's `Session-Timeout` cap.
373    ///
374    /// Every refusal is `Validation`. This is the **value** half of H3; who may
375    /// write the profile (H1, H2, H4 to H6) is the authorization function's.
376    pub fn validate(&self, segment_filter_ids: &[String], cap: u32) -> AxiamResult<()> {
377        if dictionary(&self.dictionary).is_none() {
378            return Err(invalid("unknown reply dictionary version"));
379        }
380        if self.session_timeout_secs < SESSION_TIMEOUT_FLOOR_SECS {
381            return Err(invalid(format!(
382                "session_timeout_secs must be at least {SESSION_TIMEOUT_FLOOR_SECS}"
383            )));
384        }
385        if self.session_timeout_secs > cap {
386            return Err(invalid(format!(
387                "session_timeout_secs exceeds the tenant cap of {cap}"
388            )));
389        }
390        if let Some(idle) = self.idle_timeout_secs
391            && !(IDLE_TIMEOUT_FLOOR_SECS..=self.session_timeout_secs).contains(&idle)
392        {
393            return Err(invalid(format!(
394                "idle_timeout_secs must be between {IDLE_TIMEOUT_FLOOR_SECS} and \
395                 session_timeout_secs"
396            )));
397        }
398        if let Some(filter) = &self.filter_id {
399            if !is_valid_identifier(filter) {
400                return Err(invalid("filter_id is not a valid Filter-Id name"));
401            }
402            if !segment_filter_ids.iter().any(|f| f == filter) {
403                return Err(invalid(
404                    "filter_id is outside the segment's declared filter_ids",
405                ));
406            }
407        }
408        if self.service_type.is_some_and(ServiceType::is_admin_class) {
409            return Err(invalid(
410                "the administrative Service-Type values are disabled",
411            ));
412        }
413        Ok(())
414    }
415}

433/// The value of one rendered attribute: an integer, or the one string kind the
434/// dictionary admits (a decimal VLAN or a `Filter-Id` identifier).
435#[derive(Debug, Clone, PartialEq, Eq, Serialize, utoipa::ToSchema)]
436#[schema(as = RadiusReplyValue)]
437#[serde(untagged)]
438pub enum ReplyValue {
439    /// An `integer` attribute's value.
440    Integer(u32),
441    /// A `string` attribute's value.
442    Text(String),
443}

445/// One attribute of a rendered Access-Accept.
446#[derive(Debug, Clone, PartialEq, Eq, Serialize, utoipa::ToSchema)]
447#[schema(as = RadiusReplyAttribute)]
448pub struct ReplyAttribute {
449    /// The attribute's name in FreeRADIUS's dictionary (`Tunnel-Type`, …).
450    pub name: String,
451    /// The attribute number (RFC 2865 / RFC 2868).
452    pub number: u8,
453    /// The value, typed as the dictionary types it.
454    pub value: ReplyValue,
455    /// The RFC 2868 tag octet that goes **on the wire**, when one does.
456    ///
457    /// `0` on `Tunnel-Type` and `Tunnel-Medium-Type`: an integer tunnel
458    /// attribute always carries its tag octet (`0x00`), which FreeRADIUS emits
459    /// for these with or without `:0`. **Absent** on `Tunnel-Private-Group-Id`:
460    /// the VLAN string goes on the wire with no tag octet in front of it (RFC
461    /// 2868 §3.6 allows that, since the string's first octet is above `0x1F`),
462    /// which is exactly what FreeRADIUS 3.2.8 emits for the attribute written
463    /// untagged or tagged `:0` (verified on the wire, W5.4). Absent, too, on
464    /// every untagged attribute.
465    #[serde(skip_serializing_if = "Option::is_none")]
466    pub tag: Option<u8>,
467}

482/// The values a reply is rendered from: a stored profile after the decision
483/// path re-validated it and clamped it to the tenant's current cap (design note

486pub struct EffectiveReplyProfile {
487    /// The dictionary version the profile was validated against.
488    pub dictionary: String,
489    /// `Session-Timeout`, already clamped to the tenant cap.
490    pub session_timeout_secs: u32,
491    /// Whether the stored value was above the cap and has been lowered.
492    pub session_timeout_clamped: bool,
493    /// `Termination-Action`.
494    pub termination_action: TerminationAction,
495    /// `Idle-Timeout`, never above the effective `Session-Timeout`.
496    pub idle_timeout_secs: Option<u32>,

499    /// `Service-Type`.

503/// Render the Access-Accept attributes for a device placed on a segment of
504/// VLAN `vlan_id` whose profile is `profile`.
505///
506/// The encoder accepts only dictionary values and checks them again here, the
507/// last step before the wire, whatever its caller checked: a VLAN outside 1 to
508/// 4094, a `Session-Timeout` below the floor or above the ceiling, an
509/// `Idle-Timeout` outside its bounds, a `Filter-Id` that is not an identifier,
510/// an administrative `Service-Type` or an unknown dictionary version is a
511/// `Validation` error and nothing is rendered.
512pub fn render_reply(vlan_id: u16, profile: &EffectiveReplyProfile) -> AxiamResult<RenderedReply> {
513    let Some(table) = dictionary(&profile.dictionary) else {
514        return Err(invalid("unknown reply dictionary version"));
515    };
516    if !is_valid_vlan(vlan_id) {
517        return Err(invalid("the segment's VLAN is outside 1 to 4094"));
518    }
519    if !(SESSION_TIMEOUT_FLOOR_SECS..=SESSION_TIMEOUT_CAP_CEILING_SECS)
520        .contains(&profile.session_timeout_secs)
521    {
522        return Err(invalid("Session-Timeout is outside its bounds"));
523    }
524    if let Some(idle) = profile.idle_timeout_secs
525        && !(IDLE_TIMEOUT_FLOOR_SECS..=profile.session_timeout_secs).contains(&idle)
526    {
527        return Err(invalid("Idle-Timeout is outside its bounds"));
528    }
529    if let Some(filter) = &profile.filter_id
530        && !is_valid_identifier(filter)
531    {
532        return Err(invalid("Filter-Id is not a valid identifier"));
533    }
534    if profile
535        .service_type
536        .is_some_and(ServiceType::is_admin_class)
537    {
538        return Err(invalid(
539            "the administrative Service-Type values are disabled",
540        ));
541    }
542
543    let mut attributes = Vec::with_capacity(table.len());
544    for entry in table {
545        let (value, tag) = match entry.name {
546            "Tunnel-Type" => (ReplyValue::Integer(TUNNEL_TYPE_VLAN), Some(0)),
547            "Tunnel-Medium-Type" => (ReplyValue::Integer(TUNNEL_MEDIUM_TYPE_802), Some(0)),
548            // Decimal, no sign, no leading zero (RFC 3580 §3.31), and no tag
549            // octet on the wire: see `ReplyAttribute::tag`.
550            "Tunnel-Private-Group-Id" => (ReplyValue::Text(vlan_id.to_string()), None),
551            "Session-Timeout" => (ReplyValue::Integer(profile.session_timeout_secs), None),
552            "Termination-Action" => (
553                ReplyValue::Integer(profile.termination_action.value()),
554                None,
555            ),
556            "Idle-Timeout" => match profile.idle_timeout_secs {
557                Some(idle) => (ReplyValue::Integer(idle), None),
558                None => continue,
559            },
560            "Filter-Id" => match &profile.filter_id {
561                Some(filter) => (ReplyValue::Text(filter.clone()), None),
562                None => continue,
563            },
564            "Service-Type" => match profile.service_type {
565                Some(service) => (ReplyValue::Integer(service.value()), None),
566                None => continue,
567            },
568            // A dictionary row this encoder does not know how to fill is a
569            // code defect; refusing is the fail-closed answer.
570            _ => return Err(invalid("a dictionary attribute has no encoder")),
571        };
572        attributes.push(ReplyAttribute {
573            name: entry.name.to_owned(),
574            number: entry.number,
575            value,
576            tag,
577        });
578    }
579    Ok(RenderedReply {
580        dictionary: profile.dictionary.clone(),
581        attributes,
582    })
583}

616    /// The tunnel triple of RFC 3580 §3.31 and the profile's fields, in
617    /// dictionary order; the integers carry tag 0 on the wire, the VLAN string
618    /// carries none (W5.4: FreeRADIUS 3.2.8 emits no tag octet for a string
619    /// tagged 0).
620    #[test]
621    fn the_reply_is_the_tunnel_triple_and_the_profile() {

624        assert_eq!(
625            json,
626            serde_json::json!({
627                "dictionary": "axiam-radius-reply/1",
628                "attributes": [
629                    {"name": "Tunnel-Type", "number": 64, "value": 13, "tag": 0},
630                    {"name": "Tunnel-Medium-Type", "number": 65, "value": 6, "tag": 0},
631                    {"name": "Tunnel-Private-Group-Id", "number": 81, "value": "20"},
632                    {"name": "Session-Timeout", "number": 27, "value": 900},
