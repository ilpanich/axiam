36/// Which tenants an assignment reaches, on top of the resource it is scoped to.
37///
38/// # What this adds, and why it is not a resource scope
39///
40/// A role assignment already answers *which resources* it covers. It did not
41/// answer *which tenants*, because until organization scope existed there was
42/// only ever one: the tenant the role lives in.
43///
44/// An organization-level principal broke that. Its roles live in the
45/// organization's reserved tenant and its global assignments deliberately
46/// carry into **every** tenant of the organization — which is right for an
47/// organization administrator and wrong for the case this exists to serve: an
48/// organization-level account that should administer two of the organization's
49/// twelve tenants and no others.
50///
51/// A resource scope cannot express that. Resources live *inside* a tenant, and
52/// a resource id from the organization tenant names nothing in a child tenant —
53/// `axiam_authz` says so explicitly when it drops resource-scoped assignments
54/// at a tenant boundary. Tenant reach is a different axis, so it is a different
55/// field.
56///
57/// # Semantics
58///
59/// * `None` — the assignment reaches wherever the role does. In an ordinary
60///   tenant that is that tenant; in an organization scope tenant it is every
61///   tenant of the organization. This is what every assignment written before
62///   this field existed means, and it is what an unspecified scope keeps
63///   meaning.
64/// * `Some(tenants)` — the assignment contributes **only** while acting on a
65///   tenant in the list, and nowhere else. That "nowhere else" includes the
66///   organization's own scope: an account restricted to tenants A and B is not
67///   an organization-wide administrator, and letting its grants apply while no
68///   tenant is selected would hand it back the organization-level reach the
69///   restriction exists to remove.
70/// * `Some([])` is not a meaningful restriction (it would reach nothing at all)
71///   and the API refuses it rather than writing an assignment that can never
72///   apply.
73pub type TenantScope = Option<Vec<Uuid>>;
74
75/// Whether an assignment carrying `tenant_scope` applies while acting on
76/// `tenant_id`.
77///
78/// The single rule the whole feature rests on, written once so the authorization
79/// engine, `/auth/me` and the tenant listing cannot drift apart on it.
80#[must_use]
81pub fn tenant_scope_reaches(tenant_scope: &TenantScope, tenant_id: Uuid) -> bool {
82    match tenant_scope {
83        None => true,
84        Some(tenants) => tenants.contains(&tenant_id),
85    }
86}

88/// The scope of a new role assignment: the resource it covers, and the tenants
89/// it reaches.
90///
91/// One struct rather than two positional `Option`s on every `assign_*` method:
92/// `Option<Uuid>` and `Option<Vec<Uuid>>` sitting next to each other in a call
93/// are two arguments nobody can read at the call site, and the two mean
94/// genuinely different things.
95#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
96pub struct AssignmentScope {
97    /// `None` assigns the role globally — every resource in reach.
98    pub resource_id: Option<Uuid>,
99    /// `None` reaches wherever the role does. See [`TenantScope`].
100    #[serde(default, skip_serializing_if = "Option::is_none")]
101    pub tenant_scope: Option<Vec<Uuid>>,
102    /// Whether the assignment also applies to the descendants of
103    /// `resource_id` (`true`, the default) or at that resource only (`false`).
104    #[serde(default = "default_inherit")]
105    pub inherit: bool,
106}
107
108/// The value `inherit` takes when nothing says otherwise: `true`.
109///
110/// # What the flag means
111///
112/// A resource-scoped assignment applies at its resource and — with `inherit:
113/// true`, which is what every assignment meant before the field existed — at
114/// every descendant of it. With `inherit: false` it applies at its resource
115/// **only**: "here and no further". The node itself is always in scope.
116///
117/// The flag is a property of the assignment (the `has_role` edge), not of the
118/// role's grants, because inheritance is about *where* an assignment applies,
119/// which is what `resource_id` already answers. It therefore governs allows and
120/// denies alike: a role whose grants are denies, assigned non-inheritably,
121/// denies at that node and nowhere below it.
122///
123/// It says nothing about an assignment with no resource (tenant-wide) or of a
124/// global role (everywhere by definition); the REST API refuses `false` on
125/// both rather than store a flag the engine would ignore.
126#[must_use]
127pub const fn default_inherit() -> bool {
128    true
129}
130
131impl Default for AssignmentScope {
132    fn default() -> Self {
133        Self {
134            resource_id: None,
135            tenant_scope: None,
136            inherit: default_inherit(),
137        }
138    }
139}

141impl AssignmentScope {
142    /// A global, unrestricted assignment — what every assignment was before
143    /// either field existed.
144    #[must_use]
145    pub fn global() -> Self {
146        Self::default()
147    }
148
149    /// Scoped to one resource, reaching wherever the role does.
150    #[must_use]
151    pub fn resource(resource_id: Uuid) -> Self {
152        Self {
153            resource_id: Some(resource_id),
154            tenant_scope: None,
155            inherit: default_inherit(),
156        }
157    }
158
159    /// Scoped to one resource and to that resource only — its descendants are
160    /// not reached. See [`default_inherit`].
161    #[must_use]
162    pub fn resource_only(resource_id: Uuid) -> Self {
163        Self {
164            resource_id: Some(resource_id),
165            tenant_scope: None,
166            inherit: false,
167        }
168    }
169}

209/// One `has_role` edge seen from the *role's* side: which subject holds the
210/// assignment, and the resource it is scoped to.
211///
212/// `has_role` carries a `UNIQUE(in, out)` index, so a subject holds a given
213/// role at most once — this is one row per subject, same as listing subject
214/// ids. What it adds is `resource_id`, and that field is not decoration: an
215/// unassign that does not name it deletes the edge whose `resource_id` is
216/// `NONE`, so revoking a resource-scoped grant without it silently deletes
217/// nothing.
218#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
219pub struct RoleSubjectAssignment {
220    /// The user or group holding the assignment.
221    pub subject_id: Uuid,
222    /// `None` means the role was assigned globally (no resource scope).
223    pub resource_id: Option<Uuid>,
224    /// The tenants this assignment reaches. See [`TenantScope`].
225    #[serde(default, skip_serializing_if = "Option::is_none")]
226    pub tenant_scope: Option<Vec<Uuid>>,
227    /// Whether the assignment reaches the descendants of `resource_id`
228    /// (`true`, the default) or applies at that resource only (`false`).
229    #[serde(default = "default_inherit")]
230    pub inherit: bool,
231}

233/// How far across an organization's tenants a principal's assignments reach.
234///
235/// Derived from the whole set of a principal's assignments, because reach is a
236/// property of the set and not of any one of them: holding *one* unrestricted
237/// assignment makes the principal unrestricted however many tenant-scoped ones
238/// sit beside it — those add tenants, they cannot take any away.
239#[derive(Debug, Clone, PartialEq, Eq)]
240pub enum TenantReach {
241    /// Wherever the roles themselves reach. For a principal in an
242    /// organization's scope tenant that is every tenant of the organization;
243    /// for an ordinary principal it is the one tenant it lives in.
244    Unrestricted,
245    /// Only these tenants. Possibly empty, which is a principal that can act
246    /// nowhere — the assign API refuses to create one, but a set can shrink to
247    /// empty afterwards by deleting the tenants an assignment named.
248    Restricted(BTreeSet<Uuid>),
249}
250
251impl TenantReach {
252    /// Whether acting on `tenant_id` is within reach.
253    #[must_use]
254    pub fn includes(&self, tenant_id: Uuid) -> bool {
255        match self {
256            Self::Unrestricted => true,
257            Self::Restricted(tenants) => tenants.contains(&tenant_id),
258        }
259    }
260
261    /// Whether this reach is narrower than the organization it sits in.
262    ///
263    /// The question organization-level *actions* ask: creating a tenant or
264    /// minting the organization's CA is an act on the organization as a whole,
265    /// and a principal deliberately confined to some of its tenants is not
266    /// entitled to it.
267    #[must_use]
268    pub fn is_restricted(&self) -> bool {
269        matches!(self, Self::Restricted(_))
270    }
271}

273/// The reach of a principal holding `assignments`.
274///
275/// One unrestricted assignment is enough — see [`TenantReach`].
276///
277/// # An empty set is unrestricted, not restricted-to-nothing
278///
279/// A principal with no assignments has no grants to bound, so there is nothing
280/// for a reach to describe. Reporting [`TenantReach::Restricted`] with an empty
281/// set would be arithmetically tidy and wrong in use: callers ask
282/// `is_restricted()` to mean *"this account was deliberately confined"*, and a
283/// principal that simply holds no roles was not. It would then be refused
284/// organization-level actions with "your roles are restricted to particular
285/// tenants", which names the wrong reason — and the permission check refuses it
286/// anyway, on the right one.
287#[must_use]
288pub fn tenant_reach_of(assignments: &[RoleAssignment]) -> TenantReach {
289    if assignments.is_empty() {
290        return TenantReach::Unrestricted;
291    }
292    let mut tenants = BTreeSet::new();
293    for assignment in assignments {
294        match &assignment.tenant_scope {
295            None => return TenantReach::Unrestricted,
296            Some(scope) => tenants.extend(scope.iter().copied()),
297        }
298    }
299    TenantReach::Restricted(tenants)
300}

324    /// I1 for the flag: every `has_role` edge and every request written before

327    #[test]
328    fn an_absent_inherit_field_reads_as_inheritable() {
329        let scope: AssignmentScope =
330            serde_json::from_str(r#"{"resource_id":null}"#).expect("decodes");
331        assert!(scope.inherit);
332        assert!(AssignmentScope::global().inherit);
333        assert!(AssignmentScope::resource(Uuid::new_v4()).inherit);
334        assert!(AssignmentScope::from(Some(Uuid::new_v4())).inherit);
335        assert!(!AssignmentScope::resource_only(Uuid::new_v4()).inherit);
336
337        let subject: RoleSubjectAssignment = serde_json::from_str(&format!(
338            r#"{{"subject_id":"{}","resource_id":null}}"#,
339            Uuid::new_v4()
340        ))
341        .expect("decodes");
342        assert!(subject.inherit);
343    }

345    #[test]
346    fn no_assignments_is_unrestricted_not_confined_to_nothing() {
347        // The distinction the guard rests on: "holds no roles" and "was
348        // deliberately confined" are different facts, and only the second one
349        // should produce "your roles are restricted to particular tenants".
350        assert_eq!(tenant_reach_of(&[]), TenantReach::Unrestricted);

353    #[test]
354    fn one_unrestricted_assignment_makes_the_whole_set_unrestricted() {
355        let tenant = Uuid::new_v4();
356        let reach = tenant_reach_of(&[assignment(Some(vec![tenant])), assignment(None)]);
357        assert_eq!(reach, TenantReach::Unrestricted);
358        assert!(!reach.is_restricted());
359    }

361    #[test]
362    fn scoped_assignments_union_their_tenants() {
363        let (a, b) = (Uuid::new_v4(), Uuid::new_v4());
364        let reach = tenant_reach_of(&[assignment(Some(vec![a])), assignment(Some(vec![b]))]);
365        assert!(reach.is_restricted());

373        let (a, b) = (Uuid::new_v4(), Uuid::new_v4());
374        assert!(assignment(None).reaches_tenant(a));
375        assert!(assignment(Some(vec![a])).reaches_tenant(a));
376        assert!(!assignment(Some(vec![a])).reaches_tenant(b));
377        // An empty list reaches nothing. The API refuses to create one; if a
378        // row ever holds one it must not read back as "everywhere".
379        assert!(!assignment(Some(vec![])).reaches_tenant(a));
