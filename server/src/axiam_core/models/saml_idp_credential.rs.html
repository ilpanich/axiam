1//! The SAML identity provider's signing credential (G-2, T23.2.1, D-21).
2//!
3//! One tenant signs every SAML assertion it issues with one RSA-4096 key, and
4//! the service providers it talks to pin the matching certificate from the IdP
5//! metadata. This module is the plain data of that credential; the issuance
6//! (leaf signing under the tenant's CA, key sealing) is
7//! `axiam_pki::SamlIdpCredentialService`, and the storage is
8//! [`SamlIdpCredentialRepository`](crate::repository::SamlIdpCredentialRepository).
9//!
10//! # Not a `certificate`, not on the wire
11//!
12//! The leaf carries [`CertificateType::SamlSigning`](crate::models::certificate::CertificateType::SamlSigning)
13//! in its profile but is **never a `certificate` row**, so no certificate list
14//! or get can return it, and nothing in this module derives `Serialize` or
15//! `ToSchema`: a response type for the credential is a decision for the route
16//! that exposes it, taken on purpose, not a derive away.
17//!
18//! # The private key
19//!
20//! [`SealedSamlIdpKey`] is what the repository stores and returns: ciphertext
21//! plus the custody that sealed it, with a `Debug` that prints neither. The
22//! plaintext exists only as a `Zeroizing` buffer inside the issuing service, at
23//! the moment a key is generated or an assertion is signed.
24

30/// Where a credential is in its life.
31///
32/// At most one credential per tenant is [`Self::Active`] and at most one is
33/// [`Self::Next`] — enforced by the datastore, not by the service. `Next` is
34/// the slot rotation publishes in metadata ahead of promoting it, so an SP that
35/// pins certificates has the new one before any assertion carries it; no
36/// rotation exists in W3, but the slot does, so adding it needs no migration.
37/// `Retired` is terminal and unbounded in number.
38#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
39pub enum SamlIdpCredentialStatus {
40    /// Signs the assertions the tenant issues now.
41    Active,
42    /// Published in metadata, not yet signing.
43    Next,
44    /// Out of metadata and out of use. Its key material is destroyed.
45    Retired,
46}
47
48impl SamlIdpCredentialStatus {
49    /// The spelling stored in the datastore.
50    pub fn as_str(&self) -> &'static str {
51        match self {
52            Self::Active => "active",
53            Self::Next => "next",
54            Self::Retired => "retired",
55        }
56    }
57
58    /// Parse the stored spelling.
59    pub fn from_wire(raw: &str) -> Option<Self> {
60        match raw {
61            "active" => Some(Self::Active),
62            "next" => Some(Self::Next),
63            "retired" => Some(Self::Retired),
64            _ => None,
65        }
66    }
67}

69/// A tenant's SAML signing credential, without its private key.
70///
71/// This is everything a list, a metadata document or an admin page may show.
72/// The key travels separately, as a [`SealedSamlIdpKey`], on the one path that
73/// asks for it.
74#[derive(Debug, Clone, PartialEq, Eq)]
75pub struct SamlIdpCredential {
76    /// Credential id.
77    pub id: Uuid,
78    /// The tenant this credential signs for.
79    pub tenant_id: Uuid,
80    /// The signing CA that issued the leaf.
81    pub issuer_ca_id: Uuid,
82    /// The leaf certificate, PEM. Public.
83    pub certificate_pem: String,
84    /// The certificate's serial number, lower-case hex.
85    pub serial: String,
86    /// SHA-256 fingerprint of the certificate's DER, lower-case hex.
87    pub fingerprint: String,
88    /// Start of the certificate's validity.
89    pub not_before: DateTime<Utc>,
90    /// End of the certificate's validity.
91    pub not_after: DateTime<Utc>,
92    /// Lifecycle position.
93    pub status: SamlIdpCredentialStatus,
94    /// Which custodian sealed the private key. Recorded on the row, as for CAs,
95    /// so a later custodian is a new value and not a migration.
96    pub key_custody: CaKeyCustody,
97    /// When the credential was created.
98    pub created_at: DateTime<Utc>,
99    /// When it was retired, if it was.
100    pub retired_at: Option<DateTime<Utc>>,
101}
102
103/// What a promotion did (D-42): the credential now `active`, and the one it
104/// replaced, now `retired` — `None` when the tenant had no active credential.
105///
106/// Both are without key material, like every [`SamlIdpCredential`].
107#[derive(Debug, Clone, PartialEq, Eq)]
108pub struct SamlIdpCredentialPromotion {
109    /// The former `next` credential, now `active`.
110    pub active: SamlIdpCredential,
111    /// The former `active` credential, now `retired` with its key destroyed.
112    pub retired: Option<SamlIdpCredential>,
113}

115/// A private key as the datastore holds it: sealed, with the custody that
116/// sealed it.
117///
118/// `Debug` prints neither the ciphertext nor its length.
119#[derive(Clone, PartialEq, Eq)]
120pub struct SealedSamlIdpKey {
121    /// The custodian that holds (or sealed) the key.
122    pub custody: CaKeyCustody,
123    /// Where a custodian that keeps the key itself put it; `None` for database
124    /// custody, whose material is [`Self::ciphertext`].
125    pub locator: Option<String>,
126    /// AES-256-GCM ciphertext, database custody only.
127    pub ciphertext: Option<Vec<u8>>,
128}
129
130impl std::fmt::Debug for SealedSamlIdpKey {
131    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
132        f.debug_struct("SealedSamlIdpKey")
133            .field("custody", &self.custody)
134            .field("locator", &self.locator)
135            .field(
136                "ciphertext",
137                &self.ciphertext.as_ref().map(|_| "[REDACTED]"),
138            )
139            .finish()
140    }
141}

143/// A credential and its sealed key: what the signer's lookup returns.
144///
145/// Deliberately a different type from [`SamlIdpCredential`], so the one
146/// repository method that returns key material cannot be mistaken, by type, for
147/// the ones that never do.
148#[derive(Debug, Clone, PartialEq, Eq)]
149pub struct SealedSamlIdpCredential {
150    /// The credential's public facts.
151    pub credential: SamlIdpCredential,
152    /// Its sealed private key.
153    pub key: SealedSamlIdpKey,
154}

156/// A credential to be stored.
157#[derive(Debug, Clone, PartialEq, Eq)]
158pub struct StoreSamlIdpCredential {
159    /// The id the row will have. Chosen by the caller, before the key is
160    /// sealed, so a custodian that names what it stores can name it after the
161    /// credential that owns it.
162    pub id: Uuid,
163    /// The tenant it signs for.
164    pub tenant_id: Uuid,
165    /// The signing CA that issued the leaf.
166    pub issuer_ca_id: Uuid,
167    /// The leaf certificate, PEM.
168    pub certificate_pem: String,
169    /// The certificate's serial, lower-case hex.
170    pub serial: String,
171    /// SHA-256 fingerprint of the DER, lower-case hex.
172    pub fingerprint: String,
173    /// Start of validity.
174    pub not_before: DateTime<Utc>,
175    /// End of validity.
176    pub not_after: DateTime<Utc>,
177    /// [`SamlIdpCredentialStatus::Active`] or [`SamlIdpCredentialStatus::Next`];
178    /// a credential is never created retired.
179    pub status: SamlIdpCredentialStatus,
180    /// The sealed private key.
181    pub key: SealedSamlIdpKey,
182}

184#[cfg(test)]
185mod tests {
186    use super::*;
187
188    #[test]
189    fn status_spellings_round_trip() {
190        for status in [
191            SamlIdpCredentialStatus::Active,
192            SamlIdpCredentialStatus::Next,
193            SamlIdpCredentialStatus::Retired,
194        ] {
195            assert_eq!(
196                SamlIdpCredentialStatus::from_wire(status.as_str()),
197                Some(status)
198            );
199        }
200        assert_eq!(SamlIdpCredentialStatus::from_wire("Active"), None);
201    }
202
203    #[test]
204    fn a_sealed_key_debug_prints_no_ciphertext() {
205        let marker = Uuid::new_v4().simple().to_string();
206        let sealed = SealedSamlIdpKey {
207            custody: CaKeyCustody::Database,
208            locator: None,
209            ciphertext: Some(marker.clone().into_bytes()),
210        };
211        let shown = format!(
212            "{sealed:?} {:?}",
213            StoreSamlIdpCredential {
214                id: Uuid::nil(),
215                tenant_id: Uuid::nil(),
216                issuer_ca_id: Uuid::nil(),
217                certificate_pem: String::new(),
218                serial: String::new(),
219                fingerprint: String::new(),
220                not_before: DateTime::<Utc>::UNIX_EPOCH,
221                not_after: DateTime::<Utc>::UNIX_EPOCH,
222                status: SamlIdpCredentialStatus::Active,
223                key: sealed.clone(),
224            }
225        );
226        assert!(shown.contains("[REDACTED]"));
227        assert!(!shown.contains(&marker));
228        assert!(!shown.contains(&format!("{:?}", marker.as_bytes())));
229    }
