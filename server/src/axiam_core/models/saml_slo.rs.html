1//! What SAML single logout needs the datastore to remember (G-2, T23.2.4,
2//! schema v76): which service providers hold which session, and where a logout
3//! chain stands.
4//!
5//! Two tables, both tenant-scoped, both short-lived by construction.
6//!
7//! # `saml_sp_session` — the participant record (D-37)
8//!
9//! One row per (tenant, AXIAM session, service provider), written by the SSO
10//! endpoint's second leg **before** the assertion is signed. It holds the
11//! `NameID` and the `SessionIndex` the SP was given, because:
12//!
13//! * a pairwise `NameID` is an HMAC (D-22) and cannot be resolved back to a user
14//!   without a record, and
15//! * single logout must know, for every logout, which SPs hold a session and
16//!   what each was told.
17//!
18//! The `SessionIndex` is 32 bytes from the OS CSPRNG, base64url without padding,
19//! **per SP** — so two SPs that compare notes cannot correlate a person's
20//! sessions through it (T-312) — and it is **not** the AXIAM session id. A row
21//! is not a credential: a `LogoutRequest` must be signed by the SP's registered
22//! certificate, so an index alone ends nothing.
23//!
24//! # `saml_logout_run` — the logout chain (D-39)
25//!
26//! One row per verified logout. It is the replay guard for a `LogoutRequest`
27//! `ID` (`replay_key`, UNIQUE per tenant, kept until the row expires) and the
28//! state of the front-channel propagation: the participant rows still to be
29//! told, and the SHA-256 of the **one** outbound request `ID` the browser is
30//! carrying. The raw `ID` is never stored (T-383); the response that answers it
31//! is consumed once on the X6 two-layer arbiter and must come from the SP the
32//! request went to.
33//!
34//! The queue holds participant row ids, not copies of their `NameID`s: the
35//! personal data lives in one table, and the rows it names are deleted when the
36//! chain ends.
37

42/// How long a logout run lives: ten minutes (D-38, D-39).
43///
44/// It is the life of the replay guard on a `LogoutRequest` `ID`, so it must
45/// exceed the window an `IssueInstant` is accepted in (five minutes back, the
46/// clock-skew allowance forward) — which it does with margin — and it bounds how
47/// long a stranded chain (an SP that never answers) holds its rows.
48pub const SAML_LOGOUT_RUN_TTL_SECS: i64 = 600;
49
50/// The most service providers one logout run tells (D-38, D-39). A session that
51/// took part in more ends the run `PartialLogout`.
52pub const MAX_LOGOUT_RUN_PARTICIPANTS: usize = 32;

54/// What the SSO endpoint records before it signs an assertion.
55#[derive(Debug, Clone, PartialEq, Eq)]
56pub struct NewSamlSpSession {
57    /// The tenant of the request path.
58    pub tenant_id: Uuid,
59    /// The AXIAM session the assertion is for.
60    pub session_id: Uuid,
61    /// The session's user.
62    pub user_id: Uuid,
63    /// The registered service provider.
64    pub sp_id: Uuid,
65    /// The SP's entity id, as registered (it is immutable, D-37).
66    pub sp_entity_id: String,
67    /// The `NameID` value the SP is given.
68    pub name_id: String,
69    /// The `NameID` format URN the SP is given.
70    pub name_id_format: String,
71    /// A candidate `SessionIndex`: 32 CSPRNG bytes, base64url without padding.
72    /// Used only when the session has no row for this SP yet; a second sign-on
73    /// to the same SP in one session keeps the first index (SAML Core §2.7.2.1
74    /// allows it).
75    pub session_index: String,
76    /// When the session ends; the row never outlives it.
77    pub expires_at: DateTime<Utc>,
78}

107// A participant row names a person's `NameID` (an email address at an
108// `emailAddress` SP). Never formatted, even by a stray `{:?}`.
109impl std::fmt::Debug for SamlSpSession {
110    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
111        f.debug_struct("SamlSpSession")
112            .field("id", &self.id)
113            .field("tenant_id", &self.tenant_id)
114            .field("session_id", &self.session_id)
115            .field("sp_id", &self.sp_id)
116            .field("name_id", &"[REDACTED]")
117            .field("session_index", &"[REDACTED]")
118            .finish_non_exhaustive()
119    }
120}

122/// Who started a logout run.

124pub enum SamlLogoutInitiator {

131/// What claims a run: the replay guard of a `LogoutRequest`, or a fresh
132/// IdP-initiated run.
133#[derive(Debug, Clone, PartialEq, Eq)]
134pub struct NewSamlLogoutRun {
135    /// The tenant of the request path.
136    pub tenant_id: Uuid,
137    /// Who started it.
138    pub initiator: SamlLogoutInitiator,
139    /// The initiating SP's `LogoutRequest` `ID`, for the final response's
140    /// `InResponseTo`; `None` for an IdP-initiated run. Already checked as a
141    /// request id.
142    pub initiator_request_id: Option<String>,
143    /// The initiating SP's `RelayState` (at most 80 bytes), echoed to that SP
144    /// only.
145    pub initiator_relay_state: Option<String>,
146}

148/// A stored logout run, as the chain needs it.
149#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
150pub struct SamlLogoutRun {
151    /// Record id.
152    pub id: Uuid,
153    /// The tenant.
154    pub tenant_id: Uuid,
155    /// The user whose sessions ended; `None` until the run is planned.
156    pub user_id: Option<Uuid>,
157    /// The initiating service provider; `None` for an IdP-initiated run.
158    pub initiator_sp_id: Option<Uuid>,
159    /// See [`NewSamlLogoutRun::initiator_request_id`].
160    pub initiator_request_id: Option<String>,
161    /// See [`NewSamlLogoutRun::initiator_relay_state`].
162    pub initiator_relay_state: Option<String>,
163    /// The participant rows still to be told, next first.
164    pub queue: Vec<Uuid>,
165    /// The AXIAM sessions the run ended (their participant rows go with it).
166    pub session_ids: Vec<Uuid>,
167    /// The service provider the outbound request in the browser went to.
168    pub current_sp_id: Option<Uuid>,
169    /// Whether any part of the logout is known not to have completed.
170    pub partial: bool,
171    /// How many AXIAM sessions the run ended.
172    pub sessions_ended: u32,
173    /// How many service providers have been sent a `LogoutRequest`.
174    pub sps_told: u32,
175    /// When the run was claimed.
176    pub created_at: DateTime<Utc>,
177    /// When it expires (and, after that, is swept).
178    pub expires_at: DateTime<Utc>,
179}
