1//! Outbound SCIM provisioning: the target registry, the link rows and the
2//! per-target delivery state (G-6, T23.6.1; decisions D-57 and D-58).
3//!
4//! A tenant registers downstream SCIM 2.0 service providers ([`ScimTarget`]);
5//! AXIAM pushes user and group lifecycle changes to them. This module is the
6//! plain data of that registry. What is pushed, and how, is the deliverer's
7//! (`axiam-scim`, T23.6.2); storage is [`crate::repository::ScimTargetRepository`],
8//! [`crate::repository::ScimTargetLinkRepository`] and
9//! [`crate::repository::ScimTargetStateRepository`].
10//!
11//! # Three tables, three writers
12//!
13//! * **The target** ([`ScimTarget`]) is written by an administrator only.
14//!   Its update is conditional on the `updated_at` the administrator read
15//!   ([`ScimTargetUpdate::expected_updated_at`]), as the SSF stream's is.
16//! * **The link rows** ([`ScimTargetLink`]) map an AXIAM user or group to the
17//!   downstream resource AXIAM created or adopted. They hold ids and a digest,
18//!   never an attribute of a person, so they can outlive an erasure until the
19//!   downstream `DELETE` has been sent.
20//! * **The delivery state** ([`ScimTargetState`]) is written by the deliverer
21//!   and the reconciliation job with atomic increments and plain sets, never
22//!   by read-modify-write, and **never touches the target row**.
23//!
24//! # No credential in a read model
25//!
26//! The bearer token or OAuth2 client secret is a credential **to the
27//! downstream service provider**. It is encrypted at rest by the repository and
28//! reaches a caller only through
29//! [`crate::repository::ScimTargetRepository::decrypt_credential`], the single
30//! path to the plaintext, which only the deliverer calls. [`ScimTarget`] holds
31//! no field for it; [`NewScimTarget`] and [`ScimTargetUpdate`], which carry it
32//! on the way in, redact it from `Debug`.
33//!
34//! # The credential is bound to its URL
35//!
36//! A write that changes the URL the credential is sent to — `base_url` of a
37//! bearer target, `token_url` of a client-credentials target, and that
38//! target's `base_url` too, where every access token the secret yields is sent
39//! (W5 F4 review, T-409) — or that switches the authentication kind, must
40//! supply the credential in the same write
41//! ([`ScimTargetUpdate::credential`]); the repository refuses it otherwise
42//! with `Validation`. Without that, an administrator who may edit a target
43//! but not read its credential could aim the stored one at a host they control.
44

52/// How AXIAM authenticates to the downstream service provider, without the
53/// credential itself.
54#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, utoipa::ToSchema)]
55#[serde(tag = "type", rename_all = "snake_case")]
56pub enum ScimTargetAuth {
57    /// A static bearer token, sent as `Authorization: Bearer <token>` to
58    /// `base_url`.
59    Bearer,
60    /// OAuth2 client credentials: AXIAM obtains an access token from
61    /// `token_url` with its client id and a stored client secret.
62    #[serde(rename = "oauth2_client_credentials")]
63    OAuth2ClientCredentials {
64        /// The token endpoint the client secret is sent to.
65        token_url: String,
66        /// The OAuth2 client id.
67        client_id: String,
68        /// The scope requested, if any.
69        scope: Option<String>,
70    },
71}
72
73impl ScimTargetAuth {
74    /// The spelling stored in the database and used on the wire for the kind.
75    #[must_use]
76    pub const fn kind_str(&self) -> &'static str {
77        match self {
78            Self::Bearer => "bearer",
79            Self::OAuth2ClientCredentials { .. } => "oauth2_client_credentials",
80        }
81    }
82}

171/// One registered downstream SCIM service provider, as stored and as read back.
172///
173/// Carries no credential: see the module documentation.
174#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
175pub struct ScimTarget {
176    /// The target id. Also the `target_id` of its outbound messages, link rows
177    /// and delivery state.
178    pub id: Uuid,
179    /// The owning tenant.
180    pub tenant_id: Uuid,
181    /// A human name.
182    pub name: String,
183    /// The SCIM service root of the downstream (e.g. `https://idp.example/scim/v2`).
184    pub base_url: String,
185    /// Whether AXIAM pushes to it. A disabled target receives nothing.
186    pub enabled: bool,
187    /// How AXIAM authenticates to it.
188    pub auth: ScimTargetAuth,
189    /// Which users it provisions.
190    pub scope: ScimTargetScope,
191    /// Whether groups are pushed too: every group for
192    /// [`ScimTargetScope::AllUsers`], the listed ones otherwise.
193    pub push_groups: bool,
194    /// Which attribute becomes `userName`.
195    pub user_name_from: UserNameSource,
196    /// What happens to a user who leaves scope or is no longer active.
197    pub deprovision: DeprovisionPolicy,
198    /// When the target was registered.
199    pub created_at: DateTime<Utc>,
200    /// When it was last written. The version an administrator's update is
201    /// conditional on.
202    pub updated_at: DateTime<Utc>,
203}

248/// A full replacement of a target's configuration by an administrator.
249#[derive(Clone)]
250pub struct ScimTargetUpdate {
251    /// See [`ScimTarget::name`].
252    pub name: String,
253    /// See [`ScimTarget::base_url`].
254    pub base_url: String,
255    /// See [`ScimTarget::enabled`].
256    pub enabled: bool,
257    /// See [`ScimTarget::auth`].
258    pub auth: ScimTargetAuth,
259    /// A new credential in plaintext, write-only; `None` keeps the stored one.
260    /// Required whenever the write moves the credential to a different URL or
261    /// switches the authentication kind (see the module documentation).
262    pub credential: Option<Zeroizing<String>>,
263    /// See [`ScimTarget::scope`].
264    pub scope: ScimTargetScope,
265    /// See [`ScimTarget::push_groups`].
266    pub push_groups: bool,
267    /// See [`ScimTarget::user_name_from`].
268    pub user_name_from: UserNameSource,
269    /// See [`ScimTarget::deprovision`].
270    pub deprovision: DeprovisionPolicy,
271    /// The [`ScimTarget::updated_at`] of the read this update was prepared
272    /// from: the write lands only if the target has not been written since,
273    /// and is otherwise refused with `Conflict` (T-406). `None` writes
274    /// unconditionally (against the version the repository itself reads, so
275    /// the URL-binding rule still holds); nothing in the server does that.
276    pub expected_updated_at: Option<DateTime<Utc>>,
277}
278
279impl ScimTargetUpdate {
280    /// The update that writes `target` back unchanged (credential kept), for a
281    /// caller that changes a few members — conditional on `target` still being
282    /// the target's current version.
283    #[must_use]
284    pub fn from_target(target: &ScimTarget) -> Self {
285        Self {
286            name: target.name.clone(),
287            base_url: target.base_url.clone(),
288            enabled: target.enabled,
289            auth: target.auth.clone(),
290            credential: None,
291            scope: target.scope.clone(),
292            push_groups: target.push_groups,
293            user_name_from: target.user_name_from,
294            deprovision: target.deprovision,
295            expected_updated_at: Some(target.updated_at),
296        }
297    }
298}
299

305            .field("base_url", &self.base_url)

385/// The link between an AXIAM resource and its downstream counterpart on one
386/// target. Holds ids and a digest only — no attribute of a person.
387#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
388pub struct ScimTargetLink {
389    /// The owning tenant.
390    pub tenant_id: Uuid,
391    /// The target this link belongs to.
392    pub target_id: Uuid,
393    /// User or group.
394    pub resource_type: ScimResourceType,
395    /// The AXIAM user or group id; sent downstream as `externalId`.
396    pub axiam_id: Uuid,
397    /// The id the downstream service provider assigned.
398    pub downstream_id: String,
399    /// SHA-256 (lower-case hex) of the last representation sent; an unchanged
400    /// digest skips the `PATCH`. `None` forces the next sync to send.
401    pub synced_digest: Option<String>,
402    /// Active or deprovisioned downstream.
403    pub state: ScimLinkState,
404    /// An erasure `DELETE` was dead-lettered and reconciliation must retry it.
405    pub erase_pending: bool,
406    /// When the link was made.
407    pub created_at: DateTime<Utc>,
408    /// When it was last written.
409    pub updated_at: DateTime<Utc>,
410}

428/// The audit action of a dead-lettered outbound SCIM delivery: the row the
429/// dispatcher writes (`<slug>.delivery_failed`), and the one the
430/// `scim_delivery_failed` notification event maps from.
431pub const SCIM_DEAD_LETTER_AUDIT_ACTION: &str = "scim_push.delivery_failed";
432
433/// At most one `scim_delivery_failed` notification per target per this many
434/// seconds (W5 F4 review, T-418). Every dead letter keeps its audit row and its
435/// count on [`ScimTargetState::dead_lettered_total`]; only the mail is
436/// coalesced, so a target that is down, or that refuses AXIAM's credential,
437/// mails each recipient of a rule once an hour instead of once per reference.
438pub const FAILURE_NOTIFICATION_INTERVAL_SECS: i64 = 60 * 60;

440/// What the deliverer and the reconciliation job record about a target's
441/// deliveries. One row per target, created with the target.
442///
443/// The `last_failure_reason` is a fixed-vocabulary string chosen by the
444/// deliverer; it never holds a URL, a response body or a value.
445#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
446pub struct ScimTargetState {
447    /// The target this state belongs to.
448    pub target_id: Uuid,
449    /// The owning tenant.
450    pub tenant_id: Uuid,
451    /// When a delivery last succeeded.
452    pub last_success_at: Option<DateTime<Utc>>,
453    /// When a delivery attempt last failed or dead-lettered. A delivery the
454    /// per-target breaker refused made no request and does not stamp it: the
455    /// breaker's window is measured from here (#550).
456    pub last_failure_at: Option<DateTime<Utc>>,
457    /// Why, in the deliverer's fixed vocabulary.
458    pub last_failure_reason: Option<String>,
459    /// Failed attempts since the last success.
460    pub consecutive_failures: u64,
461    /// Deliveries dead-lettered over the target's lifetime.
462    pub dead_lettered_total: u64,
463    /// When reconciliation last ran (the schedule anchor the claim is
464    /// conditional on).
465    pub last_reconciled_at: Option<DateTime<Utc>>,
466    /// When a reconciliation run last claimed the target.
467    pub reconcile_claimed_at: Option<DateTime<Utc>>,
468}
