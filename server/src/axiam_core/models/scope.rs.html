35/// The name of the scope every new resource is given.
36///
37/// A resource with no scopes at all is a resource whose permission grants can
38/// only ever be all-or-nothing: the "limit to scopes" picker in the admin UI
39/// shows an empty list, and the operator has to go and invent a scope before
40/// they can narrow anything. Seeding one named after the resource means a fresh
41/// resource is immediately usable in a scoped grant, and it names the thing it
42/// covers.
43///
44/// The rule is the resource's own name with whitespace replaced by hyphens,
45/// then an underscore and the resource's id: `"device 01"` on resource
46/// `01a03efd-…` becomes `"device-01_01a03efd-024c-7433-9c59-46cbc3a940fa"`.
47///
48/// Runs of whitespace collapse to a single hyphen and the ends are trimmed, so
49/// `"  Billing   Reports "` yields `"Billing-Reports_<id>"` rather than
50/// `"--Billing---Reports-_<id>"` — a scope name is typed into policy and pasted
51/// into SDK calls, and a doubled hyphen from an accidental double space is a
52/// footgun nobody would choose deliberately.
53///
54/// Case is left alone. Lowercasing would be a second, unasked-for
55/// transformation, and `Billing-Reports` is what an operator who typed
56/// `Billing Reports` expects to see.
57///
58/// # Why the id is in the name
59///
60/// Scope names are unique per resource, not per tenant, so the bare name was
61/// never *invalid* — but two resources called `Billing Reports` in different
62/// parts of the hierarchy both seeded a scope called `Billing-Reports`, and
63/// from then on every grant, policy line and SDK call naming that scope read
64/// identically while meaning different things. An operator reviewing a role
65/// could not tell which resource a scoped grant narrowed to without resolving
66/// the resource id by hand.
67///
68/// Appending the id makes the seeded name unique across the whole deployment
69/// and self-describing: the resource it belongs to is readable straight off
70/// the scope. The separator is an underscore precisely because the name half
71/// uses hyphens and a UUID contains them — `_` is the one character that
72/// still splits the two halves unambiguously.
73///
74/// The cost is a longer name, which is the right trade for a seeded default:
75/// it is generated, not typed, and an operator who wants a short name renames
76/// it. A name that is short and ambiguous cannot be fixed after the fact,
77/// because by then policy refers to it.
78///
79/// Returns `None` when the name has no non-whitespace content, because there is
80/// no scope name to build — the caller creates no default scope in that case
81/// rather than one called `"_<id>"`.
82///
83/// # Examples
84///
85/// ```
86/// use axiam_core::models::scope::default_scope_name;
87/// use uuid::Uuid;
88///
89/// let id = Uuid::parse_str("01a03efd-024c-7433-9c59-46cbc3a940fa").unwrap();
90///
91/// assert_eq!(
92///     default_scope_name("device 01", id),
93///     Some("device-01_01a03efd-024c-7433-9c59-46cbc3a940fa".to_string())
94/// );
95/// // Whitespace runs collapse; the ends are trimmed.
96/// assert_eq!(
97///     default_scope_name("  Billing   Reports ", id),
98///     Some("Billing-Reports_01a03efd-024c-7433-9c59-46cbc3a940fa".to_string())
99/// );
100/// // Nothing to name.
101/// assert_eq!(default_scope_name("   ", id), None);
102/// ```
103pub fn default_scope_name(resource_name: &str, resource_id: uuid::Uuid) -> Option<String> {
104    let name = resource_name
105        .split_whitespace()
106        .collect::<Vec<_>>()
107        .join("-");
108    if name.is_empty() {
109        None
110    } else {
111        Some(format!("{name}_{resource_id}"))
112    }
113}
114
115/// The description given to the scope [`default_scope_name`] names.
116///
117/// Says where it came from, so an operator who did not expect a scope to exist
118/// can tell it was seeded rather than typed by a colleague — and knows it is
119/// safe to rename or delete.
120pub fn default_scope_description(resource_name: &str) -> String {
121    format!("Default scope for the \"{resource_name}\" resource.")
122}

124#[cfg(test)]
125mod default_scope_tests {
126    use super::*;
127
128    /// A fixed id so every expectation below reads as one literal string.
129    const ID: &str = "01a03efd-024c-7433-9c59-46cbc3a940fa";
130
131    fn id() -> Uuid {
132        Uuid::parse_str(ID).unwrap()
133    }
134
135    fn name(resource_name: &str) -> Option<String> {
136        default_scope_name(resource_name, id())
137    }
138
139    #[test]
140    fn a_single_word_name_keeps_its_word_and_gains_the_id() {
141        assert_eq!(name("Billing"), Some(format!("Billing_{ID}")));
142    }
143
144    #[test]
145    fn spaces_become_hyphens() {
146        assert_eq!(
147            name("Customer Billing Reports"),
148            Some(format!("Customer-Billing-Reports_{ID}"))
149        );
150    }
151
152    #[test]
153    fn runs_of_whitespace_collapse_to_one_hyphen() {
154        // The alternative — a naive `replace(' ', "-")` — turns a double space
155        // into `--`, which then has to be typed exactly that way in every policy
156        // that references the scope.
157        assert_eq!(
158            name("Billing   Reports"),
159            Some(format!("Billing-Reports_{ID}"))
160        );
161    }
162
163    #[test]
164    fn leading_and_trailing_whitespace_is_trimmed() {
165        assert_eq!(
166            name("  Billing Reports  "),
167            Some(format!("Billing-Reports_{ID}"))
168        );
169    }
170
171    #[test]
172    fn tabs_and_newlines_count_as_whitespace() {
173        assert_eq!(
174            name("Billing\tReports\nQ1"),
175            Some(format!("Billing-Reports-Q1_{ID}"))
176        );
177    }
178
179    #[test]
180    fn case_is_preserved() {
181        // Lowercasing would be a transformation nobody asked for, and the
182        // operator who typed `Billing Reports` expects to recognise the result.
183        assert_eq!(
184            name("Billing REPORTS"),
185            Some(format!("Billing-REPORTS_{ID}"))
186        );
187    }
188
189    #[test]
190    fn existing_hyphens_survive() {
191        assert_eq!(
192            name("Multi-Region Billing"),
193            Some(format!("Multi-Region-Billing_{ID}"))
194        );
195    }
196
197    #[test]
198    fn a_blank_name_yields_no_scope() {
199        // Not `"_<id>"`: a scope whose whole name is a separator and an id
200        // tells an operator nothing, and the caller is better off seeding none.
201        assert_eq!(name(""), None);
202        assert_eq!(name("   "), None);
203        assert_eq!(name("\t\n"), None);
204    }
205
206    /// The property the id is there for.
207    #[test]
208    fn same_name_on_two_resources_yields_two_distinct_scopes() {
209        let a = default_scope_name("Billing Reports", Uuid::new_v4()).unwrap();
210        let b = default_scope_name("Billing Reports", Uuid::new_v4()).unwrap();
211        assert_ne!(
212            a, b,
213            "two resources sharing a name must not seed scopes sharing a name — \
214             every grant naming one would otherwise read identically while \
215             narrowing to a different resource"
216        );
217        assert!(a.starts_with("Billing-Reports_") && b.starts_with("Billing-Reports_"));
218    }
219
220    /// The separator has to survive the name half, which uses hyphens, and the
221    /// id half, which contains them.
222    #[test]
223    fn the_id_is_recoverable_from_the_scope_name() {
224        let n = name("Multi-Region Billing").unwrap();
225        let (prefix, suffix) = n
226            .rsplit_once('_')
227            .expect("an underscore separates the halves");
228        assert_eq!(prefix, "Multi-Region-Billing");
229        assert_eq!(Uuid::parse_str(suffix).unwrap(), id());
230    }
231
232    #[test]
233    fn the_description_names_the_resource_it_came_from() {
234        let d = default_scope_description("Billing Reports");
235        assert!(d.contains("Billing Reports"));
236    }
