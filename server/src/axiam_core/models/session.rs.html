1//! Session domain model.
2//!
3//! # Authentication evidence (X7.2, plan §4.3)
4//!
5//! A session used to record only *when the row was written*
6//! ([`Session::created_at`]) — and refresh rotation writes a new row on every
7//! refresh, so that timestamp resets several times an hour. It therefore
8//! cannot stand in for the OpenID Connect `auth_time`, which asserts when the
9//! **end user last authenticated**: an RP that sent `max_age=60` and received
10//! a token minted from a week-old login that happened to be refreshed a minute
11//! ago has been told a freshness guarantee it did not get.
12//!
13//! So the authentication event is recorded where it happens, once, and
14//! survives rotation:
15//!
16//! | Field | Written by | Copied across refresh rotation |
17//! |---|---|---|
18//! | [`Session::authenticated_at`] | the login that created the session | **yes** |
19//! | [`Session::amr`] | the same login, from the factors it actually verified | **yes** |
20//! | [`Session::browser_token_hash`] | the OP browser session (W3) | **yes** |
21//!
22//! Nothing in this wave *reads* the evidence into a token. The claims
23//! (`auth_time`, `acr`, `amr`) are emitted for no client at all — see
24//! `axiam_auth::token::IdTokenEvidence` — because emitting them is the honour
25//! lane, which is a later wave. What lands here is the record, so that when
26//! the lane opens there is a truthful answer to give.
27
28use chrono::{DateTime, Utc};
29use serde::{Deserialize, Serialize};
30use uuid::Uuid;

32/// An authentication method reference (RFC 8176), as recorded on a session.
33///
34/// A closed enum rather than a free string, for two reasons. The values are
35/// evidence a later wave turns into an `acr` claim, and a claim derived from
36/// `"otp "` or `"passkey"` is one that silently never satisfies the ACR the
37/// operator configured — a typo at one of the five login call sites would be
38/// discovered by a relying party rather than by the compiler. And a closed set
39/// is the only one that can be mapped exhaustively: `acr_for` (a later wave)
40/// must answer for every value that can reach it.
41///
42/// Only the values AXIAM can actually *prove* are modelled. Anything a stored
43/// row carries that is not one of them decodes to nothing at all
44/// ([`Amr::from_wire`] returns `None` and the decoder drops it), which is the
45/// strict direction: less evidence can only ever lower the assurance a session
46/// is credited with, never raise it.
47#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
48pub enum Amr {
49    /// RFC 8176 `pwd` — a password (or equivalent shared secret) was verified.
50    #[serde(rename = "pwd")]
51    Pwd,
52    /// RFC 8176 `otp` — a one-time passcode, i.e. AXIAM's TOTP second factor.
53    #[serde(rename = "otp")]
54    Otp,
55    /// RFC 8176 `mfa` — more than one distinct factor was verified in this
56    /// authentication. Recorded *in addition to* the factors themselves, as
57    /// RFC 8176 §2 intends.
58    #[serde(rename = "mfa")]
59    Mfa,
60    /// RFC 8176 `hwk` — proof of possession of a hardware-secured key, i.e. a
61    /// WebAuthn authenticator.
62    #[serde(rename = "hwk")]
63    Hwk,
64    /// RFC 8176 `swk` — proof of possession of a software-secured key.
65    #[serde(rename = "swk")]
66    Swk,
67    /// RFC 8176 `user` — user verification (PIN, biometric) was performed by
68    /// the authenticator, not merely user *presence*.
69    #[serde(rename = "user")]
70    User,
71    /// RFC 8176 `x509` — an X.509 certificate was verified (mTLS).
72    #[serde(rename = "x509")]
73    X509,
74    /// RFC 8176 `fed` — the authentication was performed by an upstream
75    /// identity provider and AXIAM verified its assertion.
76    #[serde(rename = "fed")]
77    Fed,
78}
79
80impl Amr {
81    /// The RFC 8176 spelling, which is what is stored and what an `amr` claim
82    /// would carry.
83    pub const fn as_str(self) -> &'static str {
84        match self {
85            Self::Pwd => "pwd",
86            Self::Otp => "otp",
87            Self::Mfa => "mfa",
88            Self::Hwk => "hwk",
89            Self::Swk => "swk",
90            Self::User => "user",
91            Self::X509 => "x509",
92            Self::Fed => "fed",
93        }
94    }
95
96    /// Parse a stored value. `None` for anything unrecognised — see the type
97    /// docs for why an unknown value is dropped rather than kept.
98    pub fn from_wire(raw: &str) -> Option<Self> {
99        match raw.trim() {
100            "pwd" => Some(Self::Pwd),
101            "otp" => Some(Self::Otp),
102            "mfa" => Some(Self::Mfa),
103            "hwk" => Some(Self::Hwk),
104            "swk" => Some(Self::Swk),
105            "user" => Some(Self::User),
106            "x509" => Some(Self::X509),
107            "fed" => Some(Self::Fed),
108            _ => None,
109        }
110    }
111
112    /// Decode a stored list, dropping values this build does not know.
113    ///
114    /// The drop is deliberate and is the whole reason this is not a `TryFrom`:
115    /// a row written by a newer binary must still be *readable* by an older
116    /// one, and the older one must not credit a session with evidence it
117    /// cannot interpret.
118    pub fn decode_list<S: AsRef<str>>(raw: &[S]) -> Vec<Self> {
119        raw.iter()
120            .filter_map(|v| Self::from_wire(v.as_ref()))
121            .collect()
122    }
123
124    /// Render a list for storage.
125    pub fn encode_list(values: &[Self]) -> Vec<String> {
126        values.iter().map(|v| v.as_str().to_owned()).collect()
127    }
128}

130/// What a login proved, and when.
131///
132/// Passed to `AuthService::create_session_and_tokens` by each of the five
133/// sign-in paths, so that the choke point every browser login funnels through
134/// records the event rather than inferring it.
135///
136/// [`Self::authenticated_at`] is separate from "now" because a federated login
137/// did not happen now: the upstream identity provider may have authenticated
138/// the user hours ago and be replaying a long-lived SSO session. Stamping the
139/// local clock there would overstate freshness, which is the one direction
140/// this record must never err in.
141#[derive(Debug, Clone, PartialEq, Eq)]
142pub struct AuthenticationEvidence {
143    /// When the end user actually authenticated.
144    pub authenticated_at: DateTime<Utc>,
145    /// Which methods were verified, RFC 8176 style.
146    pub amr: Vec<Amr>,
147}
148
149impl AuthenticationEvidence {
150    /// Evidence for an authentication that just happened here.
151    pub fn now(amr: Vec<Amr>) -> Self {
152        Self {
153            authenticated_at: Utc::now(),
154            amr,
155        }
156    }
157
158    /// Evidence for an authentication that happened elsewhere, at a time the
159    /// upstream asserted.
160    ///
161    /// `upstream` is `None` when the assertion carried no authentication
162    /// instant (a plain OAuth2 provider has no ID token; a SAML IdP need not
163    /// send `AuthnInstant`); the fallback is the moment AXIAM verified the
164    /// assertion, per plan §4.3.
165    pub fn upstream(upstream: Option<DateTime<Utc>>, amr: Vec<Amr>) -> Self {
166        Self {
167            authenticated_at: upstream.unwrap_or_else(Utc::now),
168            amr,
169        }
170    }
171}

173/// An authenticated session.
174#[derive(Debug, Clone, Serialize, Deserialize)]
175pub struct Session {
176    pub id: Uuid,
177    pub tenant_id: Uuid,
178    pub user_id: Uuid,
179    pub token_hash: String,
180    pub ip_address: Option<String>,
181    pub user_agent: Option<String>,
182    pub expires_at: DateTime<Utc>,
183    pub created_at: DateTime<Utc>,
184    /// X7.2 — when the end user authenticated, which is **not**
185    /// [`Self::created_at`] on a session produced by refresh rotation.
186    ///
187    /// A row written before schema v55 decodes to its `created_at`: the
188    /// closest truthful answer available, and never later than the real
189    /// authentication, so it can only understate freshness.
190    #[serde(default = "epoch")]
191    pub authenticated_at: DateTime<Utc>,
192    /// X7.2 — the methods [`Self::authenticated_at`] refers to.
193    ///
194    /// Empty for a row written before schema v55, and for any login path that
195    /// recorded none. An empty list is the floor: it satisfies no assurance
196    /// level above the weakest.
197    #[serde(default)]
198    pub amr: Vec<Amr>,
199    /// W3 — SHA-256 of the OP browser-session token this session is bound to.
200    ///
201    /// The column exists from schema v55; W3 is what writes it and what indexes
202    /// it (v56). It is the stored half of the `axiam_op_session` cookie: the
203    /// browser holds 256 random bits, the row holds their digest, and
204    /// `/oauth2/authorize` resolves one to the other for a client registered
205    /// `browser_sso`.
206    ///
207    /// `None` for every session created before W3, for every session created by
208    /// a path that is not a browser login, and for every deployment that never
209    /// registers a `browser_sso` client — the hash is written unconditionally
210    /// at login, but no client consults it unless it opted in.
211    ///
212    /// **Copied across refresh rotation**, like [`Self::authenticated_at`] and
213    /// [`Self::amr`] and for a related reason: rotation replaces the row, and a
214    /// browser whose OP cookie stopped resolving because its access token was
215    /// renewed would be silently signed out of the authorization endpoint
216    /// alone.
217    #[serde(default)]
218    pub browser_token_hash: Option<String>,
219}
220
221/// Serde fallback for [`Session::authenticated_at`] on a payload that predates
222/// the field.
223///
224/// The database decoder substitutes `created_at`, which is strictly better;
225/// this exists only so that a serialized `Session` from an older build
226/// deserializes at all, and it deliberately yields the *oldest* representable
227/// instant so nothing can read it as fresh.
228fn epoch() -> DateTime<Utc> {
229    DateTime::UNIX_EPOCH
230}

232/// Input for creating a session.
233#[derive(Debug, Clone)]
234pub struct CreateSession {
235    pub tenant_id: Uuid,
236    pub user_id: Uuid,
237    pub token_hash: String,
238    pub ip_address: Option<String>,
239    pub user_agent: Option<String>,
240    pub expires_at: DateTime<Utc>,
241    /// X7.2 — see [`Session::authenticated_at`].
242    ///
243    /// Refresh rotation **copies** this from the session it consumed instead
244    /// of stamping the clock: a refresh is not an authentication event, and a
245    /// freshness claim that resets on every refresh is worse than none.
246    pub authenticated_at: DateTime<Utc>,
247    /// X7.2 — see [`Session::amr`]. Copied across rotation for the same
248    /// reason.
249    pub amr: Vec<Amr>,
250    /// W3 — see [`Session::browser_token_hash`].
251    ///
252    /// `None` on every path that is not a browser sign-in, and on rotation it
253    /// carries the consumed session's value rather than a fresh one: the cookie
254    /// in the browser did not change, so neither may the digest it is matched
255    /// against.
256    pub browser_token_hash: Option<String>,
257}

259#[cfg(test)]
260mod tests {
261    use super::*;
262
263    #[test]
264    fn every_amr_value_round_trips_through_its_rfc_8176_spelling() {
265        for value in [
266            Amr::Pwd,
267            Amr::Otp,
268            Amr::Mfa,
269            Amr::Hwk,
270            Amr::Swk,
271            Amr::User,
272            Amr::X509,
273            Amr::Fed,
274        ] {
275            assert_eq!(Amr::from_wire(value.as_str()), Some(value));
276            assert_eq!(
277                serde_json::to_value(value).unwrap(),
278                serde_json::json!(value.as_str()),
279                "the JSON spelling must be the RFC 8176 one"
280            );
281        }
282    }
283
284    /// An unknown stored value is dropped, not guessed at and not an error:
285    /// less evidence lowers assurance, which is the safe direction.
286    #[test]
287    fn an_unknown_amr_value_is_dropped_rather_than_kept() {
288        let decoded = Amr::decode_list(&["pwd", "quantum-telepathy", "mfa"]);
289        assert_eq!(decoded, vec![Amr::Pwd, Amr::Mfa]);
290    }
291
292    #[test]
293    fn encode_and_decode_are_inverse_for_known_values() {
294        let values = vec![Amr::Pwd, Amr::Hwk, Amr::User];
295        assert_eq!(
296            Amr::decode_list(&Amr::encode_list(&values)),
297            values,
298            "storage round trip must preserve order and content"
299        );
300    }
301
302    /// The federated case the plan calls out: an upstream that authenticated
303    /// the user hours ago must not have its session dated "now".
304    #[test]
305    fn upstream_evidence_prefers_the_upstream_instant() {
306        let long_ago = Utc::now() - chrono::Duration::hours(9);
307        let evidence = AuthenticationEvidence::upstream(Some(long_ago), vec![Amr::Fed]);
308        assert_eq!(evidence.authenticated_at, long_ago);
309
310        let none = AuthenticationEvidence::upstream(None, vec![Amr::Fed]);
311        assert!(
312            (Utc::now() - none.authenticated_at).num_seconds() < 5,
313            "with no upstream instant the fallback is the verification moment"
314        );
315    }

317    /// A `Session` serialized by a build that predates the evidence fields
318    /// must still deserialize — and must not read as freshly authenticated.
319    #[test]
320    fn a_pre_v55_session_payload_decodes_to_the_strict_defaults() {
321        let legacy = serde_json::json!({
322            "id": Uuid::nil(),
323            "tenant_id": Uuid::nil(),
324            "user_id": Uuid::nil(),
325            "token_hash": "hash",
326            "ip_address": null,
327            "user_agent": null,
328            "expires_at": "2030-01-01T00:00:00Z",
329            "created_at": "2026-01-01T00:00:00Z",
330        });
331        let session: Session = serde_json::from_value(legacy).unwrap();
332        assert_eq!(session.authenticated_at, DateTime::UNIX_EPOCH);
333        assert!(session.amr.is_empty());
334        assert!(session.browser_token_hash.is_none());
335    }
