1//! Session domain model.
2//!
3//! # Authentication evidence (X7.2, plan §4.3)
4//!
5//! A session used to record only *when the row was written*
6//! ([`Session::created_at`]) — and refresh rotation writes a new row on every
7//! refresh, so that timestamp resets several times an hour. It therefore
8//! cannot stand in for the OpenID Connect `auth_time`, which asserts when the
9//! **end user last authenticated**: an RP that sent `max_age=60` and received
10//! a token minted from a week-old login that happened to be refreshed a minute
11//! ago has been told a freshness guarantee it did not get.
12//!
13//! So the authentication event is recorded where it happens, once, and
14//! survives rotation:
15//!
16//! | Field | Written by | Copied across refresh rotation |
17//! |---|---|---|
18//! | [`Session::authenticated_at`] | the login that created the session | **yes** |
19//! | [`Session::amr`] | the same login, from the factors it actually verified | **yes** |
20//! | [`Session::browser_token_hash`] | the OP browser session (W3) | **yes** |
21//!
22//! Nothing in this wave *reads* the evidence into a token. The claims
23//! (`auth_time`, `acr`, `amr`) are emitted for no client at all — see
24//! `axiam_auth::token::IdTokenEvidence` — because emitting them is the honour
25//! lane, which is a later wave. What lands here is the record, so that when
26//! the lane opens there is a truthful answer to give.
27
28use chrono::{DateTime, Utc};
29use serde::{Deserialize, Serialize};
30use uuid::Uuid;

32/// An authentication method reference (RFC 8176), as recorded on a session.
33///
34/// A closed enum rather than a free string, for two reasons. The values are
35/// evidence a later wave turns into an `acr` claim, and a claim derived from
36/// `"otp "` or `"passkey"` is one that silently never satisfies the ACR the
37/// operator configured — a typo at one of the five login call sites would be
38/// discovered by a relying party rather than by the compiler. And a closed set
39/// is the only one that can be mapped exhaustively: `acr_for` (a later wave)
40/// must answer for every value that can reach it.
41///
42/// Only the values AXIAM can actually *prove* are modelled. Anything a stored
43/// row carries that is not one of them decodes to nothing at all
44/// ([`Amr::from_wire`] returns `None` and the decoder drops it), which is the
45/// strict direction: less evidence can only ever lower the assurance a session
46/// is credited with, never raise it.
47#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
48pub enum Amr {
49    /// RFC 8176 `pwd` — a password (or equivalent shared secret) was verified.
50    #[serde(rename = "pwd")]
51    Pwd,
52    /// RFC 8176 `otp` — a one-time passcode, i.e. AXIAM's TOTP second factor.
53    #[serde(rename = "otp")]
54    Otp,
55    /// RFC 8176 `mfa` — more than one distinct factor was verified in this
56    /// authentication. Recorded *in addition to* the factors themselves, as
57    /// RFC 8176 §2 intends.
58    #[serde(rename = "mfa")]
59    Mfa,
60    /// RFC 8176 `hwk` — proof of possession of a hardware-secured key, i.e. a
61    /// WebAuthn authenticator.
62    #[serde(rename = "hwk")]
63    Hwk,
64    /// RFC 8176 `swk` — proof of possession of a software-secured key.
65    #[serde(rename = "swk")]
66    Swk,
67    /// RFC 8176 `user` — user verification (PIN, biometric) was performed by
68    /// the authenticator, not merely user *presence*.
69    #[serde(rename = "user")]
70    User,
71    /// RFC 8176 `x509` — an X.509 certificate was verified (mTLS).
72    #[serde(rename = "x509")]
73    X509,
74    /// RFC 8176 `fed` — the authentication was performed by an upstream
75    /// identity provider and AXIAM verified its assertion.
76    #[serde(rename = "fed")]
77    Fed,
78}
79
80impl Amr {
81    /// The RFC 8176 spelling, which is what is stored and what an `amr` claim
82    /// would carry.
83    pub const fn as_str(self) -> &'static str {
84        match self {
85            Self::Pwd => "pwd",
86            Self::Otp => "otp",
87            Self::Mfa => "mfa",
88            Self::Hwk => "hwk",
89            Self::Swk => "swk",
90            Self::User => "user",
91            Self::X509 => "x509",
92            Self::Fed => "fed",
93        }
94    }
95
96    /// Parse a stored value. `None` for anything unrecognised — see the type
97    /// docs for why an unknown value is dropped rather than kept.
98    pub fn from_wire(raw: &str) -> Option<Self> {
99        match raw.trim() {
100            "pwd" => Some(Self::Pwd),
101            "otp" => Some(Self::Otp),
102            "mfa" => Some(Self::Mfa),
103            "hwk" => Some(Self::Hwk),
104            "swk" => Some(Self::Swk),
105            "user" => Some(Self::User),
106            "x509" => Some(Self::X509),
107            "fed" => Some(Self::Fed),
108            _ => None,
109        }
110    }
111
112    /// Decode a stored list, dropping values this build does not know.
113    ///
114    /// The drop is deliberate and is the whole reason this is not a `TryFrom`:
115    /// a row written by a newer binary must still be *readable* by an older
116    /// one, and the older one must not credit a session with evidence it
117    /// cannot interpret.
118    pub fn decode_list<S: AsRef<str>>(raw: &[S]) -> Vec<Self> {
119        raw.iter()
120            .filter_map(|v| Self::from_wire(v.as_ref()))
121            .collect()
122    }
123
124    /// Render a list for storage.
125    pub fn encode_list(values: &[Self]) -> Vec<String> {
126        values.iter().map(|v| v.as_str().to_owned()).collect()
127    }
128}

130/// What a login proved, and when.
131///
132/// Passed to `AuthService::create_session_and_tokens` by each of the five
133/// sign-in paths, so that the choke point every browser login funnels through
134/// records the event rather than inferring it.
135///
136/// [`Self::authenticated_at`] is separate from "now" because a federated login
137/// did not happen now: the upstream identity provider may have authenticated
138/// the user hours ago and be replaying a long-lived SSO session. Stamping the
139/// local clock there would overstate freshness, which is the one direction
140/// this record must never err in.
141#[derive(Debug, Clone, PartialEq, Eq)]
142pub struct AuthenticationEvidence {
143    /// When the end user actually authenticated.
144    pub authenticated_at: DateTime<Utc>,
145    /// Which methods were verified, RFC 8176 style.
146    pub amr: Vec<Amr>,
147}
148
149impl AuthenticationEvidence {
150    /// Evidence for an authentication that just happened here.
151    pub fn now(amr: Vec<Amr>) -> Self {
152        Self {
153            authenticated_at: Utc::now(),
154            amr,
155        }
156    }
157
158    /// Evidence for an authentication that happened elsewhere, at a time the
159    /// upstream asserted — **never later than `verified_at`**, the moment
160    /// AXIAM verified the assertion (T23.1.2, D-10).
161    ///
162    /// `upstream` is `None` when the assertion carried no authentication
163    /// instant (a plain OAuth2 provider has no ID token; a SAML IdP need not
164    /// send `AuthnInstant`); the fallback is `verified_at`, per plan §4.3.
165    ///
166    /// The upstream instant is the provider's claim, and the provider's clock
167    /// is not ours. An instant in AXIAM's future would make the session look
168    /// younger than the assertion that produced it: `max_age` would be
169    /// satisfied by an authentication that has not yet happened by AXIAM's
170    /// clock, and the emitted `auth_time` would post-date its own `iat`. The
171    /// result is therefore `min(upstream, verified_at)`. An instant in the
172    /// past is kept as asserted, which is the point of taking it at all.
173    ///
174    /// `verified_at` is a parameter, not a `Utc::now()` call in here, so the
175    /// clamp is a pure function the tests drive with fixed instants.
176    pub fn upstream(
177        upstream: Option<DateTime<Utc>>,
178        verified_at: DateTime<Utc>,
179        amr: Vec<Amr>,
180    ) -> Self {
181        Self {
182            authenticated_at: upstream.map_or(verified_at, |t| t.min(verified_at)),
183            amr,
184        }
185    }
186
187    /// How far an upstream instant lies **ahead of** `verified_at`, or `None`
188    /// when it does not (or is absent).
189    ///
190    /// The amount [`Self::upstream`] silently removes. Callers compare it with
191    /// the federation path's clock-skew allowance to tell an IdP whose clock
192    /// runs a few seconds fast, which is ordinary, from one asserting
193    /// authentications in the future, which is worth an operator's attention.
194    pub fn upstream_ahead_by(
195        upstream: Option<DateTime<Utc>>,
196        verified_at: DateTime<Utc>,
197    ) -> Option<chrono::Duration> {
198        upstream
199            .map(|t| t - verified_at)
200            .filter(|ahead| *ahead > chrono::Duration::zero())
201    }
202}

204/// An authenticated session.
205#[derive(Debug, Clone, Serialize, Deserialize)]
206pub struct Session {
207    pub id: Uuid,
208    pub tenant_id: Uuid,
209    pub user_id: Uuid,
210    pub token_hash: String,
211    pub ip_address: Option<String>,
212    pub user_agent: Option<String>,
213    pub expires_at: DateTime<Utc>,
214    pub created_at: DateTime<Utc>,
215    /// X7.2 — when the end user authenticated, which is **not**
216    /// [`Self::created_at`] on a session produced by refresh rotation.
217    ///
218    /// A row written before schema v55 decodes to its `created_at`: the
219    /// closest truthful answer available, and never later than the real
220    /// authentication, so it can only understate freshness.
221    #[serde(default = "epoch")]
222    pub authenticated_at: DateTime<Utc>,
223    /// X7.2 — the methods [`Self::authenticated_at`] refers to.
224    ///
225    /// Empty for a row written before schema v55, and for any login path that
226    /// recorded none. An empty list is the floor: it satisfies no assurance
227    /// level above the weakest.
228    #[serde(default)]
229    pub amr: Vec<Amr>,
230    /// W3 — SHA-256 of the OP browser-session token this session is bound to.
231    ///
232    /// The column exists from schema v55; W3 is what writes it and what indexes
233    /// it (v56). It is the stored half of the `axiam_op_session` cookie: the
234    /// browser holds 256 random bits, the row holds their digest, and
235    /// `/oauth2/authorize` resolves one to the other for a client registered
236    /// `browser_sso`.
237    ///
238    /// `None` for every session created before W3, for every session created by
239    /// a path that is not a browser login, and for every deployment that never
240    /// registers a `browser_sso` client — the hash is written unconditionally
241    /// at login, but no client consults it unless it opted in.
242    ///
243    /// **Copied across refresh rotation**, like [`Self::authenticated_at`] and
244    /// [`Self::amr`] and for a related reason: rotation replaces the row, and a
245    /// browser whose OP cookie stopped resolving because its access token was
246    /// renewed would be silently signed out of the authorization endpoint
247    /// alone.
248    #[serde(default)]
249    pub browser_token_hash: Option<String>,
250    /// T-254 — when a refresh token belonging to this session was last
251    /// presented *after* it had already been rotated.
252    ///
253    /// `None` on every session that has never seen one, which is every session
254    /// in an untroubled deployment and every row written before schema v60.
255    #[serde(default)]
256    pub refresh_replay_at: Option<DateTime<Utc>>,
257    /// T-254 — how many such presentations were **accepted** under the FAPI
258    /// 2.0 §5.3.2.1-9 grace window.
259    ///
260    /// Only ever non-zero for a client registered `profile: fapi2`: no other
261    /// client has a grace window to be accepted under. Expected to be small
262    /// and benign — it is what a client retrying a rotation whose response it
263    /// lost produces — which is exactly why it is counted separately from the
264    /// refusals rather than folded in with them.
265    #[serde(default)]
266    pub refresh_replay_grace_accepted: u32,
267    /// T-254 — how many such presentations were **refused**.
268    ///
269    /// A refusal means the token had already been rotated and there was no
270    /// window to accept it in: the client is on the `standard` profile, or the
271    /// FAPI grace had run out. Nothing legitimate produces one, so any
272    /// non-zero value here is worth an operator's attention.
273    #[serde(default)]
274    pub refresh_replay_refused: u32,
275}

277/// What an operator should read into a session's refresh-replay counters
278/// (T-254).
279///
280/// Derived rather than stored, so it cannot drift from the counters it
281/// summarises. Its whole job is the "at a glance" distinction the T-254
282/// decision asks for: a FAPI client retrying a lost rotation looks nothing
283/// like a rotated token being presented where no window exists to accept it,
284/// and an operator scanning a session list must not have to compare two
285/// numbers to tell them apart.
286#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
287#[serde(rename_all = "snake_case")]
288pub enum RefreshReplayVerdict {
289    /// No refresh token of this session has ever been presented after
290    /// rotation.
291    None,
292    /// Every replay this session saw was accepted under the FAPI 2.0 grace
293    /// window. Informational: this is the retry the window exists for.
294    FapiGraceRetry,
295    /// At least one replay was refused. A rotated token was presented with no
296    /// window to accept it in — nothing a conformant client does.
297    Refused,
298}
299
300impl RefreshReplayVerdict {
301    /// The wire spelling, which is what the admin API emits and the admin UI
302    /// switches on.
303    pub const fn as_str(self) -> &'static str {
304        match self {
305            Self::None => "none",
306            Self::FapiGraceRetry => "fapi_grace_retry",
307            Self::Refused => "refused",
308        }
309    }
310}
311
312impl Session {
313    /// The T-254 badge for this session — see [`RefreshReplayVerdict`].
314    ///
315    /// A refusal outranks an accepted grace retry however the two counts
316    /// compare: one refused replay is a security event, and a hundred honest
317    /// retries alongside it do not make it less of one.
318    pub const fn refresh_replay_verdict(&self) -> RefreshReplayVerdict {
319        if self.refresh_replay_refused > 0 {
320            RefreshReplayVerdict::Refused
321        } else if self.refresh_replay_grace_accepted > 0 {
322            RefreshReplayVerdict::FapiGraceRetry
323        } else {
324            RefreshReplayVerdict::None
325        }
326    }
327}

329/// Serde fallback for [`Session::authenticated_at`] on a payload that predates
330/// the field.
331///
332/// The database decoder substitutes `created_at`, which is strictly better;
333/// this exists only so that a serialized `Session` from an older build
334/// deserializes at all, and it deliberately yields the *oldest* representable
335/// instant so nothing can read it as fresh.
336fn epoch() -> DateTime<Utc> {
337    DateTime::UNIX_EPOCH
338}

340/// Input for creating a session.
341#[derive(Debug, Clone)]
342pub struct CreateSession {
343    pub tenant_id: Uuid,
344    pub user_id: Uuid,
345    pub token_hash: String,
346    pub ip_address: Option<String>,
347    pub user_agent: Option<String>,
348    pub expires_at: DateTime<Utc>,
349    /// X7.2 — see [`Session::authenticated_at`].
350    ///
351    /// Refresh rotation **copies** this from the session it consumed instead
352    /// of stamping the clock: a refresh is not an authentication event, and a
353    /// freshness claim that resets on every refresh is worse than none.
354    pub authenticated_at: DateTime<Utc>,
355    /// X7.2 — see [`Session::amr`]. Copied across rotation for the same
356    /// reason.
357    pub amr: Vec<Amr>,
358    /// W3 — see [`Session::browser_token_hash`].
359    ///
360    /// `None` on every path that is not a browser sign-in, and on rotation it
361    /// carries the consumed session's value rather than a fresh one: the cookie
362    /// in the browser did not change, so neither may the digest it is matched
363    /// against.
364    pub browser_token_hash: Option<String>,
365}

371    #[test]
372    fn every_amr_value_round_trips_through_its_rfc_8176_spelling() {
373        for value in [
374            Amr::Pwd,
375            Amr::Otp,
376            Amr::Mfa,
377            Amr::Hwk,
378            Amr::Swk,
379            Amr::User,
380            Amr::X509,
381            Amr::Fed,

383            assert_eq!(Amr::from_wire(value.as_str()), Some(value));
384            assert_eq!(
385                serde_json::to_value(value).unwrap(),
386                serde_json::json!(value.as_str()),
387                "the JSON spelling must be the RFC 8176 one"
388            );
389        }

392    /// An unknown stored value is dropped, not guessed at and not an error:
393    /// less evidence lowers assurance, which is the safe direction.
394    #[test]
395    fn an_unknown_amr_value_is_dropped_rather_than_kept() {
396        let decoded = Amr::decode_list(&["pwd", "quantum-telepathy", "mfa"]);
397        assert_eq!(decoded, vec![Amr::Pwd, Amr::Mfa]);
398    }

405            values,

410    /// The federated case the plan calls out: an upstream that authenticated
411    /// the user hours ago must not have its session dated "now".
412    #[test]
413    fn upstream_evidence_prefers_the_upstream_instant() {
414        let verified_at = Utc::now();
415        let long_ago = verified_at - chrono::Duration::hours(9);
416        let evidence =
417            AuthenticationEvidence::upstream(Some(long_ago), verified_at, vec![Amr::Fed]);
418        assert_eq!(evidence.authenticated_at, long_ago);
419
420        let none = AuthenticationEvidence::upstream(None, verified_at, vec![Amr::Fed]);
421        assert_eq!(
422            none.authenticated_at, verified_at,
423            "with no upstream instant the fallback is the verification moment"
424        );
425    }

427    /// D-10: the evidence is never later than the moment AXIAM verified the
428    /// assertion, however far ahead the provider's clock claims to be.
429    #[test]
430    fn a_far_future_upstream_instant_records_the_verification_instant() {
431        let verified_at = Utc::now();
432        let far_future = verified_at + chrono::Duration::days(365 * 70);
433        let evidence =
434            AuthenticationEvidence::upstream(Some(far_future), verified_at, vec![Amr::Fed]);
435        assert_eq!(evidence.authenticated_at, verified_at);
436        assert_eq!(
437            AuthenticationEvidence::upstream_ahead_by(Some(far_future), verified_at),
438            Some(chrono::Duration::days(365 * 70))
439        );
440    }

442    /// An instant a few seconds ahead — an ordinary clock skew — is clamped too:
443    /// the guarantee is "never later", not "never later than the skew".
444    #[test]
445    fn an_upstream_instant_inside_the_skew_allowance_is_clamped() {
446        let verified_at = Utc::now();
447        let slightly_ahead = verified_at + chrono::Duration::seconds(30);
448        let evidence =
449            AuthenticationEvidence::upstream(Some(slightly_ahead), verified_at, vec![Amr::Fed]);

453    /// The boundary: exactly the verification instant is kept and is not
454    /// "ahead"; a past or absent instant reports no excess.
455    #[test]
456    fn only_an_instant_strictly_ahead_reports_an_excess() {
457        let verified_at = Utc::now();
458        let at = AuthenticationEvidence::upstream(Some(verified_at), verified_at, vec![Amr::Fed]);
459        assert_eq!(at.authenticated_at, verified_at);
460        assert_eq!(
461            AuthenticationEvidence::upstream_ahead_by(Some(verified_at), verified_at),
462            None
463        );
464        assert_eq!(
465            AuthenticationEvidence::upstream_ahead_by(
466                Some(verified_at - chrono::Duration::hours(1)),
467                verified_at
468            ),
469            None
470        );
471        assert_eq!(
472            AuthenticationEvidence::upstream_ahead_by(None, verified_at),
473            None
474        );
475    }
