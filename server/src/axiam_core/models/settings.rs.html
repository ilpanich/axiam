1//! Security settings model with org/tenant inheritance.
2//!
3//! Organizations set a security baseline. Tenants may override
4//! settings, but only to be **more restrictive** (never weaker).
5//! The effective settings for a tenant = org baseline merged with
6//! tenant overrides.
7

76/// Secure Remote Password policy.
77///
78/// `suite` and `ksf` are the parameters a *new* registration record is enrolled
79/// with. They deliberately do not apply retroactively: an existing record is
80/// only valid under the suite and KSF it was created with, so tightening these
81/// takes effect as users next set a password rather than invalidating
82/// everybody at once.
83#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, utoipa::ToSchema)]
84pub struct OpaquePolicy {
85    /// Whether OPAQUE is offered, and whether password login is still accepted.
86    #[schema(value_type = String, example = "disabled")]
87    pub opaque_mode: OpaqueMode,
88    /// RFC 9807 ciphersuite new records are enrolled under.
89    #[schema(value_type = String, example = "ristretto255_sha512")]
90    pub opaque_suite: OpaqueSuite,
91    /// Key-stretching function new records are enrolled under. Both variants
92    /// are memory-hard; see [`opaque_ksf_is_at_least`] for the tighten-only
93    /// ordering.
94    #[schema(value_type = String, example = "argon2id")]
95    pub opaque_ksf: OpaqueKsf,
96}

102/// Whether a settings row belongs to an organization or a tenant.
103#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, utoipa::ToSchema)]
104pub enum SettingsScope {
105    Org,
106    Tenant,
107}
108
109impl std::fmt::Display for SettingsScope {
110    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
111        match self {
112            Self::Org => write!(f, "org"),
113            Self::Tenant => write!(f, "tenant"),
114        }
115    }
116}
117
118impl std::str::FromStr for SettingsScope {
119    type Err = String;
120
121    fn from_str(s: &str) -> Result<Self, Self::Err> {
122        match s {
123            "org" => Ok(Self::Org),
124            "tenant" => Ok(Self::Tenant),
125            other => Err(format!("invalid settings scope: {other}")),
126        }
127    }
128}

299/// Validate internal invariants of organization-level settings.
300///
301/// Checks relationships between fields (e.g., max >= min) and rejects
302/// obviously invalid values (zero lifetimes where required).
303pub fn validate_org_settings(input: &SetOrgSettings) -> AxiamResult<()> {
304    let mut violations = Vec::new();
305
306    if input.max_lockout_duration_secs < input.lockout_duration_secs {
307        violations.push(format!(
308            "max_lockout_duration_secs ({}) must be >= \
309             lockout_duration_secs ({})",
310            input.max_lockout_duration_secs, input.lockout_duration_secs,
311        ));
312    }
313
314    if input.max_cert_validity_days < input.default_cert_validity_days {
315        violations.push(format!(
316            "max_cert_validity_days ({}) must be >= \
317             default_cert_validity_days ({})",
318            input.max_cert_validity_days, input.default_cert_validity_days,
319        ));
320    }
321
322    if input.lockout_backoff_multiplier < 1.0 {
323        violations.push(format!(
324            "lockout_backoff_multiplier ({}) must be >= 1.0",
325            input.lockout_backoff_multiplier,
326        ));
327    }
328
329    if input.access_token_lifetime_secs == 0 {
330        violations.push("access_token_lifetime_secs must be > 0".into());
331    }
332
333    if input.refresh_token_lifetime_secs == 0 {
334        violations.push("refresh_token_lifetime_secs must be > 0".into());
335    }
336
337    if input.mfa_challenge_lifetime_secs == 0 {
338        violations.push("mfa_challenge_lifetime_secs must be > 0".into());
339    }
340
341    if violations.is_empty() {
342        Ok(())
343    } else {
344        Err(AxiamError::Validation {
345            message: format!("Invalid org settings: {}", violations.join("; "),),
346        })
347    }
348}

454/// Validate that a tenant override is only **more restrictive** than
455/// the org baseline. Collects all violations into one error message.
456///
457/// Rules:
458/// - `tenant >= org` for: min_length, password_history_count,
459///   lockout_duration_secs, lockout_backoff_multiplier,
460///   max_lockout_duration_secs
461/// - `tenant <= org` for: max_failed_login_attempts,
462///   access_token_lifetime_secs, refresh_token_lifetime_secs,
463///   mfa_challenge_lifetime_secs, default_cert_validity_days,
464///   max_cert_validity_days, email_verification_grace_period_hours
465/// - enable-only (false->true OK, true->false NOT OK):
466///   require_uppercase/lowercase/digits/symbols, mfa_enforced,
467///   hibp_check_enabled, email_verification_required,
468///   admin_notifications_enabled
469pub fn validate_tenant_override(
470    org: &SecuritySettings,
471    overrides: &TenantSettingsOverride,
472) -> AxiamResult<()> {
473    let mut violations = Vec::new();
474
475    // --- tenant >= org (higher minimum is more restrictive) ---
476    macro_rules! check_min {
477        ($field:ident, $org_path:expr, $label:expr) => {
478            if let Some(val) = overrides.$field {
479                if val < $org_path {
480                    violations.push(format!(
481                        "{}: tenant value {} is less restrictive \
482                         than org baseline {}",
483                        $label, val, $org_path,
484                    ));
485                }
486            }
487        };
488    }
489
490    check_min!(min_length, org.password.min_length, "min_length");
491    check_min!(
492        password_history_count,
493        org.password.password_history_count,
494        "password_history_count"
495    );
496    check_min!(
497        lockout_duration_secs,
498        org.lockout.lockout_duration_secs,
499        "lockout_duration_secs"
500    );
501    check_min!(
502        max_lockout_duration_secs,
503        org.lockout.max_lockout_duration_secs,
504        "max_lockout_duration_secs"
505    );
506
507    // lockout_backoff_multiplier (f64 — compare with partial_cmp)
508    if let Some(val) = overrides.lockout_backoff_multiplier
509        && val < org.lockout.lockout_backoff_multiplier
510    {
511        violations.push(format!(
512            "lockout_backoff_multiplier: tenant value {} is \
513             less restrictive than org baseline {}",
514            val, org.lockout.lockout_backoff_multiplier,
515        ));
516    }
517
518    // --- tenant <= org (lower max / shorter lifetime is more restrictive) ---
519    macro_rules! check_max {
520        ($field:ident, $org_path:expr, $label:expr) => {
521            if let Some(val) = overrides.$field {
522                if val > $org_path {
523                    violations.push(format!(
524                        "{}: tenant value {} is less restrictive \
525                         than org baseline {}",
526                        $label, val, $org_path,
527                    ));
528                }
529            }
530        };
531    }
532
533    check_max!(
534        max_failed_login_attempts,
535        org.lockout.max_failed_login_attempts,
536        "max_failed_login_attempts"
537    );
538    check_max!(
539        access_token_lifetime_secs,
540        org.token.access_token_lifetime_secs,
541        "access_token_lifetime_secs"
542    );
543    check_max!(
544        refresh_token_lifetime_secs,
545        org.token.refresh_token_lifetime_secs,
546        "refresh_token_lifetime_secs"
547    );
548    check_max!(
549        mfa_challenge_lifetime_secs,
550        org.mfa.mfa_challenge_lifetime_secs,
551        "mfa_challenge_lifetime_secs"
552    );
553    check_max!(
554        default_cert_validity_days,
555        org.certificate.default_cert_validity_days,
556        "default_cert_validity_days"
557    );
558    check_max!(
559        max_cert_validity_days,
560        org.certificate.max_cert_validity_days,
561        "max_cert_validity_days"
562    );
563    check_max!(
564        email_verification_grace_period_hours,
565        org.email.email_verification_grace_period_hours,
566        "email_verification_grace_period_hours"
567    );
568
569    // --- enable-only (false->true OK, true->false NOT OK) ---
570    macro_rules! check_enable_only {
571        ($field:ident, $org_val:expr, $label:expr) => {
572            if let Some(val) = overrides.$field {
573                if $org_val && !val {
574                    violations.push(format!(
575                        "{}: cannot disable at tenant level when \
576                         enabled at org level",
577                        $label,
578                    ));
579                }
580            }
581        };
582    }
583
584    check_enable_only!(
585        require_uppercase,
586        org.password.require_uppercase,
587        "require_uppercase"
588    );
589    check_enable_only!(
590        require_lowercase,
591        org.password.require_lowercase,
592        "require_lowercase"
593    );
594    check_enable_only!(
595        require_digits,
596        org.password.require_digits,
597        "require_digits"
598    );
599    check_enable_only!(
600        require_symbols,
601        org.password.require_symbols,
602        "require_symbols"
603    );
604    check_enable_only!(mfa_enforced, org.mfa.mfa_enforced, "mfa_enforced");
605    check_enable_only!(
606        hibp_check_enabled,
607        org.password.hibp_check_enabled,
608        "hibp_check_enabled"
609    );
610    check_enable_only!(
611        email_verification_required,
612        org.email.email_verification_required,
613        "email_verification_required"
614    );
615    check_enable_only!(
616        admin_notifications_enabled,
617        org.notification.admin_notifications_enabled,
618        "admin_notifications_enabled"
619    );
620
621    // --- OPAQUE: tighten-only ---
622    //
623    // `OpaqueMode` derives `Ord` as Disabled < Optional < Required, which is
624    // exactly the restrictiveness order, so this is a plain comparison. Suite
625    // and KSF strength are ranked explicitly rather than by declaration order,
626    // so that adding a variant forces a decision about where it sits.
627    if let Some(mode) = overrides.opaque_mode
628        && mode < org.opaque.opaque_mode
629    {
630        violations.push(format!(
631            "opaque_mode: tenant value {} is less restrictive than org baseline {}",
632            mode, org.opaque.opaque_mode,
633        ));
634    }
635    if let Some(suite) = overrides.opaque_suite
636        && !opaque_suite_is_at_least(suite, org.opaque.opaque_suite)
637    {
638        violations.push(format!(
639            "opaque_suite: tenant value {} is weaker than org baseline {}",
640            suite, org.opaque.opaque_suite,
641        ));
642    }
643    if let Some(ksf) = overrides.opaque_ksf
644        && !opaque_ksf_is_at_least(ksf, org.opaque.opaque_ksf)
645    {
646        violations.push(format!(
647            "opaque_ksf: tenant value {} is weaker than org baseline {}",
648            ksf, org.opaque.opaque_ksf,
649        ));
650    }
651
652    if !violations.is_empty() {
653        return Err(AxiamError::Validation {
654            message: format!(
655                "Tenant override violates org baseline: {}",
656                violations.join("; "),
657            ),
658        });
659    }
660
661    // Cross-field invariant check: merge org + overrides and verify
662    // the effective policy is internally consistent.
663    let merged = effective_settings(org, overrides, Uuid::nil(), Uuid::nil());
664    let mut cross = Vec::new();
665
666    // Non-zero lifetime invariants (0 passes "more restrictive" checks
667    // but produces an unusable policy).
668    if merged.token.access_token_lifetime_secs == 0 {
669        cross.push("effective access_token_lifetime_secs must be > 0".into());
670    }
671    if merged.token.refresh_token_lifetime_secs == 0 {
672        cross.push("effective refresh_token_lifetime_secs must be > 0".into());
673    }
674    if merged.mfa.mfa_challenge_lifetime_secs == 0 {
675        cross.push("effective mfa_challenge_lifetime_secs must be > 0".into());
676    }
677
678    if merged.lockout.max_lockout_duration_secs < merged.lockout.lockout_duration_secs {
679        cross.push(format!(
680            "effective max_lockout_duration_secs ({}) must be >= \
681             lockout_duration_secs ({})",
682            merged.lockout.max_lockout_duration_secs, merged.lockout.lockout_duration_secs,
683        ));
684    }
685    if merged.certificate.max_cert_validity_days < merged.certificate.default_cert_validity_days {
686        cross.push(format!(
687            "effective max_cert_validity_days ({}) must be >= \
688             default_cert_validity_days ({})",
689            merged.certificate.max_cert_validity_days,
690            merged.certificate.default_cert_validity_days,
691        ));
692    }
693
694    if cross.is_empty() {
695        Ok(())
696    } else {
697        Err(AxiamError::Validation {
698            message: format!(
699                "Tenant override produces inconsistent effective policy: {}",
700                cross.join("; "),
701            ),
702        })
703    }
704}

706/// Compute the diff between a complete tenant settings row and the
707/// org baseline, producing a `TenantSettingsOverride` with only the
708/// fields that differ set to `Some`.
709pub fn diff_against_org(
710    org: &SecuritySettings,
711    tenant: &SecuritySettings,
712) -> TenantSettingsOverride {
713    macro_rules! diff {
714        ($field:ident, $org_path:expr, $tenant_path:expr) => {
715            if $tenant_path != $org_path {
716                Some($tenant_path)
717            } else {
718                None
719            }
720        };
721    }
722
723    TenantSettingsOverride {
724        min_length: diff!(
725            min_length,
726            org.password.min_length,
727            tenant.password.min_length
728        ),
729        require_uppercase: diff!(
730            require_uppercase,
731            org.password.require_uppercase,
732            tenant.password.require_uppercase
733        ),
734        require_lowercase: diff!(
735            require_lowercase,
736            org.password.require_lowercase,
737            tenant.password.require_lowercase
738        ),
739        require_digits: diff!(
740            require_digits,
741            org.password.require_digits,
742            tenant.password.require_digits
743        ),
744        require_symbols: diff!(
745            require_symbols,
746            org.password.require_symbols,

758        ),
759        mfa_enforced: diff!(mfa_enforced, org.mfa.mfa_enforced, tenant.mfa.mfa_enforced),

761            mfa_challenge_lifetime_secs,

834/// Build a `SecuritySettings` from a `SetOrgSettings` input.
835pub fn settings_from_org_input(id: Uuid, org_id: Uuid, input: &SetOrgSettings) -> SecuritySettings {
836    let now = Utc::now();
837    SecuritySettings {
838        id,
839        scope: SettingsScope::Org,
840        scope_id: org_id,
841        password: PasswordPolicy {
842            min_length: input.min_length,
843            require_uppercase: input.require_uppercase,
844            require_lowercase: input.require_lowercase,
845            require_digits: input.require_digits,
846            require_symbols: input.require_symbols,
847            password_history_count: input.password_history_count,
848            hibp_check_enabled: input.hibp_check_enabled,
849        },
850        mfa: MfaPolicy {
851            mfa_enforced: input.mfa_enforced,
852            mfa_challenge_lifetime_secs: input.mfa_challenge_lifetime_secs,
853        },
854        lockout: LockoutPolicy {
855            max_failed_login_attempts: input.max_failed_login_attempts,
856            lockout_duration_secs: input.lockout_duration_secs,
857            lockout_backoff_multiplier: input.lockout_backoff_multiplier,
858            max_lockout_duration_secs: input.max_lockout_duration_secs,
859        },
860        token: TokenPolicy {
861            access_token_lifetime_secs: input.access_token_lifetime_secs,
862            refresh_token_lifetime_secs: input.refresh_token_lifetime_secs,
863        },
864        email: EmailVerificationPolicy {
865            email_verification_required: input.email_verification_required,
866            email_verification_grace_period_hours: input.email_verification_grace_period_hours,
867        },
868        certificate: CertificatePolicy {
869            default_cert_validity_days: input.default_cert_validity_days,
870            max_cert_validity_days: input.max_cert_validity_days,
871        },
872        notification: NotificationPolicy {
873            admin_notifications_enabled: input.admin_notifications_enabled,
874        },
875        opaque: OpaquePolicy {
876            opaque_mode: input.opaque_mode,
877            opaque_suite: input.opaque_suite,
878            opaque_ksf: input.opaque_ksf,
879        },
880        created_at: now,
881        updated_at: now,
882    }
883}
