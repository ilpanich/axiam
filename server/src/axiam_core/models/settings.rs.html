77/// Data-retention rules that apply after a subject asks to be erased.
78#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, utoipa::ToSchema)]
79pub struct PrivacyPolicy {
80    /// How long a requested account erasure stays cancellable before the purge
81    /// runs, in days.
82    ///
83    /// The window exists so an erasure triggered by mistake, or under
84    /// coercion, can be undone — `POST /api/v1/auth/account/delete/cancel`
85    /// works for exactly this long. It was fixed at 30 days in the handler,
86    /// which meant the "cancel a pending deletion" control in the admin UI
87    /// referred to a duration no operator could see or change.
88    ///
89    /// Shorter is the more restrictive direction, so a tenant may lower it and
90    /// not raise it: it is time spent holding data the subject has already
91    /// asked to have erased, and GDPR Art. 17(1) asks for that to be
92    /// "without undue delay". The upper bound of 90 days is where Art. 12(3)'s
93    /// one-month response deadline plus its two-month extension for complex
94    /// cases runs out; anything past 30 wants a reason recorded.
95    pub deletion_grace_period_days: u32,
96}

142/// Whether, and on what terms, a client may register itself through
143/// RFC 7591 dynamic client registration (T21.4).
144///
145/// Three values rather than a `bool` plus a second `bool`, because the middle
146/// one is the whole point: a deployment that wants MCP Inspector to register
147/// itself during a demonstration and a deployment that wants Claude Code to
148/// register itself in production want different things, and the difference is
149/// whether an administrator handed out a credential first.
150///
151/// [`Disabled`](Self::Disabled) is the serde default and the system default
152/// (I1). `POST /oauth2/register` exists on every deployment and answers `403`
153/// on every one that has not chosen otherwise — the path is there, the feature
154/// is not, and a scanner learns nothing from the difference because there is
155/// none to learn.
156#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize, utoipa::ToSchema)]
157#[serde(rename_all = "snake_case")]
158pub enum DynamicRegistrationMode {
159    /// No self-registration. Every client is an administrator's decision,
160    /// which is what every AXIAM deployment does today.
161    #[default]
162    Disabled,
163    /// Self-registration, but only for a caller presenting a single-use
164    /// initial access token minted by `POST /oauth2-clients/registration-tokens`.
165    /// RFC 7591 §1.2's "protected" profile: the endpoint is open, the act is
166    /// not.
167    InitialAccessToken,
168    /// Self-registration by anybody who can reach the endpoint. RFC 7591 §1.2's
169    /// "open" profile, and the one MCP Inspector and the desktop clients
170    /// actually use. Refused while
171    /// [`OidcPolicy::external_client_allowed_resources`] is empty — see D3.
172    Anonymous,
173}
174
175impl DynamicRegistrationMode {
176    /// The stored/wire spelling.
177    pub const fn as_str(self) -> &'static str {
178        match self {
179            Self::Disabled => "disabled",
180            Self::InitialAccessToken => "initial_access_token",
181            Self::Anonymous => "anonymous",
182        }
183    }
184
185    /// Parse a stored/wire value. `None` for anything unrecognised, for the
186    /// reason `ClientProfile::from_wire` gives: a typo that degraded to a
187    /// permissive default would open an unauthenticated write endpoint.
188    pub fn from_wire(raw: &str) -> Option<Self> {
189        match raw.trim().to_ascii_lowercase().as_str() {
190            "disabled" => Some(Self::Disabled),
191            "initial_access_token" => Some(Self::InitialAccessToken),
192            "anonymous" => Some(Self::Anonymous),
193            _ => None,
194        }
195    }
196
197    /// Whether the endpoint does anything at all on this policy.
198    pub const fn is_enabled(self) -> bool {
199        !matches!(self, Self::Disabled)
200    }
201
202    /// How permissive this mode is, for the tenant-override ordering.
203    ///
204    /// The only field of [`OidcPolicy`] added by T21.4 that *has* a
205    /// restrictiveness: `disabled` refuses everybody, `anonymous` refuses
206    /// nobody, and `initial_access_token` sits between them because the set of
207    /// callers it admits is a subset of `anonymous`'s and a superset of
208    /// `disabled`'s. A tenant may move down this ladder and never up.
209    const fn permissiveness(self) -> u8 {
210        match self {
211            Self::Disabled => 0,
212            Self::InitialAccessToken => 1,
213            Self::Anonymous => 2,
214        }
215    }
216}

224/// Whether `tenant` admits no more callers than `org` does.
225pub const fn dynamic_registration_is_at_most(

232/// How many self-registered clients a tenant may hold when nothing says

234///

242/// How long a self-registered client survives without being authorized, in
243/// days, when nothing says otherwise (T21.4).
244///
245/// Thirty days: long enough that a client somebody uses monthly is not swept
246/// out from under them, short enough that a registration made once by a tool
247/// nobody kept does not sit in the table forever. The sweeper only ever
248/// touches `managed_by: dcr` rows — see `OAuth2Client::managed_by`.
249pub const DEFAULT_DCR_UNUSED_CLIENT_TTL_DAYS: u32 = 30;
250
251/// How long a self-registered client that was **never authorized** survives,
252/// in seconds (T21.8 / MCP-05). One hour.
253///
254/// # Why a second clock at all
255///
256/// [`DEFAULT_DCR_UNUSED_CLIENT_TTL_DAYS`] is sized, in its own doc comment,
257/// for "a client somebody uses monthly". A client registered and never
258/// authorized is not that client: every MCP client Phase 21 exists to serve —
259/// Inspector, Claude Code, VS Code — authorizes within seconds of registering,
260/// because registration is the first step of the same flow. A never-authorized
261/// row that is an hour old is either abandoned or hostile.
262///
263/// One TTL was serving two situations that have nothing in common, and that is
264/// what made [`OidcPolicy::dcr_max_clients`] an *availability* budget as well
265/// as a storage one: in `anonymous` mode a stranger could fill the quota in
266/// about four minutes and hold it for thirty days. The sweeper could always
267/// tell the two situations apart with no new data, because the row carries
268/// `last_authorized_at: None` — so the fix is a second clock rather than a
269/// bigger quota.
270///
271/// # Why a constant rather than a tenant setting
272///
273/// **Every other sweep window in AXIAM is a tenant setting, and this one
274/// should be too.** The owner of the decision is the tenant, not the
275/// datastore, which is the argument the sweeper's own doc comment makes about
276/// [`DEFAULT_DCR_UNUSED_CLIENT_TTL_DAYS`]; and this is the one sweep that
277/// deletes rows created by strangers, so it is the last window an operator
278/// should be unable to see or change. The T21.8 fix plan's §4 recommends the
279/// field for exactly those reasons and this constant is its stated fallback.
280///
281/// What the plan got wrong is the cost. Its §4 recorded "no migration —
282/// `OidcPolicy` lives in the settings JSON; a missing key deserialises to the
283/// default". That is true of [`OidcPolicy::cimd`], which is one
284/// `oidc_cimd_json` column, and of the tenant override, which is
285/// `overrides_json`. It is **not** true of `OidcPolicy`'s scalars: they are
286/// individual columns on a `SCHEMAFULL` `security_settings` table
287/// (`oidc_dcr_max_clients`, `oidc_dcr_unused_client_ttl_days`, both added by
288/// migration v64), so a fifth DCR number is a `DEFINE FIELD`, a migration
289/// v66 and a bump to the schema tripwire.
290///
291/// The field is therefore the maintainer's call and not this task's, and it is
292/// one migration rather than one line away. Everything else about it is
293/// already written here: the value, the mode gate, the sweeper's predicate and
294/// its tests are identical either way, so promoting the constant to
295/// `OidcPolicy::dcr_unauthorized_client_ttl_secs` is v66, eight mirrored
296/// sites, an ordering map, a range check, the admin card and a spec
297/// regeneration — and no change at all to the behaviour below.
298pub const DCR_UNAUTHORIZED_CLIENT_TTL_SECS: u32 = 3_600;

300/// OpenID Connect surface controls (X7 G8, plan §4.6/§4.8; T21.4).
301///
302/// Settings that are not password rules, here because this is the
303/// org-baseline-plus-tenant-override surface every other per-tenant control
304/// lives on. They are not all of the same kind as each other, and which is
305/// which is the whole of what [`validate_tenant_override`] and
306/// [`clamp_overrides_to_org`] read, so it is set out rather than inferred.
307///
308/// **Ordered** — a tenant may be stricter than its organization and never more
309/// permissive:
310///
311/// * [`Self::sensitive_scopes_enabled`], validated **disable-only** — the
312///   mirror image of `mfa_enforced`, because releasing personal data is the
313///   less-restrictive direction, so a tenant can turn its organization's
314///   decision off but never on.
315/// * [`Self::dynamic_registration`], on the ladder
316///   `disabled` → `initial_access_token` → `anonymous`: a tenant may move down
317///   it and never up.
318/// * [`Self::dcr_max_clients`] and [`Self::dcr_unused_client_ttl_days`], on the
319///   ordinary `tenant <= org` rule — with the wrinkle that `0` on the second
320///   means *never sweep*, which is the longest window of all and is handled by
321///   [`dcr_ttl_strictness`].
322///
323/// **Not ordered**, therefore never validated against the baseline and never
324/// clamped:
325///
326/// * [`Self::default_locale`]. A language is a presentation preference; there
327///   is no sense in which Italian is stricter than French.
328/// * [`Self::dcr_allowed_scopes`], [`Self::dcr_allowed_redirect_hosts`] and
329///   [`Self::external_client_allowed_resources`]. Each names per-tenant
330///   resources — *this* tenant's MCP servers, *this* tenant's callback hosts —
331///   and there is no sense in which one such list is stricter than another. A
332///   subset rule would force an organization to enumerate every tenant's
333///   resource servers in its own baseline before any tenant could name one.
334///
335/// The model's rule is "a tenant may only be more restrictive", which binds
336/// every field that *has* a restrictiveness; a field that has none cannot
337/// violate it.
338///
339/// One cross-field interlock spans both groups and is checked on the resolved
340/// policy rather than on either input: see [`validate_dcr_policy`].
341#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, utoipa::ToSchema)]
342pub struct OidcPolicy {
343    /// Whether `address` and `phone` may be registered on a client, requested
344    /// at the authorization endpoint, and released at UserInfo (X7 G8).
345    ///
346    /// **Off unless an organization turns it on.** The two scopes release a
347    /// postal address and a telephone number — categories of personal data
348    /// AXIAM has no other use for — so the deployment that has never thought
349    /// about them releases nothing, and the operator who has thought about
350    /// them says so once, at the organization level, where the lawful basis
351    /// for holding the data was decided.
352    ///
353    /// The switch is a *capability*, not a grant: with it on, a client still
354    /// has to register the scope, the request still has to ask for it, and the
355    /// user still has to have consented. It is the first of four gates, and it
356    /// is the only one an operator can close for everybody at once.
357    pub sensitive_scopes_enabled: bool,
358    /// The BCP 47 tag the sign-in page falls back to when the relying party's
359    /// `ui_locales` selects nothing (W5's chain, plan §4.6).
360    ///
361    /// `None` means "no tenant preference", which lands on the deployment
362    /// default (`en`) — the behaviour every deployment had before this field
363    /// existed. A tag this build does not ship also lands there: the parse is
364    /// exact rather than a language lookup, so a stored `fr-CA` reads as
365    /// "somebody wrote something this binary does not ship" rather than as a
366    /// guess at French.
367    ///
368    /// Stored as a string rather than as the `Locale` enum because that enum
369    /// lives in `axiam-oauth2`, four layers above this crate, and the crate
370    /// layering points inward.
371    #[serde(default, skip_serializing_if = "Option::is_none")]
372    pub default_locale: Option<String>,
373    /// T21.4 — whether a client may register itself (RFC 7591), and on what
374    /// terms. `disabled` unless somebody says otherwise (I1).
375    #[serde(default)]
376    #[schema(value_type = String, example = "disabled")]
377    pub dynamic_registration: DynamicRegistrationMode,
378    /// T21.4 — the scopes a self-registered client may ask for. A `scope` a
379    /// registration names that is not on this list is
380    /// `invalid_client_metadata`; an empty list means a self-registered client
381    /// gets no scopes at all, which is the honest default for a tenant that
382    /// has turned registration on without deciding what it grants.
383    ///
384    /// May not contain `address` or `phone` — see this module's
385    /// [`sensitive_scope_in_dcr_list`].
386    #[serde(default)]
387    pub dcr_allowed_scopes: Vec<String>,
388    /// T21.4 — hosts a self-registered client's `redirect_uris` may point at,
389    /// as globs (`*.example.com`, or `*` for any). The loopback hosts
390    /// (`127.0.0.1`, `[::1]`, `localhost`) are always allowed whatever this
391    /// says, because RFC 8252 §7.3 is how every desktop MCP client receives
392    /// its callback and a tenant that forbade them would have turned
393    /// registration on for nobody.
394    #[serde(default)]
395    pub dcr_allowed_redirect_hosts: Vec<String>,
396    /// **D3** — the audiences an externally registered client may address.
397    ///
398    /// The single most important field on this policy, and the reason the
399    /// settings handler refuses `dynamic_registration: anonymous` while it is
400    /// empty. A client an unrelated party registered cannot declare its own
401    /// `allowed_resources`; it inherits this list verbatim, so what a stranger
402    /// can mint a token *for* is a decision the tenant took in advance rather
403    /// than one the registration request makes.
404    ///
405    /// Empty means an externally registered client can obtain only today's
406    /// `axiam:user` tokens — which AXIAM's own APIs accept. That is why the
407    /// interlock exists: the empty list is not a safe default for an *open*
408    /// registration endpoint, it is the most dangerous one.
409    ///
410    /// Shared with T5 (CIMD), which inherits the same list for the same
411    /// reason.
412    #[serde(default)]
413    pub external_client_allowed_resources: Vec<String>,
414    /// T21.4 — how many externally registered clients this tenant may hold.
415    /// See [`DEFAULT_DCR_MAX_CLIENTS`].
416    ///
417    /// **Counted once per mechanism, against the same number** (T21.8):
418    /// `managed_by: dcr` rows and `managed_by: cimd` rows each have this many.
419    /// So a tenant running both cannot have shadow rows materialised from
420    /// documents exhaust the allowance for self-registration, or the reverse.
421    /// The CIMD count is checked *before* the document is fetched, so a tenant
422    /// at its ceiling is not an outbound amplifier either. It keeps its `dcr_`
423    /// name because dynamic registration defined it, on the same precedent as
424    /// [`Self::dcr_allowed_scopes`].
425    #[serde(default = "default_dcr_max_clients")]
426    #[schema(example = 20)]
427    pub dcr_max_clients: u32,
428    /// T21.4 — how long an externally registered client survives without
429    /// being used. See [`DEFAULT_DCR_UNUSED_CLIENT_TTL_DAYS`]. `0` disables
430    /// the sweep for this tenant, which an operator who prunes out of band
431    /// may legitimately want.
432    ///
433    /// **Two sweeps read it, over different clocks** (T21.8). A
434    /// `managed_by: dcr` row is measured from its last authorization, falling
435    /// back to when it was registered. A `managed_by: cimd` row is measured
436    /// from the last time its document was *presented*, which every authorize,
437    /// token and PAR request moves — so a document in daily use is never swept
438    /// however old its registration is, and one nobody has presented since the
439    /// window is, and re-materialises on the next request if it is still
440    /// published. Like the ceiling, it keeps its `dcr_` name.
441    #[serde(default = "default_dcr_unused_client_ttl_days")]
442    #[schema(example = 30)]
443    pub dcr_unused_client_ttl_days: u32,
444    /// T21.5 — whether a URL-shaped `client_id` is resolved by fetching the
445    /// document it names, and on what terms. See [`CimdPolicy`]; off unless
446    /// somebody turns it on (I1).
447    ///
448    /// Nested, and therefore inherited or overridden **whole**: the fields are
449    /// terms of one decision, and a half-merged posture is one neither the
450    /// organization nor the tenant wrote.
451    #[serde(default)]
452    pub cimd: CimdPolicy,
453}

469/// How strict a `dcr_unused_client_ttl_days` value is, as a number that
470/// increases with permissiveness (T21.4).
471///
472/// `0` means "never sweep", which is the **most** permissive value and not the
473/// least — an unused self-registered client kept forever is exactly what the
474/// TTL exists to prevent. A plain `tenant <= org` comparison would therefore
475/// have read `0` as the strictest possible override and let a tenant turn the
476/// sweeper off under an organization that had turned it on. Mapping `0` to the

478const fn dcr_ttl_strictness(days: u32) -> u32 {

482/// Everything wrong with a resolved dynamic-registration policy, as operator
483/// sentences (T21.4).
484///
485/// A function of the **effective** policy rather than of an input, because
486/// both ways in produce one: an organization baseline is a policy, and a
487/// tenant override merged onto its baseline is a policy. The interlock this
488/// enforces would be trivially bypassable otherwise — a tenant that set
489/// `anonymous` while inheriting an empty
490/// [`OidcPolicy::external_client_allowed_resources`] from its organization
491/// would pass a check that only looked at what the request carried.
492///
493/// # The D3 interlock
494///
495/// `anonymous` registration with an empty `external_client_allowed_resources`
496/// is refused. This is a security control, not a validation nicety. An
497/// externally registered client inherits that list as its
498/// `allowed_resources`, and an empty one leaves it able to obtain only
499/// `axiam:user` tokens — the audience **AXIAM's own APIs accept**. So the
500/// combination is not "a registration endpoint that grants nothing"; it is an
501/// unauthenticated endpoint that mints clients able to ask for tokens against
502/// AXIAM itself. The operator has to name the MCP servers this tenant fronts
503/// before strangers may register for them.
504///
505/// `initial_access_token` is deliberately **not** interlocked: a caller there
506/// presented a credential an administrator minted, so an administrator has
507/// already decided this registration should happen, and a deployment that
508/// wants exactly today's `axiam:user` behaviour for a hand-issued client is
509/// making a choice rather than leaving a door open.
510pub fn validate_dcr_policy(oidc: &OidcPolicy) -> Vec<String> {
511    let mut violations = Vec::new();
512
513    if oidc.dynamic_registration == DynamicRegistrationMode::Anonymous
514        && oidc.external_client_allowed_resources.is_empty()
515    {
516        violations.push(
517            "dynamic_registration: anonymous registration cannot be enabled while \
518             external_client_allowed_resources is empty (D3). A client registered by an \
519             unrelated party inherits that list as its allowed_resources, and an empty list \
520             leaves it able to obtain only the axiam:user tokens AXIAM's own APIs accept. \
521             Name the MCP servers this tenant fronts first"
522                .into(),
523        );
524    }

526    if let Some(scope) = sensitive_scope_in_dcr_list(&oidc.dcr_allowed_scopes) {
527        violations.push(format!(
528            "dcr_allowed_scopes: {scope} releases personal data under W7's per-client consent \
529             record and cannot be offered to a self-registered client, which already carries a \
530             forced consent record of its own (D4). Register a client for it through \
531             POST /oauth2-clients instead"
532        ));
533    }
534
535    violations
536}

538/// The GDPR-sensitive scope a `dcr_allowed_scopes` list names, if it names one
539/// (T21.4 amendment 1).
540///
541/// W7 gates `address` and `phone` behind a consent record per relying party,
542/// and D4 gates every externally registered client behind a consent record of
543/// its own. Both records live in the `oidc_scope_release:<client_id>`
544/// namespace, so a self-registered client holding `address` would need two of
545/// them — two consent screens for one authorization, or one record standing
546/// for the other while the UserInfo gate, which re-reads the sensitive record
547/// on every call, releases nothing anyway.
548///
549/// Refusing the combination is also the answer D3's reasoning points at on its
550/// own: a party that registered itself, unauthenticated, must not be able to
551/// *ask* for a postal address, whatever the tenant's sensitive-scope switch
552/// says.
553///
554/// The two names are written here rather than taken from
555/// `axiam_oauth2::sensitive::SENSITIVE_SCOPES` because that constant lives
556/// four layers out and this crate is layer 0. `axiam-oauth2`'s own test
557/// asserts the two lists agree.
558pub fn sensitive_scope_in_dcr_list(scopes: &[String]) -> Option<&str> {
559    scopes
560        .iter()
561        .find(|s| matches!(s.trim(), "address" | "phone"))
562        .map(String::as_str)
563}

569/// The shortest cache lifetime a tenant may give a client metadata document,
570/// in seconds.
571///
572/// A bound rather than a preference. The document is fetched from a URL an
573/// **unauthenticated** caller chooses, so the cache lifetime is what stands
574/// between one authorization request and one outbound HTTP request: a tenant
575/// that set it to zero would have turned its authorization endpoint into a
576/// request amplifier pointed at whatever host is on its trusted list. Sixty
577/// seconds is the floor; the shipped default is five minutes.
578pub const CIMD_MIN_CACHE_FLOOR_SECS: u64 = 60;
579
580/// The longest cache lifetime a tenant may give a client metadata document,
581/// in seconds (seven days).
582///
583/// The other end of the same control. A cached document is a *live client
584/// registration* that nobody at this deployment created, so the ceiling is the
585/// longest a stranger may pin one for after taking their document down. Seven
586/// days is an upper bound on the tenant's own `max_cache_secs`, whose shipped
587/// default is three.
588pub const CIMD_MAX_CACHE_CEILING_SECS: u64 = 604_800;
589
590/// The largest `max_metadata_bytes` a tenant may configure (64 KiB).
591///
592/// An unbounded read of an attacker-chosen URL is a memory-exhaustion
593/// primitive, and a tenant-configurable cap with no ceiling is an unbounded
594/// read with extra steps. A client metadata document that does not fit in
595/// 64 KiB is not a client metadata document; the shipped default is 5 000
596/// bytes, which is the draft's own suggestion and roughly ten times what a
597/// real one weighs.
598pub const CIMD_MAX_METADATA_BYTES_CEILING: u64 = 65_536;
599
600/// The shipped cache bounds and size cap. See [`CimdPolicy`].
601pub const DEFAULT_CIMD_MIN_CACHE_SECS: u64 = 300;
602/// See [`DEFAULT_CIMD_MIN_CACHE_SECS`].
603pub const DEFAULT_CIMD_MAX_CACHE_SECS: u64 = 259_200;
604/// See [`DEFAULT_CIMD_MIN_CACHE_SECS`].
605pub const DEFAULT_CIMD_MAX_METADATA_BYTES: u64 = 5_000;

607/// Whether, and on what terms, a `client_id` that is a URL is resolved by
608/// fetching the document it names (T21.5,
609/// `draft-ietf-oauth-client-id-metadata-document`).
610///
611/// # Why this is one nested policy rather than nine fields
612///
613/// Every field here is a term of a single decision — *do we fetch a stranger's
614/// URL and make a client out of what comes back* — and none of them means
615/// anything without [`Self::enabled`]. A tenant that states a CIMD posture
616/// states all of it; a tenant that states none inherits its organization's
617/// whole posture rather than half of one, which is the only merge that cannot
618/// produce a combination neither party wrote.
619///
620/// # The two fields that can widen, and the seven that cannot
621///
622/// [`Self::enabled`] and [`Self::allow_http`] are **ordered**: a tenant may
623/// turn either off but never on, exactly as `dynamic_registration` may only
624/// move down its ladder. Everything else names *this tenant's* domains or
625/// *this tenant's* bounds, and there is no sense in which one tenant's list of
626/// trusted publishers is stricter than another's — the same argument
627/// [`OidcPolicy`] already makes for `dcr_allowed_redirect_hosts`.
628///
629/// # Every bound here is a security control
630///
631/// [`Self::max_metadata_bytes`], [`Self::min_cache_secs`] and
632/// [`Self::max_cache_secs`] are not tuning knobs. They are, respectively, the
633/// ceiling on a read from an attacker-chosen URL, the floor under how often
634/// that read may be repeated, and the ceiling on how long its result may be
635/// trusted. Each is clamped again in code against the three constants above,
636/// so a settings row written by hand cannot lift them.
637#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, utoipa::ToSchema)]
638pub struct CimdPolicy {
639    /// **Off unless somebody turns it on** (I1). With this `false`, a
640    /// URL-shaped `client_id` is exactly today's unknown client: nothing is
641    /// fetched, nothing is materialised, and the ordinary repository lookup
642    /// answers as it always has.
643    #[serde(default)]
644    pub enabled: bool,
645    /// Permit an `http://` `client_id` and an `http://` fetch.
646    ///
647    /// **Development only, and it does more than its name says.** AXIAM's
648    /// shared SSRF guard couples the scheme rule to the address rule — the
649    /// same seam that lets an integration test point a fetch at a loopback
650    /// mock server — so a tenant that allows `http` also allows the first hop
651    /// to resolve to a private address. Redirect hops are validated strictly
652    /// whatever this says, and a public deployment that sets it has removed
653    /// the control that makes `169.254.169.254` unreachable.
654    #[serde(default)]
655    pub allow_http: bool,
656    /// The hosts whose documents this tenant will fetch at all, as globs
657    /// (`mcp.example.com`, or `*.example.com` for every host under one
658    /// domain).
659    ///
660    /// **An empty list resolves nothing**, and enabling CIMD while it is empty
661    /// is refused — see [`validate_cimd_policy`]. That is a deliberate
662    /// departure from "a URL is a client identifier, so any URL will do": the
663    /// fetch is triggered by an unauthenticated request naming the URL, so an
664    /// unrestricted list is a request-forgery primitive offered to strangers,
665    /// bounded only by the SSRF guard's address rules. Naming the publishers a
666    /// tenant actually fronts costs one settings field and removes the class.
667    ///
668    /// **`*` is refused here, and so is a wildcard over a whole top-level
669    /// domain** (`*.com`): both are the posture the empty list is refused for,
670    /// spelled differently, and a control with no second control behind it
671    /// cannot have a one-character bypass and still be the control. It is a
672    /// floor and not a public-suffix check — `*.github.io` passes, and
673    /// trusting shared hosting stays the operator's decision, bounded by the
674    /// per-tenant quota rather than by this field. `*` remains valid in
675    /// [`CimdPolicy::trusted_redirect_domains`], whose entries are not fetch
676    /// targets.
677    #[serde(default)]
678    pub trusted_client_id_domains: Vec<String>,
679    /// The hosts a document's `redirect_uris` may point at, as globs.
680    ///
681    /// The loopback hosts (`127.0.0.1`, `[::1]`, `localhost`) are always
682    /// allowed, because RFC 8252 §7.3 is how every desktop MCP client receives
683    /// its callback — so an empty list is not a refusal of everything, it is
684    /// "loopback only", which is exactly the Claude Code and VS Code profile.
685    #[serde(default)]
686    pub trusted_redirect_domains: Vec<String>,
687    /// Require every `redirect_uris` host in the document to equal the host of
688    /// the `client_id` URL itself.
689    ///
690    /// **On by default**, because the document says who the client is and a
691    /// redirect to somewhere else is the one thing a stolen or mirrored
692    /// document would want to change. It is turned **off** for the desktop MCP
693    /// clients, whose callbacks are on loopback and therefore can never share
694    /// a host with a `https://` `client_id`; `docs/admin/client-id-metadata-documents.md`
695    /// says so and says why.
696    #[serde(default = "default_true")]
697    pub restrict_same_domain: bool,
698    /// Refuse a document whose `token_endpoint_auth_method` is `none`.
699    ///
700    /// Off by default, because `none` is what every MCP desktop client is. A
701    /// tenant that turns it on accepts only `private_key_jwt` documents, which
702    /// is the posture for a deployment whose CIMD clients are servers rather
703    /// than desktops.
704    #[serde(default)]
705    pub confidential_only: bool,
706    /// The floor under a document's cache lifetime, in seconds. Clamped to
707    /// [`CIMD_MIN_CACHE_FLOOR_SECS`].
708    #[serde(default = "default_cimd_min_cache_secs")]
709    #[schema(example = 300)]
710    pub min_cache_secs: u64,
711    /// The ceiling on a document's cache lifetime, in seconds. Clamped to
712    /// [`CIMD_MAX_CACHE_CEILING_SECS`].
713    #[serde(default = "default_cimd_max_cache_secs")]
714    #[schema(example = 259_200)]
715    pub max_cache_secs: u64,
716    /// The hard cap on how many bytes of a document are read, before it is
717    /// parsed. Clamped to [`CIMD_MAX_METADATA_BYTES_CEILING`].
718    #[serde(default = "default_cimd_max_metadata_bytes")]
719    #[schema(example = 5_000)]
720    pub max_metadata_bytes: u64,
721}
722
723impl Default for CimdPolicy {
724    fn default() -> Self {
725        Self {
726            enabled: false,
727            allow_http: false,
728            trusted_client_id_domains: Vec::new(),
729            trusted_redirect_domains: Vec::new(),
730            restrict_same_domain: true,
731            confidential_only: false,
732            min_cache_secs: DEFAULT_CIMD_MIN_CACHE_SECS,
733            max_cache_secs: DEFAULT_CIMD_MAX_CACHE_SECS,
734            max_metadata_bytes: DEFAULT_CIMD_MAX_METADATA_BYTES,
735        }
736    }
737}

739impl CimdPolicy {
740    /// The cache lifetime to give a document that advertised `advertised`
741    /// seconds, or that advertised nothing (`None`).
742    ///
743    /// Clamped twice: once to this tenant's own bounds, and once to the
744    /// deployment constants, so neither a hostile `Cache-Control` nor a
745    /// hand-edited settings row can push it out of range. A document that
746    /// advertises nothing gets the floor rather than the ceiling — the least
747    /// trust for the least information.
748    pub fn clamp_cache_secs(&self, advertised: Option<u64>) -> u64 {
749        let floor = self.min_cache_secs.max(CIMD_MIN_CACHE_FLOOR_SECS);
750        let ceiling = self
751            .max_cache_secs
752            .min(CIMD_MAX_CACHE_CEILING_SECS)
753            .max(floor);
754        advertised.unwrap_or(floor).clamp(floor, ceiling)
755    }
756
757    /// The byte cap to read this tenant's documents with, clamped to
758    /// [`CIMD_MAX_METADATA_BYTES_CEILING`]. Zero is read as the shipped
759    /// default rather than as "read nothing", which no operator means.
760    pub fn effective_max_metadata_bytes(&self) -> usize {
761        let raw = if self.max_metadata_bytes == 0 {
762            DEFAULT_CIMD_MAX_METADATA_BYTES
763        } else {
764            self.max_metadata_bytes
765        };
766        usize::try_from(raw.min(CIMD_MAX_METADATA_BYTES_CEILING))
767            .unwrap_or(DEFAULT_CIMD_MAX_METADATA_BYTES as usize)
768    }
769}

792/// Everything wrong with a resolved CIMD policy, as operator sentences
793/// (T21.5).
794///
795/// A function of the **effective** policy, for the reason
796/// [`validate_dcr_policy`] gives: a tenant that enables CIMD while inheriting
797/// an empty resource list from its organization is precisely the state the
798/// first interlock exists to refuse.
799///
800/// # The two interlocks
801///
802/// **D3.** Enabling CIMD with an empty
803/// [`OidcPolicy::external_client_allowed_resources`] is refused, word for word
804/// the reasoning `dynamic_registration: anonymous` is refused for: a client
805/// materialised from a stranger's document inherits that list as its
806/// `allowed_resources`, and an empty list leaves it able to obtain only the
807/// `axiam:user` tokens AXIAM's own APIs accept.
808///
809/// **The trusted-publisher list.** Enabling CIMD with an empty
810/// [`CimdPolicy::trusted_client_id_domains`] is refused, because the fetch is
811/// reachable by an unauthenticated caller who chooses the URL.
812///
813/// The rest are range checks, and they are here rather than left to the
814/// clamping accessors so that an operator who writes an impossible bound is
815/// told, rather than quietly given a different one.
816pub fn validate_cimd_policy(oidc: &OidcPolicy) -> Vec<String> {
817    let mut violations = Vec::new();
818    let cimd = &oidc.cimd;
819
820    if !cimd.enabled {
821        // Nothing below can be reached by a request, so nothing below is
822        // refused. A tenant may stage a CIMD policy before turning it on.
823        return violations;
824    }
825
826    if oidc.external_client_allowed_resources.is_empty() {
827        violations.push(
828            "cimd.enabled: client ID metadata documents cannot be enabled while \
829             external_client_allowed_resources is empty (D3). A client materialised from a \
830             stranger's document inherits that list as its allowed_resources, and an empty \
831             list leaves it able to obtain only the axiam:user tokens AXIAM's own APIs \
832             accept. Name the MCP servers this tenant fronts first"
833                .into(),
834        );
835    }
836
837    if cimd.trusted_client_id_domains.is_empty() {
838        violations.push(
839            "cimd.trusted_client_id_domains: client ID metadata documents cannot be enabled \
840             with no trusted publisher domain. The document is fetched because an \
841             unauthenticated request named its URL, so an unrestricted list is an outbound \
842             fetch a stranger chooses the target of. Name the hosts whose documents this \
843             tenant accepts (globs are allowed: *.example.com)"
844                .into(),
845        );
846    }
847
848    // ...and the same refusal, for the spellings that mean the same thing.
849    //
850    // T21.8 / MCP-03. The refusal above was worth having for one reason: an
851    // unrestricted trusted-publisher list is a request-forgery primitive
852    // offered to strangers **and there is no second control that does that
853    // job** — the SSRF guard bounds which addresses a fetch may reach, not
854    // which hosts a caller may name. A control with no second control behind
855    // it cannot have a one-character bypass and still be the control, and
856    // `["*"]` was exactly that: refused as `[]`, admitted as `["*"]`, with the
857    // validator's own entry-shape message forty lines down recommending the
858    // spelling that produced it.
859    //
860    // A single-label wildcard suffix goes with it. `*.com` is `*` for one
861    // top-level domain, spelled longer, and the honest statement of the
862    // finding is "the list must name a publisher" rather than "the list must
863    // not contain one particular character".
864    //
865    // **It is a floor, and it is stated as one.** This is not a public-suffix
866    // check: `*.github.io` still passes, and so does `*.pages.dev`. Trusting
867    // shared hosting remains the operator's decision to make, and what bounds
868    // it is the per-tenant quota and the sweep, not this condition. If the
869    // narrower fix is wanted — refuse `*` and nothing else — deleting the
870    // `labels` arm below is one line and no test depends on it.
871    //
872    // `trusted_redirect_domains` keeps `*`, because those entries are not
873    // fetch targets: they bound where a *document* may point a browser, the
874    // loopback three are allowed whatever the list says, and an empty list is
875    // a working posture there rather than a refusal.
876    for entry in &cimd.trusted_client_id_domains {
877        let e = entry.trim();
878        let offence = if e == "*" {
879            Some("matches every host")
880        } else if let Some(suffix) = e.strip_prefix("*.")
881            && !suffix.contains('*')
882            && !suffix.is_empty()
883            && suffix.split('.').count() == 1
884        {
885            Some("is a wildcard over a whole top-level domain")
886        } else {
887            None
888        };
889        if let Some(offence) = offence {
890            violations.push(format!(
891                "cimd.trusted_client_id_domains: {entry:?} {offence}, which is the posture an \
892                 empty list is refused for. The document is fetched because an \
893                 unauthenticated request named its URL, so the list has to name a publisher: \
894                 a host (mcp.example.com) or a wildcard over one (*.example.com)"
895            ));
896        }
897    }
898
899    // A host glob, not a URL. `host_glob_matches` answers `false` for an entry
900    // carrying a scheme, a path or a port, so a tenant that typed one would
901    // have a trusted list that silently matches nothing — fail-closed, but
902    // indistinguishable from a working list until somebody tries to sign in.
903    //
904    // The two fields get different advice, because `*` is valid in one of them
905    // and refused in the other (above). Offering it here for
906    // `trusted_client_id_domains` is what made the empty-list refusal
907    // self-defeating in the first place.
908    for (field, entries, forms) in [
909        (
910            "cimd.trusted_client_id_domains",
911            &cimd.trusted_client_id_domains,
912            "Write a host (mcp.example.com) or a leftmost-label wildcard over one \
913             (*.example.com)",
914        ),
915        (
916            "cimd.trusted_redirect_domains",
917            &cimd.trusted_redirect_domains,
918            "Write a host (app.example.com), a leftmost-label wildcard (*.example.com) or *",
919        ),
920    ] {
921        for entry in entries {
922            let e = entry.trim();
923            if e.is_empty()
924                || e.contains("://")
925                || e.contains('/')
926                || e.contains(':')
927                || e.split_whitespace().count() != 1
928            {
929                violations.push(format!(
930                    "{field}: {entry:?} is not a host pattern. {forms} — not a URL, a path or \
931                     a host:port"
932                ));
933            }
934        }
935    }
936
937    if cimd.min_cache_secs < CIMD_MIN_CACHE_FLOOR_SECS {
938        violations.push(format!(
939            "cimd.min_cache_secs ({}) must be >= {CIMD_MIN_CACHE_FLOOR_SECS}: the cache \
940             lifetime is what stands between one authorization request and one outbound \
941             fetch",
942            cimd.min_cache_secs,
943        ));
944    }
945
946    if cimd.max_cache_secs > CIMD_MAX_CACHE_CEILING_SECS {
947        violations.push(format!(
948            "cimd.max_cache_secs ({}) must be <= {CIMD_MAX_CACHE_CEILING_SECS}: a cached \
949             document is a live client registration nobody here created",
950            cimd.max_cache_secs,
951        ));
952    }
953
954    if cimd.min_cache_secs > cimd.max_cache_secs {
955        violations.push(format!(
956            "cimd.min_cache_secs ({}) must be <= cimd.max_cache_secs ({})",
957            cimd.min_cache_secs, cimd.max_cache_secs,
958        ));
959    }
960
961    if cimd.max_metadata_bytes == 0 || cimd.max_metadata_bytes > CIMD_MAX_METADATA_BYTES_CEILING {
962        violations.push(format!(
963            "cimd.max_metadata_bytes ({}) must be between 1 and \
964             {CIMD_MAX_METADATA_BYTES_CEILING}: an unbounded read of an attacker-chosen URL \
965             is a memory-exhaustion primitive",
966            cimd.max_metadata_bytes,
967        ));
968    }
969
970    violations
971}

1290/// Validate internal invariants of organization-level settings.
1291///
1292/// Checks relationships between fields (e.g., max >= min) and rejects
1293/// obviously invalid values (zero lifetimes where required).
1294pub fn validate_org_settings(input: &SetOrgSettings) -> AxiamResult<()> {
1295    let mut violations = Vec::new();
1296
1297    if input.max_lockout_duration_secs < input.lockout_duration_secs {
1298        violations.push(format!(
1299            "max_lockout_duration_secs ({}) must be >= \
1300             lockout_duration_secs ({})",
1301            input.max_lockout_duration_secs, input.lockout_duration_secs,
1302        ));
1303    }
1304
1305    if input.max_cert_validity_days < input.default_cert_validity_days {
1306        violations.push(format!(
1307            "max_cert_validity_days ({}) must be >= \
1308             default_cert_validity_days ({})",
1309            input.max_cert_validity_days, input.default_cert_validity_days,
1310        ));
1311    }
1312
1313    if input.lockout_backoff_multiplier < 1.0 {
1314        violations.push(format!(
1315            "lockout_backoff_multiplier ({}) must be >= 1.0",
1316            input.lockout_backoff_multiplier,
1317        ));
1318    }
1319
1320    if input.access_token_lifetime_secs == 0 {
1321        violations.push("access_token_lifetime_secs must be > 0".into());
1322    }
1323
1324    if input.refresh_token_lifetime_secs == 0 {
1325        violations.push("refresh_token_lifetime_secs must be > 0".into());
1326    }
1327
1328    if input.mfa_challenge_lifetime_secs == 0 {
1329        violations.push("mfa_challenge_lifetime_secs must be > 0".into());
1330    }
1331
1332    // A zero-day window would purge on the same request that scheduled the
1333    // erasure, leaving the cancel link in the confirmation email pointing at
1334    // an account that is already gone.
1335    if input.deletion_grace_period_days == 0 {
1336        violations.push("deletion_grace_period_days must be >= 1".into());
1337    } else if input.deletion_grace_period_days > MAX_DELETION_GRACE_PERIOD_DAYS {
1338        violations.push(format!(
1339            "deletion_grace_period_days ({}) must be <= {} \
1340             (GDPR Art. 12(3): one month, extensible by two)",
1341            input.deletion_grace_period_days, MAX_DELETION_GRACE_PERIOD_DAYS,
1342        ));
1343    }
1344
1345    // T21.4 — the dynamic-registration interlocks, on the policy this input
1346    // resolves to. An organization baseline reaches every tenant that has not
1347    // overridden it, so a baseline naming `anonymous` with no resources is the
1348    // same open door as a tenant one; see `validate_dcr_policy`.
1349    let oidc = OidcPolicy {
1350        sensitive_scopes_enabled: input.sensitive_scopes_enabled,
1351        default_locale: input.default_locale.clone(),
1352        dynamic_registration: input.dynamic_registration,
1353        dcr_allowed_scopes: input.dcr_allowed_scopes.clone(),
1354        dcr_allowed_redirect_hosts: input.dcr_allowed_redirect_hosts.clone(),
1355        external_client_allowed_resources: input.external_client_allowed_resources.clone(),
1356        dcr_max_clients: input.dcr_max_clients,
1357        dcr_unused_client_ttl_days: input.dcr_unused_client_ttl_days,
1358        cimd: input.cimd.clone(),
1359    };
1360    violations.extend(validate_dcr_policy(&oidc));
1361    // T21.5 — the same argument, for the mechanism that reaches further: a
1362    // baseline that enables CIMD with no resource list and no trusted
1363    // publisher opens it for every tenant that has not overridden it.
1364    violations.extend(validate_cimd_policy(&oidc));
1365
1366    if violations.is_empty() {
1367        Ok(())
1368    } else {
1369        Err(AxiamError::Validation {
1370            message: format!("Invalid org settings: {}", violations.join("; "),),
1371        })
1372    }
1373}

1534/// Drop every tenant override that the org baseline has since overtaken.
1535///
1536/// Returns the names of the fields cleared, in declaration order — empty when
1537/// the override was already compliant.
1538///
1539/// # Why this exists
1540///
1541/// [`validate_tenant_override`] enforces "a tenant may only tighten" at the
1542/// moment a tenant writes one. Nothing enforced it afterwards, and an
1543/// organization baseline is not a constant: raise `min_length` to 16, or switch
1544/// `opaque_mode` from `disabled` to `required`, and every tenant that had
1545/// written an override for that field kept the old, weaker value forever. The
1546/// setting appeared to apply — the organization row said so, and tenants with no
1547/// override did inherit it — while the tenants that had ever visited their own
1548/// settings page silently did not. Which tenants those were was not visible
1549/// anywhere.
1550///
1551/// Clearing the field rather than rewriting it to the org value is deliberate:
1552/// an absent override *tracks* the baseline, so the tenant also picks up the
1553/// next change. Writing the value in would freeze it again at the new level and
1554/// reproduce the same defect one baseline later.
1555///
1556/// Overrides that are still **stricter** than the baseline are left exactly as
1557/// they are. A tenant that chose a 24-character minimum does not lose it because
1558/// the organization moved from 12 to 16.
1559pub fn clamp_overrides_to_org(
1560    org: &SecuritySettings,
1561    overrides: &mut TenantSettingsOverride,
1562) -> Vec<&'static str> {
1563    let mut cleared = Vec::new();
1564
1565    // `tenant >= org`: a higher minimum is the more restrictive one.
1566    macro_rules! clamp_min {
1567        ($field:ident, $org_path:expr, $label:expr) => {
1568            if overrides.$field.is_some_and(|val| val < $org_path) {
1569                overrides.$field = None;
1570                cleared.push($label);
1571            }
1572        };
1573    }
1574    clamp_min!(min_length, org.password.min_length, "min_length");
1575    clamp_min!(
1576        password_history_count,
1577        org.password.password_history_count,
1578        "password_history_count"
1579    );
1580    clamp_min!(
1581        lockout_duration_secs,
1582        org.lockout.lockout_duration_secs,
1583        "lockout_duration_secs"
1584    );
1585    clamp_min!(
1586        max_lockout_duration_secs,
1587        org.lockout.max_lockout_duration_secs,
1588        "max_lockout_duration_secs"
1589    );
1590    if overrides
1591        .lockout_backoff_multiplier
1592        .is_some_and(|val| val < org.lockout.lockout_backoff_multiplier)
1593    {
1594        overrides.lockout_backoff_multiplier = None;
1595        cleared.push("lockout_backoff_multiplier");
1596    }
1597
1598    // `tenant <= org`: a lower cap or shorter lifetime is the more restrictive one.
1599    macro_rules! clamp_max {
1600        ($field:ident, $org_path:expr, $label:expr) => {
1601            if overrides.$field.is_some_and(|val| val > $org_path) {
1602                overrides.$field = None;
1603                cleared.push($label);
1604            }
1605        };
1606    }
1607    clamp_max!(
1608        max_failed_login_attempts,
1609        org.lockout.max_failed_login_attempts,
1610        "max_failed_login_attempts"
1611    );
1612    clamp_max!(
1613        access_token_lifetime_secs,
1614        org.token.access_token_lifetime_secs,
1615        "access_token_lifetime_secs"
1616    );
1617    clamp_max!(
1618        refresh_token_lifetime_secs,
1619        org.token.refresh_token_lifetime_secs,
1620        "refresh_token_lifetime_secs"
1621    );
1622    clamp_max!(
1623        mfa_challenge_lifetime_secs,
1624        org.mfa.mfa_challenge_lifetime_secs,
1625        "mfa_challenge_lifetime_secs"
1626    );
1627    clamp_max!(
1628        default_cert_validity_days,
1629        org.certificate.default_cert_validity_days,
1630        "default_cert_validity_days"
1631    );
1632    clamp_max!(
1633        max_cert_validity_days,
1634        org.certificate.max_cert_validity_days,
1635        "max_cert_validity_days"
1636    );
1637    clamp_max!(
1638        email_verification_grace_period_hours,
1639        org.email.email_verification_grace_period_hours,
1640        "email_verification_grace_period_hours"
1641    );
1642    clamp_max!(
1643        deletion_grace_period_days,
1644        org.privacy.deletion_grace_period_days,
1645        "deletion_grace_period_days"
1646    );
1647
1648    // Enable-only: a tenant may turn a control on, never off.
1649    macro_rules! clamp_enable_only {
1650        ($field:ident, $org_val:expr, $label:expr) => {
1651            if $org_val && overrides.$field == Some(false) {
1652                overrides.$field = None;
1653                cleared.push($label);
1654            }
1655        };
1656    }
1657    clamp_enable_only!(
1658        require_uppercase,
1659        org.password.require_uppercase,
1660        "require_uppercase"
1661    );
1662    clamp_enable_only!(
1663        require_lowercase,
1664        org.password.require_lowercase,
1665        "require_lowercase"
1666    );
1667    clamp_enable_only!(
1668        require_digits,
1669        org.password.require_digits,
1670        "require_digits"
1671    );
1672    clamp_enable_only!(
1673        require_symbols,
1674        org.password.require_symbols,
1675        "require_symbols"
1676    );
1677    clamp_enable_only!(mfa_enforced, org.mfa.mfa_enforced, "mfa_enforced");
1678    clamp_enable_only!(
1679        hibp_check_enabled,
1680        org.password.hibp_check_enabled,
1681        "hibp_check_enabled"
1682    );
1683    clamp_enable_only!(
1684        email_verification_required,
1685        org.email.email_verification_required,
1686        "email_verification_required"
1687    );
1688    clamp_enable_only!(
1689        admin_notifications_enabled,
1690        org.notification.admin_notifications_enabled,
1691        "admin_notifications_enabled"
1692    );
1693
1694    // OPAQUE, on the same ordering `validate_tenant_override` uses.
1695    if overrides
1696        .opaque_mode
1697        .is_some_and(|mode| mode < org.opaque.opaque_mode)
1698    {
1699        overrides.opaque_mode = None;
1700        cleared.push("opaque_mode");
1701    }
1702    if overrides
1703        .opaque_suite
1704        .is_some_and(|suite| !opaque_suite_is_at_least(suite, org.opaque.opaque_suite))
1705    {
1706        overrides.opaque_suite = None;
1707        cleared.push("opaque_suite");
1708    }
1709    if overrides
1710        .opaque_ksf
1711        .is_some_and(|ksf| !opaque_ksf_is_at_least(ksf, org.opaque.opaque_ksf))
1712    {
1713        overrides.opaque_ksf = None;
1714        cleared.push("opaque_ksf");
1715    }
1716
1717    // WebAuthn user verification, on the same ordering
1718    // `validate_tenant_override` uses.
1719    if overrides.webauthn_user_verification.is_some_and(|uv| {
1720        !user_verification_is_at_least(uv, org.webauthn.webauthn_user_verification)
1721    }) {
1722        overrides.webauthn_user_verification = None;
1723        cleared.push("webauthn_user_verification");
1724    }
1725
1726    // X7 G8, and the only *disable*-only control in the model: the org value
1727    // `true` is the permissive one, so a tenant override of `false` is the
1728    // tightening direction and is kept, while `Some(true)` against an org
1729    // baseline of `false` is a tenant granting itself a release its
1730    // organization did not authorise. `default_locale` is deliberately absent
1731    // from this function — see `OidcPolicy`.
1732    if !org.oidc.sensitive_scopes_enabled && overrides.sensitive_scopes_enabled == Some(true) {
1733        overrides.sensitive_scopes_enabled = None;
1734        cleared.push("sensitive_scopes_enabled");
1735    }
1736
1737    // T21.4 — the three ordered dynamic-registration controls, on the same
1738    // ordering `validate_tenant_override` uses. The three *lists* are
1739    // deliberately absent for the reason `OidcPolicy` gives: they name
1740    // per-tenant resources (this tenant's MCP servers, this tenant's callback
1741    // hosts), and there is no sense in which one such list is stricter than
1742    // another, so an organization baseline cannot overtake one.
1743    if overrides
1744        .dynamic_registration
1745        .is_some_and(|m| !dynamic_registration_is_at_most(m, org.oidc.dynamic_registration))
1746    {
1747        overrides.dynamic_registration = None;
1748        cleared.push("dynamic_registration");
1749    }
1750    if overrides
1751        .dcr_max_clients
1752        .is_some_and(|n| n > org.oidc.dcr_max_clients)
1753    {
1754        overrides.dcr_max_clients = None;
1755        cleared.push("dcr_max_clients");
1756    }
1757    if overrides.dcr_unused_client_ttl_days.is_some_and(|d| {
1758        dcr_ttl_strictness(d) > dcr_ttl_strictness(org.oidc.dcr_unused_client_ttl_days)
1759    }) {
1760        overrides.dcr_unused_client_ttl_days = None;
1761        cleared.push("dcr_unused_client_ttl_days");
1762    }
1763    // T21.5 — the posture is overridden whole, so it is cleared whole: a
1764    // tenant CIMD block that turns on what the organization turned off is
1765    // dropped in favour of the baseline rather than half-kept.
1766    if overrides.cimd.as_ref().is_some_and(|c| {
1767        (c.enabled && !org.oidc.cimd.enabled) || (c.allow_http && !org.oidc.cimd.allow_http)
1768    }) {
1769        overrides.cimd = None;
1770        cleared.push("cimd");
1771    }
1772
1773    cleared
1774}

1776/// Validate that a tenant override is only **more restrictive** than
1777/// the org baseline. Collects all violations into one error message.
1778///
1779/// Rules:
1780/// - `tenant >= org` for: min_length, password_history_count,
1781///   lockout_duration_secs, lockout_backoff_multiplier,
1782///   max_lockout_duration_secs
1783/// - `tenant <= org` for: max_failed_login_attempts,
1784///   access_token_lifetime_secs, refresh_token_lifetime_secs,
1785///   mfa_challenge_lifetime_secs, default_cert_validity_days,
1786///   max_cert_validity_days, email_verification_grace_period_hours
1787/// - enable-only (false->true OK, true->false NOT OK):
1788///   require_uppercase/lowercase/digits/symbols, mfa_enforced,
1789///   hibp_check_enabled, email_verification_required,
1790///   admin_notifications_enabled
1791pub fn validate_tenant_override(
1792    org: &SecuritySettings,
1793    overrides: &TenantSettingsOverride,
1794) -> AxiamResult<()> {
1795    let mut violations = Vec::new();
1796
1797    // --- tenant >= org (higher minimum is more restrictive) ---
1798    macro_rules! check_min {
1799        ($field:ident, $org_path:expr, $label:expr) => {
1800            if let Some(val) = overrides.$field {
1801                if val < $org_path {
1802                    violations.push(format!(
1803                        "{}: tenant value {} is less restrictive \
1804                         than org baseline {}",
1805                        $label, val, $org_path,
1806                    ));
1807                }
1808            }
1809        };
1810    }
1811
1812    check_min!(min_length, org.password.min_length, "min_length");
1813    check_min!(
1814        password_history_count,
1815        org.password.password_history_count,
1816        "password_history_count"
1817    );
1818    check_min!(
1819        lockout_duration_secs,
1820        org.lockout.lockout_duration_secs,
1821        "lockout_duration_secs"
1822    );
1823    check_min!(
1824        max_lockout_duration_secs,
1825        org.lockout.max_lockout_duration_secs,
1826        "max_lockout_duration_secs"
1827    );
1828
1829    // lockout_backoff_multiplier (f64 — compare with partial_cmp)
1830    if let Some(val) = overrides.lockout_backoff_multiplier
1831        && val < org.lockout.lockout_backoff_multiplier
1832    {
1833        violations.push(format!(
1834            "lockout_backoff_multiplier: tenant value {} is \
1835             less restrictive than org baseline {}",
1836            val, org.lockout.lockout_backoff_multiplier,
1837        ));
1838    }
1839
1840    // --- tenant <= org (lower max / shorter lifetime is more restrictive) ---
1841    macro_rules! check_max {
1842        ($field:ident, $org_path:expr, $label:expr) => {
1843            if let Some(val) = overrides.$field {
1844                if val > $org_path {
1845                    violations.push(format!(
1846                        "{}: tenant value {} is less restrictive \
1847                         than org baseline {}",
1848                        $label, val, $org_path,
1849                    ));
1850                }
1851            }
1852        };
1853    }
1854
1855    check_max!(
1856        max_failed_login_attempts,
1857        org.lockout.max_failed_login_attempts,
1858        "max_failed_login_attempts"
1859    );
1860    check_max!(
1861        access_token_lifetime_secs,
1862        org.token.access_token_lifetime_secs,
1863        "access_token_lifetime_secs"
1864    );
1865    check_max!(
1866        refresh_token_lifetime_secs,
1867        org.token.refresh_token_lifetime_secs,
1868        "refresh_token_lifetime_secs"
1869    );
1870    check_max!(
1871        mfa_challenge_lifetime_secs,
1872        org.mfa.mfa_challenge_lifetime_secs,
1873        "mfa_challenge_lifetime_secs"
1874    );
1875    check_max!(
1876        default_cert_validity_days,
1877        org.certificate.default_cert_validity_days,
1878        "default_cert_validity_days"
1879    );
1880    check_max!(
1881        max_cert_validity_days,
1882        org.certificate.max_cert_validity_days,
1883        "max_cert_validity_days"
1884    );
1885    check_max!(
1886        email_verification_grace_period_hours,
1887        org.email.email_verification_grace_period_hours,
1888        "email_verification_grace_period_hours"
1889    );
1890    // Shorter is more restrictive: it is time spent holding data the subject
1891    // has already asked to have erased.
1892    check_max!(
1893        deletion_grace_period_days,
1894        org.privacy.deletion_grace_period_days,
1895        "deletion_grace_period_days"
1896    );
1897
1898    // --- enable-only (false->true OK, true->false NOT OK) ---
1899    macro_rules! check_enable_only {
1900        ($field:ident, $org_val:expr, $label:expr) => {
1901            if let Some(val) = overrides.$field {
1902                if $org_val && !val {
1903                    violations.push(format!(
1904                        "{}: cannot disable at tenant level when \
1905                         enabled at org level",
1906                        $label,
1907                    ));
1908                }
1909            }
1910        };
1911    }
1912
1913    check_enable_only!(
1914        require_uppercase,
1915        org.password.require_uppercase,
1916        "require_uppercase"
1917    );
1918    check_enable_only!(
1919        require_lowercase,
1920        org.password.require_lowercase,
1921        "require_lowercase"
1922    );
1923    check_enable_only!(
1924        require_digits,
1925        org.password.require_digits,
1926        "require_digits"
1927    );
1928    check_enable_only!(
1929        require_symbols,
1930        org.password.require_symbols,
1931        "require_symbols"
1932    );
1933    check_enable_only!(mfa_enforced, org.mfa.mfa_enforced, "mfa_enforced");
1934    check_enable_only!(
1935        hibp_check_enabled,
1936        org.password.hibp_check_enabled,
1937        "hibp_check_enabled"
1938    );
1939    check_enable_only!(
1940        email_verification_required,
1941        org.email.email_verification_required,
1942        "email_verification_required"
1943    );
1944    check_enable_only!(
1945        admin_notifications_enabled,
1946        org.notification.admin_notifications_enabled,
1947        "admin_notifications_enabled"
1948    );
1949
1950    // --- OPAQUE: tighten-only ---
1951    //
1952    // `OpaqueMode` derives `Ord` as Disabled < Optional < Required, which is
1953    // exactly the restrictiveness order, so this is a plain comparison. Suite
1954    // and KSF strength are ranked explicitly rather than by declaration order,
1955    // so that adding a variant forces a decision about where it sits.
1956    if let Some(mode) = overrides.opaque_mode
1957        && mode < org.opaque.opaque_mode
1958    {
1959        violations.push(format!(
1960            "opaque_mode: tenant value {} is less restrictive than org baseline {}",
1961            mode, org.opaque.opaque_mode,
1962        ));
1963    }
1964    if let Some(suite) = overrides.opaque_suite
1965        && !opaque_suite_is_at_least(suite, org.opaque.opaque_suite)
1966    {
1967        violations.push(format!(
1968            "opaque_suite: tenant value {} is weaker than org baseline {}",
1969            suite, org.opaque.opaque_suite,
1970        ));
1971    }
1972    if let Some(ksf) = overrides.opaque_ksf
1973        && !opaque_ksf_is_at_least(ksf, org.opaque.opaque_ksf)
1974    {
1975        violations.push(format!(
1976            "opaque_ksf: tenant value {} is weaker than org baseline {}",
1977            ksf, org.opaque.opaque_ksf,
1978        ));
1979    }
1980
1981    // --- WebAuthn user verification: tighten-only ---
1982    //
1983    // `required` > `preferred` > `discouraged`. A tenant handling regulated
1984    // data may demand a PIN where the organization only asks for one; it may
1985    // not quietly stop asking where the organization insists.
1986    if let Some(uv) = overrides.webauthn_user_verification
1987        && !user_verification_is_at_least(uv, org.webauthn.webauthn_user_verification)
1988    {
1989        violations.push(format!(
1990            "webauthn_user_verification: tenant value {} is less restrictive than \
1991             org baseline {}",
1992            uv, org.webauthn.webauthn_user_verification,
1993        ));
1994    }
1995
1996    // --- X7 G8 sensitive scopes: disable-only ---
1997    //
1998    // The mirror image of `check_enable_only`, and the only control in this
1999    // model that runs that way, because it is the only one whose `true` is the
2000    // permissive value. A tenant may refuse to release postal addresses and
2001    // telephone numbers that its organization allows; it may not decide on its
2002    // own to start releasing them. The lawful basis for holding the data was
2003    // established at the organization level and the tenant does not get to
2004    // widen it.
2005    //
2006    // `default_locale` is deliberately not checked here. See `OidcPolicy`.
2007    if overrides.sensitive_scopes_enabled == Some(true) && !org.oidc.sensitive_scopes_enabled {
2008        violations.push(
2009            "sensitive_scopes_enabled: cannot enable at tenant level when disabled at org level              (the address and phone scopes release personal data under the organization's              lawful basis, not the tenant's)"
2010                .into(),
2011        );
2012    }
2013
2014    // --- T21.4 dynamic client registration: three ordered controls ---
2015    //
2016    // A tenant may refuse a registration mode its organization allows; it may
2017    // not admit callers the organization does not. `dcr_max_clients` and
2018    // `dcr_unused_client_ttl_days` run the ordinary "tenant <= org" way: a
2019    // smaller ceiling and a shorter unused-client lifetime are both the
2020    // stricter direction, and a tenant raising either would be granting itself
2021    // more self-registered clients, for longer, than the organization allowed.
2022    //
2023    // The three lists are not checked, for the reason `OidcPolicy` states:
2024    // `external_client_allowed_resources` names the MCP servers *this* tenant
2025    // fronts, and a subset rule would force an organization to enumerate every
2026    // tenant's resource servers in its own baseline before any tenant could
2027    // name one.
2028    if let Some(mode) = overrides.dynamic_registration
2029        && !dynamic_registration_is_at_most(mode, org.oidc.dynamic_registration)
2030    {
2031        violations.push(format!(
2032            "dynamic_registration: tenant value {mode} admits callers the org baseline {} \
2033             does not",
2034            org.oidc.dynamic_registration,
2035        ));
2036    }
2037    if let Some(n) = overrides.dcr_max_clients
2038        && n > org.oidc.dcr_max_clients
2039    {
2040        violations.push(format!(
2041            "dcr_max_clients ({n}) must be <= org baseline ({})",
2042            org.oidc.dcr_max_clients,
2043        ));
2044    }
2045    if let Some(d) = overrides.dcr_unused_client_ttl_days
2046        && dcr_ttl_strictness(d) > dcr_ttl_strictness(org.oidc.dcr_unused_client_ttl_days)
2047    {
2048        violations.push(format!(
2049            "dcr_unused_client_ttl_days ({d}) keeps an unused self-registered client longer \
2050             than the org baseline ({}); 0 means never sweep, which is the longest value of \
2051             all",
2052            org.oidc.dcr_unused_client_ttl_days,
2053        ));
2054    }
2055
2056    // T21.5 — the two halves of the CIMD posture that can *widen*. Everything
2057    // else on `CimdPolicy` names this tenant's own publishers, callbacks and
2058    // bounds, and is unordered for the reason the three DCR lists are.
2059    if let Some(cimd) = &overrides.cimd {
2060        if cimd.enabled && !org.oidc.cimd.enabled {
2061            violations.push(
2062                "cimd.enabled: cannot enable client ID metadata documents at tenant level \
2063                 when disabled at org level (materialising a client from a stranger's \
2064                 document is a decision taken where the outbound fetch is paid for)"
2065                    .into(),
2066            );
2067        }
2068        if cimd.allow_http && !org.oidc.cimd.allow_http {
2069            violations.push(
2070                "cimd.allow_http: cannot allow plaintext metadata fetches at tenant level \
2071                 when the org baseline forbids them; the same seam also admits private \
2072                 addresses on the first hop"
2073                    .into(),
2074            );
2075        }
2076    }
2077
2078    if !violations.is_empty() {
2079        return Err(AxiamError::Validation {
2080            message: format!(
2081                "Tenant override violates org baseline: {}",
2082                violations.join("; "),
2083            ),
2084        });
2085    }
2086
2087    // Cross-field invariant check: merge org + overrides and verify
2088    // the effective policy is internally consistent.
2089    let merged = effective_settings(org, overrides, Uuid::nil(), Uuid::nil());
2090    let mut cross = Vec::new();
2091
2092    // Non-zero lifetime invariants (0 passes "more restrictive" checks
2093    // but produces an unusable policy).
2094    if merged.token.access_token_lifetime_secs == 0 {
2095        cross.push("effective access_token_lifetime_secs must be > 0".into());
2096    }
2097    if merged.token.refresh_token_lifetime_secs == 0 {
2098        cross.push("effective refresh_token_lifetime_secs must be > 0".into());
2099    }
2100    if merged.mfa.mfa_challenge_lifetime_secs == 0 {
2101        cross.push("effective mfa_challenge_lifetime_secs must be > 0".into());
2102    }
2103    // 0 passes "more restrictive" and produces an erasure that cannot be
2104    // cancelled at all, including by the link in its own confirmation email.
2105    if merged.privacy.deletion_grace_period_days == 0 {
2106        cross.push("effective deletion_grace_period_days must be >= 1".into());
2107    }
2108
2109    // T21.4 — the D3 interlock and the sensitive-scope refusal, on the policy
2110    // the tenant will actually run under. Checked on the *merge* rather than
2111    // on the override, because a tenant that names `anonymous` while
2112    // inheriting an empty resource list from its organization is exactly the
2113    // state the interlock exists to refuse.
2114    cross.extend(validate_dcr_policy(&merged.oidc));
2115    // T21.5 — the D3 interlock and the trusted-publisher refusal, on the
2116    // policy the tenant will actually run under. A tenant that enables CIMD
2117    // while inheriting an empty resource list is exactly what this refuses.
2118    cross.extend(validate_cimd_policy(&merged.oidc));
2119
2120    if merged.lockout.max_lockout_duration_secs < merged.lockout.lockout_duration_secs {
2121        cross.push(format!(
2122            "effective max_lockout_duration_secs ({}) must be >= \
2123             lockout_duration_secs ({})",
2124            merged.lockout.max_lockout_duration_secs, merged.lockout.lockout_duration_secs,
2125        ));
2126    }
2127    if merged.certificate.max_cert_validity_days < merged.certificate.default_cert_validity_days {
2128        cross.push(format!(
2129            "effective max_cert_validity_days ({}) must be >= \
2130             default_cert_validity_days ({})",
2131            merged.certificate.max_cert_validity_days,
2132            merged.certificate.default_cert_validity_days,
2133        ));
2134    }
2135
2136    if cross.is_empty() {
2137        Ok(())
2138    } else {
2139        Err(AxiamError::Validation {
2140            message: format!(
2141                "Tenant override produces inconsistent effective policy: {}",
2142                cross.join("; "),
2143            ),
2144        })
2145    }
2146}
