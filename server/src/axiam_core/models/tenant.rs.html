1//! Tenant domain model.
2//!
3//! Tenants provide full data isolation within an organization.
4//! All domain entities (users, roles, resources, etc.) are scoped to a tenant.
5//!
6//! Exactly one tenant per organization is reserved: the **organization
7//! tenant**, flagged [`TenantKind::Organization`]. It is where
8//! organization-level users, groups, roles and service accounts live — the
9//! principals whose grants apply to every tenant in the organization rather
10//! than to one. See `claude_dev/organization-scope-design.md` for why an
11//! organization scope is modelled as a tenant rather than as an
12//! `Option<Uuid>` on every scoped entity.
13
14use chrono::{DateTime, Utc};
15use serde::{Deserialize, Serialize};
16use uuid::Uuid;
17
18/// Lifecycle status of a tenant.
19///
20/// A `Suspended` tenant remains stored and its data isolated, but is treated as
21/// administratively disabled. New tenants are `Active` by default.
22#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, Default, utoipa::ToSchema)]
23pub enum TenantStatus {
24    /// The tenant is operational.
25    #[default]
26    Active,
27    /// The tenant is administratively disabled.
28    Suspended,
29}
30
31/// What a tenant *is*, as distinct from what state it is in.
32///
33/// Reserved rather than inferred: an organization has exactly one tenant of
34/// kind [`Self::Organization`], enforced by a unique index rather than by
35/// convention. Deriving it from a magic slug or from "the oldest tenant" would
36/// make the organization scope something an operator could rename or delete by
37/// accident, and it is the scope the super-admin lives in.
38#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, Default, utoipa::ToSchema)]
39#[serde(rename_all = "snake_case")]
40pub enum TenantKind {
41    /// An ordinary tenant: an isolated context holding its own users, roles,
42    /// resources and certificates. What every tenant was before organization
43    /// scope existed, and therefore the default — so every row written by an
44    /// older version reads back as exactly what it is.
45    #[default]
46    Standard,
47    /// The organization's own scope, one per organization.
48    ///
49    /// Holds the principals that operate across every tenant in the
50    /// organization. Its grants are evaluated against other tenants under one
51    /// added rule — only *global* grants cross a tenant boundary — which is
52    /// stated once, in the authorization engine.
53    ///
54    /// Created with the organization and not by hand: `POST .../tenants`
55    /// refuses to create a second one, and deleting it would strand every
56    /// organization-level principal in it.
57    Organization,
58}
59
60impl TenantKind {
61    /// Whether this is the organization's own scope.
62    pub const fn is_organization(self) -> bool {
63        matches!(self, Self::Organization)
64    }
65}
66
67impl std::fmt::Display for TenantKind {
68    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
69        match self {
70            Self::Standard => write!(f, "standard"),
71            Self::Organization => write!(f, "organization"),
72        }
73    }
74}

76/// The slug given to an organization tenant when it is created.
77///
78/// Slugs are unique per organization, so this is a reserved name: creating an
79/// ordinary tenant with it is refused. Fixed rather than derived from the
80/// organization's own slug because it appears in login URLs and operator
81/// muscle memory, and it should read the same in every deployment.
82pub const ORGANIZATION_TENANT_SLUG: &str = "organization";
83
84/// A tenant is an isolated context within an organization.
85///
86/// Each tenant has its own set of users, roles, permissions, resources,
87/// certificates, and configuration. Tenants can represent environments
88/// (dev/staging/prod) or separate business contexts.
89#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
90pub struct Tenant {
91    pub id: Uuid,
92    /// The organization this tenant belongs to.
93    pub organization_id: Uuid,
94    /// Human-readable name.
95    pub name: String,
96    /// URL-safe unique identifier within the organization (e.g., `production`).
97    pub slug: String,
98    /// Lifecycle status. New tenants default to [`TenantStatus::Active`].
99    pub status: TenantStatus,
100    /// Whether this is an ordinary tenant or the organization's own scope.
101    ///
102    /// `#[serde(default)]` so every row written before organization scope
103    /// existed reads back as [`TenantKind::Standard`], which is what it is.
104    #[serde(default)]
105    pub kind: TenantKind,
106    /// Arbitrary key-value metadata.
107    pub metadata: serde_json::Value,
108    pub created_at: DateTime<Utc>,
109    pub updated_at: DateTime<Utc>,
110}
111
112impl Tenant {
113    /// Whether this tenant is the organization's own scope.
114    ///
115    /// The question asked on the authorization path — a principal whose home
116    /// tenant answers `true` is an organization-level principal — so it is a
117    /// method rather than a field comparison spelled out at each call site.
118    pub const fn is_organization_scope(&self) -> bool {
119        self.kind.is_organization()
120    }
121}
122
123/// Fields required to create a new tenant.
124#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
125pub struct CreateTenant {
126    pub organization_id: Uuid,
127    pub name: String,
128    pub slug: String,
129    /// What kind of tenant to create.
130    ///
131    /// Not part of the public create request: the REST layer always sends
132    /// [`TenantKind::Standard`], and the organization tenant is created
133    /// alongside its organization. An API that let a caller ask for a second
134    /// organization scope would be an API whose refusal is the interesting
135    /// case.
136    #[serde(default, skip_deserializing)]
137    pub kind: TenantKind,
138    pub metadata: Option<serde_json::Value>,
139}
140
141impl CreateTenant {
142    /// The organization tenant for `organization_id`.
143    ///
144    /// One constructor rather than three call sites setting `kind` by hand —
145    /// bootstrap, the migration that backfills existing organizations, and
146    /// organization creation must all produce the same row, and a tenant that
147    /// is *nearly* the organization scope is one that the unique index will
148    /// reject at the worst possible moment.
149    pub fn organization_scope(organization_id: Uuid) -> Self {
150        Self {
151            organization_id,
152            name: "Organization".to_string(),
153            slug: ORGANIZATION_TENANT_SLUG.to_string(),
154            kind: TenantKind::Organization,
155            metadata: None,
156        }
157    }
158}
