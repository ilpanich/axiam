3use chrono::{DateTime, Utc};
4use serde::{Deserialize, Serialize};

7/// Type of WebAuthn authenticator.
8#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, utoipa::ToSchema)]
9pub enum WebauthnCredentialType {
10    /// Discoverable / platform authenticator
11    /// (1Password, Bitwarden, iCloud Keychain, Android).
12    Passkey,
13    /// Roaming / cross-platform authenticator (YubiKey, NitroKey).
14    SecurityKey,
15}
16
17impl WebauthnCredentialType {
18    /// Stable string representation for database storage.
19    pub fn as_str(&self) -> &'static str {
20        match self {
21            Self::Passkey => "Passkey",
22            Self::SecurityKey => "SecurityKey",
23        }
24    }
25}
26
27/// A registered WebAuthn credential for a user.
28#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema)]
29pub struct WebauthnCredential {
30    pub id: Uuid,
31    pub tenant_id: Uuid,
32    pub user_id: Uuid,
33    /// Base64url-encoded credential ID from the authenticator.
34    pub credential_id: String,
35    /// User-assigned friendly name (e.g., "My YubiKey", "iCloud Passkey").
36    pub name: String,
37    pub credential_type: WebauthnCredentialType,
38    /// JSON-serialized `webauthn_rs::prelude::Passkey`, AES-256-GCM
39    /// encrypted.
40    pub passkey_json: String,
41    pub created_at: DateTime<Utc>,
42    pub last_used_at: Option<DateTime<Utc>>,
43
44    // -----------------------------------------------------------------
45    // Attestation metadata (D6, X3) — additive, no migration. All four
46    // fields are `#[serde(default)]` so rows written before X3 shipped
47    // deserialize unchanged (absent JSON keys -> `None`/`false`).
48    // -----------------------------------------------------------------
49    /// The authenticator's AAGUID, when the registration ceremony resolved
50    /// one. `None` for credentials registered before X3, and for
51    /// authenticators that legitimately report no AAGUID.
52    #[serde(default)]
53    pub aaguid: Option<Uuid>,
54    /// The WebAuthn attestation statement format (`"packed"`, `"none"`,
55    /// `"tpm"`, …) recorded at registration time.
56    #[serde(default)]
57    pub attestation_format: Option<String>,
58    /// `true` if this credential was registered through the attested
59    /// ceremony (`finish_attested_passkey_registration`) and passed the
60    /// tenant's attestation policy at the time. `false` for every
61    /// credential registered under `mode: none` (today's default) or
62    /// before X3 shipped.
63    #[serde(default)]
64    pub attested: bool,
65    /// The MDS `metadataStatement.description` friendly name captured at
66    /// registration time (e.g. "YubiKey 5 NFC"), for display in credential
67    /// lists. `None` when MDS had no entry for the AAGUID, or attestation
68    /// was not requested.
69    #[serde(default)]
70    pub authenticator_name: Option<String>,
71}

73/// Input for creating a new WebAuthn credential.
74#[derive(Debug, Clone, Serialize, Deserialize)]
75pub struct CreateWebauthnCredential {
76    pub tenant_id: Uuid,
77    pub user_id: Uuid,
78    pub credential_id: String,
79    pub name: String,
80    pub credential_type: WebauthnCredentialType,
81    /// JSON-serialized and encrypted passkey data.
82    pub passkey_json: String,
83
84    /// See [`WebauthnCredential::aaguid`] (D6). `#[serde(default)]` so
85    /// existing callers that construct this via JSON without the new
86    /// fields keep working.
87    #[serde(default)]
88    pub aaguid: Option<Uuid>,
89    /// See [`WebauthnCredential::attestation_format`].
90    #[serde(default)]
91    pub attestation_format: Option<String>,
92    /// See [`WebauthnCredential::attested`].
93    #[serde(default)]
94    pub attested: bool,
95    /// See [`WebauthnCredential::authenticator_name`].
96    #[serde(default)]
97    pub authenticator_name: Option<String>,
