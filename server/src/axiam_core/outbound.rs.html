1//! The shared outbound dispatcher's ports (D-36, T23.5.1).
2//!
3//! AXIAM pushes work to systems it does not control: webhooks today, Shared
4//! Signals Framework push (G-5) and outbound SCIM operations (G-6) next. All of
5//! them need the same machinery: a durable queue, one attempt per delivery,
6//! bounded exponential retry, a dead-letter queue, an audit trail. That
7//! machinery is in `axiam-amqp`; the protocol that decides *what one attempt
8//! does* lives in the crate that owns the protocol. This module is the seam
9//! between the two, and it is layer 0 so that every protocol crate, whichever
10//! layer it sits in, can reach it.
11//!
12//! # The three pieces
13//!
14//! * [`OutboundMessage`] — the envelope that travels through the queue.
15//! * [`OutboundPublisher`] — *enqueue* a message. What a producer holds.
16//! * [`OutboundDeliverer`] — make *one attempt* at delivering a message and
17//!   report [`DeliveryOutcome`]. What a protocol crate implements.
18//!
19//! Retry, backoff, the attempt counter, the maximum-attempts decision and the
20//! dead-letter queue are **not** the deliverer's business. A deliverer never
21//! sleeps, never counts attempts and never republishes; it reports what
22//! happened and the consumer loop decides.
23//!
24//! # Adding a kind (SSF push, outbound SCIM)
25//!
26//! 1. Add one line to the `outbound_kinds!` invocation below:
27//!    `SsfPush => "ssf_push"`. The slug fixes the kind's queue names
28//!    (`axiam.ssf_push`, `.retry`, `.dlq`), its env-var prefix
29//!    (`AXIAM__<SLUG>__MAX_ATTEMPTS`, `…__BACKOFF_BASE_MS`,
30//!    `…__BACKOFF_CEILING_MS`, with the slug upper-cased) and its audit action prefix
31//!    (`ssf_push.delivery_succeeded`, `.delivery_attempt`, `.delivery_failed`, and,
32//!    from the minimal profile's in-process dispatcher only, `.delivery_abandoned`).
33//! 2. Implement [`OutboundDeliverer`] in the crate that owns the protocol,
34//!    returning that kind from [`OutboundDeliverer::kind`].
35//! 3. In `axiam-server`: call `AmqpManager::declare_outbound_topology(kind)`,
36//!    register the deliverer in `axiam_amqp::outbound::OutboundDeliverers`, and
37//!    run `axiam_amqp::outbound::run_outbound_consumer` for the kind.
38//!
39//! No change to `axiam-amqp`'s loop, topology or publisher is needed.
40//!
41//! # Implementing [`OutboundDeliverer`]
42//!
43//! ```
44//! use axiam_core::outbound::{
45//!     DeliveryOutcome, OutboundDeliverer, OutboundError, OutboundFuture, OutboundKind,
46//!     OutboundMessage,
47//! };
48//!
49//! struct Discarding;
50//!
51//! impl OutboundDeliverer for Discarding {
52//!     fn kind(&self) -> OutboundKind {
53//!         OutboundKind::Webhook
54//!     }
55//!
56//!     fn deliver_attempt<'a>(
57//!         &'a self,
58//!         msg: &'a OutboundMessage,
59//!     ) -> OutboundFuture<'a, Result<DeliveryOutcome, OutboundError>> {
60//!         Box::pin(async move {
61//!             if msg.payload.is_null() {
62//!                 // Never going to succeed; do not burn the retry budget.
63//!                 return Ok(DeliveryOutcome::DeadLetter { reason: "empty payload".into() });
64//!             }
65//!             Ok(DeliveryOutcome::Delivered { response_status: Some(204) })
66//!         })
67//!     }
68//! }
69//! ```

77/// Declares [`OutboundKind`] from `Variant => "slug"` pairs.
78///
79/// A macro so that adding a kind is one line: the enum, its serde names,
80/// [`OutboundKind::ALL`] and [`OutboundKind::as_str`] cannot drift apart.
81macro_rules! outbound_kinds {
82    ($( $(#[$meta:meta])* $variant:ident => $slug:literal ),+ $(,)?) => {
83        /// What an outbound delivery is *for*; selects the queue topology, the
84        /// retry configuration and the [`OutboundDeliverer`].
85        ///
86        /// The slug ([`Self::as_str`]) is load-bearing: it is the serde name,
87        /// the queue-name stem, the env-var prefix and the audit-action prefix.
88        /// **Never change an existing slug** — that renames queues that hold
89        /// in-flight messages and drops them on upgrade.
90        #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
91        pub enum OutboundKind {
92            $( $(#[$meta])* #[serde(rename = $slug)] $variant ),+
93        }
94
95        impl OutboundKind {
96            /// Every kind, for code that has to cover them all (topology
97            /// declaration, consumer start-up, tests).
98            pub const ALL: &'static [OutboundKind] = &[ $( OutboundKind::$variant ),+ ];
99
100            /// The kind's stable slug: lower-case ASCII letters, digits and
101            /// underscores only (it becomes part of a queue name and an
102            /// environment-variable name).
103            pub const fn as_str(self) -> &'static str {
104                match self { $( OutboundKind::$variant => $slug ),+ }
105            }
106        }
107    };
108}

110outbound_kinds! {
111    /// HMAC-signed JSON POST to a tenant-registered webhook URL.
112    Webhook => "webhook",
113    /// RFC 8935 push of one Security Event Token to an SSF receiver (G-5,
114    /// T23.5.2). `target_id` is the stream id, `delivery_id` the SET's `jti`
115    /// (as a UUID), `event_type` the event-type URI, and `payload` an
116    /// [`crate::models::ssf::SsfPendingEvent`] — the **unsigned** event: the
117    /// deliverer (`axiam-oauth2`, T23.5.3) re-reads the stream and signs at the
118    /// attempt, so a queued message never holds a signed SET and a stream
119    /// disabled in the meantime delivers nothing (D-48, D-51).
120    SsfPush => "ssf_push",
121    /// One outbound SCIM 2.0 provisioning operation toward a tenant-registered
122    /// downstream service provider (G-6, D-57). `target_id` is the
123    /// `ScimTarget` id and `payload` is a reference only,
124    /// `{resource_type, axiam_id}`: the deliverer re-reads the target, the
125    /// resource and its link at the attempt and computes the desired downstream
126    /// state then, so no attribute of a person ever sits in a queue or the DLQ.
127    ScimPush => "scim_push",
128    /// The ping-mode notification of a decided CIBA request (G-7, T23.7.2,
129    /// D-65): `POST {"auth_req_id": …}` with `Authorization: Bearer
130    /// <client_notification_token>` to the client's registered
131    /// `backchannel_client_notification_endpoint`. `target_id` is the **record
132    /// id** of the CIBA request and `payload` is empty: the deliverer
133    /// (`axiam-oauth2`) re-reads the request, opens its sealed credentials and
134    /// reads the client's endpoint at the attempt, so neither the `auth_req_id`
135    /// nor the notification token ever sits in a queue or the dead-letter queue.
136    CibaPing => "ciba_ping",
137}

152/// The envelope for one outbound delivery, as it travels through the queue.
153///
154/// Fields are the superset every kind needs; a kind that does not use one
155/// leaves it at a neutral value (e.g. `Uuid::nil()`, `Value::Null`). The
156/// envelope never carries a secret: a deliverer resolves the signing material
157/// itself from `target_id`.
158#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
159pub struct OutboundMessage {
160    /// Selects the queue, retry configuration and deliverer.
161    pub kind: OutboundKind,
162    /// Owning tenant. Every deliverer lookup is scoped by it; an envelope
163    /// without it would force a cross-tenant lookup.
164    pub tenant_id: Uuid,
165    /// What is being delivered *to*: the webhook id for
166    /// [`OutboundKind::Webhook`]; the stream id for SSF push; the target id for
167    /// outbound SCIM. Also the `resource_id` of the delivery audit records.
168    pub target_id: Uuid,
169    /// Identifies this logical delivery across all of its attempts.
170    pub delivery_id: Uuid,
171    /// The event name (`user.created`, ...). Free text to the dispatcher.
172    pub event_type: String,
173    /// The event body. Opaque to the dispatcher.
174    pub payload: serde_json::Value,
175    /// Attempts already made; `0` on first enqueue. Owned by the consumer
176    /// loop, which increments it on every republish to the retry queue.
177    pub attempt: u32,
178}

180/// What a single delivery attempt achieved. See [`OutboundDeliverer`].
181#[derive(Debug, Clone, PartialEq, Eq)]
182pub enum DeliveryOutcome {
183    /// The receiver accepted the message. The delivery is acknowledged and
184    /// never attempted again.
185    Delivered {
186        /// The receiver's protocol status (an HTTP status code), if the
187        /// protocol has one. Recorded in the success audit entry as `status`.
188        response_status: Option<u16>,
189    },
190    /// This attempt failed in a way that may succeed later (receiver down,
191    /// non-2xx, timeout, SSRF guard refused a name that may re-resolve). The
192    /// loop retries with backoff until the kind's maximum attempts, then
193    /// dead-letters.
194    Retry {
195        /// Why; recorded in the audit entry as `error`. Must not contain
196        /// secrets.
197        reason: String,
198    },
199    /// This message can never succeed (it is malformed, or its target is
200    /// permanently gone). The loop dead-letters it now, without spending the
201    /// remaining attempts.
202    DeadLetter {
203        /// Why; recorded in the audit entry as `error`. Must not contain
204        /// secrets.
205        reason: String,
206    },
207}
208
209/// Failure to enqueue, or a deliverer that could not even report an outcome.
210#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
211pub enum OutboundError {
212    /// The message could not be encoded or the broker did not confirm it.
213    #[error("outbound enqueue failed: {0}")]
214    Enqueue(String),
215    /// Bytes taken off a queue are not a valid message for that queue's kind.
216    /// The consumer loop dead-letters these without retrying.
217    #[error("outbound message malformed: {0}")]
218    Malformed(String),
219    /// A deliverer failed before it could classify the attempt. The consumer
220    /// loop treats this exactly like [`DeliveryOutcome::Retry`].
221    #[error("outbound delivery failed: {0}")]
222    Delivery(String),
223}

225/// A boxed, `Send` future: the return type that keeps the ports object-safe
226/// (`Arc<dyn OutboundPublisher>` / `Arc<dyn OutboundDeliverer>`) without a
227/// dependency on an async-trait crate in layer 0.
228pub type OutboundFuture<'a, T> = Pin<Box<dyn Future<Output = T> + Send + 'a>>;
229
230/// Enqueue an outbound delivery. The producer side of the dispatcher.
231///
232/// Best-effort from the caller's point of view: a producer is a side effect of
233/// some other operation (a revocation, a user update) and must not fail that
234/// operation because the broker is unavailable. Log the error and carry on, as
235/// `WebhookDeliveryService::emit` does.
236///
237/// Implementations route by [`OutboundMessage::kind`] and must enqueue the
238/// message with `attempt` as given (producers pass `0`). The enqueue is
239/// durable: `Ok(())` means the broker confirmed it.
240pub trait OutboundPublisher: Send + Sync {
241    /// Durably enqueue `msg` for its kind's consumer.
242    fn enqueue<'a>(
243        &'a self,
244        msg: &'a OutboundMessage,
245    ) -> OutboundFuture<'a, Result<(), OutboundError>>;
246}
247
248/// Make one delivery attempt. The protocol side of the dispatcher.
249///
250/// One implementation per [`OutboundKind`], living in the crate that owns the
251/// protocol (webhooks in `axiam-api-rest`, SSF push in `axiam-oauth2`).
252/// Registered with the consumer loop by `axiam-server`.
253///
254/// # Contract
255///
256/// * **Exactly one attempt.** No retry loop, no sleep, no republish, no
257///   attempt counting. The consumer loop owns all of those, so that backoff
258///   and dead-lettering behave identically for every kind.
259/// * **Classify, do not decide.** Return [`DeliveryOutcome::Retry`] for
260///   anything that may succeed later, and [`DeliveryOutcome::DeadLetter`] only
261///   for what can never succeed. Whether a retry is still affordable is the
262///   loop's call.
263/// * **Errors are retries.** An `Err` is treated as [`DeliveryOutcome::Retry`]
264///   with the error's text as the reason.
265/// * **Be idempotent-friendly.** At-least-once delivery means the same
266///   `delivery_id` can arrive twice (a crash between delivery and ack). Send
267///   `delivery_id` to the receiver so it can deduplicate.
268/// * **No secrets in reasons.** `reason` strings go to the audit log.
269/// * **Cancel-safe.** The future may be dropped when the consumer stops.
270pub trait OutboundDeliverer: Send + Sync {
271    /// The kind this deliverer serves. The consumer loop refuses to start for
272    /// a kind with no registered deliverer.
273    fn kind(&self) -> OutboundKind;
274
275    /// Attempt delivery of `msg` once and report what happened.
276    fn deliver_attempt<'a>(
277        &'a self,
278        msg: &'a OutboundMessage,
279    ) -> OutboundFuture<'a, Result<DeliveryOutcome, OutboundError>>;
280}

392    /// The ports must stay object-safe: `axiam-server` and the protocol crates
393    /// hold them as `Arc<dyn ...>`.
394    #[tokio::test]
395    async fn ports_are_object_safe() {
396        struct P;
397        impl OutboundPublisher for P {
398            fn enqueue<'a>(
399                &'a self,
400                _msg: &'a OutboundMessage,
401            ) -> OutboundFuture<'a, Result<(), OutboundError>> {
402                Box::pin(async { Ok(()) })
403            }
404        }
405        struct D;
406        impl OutboundDeliverer for D {
407            fn kind(&self) -> OutboundKind {
408                OutboundKind::Webhook
409            }
410            fn deliver_attempt<'a>(
411                &'a self,
412                _msg: &'a OutboundMessage,
413            ) -> OutboundFuture<'a, Result<DeliveryOutcome, OutboundError>> {
414                Box::pin(async {
415                    Ok(DeliveryOutcome::Delivered {
416                        response_status: None,
417                    })
418                })
419            }
420        }
421        let msg = OutboundMessage {
422            kind: OutboundKind::Webhook,
423            tenant_id: Uuid::nil(),
424            target_id: Uuid::nil(),
425            delivery_id: Uuid::nil(),
426            event_type: String::new(),
427            payload: serde_json::Value::Null,
428            attempt: 0,
429        };
430        let p: Arc<dyn OutboundPublisher> = Arc::new(P);
431        let d: Arc<dyn OutboundDeliverer> = Arc::new(D);
432        p.enqueue(&msg).await.unwrap();
433        assert_eq!(d.kind(), OutboundKind::Webhook);
434        assert_eq!(
435            d.deliver_attempt(&msg).await.unwrap(),
436            DeliveryOutcome::Delivered {
437                response_status: None
438            }
439        );
440    }
