1//! The declared inventory of what each column of the `user` table means
2//! (T-261).
3//!
4//! # Why this exists
5//!
6//! Four code paths decide what a `user` column means, and three of them used
7//! to decide it by writing the column name out by hand: the Art. 17 erasure
8//! statement, the administrator's tombstone behind `DELETE
9//! /api/v1/users/{id}`, and the Art. 15 export's `profile` section. A column
10//! named by none of them survives erasure and never reaches an export — which
11//! is precisely what nearly happened to `phone_number` and `address`, added by
12//! the release that also added the paths that would have stranded them. The
13//! fourth path, SCIM's no-op detection, was guarded because somebody wrote a
14//! destructure of `UpdateUser`; that guard does not generalise, because
15//! `UpdateUser` is not the `user` table.
16//!
17//! So the lists stop being lists. [`USER_COLUMNS`] is the one declaration; the
18//! two erasure statements derive their shared `SET` fragment from it
19//! ([`shared_erasure_fragment`]), and two gates make an unclassified column
20//! impossible to add quietly:
21//!
22//! 1. `user_schema_matches_the_declared_inventory` (in `axiam-db`, against a
23//!    live datastore after migrations) introspects `INFO FOR TABLE user` and
24//!    requires the field set and [`USER_COLUMNS`] to agree **in both
25//!    directions** — a removed column fails as loudly as an added one.
26//! 2. [`audit_declarations`] requires every row that classifies a column as
27//!    neither erased nor exported to say, in `note`, why.
28//!
29//! # Why `erasure` and `export` are two fields and not one flag
30//!
31//! Because a single "is personal data" boolean gets two real cases wrong.
32//! `password_hash` and `mfa_secret` are erased and are deliberately **not**
33//! exported (D-10: handing a data subject — or whoever compromised their
34//! mailbox — the offline-crackable artefact is not an Art. 15 obligation),
35//! while `created_at` is exported and deliberately not erased. Both are
36//! expressible only because the two questions are asked separately.
37//!
38//! # What this inventory does not decide
39//!
40//! Each erasure path keeps its own path-specific clauses: the terminal
41//! `status` value, the Art. 17 pipeline's `deletion_pending` /
42//! `scheduled_purge_at` reset, the tombstone's `email_verified_at`,
43//! `totp_last_used_step` and `failed_login_attempts` reset. Those asymmetries
44//! are **recorded** in the `note` of the column they belong to rather than
45//! harmonised here: harmonising them would change erasure behaviour, and this
46//! module is a gate.
47
48/// What an erasure statement writes into a column.
49///
50/// Only columns cleared by **both** erasure paths appear as `Some` on
51/// [`UserColumn::erasure`]; a column one path clears and the other does not is
52/// path-specific, stays `None`, and says so in its `note`.
53#[derive(Debug, Clone, Copy, PartialEq, Eq)]
54pub enum Erasure {
55    /// `SET <column> = NONE`.
56    ToNone,
57    /// `SET <column> = <literal>` — a SurrealQL literal such as `''`, `{}` or
58    /// `false`, written verbatim into the statement.
59    ToLiteral(&'static str),
60    /// `SET <column> = $<param>` — the calling path binds the value, because
61    /// it derives from the row: a pseudonym, an email hash, an `.invalid`
62    /// placeholder.
63    ToParam(&'static str),
64}
65
66impl Erasure {
67    /// The right-hand side this variant renders into a `SET` clause.
68    #[must_use]
69    pub fn render(&self) -> String {
70        match self {
71            Self::ToNone => "NONE".to_owned(),
72            Self::ToLiteral(literal) => (*literal).to_owned(),
73            Self::ToParam(param) => format!("${param}"),
74        }
75    }
76}

78/// One column of the `user` table, classified.
79///
80/// Every column has a row. A column with no row fails the schema-introspection
81/// gate in `axiam-db`, naming itself.
82#[derive(Debug, Clone, Copy)]
83pub struct UserColumn {
84    /// The column name exactly as `DEFINE FIELD` spells it.
85    pub name: &'static str,
86    /// What **both** erasure paths write, or `None` when the column is not
87    /// erased by both — `note` then says why.
88    pub erasure: Option<Erasure>,
89    /// The key this column appears under in the Art. 15 export's `profile`
90    /// section, or `None` when it is deliberately withheld — `note` says why.
91    ///
92    /// The key is not always the column name: `phone_number_verified_at` is
93    /// exported as the derived boolean `phone_number_verified`, because
94    /// exporting the internal column name would describe AXIAM's storage
95    /// rather than the subject's data.
96    pub export: Option<&'static str>,
97    /// Why, in the cases where either of the two above is `None`. Required to
98    /// be non-empty there, so that "nobody classified this" and "classified as
99    /// neither" are different states rather than the same silence.
100    pub note: &'static str,
101}

103/// Keys the Art. 15 `profile` section carries that are not `user` columns.
104///
105/// `id` is the record identifier. It is not a `DEFINE FIELD`, so it cannot
106/// appear in [`USER_COLUMNS`] without breaking the schema gate, and it is
107/// exported because a subject asking what is held about them is entitled to
108/// the handle everything else is keyed by.
109pub const EXPORT_KEYS_NOT_FROM_COLUMNS: &[&str] = &["id"];

277/// The `SET` clause fragment both erasure statements share, derived from
278/// [`USER_COLUMNS`].
279///
280/// Rendered without a trailing comma and without surrounding whitespace, so a
281/// caller composes it with its own path-specific clauses. The clause **set** is
282/// what this guarantees; clause order is the inventory's order, which neither
283/// erasure statement depended on (the two already disagreed about it) and
284/// SurrealDB does not observe.
285#[must_use]
286pub fn shared_erasure_fragment() -> String {
287    USER_COLUMNS
288        .iter()
289        .filter_map(|column| {
290            column
291                .erasure
292                .map(|erasure| format!("{} = {}", column.name, erasure.render()))
293        })
294        .collect::<Vec<_>>()
295        .join(", ")
296}
297
298/// Every parameter name [`shared_erasure_fragment`] expects a caller to bind.
299///
300/// A path that renders the fragment and forgets one of these produces a query
301/// SurrealDB refuses at parse time rather than one that silently writes
302/// nothing, but the list is here so a caller can be checked against it in a
303/// test rather than at runtime.
304#[must_use]
305pub fn shared_erasure_params() -> Vec<&'static str> {
306    USER_COLUMNS
307        .iter()
308        .filter_map(|column| match column.erasure {
309            Some(Erasure::ToParam(param)) => Some(param),
310            _ => None,
311        })
312        .collect()
313}

315/// Every export key the `profile` section must carry: the columns' keys plus
316/// [`EXPORT_KEYS_NOT_FROM_COLUMNS`].
317#[must_use]
318pub fn export_keys() -> Vec<&'static str> {
319    EXPORT_KEYS_NOT_FROM_COLUMNS
320        .iter()
321        .copied()
322        .chain(USER_COLUMNS.iter().filter_map(|column| column.export))
323        .collect()
324}

326/// The declaration gate: every column classified as neither erased nor
327/// exported must say why, and no column may be declared twice.
328///
329/// Returns the offending column names rather than panicking, so the test that
330/// calls it can report all of them at once.
331#[must_use]
332pub fn audit_declarations() -> Vec<&'static str> {
333    let mut problems = Vec::new();
334    for (index, column) in USER_COLUMNS.iter().enumerate() {
335        if (column.erasure.is_none() || column.export.is_none()) && column.note.is_empty() {
336            problems.push(column.name);
337        }
338        if USER_COLUMNS[..index].iter().any(|c| c.name == column.name) {
339            problems.push(column.name);
340        }
341    }
342    problems
343}

345#[cfg(test)]
346mod tests {
347    use super::*;

349    /// Gate 2. A column classified as neither erased nor exported is a
350    /// decision, and a decision with no reason recorded is how the next author
351    /// concludes it was an oversight and "fixes" it.
352    #[test]
353    fn every_unerased_or_unexported_column_says_why() {
354        assert!(
355            audit_declarations().is_empty(),
356            "these columns are not erased and/or not exported and carry no \
357             `note` saying why: {:?}",
358            audit_declarations()
359        );
360    }
361
362    #[test]
363    fn the_shared_fragment_names_every_column_that_declares_an_erasure() {
364        let fragment = shared_erasure_fragment();
365        for column in USER_COLUMNS.iter().filter(|c| c.erasure.is_some()) {
366            assert!(
367                fragment.contains(&format!("{} = ", column.name)),
368                "`{}` declares an erasure but is missing from the fragment",
369                column.name
370            );
371        }
372        // And nothing else: a column with `erasure: None` in the fragment
373        // would be an erasure nobody declared.
374        let clauses = fragment.split(", ").count();
375        let declared = USER_COLUMNS.iter().filter(|c| c.erasure.is_some()).count();
376        assert_eq!(clauses, declared);
377    }
378
379    /// The fragment is pinned verbatim. Not because the text matters to
380    /// SurrealDB — it does not — but because this is the one place a change to
381    /// what erasure writes becomes visible in a diff, and a derived statement
382    /// that nobody can see is worse than the hand-written one it replaced.
383    #[test]
384    fn the_shared_fragment_is_exactly_this() {
385        assert_eq!(
386            shared_erasure_fragment(),
387            "username = $pseudonym, \
388             email = $email_replacement, \
389             password_hash = '', \
390             mfa_enabled = false, \
391             mfa_secret = NONE, \
392             last_failed_login_at = NONE, \
393             locked_until = NONE, \
394             metadata = {}, \
395             phone_number = NONE, \
396             phone_number_verified_at = NONE, \
397             address = NONE"
398        );
399    }
400
401    #[test]
402    fn the_fragment_expects_exactly_two_bound_parameters() {
403        assert_eq!(
404            shared_erasure_params(),
405            vec!["pseudonym", "email_replacement"]
406        );
407    }

409    #[test]
410    fn export_keys_are_unique_and_include_the_record_id() {
411        let keys = export_keys();
412        let mut sorted = keys.clone();
413        sorted.sort_unstable();
414        sorted.dedup();
415        assert_eq!(sorted.len(), keys.len(), "duplicate export key");
416        assert!(keys.contains(&"id"));
417    }
