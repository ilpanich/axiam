60/// Clamp a pagination limit to [1, 200] at deserialization time.
61///
62/// Prevents resource exhaustion from unbounded page requests (SEC-010/CQ-B30).
63/// Direct struct construction is unaffected — only the serde path clamps.
64fn clamp_pagination_limit<'de, D>(de: D) -> Result<u64, D::Error>
65where
66    D: Deserializer<'de>,
67{
68    let raw = u64::deserialize(de)?;
69    Ok(raw.clamp(1, 200))
70}

181    /// code succeed at most once: returns `Ok(true)` if this call won the
182    /// CAS (the step advanced), `Ok(false)` if the guard did not match
183    /// (replay, or a concurrent caller already won). Callers MUST treat
184    /// `Ok(false)` as a rejected verification (e.g. `MfaInvalidCode`), not a
185    /// silent no-op.

198    /// Atomically increment the failed-login counter (SEC-032).
199    ///
200    /// Avoids read-modify-write TOCTOU by delegating the increment to the DB
201    /// layer via a single `UPDATE ... SET field += 1` statement.
202    ///
203    /// When the new attempt count reaches `lockout_threshold`, the account is
204    /// locked until `now + d`, where `d` grows exponentially with each lockout
205    /// beyond the threshold (brute-force protection, design-document §Security):
206    ///
207    /// ```text
208    /// d = min(base_lockout_secs * backoff_multiplier ^ (new_count - threshold),
209    ///         max_lockout_secs)
210    /// ```
211    ///
212    /// so the first lockout lasts `base_lockout_secs`, the next
213    /// `base * multiplier`, and so on, capped at `max_lockout_secs`.
214    fn increment_failed_logins(
215        &self,
216        tenant_id: Uuid,
217        user_id: Uuid,
218        lockout_threshold: u32,
219        base_lockout_secs: i64,
220        backoff_multiplier: f64,
221        max_lockout_secs: i64,
222    ) -> impl Future<Output = AxiamResult<()>> + Send;
223
224    /// Anonymize a user row in-place for GDPR Art. 17 erasure (D-05).
225    ///
226    /// Implementations MUST scrub every PII column (email, username,
227    /// password_hash, mfa_secret, metadata, lockout state) and clear
228    /// `deletion_pending`/`scheduled_purge_at` — this is the step that

458/// Storage for SCIM provisioning tokens.
459///
460/// See `axiam_core::models::scim_token` for what a token is and is not. The
461/// lookup below is the whole authentication path for `/scim/v2/*` when the
462/// caller presents a provisioning handle rather than a JWT.

465    -> impl Future<Output = AxiamResult<ScimToken>> + Send;

467    /// Resolve a presented handle by its hash, **without** filtering on
468    /// revocation or expiry.
469    ///
470    /// The caller applies [`ScimToken::is_usable`] instead, for the same
471    /// reason `consume_by_token_hash` does not filter expiry: the row is
472    /// needed in order to record *why* a request was refused, and a query that
473    /// returned `None` for "revoked" and for "never existed" alike would make
474    /// the two indistinguishable in the audit trail even though only one of
475    /// them is somebody using a credential that was taken away from them.
476    ///
477    /// Tenant-free by construction: a presented handle is the only thing the
478    /// caller knows at this point — there is no authenticated tenant yet to

578    /// This mirrors [`OAuth2ClientRepository::upgrade_client_secret_hash`],

583    /// the v1 arm of the verifier could never be retired on the strength of "no

604    /// ## Why this exists, stated plainly
605    ///
606    /// [`Self::upgrade_client_secret_hash`] can only fire on a *successful
607    /// verification*. When this count was introduced there was **no such path**
608    /// for a service account — the secret was issued at create/rotate and no
609    /// flow accepted it — so these rows could not drain lazily at all.
610    ///
611    /// The OAuth2 **client-credentials** grant now accepts a service account's
612    /// `client_id`/`client_secret`, so legacy rows *do* migrate on first use,
613    /// exactly as `oauth2_client` rows do. This count remains useful for the
614    /// case that migration cannot reach: a service account that never
615    /// authenticates. Its backlog is what decides whether the legacy hash arm
616    /// can be retired, and **rotation** clears a row that will never
617    /// authenticate on its own.
618    ///
619    /// **This counts the hash *scheme* only.** A row already in the current
620    /// scheme but keyed to a superseded pepper is indistinguishable on disk —
621    /// the stored format is identical — so it is not counted. That is correct
622    /// for the question this answers (can the legacy *arm* be retired?), and
623    /// such rows migrate on the same first authentication.
624    ///
625    /// That has one concrete consequence: the v1 arm of the verifier cannot be
626    /// retired on the strength of "no v1 `oauth2_client` rows remain", because
627    /// this table is invisible to that reasoning. This count makes the question
628    /// answerable, and the migration route is **rotation** — `rotate_secret`
629    /// writes the current scheme under the current pepper.
630    fn count_legacy_secret_hashes(
631        &self,
632        tenant_id: Option<Uuid>,
633    ) -> impl Future<Output = AxiamResult<u64>> + Send;
634}

652    /// call removed the row. Unlike [`Self::invalidate`] (idempotent, always
653    /// `Ok(())`), this is the single-use gate for refresh-token rotation
654    /// (NEW-3): two concurrent refreshes racing on the same token both pass the
655    /// prior SELECT, but only one wins this delete — the loser gets `false` and
656    /// must abort before minting a new session, preserving single-use rotation
657    /// and stolen-token reuse detection.
658    fn consume(&self, tenant_id: Uuid, id: Uuid) -> impl Future<Output = AxiamResult<bool>> + Send;
659    /// Atomically look up **and** consume a session by its refresh-token hash,
660    /// returning the row as it was immediately before deletion (`None` when no
661    /// live session matched).
662    ///
663    /// This is [`Self::get_by_token_hash`] followed by [`Self::consume`],
664    /// collapsed into one statement — and it is the shape refresh rotation
665    /// should use, for two independent reasons.
666    ///
667    /// **Correctness.** The read-then-delete pair has a window between the
668    /// SELECT and the DELETE. `consume`'s `RETURN BEFORE` already closes the
669    /// *outcome* (only one racer sees a non-empty before-image), so the pair is
670    /// safe — but it is safe by a second mechanism layered on top of a race,
671    /// rather than by not having the race. One `DELETE ... WHERE token_hash =
672    /// $h RETURN BEFORE` has no window at all: the row is selected and removed
673    /// by the same statement, and the before-image *is* the session.
674    ///
675    /// **Cost.** It halves the rotation path's datastore round trips. Refresh
676    /// is the one hot endpoint that is round-trip-bound rather than CPU-bound
677    /// (run 5: p50 88.8 ms at 545/s, a whole distribution shifted right, i.e.
678    /// added serialized work — `claude_dev/improvement-after-run5-benchmark.md`
679    /// A2/J2), so a round trip removed from it is worth more than anywhere else
680    /// in the codebase.
681    ///
682    /// Expiry is deliberately NOT filtered here. An expired token must be
683    /// consumed too — leaving the row behind would let it be replayed until
684    /// cleanup ran — so the caller receives the expired session and rejects it,
685    /// which is what the read-then-delete path did (it invalidated on the
686    /// expiry branch) with one fewer statement.

688        &self,
689        tenant_id: Uuid,
690        token_hash: &str,
691    ) -> impl Future<Output = AxiamResult<Option<Session>>> + Send;
692    /// Invalidate all sessions for a user (e.g., on password change).

699    /// change — preserve the caller's current session).

701    /// Returns the number of sessions deleted.

711    /// List all sessions for a user, tenant-scoped.
712    ///
713    /// Returns complete `Session` rows (including `token_hash`) — this method
714    /// is a raw metadata read; redacting the token/hash before surfacing the
715    /// rows to a caller (e.g. the GDPR export) is the caller's responsibility
716    /// (D-03c, SECHRD-06).
717    fn list_by_user(

1369// ---------------------------------------------------------------------------
1370// OAuth2 proof-of-possession replay (X5.1)
1371// ---------------------------------------------------------------------------
1372
1373/// Repository for the single-use `jti` values that make RFC 7523 client
1374/// assertions and RFC 9449 DPoP proofs non-replayable.
1375///
1376/// Structurally the same guarantee [`AssertionReplayRepository`] and
1377/// [`AmqpNonceRepository`] provide, and deliberately the same mechanism: a
1378/// `CREATE` against a table carrying a `UNIQUE` index, where the index
1379/// violation **is** the "already seen" answer. Nothing here reads before it
1380/// writes.
1381///
1382/// That matters more here than anywhere else in the tree. A read-then-write
1383/// replay check has a window between the two in which a second copy of the same
1384/// proof passes the read — which is precisely the race #316 and #318 spent real
1385/// effort closing for authorization codes. A proof-of-possession credential
1386/// that can be replayed once concurrently is not a proof of anything.
1387///
1388/// # Two scopes, one mechanism
1389///
1390/// The two kinds are separated by their `scope` rather than by two tables,
1391/// because the uniqueness key differs only in what the second component means:
1392///
1393/// | Kind | Scope | Why |
1394/// |---|---|---|
1395/// | client assertion | `client_id` | RFC 7523 §3 scopes `jti` uniqueness to the issuer, and the issuer of a client assertion *is* the client. Two clients that both pick `jti: "1"` have not replayed anything. |
1396/// | DPoP proof | `jkt` | RFC 9449 §11.1 scopes it to the proof key. The same reasoning: the key is the issuer. |
1397///
1398/// Both are further scoped by tenant, like everything else in AXIAM.
1399pub trait ProofReplayRepository: Send + Sync {
1400    /// Record a consumed `jti`.
1401    ///
1402    /// Returns `Ok(())` the first time a `(tenant_id, kind, scope, jti)` tuple
1403    /// is seen and `Err(AxiamError::ReplayDetected)` on every subsequent
1404    /// attempt — decided by the UNIQUE index, not by a preceding read.
1405    fn insert_proof_jti(
1406        &self,
1407        tenant_id: Uuid,
1408        kind: ProofKind,
1409        scope: &str,
1410        jti: &str,
1411        expires_at: chrono::DateTime<chrono::Utc>,
1412    ) -> impl Future<Output = AxiamResult<()>> + Send;
1413
1414    /// Delete expired rows across all tenants. Returns the number deleted.
1415    ///
1416    /// A row is only useful until the credential it records could no longer be
1417    /// accepted anyway: past that point replaying it fails on `exp` (assertions)
1418    /// or `iat` freshness (DPoP proofs) without consulting this table at all.
1419    fn cleanup_expired_proofs(&self) -> impl Future<Output = AxiamResult<u64>> + Send;
1420}
1421
1422/// Which proof-of-possession credential a replay row records.
1423///
1424/// An explicit discriminator rather than two tables because the two rows are
1425/// identical in shape and lifetime, and one table means one cleanup job and one
1426/// index to reason about. It is *not* merely cosmetic: without it, a client
1427/// whose `client_id` happened to equal some key's `jkt` would share a
1428/// uniqueness namespace with it.
1429#[derive(Debug, Clone, Copy, PartialEq, Eq)]
1430pub enum ProofKind {
1431    /// RFC 7523 §2.2 `client_assertion`, scoped by `client_id`.
1432    ClientAssertion,
1433    /// RFC 9449 DPoP proof, scoped by the proof key's `jkt`.
1434    DpopProof,
1435}
1436
1437impl ProofKind {
1438    pub const fn as_str(self) -> &'static str {
1439        match self {
1440            Self::ClientAssertion => "client_assertion",
1441            Self::DpopProof => "dpop_proof",
1442        }
1443    }
1444}

2342    // Pagination (SEC-010 / CQ-B30)
2343    //
2344    // `clamp_pagination_limit` is a resource-exhaustion control, not a
2345    // convenience: it is the only thing standing between `?limit=100000000`
2346    // and a query that tries to materialise the whole table. It runs on the
2347    // SERDE path only — direct struct construction is deliberately unaffected,
2348    // because internal callers that ask for a large page have already been
2349    // reasoned about, while a query string has not.
2350    //
2351    // Both halves of that split are asserted below. A refactor that moved the
2352    // clamp into the struct would break internal callers silently; one that
2353    // dropped it from serde would reopen the DoS with no visible change.
2354    // -----------------------------------------------------------------------
2355
2356    fn from_query(json: &str) -> Pagination {
2357        serde_json::from_str(json).expect("deserialises")
2358    }

2360    #[test]
2361    fn a_deserialised_limit_is_clamped_to_the_documented_bounds() {
2362        assert_eq!(from_query(r#"{"offset":0,"limit":100000000}"#).limit, 200);
2363        assert_eq!(from_query(r#"{"offset":0,"limit":201}"#).limit, 200);
2364        assert_eq!(from_query(r#"{"offset":0,"limit":200}"#).limit, 200);
2365        // A zero-size page is not a page; it would loop forever in any caller
2366        // that pages until it sees fewer rows than it asked for.
2367        assert_eq!(from_query(r#"{"offset":0,"limit":0}"#).limit, 1);
2368        assert_eq!(from_query(r#"{"offset":0,"limit":1}"#).limit, 1);
2369        // Anything already inside the range passes through untouched.
2370        assert_eq!(from_query(r#"{"offset":0,"limit":50}"#).limit, 50);
2371        assert_eq!(from_query(r#"{"offset":0,"limit":199}"#).limit, 199);
2372        // u64::MAX is the value an attacker actually sends.
2373        let max = format!(r#"{{"offset":0,"limit":{}}}"#, u64::MAX);
2374        assert_eq!(from_query(&max).limit, 200);
2375    }

2377    #[test]
2378    fn the_offset_is_not_clamped_because_it_bounds_no_work() {
2379        // Only `limit` decides how many rows come back, so only `limit` is a
2380        // DoS lever. A huge offset returns an empty page cheaply, and clamping
2381        // it would silently hand back page 1 to a caller that asked for page
2382        // 10 000 — a correctness bug in service of no security gain.
2383        assert_eq!(
