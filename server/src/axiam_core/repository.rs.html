60/// Clamp a pagination limit to [1, 200] at deserialization time.
61///
62/// Prevents resource exhaustion from unbounded page requests (SEC-010/CQ-B30).
63/// Direct struct construction is unaffected — only the serde path clamps.
64fn clamp_pagination_limit<'de, D>(de: D) -> Result<u64, D::Error>
65where
66    D: Deserializer<'de>,
67{
68    let raw = u64::deserialize(de)?;
69    Ok(raw.clamp(1, 200))
70}

72/// Normalise a free-text search term at deserialization time.
73///
74/// Trimmed, and blank becomes `None`. A UI that sends `?search=` on every
75/// keystroke — including after the box is cleared — should not thereby ask for
76/// "rows whose name contains the empty string", which is a different query
77/// plan for the same intent.
78///
79/// Length-capped because the term reaches a `string::contains` over every row
80/// in the tenant: an unbounded one is CPU an unauthenticated-adjacent caller
81/// gets to spend. The cap is generous enough for a full UUID, which is the
82/// longest thing anyone pastes in.
83fn normalize_search<'de, D>(de: D) -> Result<Option<String>, D::Error>
84where
85    D: Deserializer<'de>,
86{
87    const MAX_SEARCH_LEN: usize = 128;
88    let raw = Option::<String>::deserialize(de)?;
89    Ok(raw
90        .map(|s| s.trim().chars().take(MAX_SEARCH_LEN).collect::<String>())
91        .filter(|s| !s.is_empty()))
92}

94/// Pagination parameters for list queries.
95#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema, utoipa::IntoParams)]
96#[serde(default)]
97pub struct Pagination {
98    pub offset: u64,
99    #[serde(deserialize_with = "clamp_pagination_limit")]
100    pub limit: u64,
101    /// Free-text filter applied before paging.
102    ///
103    /// Matched case-insensitively against the identifying fields of whatever is
104    /// being listed — a name or username, plus the record id, so an operator
105    /// who has a UUID from a log line can paste it into the same box.
106    /// Which fields exactly is each repository's business; the shape is shared
107    /// so every list endpoint spells it the same way.
108    ///
109    /// **Applied before `offset`/`limit`, and `total` counts matches, not
110    /// rows.** Any other order gives a pager whose page count belongs to a
111    /// different result set than the page it is showing.
112    ///
113    /// `None` means no filter. Blank and whitespace-only values deserialize to
114    /// `None` — see [`normalize_search`].
115    #[serde(default, deserialize_with = "normalize_search")]
116    pub search: Option<String>,
117}
118
119impl Default for Pagination {
120    fn default() -> Self {
121        Self {
122            offset: 0,
123            limit: 50,
124            search: None,
125        }
126    }
127}
128
129impl Pagination {
130    /// The search term — trimmed, lowercased — or `None` when there is nothing
131    /// to search for.
132    ///
133    /// Lowercasing here rather than in each repository means "case-insensitive"
134    /// is decided once. A repository that compared raw would be
135    /// case-*sensitive* for exactly as long as nobody noticed.
136    ///
137    /// Trimming here as well as in [`normalize_search`] is not redundant. That
138    /// one runs on the serde path only, so it catches a query string and misses
139    /// every `Pagination` built in code — a test, an internal caller, an SDK
140    /// assembling the struct. `Some("   ")` reaching a repository asks for rows
141    /// containing three spaces, which is not what any caller meant and returns
142    /// nothing while looking like a working filter. This is the accessor every
143    /// repository actually calls, so making it authoritative is what makes the
144    /// rule true rather than usually true.
145    pub fn search_term(&self) -> Option<String> {
146        self.search
147            .as_ref()
148            .map(|s| s.trim().to_lowercase())
149            .filter(|s| !s.is_empty())
150    }
151}

280    ///
281    /// Avoids read-modify-write TOCTOU by delegating the increment to the DB
282    /// layer via a single `UPDATE ... SET field += 1` statement.
283    ///
284    /// When the new attempt count reaches `lockout_threshold`, the account is
285    /// locked until `now + d`, where `d` grows exponentially with each lockout
286    /// beyond the threshold (brute-force protection, design-document §Security):
287    ///
288    /// ```text
289    /// d = min(base_lockout_secs * backoff_multiplier ^ (new_count - threshold),
290    ///         max_lockout_secs)
291    /// ```
292    ///
293    /// so the first lockout lasts `base_lockout_secs`, the next
294    /// `base * multiplier`, and so on, capped at `max_lockout_secs`.
295    fn increment_failed_logins(
296        &self,
297        tenant_id: Uuid,
298        user_id: Uuid,
299        lockout_threshold: u32,
300        base_lockout_secs: i64,
301        backoff_multiplier: f64,
302        max_lockout_secs: i64,
303    ) -> impl Future<Output = AxiamResult<()>> + Send;

845    -> impl Future<Output = AxiamResult<()>> + Send;
846    /// Atomically consume (delete) a single session, returning `true` iff this
847    /// call removed the row. Unlike [`Self::invalidate`] (idempotent, always
848    /// `Ok(())`), this is the single-use gate for refresh-token rotation
849    /// (NEW-3): two concurrent refreshes racing on the same token both pass the
850    /// prior SELECT, but only one wins this delete — the loser gets `false` and
851    /// must abort before minting a new session, preserving single-use rotation
852    /// and stolen-token reuse detection.
853    fn consume(&self, tenant_id: Uuid, id: Uuid) -> impl Future<Output = AxiamResult<bool>> + Send;
854    /// Atomically look up **and** consume a session by its refresh-token hash,
855    /// returning the row as it was immediately before deletion (`None` when no
856    /// live session matched).
857    ///
858    /// This is [`Self::get_by_token_hash`] followed by [`Self::consume`],
859    /// collapsed into one statement — and it is the shape refresh rotation
860    /// should use, for two independent reasons.
861    ///
862    /// **Correctness.** The read-then-delete pair has a window between the
863    /// SELECT and the DELETE. `consume`'s `RETURN BEFORE` already closes the
864    /// *outcome* (only one racer sees a non-empty before-image), so the pair is
865    /// safe — but it is safe by a second mechanism layered on top of a race,
866    /// rather than by not having the race. One `DELETE ... WHERE token_hash =
867    /// $h RETURN BEFORE` has no window at all: the row is selected and removed
868    /// by the same statement, and the before-image *is* the session.
869    ///
870    /// **Cost.** It halves the rotation path's datastore round trips. Refresh
871    /// is the one hot endpoint that is round-trip-bound rather than CPU-bound
872    /// (run 5: p50 88.8 ms at 545/s, a whole distribution shifted right, i.e.
873    /// added serialized work — `claude_dev/improvement-after-run5-benchmark.md`
874    /// A2/J2), so a round trip removed from it is worth more than anywhere else
875    /// in the codebase.
876    ///
877    /// Expiry is deliberately NOT filtered here. An expired token must be
878    /// consumed too — leaving the row behind would let it be replayed until
879    /// cleanup ran — so the caller receives the expired session and rejects it,
880    /// which is what the read-then-delete path did (it invalidated on the
881    /// expiry branch) with one fewer statement.
882    fn consume_by_token_hash(
883        &self,
884        tenant_id: Uuid,
885        token_hash: &str,
886    ) -> impl Future<Output = AxiamResult<Option<Session>>> + Send;

1668// ---------------------------------------------------------------------------
1669// OAuth2 proof-of-possession replay (X5.1)
1670// ---------------------------------------------------------------------------
1671
1672/// Repository for the single-use `jti` values that make RFC 7523 client
1673/// assertions and RFC 9449 DPoP proofs non-replayable.
1674///
1675/// Structurally the same guarantee [`AssertionReplayRepository`] and
1676/// [`AmqpNonceRepository`] provide, and deliberately the same mechanism: a
1677/// `CREATE` against a table carrying a `UNIQUE` index, where the index
1678/// violation **is** the "already seen" answer. Nothing here reads before it
1679/// writes.
1680///
1681/// That matters more here than anywhere else in the tree. A read-then-write
1682/// replay check has a window between the two in which a second copy of the same
1683/// proof passes the read — which is precisely the race #316 and #318 spent real
1684/// effort closing for authorization codes. A proof-of-possession credential
1685/// that can be replayed once concurrently is not a proof of anything.
1686///
1687/// # Two scopes, one mechanism
1688///
1689/// The two kinds are separated by their `scope` rather than by two tables,
1690/// because the uniqueness key differs only in what the second component means:
1691///
1692/// | Kind | Scope | Why |
1693/// |---|---|---|
1694/// | client assertion | `client_id` | RFC 7523 §3 scopes `jti` uniqueness to the issuer, and the issuer of a client assertion *is* the client. Two clients that both pick `jti: "1"` have not replayed anything. |
1695/// | DPoP proof | `jkt` | RFC 9449 §11.1 scopes it to the proof key. The same reasoning: the key is the issuer. |
1696///
1697/// Both are further scoped by tenant, like everything else in AXIAM.
1698pub trait ProofReplayRepository: Send + Sync {
1699    /// Record a consumed `jti`.
1700    ///
1701    /// Returns `Ok(())` the first time a `(tenant_id, kind, scope, jti)` tuple
1702    /// is seen and `Err(AxiamError::ReplayDetected)` on every subsequent
1703    /// attempt — decided by the UNIQUE index, not by a preceding read.
1704    fn insert_proof_jti(
1705        &self,
1706        tenant_id: Uuid,
1707        kind: ProofKind,
1708        scope: &str,
1709        jti: &str,
1710        expires_at: chrono::DateTime<chrono::Utc>,
1711    ) -> impl Future<Output = AxiamResult<()>> + Send;
1712
1713    /// Delete expired rows across all tenants. Returns the number deleted.
1714    ///
1715    /// A row is only useful until the credential it records could no longer be
1716    /// accepted anyway: past that point replaying it fails on `exp` (assertions)
1717    /// or `iat` freshness (DPoP proofs) without consulting this table at all.
1718    fn cleanup_expired_proofs(&self) -> impl Future<Output = AxiamResult<u64>> + Send;
1719}
1720
1721/// Which proof-of-possession credential a replay row records.
1722///
1723/// An explicit discriminator rather than two tables because the two rows are
1724/// identical in shape and lifetime, and one table means one cleanup job and one
1725/// index to reason about. It is *not* merely cosmetic: without it, a client
1726/// whose `client_id` happened to equal some key's `jkt` would share a

2571// ---------------------------------------------------------------------------
2572//

2574// effective policy (resolved by its caller from `WebauthnAttestationPolicyRepository`
