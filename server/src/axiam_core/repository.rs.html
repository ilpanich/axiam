74/// Clamp a pagination limit to [1, 200] at deserialization time.
75///
76/// Prevents resource exhaustion from unbounded page requests (SEC-010/CQ-B30).
77/// Direct struct construction is unaffected — only the serde path clamps.
78fn clamp_pagination_limit<'de, D>(de: D) -> Result<u64, D::Error>
79where
80    D: Deserializer<'de>,
81{
82    let raw = u64::deserialize(de)?;
83    Ok(raw.clamp(1, 200))
84}

86/// Normalise a free-text search term at deserialization time.
87///
88/// Trimmed, and blank becomes `None`. A UI that sends `?search=` on every
89/// keystroke — including after the box is cleared — should not thereby ask for
90/// "rows whose name contains the empty string", which is a different query
91/// plan for the same intent.
92///
93/// Length-capped because the term reaches a `string::contains` over every row
94/// in the tenant: an unbounded one is CPU an unauthenticated-adjacent caller
95/// gets to spend. The cap is generous enough for a full UUID, which is the
96/// longest thing anyone pastes in.
97fn normalize_search<'de, D>(de: D) -> Result<Option<String>, D::Error>
98where
99    D: Deserializer<'de>,
100{
101    const MAX_SEARCH_LEN: usize = 128;
102    let raw = Option::<String>::deserialize(de)?;
103    Ok(raw
104        .map(|s| s.trim().chars().take(MAX_SEARCH_LEN).collect::<String>())
105        .filter(|s| !s.is_empty()))
106}

108/// Pagination parameters for list queries.
109#[derive(Debug, Clone, Serialize, Deserialize, utoipa::ToSchema, utoipa::IntoParams)]
110#[serde(default)]
111pub struct Pagination {
112    pub offset: u64,
113    #[serde(deserialize_with = "clamp_pagination_limit")]
114    pub limit: u64,
115    /// Free-text filter applied before paging.
116    ///
117    /// Matched case-insensitively against the identifying fields of whatever is
118    /// being listed — a name or username, plus the record id, so an operator
119    /// who has a UUID from a log line can paste it into the same box.
120    /// Which fields exactly is each repository's business; the shape is shared
121    /// so every list endpoint spells it the same way.
122    ///
123    /// **Applied before `offset`/`limit`, and `total` counts matches, not
124    /// rows.** Any other order gives a pager whose page count belongs to a
125    /// different result set than the page it is showing.
126    ///
127    /// `None` means no filter. Blank and whitespace-only values deserialize to
128    /// `None` — see [`normalize_search`].
129    #[serde(default, deserialize_with = "normalize_search")]
130    pub search: Option<String>,
131}
132
133impl Default for Pagination {
134    fn default() -> Self {
135        Self {
136            offset: 0,
137            limit: 50,
138            search: None,
139        }
140    }
141}
142
143impl Pagination {
144    /// The search term — trimmed, lowercased — or `None` when there is nothing
145    /// to search for.
146    ///
147    /// Lowercasing here rather than in each repository means "case-insensitive"
148    /// is decided once. A repository that compared raw would be
149    /// case-*sensitive* for exactly as long as nobody noticed.
150    ///
151    /// Trimming here as well as in [`normalize_search`] is not redundant. That
152    /// one runs on the serde path only, so it catches a query string and misses
153    /// every `Pagination` built in code — a test, an internal caller, an SDK
154    /// assembling the struct. `Some("   ")` reaching a repository asks for rows
155    /// containing three spaces, which is not what any caller meant and returns
156    /// nothing while looking like a working filter. This is the accessor every
157    /// repository actually calls, so making it authoritative is what makes the
158    /// rule true rather than usually true.
159    pub fn search_term(&self) -> Option<String> {
160        self.search
161            .as_ref()
162            .map(|s| s.trim().to_lowercase())
163            .filter(|s| !s.is_empty())
164    }
165}

1117    /// should use, for two independent reasons.
1118    ///
1119    /// **Correctness.** The read-then-delete pair has a window between the
1120    /// SELECT and the DELETE. `consume`'s `RETURN BEFORE` already closes the
1121    /// *outcome* (only one racer sees a non-empty before-image), so the pair is
1122    /// safe — but it is safe by a second mechanism layered on top of a race,

1128    /// is the one hot endpoint that is round-trip-bound rather than CPU-bound
1129    /// (run 5: p50 88.8 ms at 545/s, a whole distribution shifted right, i.e.

2198
2199    /// Rotate a refresh token out of use, leaving it usable until `grace_until`.
2200    ///
2201    /// # Why rotation is not revocation
2202    ///
2203    /// FAPI 2.0 Security Profile §5.3.2.1-9 requires an authorization server
2204    /// that rotates refresh tokens to keep accepting the previous one for a
2205    /// short period after the new one is issued. The case it exists for is a
2206    /// client that never received the rotation response — a dropped connection
2207    /// after the server committed the write — which under immediate revocation
2208    /// is locked out permanently, holding a token the server has destroyed and
2209    /// no way to ask for another.
2210    ///
2211    /// # Why this is expressed as an expiry and not a second flag
2212    ///
2213    /// [`Self::get_by_token_hash`] already refuses an expired token, so a
2214    /// shortened `expires_at` needs no new condition on the read path, no new
2215    /// column, and no change to the several other callers of [`Self::revoke`]
2216    /// — every one of which means "this grant is over" rather than "this token
2217    /// has been succeeded", and none of which should acquire a grace period by
2218    /// sharing a code path with this.
2219    ///
2220    /// Implementations MUST only ever bring `expires_at` forward. A refresh
2221    /// token issued with thirty days on it must not have its life *extended*
2222    /// to the grace instant by a rotation, and a caller passing a distant
2223    /// `grace_until` must not be able to resurrect one that has already run
2224    /// out.
2225    ///
2226    /// Returns `NotFound` when no live token matched, exactly as
2227    /// [`Self::revoke`] does, so that concurrent use is still detectable.
2228    ///
2229    /// # T-254: this also stamps `rotated_at`
2230    ///
2231    /// A superseded token stays redeemable, so the only thing that
2232    /// distinguishes the honest retry the grace exists for from a leaked token
2233    /// being replayed is that the row *has a successor*. Implementations MUST
2234    /// therefore set [`RefreshToken::rotated_at`] in the same statement that
2235    /// brings the expiry forward — in the same statement, because a second
2236    /// write could be lost and leave a rotated token that does not say so.
2237    ///
2238    /// Only ever called for a client whose registration is `profile: fapi2`.

2246    ) -> impl Future<Output = AxiamResult<()>> + Send;

2249    /// (T-254).
2250    ///
2251    /// The rotation path for every client that is **not** on the `fapi2`
2252    /// profile — which is every client registered today. The token is revoked
2253    /// outright, exactly as it was before 065f37c: a second presentation finds
2254    /// no live row and is refused.
2255    ///
2256    /// # Why not just [`Self::revoke`]
2257    ///
2258    /// Because the two mean different things and T-254 needs to tell them
2259    /// apart. `revoke` is "this grant is over" — logout, a password reset, an
2260    /// RFC 7009 request — and a later presentation of such a token is an
2261    /// ordinary stale credential. This is "this token has been succeeded", and
2262    /// a later presentation of *this* is a replay. The difference is recorded

2264    /// does not touch.
2265    ///

2281    /// the same reason: when a presented credential is refused, the server
2282    /// still has to be able to say *which session* was replayed against, and
2283    /// the read path that refuses it has already filtered the row away.
2284    ///
2285    /// Neither `revoked` nor `expires_at` is filtered — a replay of a token

2497/// A one-shot code that converts a cross-site SSO return into a same-site
2498/// session issuance.
2499///
2500/// # Why this exists
2501///
2502/// AXIAM's session cookies are `SameSite=Strict`. SAML and Apple's
2503/// `response_mode=form_post` both have the IdP perform a **cross-site form
2504/// POST** directly to an AXIAM endpoint; setting `Strict` cookies on that
2505/// response is permitted, but the browser will not *send* them on the
2506/// navigation that follows, so the user lands back in the SPA with a session
2507/// they cannot use.
2508///
2509/// So the receiving endpoint mints one of these instead of setting cookies, and
2510/// `303`-redirects to the SPA with it. The SPA posts it back **same-origin**,
2511/// and that response sets the cookies. The alternative — weakening the cookies
2512/// to `SameSite=Lax` — would re-open the CSRF surface `Strict` closes, across
2513/// every endpoint, permanently, to serve two flows.
2514///
2515/// The row carries **no token material**: the access and refresh tokens are
2516/// created at redemption, from `user_id` and `tenant_id`, not stored here.
2517/// The stored value is a SHA-256 hash of the code, so a database read does not
2518/// yield a usable credential.
2519///
2520/// See `claude_dev/federation-sso-login-design.md` §5.2.
2521#[derive(Debug, Clone)]
2522pub struct SsoHandoffCode {
2523    /// Lowercase hex SHA-256 of the code. The code itself is never stored.
2524    pub code_hash: String,
2525    pub tenant_id: uuid::Uuid,
2526    pub user_id: uuid::Uuid,
2527    /// SPA destination, carried through from the login state row.
2528    pub redirect_uri: String,
2529    pub expires_at: chrono::DateTime<chrono::Utc>,
2530    /// X7.2 — when the *upstream* identity provider authenticated the user.
2531    ///
2532    /// Carried across the handoff hop because the session is created on the
2533    /// next request, up to 60 seconds later, and by a handler that never saw
2534    /// the assertion. Stamping the clock at redemption would date a federated
2535    /// login by AXIAM's clock rather than the provider's, which is precisely
2536    /// the overstatement plan §4.3 forbids.
2537    ///
2538    /// `None` on a row written before schema v55 (at most 60 seconds old at
2539    /// any deploy) and whenever the assertion carried no authentication
2540    /// instant; redemption then falls back to its own clock.
2541    pub authenticated_at: Option<chrono::DateTime<chrono::Utc>>,
2542}
2543
2544/// How long a handoff code is valid.
2545///
2546/// It exists to survive exactly one redirect. Sixty seconds is generous for
2547/// that and short enough that a code caught in browser history or a `Referer`
2548/// header is almost certainly already dead. The login state row it descends
2549/// from gets ten minutes because a human is reading an IdP consent screen in
2550/// the middle of it; nobody is reading anything during this hop.
2551pub const SSO_HANDOFF_TTL_SECS: i64 = 60;

2661// OAuth2 proof-of-possession replay (X5.1)
2662// ---------------------------------------------------------------------------
2663
2664/// Repository for the single-use `jti` values that make RFC 7523 client
2665/// assertions and RFC 9449 DPoP proofs non-replayable.
2666///
2667/// Structurally the same guarantee [`AssertionReplayRepository`] and
2668/// [`AmqpNonceRepository`] provide, and deliberately the same mechanism: a
2669/// `CREATE` against a table carrying a `UNIQUE` index, where the index
2670/// violation **is** the "already seen" answer. Nothing here reads before it
2671/// writes.
2672///
2673/// That matters more here than anywhere else in the tree. A read-then-write
2674/// replay check has a window between the two in which a second copy of the same
2675/// proof passes the read — which is precisely the race #316 and #318 spent real
2676/// effort closing for authorization codes. A proof-of-possession credential
2677/// that can be replayed once concurrently is not a proof of anything.
2678///
2679/// # Two scopes, one mechanism
2680///
2681/// The two kinds are separated by their `scope` rather than by two tables,
2682/// because the uniqueness key differs only in what the second component means:
2683///
2684/// | Kind | Scope | Why |
2685/// |---|---|---|
2686/// | client assertion | `client_id` | RFC 7523 §3 scopes `jti` uniqueness to the issuer, and the issuer of a client assertion *is* the client. Two clients that both pick `jti: "1"` have not replayed anything. |
2687/// | DPoP proof | `jkt` | RFC 9449 §11.1 scopes it to the proof key. The same reasoning: the key is the issuer. |
2688///
2689/// Both are further scoped by tenant, like everything else in AXIAM.
2690pub trait ProofReplayRepository: Send + Sync {
2691    /// Record a consumed `jti`.
2692    ///
2693    /// Returns `Ok(())` the first time a `(tenant_id, kind, scope, jti)` tuple
2694    /// is seen and `Err(AxiamError::ReplayDetected)` on every subsequent
2695    /// attempt — decided by the UNIQUE index, not by a preceding read.
2696    fn insert_proof_jti(
2697        &self,
2698        tenant_id: Uuid,
2699        kind: ProofKind,
2700        scope: &str,
2701        jti: &str,
2702        expires_at: chrono::DateTime<chrono::Utc>,
2703    ) -> impl Future<Output = AxiamResult<()>> + Send;
2704
2705    /// Delete expired rows across all tenants. Returns the number deleted.
2706    ///
2707    /// A row is only useful until the credential it records could no longer be
2708    /// accepted anyway: past that point replaying it fails on `exp` (assertions)
2709    /// or `iat` freshness (DPoP proofs) without consulting this table at all.
2710    fn cleanup_expired_proofs(&self) -> impl Future<Output = AxiamResult<u64>> + Send;
2711}

2718/// whose `client_id` happened to equal some key's `jkt` would share a
2719/// uniqueness namespace with it.
2720#[derive(Debug, Clone, Copy, PartialEq, Eq)]
2721pub enum ProofKind {
2722    /// RFC 7523 §2.2 `client_assertion`, scoped by `client_id`.

2729}

4715    /// # Why this table gains a delete, and why that is not a hole

4717    /// The rest of this trait treats a consent record as write-once, and the

4731    /// than by every caller remembering it.
4732    ///
4733    /// The record of what happened is not lost with the row. Granting and
4734    /// withdrawing both emit audit events, and the audit log is the
4735    /// append-only, signature-chained store — which is where a history belongs.
4736    /// This table holds *live* consent: the question the UserInfo endpoint asks
