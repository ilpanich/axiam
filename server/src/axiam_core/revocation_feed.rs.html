1//! The session-revocation feed (T-39, T-143).
2//!
3//! # The gap this narrows
4//!
5//! An AXIAM access token is self-contained and valid for up to fifteen
6//! minutes, and an SDK route guard verifies it locally. A role removal, an
7//! account disable or a logout therefore does not reach a token already in a
8//! caller's hands: the two threats are the server-side and client-side faces
9//! of one trade, and both recorded the same remedy — call gRPC introspection
10//! instead of verifying locally. That remedy is real and costs a network round
11//! trip **per request**, which is why integrators demonstrably do not adopt it.
12//!
13//! The feed is the cheaper shape of the same answer: a small, bounded,
14//! unlinkable list of revoked sessions, published beside the JWKS, polled on an
15//! interval rather than per request. A guard that polls it rejects a revoked
16//! session within one poll interval instead of within one token lifetime.
17//!
18//! # Five properties, each load-bearing
19//!
20//! 1. **Hashes, never identifiers.** An entry is
21//!    [`revocation_hash`] of the session id — base64url-unpadded SHA-256, the
22//!    same encoding `cnf.x5t#S256` and `jkt` already use, so no SDK needs a new
23//!    primitive. A `sid` is a session id and not a subject, so the feed
24//!    discloses neither who was revoked nor how many distinct users are behind
25//!    the entries. This is not a privacy *guarantee* — it is a
26//!    non-enumerability argument, and it holds because a `sid` is a UUIDv4 and
27//!    the preimage space is not walkable, not because a hash is magic.
28//! 2. **Bounded.** An entry lives exactly one access-token lifetime past the
29//!    revocation. After that every token naming that session has expired on its
30//!    own `exp` and the entry proves nothing. So the document's size tracks the
31//!    revocation rate over fifteen minutes, never the deployment's history.
32//! 3. **Rate-limited and cacheable**, like the JWKS beside it.
33//! 4. **Never fail closed on the feed.** A guard that cannot fetch it, or that
34//!    fetches something malformed, MUST behave exactly as it does today. This
35//!    is the rule that makes the feature safe to ship at all: a revocation feed
36//!    that can deny requests when it is unreachable turns a network blip into
37//!    an outage, and would be a worse control than the window it narrows. The
38//!    feed can only ever turn an accept into a reject, never the reverse.
39//! 5. **Additive to the token.** Nothing about the JWT changes. `sid` has been
40//!    there since T-249.
41//!
42//! Off by default, per deployment. A deployment that does not enable it writes
43//! no row, serves no route, and is byte-identical to one built before this
44//! module existed.
45
46use base64::Engine;
47use base64::engine::general_purpose::URL_SAFE_NO_PAD;
48use sha2::{Digest, Sha256};
49use uuid::Uuid;
50
51/// The digest algorithm the feed publishes, and the only value a client may
52/// accept in the document's `alg` member.
53///
54/// Published so a client can refuse a document it cannot interpret rather than
55/// silently matching nothing — which is the failure mode that would make a
56/// guard *appear* to honour revocations while honouring none.
57pub const REVOCATION_HASH_ALG: &str = "SHA-256";
58
59/// The feed entry for a session id.
60///
61/// Base64url without padding, over the session id's **canonical hyphenated
62/// lowercase text form** — the same bytes that appear in the token's `sid`
63/// claim. Hashing the UUID's raw 16 bytes instead would be equally secure and
64/// would not interoperate, because a guard computes this from the string it
65/// read out of a JWT.
66#[must_use]
67pub fn revocation_hash(session_id: Uuid) -> String {
68    revocation_hash_of(&session_id.to_string())
69}
70
71/// [`revocation_hash`] for a `sid` that has already been read as text.
72///
73/// What an SDK guard calls: it holds the claim, not a parsed UUID, and
74/// requiring it to parse first would make a guard's answer depend on its UUID
75/// parser's strictness rather than on the feed.
76#[must_use]
77pub fn revocation_hash_of(sid: &str) -> String {
78    let mut hasher = Sha256::new();
79    hasher.update(sid.as_bytes());
80    URL_SAFE_NO_PAD.encode(hasher.finalize())
81}

83#[cfg(test)]
84mod tests {
85    use super::*;
86
87    /// The wire format, pinned. Eleven SDKs compute this independently, and a
88    /// change here is a change every one of them silently stops matching —
89    /// which presents as "revocation stopped working" with nothing failing.
90    #[test]
91    fn the_hash_is_base64url_unpadded_sha256_of_the_canonical_text_form() {
92        let sid = Uuid::parse_str("6f3e0a5c-1b2d-4e8f-9a7b-0c1d2e3f4a5b").unwrap();
93        let hash = revocation_hash(sid);
94
95        // 32 bytes base64url-unpadded is 43 characters, and none of them may be
96        // `+`, `/` or `=` — a standard-alphabet encoding here would be a subtle
97        // mismatch that only shows up for some inputs.
98        assert_eq!(hash.len(), 43);
99        assert!(!hash.contains('+') && !hash.contains('/') && !hash.contains('='));
100
101        // Computed over the hyphenated lowercase text, which is what the `sid`
102        // claim carries.
103        assert_eq!(
104            hash,
105            revocation_hash_of("6f3e0a5c-1b2d-4e8f-9a7b-0c1d2e3f4a5b")
106        );
107        assert_eq!(hash, "i9N2lYMTV4FhA0husWjGYCqJXXTb7_fMBuomhWjSsgQ");
108    }
109
110    /// A guard reads the claim as text and must reach the same entry as the
111    /// server, which parsed a UUID. An uppercase or braced rendering is a
112    /// *different* string and hashes differently — which is correct, and is
113    /// why the contract says the canonical form.
114    #[test]
115    fn the_text_form_is_the_canonical_one_and_nothing_else() {
116        let sid = Uuid::parse_str("6F3E0A5C-1B2D-4E8F-9A7B-0C1D2E3F4A5B").unwrap();
117        // `Uuid::to_string` normalises, so the server's entry is the lowercase
118        // one whatever case the id was written in.
119        assert_eq!(
120            revocation_hash(sid),
121            revocation_hash_of("6f3e0a5c-1b2d-4e8f-9a7b-0c1d2e3f4a5b")
122        );
123        assert_ne!(
124            revocation_hash(sid),
125            revocation_hash_of("6F3E0A5C-1B2D-4E8F-9A7B-0C1D2E3F4A5B")
126        );
127    }
128
129    /// Distinct sessions produce distinct entries, and the same session
130    /// produces the same one — so a session revoked twice is one row, not two.
131    #[test]
132    fn the_hash_is_deterministic_and_distinguishing() {
133        let a = Uuid::new_v4();
134        let b = Uuid::new_v4();
135        assert_eq!(revocation_hash(a), revocation_hash(a));
136        assert_ne!(revocation_hash(a), revocation_hash(b));
137    }
138
139    /// The entry must not be the id. Stated as a test because the cheap
140    /// "optimisation" — publish the `sid` and skip the hashing on both sides —
141    /// is exactly what turns a public document into a disclosure.
142    #[test]
143    fn the_entry_never_contains_the_session_id() {
144        let sid = Uuid::new_v4();
145        let text = sid.to_string();
146        let hash = revocation_hash(sid);
147        assert!(!hash.contains(&text));
148        for window in text
149            .as_bytes()
150            .windows(4)
151            .map(|w| std::str::from_utf8(w).unwrap())
152        {
153            assert!(!hash.contains(window), "the entry leaks {window:?}");
154        }
155    }
