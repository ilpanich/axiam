1//! Adapter implementing `axiam_core::repository::AttestationMetadataSource`
2//! over any [`MdsRepository`] (X3 wave 2, W2-D4).
3//!
4//! This is the "implemented over the repositories" half of W2-D4: it exists
5//! so `axiam-auth` can depend on the small, data-only
6//! [`axiam_core::repository::AttestationMetadataSource`] trait instead of a
7//! hard `axiam-pki`/`axiam-db` dependency, while the real lookup still goes
8//! through the same `mds_entry` table `MdsRepository` reads.
9
10use axiam_core::error::AxiamResult;
11use axiam_core::models::mds::MdsEntry;
12use axiam_core::repository::{AttestationMetadataSource, AttestationRootMaterial, MdsRepository};
13use base64::Engine;
14use base64::engine::general_purpose::STANDARD;
15use uuid::Uuid;
16
17/// Wraps any [`MdsRepository`] to serve [`AttestationMetadataSource`].
18///
19/// `attestation_roots` decodes each entry's `attestation_root_certificates`
20/// (base64 DER, as MDS ships them — see `axiam_core::models::mds::MdsEntry`
21/// docs) into raw DER bytes, flattened to one `AttestationRootMaterial` per
22/// `(aaguid, root certificate)` pair. A cert that fails to base64-decode is
23/// skipped and logged rather than failing the whole lookup — one malformed
24/// MDS entry must not take down attestation for every other authenticator.
25#[derive(Clone)]
26pub struct MdsAttestationMetadataSource<M: MdsRepository> {
27    mds: M,
28}
29
30impl<M: MdsRepository> MdsAttestationMetadataSource<M> {
31    pub fn new(mds: M) -> Self {
32        Self { mds }
33    }
34}
35
36impl<M: MdsRepository + Send + Sync> AttestationMetadataSource for MdsAttestationMetadataSource<M> {
37    async fn get_entry(&self, aaguid: Uuid) -> AxiamResult<Option<MdsEntry>> {
38        self.mds.get_by_aaguid(aaguid).await
39    }
40
41    async fn attestation_roots(
42        &self,
43        allowed_aaguids: Option<&[Uuid]>,
44    ) -> AxiamResult<Vec<AttestationRootMaterial>> {
45        let entries: Vec<MdsEntry> = match allowed_aaguids {
46            Some(allowed) => {
47                let mut out = Vec::with_capacity(allowed.len());
48                for &aaguid in allowed {
49                    if let Some(entry) = self.mds.get_by_aaguid(aaguid).await? {
50                        out.push(entry);
51                    }
52                }
53                out
54            }
55            None => self.mds.list_all().await?,
56        };
57
58        let mut roots = Vec::new();
59        for entry in entries {
60            for cert_b64 in &entry.attestation_root_certificates {
61                match STANDARD.decode(cert_b64) {
62                    Ok(der) => roots.push(AttestationRootMaterial {
63                        aaguid: entry.aaguid,
64                        der,
65                        description: entry
66                            .description
67                            .clone()
68                            .unwrap_or_else(|| entry.aaguid.to_string()),
69                    }),
70                    Err(e) => {
71                        tracing::warn!(
72                            aaguid = %entry.aaguid,
73                            error = %e,
74                            "skipping unparsable MDS attestation root certificate"
75                        );
76                    }
77                }
78            }
79        }
80        Ok(roots)
81    }
82}
