1//! [`DbHandle`] — a **live** reference to one pooled SurrealDB connection.
2//!
3//! ## The bug this type exists to close
4//!
5//! Before this type, every repository was constructed from a one-time
6//! `pool.handle_for_repo().await` **clone** of the pooled `Surreal<C>` — a
7//! snapshot taken once at boot and then held for the process lifetime. That is
8//! fine right up until the pool's reconnect loop
9//! (`DbManager::spawn_reconnect_loop`, PERF-04/D-12) decides the pooled
10//! connection is poisoned and swaps a brand-new one into the pool slot.
11//!
12//! The swap replaced what the *pool* pointed at, but every repository was still
13//! holding its own clone of the **evicted** connection. Those clones share the
14//! old `Arc<inner>` — the old router task and the old `reqwest` client with the
15//! old (now-rejected) cached auth token — so from that moment on every
16//! repository query 401s, permanently, until the process restarts. Observed
17//! ~7 minutes into a sustained load run during the rate-limit verification: a
18//! cell starts clean and then produces a wall of 401s with the server otherwise
19//! reporting healthy, cleared instantly by `docker restart`.
20//!
21//! Readiness did not catch it either: `DbPool::health_check` probes through the
22//! pool slot, so it observes the *fresh* handle and reports healthy while the
23//! repositories are all talking to the dead one.
24//!
25//! ## The fix
26//!
27//! [`DbHandle`] holds the pool **slot** (`Arc<ArcSwap<Surreal<C>>>`) rather than
28//! a connection cloned out of it, and resolves the current connection at every
29//! call via [`DbHandle::current`]. A reconnect-loop swap is therefore observed
30//! by the very next query on every repository already built from that slot —
31//! no restart, no permanent 401 wall.
32//!
33//! ## Why `ArcSwap` and not `RwLock`
34//!
35//! The slot is read on every single database call and written essentially never
36//! (only by a reconnect). `ArcSwap::load_full` is a lock-free atomic read, so
37//! the hot path takes no lock, cannot be delayed behind a queued writer, and —
38//! importantly for the ~250 repository call sites — stays **synchronous**, so
39//! resolving the current handle adds no `.await` point to any query path.
40
41use std::sync::Arc;
42
43use arc_swap::ArcSwap;
44use surrealdb::engine::remote::http::Client;
45use surrealdb::{Connection, Surreal};
46
47/// The swappable pool slot shared between a pooled handle, its background
48/// refresh/reconnect tasks, and every [`DbHandle`] bound to it.
49///
50/// Writing this slot (a reconnect-loop eviction) is atomically visible to every
51/// holder's next [`DbHandle::current`] call — that visibility *is* the fix.
52pub(crate) type DbSlot<C> = Arc<ArcSwap<Surreal<C>>>;
53
54/// Build a fresh slot around an already-connected handle.
55pub(crate) fn new_slot<C: Connection>(db: Surreal<C>) -> DbSlot<C> {
56    Arc::new(ArcSwap::from_pointee(db))
57}
58
59/// A live reference to one pooled SurrealDB connection.
60///
61/// Repositories hold this instead of an owned `Surreal<C>` so that a
62/// reconnect-loop handle swap is picked up on their next query rather than
63/// leaving them pinned to an evicted connection forever (see the module docs).
64///
65/// Cloning is cheap — clones share the same slot, so they all observe the same
66/// swaps. `Surreal<C>` converts into a `DbHandle<C>` via [`From`], which keeps
67/// tests (and any caller that legitimately owns a standalone connection, such
68/// as an embedded `kv-mem` instance) able to construct repositories exactly as
69/// before; such a handle simply has a slot that nothing ever swaps.
70pub struct DbHandle<C: Connection = Client> {
71    slot: DbSlot<C>,
72}
73
74impl<C: Connection> DbHandle<C> {
75    /// Wrap an existing pool slot. Crate-internal: outside this crate a
76    /// `DbHandle` is obtained from [`crate::DbPool::handle_for_repo`] (live) or
77    /// via `From<Surreal<C>>` (standalone).
78    pub(crate) fn from_slot(slot: DbSlot<C>) -> Self {
79        Self { slot }
80    }
81
82    /// The connection to use for **this** operation.
83    ///
84    /// Resolves the slot on every call, so a reconnect-loop swap is observed
85    /// immediately. The returned `Arc<Surreal<C>>` derefs to the connection, so
86    /// call sites read as `self.db.current().query(..)`.
87    ///
88    /// Lock-free (a single atomic load), so this is safe and cheap to call on
89    /// every query and never blocks behind the reconnect loop.
90    pub fn current(&self) -> Arc<Surreal<C>> {
91        self.slot.load_full()
92    }
93}
94
95impl<C: Connection> Clone for DbHandle<C> {
96    /// Clones share the SAME slot — every clone observes every swap. (Derived
97    /// `Clone` would demand `C: Clone`, which `Connection` does not imply.)
98    fn clone(&self) -> Self {
99        Self {
100            slot: Arc::clone(&self.slot),
101        }
102    }
103}
104
105impl<C: Connection> From<Surreal<C>> for DbHandle<C> {
106    /// Wrap a standalone connection in its own slot. Nothing swaps this slot,
107    /// so the result behaves exactly like holding the connection directly —
108    /// this is the path tests and embedded `kv-mem` callers take.
109    fn from(db: Surreal<C>) -> Self {
110        Self::from_slot(new_slot(db))
111    }
112}
113
114impl<C: Connection> std::fmt::Debug for DbHandle<C> {
115    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
116        f.debug_struct("DbHandle").finish_non_exhaustive()
117    }
118}

120#[cfg(test)]
121mod tests {
122    use super::*;
123    use surrealdb::engine::local::{Db, Mem};

125    async fn marked(marker: &str) -> Surreal<Db> {
126        let db = Surreal::new::<Mem>(()).await.expect("in-memory connect");
127        db.use_ns("test").use_db("test").await.expect("use ns/db");
128        db.query(format!("CREATE marker SET value = '{marker}'"))
129            .await
130            .and_then(|r| r.check())
131            .expect("seed marker row");
132        db
133    }
134
135    async fn read_marker(db: &Surreal<Db>) -> Vec<String> {
136        let mut result = db
137            .query("SELECT VALUE value FROM marker")
138            .await
139            .and_then(|r| r.check())
140            .expect("read marker rows");
141        result.take(0).expect("deserialize marker values")
142    }
143
144    /// THE regression test for the stale-handle bug: a handle taken *before* a
145    /// slot swap must observe the new connection on its next `current()`, not
146    /// stay pinned to the evicted one.
147    #[tokio::test]
148    async fn current_observes_a_swap_made_after_the_handle_was_taken() {
149        let slot = new_slot(marked("old").await);
150        // Bound once, up-front — exactly how a repository binds at boot.
151        let handle = DbHandle::from_slot(Arc::clone(&slot));
152        assert_eq!(
153            read_marker(&handle.current()).await,
154            vec!["old".to_string()]
155        );
156
157        // The reconnect loop evicts the poisoned connection.
158        slot.store(Arc::new(marked("new").await));
159
160        assert_eq!(
161            read_marker(&handle.current()).await,
162            vec!["new".to_string()],
163            "a handle bound before the swap must follow the slot, not stay \
164             pinned to the evicted connection"
165        );
166    }
167
168    /// Clones share the slot, so a swap reaches every repository built from it
169    /// (repositories are `.clone()`d liberally into per-worker app state).
170    #[tokio::test]
171    async fn clones_share_the_slot_and_all_observe_the_swap() {
172        let slot = new_slot(marked("old").await);
173        let a = DbHandle::from_slot(Arc::clone(&slot));
174        let b = a.clone();
175        let c = b.clone();
176
177        slot.store(Arc::new(marked("new").await));
178
179        for (name, h) in [("a", &a), ("b", &b), ("c", &c)] {
180            assert_eq!(
181                read_marker(&h.current()).await,
182                vec!["new".to_string()],
183                "clone {name} must observe the swap"
184            );
185        }
186    }
187
188    /// `From<Surreal<C>>` yields a working, never-swapped handle — the shape
189    /// tests and embedded callers rely on.
190    #[tokio::test]
191    async fn from_surreal_yields_a_usable_standalone_handle() {
192        let handle: DbHandle<Db> = marked("standalone").await.into();
193        assert_eq!(
194            read_marker(&handle.current()).await,
195            vec!["standalone".to_string()]
196        );
197    }
