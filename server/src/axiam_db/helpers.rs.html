29/// Parse a UUID string coming out of SurrealDB, embedding the field name in
30/// the error message for easier debugging.
31///
32/// Unlike the inline `Uuid::parse_str(..).map_err(|e| DbError::Migration(…))`
33/// pattern scattered across ~25 repos, this function names the offending
34/// field and reports a corrupt-data read as `DbError::Serialization` — a
35/// malformed value coming back out of a read is not a schema-migration
36/// failure (QUAL-03/D-10).
37pub fn parse_uuid(s: &str, field: &str) -> Result<Uuid, DbError> {
38    s.parse::<Uuid>()
39        .map_err(|e| DbError::Serialization(format!("invalid {field} UUID: {e}")))
40}

43// classify_write_error — centralized index-violation detection (QUAL-03/D-09)

46/// Classify a write-path error (from a `CREATE`/`RELATE` statement's
47/// `.check()` or an upstream mapped error) into the correct [`DbError`]
48/// variant.
49///
50/// Reuses the exact marker-string set already proven correct against this
51/// codebase's SurrealDB version by `saml_replay.rs`, `federation_login_state.rs`,
52/// and `seeder.rs`: a genuine unique/index violation surfaces one of
53/// `"already contains"`, `"already exists"`, or `"unique"` in the error text.
54/// Those map to [`DbError::AlreadyExists`] (→ HTTP 409). Anything else —
55/// crucially including a DB outage/connection error, which contains none of
56/// these markers — falls through to [`DbError::Migration`] (→ 5xx). A DB
57/// outage must never be misclassified as a false 409.
58///
59/// Generic over `E: Display` (rather than a single concrete error type) so
60/// the same centralized detector can be called uniformly from every
61/// `.map_err` site on the write paths this phase routes through it,
62/// regardless of whether the site's error came from `Response::check()`
63/// (`surrealdb::Error`) or another fallible step in the same write path.
64///
65/// Per D-09, this is the ONLY place that inspects error text for these
66/// markers — call sites must not add their own inline `contains(...)` checks.
67pub fn classify_write_error<E: std::fmt::Display>(err: E, entity: &str) -> DbError {
68    let msg = err.to_string();
69    if is_unique_violation(&msg) {
70        DbError::AlreadyExists {
71            entity: entity.to_string(),
72        }
73    } else if is_write_conflict(&msg) {
74        // Ordered AFTER the unique check on purpose. A UNIQUE violation is a
75        // statement about the caller's request and must win: it is a 409 the
76        // client can act on, and retrying it would only fail again identically.
77        // A write conflict says nothing about the request — only that it lost a
78        // race — so it is the weaker, more retryable classification of the two.
79        DbError::Conflict(msg)
80    } else {
81        DbError::Migration(msg)
82    }
83}

85/// Run `op`, retrying while it fails with a retryable write conflict.
86///
87/// This is the helper the [`WRITE_CONFLICT_MARKERS`] documentation has linked
88/// to since the markers were introduced, and which was never actually defined:
89/// [`MAX_WRITE_ATTEMPTS`], [`write_conflict_backoff`] and [`is_write_conflict`]
90/// all shipped, tested, with exactly one hand-rolled loop assembling them (in
91/// `UserRepository::increment_failed_logins`) and no way to reuse it. Every
92/// other contended write therefore had no retry at all — which is how a
93/// concurrent SCIM `PATCH /scim/v2/Users/{id}` came to answer HTTP 500 on 2.2%
94/// of a benchmark flood, for a write the engine labelled retryable.
95///
96/// `op` is a closure rather than a future because a future is consumed by the
97/// first `.await`: a retry needs a *new* one per attempt, and SurrealDB's
98/// `bind` takes owned values, so each attempt must rebuild its own statement.
99///
100/// Generic over the error type rather than fixed to [`DbError`] because the two
101/// kinds of call site disagree about it: a repository method that returns
102/// `AxiamResult<T>` has already widened to `AxiamError` by the time it can fail,
103/// while one working inside the db layer still holds a `DbError`. Both render
104/// the engine's text, which is all the retry predicate reads, so neither has to
105/// convert at the boundary just to be retried.
106///
107/// # Safety of replaying `op`
108///
109/// Only [`is_write_conflict`] is retried, and a conflicted transaction commits
110/// NOTHING — that is what the abort means. So re-running `op` cannot
111/// double-apply, even when the statement is not idempotent (`x += 1`). Every
112/// other error returns on the first attempt, so an outage or a malformed
113/// statement fails fast instead of slowly.
114pub async fn retry_on_write_conflict<T, E, F, Fut>(mut op: F) -> Result<T, E>
115where
116    F: FnMut() -> Fut,
117    Fut: std::future::Future<Output = Result<T, E>>,
118    E: std::fmt::Display,
119{
120    let mut attempt = 1;
121    loop {
122        match op().await {
123            // Matched on the rendered message rather than on `DbError::Conflict`
124            // because a conflict can arrive as any of three variants depending
125            // on where in the write path it surfaced: `Conflict` from
126            // `classify_write_error`, `Surreal` from a bare `DbError::from`, or
127            // `Migration` from a call site that has not been routed through the
128            // classifier. Every one of them renders the engine's own text, so
129            // the message is the reliable discriminator; the variant is not.
130            Err(e) if attempt < MAX_WRITE_ATTEMPTS && is_write_conflict(&e.to_string()) => {
131                tokio::time::sleep(write_conflict_backoff(attempt)).await;
132                attempt += 1;
133            }
134            outcome => return outcome,
135        }
136    }
137}

139// ---------------------------------------------------------------------------
140// The unique-violation marker set — one definition, three classifiers (F5)
141// ---------------------------------------------------------------------------
142
143/// Substrings that identify a SurrealDB UNIQUE-index violation.
144///
145/// SurrealDB v3 reports one as ``Database index `idx_x_uniq` already contains
146/// [...]``. `"already exists"` and `"unique"` are kept as fallbacks: they
147/// covered earlier server versions and cost nothing to keep, and a conflict
148/// this set fails to recognise is not a benign miss — it is a replayed
149/// assertion accepted as fresh.
150///
151/// **The set is narrow on purpose.** Everything it does not match falls
152/// through to a 5xx, and that is the safe direction: a datastore outage
153/// contains none of these markers and must never be reported to a caller as
154/// "that already exists" or "that was a replay". Widening this set trades a
155/// clearer conflict message for the risk of answering an outage with a 409.
156const UNIQUE_VIOLATION_MARKERS: [&str; 3] = ["already contains", "already exists", "unique"];
157
158/// Whether a datastore error message reports a UNIQUE-index violation.
159///
160/// # Why this is a function and not three `contains` calls
161///
162/// [`classify_write_error`] has always documented itself as "the ONLY place
163/// that inspects error text for these markers — call sites must not add their
164/// own inline `contains(...)` checks" (D-09). Five call sites did anyway:
165/// `saml_replay`, `amqp_nonce_replay`, `oauth2_proof_replay`,
166/// `federation_login_state` and the REST bootstrap handler, each with the
167/// three markers written out again and a comment pointing at one of the
168/// others as its source of truth.
169///
170/// Nobody did that carelessly — the copies were kept deliberately identical,
171/// and `oauth2_proof_replay` says so in as many words: "kept identical so the
172/// three replay guards cannot drift into disagreeing about what a conflict
173/// looks like." That is the correct worry and the wrong mechanism. Every one
174/// of those five sites decides whether a replayed credential is refused or
175/// accepted; a marker added to four of them is a security fix with a hole in
176/// it, and no test would notice.
177///
178/// So the markers live here, once, and the five sites call one of the three
179/// classifiers below. See `claude_dev/conflict-classification.md`, and
180/// `scripts/check-conflict-markers.py`, which fails CI if a sixth inline copy
181/// appears.
182pub fn is_unique_violation(msg: &str) -> bool {
183    UNIQUE_VIOLATION_MARKERS.iter().any(|m| msg.contains(m))
184}

187// Optimistic-concurrency write conflicts — transient, and explicitly retryable
188// ---------------------------------------------------------------------------
189
190/// Substrings that identify a SurrealDB optimistic-concurrency write conflict.
191///
192/// SurrealDB is optimistic: concurrent transactions touching the same record
193/// both proceed, and the loser is aborted at commit with
194/// ``There was a problem with the key-value store: Transaction conflict:
195/// Transaction write conflict. This transaction can be retried`` (some
196/// versions phrase it ``Failed to commit transaction due to a read or write
197/// conflict``). Both phrasings are matched.
198///
199/// **This class of failure is not an error to report — it is an instruction to
200/// retry**, and the datastore says so in the message itself. It is emphatically
201/// NOT a [`is_unique_violation`]: no constraint was broken and nothing about
202/// the caller's request was wrong; the write simply lost a race and never
203/// committed. Because it commits nothing, replaying it cannot double-apply —
204/// which is what makes [`retry_on_write_conflict`] safe even for a
205/// non-idempotent statement like `failed_login_attempts += 1`.
206///
207/// # The last two markers, and why they were a hole
208///
209/// `"failed transaction"` and `"Failed to commit transaction"` were the ONLY
210/// two literals [`is_transaction_conflict`] matched, in a set of its own. That
211/// helper guards the single-use consume on four replay-sensitive paths —
212/// `device_grant`, `permission_ticket`, `pushed_auth_request` and
213/// `oauth2_auth_code` — and the message SurrealDB v3 actually emits for a
214/// contended write (`Transaction conflict: Transaction write conflict. This
215/// transaction can be retried`) contains neither of them. So the branch those
216/// four paths rely on to answer "someone else got there first" could not fire
217/// on the phrasing the deployed engine produces, and a correctly-refused
218/// replay surfaced as a 500 instead of as "no row consumed".
219///
220/// That direction is fail-CLOSED — a 500 mints no token — so it was a
221/// robustness defect rather than a security hole. But it is exactly the
222/// drift-between-copies failure D-09 and `scripts/check-conflict-markers.py`
223/// exist to prevent, and it survived because the two sets were never one set.
224/// They are one set now, and [`is_transaction_conflict`] delegates here.
225const WRITE_CONFLICT_MARKERS: [&str; 4] = [
226    "Transaction conflict",
227    "read or write conflict",
228    "failed transaction",
229    "Failed to commit transaction",
230];
231
232/// Whether a datastore error message reports a retryable write conflict.
233///
234/// Kept beside [`is_unique_violation`] and for the same reason (D-09): the
235/// decision "was this transient?" is made from free error text, so it gets one
236/// definition rather than a `contains(...)` at every hot-row write site.
237pub fn is_write_conflict(msg: &str) -> bool {
238    WRITE_CONFLICT_MARKERS.iter().any(|m| msg.contains(m))
239}
240
241/// How many attempts a contended write gets before its conflict surfaces.
242///
243/// Four is chosen against the shape of the failure rather than as a round
244/// number: a conflict means some other writer *did* commit, so the contended
245/// record is free almost immediately and the overwhelming majority of retries
246/// succeed on the first one. The extra attempts cover a burst of writers
247/// landing on the same record at once; past that, retrying is no longer
248/// masking a race but hiding sustained contention that should surface.
249///
250/// Retrying only ever applies to [`is_write_conflict`]. Every other error — an
251/// outage, a malformed statement, a constraint violation — must return on the
252/// first attempt, so a hard failure is never turned into a slow one.
253pub const MAX_WRITE_ATTEMPTS: u32 = 4;
254
255/// How long to wait before write attempt `attempt + 1` (2 ms, 4 ms, 8 ms).
256///
257/// Not there to wait out the winner — it has already committed by the time the
258/// loser learns it lost — but to stop a burst of contending writers from
259/// re-colliding in lockstep on the retry.
260pub fn write_conflict_backoff(attempt: u32) -> std::time::Duration {
261    std::time::Duration::from_millis(1u64 << attempt.min(6))
262}

264/// Classify a failed single-use `CREATE` on a replay-guard table.
265///
266/// A UNIQUE violation here is not an error condition — it is the answer. The
267/// row was already present, so the assertion / nonce / proof `jti` being
268/// inserted has been seen before, and the caller must refuse it. Anything else
269/// is a real datastore failure and propagates as [`AxiamError::Database`], so
270/// an outage surfaces as a 5xx rather than as a silent "replay detected" that
271/// would refuse legitimate traffic.
272///
273/// Used by all three replay repositories (`saml_replay`, `amqp_nonce_replay`,
274/// `oauth2_proof_replay`), which is the point: there is one definition of what
275/// a replay looks like.
276pub fn classify_replay_write_error<E: std::fmt::Display>(err: E) -> AxiamError {
277    let msg = err.to_string();
278    if is_unique_violation(&msg) {
279        AxiamError::ReplayDetected
280    } else {
281        AxiamError::Database(msg)
282    }
283}

285/// Classify a failed `CREATE` where a UNIQUE violation means "already taken".
286///
287/// The [`AxiamError`]-returning counterpart to [`classify_write_error`], for
288/// the write paths that build an `AxiamError` directly rather than a
289/// [`DbError`]. `entity` names what collided and reaches the client in the
290/// 409 body, so it should identify the constrained thing
291/// (`"federation_login_state.state"`), not merely the table.
292pub fn classify_conflict_write_error<E: std::fmt::Display>(err: E, entity: &str) -> AxiamError {
293    let msg = err.to_string();
294    if is_unique_violation(&msg) {
295        AxiamError::AlreadyExists {
296            entity: entity.to_string(),
297        }
298    } else {
299        AxiamError::Database(msg)
300    }
301}

304// cleanup_expired_rows — one sweep for every replay-guard table (F5)
305// ---------------------------------------------------------------------------
306
307/// Count and then delete every row of `table` whose `expires_at` has passed,
308/// returning how many were removed.
309///
310/// The three replay tables carried a byte-identical copy of this, differing
311/// only in the table name. It is not security-critical the way
312/// [`classify_replay_write_error`] is — a sweep that under-deletes leaves rows
313/// that only cost space, and the UNIQUE index keeps doing its job either way —
314/// but three copies of one query is three places to fix when the count and the
315/// delete need to become one statement.
316///
317/// `table` is interpolated into the query rather than bound, because SurrealDB
318/// has no bind form for a table name. It is therefore **never** caller- or
319/// request-derived: every call site passes a `&'static str` literal naming one
320/// of this crate's own tables.
321pub async fn cleanup_expired_rows<C: Connection>(
322    db: &crate::handle::DbHandle<C>,
323    table: &'static str,
324) -> AxiamResult<u64> {
325    let mut count_result = db
326        .current()
327        .query(format!(
328            "SELECT count() AS total FROM {table} \
329             WHERE expires_at < time::now() GROUP ALL"
330        ))
331        .await
332        .map_err(DbError::from)?;
333
334    let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
335    let total = count_rows.first().map(|r| r.total).unwrap_or(0);
336
337    db.current()
338        .query(format!("DELETE {table} WHERE expires_at < time::now()"))
339        .await
340        .map_err(DbError::from)?;
341
342    Ok(total)
343}

349/// Take the first element of `items` or return `DbError::NotFound`.
350///
351/// Replaces the `into_iter().next().ok_or_else(|| DbError::NotFound{…})`
352/// pattern repeated in every `get_by_id` / find method.
353/// Whether a SurrealDB error is a write-write transaction conflict.
354///
355/// # Why this exists
356///
357/// A multi-statement query (`LET $x = (UPDATE ...); SELECT ... FROM $x`) is
358/// **not** atomic in SurrealDB — each statement runs in its own transaction —
359/// so a single-use `UPDATE ... WHERE consumed = false` guard does not
360/// serialise concurrent callers on its own. Measured: eight concurrent
361/// redemptions of one row, up to four of them "winning".
362///
363/// Wrapping the statements in `BEGIN`/`COMMIT` makes a persistent datastore
364/// detect the conflict and abort every loser with this error — 54% of
365/// contended attempts on `surrealkv`, with zero double winners in 40 000
366/// (`tools/surreal-race-probe`). That abort is the first of the two layers the
367/// three single-use consume paths run since X6 (#302); the second is a nonce
368/// read back after the commit, which is why the read-back lives in a query of
369/// its own and this error can also surface on that second query.
370///
371/// For a single-use consume the abort is not a fault — it is the datastore
372/// reporting that someone else got there first, which is the answer the caller
373/// wanted. Callers translate it to "no row consumed"; propagating it would turn
374/// a correctly-refused replay into a 500.
375///
376/// Matched on the message because the driver surfaces it as an opaque error
377/// rather than a typed variant. Deliberately narrow: only a conflict is
378/// swallowed, and only into "lost the race" — every other failure propagates.
379///
380/// # Why this delegates instead of holding its own markers
381///
382/// It used to match two literals of its own — `"failed transaction"` and
383/// `"Failed to commit transaction"` — neither of which appears in the message
384/// SurrealDB v3 emits for a contended write. The four callers listed above
385/// therefore could not recognise the live phrasing, and answered a refused
386/// replay with a 500 rather than with "no row consumed". Both literals are now
387/// in [`WRITE_CONFLICT_MARKERS`] alongside the v3 phrasing, and this is a thin
388/// `&E`-taking adapter over [`is_write_conflict`] so the two can never again
389/// disagree about what a conflict looks like — the exact drift D-09 forbids.
390pub fn is_transaction_conflict<E: std::fmt::Display>(err: &E) -> bool {
391    is_write_conflict(&err.to_string())
392}

394pub fn take_first_or_not_found<T>(items: Vec<T>, entity: &str, id: &str) -> Result<T, DbError> {

419        items,
420        total,
421        offset: pagination.offset,
422        limit: pagination.limit,

430/// The SurrealQL fragment that narrows a list query to [`Pagination::search`].
431///
432/// Returns a clause to splice in after an existing `WHERE`, beginning with
433/// ` AND ` and empty when no term was given — so a caller concatenates it
434/// unconditionally and a list with no search runs exactly the query it ran
435/// before this existed.
436///
437/// # The binding is unconditional
438///
439/// `$search` is always bound (to the empty string when there is no term), even
440/// though the clause that reads it is then absent. The alternative — binding
441/// only when the clause is present — makes every call site a branch, and a
442/// caller that got the branch wrong would fail at runtime on a query variant
443/// nobody runs locally. An unused binding costs nothing.
444///
445/// # Why `CONTAINS` and not an index
446///
447/// A substring match over a tenant's rows, which is what an operator typing
448/// into a box means. It is not indexed, and the honest reason is that the
449/// alternative — SurrealDB's full-text search — tokenises, so `01a03efd` would
450/// not match `device-01_01a03efd-…` and pasting an id from a log line would
451/// silently find nothing. That is the search operators actually perform. The
452/// term is length-capped at deserialization (see `normalize_search`) so the
453/// scan is bounded.
454///
455/// Both sides are lowercased: the caller's term by
456/// [`Pagination::search_term`], the column by `string::lowercase` here.
457///
458/// # Matching the record id
459///
460/// `meta::id(id)` is always searched in addition to `fields`, so the id an
461/// operator copied out of a log line, an audit entry or a URL finds its row
462/// in the same box as a name. It is compared unlowercased — a UUID has no
463/// case — via a separate disjunct.
464pub fn search_clause(fields: &[&str]) -> String {
465    let mut disjuncts: Vec<String> = fields
466        .iter()
467        .map(|f| format!("string::lowercase({f} ?? '') CONTAINS $search"))
468        .collect();
469    disjuncts.push("meta::id(id) CONTAINS $search".to_string());
470    format!(" AND ({})", disjuncts.join(" OR "))
471}
472
473/// [`search_clause`] when `pagination` carries a term, otherwise empty.
474pub fn search_filter(pagination: &Pagination, fields: &[&str]) -> String {
475    if pagination.search_term().is_some() {
476        search_clause(fields)
477    } else {
478        String::new()
479    }
480}
481
482/// The value to bind to `$search` — lowercased, or empty when unsearched.
483pub fn search_bind(pagination: &Pagination) -> String {
484    pagination.search_term().unwrap_or_default()
485}

557// ---------------------------------------------------------------------------
558// Delete-target existence guard (SEC-104)
559// ---------------------------------------------------------------------------
560
561/// The sentinel a delete transaction `THROW`s when its target row does not
562/// exist in the caller's tenant (SEC-104).
563///
564/// Not a human sentence: it is matched, not read, and a message that reads
565/// like prose invites someone to reword it.
566pub const DELETE_TARGET_MISSING: &str = "axiam:delete_target_missing";
567
568/// The first statement of a delete transaction: abort unless the target row
569/// exists **in this tenant** (SEC-104).
570///
571/// # Why the guard rather than inspecting the DELETE's result
572///
573/// Five repositories (`group`, `role`, `resource`, `permission`,
574/// `service_account`) delete a record plus its edges inside one transaction and
575/// returned `Ok(())` without looking at whether anything matched. The
576/// `WHERE tenant_id = $tenant_id` predicate was — and is — correct, so no other
577/// tenant's data was ever touched, and because the answer was a uniform 204 for
578/// a foreign id, an unknown id and a real one, it was not an existence oracle
579/// either. What it cost is an administrator who deletes the wrong id, is told
580/// it worked, and believes the group is gone. `user` and `webhook` already got
581/// this right with `RETURN BEFORE` + an empty-row check; this makes the other
582/// five agree, in the repository, so `axiam-scim`'s hand-written pre-check can
583/// be deleted rather than recopied for every future resource type.
584///
585/// Placing the check **inside** the transaction, as its first statement, is
586/// what makes it more than a pre-check: a `THROW` aborts, so a delete against
587/// a missing record leaves the edge deletes unexecuted too, rather than
588/// stripping edges belonging to nothing and then reporting failure.
589///
590/// `table` is a literal table name from the calling repository, never caller
591/// input; `$id` and `$tenant_id` are bound parameters.
592pub fn delete_existence_guard(table: &str) -> String {
593    format!(
594        "LET $existing = (SELECT VALUE id FROM type::record('{table}', $id) \
595           WHERE tenant_id = $tenant_id); \
596         IF array::len($existing) == 0 {{ THROW '{DELETE_TARGET_MISSING}'; }};"
597    )
598}
599
600/// Turn a delete transaction's statement errors into the right [`DbError`]
601/// (SEC-104).
602///
603/// Scans **every** statement's error rather than trusting `Response::check()`,
604/// for the reason `resource::delete` already documents: a `THROW` fires on its
605/// own slot while the trailing statements report the generic "not executed due
606/// to a failed transaction", and `check()` can surface one of those instead —
607/// which would silently downgrade a precise `404` into an opaque `500`.
608///
609/// `extra` lets a repository claim a `THROW` of its own (`resource`'s
610/// child-count guard) before the generic mapping runs.
611pub fn map_delete_errors(
612    errors: std::collections::HashMap<usize, surrealdb::Error>,
613    entity: &str,
614    id: &str,
615) -> Result<(), DbError> {
616    if errors.is_empty() {
617        return Ok(());
618    }
619    let combined = errors
620        .into_values()
621        .map(|e| e.to_string())
622        .collect::<Vec<_>>()
623        .join("; ");
624    if combined.contains(DELETE_TARGET_MISSING) {
625        return Err(DbError::NotFound {
626            entity: entity.to_string(),
627            id: id.to_string(),
628        });
629    }
630    Err(DbError::Migration(combined))
631}
