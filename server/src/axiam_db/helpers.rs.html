26// parse_uuid — typed parse, no longer misusing DbError::Migration (CQ-B11)
27// ---------------------------------------------------------------------------
28
29/// Parse a UUID string coming out of SurrealDB, embedding the field name in
30/// the error message for easier debugging.
31///
32/// Unlike the inline `Uuid::parse_str(..).map_err(|e| DbError::Migration(…))`
33/// pattern scattered across ~25 repos, this function names the offending
34/// field and reports a corrupt-data read as `DbError::Serialization` — a
35/// malformed value coming back out of a read is not a schema-migration
36/// failure (QUAL-03/D-10).
37pub fn parse_uuid(s: &str, field: &str) -> Result<Uuid, DbError> {
38    s.parse::<Uuid>()
39        .map_err(|e| DbError::Serialization(format!("invalid {field} UUID: {e}")))
40}

43// classify_write_error — centralized index-violation detection (QUAL-03/D-09)
44// ---------------------------------------------------------------------------
45
46/// Classify a write-path error (from a `CREATE`/`RELATE` statement's
47/// `.check()` or an upstream mapped error) into the correct [`DbError`]
48/// variant.
49///
50/// Reuses the exact marker-string set already proven correct against this
51/// codebase's SurrealDB version by `saml_replay.rs`, `federation_login_state.rs`,
52/// and `seeder.rs`: a genuine unique/index violation surfaces one of
53/// `"already contains"`, `"already exists"`, or `"unique"` in the error text.
54/// Those map to [`DbError::AlreadyExists`] (→ HTTP 409). Anything else —
55/// crucially including a DB outage/connection error, which contains none of
56/// these markers — falls through to [`DbError::Migration`] (→ 5xx). A DB
57/// outage must never be misclassified as a false 409.
58///
59/// Generic over `E: Display` (rather than a single concrete error type) so
60/// the same centralized detector can be called uniformly from every
61/// `.map_err` site on the write paths this phase routes through it,
62/// regardless of whether the site's error came from `Response::check()`
63/// (`surrealdb::Error`) or another fallible step in the same write path.
64///
65/// Per D-09, this is the ONLY place that inspects error text for these
66/// markers — call sites must not add their own inline `contains(...)` checks.
67pub fn classify_write_error<E: std::fmt::Display>(err: E, entity: &str) -> DbError {
68    let msg = err.to_string();
69    if is_unique_violation(&msg) {
70        DbError::AlreadyExists {
71            entity: entity.to_string(),
72        }
73    } else {
74        DbError::Migration(msg)
75    }
76}

78// ---------------------------------------------------------------------------
79// The unique-violation marker set — one definition, three classifiers (F5)
80// ---------------------------------------------------------------------------
81
82/// Substrings that identify a SurrealDB UNIQUE-index violation.
83///
84/// SurrealDB v3 reports one as ``Database index `idx_x_uniq` already contains
85/// [...]``. `"already exists"` and `"unique"` are kept as fallbacks: they
86/// covered earlier server versions and cost nothing to keep, and a conflict
87/// this set fails to recognise is not a benign miss — it is a replayed
88/// assertion accepted as fresh.
89///
90/// **The set is narrow on purpose.** Everything it does not match falls
91/// through to a 5xx, and that is the safe direction: a datastore outage
92/// contains none of these markers and must never be reported to a caller as
93/// "that already exists" or "that was a replay". Widening this set trades a
94/// clearer conflict message for the risk of answering an outage with a 409.
95const UNIQUE_VIOLATION_MARKERS: [&str; 3] = ["already contains", "already exists", "unique"];
96
97/// Whether a datastore error message reports a UNIQUE-index violation.
98///
99/// # Why this is a function and not three `contains` calls
100///
101/// [`classify_write_error`] has always documented itself as "the ONLY place
102/// that inspects error text for these markers — call sites must not add their
103/// own inline `contains(...)` checks" (D-09). Five call sites did anyway:
104/// `saml_replay`, `amqp_nonce_replay`, `oauth2_proof_replay`,
105/// `federation_login_state` and the REST bootstrap handler, each with the
106/// three markers written out again and a comment pointing at one of the
107/// others as its source of truth.
108///
109/// Nobody did that carelessly — the copies were kept deliberately identical,
110/// and `oauth2_proof_replay` says so in as many words: "kept identical so the
111/// three replay guards cannot drift into disagreeing about what a conflict
112/// looks like." That is the correct worry and the wrong mechanism. Every one
113/// of those five sites decides whether a replayed credential is refused or
114/// accepted; a marker added to four of them is a security fix with a hole in
115/// it, and no test would notice.
116///
117/// So the markers live here, once, and the five sites call one of the three
118/// classifiers below. See `claude_dev/conflict-classification.md`, and
119/// `scripts/check-conflict-markers.py`, which fails CI if a sixth inline copy
120/// appears.
121pub fn is_unique_violation(msg: &str) -> bool {
122    UNIQUE_VIOLATION_MARKERS.iter().any(|m| msg.contains(m))
123}
124
125/// Classify a failed single-use `CREATE` on a replay-guard table.
126///
127/// A UNIQUE violation here is not an error condition — it is the answer. The
128/// row was already present, so the assertion / nonce / proof `jti` being
129/// inserted has been seen before, and the caller must refuse it. Anything else
130/// is a real datastore failure and propagates as [`AxiamError::Database`], so
131/// an outage surfaces as a 5xx rather than as a silent "replay detected" that
132/// would refuse legitimate traffic.
133///
134/// Used by all three replay repositories (`saml_replay`, `amqp_nonce_replay`,
135/// `oauth2_proof_replay`), which is the point: there is one definition of what
136/// a replay looks like.
137pub fn classify_replay_write_error<E: std::fmt::Display>(err: E) -> AxiamError {
138    let msg = err.to_string();
139    if is_unique_violation(&msg) {
140        AxiamError::ReplayDetected
141    } else {
142        AxiamError::Database(msg)
143    }
144}
145
146/// Classify a failed `CREATE` where a UNIQUE violation means "already taken".
147///
148/// The [`AxiamError`]-returning counterpart to [`classify_write_error`], for
149/// the write paths that build an `AxiamError` directly rather than a
150/// [`DbError`]. `entity` names what collided and reaches the client in the
151/// 409 body, so it should identify the constrained thing
152/// (`"federation_login_state.state"`), not merely the table.
153pub fn classify_conflict_write_error<E: std::fmt::Display>(err: E, entity: &str) -> AxiamError {
154    let msg = err.to_string();
155    if is_unique_violation(&msg) {
156        AxiamError::AlreadyExists {
157            entity: entity.to_string(),
158        }
159    } else {
160        AxiamError::Database(msg)
161    }
162}

165// cleanup_expired_rows — one sweep for every replay-guard table (F5)
166// ---------------------------------------------------------------------------
167
168/// Count and then delete every row of `table` whose `expires_at` has passed,
169/// returning how many were removed.
170///
171/// The three replay tables carried a byte-identical copy of this, differing
172/// only in the table name. It is not security-critical the way
173/// [`classify_replay_write_error`] is — a sweep that under-deletes leaves rows
174/// that only cost space, and the UNIQUE index keeps doing its job either way —
175/// but three copies of one query is three places to fix when the count and the
176/// delete need to become one statement.
177///
178/// `table` is interpolated into the query rather than bound, because SurrealDB
179/// has no bind form for a table name. It is therefore **never** caller- or
180/// request-derived: every call site passes a `&'static str` literal naming one
181/// of this crate's own tables.
182pub async fn cleanup_expired_rows<C: Connection>(
183    db: &crate::handle::DbHandle<C>,
184    table: &'static str,
185) -> AxiamResult<u64> {
186    let mut count_result = db
187        .current()
188        .query(format!(
189            "SELECT count() AS total FROM {table} \
190             WHERE expires_at < time::now() GROUP ALL"
191        ))
192        .await
193        .map_err(DbError::from)?;
194
195    let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
196    let total = count_rows.first().map(|r| r.total).unwrap_or(0);
197
198    db.current()
199        .query(format!("DELETE {table} WHERE expires_at < time::now()"))
200        .await
201        .map_err(DbError::from)?;
202
203    Ok(total)
204}

207// take_first_or_not_found — unified "0 rows → NotFound" helper
208// ---------------------------------------------------------------------------
209
210/// Take the first element of `items` or return `DbError::NotFound`.
211///
212/// Replaces the `into_iter().next().ok_or_else(|| DbError::NotFound{…})`
213/// pattern repeated in every `get_by_id` / find method.
214/// Whether a SurrealDB error is a write-write transaction conflict.
215///
216/// # Why this exists
217///
218/// A multi-statement query (`LET $x = (UPDATE ...); SELECT ... FROM $x`) is
219/// **not** atomic in SurrealDB — each statement runs in its own transaction —
220/// so a single-use `UPDATE ... WHERE consumed = false` guard does not
221/// serialise concurrent callers on its own. Measured: eight concurrent
222/// redemptions of one row, up to four of them "winning".
223///
224/// Wrapping the statements in `BEGIN`/`COMMIT` makes a persistent datastore
225/// detect the conflict and abort every loser with this error — 54% of
226/// contended attempts on `surrealkv`, with zero double winners in 40 000
227/// (`tools/surreal-race-probe`). That abort is the first of the two layers the
228/// three single-use consume paths run since X6 (#302); the second is a nonce
229/// read back after the commit, which is why the read-back lives in a query of
230/// its own and this error can also surface on that second query.
231///
232/// For a single-use consume the abort is not a fault — it is the datastore
233/// reporting that someone else got there first, which is the answer the caller
234/// wanted. Callers translate it to "no row consumed"; propagating it would turn
235/// a correctly-refused replay into a 500.
236///
237/// Matched on the message because the driver surfaces it as an opaque error
238/// rather than a typed variant. Deliberately narrow: only a conflict is
239/// swallowed, and only into "lost the race" — every other failure propagates.
240pub fn is_transaction_conflict<E: std::fmt::Display>(err: &E) -> bool {
241    let msg = err.to_string();
242    msg.contains("failed transaction") || msg.contains("Failed to commit transaction")
243}
244
245pub fn take_first_or_not_found<T>(items: Vec<T>, entity: &str, id: &str) -> Result<T, DbError> {
246    items.into_iter().next().ok_or_else(|| DbError::NotFound {
247        entity: entity.to_string(),
248        id: id.to_string(),
249    })
250}

253// paginate — shared PaginatedResult<T> construction (QUAL-02/CQ-B10)
254// ---------------------------------------------------------------------------
255
256/// Assemble a [`PaginatedResult<T>`] from already-fetched `items`, the
257/// `count_rows` from a `SELECT count() AS total ... GROUP ALL` query, and the
258/// `pagination` window that was applied to the data query.
259///
260/// Replaces the `let total = count_rows.first().map(|r| r.total).unwrap_or(0);
261/// Ok(PaginatedResult { items, total, offset, limit })` tail duplicated across
262/// every repo's list method (CQ-B10).
263pub fn paginate<T>(
264    items: Vec<T>,
265    count_rows: Vec<CountRow>,
266    pagination: &Pagination,
267) -> PaginatedResult<T> {
268    let total = count_rows.first().map(|r| r.total).unwrap_or(0);
269    PaginatedResult {
270        items,
271        total,
272        offset: pagination.offset,
273        limit: pagination.limit,
274    }
275}

277// ---------------------------------------------------------------------------
278// Delete-target existence guard (SEC-104)
279// ---------------------------------------------------------------------------
280
281/// The sentinel a delete transaction `THROW`s when its target row does not
282/// exist in the caller's tenant (SEC-104).
283///
284/// Not a human sentence: it is matched, not read, and a message that reads
285/// like prose invites someone to reword it.
286pub const DELETE_TARGET_MISSING: &str = "axiam:delete_target_missing";
287
288/// The first statement of a delete transaction: abort unless the target row
289/// exists **in this tenant** (SEC-104).
290///
291/// # Why the guard rather than inspecting the DELETE's result
292///
293/// Five repositories (`group`, `role`, `resource`, `permission`,
294/// `service_account`) delete a record plus its edges inside one transaction and
295/// returned `Ok(())` without looking at whether anything matched. The
296/// `WHERE tenant_id = $tenant_id` predicate was — and is — correct, so no other
297/// tenant's data was ever touched, and because the answer was a uniform 204 for
298/// a foreign id, an unknown id and a real one, it was not an existence oracle
299/// either. What it cost is an administrator who deletes the wrong id, is told
300/// it worked, and believes the group is gone. `user` and `webhook` already got
301/// this right with `RETURN BEFORE` + an empty-row check; this makes the other
302/// five agree, in the repository, so `axiam-scim`'s hand-written pre-check can
303/// be deleted rather than recopied for every future resource type.
304///
305/// Placing the check **inside** the transaction, as its first statement, is
306/// what makes it more than a pre-check: a `THROW` aborts, so a delete against
307/// a missing record leaves the edge deletes unexecuted too, rather than
308/// stripping edges belonging to nothing and then reporting failure.
309///
310/// `table` is a literal table name from the calling repository, never caller
311/// input; `$id` and `$tenant_id` are bound parameters.
312pub fn delete_existence_guard(table: &str) -> String {
313    format!(
314        "LET $existing = (SELECT VALUE id FROM type::record('{table}', $id) \
315           WHERE tenant_id = $tenant_id); \
316         IF array::len($existing) == 0 {{ THROW '{DELETE_TARGET_MISSING}'; }};"
317    )
318}
319
320/// Turn a delete transaction's statement errors into the right [`DbError`]
321/// (SEC-104).
322///
323/// Scans **every** statement's error rather than trusting `Response::check()`,
324/// for the reason `resource::delete` already documents: a `THROW` fires on its
325/// own slot while the trailing statements report the generic "not executed due
326/// to a failed transaction", and `check()` can surface one of those instead —
327/// which would silently downgrade a precise `404` into an opaque `500`.
328///
329/// `extra` lets a repository claim a `THROW` of its own (`resource`'s
330/// child-count guard) before the generic mapping runs.
331pub fn map_delete_errors(
332    errors: std::collections::HashMap<usize, surrealdb::Error>,
333    entity: &str,
334    id: &str,
335) -> Result<(), DbError> {
336    if errors.is_empty() {
337        return Ok(());
338    }
339    let combined = errors
340        .into_values()
341        .map(|e| e.to_string())
342        .collect::<Vec<_>>()
343        .join("; ");
344    if combined.contains(DELETE_TARGET_MISSING) {
345        return Err(DbError::NotFound {
346            entity: entity.to_string(),
347            id: id.to_string(),
348        });
349    }
350    Err(DbError::Migration(combined))
351}
