26// parse_uuid — typed parse, no longer misusing DbError::Migration (CQ-B11)

28
29/// Parse a UUID string coming out of SurrealDB, embedding the field name in
30/// the error message for easier debugging.
31///
32/// Unlike the inline `Uuid::parse_str(..).map_err(|e| DbError::Migration(…))`
33/// pattern scattered across ~25 repos, this function names the offending
34/// field and reports a corrupt-data read as `DbError::Serialization` — a
35/// malformed value coming back out of a read is not a schema-migration
36/// failure (QUAL-03/D-10).
37pub fn parse_uuid(s: &str, field: &str) -> Result<Uuid, DbError> {
38    s.parse::<Uuid>()
39        .map_err(|e| DbError::Serialization(format!("invalid {field} UUID: {e}")))
40}

43// classify_write_error — centralized index-violation detection (QUAL-03/D-09)
44// ---------------------------------------------------------------------------
45
46/// Classify a write-path error (from a `CREATE`/`RELATE` statement's
47/// `.check()` or an upstream mapped error) into the correct [`DbError`]
48/// variant.
49///
50/// Reuses the exact marker-string set already proven correct against this
51/// codebase's SurrealDB version by `saml_replay.rs`, `federation_login_state.rs`,
52/// and `seeder.rs`: a genuine unique/index violation surfaces one of
53/// `"already contains"`, `"already exists"`, or `"unique"` in the error text.
54/// Those map to [`DbError::AlreadyExists`] (→ HTTP 409). Anything else —
55/// crucially including a DB outage/connection error, which contains none of
56/// these markers — falls through to [`DbError::Migration`] (→ 5xx). A DB
57/// outage must never be misclassified as a false 409.
58///
59/// Generic over `E: Display` (rather than a single concrete error type) so
60/// the same centralized detector can be called uniformly from every
61/// `.map_err` site on the write paths this phase routes through it,
62/// regardless of whether the site's error came from `Response::check()`
63/// (`surrealdb::Error`) or another fallible step in the same write path.
64///
65/// Per D-09, this is the ONLY place that inspects error text for these
66/// markers — call sites must not add their own inline `contains(...)` checks.
67pub fn classify_write_error<E: std::fmt::Display>(err: E, entity: &str) -> DbError {
68    let msg = err.to_string();
69    if is_unique_violation(&msg) {
70        DbError::AlreadyExists {
71            entity: entity.to_string(),
72        }
73    } else {
74        DbError::Migration(msg)
75    }
76}

78// ---------------------------------------------------------------------------
79// The unique-violation marker set — one definition, three classifiers (F5)
80// ---------------------------------------------------------------------------
81
82/// Substrings that identify a SurrealDB UNIQUE-index violation.
83///
84/// SurrealDB v3 reports one as ``Database index `idx_x_uniq` already contains
85/// [...]``. `"already exists"` and `"unique"` are kept as fallbacks: they
86/// covered earlier server versions and cost nothing to keep, and a conflict
87/// this set fails to recognise is not a benign miss — it is a replayed
88/// assertion accepted as fresh.
89///
90/// **The set is narrow on purpose.** Everything it does not match falls
91/// through to a 5xx, and that is the safe direction: a datastore outage
92/// contains none of these markers and must never be reported to a caller as
93/// "that already exists" or "that was a replay". Widening this set trades a
94/// clearer conflict message for the risk of answering an outage with a 409.
95const UNIQUE_VIOLATION_MARKERS: [&str; 3] = ["already contains", "already exists", "unique"];
96
97/// Whether a datastore error message reports a UNIQUE-index violation.
98///
99/// # Why this is a function and not three `contains` calls
100///
101/// [`classify_write_error`] has always documented itself as "the ONLY place
102/// that inspects error text for these markers — call sites must not add their
103/// own inline `contains(...)` checks" (D-09). Five call sites did anyway:
104/// `saml_replay`, `amqp_nonce_replay`, `oauth2_proof_replay`,
105/// `federation_login_state` and the REST bootstrap handler, each with the
106/// three markers written out again and a comment pointing at one of the
107/// others as its source of truth.
108///
109/// Nobody did that carelessly — the copies were kept deliberately identical,
110/// and `oauth2_proof_replay` says so in as many words: "kept identical so the
111/// three replay guards cannot drift into disagreeing about what a conflict
112/// looks like." That is the correct worry and the wrong mechanism. Every one
113/// of those five sites decides whether a replayed credential is refused or
114/// accepted; a marker added to four of them is a security fix with a hole in
115/// it, and no test would notice.
116///
117/// So the markers live here, once, and the five sites call one of the three
118/// classifiers below. See `claude_dev/conflict-classification.md`, and
119/// `scripts/check-conflict-markers.py`, which fails CI if a sixth inline copy
120/// appears.
121pub fn is_unique_violation(msg: &str) -> bool {
122    UNIQUE_VIOLATION_MARKERS.iter().any(|m| msg.contains(m))
123}

125// ---------------------------------------------------------------------------
126// Optimistic-concurrency write conflicts — transient, and explicitly retryable
127// ---------------------------------------------------------------------------
128
129/// Substrings that identify a SurrealDB optimistic-concurrency write conflict.
130///
131/// SurrealDB is optimistic: concurrent transactions touching the same record
132/// both proceed, and the loser is aborted at commit with
133/// ``There was a problem with the key-value store: Transaction conflict:
134/// Transaction write conflict. This transaction can be retried`` (some
135/// versions phrase it ``Failed to commit transaction due to a read or write
136/// conflict``). Both phrasings are matched.
137///
138/// **This class of failure is not an error to report — it is an instruction to
139/// retry**, and the datastore says so in the message itself. It is emphatically
140/// NOT a [`is_unique_violation`]: no constraint was broken and nothing about
141/// the caller's request was wrong; the write simply lost a race and never
142/// committed. Because it commits nothing, replaying it cannot double-apply —
143/// which is what makes [`retry_on_write_conflict`] safe even for a
144/// non-idempotent statement like `failed_login_attempts += 1`.
145const WRITE_CONFLICT_MARKERS: [&str; 2] = ["Transaction conflict", "read or write conflict"];
146
147/// Whether a datastore error message reports a retryable write conflict.
148///
149/// Kept beside [`is_unique_violation`] and for the same reason (D-09): the
150/// decision "was this transient?" is made from free error text, so it gets one
151/// definition rather than a `contains(...)` at every hot-row write site.
152pub fn is_write_conflict(msg: &str) -> bool {
153    WRITE_CONFLICT_MARKERS.iter().any(|m| msg.contains(m))
154}
155
156/// How many attempts a contended write gets before its conflict surfaces.
157///
158/// Four is chosen against the shape of the failure rather than as a round
159/// number: a conflict means some other writer *did* commit, so the contended
160/// record is free almost immediately and the overwhelming majority of retries
161/// succeed on the first one. The extra attempts cover a burst of writers
162/// landing on the same record at once; past that, retrying is no longer
163/// masking a race but hiding sustained contention that should surface.
164///
165/// Retrying only ever applies to [`is_write_conflict`]. Every other error — an
166/// outage, a malformed statement, a constraint violation — must return on the
167/// first attempt, so a hard failure is never turned into a slow one.
168pub const MAX_WRITE_ATTEMPTS: u32 = 4;
169
170/// How long to wait before write attempt `attempt + 1` (2 ms, 4 ms, 8 ms).
171///
172/// Not there to wait out the winner — it has already committed by the time the
173/// loser learns it lost — but to stop a burst of contending writers from
174/// re-colliding in lockstep on the retry.
175pub fn write_conflict_backoff(attempt: u32) -> std::time::Duration {
176    std::time::Duration::from_millis(1u64 << attempt.min(6))
177}

179/// Classify a failed single-use `CREATE` on a replay-guard table.
180///
181/// A UNIQUE violation here is not an error condition — it is the answer. The
182/// row was already present, so the assertion / nonce / proof `jti` being
183/// inserted has been seen before, and the caller must refuse it. Anything else
184/// is a real datastore failure and propagates as [`AxiamError::Database`], so
185/// an outage surfaces as a 5xx rather than as a silent "replay detected" that
186/// would refuse legitimate traffic.
187///
188/// Used by all three replay repositories (`saml_replay`, `amqp_nonce_replay`,
189/// `oauth2_proof_replay`), which is the point: there is one definition of what
190/// a replay looks like.
191pub fn classify_replay_write_error<E: std::fmt::Display>(err: E) -> AxiamError {
192    let msg = err.to_string();
193    if is_unique_violation(&msg) {
194        AxiamError::ReplayDetected
195    } else {
196        AxiamError::Database(msg)
197    }
198}
199
200/// Classify a failed `CREATE` where a UNIQUE violation means "already taken".
201///
202/// The [`AxiamError`]-returning counterpart to [`classify_write_error`], for
203/// the write paths that build an `AxiamError` directly rather than a
204/// [`DbError`]. `entity` names what collided and reaches the client in the
205/// 409 body, so it should identify the constrained thing
206/// (`"federation_login_state.state"`), not merely the table.
207pub fn classify_conflict_write_error<E: std::fmt::Display>(err: E, entity: &str) -> AxiamError {
208    let msg = err.to_string();
209    if is_unique_violation(&msg) {
210        AxiamError::AlreadyExists {
211            entity: entity.to_string(),
212        }
213    } else {
214        AxiamError::Database(msg)
215    }
216}

219// cleanup_expired_rows — one sweep for every replay-guard table (F5)
220// ---------------------------------------------------------------------------
221
222/// Count and then delete every row of `table` whose `expires_at` has passed,
223/// returning how many were removed.
224///
225/// The three replay tables carried a byte-identical copy of this, differing
226/// only in the table name. It is not security-critical the way
227/// [`classify_replay_write_error`] is — a sweep that under-deletes leaves rows
228/// that only cost space, and the UNIQUE index keeps doing its job either way —
229/// but three copies of one query is three places to fix when the count and the
230/// delete need to become one statement.
231///
232/// `table` is interpolated into the query rather than bound, because SurrealDB
233/// has no bind form for a table name. It is therefore **never** caller- or
234/// request-derived: every call site passes a `&'static str` literal naming one
235/// of this crate's own tables.
236pub async fn cleanup_expired_rows<C: Connection>(
237    db: &crate::handle::DbHandle<C>,
238    table: &'static str,
239) -> AxiamResult<u64> {
240    let mut count_result = db
241        .current()
242        .query(format!(
243            "SELECT count() AS total FROM {table} \
244             WHERE expires_at < time::now() GROUP ALL"
245        ))
246        .await
247        .map_err(DbError::from)?;
248
249    let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
250    let total = count_rows.first().map(|r| r.total).unwrap_or(0);
251
252    db.current()
253        .query(format!("DELETE {table} WHERE expires_at < time::now()"))
254        .await
255        .map_err(DbError::from)?;
256
257    Ok(total)
258}

261// take_first_or_not_found — unified "0 rows → NotFound" helper
262// ---------------------------------------------------------------------------
263
264/// Take the first element of `items` or return `DbError::NotFound`.
265///
266/// Replaces the `into_iter().next().ok_or_else(|| DbError::NotFound{…})`
267/// pattern repeated in every `get_by_id` / find method.
268/// Whether a SurrealDB error is a write-write transaction conflict.
269///
270/// # Why this exists
271///
272/// A multi-statement query (`LET $x = (UPDATE ...); SELECT ... FROM $x`) is
273/// **not** atomic in SurrealDB — each statement runs in its own transaction —
274/// so a single-use `UPDATE ... WHERE consumed = false` guard does not
275/// serialise concurrent callers on its own. Measured: eight concurrent
276/// redemptions of one row, up to four of them "winning".
277///
278/// Wrapping the statements in `BEGIN`/`COMMIT` makes a persistent datastore
279/// detect the conflict and abort every loser with this error — 54% of
280/// contended attempts on `surrealkv`, with zero double winners in 40 000
281/// (`tools/surreal-race-probe`). That abort is the first of the two layers the
282/// three single-use consume paths run since X6 (#302); the second is a nonce
283/// read back after the commit, which is why the read-back lives in a query of
284/// its own and this error can also surface on that second query.
285///
286/// For a single-use consume the abort is not a fault — it is the datastore
287/// reporting that someone else got there first, which is the answer the caller
288/// wanted. Callers translate it to "no row consumed"; propagating it would turn
289/// a correctly-refused replay into a 500.
290///
291/// Matched on the message because the driver surfaces it as an opaque error
292/// rather than a typed variant. Deliberately narrow: only a conflict is
293/// swallowed, and only into "lost the race" — every other failure propagates.
294pub fn is_transaction_conflict<E: std::fmt::Display>(err: &E) -> bool {
295    let msg = err.to_string();
296    msg.contains("failed transaction") || msg.contains("Failed to commit transaction")
297}
298
299pub fn take_first_or_not_found<T>(items: Vec<T>, entity: &str, id: &str) -> Result<T, DbError> {
300    items.into_iter().next().ok_or_else(|| DbError::NotFound {
301        entity: entity.to_string(),
302        id: id.to_string(),
303    })
304}

310/// Assemble a [`PaginatedResult<T>`] from already-fetched `items`, the
311/// `count_rows` from a `SELECT count() AS total ... GROUP ALL` query, and the
312/// `pagination` window that was applied to the data query.
313///
314/// Replaces the `let total = count_rows.first().map(|r| r.total).unwrap_or(0);
315/// Ok(PaginatedResult { items, total, offset, limit })` tail duplicated across

317pub fn paginate<T>(
318    items: Vec<T>,
319    count_rows: Vec<CountRow>,
320    pagination: &Pagination,
321) -> PaginatedResult<T> {
322    let total = count_rows.first().map(|r| r.total).unwrap_or(0);
323    PaginatedResult {
324        items,
325        total,
326        offset: pagination.offset,
327        limit: pagination.limit,
328    }

332// Free-text search
333// ---------------------------------------------------------------------------
334
335/// The SurrealQL fragment that narrows a list query to [`Pagination::search`].
336///
337/// Returns a clause to splice in after an existing `WHERE`, beginning with
338/// ` AND ` and empty when no term was given — so a caller concatenates it
339/// unconditionally and a list with no search runs exactly the query it ran
340/// before this existed.
341///
342/// # The binding is unconditional
343///
344/// `$search` is always bound (to the empty string when there is no term), even
345/// though the clause that reads it is then absent. The alternative — binding
346/// only when the clause is present — makes every call site a branch, and a
347/// caller that got the branch wrong would fail at runtime on a query variant
348/// nobody runs locally. An unused binding costs nothing.
349///
350/// # Why `CONTAINS` and not an index
351///
352/// A substring match over a tenant's rows, which is what an operator typing
353/// into a box means. It is not indexed, and the honest reason is that the
354/// alternative — SurrealDB's full-text search — tokenises, so `01a03efd` would
355/// not match `device-01_01a03efd-…` and pasting an id from a log line would
356/// silently find nothing. That is the search operators actually perform. The
357/// term is length-capped at deserialization (see `normalize_search`) so the
358/// scan is bounded.
359///
360/// Both sides are lowercased: the caller's term by
361/// [`Pagination::search_term`], the column by `string::lowercase` here.
362///
363/// # Matching the record id
364///
365/// `meta::id(id)` is always searched in addition to `fields`, so the id an
366/// operator copied out of a log line, an audit entry or a URL finds its row
367/// in the same box as a name. It is compared unlowercased — a UUID has no
368/// case — via a separate disjunct.
369pub fn search_clause(fields: &[&str]) -> String {
370    let mut disjuncts: Vec<String> = fields
371        .iter()
372        .map(|f| format!("string::lowercase({f} ?? '') CONTAINS $search"))
373        .collect();
374    disjuncts.push("meta::id(id) CONTAINS $search".to_string());
375    format!(" AND ({})", disjuncts.join(" OR "))
376}
377
378/// [`search_clause`] when `pagination` carries a term, otherwise empty.
379pub fn search_filter(pagination: &Pagination, fields: &[&str]) -> String {
380    if pagination.search_term().is_some() {
381        search_clause(fields)
382    } else {
383        String::new()
384    }
385}
386
387/// The value to bind to `$search` — lowercased, or empty when unsearched.
388pub fn search_bind(pagination: &Pagination) -> String {
389    pagination.search_term().unwrap_or_default()
390}

463// Delete-target existence guard (SEC-104)
464// ---------------------------------------------------------------------------
465
466/// The sentinel a delete transaction `THROW`s when its target row does not
467/// exist in the caller's tenant (SEC-104).
468///
469/// Not a human sentence: it is matched, not read, and a message that reads
470/// like prose invites someone to reword it.
471pub const DELETE_TARGET_MISSING: &str = "axiam:delete_target_missing";
472
473/// The first statement of a delete transaction: abort unless the target row
474/// exists **in this tenant** (SEC-104).
475///
476/// # Why the guard rather than inspecting the DELETE's result
477///
478/// Five repositories (`group`, `role`, `resource`, `permission`,
479/// `service_account`) delete a record plus its edges inside one transaction and
480/// returned `Ok(())` without looking at whether anything matched. The
481/// `WHERE tenant_id = $tenant_id` predicate was — and is — correct, so no other
482/// tenant's data was ever touched, and because the answer was a uniform 204 for
483/// a foreign id, an unknown id and a real one, it was not an existence oracle
484/// either. What it cost is an administrator who deletes the wrong id, is told
485/// it worked, and believes the group is gone. `user` and `webhook` already got
486/// this right with `RETURN BEFORE` + an empty-row check; this makes the other
487/// five agree, in the repository, so `axiam-scim`'s hand-written pre-check can
488/// be deleted rather than recopied for every future resource type.
489///
490/// Placing the check **inside** the transaction, as its first statement, is
491/// what makes it more than a pre-check: a `THROW` aborts, so a delete against
492/// a missing record leaves the edge deletes unexecuted too, rather than
493/// stripping edges belonging to nothing and then reporting failure.
494///
495/// `table` is a literal table name from the calling repository, never caller
496/// input; `$id` and `$tenant_id` are bound parameters.
497pub fn delete_existence_guard(table: &str) -> String {
498    format!(
499        "LET $existing = (SELECT VALUE id FROM type::record('{table}', $id) \
500           WHERE tenant_id = $tenant_id); \
501         IF array::len($existing) == 0 {{ THROW '{DELETE_TARGET_MISSING}'; }};"
502    )
503}
504
505/// Turn a delete transaction's statement errors into the right [`DbError`]
506/// (SEC-104).
507///
508/// Scans **every** statement's error rather than trusting `Response::check()`,
509/// for the reason `resource::delete` already documents: a `THROW` fires on its
510/// own slot while the trailing statements report the generic "not executed due
511/// to a failed transaction", and `check()` can surface one of those instead —
512/// which would silently downgrade a precise `404` into an opaque `500`.
513///
514/// `extra` lets a repository claim a `THROW` of its own (`resource`'s
515/// child-count guard) before the generic mapping runs.
516pub fn map_delete_errors(
517    errors: std::collections::HashMap<usize, surrealdb::Error>,
518    entity: &str,
519    id: &str,
520) -> Result<(), DbError> {
521    if errors.is_empty() {
522        return Ok(());
523    }
524    let combined = errors
525        .into_values()
526        .map(|e| e.to_string())
527        .collect::<Vec<_>>()
528        .join("; ");
529    if combined.contains(DELETE_TARGET_MISSING) {
530        return Err(DbError::NotFound {
531            entity: entity.to_string(),
532            id: id.to_string(),
533        });
534    }
535    Err(DbError::Migration(combined))
536}
