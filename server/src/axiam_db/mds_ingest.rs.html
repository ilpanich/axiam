1//! FIDO MDS3 ingestion orchestrator (X3 wave 2) — ties `axiam-pki`'s
2//! fetch/verify pipeline to `MdsRepository` storage.
3//!
4//! `axiam-pki::mds::verify_and_parse` and `decide_ingest_outcome` are pure
5//! (wave 1, no storage access — see their module docs for why `no stored
6//! `no`` lives outside their signature). This module is the "ingestion
7//! orchestrator" the wave-1 docs call out as later-wave work: it reads the
8//! currently stored serial via [`MdsRepository::get_meta`], asks
9//! `decide_ingest_outcome` what that means, and only then calls
10//! [`MdsRepository::replace_entries`] / [`MdsRepository::touch_refreshed_at`]
11//! — exactly the order D4 step 8 requires, and never in the other order (a
12//! blind replace would defeat the rollback guard entirely).
13//!
14//! Lives in `axiam-db` rather than `axiam-pki` (spec's stated scope for wave
15//! 2, section A) — `axiam-db` gained a regular dependency on `axiam-pki` for
16//! this (see its `Cargo.toml`); the direction is acyclic since `axiam-pki`
17//! depends only on `axiam-core` in its own regular dependencies.
18//!
19//! Background-job/admin-endpoint wiring (calling these functions on a
20//! schedule or from `POST /api/v1/mds/refresh`) is `axiam-server`/REST work,
21//! out of scope for this wave.
22
23use axiam_core::error::{AxiamError, AxiamResult};
24use axiam_core::repository::MdsRepository;
25use axiam_pki::mds::{BlobIngestOutcome, MdsBlob, MdsError, decide_ingest_outcome};
26
27/// The outcome of one ingestion attempt, covering all four
28/// [`BlobIngestOutcome`] cases plus enough context to report them (D11:
29/// `mds.refreshed` / `mds.refresh_failed` audit actions).
30#[derive(Debug, Clone, PartialEq, Eq)]
31pub enum MdsIngestOutcome {
32    /// First-ever ingestion — no stored serial existed.
33    Initial { no: i64, entry_count: usize },
34    /// A newer BLOB replaced the stored entries.
35    Replaced { no: i64, entry_count: usize },
36    /// The fetched BLOB's `no` equalled the stored serial — only
37    /// `last_refreshed_at` was bumped, entries were already current.
38    NoOpRefresh { no: i64 },
39    /// The fetched BLOB's `no` was lower than the stored serial — rejected as
40    /// a rollback attempt. Nothing was written.
41    RollbackRejected { attempted_no: i64, stored_no: i64 },
42}
43
44/// Ingest an already-verified [`MdsBlob`] (D4 step 8, applied to storage).
45///
46/// Split out from the fetch/load entry points below so ingestion-outcome
47/// logic has exactly one implementation regardless of transport (network vs.
48/// air-gapped local file) and is directly testable against an in-memory
49/// `MdsRepository` without a network or filesystem seam.
50pub async fn ingest_blob<M: MdsRepository>(
51    mds_repo: &M,
52    blob: MdsBlob,
53) -> AxiamResult<MdsIngestOutcome> {
54    let stored_meta = mds_repo.get_meta().await?;
55    let stored_no = stored_meta.as_ref().map(|m| m.no);
56
57    match decide_ingest_outcome(blob.meta.no, stored_no) {
58        BlobIngestOutcome::Initial => {
59            let entry_count = blob.entries.len();
60            let no = blob.meta.no;
61            mds_repo.replace_entries(blob.entries, blob.meta).await?;
62            Ok(MdsIngestOutcome::Initial { no, entry_count })
63        }
64        BlobIngestOutcome::Replace => {
65            let entry_count = blob.entries.len();
66            let no = blob.meta.no;
67            mds_repo.replace_entries(blob.entries, blob.meta).await?;
68            Ok(MdsIngestOutcome::Replaced { no, entry_count })
69        }
70        BlobIngestOutcome::NoOpRefresh => {
71            let no = blob.meta.no;
72            mds_repo
73                .touch_refreshed_at(blob.meta.last_refreshed_at)
74                .await?;
75            Ok(MdsIngestOutcome::NoOpRefresh { no })
76        }
77        BlobIngestOutcome::RollbackRejected => {
78            let attempted_no = blob.meta.no;
79            let stored_no = stored_no.unwrap_or_default();
80            tracing::warn!(
81                action = "mds.refresh_failed",
82                reason = "rollback",
83                attempted_no,
84                stored_no,
85                "rejected an MDS BLOB whose serial is older than the stored one"
86            );
87            Ok(MdsIngestOutcome::RollbackRejected {
88                attempted_no,
89                stored_no,
90            })
91        }
92    }
93}

95fn log_fetch_failure(source: &str, err: &MdsError) {
96    tracing::error!(
97        action = "mds.refresh_failed",
98        reason = "verify_failed",
99        source,
100        error = %err,
101        "FIDO MDS3 BLOB fetch/verification failed"
102    );
103}

105/// Fetch, verify, and ingest the MDS3 BLOB from `url` (D10's network path,
106/// through `axiam-pki`'s SSRF-guarded fetch).
107pub async fn ingest_from_url<M: MdsRepository>(
108    mds_repo: &M,
109    url: &str,
110    expected_leaf_dns: &str,
111    allow_private: bool,
112) -> AxiamResult<MdsIngestOutcome> {
113    let blob = axiam_pki::mds::fetch_verified_blob(url, expected_leaf_dns, allow_private)
114        .await
115        .map_err(|e| {
116            log_fetch_failure(url, &e);
117            AxiamError::ServiceUnavailable(format!("MDS BLOB fetch/verification failed: {e}"))
118        })?;
119    ingest_blob(mds_repo, blob).await
120}

122/// Load, verify, and ingest the MDS3 BLOB from a local file — the air-gapped
123/// deployment path (`AXIAM__PKI__MDS_BLOB_PATH`, D10). No network I/O.
124pub async fn ingest_from_file<M: MdsRepository>(
125    mds_repo: &M,
126    path: &std::path::Path,
127    expected_leaf_dns: &str,
128) -> AxiamResult<MdsIngestOutcome> {
129    let blob =
130        axiam_pki::mds::load_verified_blob_from_file(path, expected_leaf_dns).map_err(|e| {
131            log_fetch_failure(&path.display().to_string(), &e);
132            AxiamError::ServiceUnavailable(format!("MDS BLOB load/verification failed: {e}"))
133        })?;
134    ingest_blob(mds_repo, blob).await
135}
