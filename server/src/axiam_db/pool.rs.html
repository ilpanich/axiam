1//! Database connection pool (F2) — N independent `Surreal<Client>` handles.
2//!
3//! See `claude_dev/db-pool-design.md` for the full design rationale. In short:
4//!
5//! * The pinned `surrealdb` 3.2.1 HTTP engine offers **no** native connection
6//!   pool, and every `Surreal::clone()` shares one `Arc<inner>` (one router
7//!   task + one `reqwest` client). So the only way to get genuine dispatch
8//!   parallelism, N independent TCP pools, and independently-renewable sessions
9//!   is N separate `Surreal::new::<Http>(..) + signin` connections — exactly
10//!   what [`DbManager::connect_handle`] builds. [`DbPool`] holds N of them.
11//! * Repositories bind a LIVE [`DbHandle`] over a pool slot rather than a
12//!   `Surreal<C>` clone taken at boot, so a reconnect-loop eviction is observed
13//!   by their next query instead of pinning them to the dead connection (see
14//!   [`DbPool::handle_for_repo`] and `handle.rs`).
15//! * Each pooled handle carries its OWN proactive re-signin + reconnect loop
16//!   (the D-04/PERF-04 machinery previously applied only to the manager's
17//!   handle), so **CQ-B48 is closed**: no request can reach an un-renewed
18//!   snapshot session, and the former restart-only ~4-week token outage is gone.
19//! * A process-wide `tokio::sync::Semaphore` optionally bounds total concurrent
20//!   in-flight DB ops (the stampede fix, analogous to B1's Argon2id gate). On
21//!   acquire-timeout the checkout returns the **existing** overload error,
22//!   [`AxiamError::ServiceUnavailable`] → HTTP 503 — reusing B1's taxonomy
23//!   exactly (see [`acquire_db_permit`]).
24//!
25//! ## Safe-rollout invariant (`pool_size = 1`, cap disabled — the default)
26//!
27//! With the default [`DbConfig`] (`pool_size = 1`, `pool_max_in_flight = 0`)
28//! the pool is observably identical to today: exactly one handle (so
29//! least-in-flight selection and round-robin are no-ops), and **no** semaphore
30//! is built — [`DbPool::checkout`] never acquires a permit and the
31//! acquire-timeout path can never fire, preserving today's unbounded DB
32//! concurrency byte-for-byte. `pool_size > 1` / a positive cap is opt-in.
33

52/// One pooled handle: an independent `Surreal<C>` connection behind the same
53/// swappable [`DbSlot`] [`DbManager`] uses (D-12, so its own reconnect loop can
54/// evict a poisoned handle), plus this handle's in-flight count
55/// (least-in-flight checkout input) and its two owned background tasks.
56struct PooledHandle<C: Connection> {
57    /// Swappable so the per-handle reconnect loop can atomically evict a
58    /// poisoned handle (D-12) — identical to `DbManager::db`. Repositories
59    /// bind a [`DbHandle`] over THIS slot, so an eviction is observed by their
60    /// very next query instead of leaving them on the dead connection.
61    db: DbSlot<C>,
62    /// In-flight count for THIS handle — the least-in-flight selection input.
63    /// Incremented by [`DbPool::checkout`], decremented when the returned
64    /// [`DbCheckout`] guard drops.
65    in_flight: Arc<AtomicUsize>,
66    /// Proactive re-signin task (D-04) for this handle. `None` for test-only
67    /// pools built over the embedded `kv-mem` engine, whose tokens never expire.
68    refresh_handle: Option<JoinHandle<()>>,
69    /// Reconnect loop (PERF-04) for this handle. `None` for `kv-mem` test pools.
70    reconnect_handle: Option<JoinHandle<()>>,
71}

73/// A pool of N independent, individually-renewable SurrealDB handles.
74///
75/// Generic over the connection type so the checkout/selection logic can be
76/// unit-tested against the embedded `kv-mem` engine (`DbPool<Db>`) where a live
77/// server would otherwise be required; production always uses the default
78/// `DbPool<Client>` (the HTTP engine) built via [`DbPool::connect`].
79pub struct DbPool<C: Connection = Client> {
80    handles: Vec<PooledHandle<C>>,
81    /// `None` when the in-flight cap is disabled (`pool_max_in_flight == 0`,
82    /// the default) — checkout then never acquires a permit, so today's
83    /// unbounded concurrency is preserved exactly.
84    semaphore: Option<Arc<Semaphore>>,
85    /// How long [`DbPool::checkout`] waits for a permit before returning the
86    /// overload error. Only consulted when `semaphore` is `Some`.
87    acquire_timeout: Duration,
88    /// Round-robin cursor for [`DbPool::handle_for_repo`] binding (spreads
89    /// repositories evenly across handles at startup).
90    rr: AtomicUsize,
91}
92
93impl DbPool<Client> {
94    /// Connect and build the pool using the fixed production
95    /// [`ROOT_TOKEN_DURATION`]. Mirrors [`DbManager::connect`].
96    pub async fn connect(config: &DbConfig) -> Result<Self, surrealdb::Error> {
97        Self::connect_with_ttl(config, ROOT_TOKEN_DURATION).await
98    }
99
100    /// Build a pool of `config.pool_size` INDEPENDENT handles (each via the
101    /// shared [`DbManager::connect_handle`] path — its own router task, its own
102    /// `reqwest` client, its own proactive re-signin + reconnect loop). The
103    /// root-token duration is extended once up-front (the DEFINE USER is
104    /// idempotent), then each handle signs in fresh and minting its own
105    /// long-lived token.
106    ///
107    /// `ttl` is a test hook (short TTLs prove per-handle session renewal without
108    /// waiting four weeks); production passes [`ROOT_TOKEN_DURATION`] via
109    /// [`connect`](Self::connect).
110    pub async fn connect_with_ttl(
111        config: &DbConfig,
112        ttl: Duration,
113    ) -> Result<Self, surrealdb::Error> {
114        // Guard against a misconfigured `pool_size = 0`: at least one handle.
115        let size = config.pool_size.max(1);
116        tracing::info!(
117            pool_size = size,
118            pool_max_in_flight = config.pool_max_in_flight,
119            "Building SurrealDB connection pool"
120        );
121
122        // Idempotent, once per process even for N handles (D-04/CORR-02).
123        DbManager::extend_root_token_duration(config, ttl).await;
124
125        let mut handles = Vec::with_capacity(size);
126        for _ in 0..size {
127            let (db, refresh, reconnect) = DbManager::connect_handle(config, ttl).await?;
128            handles.push(PooledHandle {
129                db,
130                in_flight: Arc::new(AtomicUsize::new(0)),
131                refresh_handle: Some(refresh),
132                reconnect_handle: Some(reconnect),
133            });
134        }
135
136        tracing::info!(pool_size = size, "SurrealDB connection pool ready");
137        Ok(Self::assemble(
138            handles,
139            config.pool_max_in_flight,
140            Duration::from_secs(config.pool_acquire_timeout_secs),
141        ))
142    }
143
144    /// Assemble the pool struct from already-built handles. The semaphore is
145    /// built ONLY when `max_in_flight > 0` — a `0` (default) cap means no
146    /// semaphore at all, so the acquire path is entirely bypassed.
147    fn assemble(
148        handles: Vec<PooledHandle<Client>>,
149        max_in_flight: usize,
150        acquire_timeout: Duration,
151    ) -> Self {
152        Self {
153            handles,
154            semaphore: (max_in_flight > 0).then(|| Arc::new(Semaphore::new(max_in_flight))),
155            acquire_timeout,
156            rr: AtomicUsize::new(0),
157        }
158    }
159}

161impl<C: Connection> DbPool<C> {
162    /// Number of pooled handles.
163    pub fn size(&self) -> usize {
164        self.handles.len()
165    }
166
167    /// Bind a repository to one pooled handle — the minimal-churn repo seam
168    /// (design §6, Option B). Returns a **live** [`DbHandle`] over the chosen
169    /// pool slot, picked round-robin so repositories spread evenly across
170    /// handles at startup. The repository then calls
171    /// `self.db.current().query(..)`, resolving the slot per operation.
172    ///
173    /// Returning a live reference rather than a `Surreal<C>` **clone** is what
174    /// closes the stale-handle bug: a clone is a snapshot of whatever
175    /// connection occupied the slot at boot, so when this handle's reconnect
176    /// loop evicted a poisoned connection (~7 minutes into a sustained load
177    /// run, in the observed case) every repository stayed pinned to the dead
178    /// one and 401'd permanently until the process restarted — while
179    /// [`health_check`](Self::health_check), which probes through the slot,
180    /// still reported the pool healthy. A [`DbHandle`] follows the slot, so
181    /// the swap is picked up on the next query. See `handle.rs` for the full
182    /// account.
183    ///
184    /// Binding is still round-robin at construction time, so per-repository
185    /// handle spread across the pool is unchanged.
186    pub fn handle_for_repo(&self) -> DbHandle<C> {
187        // Preserve the F1 checkout counter semantics (handles handed to repos).
188        metrics::record_handle_checkout();
189        let idx = self.rr.fetch_add(1, Ordering::Relaxed) % self.handles.len();
190        DbHandle::from_slot(Arc::clone(&self.handles[idx].db))
191    }
192
193    /// Per-operation checkout with least-in-flight handle selection and, when
194    /// the cap is enabled, a process-wide concurrency bound.
195    ///
196    /// 1. If the in-flight cap is enabled, acquire a semaphore permit within
197    ///    `acquire_timeout`; on timeout return [`AxiamError::ServiceUnavailable`]
198    ///    (HTTP 503, B1's overload taxonomy — see [`acquire_db_permit`]). When
199    ///    the cap is disabled (default) this step is skipped entirely.
200    /// 2. Pick the handle with the smallest current in-flight count (ties → the
201    ///    lowest index); this is load-aware and steers work away from a
202    ///    momentarily-congested handle. It is a no-op at `pool_size = 1`.
203    /// 3. Increment that handle's in-flight count and return a [`DbCheckout`]
204    ///    RAII guard that holds the permit and decrements the count on drop.
205    pub async fn checkout(&self) -> Result<DbCheckout<C>, AxiamError> {
206        let permit = match &self.semaphore {
207            Some(sem) => Some(acquire_db_permit(sem, self.acquire_timeout).await?),
208            None => None,
209        };
210
211        let idx = self.select_least_in_flight();
212        let slot = &self.handles[idx];
213        slot.in_flight.fetch_add(1, Ordering::Relaxed);
214        let handle = (*slot.db.load_full()).clone();
215
216        Ok(DbCheckout {
217            handle,
218            in_flight: Arc::clone(&slot.in_flight),
219            _permit: permit,
220        })
221    }
222
223    /// Index of the least-in-flight handle (ties broken by lowest index).
224    /// Reads each count `Relaxed` — an approximate hint is fine, so no lock is
225    /// taken on the hot path (design §3.1).
226    fn select_least_in_flight(&self) -> usize {
227        let mut best = 0usize;
228        let mut best_load = usize::MAX;
229        for (i, h) in self.handles.iter().enumerate() {
230            let load = h.in_flight.load(Ordering::Relaxed);
231            if load < best_load {
232                best_load = load;
233                best = i;
234            }
235        }
236        best
237    }
238
239    /// Readiness probe: query EVERY pooled handle so readiness reflects the
240    /// whole pool (an auth-expired or poisoned handle anywhere trips the gate).
241    /// Each probe is routed through the F1 [`metrics::instrument_query`] gauge.
242    /// An auth failure classifies as [`DbError::Unhealthy`] (D-05) so `/ready`
243    /// alarms rather than treating it as a transient query error.
244    pub async fn health_check(&self) -> Result<(), DbError> {
245        for slot in &self.handles {
246            let current = slot.db.load_full();
247            let result = metrics::instrument_query("health_check.pool", current.query("RETURN 1"))
248                .await
249                .map_err(DbManager::classify_query_error)?;
250            result.check().map_err(DbManager::classify_query_error)?;
251        }
252        Ok(())
253    }
254
255    /// Test-only constructor: build a pool directly over already-open handles
256    /// (e.g. embedded `kv-mem` instances) with no background renewal tasks, so
257    /// the checkout/selection/cap logic can be exercised without a live server.
258    #[cfg(test)]
259    fn from_handles(raw: Vec<Surreal<C>>, max_in_flight: usize, acquire_timeout: Duration) -> Self {
260        let handles = raw
261            .into_iter()
262            .map(|db| PooledHandle {
263                db: new_slot(db),
264                in_flight: Arc::new(AtomicUsize::new(0)),
265                refresh_handle: None,
266                reconnect_handle: None,
267            })
268            .collect();
269        Self {
270            handles,
271            semaphore: (max_in_flight > 0).then(|| Arc::new(Semaphore::new(max_in_flight))),
272            acquire_timeout,
273            rr: AtomicUsize::new(0),
274        }
275    }
276}

278impl<C: Connection> Drop for DbPool<C> {
279    fn drop(&mut self) {
280        // Explicitly stop each handle's background tasks (mirrors
281        // `DbManager::drop`) so a dropped pool never leaks detached tasks.
282        for slot in &self.handles {
283            if let Some(t) = &slot.refresh_handle {
284                t.abort();
285            }
286            if let Some(t) = &slot.reconnect_handle {
287                t.abort();
288            }
289        }
290    }
291}

293/// RAII checkout guard. `Deref`s to the chosen `Surreal<C>` so callers use it
294/// exactly like the owned handle they hold today (`checkout().query(..)`).
295/// Holds the semaphore permit (released on drop) and decrements the chosen
296/// handle's in-flight count on drop.
297pub struct DbCheckout<C: Connection = Client> {
298    /// Owned clone of the chosen pooled handle, read once out of its `RwLock`
299    /// at checkout time (a later reconnect-loop swap is observed by the NEXT
300    /// checkout, not this in-flight one).
301    handle: Surreal<C>,
302    in_flight: Arc<AtomicUsize>,
303    /// `None` when the cap is disabled; `Some` holds a permit released on drop.
304    _permit: Option<OwnedSemaphorePermit>,
305}
306
307impl<C: Connection> Deref for DbCheckout<C> {
308    type Target = Surreal<C>;
309    fn deref(&self) -> &Self::Target {
310        &self.handle
311    }
312}
313
314impl<C: Connection> Drop for DbCheckout<C> {
315    fn drop(&mut self) {
316        self.in_flight.fetch_sub(1, Ordering::Relaxed);
317        // `_permit` (if any) is released as the field drops.
318    }
319}

321/// Acquire a permit from the pool's in-flight semaphore, bounded by `timeout`.
322///
323/// Mirrors `axiam_auth::crypto_gate::acquire_hash_permit` EXACTLY so the DB
324/// pool and the Argon2id gate (B1) share ONE overload taxonomy:
325///
326/// * success → the held [`OwnedSemaphorePermit`] (drop to release),
327/// * acquire-timeout → [`AxiamError::ServiceUnavailable`] (the REST layer maps
328///   this to **HTTP 503** — a transient server-capacity condition, not a
329///   per-client rate-limit),
330/// * a closed semaphore (never happens — the pool never closes it) →
331///   [`AxiamError::Internal`].
332///
333/// `DbError` intentionally grows NO `ServiceUnavailable` variant — surfacing the
334/// existing `AxiamError` here keeps a single overload shape across B1 and F2
335/// (design §3.2).
336async fn acquire_db_permit(
337    sem: &Arc<Semaphore>,
338    timeout: Duration,
339) -> Result<OwnedSemaphorePermit, AxiamError> {
340    match tokio::time::timeout(timeout, Arc::clone(sem).acquire_owned()).await {
341        Ok(Ok(permit)) => Ok(permit),
342        Ok(Err(_closed)) => Err(AxiamError::Internal("db pool semaphore closed".into())),
343        Err(_elapsed) => Err(AxiamError::ServiceUnavailable(
344            "database is at capacity; please retry shortly".into(),
345        )),
346    }
347}

579    /// Dropping a `DbPool` must abort every pooled handle's background
580    /// refresh/reconnect tasks (mirrors `DbManager::drop`) so no detached
581    /// task is leaked. Builds handles with real spawned tasks (bypassing
582    /// `from_handles`, which always sets these to `None` for `kv-mem` pools)
583    /// to prove the `Some(..)` abort branches actually run.
584    #[tokio::test]
585    async fn dropping_pool_aborts_background_tasks_on_every_handle() {
586        use tokio::sync::Notify;
587
588        let notify_a = Arc::new(Notify::new());
589        let notify_b = Arc::new(Notify::new());
590        let (na, nb) = (Arc::clone(&notify_a), Arc::clone(&notify_b));
591
592        let refresh_a = tokio::spawn(async move {
593            na.notified().await;
594        });
595        let reconnect_a = tokio::spawn(async move {
596            std::future::pending::<()>().await;
597        });
598        let refresh_b = tokio::spawn(async move {
599            nb.notified().await;
600        });
601        let reconnect_b = tokio::spawn(async move {
602            std::future::pending::<()>().await;
603        });
604
605        let pool = DbPool {
606            handles: vec![
607                PooledHandle {
608                    db: new_slot(new_mem().await),
609                    in_flight: Arc::new(AtomicUsize::new(0)),
610                    refresh_handle: Some(refresh_a),
611                    reconnect_handle: Some(reconnect_a),
612                },
613                PooledHandle {
614                    db: new_slot(new_mem().await),
615                    in_flight: Arc::new(AtomicUsize::new(0)),
616                    refresh_handle: Some(refresh_b),
617                    reconnect_handle: Some(reconnect_b),
618                },
619            ],
620            semaphore: None,
621            acquire_timeout: Duration::from_millis(20),
622            rr: AtomicUsize::new(0),
623        };
624
625        // Dropping the pool must call `.abort()` on every handle's
626        // refresh_handle/reconnect_handle (both `Some(..)` branches here).
627        // Two of the four tasks would otherwise run forever
628        // (`pending::<()>()`); if `Drop` failed to abort them the test
629        // binary would hang at process exit waiting on the tokio runtime.
630        // Reaching this point is itself proof the abort ran.
631        drop(pool);
632        tokio::time::sleep(Duration::from_millis(20)).await;
633    }

635    /// THE regression test for the stale-handle bug (§7 of the run-4 benchmark
636    /// instructions): a repository built ONCE at boot must follow a
637    /// reconnect-loop handle swap, not stay pinned to the evicted connection.
638    ///
639    /// Before the fix, `handle_for_repo` handed out an owned `Surreal<C>`
640    /// **clone**; when the reconnect loop replaced the pooled connection, every
641    /// repository kept querying the dead one and 401'd permanently until the
642    /// process restarted — while `health_check`, which probes through the pool
643    /// slot, still reported the pool healthy.
644    ///
645    /// Two independent `kv-mem` instances stand in for the evicted and the
646    /// freshly-reconnected connection (same substitution
647    /// `poisoned_handle_evicted_per_handle_others_unaffected` makes, and for the
648    /// same reason: constructing a real `Surreal<Client>` requires a live
649    /// server). Writing through the repository AFTER the swap must land in the
650    /// new instance, and a record written before the swap must no longer be
651    /// visible — which is only true if the repository resolved the slot rather
652    /// than holding a snapshot.
653    #[tokio::test]
654    async fn repository_bound_at_boot_follows_a_later_handle_swap() {
655        use axiam_core::models::organization::CreateOrganization;
656        use axiam_core::repository::OrganizationRepository;
657
658        use crate::repository::SurrealOrganizationRepository;
659        use crate::schema::run_migrations;
660
661        async fn migrated() -> Surreal<Db> {
662            let db = new_mem().await;
663            run_migrations(&db).await.expect("run migrations");
664            db
665        }
666
667        let evicted = migrated().await;
668        let pool = DbPool::from_handles(vec![evicted], 0, Duration::from_millis(20));
669
670        // Bound ONCE, at "boot" — the repository never sees the pool again.
671        let repo = SurrealOrganizationRepository::new(pool.handle_for_repo());
672
673        repo.create(CreateOrganization {
674            name: "Before".into(),
675            slug: "before-swap".into(),
676            metadata: None,
677        })
678        .await
679        .expect("write before the swap");
680        repo.get_by_slug("before-swap")
681            .await
682            .expect("readable before the swap");
683
684        // The reconnect loop evicts the poisoned connection and installs a
685        // fresh one (`db.store(Arc::new(fresh))`, exactly as PERF-04 does).
686        pool.handles[0].db.store(Arc::new(migrated().await));
687
688        // The already-built repository must now be talking to the NEW
689        // connection: a write lands there...
690        repo.create(CreateOrganization {
691            name: "After".into(),
692            slug: "after-swap".into(),
693            metadata: None,
694        })
695        .await
696        .expect(
697            "a repository bound before the swap must keep working afterwards — \
698             this is the permanent-401 failure mode",
699        );
700        repo.get_by_slug("after-swap")
701            .await
702            .expect("post-swap write must be readable through the same repository");
703
704        // ...and the pre-swap record, which lives only in the evicted
705        // connection, is correctly no longer visible.
706        assert!(
707            repo.get_by_slug("before-swap").await.is_err(),
708            "the repository must be reading the NEW connection after the swap, \
709             not the evicted one"
710        );
711    }

713    /// `handle_for_repo` round-robins across ALL pooled handles (not just the
714    /// `pool_size = 1` case covered above) — each successive call advances
715    /// the cursor and wraps back to the first handle after a full cycle.
716    #[tokio::test]
717    async fn handle_for_repo_round_robins_across_handles() {
718        let pool = DbPool::from_handles(
719            vec![
720                new_marked("h0").await,
721                new_marked("h1").await,
722                new_marked("h2").await,
723            ],
724            0,
725            Duration::from_millis(20),
726        );
727
728        let mut seen = Vec::new();

730            let handle = pool.handle_for_repo();
731            seen.push(read_marker(&handle.current()).await.remove(0));
732        }
733        assert_eq!(
734            seen,
735            vec!["h0", "h1", "h2", "h0", "h1", "h2", "h0"],
736            "handle_for_repo must round-robin across every pooled handle"
737        );
738    }
