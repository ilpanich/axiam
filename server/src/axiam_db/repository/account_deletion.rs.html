45fn parse_status(s: &str) -> Result<AccountDeletionStatus, DbError> {
46    match s {
47        "pending" => Ok(AccountDeletionStatus::Pending),
48        "cancelled" => Ok(AccountDeletionStatus::Cancelled),
49        "completed" => Ok(AccountDeletionStatus::Completed),
50        other => Err(DbError::Migration(format!(
51            "unknown account_deletion status: {other}"
52        ))),
53    }
54}

172    /// D-14: atomically mark a user deletion-pending AND create the
173    /// `account_deletion` row (holding the `cancel_token_hash`) in ONE
174    /// transaction.
175    ///
176    /// Before this method existed, `gdpr.rs::request_account_delete` issued
177    /// two fully independent DB round-trips — `user_repo.mark_deletion_pending`
178    /// then `account_deletion_repo.create` — so a failure on the second call
179    /// (after the first already committed) left the user permanently
180    /// `deletion_pending = true` with no `account_deletion` row to hold a
181    /// cancel token: an uncancellable purge (CQ-B39 residual). Wrapping both
182    /// statements in `BEGIN`/`COMMIT` guarantees they succeed or fail
183    /// together.
184    ///
185    /// Also adds a duplicate-pending-request guard (Rule 2: without it, a
186    /// double-submit race could mint two live `account_deletion` rows — two
187    /// outstanding cancel tokens — for the same user). The guard is
188    /// evaluated *inside* the same transaction: if a pending deletion
189    /// already exists for this user, the whole transaction — including the
190    /// just-issued `UPDATE` on the user row — rolls back, so
191    /// `deletion_pending` is never left `true` without a matching
192    /// cancellable row.
193    ///
194    /// Result slots: BEGIN=0, UPDATE user=1, LET $existing=2, IF/THROW=3,
195    /// CREATE account_deletion=4, COMMIT=5. Every field of the returned
196    /// `AccountDeletion` is already known client-side (the id is generated
197    /// here, everything else is an input or a literal), so we don't
198    /// `.take()` any statement result — `.check()` alone proves the whole
199    /// transaction committed.
200    pub async fn create_with_pending_flag(
201        &self,
202        tenant_id: Uuid,
203        user_id: Uuid,
204        scheduled_purge_at: DateTime<Utc>,
205        cancel_token_hash: String,
206    ) -> AxiamResult<AccountDeletion> {
207        let id = new_id();
208        let created_at = Utc::now();
209
210        let query = "BEGIN TRANSACTION; \
211            UPDATE type::record('user', $user_id) SET \
212                deletion_pending = true, \
213                scheduled_purge_at = $purge_at, \
214                status = 'Inactive', \
215                updated_at = time::now() \
216                WHERE tenant_id = $tenant_id; \
217            LET $existing = (SELECT id FROM account_deletion \
218                WHERE tenant_id = $tenant_id AND user_id = $user_id \
219                AND status = 'pending'); \
220            IF array::len($existing) > 0 { \
221                THROW 'pending deletion request already exists'; \
222            }; \
223            CREATE type::record('account_deletion', $ad_id) SET \
224                tenant_id = $tenant_id, \
225                user_id = $user_id, \
226                cancel_token_hash = $hash, \
227                scheduled_purge_at = $purge_at, \
228                status = 'pending', \
229                created_at = $created_at; \
230            COMMIT TRANSACTION";
231
232        let result = self
233            .db
234            .current()
235            .query(query)
236            .bind(("user_id", user_id.to_string()))
237            .bind(("tenant_id", tenant_id.to_string()))
238            .bind(("purge_at", scheduled_purge_at))
239            .bind(("ad_id", id.to_string()))
240            .bind(("hash", cancel_token_hash.clone()))
241            .bind(("created_at", created_at))
242            .await
243            .map_err(DbError::from)?;
244
245        result
246            .check()
247            .map_err(|e| classify_write_error(e.to_string(), "account_deletion"))?;
248
249        // The user is `Inactive` now: a provisioned field changed.
250        if let Some(sink) = &self.provisioning_sink {
251            sink.user_changed(tenant_id, user_id).await;
252        }
253        Ok(AccountDeletion {
254            id,
255            tenant_id,
256            user_id,
257            cancel_token_hash,
258            scheduled_purge_at,
259            status: AccountDeletionStatus::Pending,
260            created_at,
261        })
262    }
263}

265// ---------------------------------------------------------------------------
266// Cancel vs. purge (item 11 of the 1.1.0 atomicity audit)
267// ---------------------------------------------------------------------------
268//
269// A cancel and the purge sweep decided on reads made before their writes, and
270// every write was unconditional: a cancel that read "pending, inside the grace
271// window" just before the deadline could land after the sweep had selected the
272// account, and the caller was told "cancelled" while the sweep erased the
273// account — or, worse, after the sweep had already revoked its credentials,
274// group memberships, roles and certificates, which a cancel does not restore.
275//
276// Now the sweep **claims** a request before its first destructive step
277// (`claim_for_purge`: `purge_claimed_at`, schema v107, written in a transaction
278// that re-checks the account is still due), and a cancel is one transaction
279// that re-checks the request is pending, unclaimed and inside its window, and
280// then clears the account's pending flag (`cancel_pending`). Both write the
281// `account_deletion` row, so the engine serialises them: a claim committed
282// first makes the cancel answer [`CancelOutcome::Erasing`]; a cancel committed
283// first makes the claim find the account no longer due, and the sweep skips
284// it. The loser of an overlap is retried on a fresh snapshot, where its
285// re-check sees the winner.
286//
287// The claim is also exclusive (#738): every replica runs the sweep, and a
288// claim was idempotent, so two sweeps that read one due account both ran its
289// erasure. Now a claim younger than the lease answers `Held` to a second
290// claimant, and the unit releases it on every exit before the anonymization.
291
292const DELETION_GONE: &str = "axiam:account_deletion_gone";
293const DELETION_CANCELLED: &str = "axiam:account_deletion_cancelled";
294const DELETION_ERASING: &str = "axiam:account_deletion_erasing";
295const DELETION_EXPIRED: &str = "axiam:account_deletion_expired";
296const PURGE_NOT_DUE: &str = "axiam:account_purge_not_due";
297const PURGE_HELD: &str = "axiam:account_purge_held";
298
299/// What a cancel that is no longer possible answers through
300/// [`AccountDeletionRepository::mark_cancelled`].
301const NOT_CANCELLABLE: &str = "the deletion request is no longer pending: it was cancelled already, or its erasure has \
302     started";
303
304/// What [`SurrealAccountDeletionRepository::claim_for_purge`] found.
305#[derive(Debug, Clone, Copy, PartialEq, Eq)]
306pub enum PurgeClaimOutcome {
307    /// The claim is this caller's, stamped with its `now`, until it erases the
308    /// account, releases the claim, or the lease runs out.
309    Claimed,
310    /// The account is no longer deletion-pending and due: its deletion was
311    /// cancelled, or another unit has erased it.
312    NotDue,
313    /// Another unit holds a claim on the account younger than the lease.
314    Held,
315}
316
317/// What [`SurrealAccountDeletionRepository::cancel_pending`] found.
318#[derive(Debug, Clone, Copy, PartialEq, Eq)]
319pub enum CancelOutcome {
320    /// The request is cancelled and the account active again.
321    Cancelled,
322    /// The request was cancelled before; nothing changed.
323    AlreadyCancelled,
324    /// The grace window has ended; nothing changed.
325    Expired,
326    /// The purge sweep has claimed the request (or completed it): the erasure
327    /// has started, and nothing changed.
328    Erasing,
329}

331impl<C: Connection> SurrealAccountDeletionRepository<C> {
332    /// Cancel the pending deletion request `id`, and re-enable its account, as
333    /// one transaction — or say why not.
334    ///
335    /// The request must still be `pending`, unclaimed by the purge sweep
336    /// (`purge_claimed_at = NONE`) and inside its grace window when the
337    /// transaction runs, and the account must still be deletion-pending; the
338    /// checks and both writes are one transaction. `NotFound` when there is no
339    /// such request in the tenant.
340    pub async fn cancel_pending(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<CancelOutcome> {
341        let outcome = retry_on_write_conflict(|| self.cancel_tx(tenant_id, id))
342            .await
343            .map_err(AxiamError::from)?;
344        if let (CancelOutcome::Cancelled, Some(sink), Some(user_id)) =
345            (outcome.0, &self.provisioning_sink, outcome.1)
346        {
347            // The account is `Active` again: a provisioned field changed.
348            sink.user_changed(tenant_id, user_id).await;
349        }
350        Ok(outcome.0)
351    }
352
353    async fn cancel_tx(
354        &self,
355        tenant_id: Uuid,
356        id: Uuid,
357    ) -> Result<(CancelOutcome, Option<Uuid>), TxFail> {
358        // Slots: BEGIN=0, LET $row=1, four IF/THROW=2..=5, UPDATE deletion=6,
359        // LET $user=7, IF/THROW=8, $user=9, COMMIT=10.
360        let query = format!(
361            "BEGIN TRANSACTION; \
362             LET $row = (SELECT status, purge_claimed_at, scheduled_purge_at, user_id \
363                 FROM type::record('account_deletion', $id) WHERE tenant_id = $tenant_id); \
364             IF array::len($row) = 0 {{ THROW '{DELETION_GONE}'; }}; \
365             IF $row[0].status = 'cancelled' {{ THROW '{DELETION_CANCELLED}'; }}; \
366             IF $row[0].status != 'pending' OR $row[0].purge_claimed_at != NONE \
367                 {{ THROW '{DELETION_ERASING}'; }}; \
368             IF $row[0].scheduled_purge_at <= time::now() {{ THROW '{DELETION_EXPIRED}'; }}; \
369             UPDATE type::record('account_deletion', $id) SET status = 'cancelled'; \
370             LET $user = (UPDATE type::record('user', $row[0].user_id) SET \
371                 deletion_pending = false, scheduled_purge_at = NONE, status = 'Active', \
372                 updated_at = time::now() \
373                 WHERE tenant_id = $tenant_id AND deletion_pending = true \
374                 RETURN VALUE meta::id(id)); \
375             IF array::len($user) = 0 {{ THROW '{DELETION_ERASING}'; }}; \
376             $user; \
377             COMMIT TRANSACTION"
378        );
379        let mut result = self
380            .db
381            .current()
382            .query(query)
383            .bind(("id", id.to_string()))
384            .bind(("tenant_id", tenant_id.to_string()))
385            .await?;
386        let errors = result.take_errors();
387        if !errors.is_empty() {
388            let combined = errors
389                .into_values()
390                .map(|e| e.to_string())
391                .collect::<Vec<_>>()
392                .join("; ");
393            if combined.contains(DELETION_GONE) {
394                return Err(TxFail::Refused(AxiamError::NotFound {
395                    entity: "account_deletion".into(),
396                    id: id.to_string(),
397                }));
398            }
399            // The refusals are matched before anything else: the statements
400            // after a THROW report "not executed due to a failed transaction",
401            // which is also a write-conflict marker.
402            let refused = if combined.contains(DELETION_CANCELLED) {
403                CancelOutcome::AlreadyCancelled
404            } else if combined.contains(DELETION_ERASING) {
405                CancelOutcome::Erasing
406            } else if combined.contains(DELETION_EXPIRED) {
407                CancelOutcome::Expired
408            } else {
409                return Err(TxFail::Db(classify_write_error(
410                    combined,
411                    "account_deletion",
412                )));
413            };
414            return Ok((refused, None));
415        }
416        let users: Vec<String> = result.take(9).map_err(DbError::from)?;
417        let user_id = users.first().map(|u| parse_uuid(u, "user")).transpose()?;
418        Ok((CancelOutcome::Cancelled, user_id))
419    }
420
421    /// Claim `user_id`'s erasure for the purge sweep, before its first
422    /// destructive step, exclusively and for `lease` (#738).
423    ///
424    /// [`PurgeClaimOutcome::NotDue`] when the account is no longer
425    /// deletion-pending and due at `now` (its deletion was cancelled since the
426    /// sweep's read), and [`PurgeClaimOutcome::Held`] when another unit holds a
427    /// claim on it younger than `lease`: the sweep skips the account either
428    /// way. Otherwise the claim is taken, stamped with `now`, and the answer is
429    /// [`PurgeClaimOutcome::Claimed`].
430    ///
431    /// The claim lives on the pending request, as `purge_claimed_at`; an
432    /// erasure without a request row is claimed by the same stamp on the user
433    /// row, in `erasure_audit_pending` — the field the anonymization overwrites
434    /// with its record marker, and the recovery reads only on an `Anonymized`
435    /// row, so a claim on an account not yet erased is never taken for an owed
436    /// record. A claim is free when its stamp is `NONE`, released (the Unix
437    /// epoch, [`Self::release_purge_claim`]) or at or before `now - lease`:
438    /// two sweeps that read one due account both reach this transaction, the
439    /// engine serialises their writes to the one row, and the second, retried
440    /// on a fresh snapshot, finds the first's claim live.
441    ///
442    /// Once claimed, a request can no longer be cancelled, released or not:
443    /// the steps before the point of no return revoke what a cancel does not
444    /// restore.
445    pub async fn claim_for_purge(
446        &self,
447        tenant_id: Uuid,
448        user_id: Uuid,
449        now: DateTime<Utc>,
450        lease: chrono::TimeDelta,
451    ) -> AxiamResult<PurgeClaimOutcome> {
452        retry_on_write_conflict(|| self.claim_tx(tenant_id, user_id, now, lease))
453            .await
454            .map_err(AxiamError::from)
455    }
456
457    async fn claim_tx(
458        &self,
459        tenant_id: Uuid,
460        user_id: Uuid,
461        now: DateTime<Utc>,
462        lease: chrono::TimeDelta,
463    ) -> Result<PurgeClaimOutcome, TxFail> {
464        // A pending request is the claim's home; the user row's stamp stands
465        // in only where there is none. A live stamp on either holds the
466        // account, so a request created after a user-row claim does not open
467        // a second way in.
468        let query = format!(
469            "BEGIN TRANSACTION; \
470             LET $due = (SELECT VALUE meta::id(id) FROM type::record('user', $user_id) \
471                 WHERE tenant_id = $tenant_id AND deletion_pending = true \
472                 AND scheduled_purge_at <= $now); \
473             IF array::len($due) = 0 {{ THROW '{PURGE_NOT_DUE}'; }}; \
474             LET $held = (SELECT VALUE meta::id(id) FROM account_deletion \
475                 WHERE tenant_id = $tenant_id AND user_id = $user_id AND status = 'pending' \
476                 AND purge_claimed_at != NONE AND purge_claimed_at > $cutoff); \
477             LET $user_held = (SELECT VALUE meta::id(id) FROM type::record('user', $user_id) \
478                 WHERE erasure_audit_pending != NONE AND erasure_audit_pending > $cutoff); \
479             IF array::len($held) > 0 OR array::len($user_held) > 0 \
480                 {{ THROW '{PURGE_HELD}'; }}; \
481             LET $taken = (UPDATE account_deletion SET purge_claimed_at = $now \
482                 WHERE tenant_id = $tenant_id AND user_id = $user_id AND status = 'pending' \
483                 RETURN VALUE meta::id(id)); \
484             IF array::len($taken) = 0 {{ \
485                 UPDATE type::record('user', $user_id) SET erasure_audit_pending = $now \
486                 WHERE tenant_id = $tenant_id RETURN NONE; \
487             }}; \
488             COMMIT TRANSACTION"
489        );
490        let mut result = self
491            .db
492            .current()
493            .query(query)
494            .bind(("user_id", user_id.to_string()))
495            .bind(("tenant_id", tenant_id.to_string()))
496            .bind(("now", now))
497            .bind(("cutoff", now - lease))
498            .await?;
499        let errors = result.take_errors();
500        if errors.is_empty() {
501            return Ok(PurgeClaimOutcome::Claimed);
502        }
503        let combined = errors
504            .into_values()
505            .map(|e| e.to_string())
506            .collect::<Vec<_>>()
507            .join("; ");
508        // The refusals first: the statements after a THROW report "not
509        // executed due to a failed transaction", a write-conflict marker too.
510        if combined.contains(PURGE_NOT_DUE) {
511            return Ok(PurgeClaimOutcome::NotDue);
512        }
513        if combined.contains(PURGE_HELD) {
514            return Ok(PurgeClaimOutcome::Held);
515        }
516        Err(TxFail::Db(classify_write_error(
517            combined,
518            "account_deletion",
519        )))
520    }
521
522    /// Give up the claim `stamp` on `user_id`'s erasure without erasing it,
523    /// so the next sweep takes the account at once rather than after the
524    /// lease (#738): the request's `purge_claimed_at` becomes the Unix epoch —
525    /// older than any lease, and still not `NONE`, so the request stays
526    /// uncancellable — and a user-row stamp is cleared.
527    ///
528    /// Only while the account is still deletion-pending: once the
529    /// anonymization — the point of no return — has cleared that flag, the
530    /// user row's `erasure_audit_pending` is the erasure's record marker, not
531    /// this claim, and is left alone; so is a claim another unit has since
532    /// taken over, which carries a different stamp.
533    ///
534    /// # Errors
535    ///
536    /// A datastore failure; the claim then lapses with its lease.
537    pub async fn release_purge_claim(
538        &self,
539        tenant_id: Uuid,
540        user_id: Uuid,
541        stamp: DateTime<Utc>,
542    ) -> AxiamResult<()> {
543        retry_on_write_conflict(|| async {
544            let mut result = self
545                .db
546                .current()
547                .query(
548                    "BEGIN TRANSACTION; \
549                     LET $due = (SELECT VALUE meta::id(id) FROM type::record('user', $user_id) \
550                         WHERE tenant_id = $tenant_id AND deletion_pending = true); \
551                     IF array::len($due) > 0 { \
552                         UPDATE account_deletion SET purge_claimed_at = $epoch \
553                         WHERE tenant_id = $tenant_id AND user_id = $user_id \
554                           AND status = 'pending' AND purge_claimed_at = $stamp RETURN NONE; \
555                         UPDATE type::record('user', $user_id) SET erasure_audit_pending = NONE \
556                         WHERE tenant_id = $tenant_id AND deletion_pending = true \
557                           AND erasure_audit_pending = $stamp RETURN NONE; \
558                     }; \
559                     COMMIT TRANSACTION;",
560                )
561                .bind(("user_id", user_id.to_string()))
562                .bind(("tenant_id", tenant_id.to_string()))
563                .bind(("stamp", stamp))
564                .bind(("epoch", DateTime::<Utc>::UNIX_EPOCH))
565                .await?;
566            let errors = result.take_errors();
567            if errors.is_empty() {
568                return Ok(());
569            }
570            let combined = errors
571                .into_values()
572                .map(|e| e.to_string())
573                .collect::<Vec<_>>()
574                .join("; ");
575            Err(TxFail::Db(classify_write_error(
576                combined,
577                "account_deletion",
578            )))
579        })
580        .await
581        .map_err(AxiamError::from)
582    }

585/// An erasure whose `gdpr.user_pseudonymized` record is owed (#689), claimed by
586/// [`SurrealAccountDeletionRepository::claim_owed_erasure_records`].
587#[derive(Debug, Clone, PartialEq, Eq)]
588pub struct OwedErasureRecord {
589    /// The erased account's tenant.
590    pub tenant_id: Uuid,
591    /// The erased account.
592    pub user_id: Uuid,
593    /// Its GDPR pseudonym, as the anonymization wrote it over the username.
594    pub pseudonym: String,
595    /// The claim this call holds on the account's marker.
596    pub claimed_at: DateTime<Utc>,
597    /// The marker's stamp as this claim read it, when it was one a unit or an
598    /// earlier sweep set (never before the anonymization): `None` when the
599    /// marker was released (the Unix epoch) or unset (an erasure cut before
600    /// schema v108). The recovery dates a missing erasure proof from it
601    /// (#737).
602    pub marked_at: Option<DateTime<Utc>>,
603}
604
605/// The erasure-record marker, `user.erasure_audit_pending` (#689, schema v108).
606///
607/// The anonymization — the erasure's point of no return — sets it in its own
608/// transaction to a stamp the erasure chose and armed its guard from before
609/// the write (`UserRepository::anonymize_user_marked`, #737); the value is
610/// the claim's stamp. The erasure writes its proof and
611/// `gdpr.user_pseudonymized`, and then clears the marker, completing the
612/// account's deletion request in the same transaction
613/// ([`Self::settle_erasure_record`]). An erasure cut in between leaves it
614/// set, and the next GDPR sweep, on any replica, claims it
615/// ([`Self::claim_owed_erasure_records`]) once it is released
616/// ([`Self::release_erasure_record`]: the Unix epoch) or older than the lease,
617/// writes what the erasure still owed — the proof unless it exists, the
618/// record unless it is already there ([`Self::erasure_recorded`]) — and
619/// settles it. Every write is conditional on the stamp read, so the erasure
620/// and a sweep, or two sweeps, never both own the marker.
621impl<C: Connection> SurrealAccountDeletionRepository<C> {
622    /// The erasure's record is written: clear `user_id`'s marker and, in the
623    /// same transaction, complete its pending deletion request — only while
624    /// the marker still carries `stamp`, the claim the caller holds. `true`
625    /// when this call cleared it; `false` when the marker was taken over (the
626    /// replica that took it settles it) or is already clear.
627    ///
628    /// # Errors
629    ///
630    /// A datastore failure; nothing changed, and the next sweep finds the
631    /// record written and settles the marker.
632    pub async fn settle_erasure_record(
633        &self,
634        tenant_id: Uuid,
635        user_id: Uuid,
636        stamp: DateTime<Utc>,
637    ) -> AxiamResult<bool> {
638        retry_on_write_conflict(|| self.settle_tx(tenant_id, user_id, stamp))
639            .await
640            .map_err(AxiamError::from)
641    }
642
643    async fn settle_tx(
644        &self,
645        tenant_id: Uuid,
646        user_id: Uuid,
647        stamp: DateTime<Utc>,
648    ) -> Result<bool, TxFail> {
649        let mut result = self
650            .db
651            .current()
652            .query(
653                "BEGIN TRANSACTION; \
654                 LET $cleared = (UPDATE type::record('user', $user_id) \
655                     SET erasure_audit_pending = NONE \
656                     WHERE tenant_id = $tenant_id AND erasure_audit_pending = $stamp \
657                     RETURN VALUE meta::id(id)); \
658                 IF array::len($cleared) > 0 { \
659                     UPDATE account_deletion SET status = 'completed' \
660                     WHERE tenant_id = $tenant_id AND user_id = $user_id \
661                       AND status = 'pending' RETURN NONE; \
662                 }; \
663                 RETURN array::len($cleared); \
664                 COMMIT TRANSACTION;",
665            )
666            .bind(("user_id", user_id.to_string()))
667            .bind(("tenant_id", tenant_id.to_string()))
668            .bind(("stamp", stamp))
669            .await?;
670        let errors = result.take_errors();
671        if !errors.is_empty() {
672            let combined = errors
673                .into_values()
674                .map(|e| e.to_string())
675                .collect::<Vec<_>>()
676                .join("; ");
677            return Err(TxFail::Db(classify_write_error(combined, "user")));
678        }
679        let cleared: Option<i64> = result.take(3)?;
680        Ok(cleared.unwrap_or(0) > 0)
681    }
682
683    /// Give up the claim `stamp` on `user_id`'s marker without recording it,
684    /// so the next sweep takes it over at once rather than after the lease: the
685    /// stamp becomes the Unix epoch, older than any lease. A marker another
686    /// replica has since taken over carries a different stamp and is left
687    /// alone.
688    ///
689    /// # Errors
690    ///
691    /// A datastore failure; the marker then lapses with its lease.
692    pub async fn release_erasure_record(
693        &self,
694        tenant_id: Uuid,
695        user_id: Uuid,
696        stamp: DateTime<Utc>,
697    ) -> AxiamResult<()> {
698        retry_hot_row(|| async {
699            self.db
700                .current()
701                .query(
702                    "UPDATE type::record('user', $user_id) SET erasure_audit_pending = $epoch \
703                     WHERE tenant_id = $tenant_id AND erasure_audit_pending = $stamp",
704                )
705                .bind(("user_id", user_id.to_string()))
706                .bind(("tenant_id", tenant_id.to_string()))
707                .bind(("stamp", stamp))
708                .bind(("epoch", DateTime::<Utc>::UNIX_EPOCH))
709                .await
710                .map_err(DbError::from)?
711                .check()
712                .map_err(|e| classify_write_error(e, "user"))?;
713            Ok::<_, DbError>(())
714        })
715        .await
716        .map_err(Into::into)
717    }
718
719    /// Claim, at `now`, every erasure whose record is owed and that nobody is
720    /// recording: an anonymized account whose marker is released or stamped at
721    /// or before `now - lease`, and (#689 (i)) an anonymized account with no
722    /// marker whose deletion request is still pending — an erasure cut before
723    /// schema v108 between its anonymization and its completion. Each claim
724    /// stamps the marker with `now`, conditionally on the value read, so of
725    /// two sweeps that read one account exactly one takes it. What is returned
726    /// is this call's to record, then to settle
727    /// ([`Self::settle_erasure_record`]) or give up
728    /// ([`Self::release_erasure_record`]).
729    ///
730    /// # Errors
731    ///
732    /// A datastore failure.
733    pub async fn claim_owed_erasure_records(
734        &self,
735        now: DateTime<Utc>,
736        lease: chrono::TimeDelta,
737    ) -> AxiamResult<Vec<OwedErasureRecord>> {
738        #[derive(Debug, SurrealValue)]
739        struct OwedRow {
740            record_id: String,
741            tenant_id: String,
742            username: String,
743            stamp: Option<DateTime<Utc>>,
744        }
745        let mut result = self
746            .db
747            .current()
748            .query(
749                // The pending requests are read once into `$pending` and their
750                // users fetched by record id: the cost follows the requests,
751                // not the users (R111W1-01 — a subquery in the `WHERE` of a
752                // scan of `user` is evaluated again for every row).
753                "LET $pending = (SELECT VALUE user_id FROM account_deletion \
754                     WHERE status = 'pending'); \
755                 SELECT meta::id(id) AS record_id, tenant_id, username, \
756                     erasure_audit_pending AS stamp \
757                 FROM user WHERE erasure_audit_pending != NONE \
758                   AND erasure_audit_pending <= $cutoff AND status = 'Anonymized'; \
759                 SELECT meta::id(id) AS record_id, tenant_id, username, \
760                     erasure_audit_pending AS stamp \
761                 FROM array::map($pending, |$u| type::record('user', $u)) \
762                 WHERE status = 'Anonymized' AND erasure_audit_pending = NONE;",
763            )
764            .bind(("cutoff", now - lease))
765            .await
766            .map_err(DbError::from)?
767            .check()
768            .map_err(DbError::from)?;
769        let mut owed: Vec<OwedRow> = result.take(1).map_err(DbError::from)?;
770        let unmarked: Vec<OwedRow> = result.take(2).map_err(DbError::from)?;
771        owed.extend(unmarked);
772
773        let mut claimed = Vec::new();
774        for row in owed {
775            let (Ok(user_id), Ok(tenant_id)) = (
776                Uuid::parse_str(&row.record_id),
777                Uuid::parse_str(&row.tenant_id),
778            ) else {
779                continue;
780            };
781            // Conditional on the marker read: NONE, or the stale stamp.
782            let condition = if row.stamp.is_some() {
783                "erasure_audit_pending = $read"
784            } else {
785                "erasure_audit_pending = NONE"
786            };
787            let statement = format!(
788                "UPDATE type::record('user', $user_id) SET erasure_audit_pending = $now \
789                 WHERE tenant_id = $tenant_id AND status = 'Anonymized' AND {condition} \
790                 RETURN VALUE meta::id(id)"
791            );
792            let won = retry_hot_row(|| async {
793                let mut result = self
794                    .db
795                    .current()
796                    .query(statement.as_str())
797                    .bind(("user_id", row.record_id.clone()))
798                    .bind(("tenant_id", row.tenant_id.clone()))
799                    .bind(("now", now))
800                    .bind(("read", row.stamp))
801                    .await
802                    .map_err(DbError::from)?
803                    .check()
804                    .map_err(|e| classify_write_error(e, "user"))?;
805                let ids: Vec<String> = result.take(0).map_err(DbError::from)?;
806                Ok::<_, DbError>(!ids.is_empty())
807            })
808            .await?;
809            if won {
810                claimed.push(OwedErasureRecord {
811                    tenant_id,
812                    user_id,
813                    pseudonym: row.username,
814                    claimed_at: now,
815                    marked_at: row
816                        .stamp
817                        .filter(|stamp| *stamp != DateTime::<Utc>::UNIX_EPOCH),
818                });
819            }
820        }
821        Ok(claimed)
822    }
823
824    /// Whether `tenant_id`'s audit trail holds `action` written for the
825    /// erasure of `user_id` — for `gdpr.user_pseudonymized`, whether the
826    /// erasure's record was written before its marker could be cleared
827    /// (#689). An account is erased once, so the first such row is the one.
828    ///
829    /// The row is written with `user_id` as its actor and `pseudonym` in its
830    /// metadata; it is matched by either. A unit that stalled past its claim's
831    /// lease and then ran its audit scrub after another unit had recorded the
832    /// erasure nils that row's `actor_id` (#737): the pseudonym, the keyed
833    /// HMAC of the tenant and the account and so the account's alone, still
834    /// names it.
835    ///
836    /// # Errors
837    ///
838    /// A datastore failure.
839    pub async fn erasure_recorded(
840        &self,
841        tenant_id: Uuid,
842        user_id: Uuid,
843        pseudonym: &str,
844        action: &str,
845    ) -> AxiamResult<bool> {
846        // Two reads, each on `idx_audit_tenant_actor` (the tenant and one
847        // actor): the first is 1.1.1's form, the account as the row's actor;
848        // the second the row a stalled unit's scrub left, whose actor is the
849        // nil UUID and which only the pseudonym still names. Written as one
850        // `actor_id = … OR metadata.pseudonym = …` the planner read the whole
851        // tenant's trail (R112W1-03: 0.8–1.1 s per owed erasure at 100 000
852        // rows, against 2 ms for the first form alone). The second read runs
853        // only when the first found nothing, and scans the nil actor's rows
854        // rather than the tenant's.
855        let tenant = tenant_id.to_string();
856        for (actor_id, pseudonym_only) in [(user_id, false), (Uuid::nil(), true)] {
857            let sql = if pseudonym_only {
858                "SELECT count() AS total FROM audit_log \
859                 WHERE tenant_id = $tenant_id AND actor_id = $actor_id \
860                   AND action = $action AND metadata.pseudonym = $pseudonym \
861                 GROUP ALL"
862            } else {
863                "SELECT count() AS total FROM audit_log \
864                 WHERE tenant_id = $tenant_id AND actor_id = $actor_id \
865                   AND action = $action \
866                 GROUP ALL"
867            };
868            let mut result = self
869                .db
870                .current()
871                .query(sql)
872                .bind(("tenant_id", tenant.clone()))
873                .bind(("actor_id", actor_id.to_string()))
874                .bind(("pseudonym", pseudonym.to_string()))
875                .bind(("action", action.to_string()))
876                .await
877                .map_err(DbError::from)?
878                .check()
879                .map_err(DbError::from)?;
880            let rows: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
881            if rows.first().is_some_and(|r| r.total > 0) {
882                return Ok(true);
883            }
884        }
885        Ok(false)
886    }
887}
