1//! SurrealDB implementation of [`AuditLogRepository`].
2//!
3//! Audit logs are append-only, and this module is where that is actually
4//! enforced. The table's `FOR update NONE` / `FOR delete NONE` permissions
5//! read like the guarantee but do not provide it: `axiam-server` connects as
6//! a SurrealDB **root** user, and table permission clauses apply to record and
7//! scope users, not to root. The real control is that only three methods here
8//! write anything other than an INSERT, each deliberate and each documented:
9//!
10//! - [`AuditLogRepository::pseudonymize_actor`] — the GDPR erasure scrub
11//!   (D-03/D-04), which anonymises the subject in place rather than deleting
12//!   the record.
13//! - [`AuditLogRepository::prune_older_than`] — retention (T-119), driven by
14//!   the background sweep on a clock and reachable from no HTTP handler.
15//!
16//! One deletion lives outside this module: the tenant purge
17//! (`super::tenant_purge`, run by the cleanup job after a tenant's deletion)
18//! deletes that tenant's rows at or before the time of the export its
19//! deletion required (T-118, R1W1-04), and the signatures over nothing else.
20//! It deletes nothing the export does not hold and updates no row.
21//!
22//! Adding a third is a security decision, not a refactor.
23

43/// Query 2b of `pseudonymize_actor` (#808): one `SELECT` over the four
44/// network-role actions, then one `UPDATE` that clears, per row, the field its
45/// action names. The `IF` per field leaves every other field as it was.
46const ERASE_GRANTEE_SQL: &str = "\
47    LET $rows = (SELECT VALUE id FROM audit_log \
48        WHERE tenant_id = $tenant_id \
49        AND action IN ['network_role.assigned', 'network_role.unassigned', \
50                       'network_role.assign_refused', 'network_role.unassign_refused'] \
51        AND ((action IN ['network_role.assigned', 'network_role.unassigned'] \
52              AND metadata.grantee.kind = 'user' AND metadata.grantee.id = $user_id) \
53             OR (action IN ['network_role.assign_refused', 'network_role.unassign_refused'] \
54                 AND (metadata.after.user = $user_id OR metadata.before.user = $user_id)))); \
55    UPDATE $rows SET \
56        metadata.grantee.id = IF action IN ['network_role.assigned', 'network_role.unassigned'] \
57            AND metadata.grantee.kind = 'user' AND metadata.grantee.id = $user_id \
58            THEN $nil_uuid ELSE metadata.grantee.id END, \
59        metadata.after.user = IF action IN ['network_role.assign_refused', \
60                'network_role.unassign_refused'] AND metadata.after.user = $user_id \
61            THEN $nil_uuid ELSE metadata.after.user END, \
62        metadata.before.user = IF action IN ['network_role.assign_refused', \
63                'network_role.unassign_refused'] AND metadata.before.user = $user_id \
64            THEN $nil_uuid ELSE metadata.before.user END";

97fn parse_actor_type(s: &str) -> Result<ActorType, DbError> {
98    match s {
99        "User" => Ok(ActorType::User),
100        "ServiceAccount" => Ok(ActorType::ServiceAccount),
101        "System" => Ok(ActorType::System),
102        other => Err(DbError::Migration(format!("unknown actor type: {other}"))),
103    }
104}
105
106fn actor_type_str(t: &ActorType) -> &'static str {
107    match t {
108        ActorType::User => "User",
109        ActorType::ServiceAccount => "ServiceAccount",
110        ActorType::System => "System",
111    }
112}

114fn parse_outcome(s: &str) -> Result<AuditOutcome, DbError> {
115    match s {
116        "Success" => Ok(AuditOutcome::Success),
117        "Failure" => Ok(AuditOutcome::Failure),
118        "Denied" => Ok(AuditOutcome::Denied),
119        other => Err(DbError::Migration(format!(
120            "unknown audit outcome: {other}"
121        ))),
122    }
123}
124
125fn outcome_str(o: &AuditOutcome) -> &'static str {
126    match o {
127        AuditOutcome::Success => "Success",
128        AuditOutcome::Failure => "Failure",
129        AuditOutcome::Denied => "Denied",

139        let tenant_id = Uuid::parse_str(&self.tenant_id)

142            .map_err(|e| DbError::Migration(format!("invalid actor UUID: {e}")))?;

145            .as_deref()
146            .map(Uuid::parse_str)
147            .transpose()
148            .map_err(|e| DbError::Migration(format!("invalid resource UUID: {e}")))?;
149
150        Ok(AuditLogEntry {
151            id,
152            tenant_id,
153            actor_id,
154            actor_type: parse_actor_type(&self.actor_type)?,
155            action: self.action,
156            resource_id,
157            outcome: parse_outcome(&self.outcome)?,
158            ip_address: self.ip_address,
159            metadata: self.metadata,
160            timestamp: self.timestamp,
161        })
162    }
163}

199/// Pre-computed bind values for filter parameters.
200enum FilterBind {
201    Str(&'static str, String),
202    DateTime(&'static str, DateTime<Utc>),
203}
204
205/// Build the WHERE clause and collected bind values from an `AuditLogFilter`.
206///
207/// Always includes `tenant_id = $tenant_id` (bound separately by the caller).
208fn build_filter_clause(filter: &AuditLogFilter) -> (String, Vec<FilterBind>) {
209    let mut conditions = vec!["tenant_id = $tenant_id".to_string()];
210    let mut binds: Vec<FilterBind> = Vec::new();
211
212    if let Some(actor_id) = &filter.actor_id {
213        conditions.push("actor_id = $actor_id".into());
214        binds.push(FilterBind::Str("actor_id", actor_id.to_string()));
215    }
216    if let Some(action) = &filter.action {
217        conditions.push("action = $action".into());
218        binds.push(FilterBind::Str("action", action.clone()));
219    }
220    if let Some(outcome) = &filter.outcome {
221        conditions.push("outcome = $outcome".into());
222        binds.push(FilterBind::Str("outcome", outcome_str(outcome).into()));
223    }
224    if let Some(resource_id) = &filter.resource_id {
225        conditions.push("resource_id = $resource_id".into());
226        binds.push(FilterBind::Str("resource_id", resource_id.to_string()));
227    }
228    if let Some(from) = &filter.from {
229        conditions.push("timestamp >= $from_ts".into());
230        binds.push(FilterBind::DateTime("from_ts", *from));
231    }
232    if let Some(to) = &filter.to {
233        conditions.push("timestamp <= $to_ts".into());
234        binds.push(FilterBind::DateTime("to_ts", *to));
235    }
236
237    (conditions.join(" AND "), binds)
238}
239
240/// Apply pre-computed filter binds to a query.
241fn apply_filter_binds<'a, C: Connection>(
242    mut query: surrealdb::method::Query<'a, C>,
243    binds: &'a [FilterBind],
244) -> surrealdb::method::Query<'a, C> {
245    for bind in binds {
246        match bind {
247            FilterBind::Str(key, val) => {
248                query = query.bind((*key, val.clone()));
249            }
250            FilterBind::DateTime(key, val) => {
251                query = query.bind((*key, *val));
252            }
253        }
254    }
255    query
256}

262/// SurrealDB implementation of the audit log repository.
263#[derive(Clone)]
264pub struct SurrealAuditLogRepository<C: Connection> {
265    db: DbHandle<C>,
266    /// T-110 — whether this deployment minimises what it collects.
267    ///
268    /// Held **here**, on the repository, and not on the audit middleware,
269    /// because the middleware is one producer among eighteen: the OAuth2
270    /// replay record, the GDPR erasure proof, the webhook consumer, the
271    /// federation secret backfill and the rest all call
272    /// [`AuditLogRepository::append`] directly. This is the only code every
273    /// audit row passes through, and "before the append-only write" has to
274    /// mean *every* write or it means nothing.
275    minimisation: AuditMinimisation,
276}
277
278impl<C: Connection> SurrealAuditLogRepository<C> {
279    /// Without minimisation — today's behaviour, and what every test and every
280    /// caller that has no opinion gets.
281    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
282        let db = db.into();
283        Self {
284            db,
285            minimisation: AuditMinimisation::default(),
286        }
287    }
288
289    /// With the deployment's minimisation policy (T-110).
290    ///
291    /// A builder rather than a second constructor parameter, so the several
292    /// dozen existing `new` call sites stay unchanged and the composition root
293    /// is the one place that opts in.
294    #[must_use]
295    pub fn with_minimisation(mut self, minimisation: AuditMinimisation) -> Self {
296        self.minimisation = minimisation;
297        self
298    }

302    async fn datastore_now(&self) -> AxiamResult<DateTime<Utc>> {
303        // The same function that stamps `audit_log.timestamp` (`DEFAULT
304        // time::now()`), read from the same server: a bound taken here and a
305        // row stamped by an append share one clock, whatever the application
306        // host's says.
307        let mut result = self
308            .db
309            .current()
310            .query("RETURN time::now()")
311            .await
312            .map_err(DbError::from)?
313            .check()
314            .map_err(|e| DbError::Migration(e.to_string()))?;
315        let now: Option<DateTime<Utc>> = result.take(0).map_err(DbError::from)?;
316        Ok(now.ok_or_else(|| DbError::Migration("RETURN time::now() returned no value".into()))?)
317    }

319    async fn append(&self, input: CreateAuditLogEntry) -> AxiamResult<AuditLogEntry> {
320        let id = new_id();
321        let id_str = id.to_string();
322
323        // T-110: the last thing before the write, because the table is
324        // append-only and there is no second chance by construction.
325        let mut input = input;
326        self.minimisation.apply(&mut input);
327
328        let metadata = input
329            .metadata
330            .unwrap_or(serde_json::Value::Object(Default::default()));
331
332        let resource_id_str = input.resource_id.map(|r| r.to_string());
333
334        let result = self
335            .db
336            .current()
337            .query(
338                "CREATE type::record('audit_log', $id) SET \
339                 tenant_id = $tenant_id, \
340                 actor_id = $actor_id, \
341                 actor_type = $actor_type, \
342                 action = $action, \
343                 resource_id = $resource_id, \
344                 outcome = $outcome, \
345                 ip_address = $ip_address, \
346                 metadata = $metadata",
347            )
348            .bind(("id", id_str.clone()))
349            .bind(("tenant_id", input.tenant_id.to_string()))
350            .bind(("actor_id", input.actor_id.to_string()))
351            .bind(("actor_type", actor_type_str(&input.actor_type).to_string()))
352            .bind(("action", input.action))
353            .bind(("resource_id", resource_id_str))
354            .bind(("outcome", outcome_str(&input.outcome).to_string()))
355            .bind(("ip_address", input.ip_address))
356            .bind(("metadata", metadata))
357            .await
358            .map_err(DbError::from)?;
359
360        let mut result = result
361            .check()
362            .map_err(|e| DbError::Migration(e.to_string()))?;
363
364        let rows: Vec<AuditLogRow> = result.take(0).map_err(DbError::from)?;
365        let row = take_first_or_not_found(rows, "audit_log", &id_str)?;
366
367        Ok(row.into_entry(id)?)
368    }

370    async fn list(
371        &self,
372        tenant_id: Uuid,
373        filter: AuditLogFilter,
374        pagination: Pagination,
375    ) -> AxiamResult<PaginatedResult<AuditLogEntry>> {
376        let tenant_id_str = tenant_id.to_string();
377
378        // Build dynamic WHERE clause.
379        let (where_clause, binds) = build_filter_clause(&filter);
380
381        // Count query.
382        let count_sql =
383            format!("SELECT count() AS total FROM audit_log WHERE {where_clause} GROUP ALL");
384        let db = self.db.current();
385        let mut count_query = db.query(&count_sql);
386        count_query = count_query.bind(("tenant_id", tenant_id_str.clone()));
387        count_query = apply_filter_binds(count_query, &binds);
388        let mut count_result = count_query.await.map_err(DbError::from)?;
389        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
390
391        // Data query.
392        let data_sql = format!(
393            "SELECT meta::id(id) AS record_id, * FROM audit_log \
394             WHERE {where_clause} \
395             ORDER BY timestamp DESC \
396             LIMIT $limit START $offset"
397        );
398        let db = self.db.current();
399        let mut data_query = db.query(&data_sql);
400        data_query = data_query.bind(("tenant_id", tenant_id_str));
401        data_query = apply_filter_binds(data_query, &binds);
402        data_query = data_query.bind(("limit", pagination.limit));
403        data_query = data_query.bind(("offset", pagination.offset));
404
405        let mut data_result = data_query.await.map_err(DbError::from)?;
406        let rows: Vec<AuditLogRowWithId> = data_result.take(0).map_err(DbError::from)?;
407
408        let items: Vec<AuditLogEntry> = rows
409            .into_iter()
410            .map(|r| r.try_into_entry())
411            .collect::<Result<_, _>>()?;
412
413        Ok(paginate(items, count_rows, &pagination))
414    }

425    async fn pseudonymize_actor(
426        &self,
427        tenant_id: Uuid,
428        user_id: Uuid,
429        pseudonym: &str,
430    ) -> AxiamResult<u64> {
431        // This is the SINGLE sanctioned non-INSERT write on audit_log (D-04).
432        // It requires the v15 schema permission: FOR update WHERE $auth.role = 'gdpr_pseudonymizer'.
433        //
434        // Full D-03 scrub:
435        // (a) actor_id → nil UUID
436        // (b) metadata.actor_pseudonym → pseudonym string (the new correlation key)
437        // (c) ip_address → NULL
438        // (d) known PII metadata keys redacted to "[redacted]"
439        // (e) resource_id → nil UUID in entries where resource_id == user_id
440        // (f) metadata.grantee.id → nil UUID in `network_role.assigned` /
441        //     `.unassigned` rows naming the user as grantee (#758 (e))
442        //
443        // NOTE: In SurrealDB in-memory engine (used by tests) the $auth context is not
444        // set, so the permission check may be bypassed. The application-layer guard
445        // (this method is the only path) is the true enforcement mechanism.
446        let nil_uuid = Uuid::nil().to_string();
447        let user_id_str = user_id.to_string();
448        let tenant_id_str = tenant_id.to_string();
449        let pii_keys = ["email", "username", "name", "display_name", "phone"];
450        let redacted_value = json!("[redacted]");
451
452        // Build a SET clause for metadata PII keys using SurrealDB object merge.
453        // We set each known PII key to "[redacted]" if present.
454        let pii_clauses: Vec<String> = pii_keys
455            .iter()
456            .map(|k| format!("metadata.{k} = IF metadata.{k} != NONE THEN '[redacted]' END"))
457            .collect();
458        let pii_set = if pii_clauses.is_empty() {
459            String::new()
460        } else {
461            format!(", {}", pii_clauses.join(", "))
462        };
463
464        // Query 1: scrub actor fields + metadata PII on all entries for this actor.
465        // `metadata.forwarded_for_claimed` is the request row's second client
466        // address (the caller's leftmost `X-Forwarded-For` entry, #752), so it
467        // goes with `ip_address` (#794).
468        // Not the erasure's own record (#784): it names the erased account as
469        // its actor on purpose, is written after the scrub, and carries only
470        // the pseudonym — a unit stalled past its claim's lease that scrubs
471        // after another unit recorded the erasure must not nil it. Matched by
472        // the pseudonym too, a keyed HMAC no other writer of the trail holds.
473        let q1_sql = format!(
474            "UPDATE audit_log SET \
475             actor_id = $nil_uuid, \
476             metadata.actor_pseudonym = $pseudonym, \
477             ip_address = NONE, \
478             metadata.forwarded_for_claimed = NONE{} \
479             WHERE tenant_id = $tenant_id AND actor_id = $user_id \
480             AND NOT (action = 'gdpr.user_pseudonymized' AND metadata.pseudonym = $pseudonym)",
481            pii_set
482        );
483        let _ = self
484            .db
485            .current()
486            .query(&q1_sql)
487            .bind(("nil_uuid", nil_uuid.clone()))
488            .bind(("pseudonym", pseudonym.to_string()))
489            .bind(("tenant_id", tenant_id_str.clone()))
490            .bind(("user_id", user_id_str.clone()))
491            .await
492            .map_err(DbError::from)?
493            .check()
494            .map_err(|e| DbError::Migration(e.to_string()))?;
495
496        // Query 2: null-out resource_id in entries where someone acted ON this user.
497        let _ = self
498            .db
499            .current()
500            .query(
501                "UPDATE audit_log SET resource_id = $nil_uuid \
502                 WHERE tenant_id = $tenant_id AND resource_id = $user_id",
503            )
504            .bind(("nil_uuid", nil_uuid.clone()))
505            .bind(("tenant_id", tenant_id_str.clone()))
506            .bind(("user_id", user_id_str.clone()))
507            .await
508            .map_err(DbError::from)?
509            .check()
510            .map_err(|e| DbError::Migration(e.to_string()))?;
511
512        // Query 2b: the grantee of a `network_role.assigned` / `.unassigned`
513        // row (and of a refused one, below) is the subject, whichever actor wrote it (an administrator, or the
514        // erasure itself). Cleared to the nil UUID like the other identifiers;
515        // idempotent, since the match is on the user's id, which a cleared row
516        // no longer carries.
517        //
518        // No index serves the grantee, so the rows are chosen first by the
519        // cheap predicates (a plain `SELECT VALUE id` that tests the action
520        // before it reaches into `metadata`) and the `UPDATE` then writes only
521        // the chosen ids; the same predicates in the `UPDATE`'s own `WHERE`
522        // took 4.3 s per erasure at 100 000 rows in the tenant (R112W1-03).
523        //
524        // The gate's refusal rows (`network_role.assign_refused` /
525        // `.unassign_refused`) name a user grantee by the `(kind, id)` pair of
526        // the refused write: `metadata.after.user` for an assignment, and
527        // `metadata.before.user` for an unassignment (the gate's and the
528        // stale-claim refusal alike). That id is cleared the same way (#802,
529        // R112-09 (a)). A service account's or a group's key is not `user`,
530        // so it is not matched.
531        //
532        // One `SELECT` reads the trail for all four actions (#808, R112-15:
533        // three of them read the tenant's whole prefix three times), and one
534        // `UPDATE` clears each field only where its own action and predicate
535        // hold, so a row is changed exactly as the three statements did.
536        let _ = self
537            .db
538            .current()
539            .query(ERASE_GRANTEE_SQL)
540            .bind(("nil_uuid", nil_uuid.clone()))
541            .bind(("tenant_id", tenant_id_str.clone()))
542            .bind(("user_id", user_id_str.clone()))
543            .await
544            .map_err(DbError::from)?
545            .check()
546            .map_err(|e| DbError::Migration(e.to_string()))?;
547
548        // Query 3: count entries that now have nil actor_id (as a proxy for rows updated).
549        let mut count_result = self
550            .db
551            .current()
552            .query(
553                "SELECT count() AS total FROM audit_log \
554                 WHERE tenant_id = $tenant_id AND actor_id = $nil_uuid GROUP ALL",
555            )
556            .bind(("tenant_id", tenant_id_str))
557            .bind(("nil_uuid", nil_uuid))
558            .await
559            .map_err(DbError::from)?
560            .check()
561            .map_err(|e| DbError::Migration(e.to_string()))?;
562
563        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
564        let _ = redacted_value; // used conceptually in the pii_clauses above
565        Ok(count_rows.first().map(|r| r.total).unwrap_or(0))
566    }

568    async fn prune_older_than(&self, cutoff: DateTime<Utc>) -> AxiamResult<u64> {
569        // T-119. Counted BEFORE deleting: SurrealDB's DELETE returns an empty
570        // result set rather than a row count, and counting afterwards would
571        // report zero no matter how much was removed.
572        let mut count_result = self
573            .db
574            .current()
575            .query("SELECT count() AS total FROM audit_log WHERE timestamp < $cutoff GROUP ALL")
576            .bind(("cutoff", cutoff))
577            .await
578            .map_err(DbError::from)?
579            .check()
580            .map_err(|e| DbError::Migration(e.to_string()))?;
581        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
582        let doomed = count_rows.first().map(|r| r.total).unwrap_or(0);
583        if doomed == 0 {
584            return Ok(0);
585        }
586
587        self.db
588            .current()
589            .query("DELETE audit_log WHERE timestamp < $cutoff")
590            .bind(("cutoff", cutoff))
591            .await
592            .map_err(DbError::from)?
593            .check()
594            .map_err(|e| DbError::Migration(e.to_string()))?;
595
596        // `audit_signature` is currently defined in the schema but never
597        // written by any code, so today this clause deletes nothing. It is
598        // here so that retention already covers signatures on the day signing
599        // is implemented, rather than silently leaving rows that reference
600        // entries this method has just removed. `signed_at < cutoff` is the
601        // conservative boundary: a signature is written no earlier than the
602        // entries it covers, so anything signed before the cutoff can only
603        // cover entries that are themselves past it.
604        self.db
605            .current()
606            .query("DELETE audit_signature WHERE signed_at < $cutoff")
607            .bind(("cutoff", cutoff))
608            .await
609            .map_err(DbError::from)?
610            .check()
611            .map_err(|e| DbError::Migration(e.to_string()))?;
612
613        Ok(doomed)
614    }

616    async fn get_by_ids(&self, tenant_id: Uuid, ids: &[Uuid]) -> AxiamResult<Vec<AuditLogEntry>> {
617        let id_strings: Vec<String> = ids.iter().map(|id| id.to_string()).collect();
618        let r = self
619            .db
620            .current()
621            .query(
622                "SELECT meta::id(id) AS record_id, * FROM audit_log \
623                 WHERE tenant_id = $tenant_id \
624                 AND meta::id(id) IN $ids",
625            )
626            .bind(("tenant_id", tenant_id.to_string()))
627            .bind(("ids", id_strings))
628            .await
629            .map_err(DbError::from)?;
630        let mut result = r.check().map_err(|e| DbError::Migration(e.to_string()))?;
631        let rows: Vec<AuditLogRowWithId> = result.take(0).map_err(DbError::from)?;
632        let entries: Vec<AuditLogEntry> = rows
633            .into_iter()
634            .map(|r| r.try_into_entry().map_err(Into::into))
635            .collect::<AxiamResult<Vec<_>>>()?;
636
637        // Preserve caller-provided order for deterministic signing
638        let index: std::collections::HashMap<Uuid, AuditLogEntry> =
639            entries.into_iter().map(|e| (e.id, e)).collect();
640        Ok(ids.iter().filter_map(|id| index.get(id).cloned()).collect())
641    }
