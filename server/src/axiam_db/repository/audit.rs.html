1//! SurrealDB implementation of [`AuditLogRepository`].
2//!
3//! Audit logs are append-only, and this module is where that is actually
4//! enforced. The table's `FOR update NONE` / `FOR delete NONE` permissions
5//! read like the guarantee but do not provide it: `axiam-server` connects as
6//! a SurrealDB **root** user, and table permission clauses apply to record and
7//! scope users, not to root. The real control is that only three methods here
8//! write anything other than an INSERT, each deliberate and each documented:
9//!
10//! - [`AuditLogRepository::pseudonymize_actor`] — the GDPR erasure scrub
11//!   (D-03/D-04), which anonymises the subject in place rather than deleting
12//!   the record.
13//! - [`AuditLogRepository::prune_older_than`] — retention (T-119), driven by
14//!   the background sweep on a clock and reachable from no HTTP handler.
15//!
16//! Adding a third is a security decision, not a refactor.
17

68fn parse_actor_type(s: &str) -> Result<ActorType, DbError> {
69    match s {
70        "User" => Ok(ActorType::User),
71        "ServiceAccount" => Ok(ActorType::ServiceAccount),
72        "System" => Ok(ActorType::System),
73        other => Err(DbError::Migration(format!("unknown actor type: {other}"))),
74    }
75}

78    match t {
79        ActorType::User => "User",
80        ActorType::ServiceAccount => "ServiceAccount",
81        ActorType::System => "System",
82    }

116            .as_deref()
117            .map(Uuid::parse_str)
118            .transpose()
119            .map_err(|e| DbError::Migration(format!("invalid resource UUID: {e}")))?;

121        Ok(AuditLogEntry {
122            id,
123            tenant_id,
124            actor_id,

126            action: self.action,
127            resource_id,
128            outcome: parse_outcome(&self.outcome)?,
129            ip_address: self.ip_address,
130            metadata: self.metadata,
131            timestamp: self.timestamp,
132        })
133    }

170/// Pre-computed bind values for filter parameters.
171enum FilterBind {
172    Str(&'static str, String),
173    DateTime(&'static str, DateTime<Utc>),
174}
175
176/// Build the WHERE clause and collected bind values from an `AuditLogFilter`.
177///
178/// Always includes `tenant_id = $tenant_id` (bound separately by the caller).
179fn build_filter_clause(filter: &AuditLogFilter) -> (String, Vec<FilterBind>) {
180    let mut conditions = vec!["tenant_id = $tenant_id".to_string()];
181    let mut binds: Vec<FilterBind> = Vec::new();
182
183    if let Some(actor_id) = &filter.actor_id {
184        conditions.push("actor_id = $actor_id".into());
185        binds.push(FilterBind::Str("actor_id", actor_id.to_string()));
186    }
187    if let Some(action) = &filter.action {
188        conditions.push("action = $action".into());
189        binds.push(FilterBind::Str("action", action.clone()));
190    }
191    if let Some(outcome) = &filter.outcome {
192        conditions.push("outcome = $outcome".into());
193        binds.push(FilterBind::Str("outcome", outcome_str(outcome).into()));
194    }
195    if let Some(resource_id) = &filter.resource_id {
196        conditions.push("resource_id = $resource_id".into());
197        binds.push(FilterBind::Str("resource_id", resource_id.to_string()));
198    }
199    if let Some(from) = &filter.from {
200        conditions.push("timestamp >= $from_ts".into());
201        binds.push(FilterBind::DateTime("from_ts", *from));
202    }
203    if let Some(to) = &filter.to {
204        conditions.push("timestamp <= $to_ts".into());
205        binds.push(FilterBind::DateTime("to_ts", *to));
206    }
207
208    (conditions.join(" AND "), binds)
209}
210
211/// Apply pre-computed filter binds to a query.
212fn apply_filter_binds<'a, C: Connection>(
213    mut query: surrealdb::method::Query<'a, C>,
214    binds: &'a [FilterBind],
215) -> surrealdb::method::Query<'a, C> {
216    for bind in binds {
217        match bind {
218            FilterBind::Str(key, val) => {
219                query = query.bind((*key, val.clone()));
220            }
221            FilterBind::DateTime(key, val) => {
222                query = query.bind((*key, *val));
223            }
224        }
225    }
226    query
227}

233/// SurrealDB implementation of the audit log repository.
234#[derive(Clone)]
235pub struct SurrealAuditLogRepository<C: Connection> {
236    db: DbHandle<C>,
237    /// T-110 — whether this deployment minimises what it collects.
238    ///
239    /// Held **here**, on the repository, and not on the audit middleware,
240    /// because the middleware is one producer among eighteen: the OAuth2
241    /// replay record, the GDPR erasure proof, the webhook consumer, the
242    /// federation secret backfill and the rest all call
243    /// [`AuditLogRepository::append`] directly. This is the only code every
244    /// audit row passes through, and "before the append-only write" has to
245    /// mean *every* write or it means nothing.
246    minimisation: AuditMinimisation,
247}
248
249impl<C: Connection> SurrealAuditLogRepository<C> {
250    /// Without minimisation — today's behaviour, and what every test and every
251    /// caller that has no opinion gets.
252    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
253        let db = db.into();
254        Self {
255            db,
256            minimisation: AuditMinimisation::default(),
257        }
258    }
259
260    /// With the deployment's minimisation policy (T-110).
261    ///
262    /// A builder rather than a second constructor parameter, so the several
263    /// dozen existing `new` call sites stay unchanged and the composition root
264    /// is the one place that opts in.
265    #[must_use]
266    pub fn with_minimisation(mut self, minimisation: AuditMinimisation) -> Self {
267        self.minimisation = minimisation;
268        self
269    }
270}

272impl<C: Connection> AuditLogRepository for SurrealAuditLogRepository<C> {
273    async fn append(&self, input: CreateAuditLogEntry) -> AxiamResult<AuditLogEntry> {
274        let id = new_id();
275        let id_str = id.to_string();
276
277        // T-110: the last thing before the write, because the table is
278        // append-only and there is no second chance by construction.
279        let mut input = input;
280        self.minimisation.apply(&mut input);
281
282        let metadata = input
283            .metadata
284            .unwrap_or(serde_json::Value::Object(Default::default()));
285
286        let resource_id_str = input.resource_id.map(|r| r.to_string());
287
288        let result = self
289            .db
290            .current()
291            .query(
292                "CREATE type::record('audit_log', $id) SET \
293                 tenant_id = $tenant_id, \
294                 actor_id = $actor_id, \
295                 actor_type = $actor_type, \
296                 action = $action, \
297                 resource_id = $resource_id, \
298                 outcome = $outcome, \
299                 ip_address = $ip_address, \
300                 metadata = $metadata",
301            )
302            .bind(("id", id_str.clone()))
303            .bind(("tenant_id", input.tenant_id.to_string()))
304            .bind(("actor_id", input.actor_id.to_string()))
305            .bind(("actor_type", actor_type_str(&input.actor_type).to_string()))
306            .bind(("action", input.action))
307            .bind(("resource_id", resource_id_str))
308            .bind(("outcome", outcome_str(&input.outcome).to_string()))
309            .bind(("ip_address", input.ip_address))
310            .bind(("metadata", metadata))
311            .await
312            .map_err(DbError::from)?;
313
314        let mut result = result
315            .check()
316            .map_err(|e| DbError::Migration(e.to_string()))?;
317
318        let rows: Vec<AuditLogRow> = result.take(0).map_err(DbError::from)?;
319        let row = take_first_or_not_found(rows, "audit_log", &id_str)?;
320
321        Ok(row.into_entry(id)?)
322    }

324    async fn list(
325        &self,
326        tenant_id: Uuid,
327        filter: AuditLogFilter,
328        pagination: Pagination,
329    ) -> AxiamResult<PaginatedResult<AuditLogEntry>> {
330        let tenant_id_str = tenant_id.to_string();
331
332        // Build dynamic WHERE clause.
333        let (where_clause, binds) = build_filter_clause(&filter);
334
335        // Count query.
336        let count_sql =
337            format!("SELECT count() AS total FROM audit_log WHERE {where_clause} GROUP ALL");
338        let db = self.db.current();
339        let mut count_query = db.query(&count_sql);
340        count_query = count_query.bind(("tenant_id", tenant_id_str.clone()));
341        count_query = apply_filter_binds(count_query, &binds);
342        let mut count_result = count_query.await.map_err(DbError::from)?;
343        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
344
345        // Data query.
346        let data_sql = format!(
347            "SELECT meta::id(id) AS record_id, * FROM audit_log \
348             WHERE {where_clause} \
349             ORDER BY timestamp DESC \
350             LIMIT $limit START $offset"
351        );
352        let db = self.db.current();
353        let mut data_query = db.query(&data_sql);
354        data_query = data_query.bind(("tenant_id", tenant_id_str));
355        data_query = apply_filter_binds(data_query, &binds);
356        data_query = data_query.bind(("limit", pagination.limit));
357        data_query = data_query.bind(("offset", pagination.offset));
358
359        let mut data_result = data_query.await.map_err(DbError::from)?;
360        let rows: Vec<AuditLogRowWithId> = data_result.take(0).map_err(DbError::from)?;
361
362        let items: Vec<AuditLogEntry> = rows
363            .into_iter()
364            .map(|r| r.try_into_entry())
365            .collect::<Result<_, _>>()?;
366
367        Ok(paginate(items, count_rows, &pagination))
368    }

379    async fn pseudonymize_actor(
380        &self,
381        tenant_id: Uuid,
382        user_id: Uuid,
383        pseudonym: &str,
384    ) -> AxiamResult<u64> {
385        // This is the SINGLE sanctioned non-INSERT write on audit_log (D-04).
386        // It requires the v15 schema permission: FOR update WHERE $auth.role = 'gdpr_pseudonymizer'.
387        //
388        // Full D-03 scrub:
389        // (a) actor_id → nil UUID
390        // (b) metadata.actor_pseudonym → pseudonym string (the new correlation key)
391        // (c) ip_address → NULL
392        // (d) known PII metadata keys redacted to "[redacted]"
393        // (e) resource_id → nil UUID in entries where resource_id == user_id
394        //
395        // NOTE: In SurrealDB in-memory engine (used by tests) the $auth context is not
396        // set, so the permission check may be bypassed. The application-layer guard
397        // (this method is the only path) is the true enforcement mechanism.
398        let nil_uuid = Uuid::nil().to_string();
399        let user_id_str = user_id.to_string();
400        let tenant_id_str = tenant_id.to_string();
401        let pii_keys = ["email", "username", "name", "display_name", "phone"];
402        let redacted_value = json!("[redacted]");
403
404        // Build a SET clause for metadata PII keys using SurrealDB object merge.
405        // We set each known PII key to "[redacted]" if present.
406        let pii_clauses: Vec<String> = pii_keys
407            .iter()
408            .map(|k| format!("metadata.{k} = IF metadata.{k} != NONE THEN '[redacted]' END"))
409            .collect();
410        let pii_set = if pii_clauses.is_empty() {
411            String::new()
412        } else {
413            format!(", {}", pii_clauses.join(", "))
414        };
415
416        // Query 1: scrub actor fields + metadata PII on all entries for this actor.
417        let q1_sql = format!(
418            "UPDATE audit_log SET \
419             actor_id = $nil_uuid, \
420             metadata.actor_pseudonym = $pseudonym, \
421             ip_address = NONE{} \
422             WHERE tenant_id = $tenant_id AND actor_id = $user_id",
423            pii_set
424        );
425        let _ = self
426            .db
427            .current()
428            .query(&q1_sql)
429            .bind(("nil_uuid", nil_uuid.clone()))
430            .bind(("pseudonym", pseudonym.to_string()))
431            .bind(("tenant_id", tenant_id_str.clone()))
432            .bind(("user_id", user_id_str.clone()))
433            .await
434            .map_err(DbError::from)?
435            .check()
436            .map_err(|e| DbError::Migration(e.to_string()))?;
437
438        // Query 2: null-out resource_id in entries where someone acted ON this user.
439        let _ = self
440            .db
441            .current()
442            .query(
443                "UPDATE audit_log SET resource_id = $nil_uuid \
444                 WHERE tenant_id = $tenant_id AND resource_id = $user_id",
445            )
446            .bind(("nil_uuid", nil_uuid.clone()))
447            .bind(("tenant_id", tenant_id_str.clone()))
448            .bind(("user_id", user_id_str.clone()))
449            .await
450            .map_err(DbError::from)?
451            .check()
452            .map_err(|e| DbError::Migration(e.to_string()))?;
453
454        // Query 3: count entries that now have nil actor_id (as a proxy for rows updated).
455        let mut count_result = self
456            .db
457            .current()
458            .query(
459                "SELECT count() AS total FROM audit_log \
460                 WHERE tenant_id = $tenant_id AND actor_id = $nil_uuid GROUP ALL",
461            )
462            .bind(("tenant_id", tenant_id_str))
463            .bind(("nil_uuid", nil_uuid))
464            .await
465            .map_err(DbError::from)?
466            .check()
467            .map_err(|e| DbError::Migration(e.to_string()))?;
468
469        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
470        let _ = redacted_value; // used conceptually in the pii_clauses above
471        Ok(count_rows.first().map(|r| r.total).unwrap_or(0))
472    }

474    async fn prune_older_than(&self, cutoff: DateTime<Utc>) -> AxiamResult<u64> {
475        // T-119. Counted BEFORE deleting: SurrealDB's DELETE returns an empty
476        // result set rather than a row count, and counting afterwards would
477        // report zero no matter how much was removed.
478        let mut count_result = self
479            .db
480            .current()
481            .query("SELECT count() AS total FROM audit_log WHERE timestamp < $cutoff GROUP ALL")
482            .bind(("cutoff", cutoff))
483            .await
484            .map_err(DbError::from)?
485            .check()
486            .map_err(|e| DbError::Migration(e.to_string()))?;
487        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
488        let doomed = count_rows.first().map(|r| r.total).unwrap_or(0);
489        if doomed == 0 {
490            return Ok(0);
491        }
492
493        self.db
494            .current()
495            .query("DELETE audit_log WHERE timestamp < $cutoff")
496            .bind(("cutoff", cutoff))
497            .await
498            .map_err(DbError::from)?
499            .check()
500            .map_err(|e| DbError::Migration(e.to_string()))?;
501
502        // `audit_signature` is currently defined in the schema but never
503        // written by any code, so today this clause deletes nothing. It is
504        // here so that retention already covers signatures on the day signing
505        // is implemented, rather than silently leaving rows that reference
506        // entries this method has just removed. `signed_at < cutoff` is the
507        // conservative boundary: a signature is written no earlier than the
508        // entries it covers, so anything signed before the cutoff can only
509        // cover entries that are themselves past it.
510        self.db
511            .current()
512            .query("DELETE audit_signature WHERE signed_at < $cutoff")
513            .bind(("cutoff", cutoff))
514            .await
515            .map_err(DbError::from)?
516            .check()
517            .map_err(|e| DbError::Migration(e.to_string()))?;
518
519        Ok(doomed)
520    }

522    async fn get_by_ids(&self, tenant_id: Uuid, ids: &[Uuid]) -> AxiamResult<Vec<AuditLogEntry>> {
523        let id_strings: Vec<String> = ids.iter().map(|id| id.to_string()).collect();
524        let r = self
525            .db
526            .current()
527            .query(
528                "SELECT meta::id(id) AS record_id, * FROM audit_log \
529                 WHERE tenant_id = $tenant_id \
530                 AND meta::id(id) IN $ids",
531            )
532            .bind(("tenant_id", tenant_id.to_string()))
533            .bind(("ids", id_strings))
534            .await
535            .map_err(DbError::from)?;
536        let mut result = r.check().map_err(|e| DbError::Migration(e.to_string()))?;
537        let rows: Vec<AuditLogRowWithId> = result.take(0).map_err(DbError::from)?;
538        let entries: Vec<AuditLogEntry> = rows
539            .into_iter()
540            .map(|r| r.try_into_entry().map_err(Into::into))
541            .collect::<AxiamResult<Vec<_>>>()?;
542
543        // Preserve caller-provided order for deterministic signing
544        let index: std::collections::HashMap<Uuid, AuditLogEntry> =
545            entries.into_iter().map(|e| (e.id, e)).collect();
546        Ok(ids.iter().filter_map(|id| index.get(id).cloned()).collect())
547    }

793    /// The Art. 15 export keeps working on minimised rows, and this is the
794    /// precise form of that claim.
795    ///
796    /// The export's `audit_entries` section reads exactly four fields —
797    /// `action`, `outcome`, `timestamp`, `resource_id`
798    /// (`axiam_server::cleanup::aggregate_export_data`) — and **not**
799    /// `ip_address`. So minimisation is invisible to Art. 15: a data subject's
800    /// inventory is structurally and materially identical whether or not the
801    /// deployment minimises.
802    ///
803    /// Asserting it here rather than in the export job is deliberate. The
804    /// property is about what the repository hands the export, the export
805    /// reads those four fields from this row, and a test at this level fails
806    /// the moment a minimisation is added that touches one of them — which is
807    /// the change that would break Art. 15 without anyone noticing.
808    #[tokio::test]
809    async fn minimisation_leaves_every_field_the_art_15_export_reads() {
810        let db = setup_db().await;
811        let tenant_id = Uuid::new_v4();
812        let resource = Uuid::new_v4();
813
814        let mut entry = an_entry(tenant_id, Some("203.0.113.42"));
815        entry.resource_id = Some(resource);
816
817        let minimised = SurrealAuditLogRepository::new(db.clone())
818            .with_minimisation(AuditMinimisation::new(true))
819            .append(entry.clone())
820            .await
821            .unwrap();
822        let whole = SurrealAuditLogRepository::new(db.clone())
823            .append(entry)
824            .await
825            .unwrap();
826
827        assert_eq!(minimised.action, whole.action);
828        assert_eq!(minimised.outcome, whole.outcome);
829        assert_eq!(minimised.resource_id, whole.resource_id);
830        assert_eq!(minimised.actor_id, whole.actor_id);
831        // The one field that differs is the one the export never reads.
832        assert_ne!(minimised.ip_address, whole.ip_address);
833    }
