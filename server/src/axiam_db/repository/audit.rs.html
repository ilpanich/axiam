1//! SurrealDB implementation of [`AuditLogRepository`].
2//!
3//! Audit logs are append-only, and this module is where that is actually
4//! enforced. The table's `FOR update NONE` / `FOR delete NONE` permissions
5//! read like the guarantee but do not provide it: `axiam-server` connects as
6//! a SurrealDB **root** user, and table permission clauses apply to record and
7//! scope users, not to root. The real control is that only three methods here
8//! write anything other than an INSERT, each deliberate and each documented:
9//!
10//! - [`AuditLogRepository::pseudonymize_actor`] — the GDPR erasure scrub
11//!   (D-03/D-04), which anonymises the subject in place rather than deleting
12//!   the record.
13//! - [`AuditLogRepository::prune_older_than`] — retention (T-119), driven by
14//!   the background sweep on a clock and reachable from no HTTP handler.
15//!
16//! Adding a third is a security decision, not a refactor.
17

67fn parse_actor_type(s: &str) -> Result<ActorType, DbError> {
68    match s {
69        "User" => Ok(ActorType::User),
70        "ServiceAccount" => Ok(ActorType::ServiceAccount),
71        "System" => Ok(ActorType::System),
72        other => Err(DbError::Migration(format!("unknown actor type: {other}"))),
73    }
74}
75
76fn actor_type_str(t: &ActorType) -> &'static str {
77    match t {
78        ActorType::User => "User",
79        ActorType::ServiceAccount => "ServiceAccount",
80        ActorType::System => "System",
81    }
82}

84fn parse_outcome(s: &str) -> Result<AuditOutcome, DbError> {
85    match s {
86        "Success" => Ok(AuditOutcome::Success),
87        "Failure" => Ok(AuditOutcome::Failure),
88        "Denied" => Ok(AuditOutcome::Denied),
89        other => Err(DbError::Migration(format!(
90            "unknown audit outcome: {other}"
91        ))),
92    }
93}

95fn outcome_str(o: &AuditOutcome) -> &'static str {
96    match o {
97        AuditOutcome::Success => "Success",
98        AuditOutcome::Failure => "Failure",
99        AuditOutcome::Denied => "Denied",

107impl AuditLogRow {
108    fn into_entry(self, id: Uuid) -> Result<AuditLogEntry, DbError> {
109        let tenant_id = Uuid::parse_str(&self.tenant_id)
110            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
111        let actor_id = Uuid::parse_str(&self.actor_id)
112            .map_err(|e| DbError::Migration(format!("invalid actor UUID: {e}")))?;
113        let resource_id = self
114            .resource_id
115            .as_deref()
116            .map(Uuid::parse_str)
117            .transpose()
118            .map_err(|e| DbError::Migration(format!("invalid resource UUID: {e}")))?;
119
120        Ok(AuditLogEntry {
121            id,
122            tenant_id,
123            actor_id,
124            actor_type: parse_actor_type(&self.actor_type)?,
125            action: self.action,
126            resource_id,
127            outcome: parse_outcome(&self.outcome)?,
128            ip_address: self.ip_address,
129            metadata: self.metadata,
130            timestamp: self.timestamp,
131        })
132    }
133}

169/// Pre-computed bind values for filter parameters.
170enum FilterBind {
171    Str(&'static str, String),
172    DateTime(&'static str, DateTime<Utc>),
173}
174
175/// Build the WHERE clause and collected bind values from an `AuditLogFilter`.
176///
177/// Always includes `tenant_id = $tenant_id` (bound separately by the caller).
178fn build_filter_clause(filter: &AuditLogFilter) -> (String, Vec<FilterBind>) {
179    let mut conditions = vec!["tenant_id = $tenant_id".to_string()];
180    let mut binds: Vec<FilterBind> = Vec::new();
181
182    if let Some(actor_id) = &filter.actor_id {
183        conditions.push("actor_id = $actor_id".into());
184        binds.push(FilterBind::Str("actor_id", actor_id.to_string()));
185    }
186    if let Some(action) = &filter.action {
187        conditions.push("action = $action".into());
188        binds.push(FilterBind::Str("action", action.clone()));
189    }
190    if let Some(outcome) = &filter.outcome {
191        conditions.push("outcome = $outcome".into());
192        binds.push(FilterBind::Str("outcome", outcome_str(outcome).into()));
193    }
194    if let Some(resource_id) = &filter.resource_id {
195        conditions.push("resource_id = $resource_id".into());
196        binds.push(FilterBind::Str("resource_id", resource_id.to_string()));
197    }
198    if let Some(from) = &filter.from {
199        conditions.push("timestamp >= $from_ts".into());
200        binds.push(FilterBind::DateTime("from_ts", *from));
201    }
202    if let Some(to) = &filter.to {
203        conditions.push("timestamp <= $to_ts".into());
204        binds.push(FilterBind::DateTime("to_ts", *to));
205    }
206
207    (conditions.join(" AND "), binds)
208}
209
210/// Apply pre-computed filter binds to a query.
211fn apply_filter_binds<'a, C: Connection>(
212    mut query: surrealdb::method::Query<'a, C>,
213    binds: &'a [FilterBind],
214) -> surrealdb::method::Query<'a, C> {
215    for bind in binds {
216        match bind {
217            FilterBind::Str(key, val) => {
218                query = query.bind((*key, val.clone()));
219            }
220            FilterBind::DateTime(key, val) => {
221                query = query.bind((*key, *val));
222            }
223        }
224    }
225    query
226}

245impl<C: Connection> AuditLogRepository for SurrealAuditLogRepository<C> {
246    async fn append(&self, input: CreateAuditLogEntry) -> AxiamResult<AuditLogEntry> {
247        let id = new_id();
248        let id_str = id.to_string();
249
250        let metadata = input
251            .metadata
252            .unwrap_or(serde_json::Value::Object(Default::default()));
253
254        let resource_id_str = input.resource_id.map(|r| r.to_string());
255
256        let result = self
257            .db
258            .current()
259            .query(
260                "CREATE type::record('audit_log', $id) SET \
261                 tenant_id = $tenant_id, \
262                 actor_id = $actor_id, \
263                 actor_type = $actor_type, \
264                 action = $action, \
265                 resource_id = $resource_id, \
266                 outcome = $outcome, \
267                 ip_address = $ip_address, \
268                 metadata = $metadata",
269            )
270            .bind(("id", id_str.clone()))
271            .bind(("tenant_id", input.tenant_id.to_string()))
272            .bind(("actor_id", input.actor_id.to_string()))
273            .bind(("actor_type", actor_type_str(&input.actor_type).to_string()))
274            .bind(("action", input.action))
275            .bind(("resource_id", resource_id_str))
276            .bind(("outcome", outcome_str(&input.outcome).to_string()))
277            .bind(("ip_address", input.ip_address))
278            .bind(("metadata", metadata))
279            .await
280            .map_err(DbError::from)?;
281
282        let mut result = result
283            .check()
284            .map_err(|e| DbError::Migration(e.to_string()))?;
285
286        let rows: Vec<AuditLogRow> = result.take(0).map_err(DbError::from)?;
287        let row = take_first_or_not_found(rows, "audit_log", &id_str)?;
288
289        Ok(row.into_entry(id)?)
290    }

292    async fn list(
293        &self,
294        tenant_id: Uuid,
295        filter: AuditLogFilter,
296        pagination: Pagination,
297    ) -> AxiamResult<PaginatedResult<AuditLogEntry>> {
298        let tenant_id_str = tenant_id.to_string();
299
300        // Build dynamic WHERE clause.
301        let (where_clause, binds) = build_filter_clause(&filter);
302
303        // Count query.
304        let count_sql =
305            format!("SELECT count() AS total FROM audit_log WHERE {where_clause} GROUP ALL");
306        let db = self.db.current();
307        let mut count_query = db.query(&count_sql);
308        count_query = count_query.bind(("tenant_id", tenant_id_str.clone()));
309        count_query = apply_filter_binds(count_query, &binds);
310        let mut count_result = count_query.await.map_err(DbError::from)?;
311        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
312
313        // Data query.
314        let data_sql = format!(
315            "SELECT meta::id(id) AS record_id, * FROM audit_log \
316             WHERE {where_clause} \
317             ORDER BY timestamp DESC \
318             LIMIT $limit START $offset"
319        );
320        let db = self.db.current();
321        let mut data_query = db.query(&data_sql);
322        data_query = data_query.bind(("tenant_id", tenant_id_str));
323        data_query = apply_filter_binds(data_query, &binds);
324        data_query = data_query.bind(("limit", pagination.limit));
325        data_query = data_query.bind(("offset", pagination.offset));
326
327        let mut data_result = data_query.await.map_err(DbError::from)?;
328        let rows: Vec<AuditLogRowWithId> = data_result.take(0).map_err(DbError::from)?;
329
330        let items: Vec<AuditLogEntry> = rows
331            .into_iter()
332            .map(|r| r.try_into_entry())
333            .collect::<Result<_, _>>()?;
334
335        Ok(paginate(items, count_rows, &pagination))
336    }

347    async fn pseudonymize_actor(
348        &self,
349        tenant_id: Uuid,
350        user_id: Uuid,
351        pseudonym: &str,
352    ) -> AxiamResult<u64> {
353        // This is the SINGLE sanctioned non-INSERT write on audit_log (D-04).
354        // It requires the v15 schema permission: FOR update WHERE $auth.role = 'gdpr_pseudonymizer'.
355        //
356        // Full D-03 scrub:
357        // (a) actor_id → nil UUID
358        // (b) metadata.actor_pseudonym → pseudonym string (the new correlation key)
359        // (c) ip_address → NULL
360        // (d) known PII metadata keys redacted to "[redacted]"
361        // (e) resource_id → nil UUID in entries where resource_id == user_id
362        //
363        // NOTE: In SurrealDB in-memory engine (used by tests) the $auth context is not
364        // set, so the permission check may be bypassed. The application-layer guard
365        // (this method is the only path) is the true enforcement mechanism.
366        let nil_uuid = Uuid::nil().to_string();
367        let user_id_str = user_id.to_string();
368        let tenant_id_str = tenant_id.to_string();
369        let pii_keys = ["email", "username", "name", "display_name", "phone"];
370        let redacted_value = json!("[redacted]");
371
372        // Build a SET clause for metadata PII keys using SurrealDB object merge.
373        // We set each known PII key to "[redacted]" if present.
374        let pii_clauses: Vec<String> = pii_keys
375            .iter()
376            .map(|k| format!("metadata.{k} = IF metadata.{k} != NONE THEN '[redacted]' END"))
377            .collect();
378        let pii_set = if pii_clauses.is_empty() {
379            String::new()
380        } else {
381            format!(", {}", pii_clauses.join(", "))
382        };
383
384        // Query 1: scrub actor fields + metadata PII on all entries for this actor.
385        let q1_sql = format!(
386            "UPDATE audit_log SET \
387             actor_id = $nil_uuid, \
388             metadata.actor_pseudonym = $pseudonym, \
389             ip_address = NONE{} \
390             WHERE tenant_id = $tenant_id AND actor_id = $user_id",
391            pii_set
392        );
393        let _ = self
394            .db
395            .current()
396            .query(&q1_sql)
397            .bind(("nil_uuid", nil_uuid.clone()))
398            .bind(("pseudonym", pseudonym.to_string()))
399            .bind(("tenant_id", tenant_id_str.clone()))
400            .bind(("user_id", user_id_str.clone()))
401            .await
402            .map_err(DbError::from)?
403            .check()
404            .map_err(|e| DbError::Migration(e.to_string()))?;
405
406        // Query 2: null-out resource_id in entries where someone acted ON this user.
407        let _ = self
408            .db
409            .current()
410            .query(
411                "UPDATE audit_log SET resource_id = $nil_uuid \
412                 WHERE tenant_id = $tenant_id AND resource_id = $user_id",
413            )
414            .bind(("nil_uuid", nil_uuid.clone()))
415            .bind(("tenant_id", tenant_id_str.clone()))
416            .bind(("user_id", user_id_str.clone()))
417            .await
418            .map_err(DbError::from)?
419            .check()
420            .map_err(|e| DbError::Migration(e.to_string()))?;
421
422        // Query 3: count entries that now have nil actor_id (as a proxy for rows updated).
423        let mut count_result = self
424            .db
425            .current()
426            .query(
427                "SELECT count() AS total FROM audit_log \
428                 WHERE tenant_id = $tenant_id AND actor_id = $nil_uuid GROUP ALL",
429            )
430            .bind(("tenant_id", tenant_id_str))
431            .bind(("nil_uuid", nil_uuid))
432            .await
433            .map_err(DbError::from)?
434            .check()
435            .map_err(|e| DbError::Migration(e.to_string()))?;
436
437        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
438        let _ = redacted_value; // used conceptually in the pii_clauses above
439        Ok(count_rows.first().map(|r| r.total).unwrap_or(0))
440    }

442    async fn prune_older_than(&self, cutoff: DateTime<Utc>) -> AxiamResult<u64> {
443        // T-119. Counted BEFORE deleting: SurrealDB's DELETE returns an empty
444        // result set rather than a row count, and counting afterwards would
445        // report zero no matter how much was removed.
446        let mut count_result = self
447            .db
448            .current()
449            .query("SELECT count() AS total FROM audit_log WHERE timestamp < $cutoff GROUP ALL")
450            .bind(("cutoff", cutoff))
451            .await
452            .map_err(DbError::from)?
453            .check()
454            .map_err(|e| DbError::Migration(e.to_string()))?;
455        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
456        let doomed = count_rows.first().map(|r| r.total).unwrap_or(0);
457        if doomed == 0 {
458            return Ok(0);
459        }
460
461        self.db
462            .current()
463            .query("DELETE audit_log WHERE timestamp < $cutoff")
464            .bind(("cutoff", cutoff))
465            .await
466            .map_err(DbError::from)?
467            .check()
468            .map_err(|e| DbError::Migration(e.to_string()))?;
469
470        // `audit_signature` is currently defined in the schema but never
471        // written by any code, so today this clause deletes nothing. It is
472        // here so that retention already covers signatures on the day signing
473        // is implemented, rather than silently leaving rows that reference
474        // entries this method has just removed. `signed_at < cutoff` is the
475        // conservative boundary: a signature is written no earlier than the
476        // entries it covers, so anything signed before the cutoff can only
477        // cover entries that are themselves past it.
478        self.db
479            .current()
480            .query("DELETE audit_signature WHERE signed_at < $cutoff")
481            .bind(("cutoff", cutoff))
482            .await
483            .map_err(DbError::from)?
484            .check()
485            .map_err(|e| DbError::Migration(e.to_string()))?;
486
487        Ok(doomed)
488    }

490    async fn get_by_ids(&self, tenant_id: Uuid, ids: &[Uuid]) -> AxiamResult<Vec<AuditLogEntry>> {
491        let id_strings: Vec<String> = ids.iter().map(|id| id.to_string()).collect();
492        let r = self
493            .db
494            .current()
495            .query(
496                "SELECT meta::id(id) AS record_id, * FROM audit_log \
497                 WHERE tenant_id = $tenant_id \
498                 AND meta::id(id) IN $ids",
499            )
500            .bind(("tenant_id", tenant_id.to_string()))
501            .bind(("ids", id_strings))
502            .await
503            .map_err(DbError::from)?;
504        let mut result = r.check().map_err(|e| DbError::Migration(e.to_string()))?;
505        let rows: Vec<AuditLogRowWithId> = result.take(0).map_err(DbError::from)?;
506        let entries: Vec<AuditLogEntry> = rows
507            .into_iter()
508            .map(|r| r.try_into_entry().map_err(Into::into))
509            .collect::<AxiamResult<Vec<_>>>()?;
510
511        // Preserve caller-provided order for deterministic signing
512        let index: std::collections::HashMap<Uuid, AuditLogEntry> =
513            entries.into_iter().map(|e| (e.id, e)).collect();
514        Ok(ids.iter().filter_map(|id| index.get(id).cloned()).collect())
515    }

629            actor_id: Uuid::new_v4(),
630            actor_type: ActorType::System,
631            action: action.into(),
632            resource_id: None,
633            outcome: AuditOutcome::Success,

636        };

640        repo.append(entry("recent.event")).await.unwrap();

655                axiam_core::repository::Pagination::default(),

657            .await
658            .unwrap();

660        assert_eq!(remaining.items[0].action, "recent.event");
