134            not_after: self.not_after,
135            status: self.status,
136            encrypted_private_key: self.encrypted_private_key,
137            key_custody: self.key_custody,
138            key_locator: self.key_locator,
139            mtls_trust_anchor: self.mtls_trust_anchor,
140            created_at: self.created_at,
141        }
142        .try_into_entry()?;
143        Ok(CaVaultForwardCandidate {
144            ca,
145            // Rows from before v96 carry no count: none attempted, due now (D-6).
146            attempts: attempts.map_or(0, |n| u32::try_from(n.max(0)).unwrap_or(u32::MAX)),
147            next_attempt_at,
148            last_error,
149        })

172/// Resolve a row's custodian, tolerating one written before schema v45.
173///
174/// Same rule as the migration's backfill, and here for the same reason it is
175/// there: a row holding ciphertext has its key in itself, and one holding none
176/// never had a key here at all. Reading the second as `Database` would produce
177/// a CA that claims a key it does not have and fails at issuance with a
178/// decryption error rather than a sentence.
179///
180/// An unparseable value falls back the same way rather than failing the read: a
181/// cosmetic column must not take CA listing down for the whole organization.
182fn decode_custody(stored: Option<&str>, has_ciphertext: bool) -> CaKeyCustody {
183    match stored.and_then(|v| v.parse::<CaKeyCustody>().ok()) {
184        Some(custody) => custody,
185        None if has_ciphertext => CaKeyCustody::Database,
186        None => CaKeyCustody::External,
187    }
188}
189
190/// Read a nullable UUID column.
191///
192/// A stored value that will not parse is a corrupt row rather than an absent
193/// one, and is reported as such: silently reading it as `None` would turn a
194/// tenant signing CA into an organization CA, which is a change in who is
195/// allowed to see it.
196fn parse_optional_uuid(stored: Option<&str>, column: &str) -> Result<Option<Uuid>, DbError> {
197    stored
198        .filter(|v| !v.is_empty())
199        .map(|v| {
200            Uuid::parse_str(v)
201                .map_err(|e| DbError::Migration(format!("invalid {column} UUID: {e}")))
202        })
203        .transpose()
204}
205
206fn parse_status(s: &str) -> Result<CertificateStatus, DbError> {
207    match s {
208        "Active" => Ok(CertificateStatus::Active),
209        "Revoked" => Ok(CertificateStatus::Revoked),
210        "Expired" => Ok(CertificateStatus::Expired),
211        other => Err(DbError::Migration(format!(
212            "unknown certificate status: {other}"
213        ))),
214    }
215}
216
217fn status_str(s: &CertificateStatus) -> &'static str {
218    match s {
219        CertificateStatus::Active => "Active",
220        CertificateStatus::Revoked => "Revoked",
221        CertificateStatus::Expired => "Expired",
222    }
223}
224
225fn parse_key_algorithm(s: &str) -> Result<KeyAlgorithm, DbError> {
226    match s {
227        "Rsa4096" => Ok(KeyAlgorithm::Rsa4096),
228        "Ed25519" => Ok(KeyAlgorithm::Ed25519),
229        other => Err(DbError::Migration(format!(
230            "unknown key algorithm: {other}"
231        ))),
232    }
233}
234
235fn key_algorithm_str(k: &KeyAlgorithm) -> &'static str {
236    match k {
237        KeyAlgorithm::Rsa4096 => "Rsa4096",
238        KeyAlgorithm::Ed25519 => "Ed25519",
239    }
240}

246impl CaCertificateRow {
247    fn into_entry(self, id: Uuid) -> Result<CaCertificate, DbError> {
248        let organization_id = Uuid::parse_str(&self.organization_id)
249            .map_err(|e| DbError::Migration(format!("invalid org UUID: {e}")))?;
250        Ok(CaCertificate {
251            id,
252            organization_id,
253            tenant_id: parse_optional_uuid(self.tenant_id.as_deref(), "tenant_id")?,
254            parent_ca_id: parse_optional_uuid(self.parent_ca_id.as_deref(), "parent_ca_id")?,
255            subject: self.subject,
256            public_cert_pem: self.public_cert_pem,
257            chain_pem: self.chain_pem,
258            fingerprint: self.fingerprint,
259            key_algorithm: parse_key_algorithm(&self.key_algorithm)?,
260            not_before: self.not_before,
261            not_after: self.not_after,
262            status: parse_status(&self.status)?,
263            key_custody: decode_custody(
264                self.key_custody.as_deref(),
265                self.encrypted_private_key.is_some(),
266            ),
267            key_locator: self.key_locator,
268            mtls_trust_anchor: self.mtls_trust_anchor.unwrap_or(false),
269            encrypted_private_key: self.encrypted_private_key.map(|b| b.into_inner().to_vec()),
270            created_at: self.created_at,
271        })
272    }
273}

322impl<C: Connection> CaCertificateRepository for SurrealCaCertificateRepository<C> {
323    async fn create(&self, input: StoreCaCertificate) -> AxiamResult<CaCertificate> {
324        // The id comes from the caller, not from `new_id()` here: a custodian
325        // outside the database addresses the key by CA id, so the id has to
326        // exist before the key is stored, and the key has to be stored before
327        // the row.
328        let id = input.id;
329        let id_str = id.to_string();
330
331        let result = self
332            .db
333            .current()
334            .query(
335                "CREATE type::record('ca_certificate', $id) SET \
336                 organization_id = $org_id, \
337                 tenant_id = $tenant_id, \
338                 parent_ca_id = $parent_ca_id, \
339                 subject = $subject, \
340                 public_cert_pem = $public_cert_pem, \
341                 chain_pem = $chain_pem, \
342                 fingerprint = $fingerprint, \
343                 key_algorithm = $key_algorithm, \
344                 not_before = $not_before, \
345                 not_after = $not_after, \
346                 status = $status, \
347                 encrypted_private_key = $encrypted_private_key, \
348                 key_custody = $key_custody, \
349                 key_locator = $key_locator, \
350                 serial = $serial, \
351                 mtls_trust_anchor = false",
352            )
353            .bind(("id", id_str))
354            // #653: recorded only for a CA with a parent in AXIAM, whose serial
355            // that parent chose; the unique index is `(parent_ca_id, serial)`.
356            // A root's serial is its (possibly foreign) issuer's choice and is
357            // not unique across roots, so it is left unrecorded.
358            .bind((
359                "serial",
360                input
361                    .parent_ca_id
362                    .and_then(|_| axiam_pki::certificate_serial_hex(&input.public_cert_pem)),
363            ))
364            .bind(("org_id", input.organization_id.to_string()))
365            .bind(("tenant_id", input.tenant_id.map(|t| t.to_string())))
366            .bind(("parent_ca_id", input.parent_ca_id.map(|p| p.to_string())))
367            .bind(("subject", input.subject))
368            .bind(("public_cert_pem", input.public_cert_pem))
369            .bind(("chain_pem", input.chain_pem))
370            .bind(("fingerprint", input.fingerprint))
371            .bind(("key_algorithm", key_algorithm_str(&input.key_algorithm)))
372            .bind(("not_before", input.not_before))
373            .bind(("not_after", input.not_after))
374            .bind(("status", status_str(&CertificateStatus::Active)))
375            .bind((
376                "encrypted_private_key",
377                input
378                    .encrypted_private_key
379                    .map(surrealdb_types::Bytes::from),
380            ))
381            .bind(("key_custody", input.key_custody.to_string()))
382            .bind(("key_locator", input.key_locator))
383            .await
384            .map_err(DbError::from)?;
385
386        let mut result = result
387            .check()
388            .map_err(|e| DbError::Migration(e.to_string()))?;
389        let row: Option<CaCertificateRow> = result.take(0).map_err(DbError::from)?;
390        let row = row.ok_or_else(|| DbError::NotFound {
391            entity: "ca_certificate".into(),
392            id: id.to_string(),
393        })?;
394
395        Ok(row.into_entry(id)?)
396    }

456    async fn list_by_organization(
457        &self,
458        organization_id: Uuid,
459        pagination: Pagination,
460    ) -> AxiamResult<PaginatedResult<CaCertificate>> {
461        let org_id_str = organization_id.to_string();
462
463        // Count query.
464        let count_sql = "SELECT count() AS total FROM ca_certificate \
465                         WHERE organization_id = $org_id GROUP ALL";
466        let count_result = self
467            .db
468            .current()
469            .query(count_sql)
470            .bind(("org_id", org_id_str.clone()))
471            .await
472            .map_err(DbError::from)?;
473        let mut count_result = count_result
474            .check()
475            .map_err(|e| DbError::Migration(e.to_string()))?;
476        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
477
478        // Data query.
479        let data_sql = "SELECT meta::id(id) AS record_id, * FROM ca_certificate \
480                        WHERE organization_id = $org_id \
481                        ORDER BY created_at DESC \
482                        LIMIT $limit START $offset";
483        let data_result = self
484            .db
485            .current()
486            .query(data_sql)
487            .bind(("org_id", org_id_str))
488            .bind(("limit", pagination.limit))
489            .bind(("offset", pagination.offset))
490            .await
491            .map_err(DbError::from)?;
492        let mut data_result = data_result
493            .check()
494            .map_err(|e| DbError::Migration(e.to_string()))?;
495        let rows: Vec<CaCertificateRowWithId> = data_result.take(0).map_err(DbError::from)?;
496
497        let items: Vec<CaCertificate> = rows
498            .into_iter()
499            .map(|r| r.try_into_entry())
500            .collect::<Result<_, _>>()?;
501
502        Ok(paginate(items, count_rows, &pagination))
503    }

581    async fn set_mtls_trust_anchor(
582        &self,
583        organization_id: Uuid,
584        id: Uuid,
585        enabled: bool,
586    ) -> AxiamResult<CaCertificate> {
587        // The organization guard is on the UPDATE itself, not a read-then-write:
588        // a caller supplying another organization's CA id must match zero rows
589        // and get a not-found, not flip a flag on a CA they cannot see.
590        let result = self
591            .db
592            .current()
593            .query(
594                "UPDATE type::record('ca_certificate', $id) SET \
595                 mtls_trust_anchor = $enabled \
596                 WHERE organization_id = $org_id",
597            )
598            .bind(("id", id.to_string()))
599            .bind(("org_id", organization_id.to_string()))
600            .bind(("enabled", enabled))
601            .await
602            .map_err(DbError::from)?;

725        // Not organization-scoped, by design — see the trait's doc comment.

813    async fn claim_vault_forward(
814        &self,
815        organization_id: Uuid,
816        id: Uuid,
817        seen_next_attempt_at: Option<DateTime<Utc>>,
818        lease_until: DateTime<Utc>,
819    ) -> AxiamResult<bool> {
820        // The leaf table's claim (#603): the value read is the write's
821        // precondition, so of two replicas that read one row exactly one write
822        // matches. A forwarded row, or another organization's, matches nothing.
823        let seen = if seen_next_attempt_at.is_some() {
824            "vault_next_attempt_at = $seen"
825        } else {
826            "vault_next_attempt_at = NONE"
827        };
828        let query = format!(
829            "UPDATE type::record('ca_certificate', $id) SET vault_next_attempt_at = $lease \
830             WHERE organization_id = $org_id \
831               AND status = 'Revoked' \
832               AND vault_revoked_at = NONE \
833               AND {seen} \
834             RETURN VALUE meta::id(id)"
835        );
836        retry_hot_row(|| async {
837            let mut result = self
838                .db
839                .current()
840                .query(query.clone())
841                .bind(("id", id.to_string()))
842                .bind(("org_id", organization_id.to_string()))
843                .bind(("seen", seen_next_attempt_at))
844                .bind(("lease", lease_until))
845                .await
846                .map_err(DbError::from)?
847                .check()
848                .map_err(|e| classify_write_error(e, "ca_certificate"))?;
849            let claimed: Vec<String> = result.take(0).map_err(DbError::from)?;
850            Ok::<_, DbError>(!claimed.is_empty())
851        })
852        .await
853        .map_err(Into::into)
854    }

856    async fn record_vault_forward_failure(
857        &self,
858        organization_id: Uuid,
859        id: Uuid,
860        claimed_until: DateTime<Utc>,
861        attempts: u32,
862        next_attempt_at: DateTime<Utc>,
863        error: &str,
864    ) -> AxiamResult<bool> {
865        // R101W1-09 (d): conditional on the lease the caller's claim wrote, as
866        // the claim is on the value it read. A replica whose attempt outlived
867        // its lease finds another replica's lease (or a forwarded row) and
868        // writes nothing; a write conflict with that replica's claim is
869        // retried and then finds the new lease.
870        let error: String = error.chars().take(MAX_FORWARD_ERROR_CHARS).collect();
871        retry_hot_row(|| async {
872            let mut result = self
873                .db
874                .current()
875                .query(
876                    "UPDATE type::record('ca_certificate', $id) SET \
877                       vault_forward_attempts = $attempts, \
878                       vault_forward_last_error = $error, \
879                       vault_next_attempt_at = $next \
880                     WHERE organization_id = $org_id \
881                       AND vault_revoked_at = NONE \
882                       AND vault_next_attempt_at = $lease \
883                     RETURN VALUE meta::id(id)",
884                )
885                .bind(("id", id.to_string()))
886                .bind(("org_id", organization_id.to_string()))
887                .bind(("lease", claimed_until))
888                .bind(("attempts", i64::from(attempts)))
889                .bind(("error", error.clone()))
890                .bind(("next", next_attempt_at))
891                .await
892                .map_err(DbError::from)?
893                .check()
894                .map_err(|e| classify_write_error(e, "ca_certificate"))?;
895            let written: Vec<String> = result.take(0).map_err(DbError::from)?;
896            Ok::<_, DbError>(!written.is_empty())
897        })
898        .await
899        .map_err(Into::into)
900    }
