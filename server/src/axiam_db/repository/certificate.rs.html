86impl VaultForwardRow {
87    fn try_into_candidate(self) -> Result<VaultForwardCandidate, DbError> {
88        let attempts = self.vault_forward_attempts;
89        let last_error = self.vault_forward_last_error;
90        let next_attempt_at = self.vault_next_attempt_at;
91        let certificate = CertificateRowWithId {
92            record_id: self.record_id,
93            tenant_id: self.tenant_id,
94            issuer_ca_id: self.issuer_ca_id,
95            subject: self.subject,
96            public_cert_pem: self.public_cert_pem,
97            fingerprint: self.fingerprint,
98            cert_type: self.cert_type,
99            key_algorithm: self.key_algorithm,
100            not_before: self.not_before,
101            not_after: self.not_after,
102            status: self.status,
103            metadata: self.metadata,
104            user_id: self.user_id,
105            created_at: self.created_at,
106        }
107        .try_into_entry()?;
108        Ok(VaultForwardCandidate {
109            certificate,
110            // Rows from before v95 carry no count: none attempted, due now (D-6).
111            attempts: attempts.map_or(0, |n| u32::try_from(n.max(0)).unwrap_or(u32::MAX)),
112            next_attempt_at,
113            last_error,
114        })
115    }

118/// The longest error text a row keeps (#603): a status and a short message.
119/// The sweep sanitises before it writes; this bound holds whatever the caller
120/// sent.
121const MAX_FORWARD_ERROR_CHARS: usize = 256;

123/// The unforwarded revocations of active `vault_pki` CAs: the rows the sweep
124/// is about. One definition for the list, the due list and the count.
125const UNFORWARDED_WHERE: &str = "status = 'Revoked' \
126       AND vault_revoked_at = NONE \
127       AND not_after > time::now() \
128       AND issuer_ca_id IN (SELECT VALUE meta::id(id) FROM ca_certificate \
129                            WHERE key_custody = 'vault_pki' \
130                              AND status = 'Active')";
131
132/// `$taken` and `$unambiguous` for [`BELONGS_TO_ACCOUNT`]: the collision rule of
133/// `SurrealUserRepository::identity_collision` — both columns, folded, every
134/// status, the account itself left out — computed in the same query as the
135/// write that uses it, so no caller can skip it (R101W1-01). Binds `$tenant_id`,
136/// `$user_id`, `$names`.
137const ACCOUNT_NAMES_PREFIX: &str = "LET $taken = array::flatten( \
138         SELECT VALUE [string::lowercase(username), string::lowercase(email)] \
139         FROM user \
140         WHERE tenant_id = $tenant_id \
141           AND meta::id(id) != $user_id \
142           AND (string::lowercase(username) IN $names \
143                OR string::lowercase(email) IN $names)); \
144     LET $unambiguous = array::complement($names, $taken);";
145
146/// The condition under which a `User` certificate belongs to the account being
147/// revoked or erased (#533, R101W1-01): the binding, and for a row without one
148/// the convention — `metadata.user_id`, or a common name equal to an
149/// unambiguous name of the account on a row created since the account was.
150/// One definition for the revocation and the metadata scrub, so the second can
151/// never reach a row the first would not. Binds `$user_id`, `$user_created_at`.
152const BELONGS_TO_ACCOUNT: &str = "(user_id = $user_id \
153         OR (user_id = NONE \
154             AND (metadata.user_id = $user_id \
155                  OR (created_at >= $user_created_at \
156                      AND string::lowercase(subject) IN $unambiguous))))";
157
158/// An account's username and email, trimmed and folded, without the empty ones.
159fn folded_names(username: &str, email: &str) -> Vec<String> {
160    [username, email]
161        .into_iter()
162        .map(|name| name.trim().to_lowercase())
163        .filter(|name| !name.is_empty())
164        .collect()
165}

220fn parse_status(s: &str) -> Result<CertificateStatus, DbError> {
221    match s {
222        "Active" => Ok(CertificateStatus::Active),
223        "Revoked" => Ok(CertificateStatus::Revoked),
224        "Expired" => Ok(CertificateStatus::Expired),
225        other => Err(DbError::Migration(format!(
226            "unknown certificate status: {other}"
227        ))),
228    }
229}
230
231fn status_str(s: &CertificateStatus) -> &'static str {
232    match s {
233        CertificateStatus::Active => "Active",
234        CertificateStatus::Revoked => "Revoked",
235        CertificateStatus::Expired => "Expired",
236    }
237}

239fn parse_key_algorithm(s: &str) -> Result<KeyAlgorithm, DbError> {
240    match s {
241        "Rsa4096" => Ok(KeyAlgorithm::Rsa4096),
242        "Ed25519" => Ok(KeyAlgorithm::Ed25519),
243        other => Err(DbError::Migration(format!(
244            "unknown key algorithm: {other}"
245        ))),
246    }
247}

249fn key_algorithm_str(k: &KeyAlgorithm) -> &'static str {
250    match k {
251        KeyAlgorithm::Rsa4096 => "Rsa4096",
252        KeyAlgorithm::Ed25519 => "Ed25519",
253    }
254}
255
256fn parse_cert_type(s: &str) -> Result<CertificateType, DbError> {
257    match s {
258        "User" => Ok(CertificateType::User),
259        "Service" => Ok(CertificateType::Service),
260        "Device" => Ok(CertificateType::Device),
261        "Server" => Ok(CertificateType::Server),
262        // Reads what a bypass of the service layer might have left behind, so a
263        // door refuses it *by type* (and says so) rather than failing opaquely
264        // on the read. Nothing writes it: see `cert_type_str`.
265        "SamlSigning" => Ok(CertificateType::SamlSigning),
266        other => Err(DbError::Migration(format!(
267            "unknown certificate type: {other}"
268        ))),
269    }
270}
271
272fn cert_type_str(t: &CertificateType) -> &'static str {
273    match t {
274        CertificateType::User => "User",
275        CertificateType::Service => "Service",
276        CertificateType::Device => "Device",
277        CertificateType::Server => "Server",
278        // Never stored (D-21): the SAML signing leaf lives in
279        // `saml_idp_credential`. The arm exists so the match is exhaustive, and
280        // the schema's `cert_type` assertion refuses the value if anything ever
281        // tries to write it.
282        CertificateType::SamlSigning => "SamlSigning",
283    }
284}

314            metadata: self.metadata,

364impl<C: Connection> CertificateRepository for SurrealCertificateRepository<C> {
365    async fn create(&self, input: StoreCertificate) -> AxiamResult<Certificate> {
366        let id = new_id();
367        let id_str = id.to_string();
368
369        let result = self
370            .db
371            .current()
372            .query(
373                "CREATE type::record('certificate', $id) SET \
374                 tenant_id = $tenant_id, \
375                 issuer_ca_id = $issuer_ca_id, \
376                 subject = $subject, \
377                 public_cert_pem = $public_cert_pem, \
378                 fingerprint = $fingerprint, \
379                 cert_type = $cert_type, \
380                 key_algorithm = $key_algorithm, \
381                 not_before = $not_before, \
382                 not_after = $not_after, \
383                 status = $status, \
384                 metadata = $metadata, \
385                 user_id = $user_id, \
386                 serial = $serial",
387            )
388            .bind(("id", id_str))
389            .bind(("tenant_id", input.tenant_id.to_string()))
390            .bind(("issuer_ca_id", input.issuer_ca_id.to_string()))
391            .bind(("subject", input.subject))
392            // #653: the serial the certificate carries, recorded so the unique
393            // index on `(issuer_ca_id, serial)` refuses a repeat under one
394            // issuer. `None` for text that is not a parseable certificate.
395            .bind((
396                "serial",
397                axiam_pki::certificate_serial_hex(&input.public_cert_pem),
398            ))
399            .bind(("public_cert_pem", input.public_cert_pem))
400            .bind(("fingerprint", input.fingerprint))
401            .bind(("cert_type", cert_type_str(&input.cert_type)))
402            .bind(("key_algorithm", key_algorithm_str(&input.key_algorithm)))
403            .bind(("not_before", input.not_before))
404            .bind(("not_after", input.not_after))
405            .bind(("status", status_str(&CertificateStatus::Active)))
406            .bind(("metadata", input.metadata))
407            .bind(("user_id", input.user_id.map(|id| id.to_string())))
408            .await
409            .map_err(DbError::from)?;
410
411        let mut result = result
412            .check()
413            .map_err(|e| DbError::Migration(e.to_string()))?;
414        let row: Option<CertificateRow> = result.take(0).map_err(DbError::from)?;
415        let row = row.ok_or_else(|| DbError::NotFound {
416            entity: "certificate".into(),
417            id: id.to_string(),
418        })?;
419
420        // Create the signed_by edge (RELATE doesn't accept type::record()).
421        let relate_sql = format!(
422            "RELATE certificate:`{}`->signed_by->ca_certificate:`{}`",
423            id, input.issuer_ca_id,
424        );
425        self.db
426            .current()
427            .query(&relate_sql)
428            .await
429            .map_err(DbError::from)?
430            .check()
431            .map_err(|e| DbError::Migration(e.to_string()))?;
432
433        Ok(row.into_entry(id)?)
434    }

487    async fn revoke(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
488        let result = self
489            .db
490            .current()
491            .query(
492                // #565: the first revocation's date stands — a CRL entry says
493                // when the CA processed the revocation, and a second revoke of
494                // the same certificate is not a second revocation.
495                "UPDATE type::record('certificate', $id) SET \
496                 status = $status, \
497                 revoked_at = revoked_at ?? time::now() \
498                 WHERE tenant_id = $tenant_id",
499            )
500            .bind(("id", id.to_string()))
501            .bind(("tenant_id", tenant_id.to_string()))

508            .map_err(|e| DbError::Migration(e.to_string()))?;

510        if row.is_none() {
511            return Err(DbError::NotFound {

575        // A row never attempted has no `vault_next_attempt_at`, which sorts
576        // first; then the soonest due, then the oldest revocation.

578            "SELECT meta::id(id) AS record_id, * FROM certificate \
579             WHERE {UNFORWARDED_WHERE} \
580             ORDER BY vault_next_attempt_at, revoked_at \
581             LIMIT $limit"
582        );
583        let result = self
584            .db
585            .current()
586            .query(query)
587            .bind(("limit", limit))
588            .await
589            .map_err(DbError::from)?;

591            .check()
592            .map_err(|e| DbError::Migration(e.to_string()))?;

594        rows.into_iter()
595            .map(|row| row.try_into_entry().map_err(Into::into))
596            .collect()
597    }

628    async fn claim_vault_forward(
629        &self,
630        tenant_id: Uuid,
631        id: Uuid,
632        seen_next_attempt_at: Option<DateTime<Utc>>,
633        lease_until: DateTime<Utc>,
634    ) -> AxiamResult<bool> {
635        // The value read is the precondition of the write (the SCIM
636        // `claim_reconciliation` pattern): of two replicas that read the same
637        // row, one write matches and moves the date; the other either matches
638        // nothing or loses the optimistic write, retries, and then finds the
639        // winner's date. A forwarded row, or another tenant's, matches nothing.
640        let seen = if seen_next_attempt_at.is_some() {
641            "vault_next_attempt_at = $seen"
642        } else {
643            "vault_next_attempt_at = NONE"
644        };
645        let query = format!(
646            "UPDATE type::record('certificate', $id) SET vault_next_attempt_at = $lease \
647             WHERE tenant_id = $tenant_id \
648               AND status = 'Revoked' \
649               AND vault_revoked_at = NONE \
650               AND {seen} \
651             RETURN VALUE meta::id(id)"
652        );
653        retry_hot_row(|| async {
654            let mut result = self
655                .db
656                .current()
657                .query(query.clone())
658                .bind(("id", id.to_string()))
659                .bind(("tenant_id", tenant_id.to_string()))
660                .bind(("seen", seen_next_attempt_at))
661                .bind(("lease", lease_until))
662                .await
663                .map_err(DbError::from)?
664                .check()
665                .map_err(|e| classify_write_error(e, "certificate"))?;
666            let claimed: Vec<String> = result.take(0).map_err(DbError::from)?;
667            Ok::<_, DbError>(!claimed.is_empty())
668        })
669        .await
670        .map_err(Into::into)
671    }

673    async fn record_vault_forward_failure(
674        &self,
675        tenant_id: Uuid,
676        id: Uuid,
677        claimed_until: DateTime<Utc>,
678        attempts: u32,
679        next_attempt_at: DateTime<Utc>,
680        error: &str,
681    ) -> AxiamResult<bool> {
682        // R101W1-09 (d): conditional on the lease the caller's claim wrote, as
683        // the claim is on the value it read. A replica whose attempt outlived
684        // its lease finds another replica's lease (or a forwarded row) and
685        // writes nothing; a write conflict with that replica's claim is
686        // retried and then finds the new lease.
687        let error: String = error.chars().take(MAX_FORWARD_ERROR_CHARS).collect();
688        retry_hot_row(|| async {
689            let mut result = self
690                .db
691                .current()
692                .query(
693                    "UPDATE type::record('certificate', $id) SET \
694                       vault_forward_attempts = $attempts, \
695                       vault_forward_last_error = $error, \
696                       vault_next_attempt_at = $next \
697                     WHERE tenant_id = $tenant_id \
698                       AND vault_revoked_at = NONE \
699                       AND vault_next_attempt_at = $lease \
700                     RETURN VALUE meta::id(id)",
701                )
702                .bind(("id", id.to_string()))
703                .bind(("tenant_id", tenant_id.to_string()))
704                .bind(("lease", claimed_until))
705                .bind(("attempts", i64::from(attempts)))
706                .bind(("error", error.clone()))
707                .bind(("next", next_attempt_at))
708                .await
709                .map_err(DbError::from)?
710                .check()
711                .map_err(|e| classify_write_error(e, "certificate"))?;
712            let written: Vec<String> = result.take(0).map_err(DbError::from)?;
713            Ok::<_, DbError>(!written.is_empty())
714        })
715        .await
716        .map_err(Into::into)
717    }

793    async fn revoke_user_certificates(
794        &self,
795        tenant_id: Uuid,
796        user_id: Uuid,
797        username: &str,
798        email: &str,
799        user_created_at: DateTime<Utc>,
800    ) -> AxiamResult<u64> {
801        // See the trait: the binding first (#533), and the convention —
802        // `metadata.user_id`, or the subject common name equalling the
803        // account's username or email, ignoring case — only for a row with no
804        // binding, so another account's certificate is never matched by name.
805        // Only `User`-type certificates that are still active.
806        //
807        // R101W1-01: the name match is further limited to rows created at or
808        // after the account, and to names no other account of the tenant
809        // holds. `$taken` is the collision rule of
810        // `SurrealUserRepository::identity_collision` — both columns, folded,
811        // every status, the account itself left out — computed in the same
812        // query as the revocation, so no caller can skip it. The binding and
813        // the `metadata.user_id` match (a random id nobody can be named
814        // after) are not bounded.
815        let names = folded_names(username, email);
816        let result = self
817            .db
818            .current()
819            .query(format!(
820                "{ACCOUNT_NAMES_PREFIX} \
821                 SELECT meta::id(id) AS record_id FROM \
822                 (UPDATE certificate SET status = 'Revoked', revoked_at = time::now() \
823                  WHERE tenant_id = $tenant_id \
824                    AND cert_type = 'User' \
825                    AND status = 'Active' \
826                    AND {BELONGS_TO_ACCOUNT})"
827            ))
828            .bind(("tenant_id", tenant_id.to_string()))
829            .bind(("user_id", user_id.to_string()))
830            .bind(("names", names))
831            .bind(("user_created_at", user_created_at))
832            .await
833            .map_err(DbError::from)?;
834        let mut result = result
835            .check()
836            .map_err(|e| DbError::Migration(e.to_string()))?;
837        // LET, LET, SELECT.
838        let rows: Vec<RevokedCertificateRow> = result.take(2).map_err(DbError::from)?;
839        Ok(rows.len() as u64)
840    }

882    async fn delete_past_retention(&self, limit: u32) -> AxiamResult<u64> {
883        // The batch is chosen first and deleted by id, so `LIMIT` bounds the
884        // write; the edge goes before the row it hangs from. Counted from what
885        // `DELETE … RETURN BEFORE` returns, i.e. what this call removed: a
886        // replica that lost the race for a row deletes nothing and counts
887        // nothing. Live tenants only (a tenant row without `deleted_at`).
888        //
889        // The live tenants are read once into `$live`: written inline as
890        // `tenant_id IN (SELECT …)` the subquery is evaluated again for every
891        // candidate row (36 s for 3 000 rows and 301 tenants, R110W1-01).
892        //
893        // Two replicas choosing overlapping batches conflict on the rows they
894        // both delete, and the datastore aborts one transaction having
895        // committed nothing; replaying it re-chooses its batch from what is
896        // left, so a conflict is retried here rather than failing the sweep.
897        let removed = retry_on_write_conflict(|| async {
898            let mut result = self
899                .db
900                .current()
901                .query(format!(
902                    "LET $live = (SELECT VALUE meta::id(id) FROM tenant \
903                                  WHERE deleted_at = NONE); \
904                     LET $doomed = (SELECT VALUE id FROM certificate \
905                         WHERE {PAST_REVOCATION_GRACE} \
906                           AND tenant_id IN $live \
907                         LIMIT $limit); \
908                     DELETE cert_bound_to WHERE in IN $doomed; \
909                     LET $removed = (DELETE $doomed RETURN BEFORE); \
910                     SELECT meta::id(id) AS record_id FROM $removed"
911                ))
912                .bind(("limit", limit))
913                .await
914                .map_err(DbError::from)?
915                .check()
916                .map_err(|e| DbError::Migration(e.to_string()))?;
917            // LET, LET, DELETE, LET, SELECT.
918            let rows: Vec<RevokedCertificateRow> = result.take(4).map_err(DbError::from)?;
919            Ok::<_, DbError>(rows.len() as u64)
920        })
921        .await?;
922        Ok(removed)
923    }

925    async fn list(
926        &self,
927        tenant_id: Uuid,
928        pagination: Pagination,
929    ) -> AxiamResult<PaginatedResult<Certificate>> {
930        let tenant_id_str = tenant_id.to_string();
931
932        // Free-text filter, applied to BOTH queries below so the
933        // total counts matches rather than rows — a pager whose page
934        // count belongs to a different result set than the page it
935        // shows is worse than no pager. Empty when unsearched, so an
936        // unfiltered list runs exactly the query it always ran.
937        let search = search_filter(&pagination, &["subject", "fingerprint"]);
938        let search_term = search_bind(&pagination);
939
940        let count_sql = format!(
941            "SELECT count() AS total FROM certificate \
942                         WHERE tenant_id = $tenant_id{search} GROUP ALL"
943        );
944        let count_result = self
945            .db
946            .current()
947            .query(count_sql)
948            .bind(("tenant_id", tenant_id_str.clone()))
949            .bind(("search", search_term.clone()))
950            .await
951            .map_err(DbError::from)?;
952        let mut count_result = count_result
953            .check()
954            .map_err(|e| DbError::Migration(e.to_string()))?;
955        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
956
957        let data_sql = format!(
958            "SELECT meta::id(id) AS record_id, * FROM certificate \
959                        WHERE tenant_id = $tenant_id{search} \
960                        ORDER BY created_at DESC \
961                        LIMIT $limit START $offset"
962        );
963        let data_result = self
964            .db
965            .current()
966            .query(data_sql)
967            .bind(("tenant_id", tenant_id_str))
968            .bind(("search", search_term))
969            .bind(("limit", pagination.limit))
970            .bind(("offset", pagination.offset))
971            .await
972            .map_err(DbError::from)?;
973        let mut data_result = data_result
974            .check()
975            .map_err(|e| DbError::Migration(e.to_string()))?;
976        let rows: Vec<CertificateRowWithId> = data_result.take(0).map_err(DbError::from)?;
977
978        let items: Vec<Certificate> = rows
979            .into_iter()
980            .map(|r| r.try_into_entry())
981            .collect::<Result<_, _>>()?;
982
983        Ok(paginate(items, count_rows, &pagination))
984    }

1007    async fn bind_to_service_account(
1008        &self,
1009        tenant_id: Uuid,
1010        cert_id: Uuid,
1011        sa_id: Uuid,
1012    ) -> AxiamResult<()> {
1013        // Verify both certificate and service account belong to the same tenant
1014        // before creating the binding.
1015        let verify_sql = format!(
1016            "LET $cert = (SELECT tenant_id FROM certificate:`{cert_id}` WHERE tenant_id = $tid);\
1017             LET $sa = (SELECT tenant_id FROM service_account:`{sa_id}` WHERE tenant_id = $tid);\
1018             IF array::len($cert) = 0 OR array::len($sa) = 0 {{ \
1019                 THROW 'cross-tenant binding denied'; \
1020             }};\
1021             RELATE certificate:`{cert_id}`->cert_bound_to->service_account:`{sa_id}`",
1022        );
1023        let result = self
1024            .db
1025            .current()
1026            .query(&verify_sql)
1027            .bind(("tid", tenant_id.to_string()))
1028            .await
1029            .map_err(DbError::from)?;
1030
1031        if let Err(e) = result.check() {
1032            let msg = e.to_string();
1033            if msg.contains("cross-tenant binding denied") {
1034                return Err(AxiamError::AuthorizationDenied {
1035                    reason: "cross-tenant certificate-to-service-account binding denied".into(),
1036                    action: None,
1037                    resource_id: None,
1038                });
1039            }
1040            return Err(classify_write_error(msg, "certificate_binding").into());
1041        }
1042        Ok(())
1043    }

1045    async fn bound_service_accounts(
1046        &self,
1047        cert_ids: &[Uuid],
1048    ) -> AxiamResult<std::collections::HashMap<Uuid, Uuid>> {
1049        use std::collections::HashMap;
1050
1051        if cert_ids.is_empty() {
1052            return Ok(HashMap::new());
1053        }
1054
1055        #[derive(Debug, SurrealValue)]
1056        struct EdgeRow {
1057            cert_id: String,
1058            sa_id: String,
1059        }
1060
1061        // One query for the whole page. The `in` side is compared as a record
1062        // id, so the list is built from the ids rather than bound as strings —
1063        // they are UUIDs that came from the rows this method's caller just
1064        // read, so there is nothing here an outside caller can shape.
1065        let targets = cert_ids
1066            .iter()
1067            .map(|id| format!("certificate:`{id}`"))
1068            .collect::<Vec<_>>()
1069            .join(", ");
1070        let sql = format!(
1071            "SELECT meta::id(in) AS cert_id, meta::id(out) AS sa_id \
1072             FROM cert_bound_to WHERE in IN [{targets}]"
1073        );
1074
1075        let result = self.db.current().query(&sql).await.map_err(DbError::from)?;
1076        let mut result = result
1077            .check()
1078            .map_err(|e| DbError::Migration(e.to_string()))?;
1079        let rows: Vec<EdgeRow> = result.take(0).map_err(DbError::from)?;
1080
1081        let mut out = HashMap::with_capacity(rows.len());
1082        for row in rows {
1083            let cert = Uuid::parse_str(&row.cert_id)
1084                .map_err(|e| DbError::Migration(format!("invalid certificate UUID: {e}")))?;
1085            let sa = Uuid::parse_str(&row.sa_id)
1086                .map_err(|e| DbError::Migration(format!("invalid service account UUID: {e}")))?;
1087            out.insert(cert, sa);
1088        }
1089        Ok(out)
1090    }
