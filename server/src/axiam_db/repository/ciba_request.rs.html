1//! SurrealDB implementation of [`CibaRequestRepository`] (G-7, schema v80).
2//!
3//! # Two writers, so every transition carries its precondition
4//!
5//! A CIBA request is written by two parties who never coordinate: the user,
6//! deciding on the identity pages, and the client, asking the token endpoint
7//! whether the user has decided. Every write below is therefore one statement
8//! whose `WHERE` clause holds the state it was prepared from (T-406's lesson):
9//!
10//! | write | conditional on |
11//! |---|---|
12//! | approve / deny | `status = 'pending'`, `version` the page read, the request's user, unexpired |
13//! | mark expired | `status IN ['pending','approved']`, `version`, past `expires_at` |
14//! | record a poll | the `last_polled_at` the token endpoint read (compare-and-set) |
15//! | redeem | `status = 'approved'`, the client that started it, unexpired — the X6 arbiter |
16//!
17//! Polls deliberately do **not** bump `version`: a client polling every five
18//! seconds would otherwise make every approval page's read stale before the
19//! user could click.
20//!
21//! # Single use — the X6 two-layer arbiter
22//!
23//! [`CibaRequestRepository::redeem`] is `device_grant.redeem` with one more
24//! guard (the client): a guarded `UPDATE … RETURN BEFORE` inside an explicit
25//! transaction, so two concurrent redemptions conflict on one key and the
26//! engine aborts the loser, and then a per-attempt nonce read back **after**
27//! the commit in a query of its own, so a conflict the engine missed is still
28//! caught unless the two commits interleave around the read-back. See
29//! `device_grant.rs` for the measurements and why the read-back must stay
30//! outside the transaction.
31//!
32//! # Ping-mode credentials are sealed
33//!
34//! A ping-mode request must keep its `auth_req_id` recoverable (the
35//! notification body is that identifier) and the client's
36//! `client_notification_token` (the bearer the notification presents). Both go
37//! into one AES-256-GCM blob under `pki_encryption_key`, nonce in its own
38//! column; no read of the table projects either column except
39//! [`CibaRequestRepository::ping_credentials`]. With no key configured a
40//! ping-mode request is refused with `ServiceUnavailable` rather than stored in
41//! clear.

99impl CibaRequestRow {
100    fn try_into_request(self) -> Result<CibaRequest, DbError> {
101        let status = CibaRequestStatus::from_wire(&self.status).ok_or_else(|| {
102            DbError::Migration(format!(
103                "unrecognised CIBA request status {:?}",
104                self.status
105            ))
106        })?;
107        let delivery_mode = CibaDeliveryMode::from_wire(&self.delivery_mode).ok_or_else(|| {
108            DbError::Migration(format!(
109                "unrecognised CIBA delivery mode {:?}",
110                self.delivery_mode
111            ))
112        })?;
113        let user_id = match self.user_id.as_deref() {
114            None | Some("") => None,
115            Some(raw) => Some(parse_uuid(raw, "user")?),
116        };
117        // The evidence is all or nothing: a row with a session but no time is
118        // not evidence of anything, and is read as none.
119        let approval = match (
120            self.approval_session_id.as_deref(),
121            self.auth_time,
122            self.acr,
123        ) {
124            (Some(session), Some(auth_time), Some(acr)) => Some(CibaApprovalEvidence {
125                session_id: parse_uuid(session, "session")?,
126                auth_time,
127                acr,
128                amr: Amr::decode_list(&self.amr),
129            }),
130            _ => None,
131        };
132        Ok(CibaRequest {
133            id: parse_uuid(&self.record_id, "CIBA request")?,
134            tenant_id: parse_uuid(&self.tenant_id, "tenant")?,
135            client_id: self.client_id,
136            auth_req_id_hash: self.auth_req_id_hash,
137            user_id,
138            scopes: self.scopes,
139            binding_message: self.binding_message,
140            acr_values: self.acr_values,
141            resource: self.resource,
142            delivery_mode,
143            status,
144            version: self.version.max(0) as u64,
145            interval_secs: self.interval_secs.max(1) as u64,
146            last_polled_at: self.last_polled_at,
147            expires_at: self.expires_at,
148            approval,
149            decided_at: self.decided_at,
150            created_at: self.created_at,
151        })
152    }
153}

169pub struct SurrealCibaRequestRepository<C: Connection> {

206    fn missing_key() -> AxiamError {
207        AxiamError::ServiceUnavailable(format!(
208            "a ping-mode CIBA request cannot be stored: {PKI_ENCRYPTION_KEY} ({}) is not \

214    async fn fetch_one(
215        &self,
216        query: String,
217        binds: Vec<(&'static str, String)>,
218    ) -> AxiamResult<Option<CibaRequest>> {
219        let db = self.db.current();
220        let mut builder = db.query(query);
221        for bind in binds {
222            builder = builder.bind(bind);
223        }
224        let mut result = builder.await.map_err(DbError::from)?;
225        let rows: Vec<CibaRequestRow> = result.take(0).map_err(DbError::from)?;
226        rows.into_iter()
227            .next()
228            .map(CibaRequestRow::try_into_request)
229            .transpose()
230            .map_err(Into::into)
231    }
232}

234/// Whether a guarded `UPDATE … RETURN meta::id(id) AS record_id` matched.
235///
236/// A conflicting concurrent write is "the precondition failed", not a fault.
237fn matched(result: Result<surrealdb::IndexedResults, surrealdb::Error>) -> AxiamResult<bool> {
238    let mut result = match result {
239        Ok(r) => r,
240        Err(e) if is_transaction_conflict(&e) => return Ok(false),
241        Err(e) => return Err(DbError::from(e).into()),
242    };
243    let rows: Vec<IdRow> = match result.take(0) {
244        Ok(rows) => rows,
245        Err(e) if is_transaction_conflict(&e) => return Ok(false),
246        Err(e) => return Err(DbError::from(e).into()),
247    };
248    Ok(!rows.is_empty())
249}

251impl<C: Connection> CibaRequestRepository for SurrealCibaRequestRepository<C> {
252    async fn create(&self, input: CreateCibaRequest) -> AxiamResult<CibaRequest> {
253        let id = new_id();
254        let (ping_nonce, ping_ciphertext) = match &input.ping {
255            None => (None, None),
256            Some(creds) => {
257                let key = self.key.as_ref().ok_or_else(Self::missing_key)?;
258                let plaintext = Zeroizing::new(
259                    serde_json::to_vec(&PingPlaintext {
260                        auth_req_id: creds.auth_req_id.clone(),
261                        client_notification_token: creds.client_notification_token.clone(),
262                    })
263                    .map_err(|_| {
264                        AxiamError::Crypto("the ping credentials could not be encoded".into())
265                    })?,
266                );
267                let (nonce, ciphertext) = encrypt_separate(key, &plaintext).map_err(|_| {
268                    AxiamError::Crypto("the ping credentials could not be encrypted".into())
269                })?;
270                (Some(nonce), Some(ciphertext))
271            }
272        };
273
274        let mut result = self
275            .db
276            .current()
277            .query(format!(
278                "CREATE type::record('ciba_request', $id) SET \
279                 tenant_id = $tenant_id, \
280                 client_id = $client_id, \
281                 auth_req_id_hash = $hash, \
282                 user_id = $user_id, \
283                 scopes = $scopes, \
284                 binding_message = $binding_message, \
285                 acr_values = $acr_values, \
286                 resource = $resource, \
287                 delivery_mode = $delivery_mode, \
288                 ping_ciphertext = $ping_ciphertext, \
289                 ping_nonce = $ping_nonce, \
290                 status = 'pending', \
291                 version = 0, \
292                 interval_secs = $interval_secs, \
293                 last_polled_at = NONE, \
294                 expires_at = $expires_at, \
295                 approval_session_id = NONE, \
296                 auth_time = NONE, \
297                 acr = NONE, \
298                 amr = [], \
299                 decided_at = NONE, \
300                 redemption_id = NONE, \
301                 created_at = time::now() \
302                 RETURN {SELECT_FIELDS}"
303            ))
304            .bind(("id", id.to_string()))
305            .bind(("tenant_id", input.tenant_id.to_string()))
306            .bind(("client_id", input.client_id))
307            .bind(("hash", input.auth_req_id_hash))
308            .bind(("user_id", input.user_id.map(|u| u.to_string())))
309            .bind(("scopes", input.scopes))
310            .bind(("binding_message", input.binding_message))
311            .bind(("acr_values", input.acr_values))
312            .bind(("resource", input.resource))
313            .bind(("delivery_mode", input.delivery_mode.as_str().to_owned()))
314            .bind(("ping_ciphertext", ping_ciphertext))
315            .bind(("ping_nonce", ping_nonce))
316            .bind(("interval_secs", input.interval_secs as i64))
317            .bind(("expires_at", input.expires_at))
318            .await
319            .map_err(DbError::from)?;
320
321        let rows: Vec<CibaRequestRow> = result.take(0).map_err(DbError::from)?;
322        rows.into_iter()
323            .next()
324            .ok_or_else(|| DbError::Migration("CIBA request not returned after create".into()))?
325            .try_into_request()
326            .map_err(Into::into)
327    }

440    async fn record_poll(
441        &self,
442        tenant_id: Uuid,
443        id: Uuid,
444        seen_last_polled_at: Option<DateTime<Utc>>,
445        polled_at: DateTime<Utc>,
446        interval_secs: u64,
447    ) -> AxiamResult<bool> {
448        // Compare-and-set on the value the caller read. `last_polled_at = NONE`
449        // is its own spelling because NONE does not compare equal to a bound
450        // value in SurrealQL.
451        let guard = if seen_last_polled_at.is_some() {
452            "last_polled_at = $seen"
453        } else {
454            "last_polled_at = NONE"
455        };
456        matched(
457            self.db
458                .current()
459                .query(format!(
460                    "UPDATE type::record('ciba_request', $id) SET \
461                         last_polled_at = $polled_at, interval_secs = $interval \
462                     WHERE tenant_id = $tenant_id AND {guard} \
463                     RETURN meta::id(id) AS record_id"
464                ))
465                .bind(("id", id.to_string()))
466                .bind(("tenant_id", tenant_id.to_string()))
467                .bind(("polled_at", polled_at))
468                .bind(("interval", interval_secs as i64))
469                .bind(("seen", seen_last_polled_at))
470                .await,
471        )
472    }

474    async fn redeem(
475        &self,
476        tenant_id: Uuid,
477        auth_req_id_hash: &str,
478        client_id: &str,
479    ) -> AxiamResult<Option<CibaRequest>> {
480        // Layer 1: the guarded UPDATE inside an explicit transaction.
481        let nonce = new_id().to_string();
482        let result = self
483            .db
484            .current()
485            .query(format!(
486                "BEGIN TRANSACTION; \
487                 LET $before = (UPDATE ciba_request \
488                     SET status = 'redeemed', version = version + 1, redemption_id = $nonce \
489                     WHERE tenant_id = $tenant_id AND auth_req_id_hash = $hash \
490                     AND client_id = $client_id \
491                     AND status = 'approved' AND expires_at > time::now() \
492                     RETURN BEFORE); \
493                 SELECT {SELECT_FIELDS} FROM $before; \
494                 COMMIT TRANSACTION"
495            ))
496            .bind(("tenant_id", tenant_id.to_string()))
497            .bind(("hash", auth_req_id_hash.to_owned()))
498            .bind(("client_id", client_id.to_owned()))
499            .bind(("nonce", nonce.clone()))
500            .await;
501        let mut result = match result {
502            Ok(r) => r,
503            Err(e) if is_transaction_conflict(&e) => return Ok(None),
504            Err(e) => return Err(DbError::from(e).into()),
505        };
506        // BEGIN=0, LET=1, SELECT=2, COMMIT=3.
507        let rows: Vec<CibaRequestRow> = match result.take::<Vec<CibaRequestRow>>(2) {
508            Ok(rows) => rows,
509            Err(e) if is_transaction_conflict(&e) => return Ok(None),
510            Err(e) => return Err(DbError::from(e).into()),
511        };
512        if rows.is_empty() {
513            return Ok(None);
514        }
515
516        // Layer 2: outside, and after, the transaction above.
517        let stored = self
518            .db
519            .current()
520            .query(
521                "SELECT VALUE redemption_id FROM ciba_request \
522                 WHERE tenant_id = $tenant_id AND auth_req_id_hash = $hash LIMIT 1",
523            )
524            .bind(("tenant_id", tenant_id.to_string()))
525            .bind(("hash", auth_req_id_hash.to_owned()))
526            .await;
527        let mut stored = match stored {
528            Ok(r) => r,
529            Err(e) if is_transaction_conflict(&e) => return Ok(None),
530            Err(e) => return Err(DbError::from(e).into()),
531        };
532        let stored: Vec<Option<String>> = match stored.take::<Vec<Option<String>>>(0) {
533            Ok(v) => v,
534            Err(e) if is_transaction_conflict(&e) => return Ok(None),
535            Err(e) => return Err(DbError::from(e).into()),
536        };
537        if stored.into_iter().flatten().next().as_deref() != Some(nonce.as_str()) {
538            // Another redemption's write landed after ours: it holds the
539            // request, and this one must not also mint a token set.
540            return Ok(None);
541        }
542
543        rows.into_iter()
544            .next()
545            .map(CibaRequestRow::try_into_request)
546            .transpose()
547            .map_err(Into::into)
548    }

550    async fn ping_credentials(
551        &self,
552        tenant_id: Uuid,
553        id: Uuid,
554    ) -> AxiamResult<Option<CibaPingCredentials>> {
555        #[derive(SurrealValue)]
556        struct SealedRow {
557            ping_ciphertext: Option<String>,
558            ping_nonce: Option<String>,
559        }
560        let mut result = self
561            .db
562            .current()
563            .query(
564                "SELECT ping_ciphertext, ping_nonce FROM ciba_request \
565                 WHERE meta::id(id) = $id AND tenant_id = $tenant_id",
566            )
567            .bind(("id", id.to_string()))
568            .bind(("tenant_id", tenant_id.to_string()))
569            .await
570            .map_err(DbError::from)?;
571        let rows: Vec<SealedRow> = result.take(0).map_err(DbError::from)?;
572        let Some(row) = rows.into_iter().next() else {
573            return Ok(None);
574        };
575        let (Some(ciphertext), Some(nonce)) = (row.ping_ciphertext, row.ping_nonce) else {
576            return Ok(None);
577        };
578        let key = self.key.as_ref().ok_or_else(Self::missing_key)?;
579        let plaintext =
580            Zeroizing::new(decrypt_separate(key, &nonce, &ciphertext).map_err(|_| {
581                AxiamError::Crypto(
582                    "the CIBA ping credentials could not be decrypted: the stored value does \
583                     not match the configured pki_encryption_key"
584                        .into(),
585                )
586            })?);
587        let decoded: PingPlaintext = serde_json::from_slice(&plaintext)
588            .map_err(|_| AxiamError::Crypto("the CIBA ping credentials are malformed".into()))?;
589        Ok(Some(CibaPingCredentials {
590            auth_req_id: decoded.auth_req_id,
591            client_notification_token: decoded.client_notification_token,
592        }))
593    }

595    async fn sweep_expired(
596        &self,
597        now: DateTime<Utc>,
598        retention: chrono::Duration,
599    ) -> AxiamResult<u64> {
600        // First mark, then delete: a request that expired a moment ago stays
601        // long enough (`retention`) for a client still polling to be told
602        // `expired_token` rather than `invalid_grant`.
603        let cutoff = now - retention;
604        let mut result = self
605            .db
606            .current()
607            .query(
608                "UPDATE ciba_request SET status = 'expired', version = version + 1 \
609                     WHERE status IN ['pending', 'approved'] AND expires_at <= $now \
610                     RETURN NONE; \
611                 LET $removed = (DELETE ciba_request WHERE expires_at <= $cutoff RETURN BEFORE); \
612                 SELECT meta::id(id) AS record_id FROM $removed",
613            )
614            .bind(("now", now))
615            .bind(("cutoff", cutoff))
616            .await
617            .map_err(DbError::from)?;
618        // UPDATE=0, LET=1, SELECT=2.
619        let removed: Vec<IdRow> = result.take(2).map_err(DbError::from)?;
620        Ok(removed.len() as u64)
621    }
622}
