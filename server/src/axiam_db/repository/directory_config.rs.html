1//! SurrealDB implementation of [`DirectoryConfigRepository`] (G-3, D-15).
2//!
3//! One row per tenant. The bind secret is encrypted at rest with AES-256-GCM,
4//! a fresh nonce per write, exactly as the per-tenant SMTP password is in
5//! `email_config.rs`; the 256-bit key is fetched from the secret provider under
6//! `directory_encryption_key` and handed to [`SurrealDirectoryConfigRepository::new`]
7//! by the composition root. It is **optional**: a repository built without it
8//! fails closed on `create`, `update` and `decrypt_bind_secret` with an error
9//! naming the key, and still serves reads, `delete` and `list_enabled`.
10//!
11//! The secret is write-only. Every read projects the non-secret columns
12//! explicitly, so the ciphertext never leaves the datastore except through
13//! [`SurrealDirectoryConfigRepository::decrypt_bind_secret`], the single path
14//! to the plaintext, which the bind path (T23.3.2) calls.
15

35/// The key version written today. The column exists so a future rotation can
36/// tell which key sealed a row; `email_config` writes the same constant.
37const SECRET_KEY_VERSION: i64 = 1;

40/// hydrates `bind_secret_ciphertext` or `bind_secret_nonce`.
41const PUBLIC_COLUMNS: &str = "meta::id(id) AS record_id, tenant_id, enabled, kind, url, \

45    jit_provisioning, trust_anchors_pem, created_at, updated_at";

98impl DirectoryRow {
99    fn into_domain(self) -> Result<DirectoryConfig, DbError> {
100        let kind = DirectoryKind::from_wire(&self.kind).ok_or_else(|| {
101            DbError::Serialization("directory_config row has an unknown kind".to_string())
102        })?;
103        let depth = u8::try_from(self.group_nesting_depth).map_err(|_| {
104            DbError::Serialization("directory_config group_nesting_depth is out of range".into())
105        })?;
106        let interval = u64::try_from(self.sync_interval_secs).map_err(|_| {
107            DbError::Serialization("directory_config sync_interval_secs is out of range".into())
108        })?;
109        let group_mappings = self
110            .group_mappings
111            .unwrap_or_default()
112            .into_iter()
113            .map(|row| {
114                Ok(GroupMapping {
115                    directory_group_dn: row.directory_group_dn,
116                    group_id: parse_uuid(&row.group_id, "group")?,
117                })
118            })
119            .collect::<Result<Vec<_>, DbError>>()?;
120        Ok(DirectoryConfig {
121            id: parse_uuid(&self.record_id, "directory_config")?,
122            tenant_id: parse_uuid(&self.tenant_id, "tenant")?,
123            enabled: self.enabled,
124            kind,
125            url: self.url,
126            start_tls: self.start_tls,
127            bind_dn: self.bind_dn,
128            base_dn: self.base_dn,
129            user_filter: self.user_filter,
130            user_attribute_map: UserAttributeMap {
131                username: self.attr_username,
132                email: self.attr_email,
133                display_name: self.attr_display_name,
134                external_id: self.attr_external_id,
135            },
136            group_base_dn: self.group_base_dn,
137            group_filter: self.group_filter,
138            group_member_attribute: self.group_member_attribute,
139            group_nesting_depth: depth,
140            group_mappings,
141            sync_interval_secs: interval,
142            jit_provisioning: self.jit_provisioning,
143            trust_anchors_pem: self.trust_anchors_pem,
144            created_at: self.created_at,
145            updated_at: self.updated_at,
146        })
147    }
148}

174impl<C: Connection> SurrealDirectoryConfigRepository<C> {
175    /// Build the repository. `key` is the 256-bit key the secret provider holds
176    /// as `directory_encryption_key`, or `None` when it is not configured, in
177    /// which case the directory feature is unavailable (see the module docs).
178    pub fn new(db: impl Into<DbHandle<C>>, key: Option<[u8; 32]>) -> Self {
179        Self { db: db.into(), key }
180    }
181
182    /// Whether the encryption key is configured: the one fact a management
183    /// route needs to answer "this deployment does not have the feature" before
184    /// it validates anything, without learning or logging the key itself.
185    #[must_use]
186    pub fn has_encryption_key(&self) -> bool {
187        self.key.is_some()
188    }
189
190    /// The text every missing-key refusal carries: the logical name and the
191    /// variable the default provider reads, so the operator can act on it.
192    fn missing_key_message() -> String {
193        format!(
194            "the directory feature is unavailable: {DIRECTORY_ENCRYPTION_KEY} \
195             ({}) is not configured",
196            env_var_name(DIRECTORY_ENCRYPTION_KEY)
197        )
198    }
199
200    /// The key for a write, or the fail-closed refusal.
201    fn key_for_write(&self) -> AxiamResult<&[u8; 32]> {
202        self.key.as_ref().ok_or_else(|| AxiamError::Validation {
203            message: Self::missing_key_message(),
204        })
205    }
206
207    /// The write-path rules of the mapping table that need the datastore
208    /// (D-30): at most [`GROUP_MAPPINGS_MAX`] rows, and **every `group_id` a
209    /// group of `tenant_id`**. A mapping naming another tenant's group, or none
210    /// at all, is refused before the row is touched, so a tenant administrator
211    /// can never point their directory at a group they do not own.
212    ///
213    /// The shape of each DN is `axiam-directory::config::validate`'s rule (this
214    /// layer cannot reach it); the two together are what `T23.3.8` calls.
215    async fn check_group_mappings(
216        &self,
217        tenant_id: Uuid,
218        mappings: &[GroupMapping],
219    ) -> AxiamResult<()> {
220        if mappings.len() > GROUP_MAPPINGS_MAX {
221            return Err(AxiamError::Validation {
222                message: format!(
223                    "directory group_mappings may hold at most {GROUP_MAPPINGS_MAX} entries"
224                ),
225            });
226        }
227        let mut wanted: Vec<String> = mappings.iter().map(|m| m.group_id.to_string()).collect();
228        wanted.sort_unstable();
229        wanted.dedup();
230        if wanted.is_empty() {
231            return Ok(());
232        }
233        let mut result = self
234            .db
235            .current()
236            .query(
237                "SELECT meta::id(id) AS record_id FROM group \
238                 WHERE tenant_id = $tenant_id AND meta::id(id) IN $ids",
239            )
240            .bind(("tenant_id", tenant_id.to_string()))
241            .bind(("ids", wanted.clone()))
242            .await
243            .map_err(DbError::from)?;
244        let found: Vec<GroupIdRow> = result.take(0).map_err(DbError::from)?;
245        if found.len() != wanted.len() {
246            return Err(AxiamError::Validation {
247                message: "directory group_mappings names a group that is not a group of this \
248                          tenant"
249                    .into(),
250            });
251        }
252        Ok(())
253    }
254
255    async fn fetch_public(&self, tenant_id: Uuid) -> AxiamResult<Option<DirectoryConfig>> {
256        let mut result = self
257            .db
258            .current()
259            .query(format!(
260                "SELECT {PUBLIC_COLUMNS} FROM directory_config WHERE tenant_id = $tenant_id"
261            ))
262            .bind(("tenant_id", tenant_id.to_string()))
263            .await
264            .map_err(DbError::from)?;
265        let rows: Vec<DirectoryRow> = result.take(0).map_err(DbError::from)?;
266        match rows.into_iter().next() {
267            Some(row) => Ok(Some(row.into_domain()?)),
268            None => Ok(None),
269        }
270    }
271}

273/// Encrypt the bind secret, returning `(nonce_b64, ciphertext_b64)`.
274///
275/// The error carries no detail: whatever the cipher says, none of it belongs in
276/// a response or a log next to a credential.
277fn seal(key: &[u8; 32], secret: &str) -> AxiamResult<(String, String)> {
278    encrypt_separate(key, secret.as_bytes())
279        .map_err(|_| AxiamError::Crypto("the directory bind secret could not be encrypted".into()))
280}

282impl<C: Connection> DirectoryConfigRepository for SurrealDirectoryConfigRepository<C> {
283    async fn create(&self, input: NewDirectoryConfig) -> AxiamResult<DirectoryConfig> {
284        let key = self.key_for_write()?;
285        let secret = input
286            .bind_secret
287            .as_ref()
288            .ok_or_else(|| AxiamError::Validation {
289                message: "a directory configuration requires a bind secret when it is created"
290                    .into(),
291            })?;
292        let (nonce, ciphertext) = seal(key, secret)?;
293        self.check_group_mappings(input.tenant_id, &input.group_mappings)
294            .await?;
295        let mapping_rows: Vec<GroupMappingRow> = input
296            .group_mappings
297            .iter()
298            .map(|m| GroupMappingRow {
299                directory_group_dn: m.directory_group_dn.clone(),
300                group_id: m.group_id.to_string(),
301            })
302            .collect();
303
304        let id = new_id();
305        let tenant_id = input.tenant_id;
306        let result = self
307            .db
308            .current()
309            .query(
310                "CREATE type::record('directory_config', $id) SET \
311                 tenant_id = $tenant_id, enabled = $enabled, kind = $kind, \
312                 url = $url, start_tls = $start_tls, bind_dn = $bind_dn, \
313                 bind_secret_ciphertext = $ciphertext, bind_secret_nonce = $nonce, \
314                 secret_key_version = $key_version, base_dn = $base_dn, \
315                 user_filter = $user_filter, attr_username = $attr_username, \
316                 attr_email = $attr_email, attr_display_name = $attr_display_name, \
317                 attr_external_id = $attr_external_id, \
318                 group_base_dn = $group_base_dn, group_filter = $group_filter, \
319                 group_member_attribute = $group_member_attribute, \
320                 group_nesting_depth = $group_nesting_depth, \
321                 group_mappings = $group_mappings, \
322                 sync_interval_secs = $sync_interval_secs, \
323                 jit_provisioning = $jit_provisioning, \
324                 trust_anchors_pem = $trust_anchors_pem, \
325                 created_at = time::now(), updated_at = time::now()",
326            )
327            .bind(("id", id.to_string()))
328            .bind(("tenant_id", tenant_id.to_string()))
329            .bind(("enabled", input.enabled))
330            .bind(("kind", input.kind.as_str().to_string()))
331            .bind(("url", input.url))
332            .bind(("start_tls", input.start_tls))
333            .bind(("bind_dn", input.bind_dn))
334            .bind(("ciphertext", ciphertext))
335            .bind(("nonce", nonce))
336            .bind(("key_version", SECRET_KEY_VERSION))
337            .bind(("base_dn", input.base_dn))
338            .bind(("user_filter", input.user_filter))
339            .bind(("attr_username", input.user_attribute_map.username))
340            .bind(("attr_email", input.user_attribute_map.email))
341            .bind(("attr_display_name", input.user_attribute_map.display_name))
342            .bind(("attr_external_id", input.user_attribute_map.external_id))
343            .bind(("group_base_dn", input.group_base_dn))
344            .bind(("group_filter", input.group_filter))
345            .bind(("group_member_attribute", input.group_member_attribute))
346            .bind(("group_nesting_depth", i64::from(input.group_nesting_depth)))
347            .bind(("group_mappings", mapping_rows))
348            .bind((
349                "sync_interval_secs",
350                i64::try_from(input.sync_interval_secs).map_err(|_| AxiamError::Validation {
351                    message: "sync_interval_secs is out of range".into(),
352                })?,
353            ))
354            .bind(("jit_provisioning", input.jit_provisioning))
355            .bind(("trust_anchors_pem", input.trust_anchors_pem))
356            .await
357            .map_err(DbError::from)?;
358        // The unique index on `tenant_id` is what makes a second create for the
359        // same tenant an `AlreadyExists` rather than a second configuration.
360        result
361            .check()
362            .map_err(|e| classify_write_error(e, "directory_config"))?;
363
364        let created = self.fetch_public(tenant_id).await?;
365        Ok(take_first_or_not_found(
366            created.into_iter().collect(),
367            "directory_config",
368            &tenant_id.to_string(),
369        )?)
370    }

372    async fn update(&self, input: NewDirectoryConfig) -> AxiamResult<DirectoryConfig> {
373        // `None` keeps the stored ciphertext and nonce untouched; `Some` writes
374        // both, under a fresh nonce. **The key is needed only to seal**: an
375        // update that carries no secret (enabling or disabling the directory,
376        // editing a filter) neither reads nor writes one, so a deployment that
377        // later lost its key can still switch its directory off (T23.3.8,
378        // CONTRACT §30.3 rule 4: only a write that carries a secret is `503`).
379        let sealed = match &input.bind_secret {
380            Some(secret) => Some(seal(self.key_for_write()?, secret)?),
381            None => None,
382        };
383        // D-30: before anything is written, so a refused table changes nothing.
384        self.check_group_mappings(input.tenant_id, &input.group_mappings)
385            .await?;
386        let mapping_rows: Vec<GroupMappingRow> = input
387            .group_mappings
388            .iter()
389            .map(|m| GroupMappingRow {
390                directory_group_dn: m.directory_group_dn.clone(),
391                group_id: m.group_id.to_string(),
392            })
393            .collect();
394
395        let secret_sets = if sealed.is_some() {
396            "bind_secret_ciphertext = $ciphertext, bind_secret_nonce = $nonce, \
397             secret_key_version = $key_version, "
398        } else {
399            ""
400        };
401
402        // P23W2-01: a stored secret is never sent to a server it was not
403        // entered for. Without a new secret, the fields that decide where and
404        // how the service bind goes — the URL, StartTLS, the bind DN and the
405        // trust anchors — must equal the stored ones, or the update matches no
406        // row. Otherwise whoever may edit the configuration could point it at a
407        // host they control, with their own CA as the anchor, and receive the
408        // write-only secret (D-15) in the next service bind. The guard is in
409        // the `WHERE`, so the comparison and the write are one statement.
410        let connection_guard = if sealed.is_some() {
411            ""
412        } else {
413            " AND url = $url AND start_tls = $start_tls AND bind_dn = $bind_dn \
414             AND trust_anchors_pem = $trust_anchors_pem"
415        };
416        let moved = sealed.is_none().then(|| {
417            (
418                input.url.clone(),
419                input.start_tls,
420                input.bind_dn.clone(),
421                input.trust_anchors_pem.clone(),
422            )
423        });
424
425        let tenant_id = input.tenant_id;
426        let interval =
427            i64::try_from(input.sync_interval_secs).map_err(|_| AxiamError::Validation {
428                message: "sync_interval_secs is out of range".into(),
429            })?;
430        let db = self.db.current();
431        let mut query = db
432            .query(format!(
433                "UPDATE directory_config SET {secret_sets}\
434                 enabled = $enabled, kind = $kind, url = $url, \
435                 start_tls = $start_tls, bind_dn = $bind_dn, base_dn = $base_dn, \
436                 user_filter = $user_filter, attr_username = $attr_username, \
437                 attr_email = $attr_email, attr_display_name = $attr_display_name, \
438                 attr_external_id = $attr_external_id, \
439                 group_base_dn = $group_base_dn, group_filter = $group_filter, \
440                 group_member_attribute = $group_member_attribute, \
441                 group_nesting_depth = $group_nesting_depth, \
442                 group_mappings = $group_mappings, \
443                 sync_interval_secs = $sync_interval_secs, \
444                 jit_provisioning = $jit_provisioning, \
445                 trust_anchors_pem = $trust_anchors_pem, \
446                 updated_at = time::now() \
447                 WHERE tenant_id = $tenant_id{connection_guard}"
448            ))
449            .bind(("tenant_id", tenant_id.to_string()))
450            .bind(("enabled", input.enabled))
451            .bind(("kind", input.kind.as_str().to_string()))
452            .bind(("url", input.url))
453            .bind(("start_tls", input.start_tls))
454            .bind(("bind_dn", input.bind_dn))
455            .bind(("base_dn", input.base_dn))
456            .bind(("user_filter", input.user_filter))
457            .bind(("attr_username", input.user_attribute_map.username))
458            .bind(("attr_email", input.user_attribute_map.email))
459            .bind(("attr_display_name", input.user_attribute_map.display_name))
460            .bind(("attr_external_id", input.user_attribute_map.external_id))
461            .bind(("group_base_dn", input.group_base_dn))
462            .bind(("group_filter", input.group_filter))
463            .bind(("group_member_attribute", input.group_member_attribute))
464            .bind(("group_nesting_depth", i64::from(input.group_nesting_depth)))
465            .bind(("group_mappings", mapping_rows))
466            .bind(("sync_interval_secs", interval))
467            .bind(("jit_provisioning", input.jit_provisioning))
468            .bind(("trust_anchors_pem", input.trust_anchors_pem));
469        if let Some((nonce, ciphertext)) = sealed {
470            query = query
471                .bind(("ciphertext", ciphertext))
472                .bind(("nonce", nonce))
473                .bind(("key_version", SECRET_KEY_VERSION));
474        }
475        let result = query.await.map_err(DbError::from)?;
476        result
477            .check()
478            .map_err(|e| classify_write_error(e, "directory_config"))?;
479
480        let updated = self.fetch_public(tenant_id).await?;
481        let updated = take_first_or_not_found(
482            updated.into_iter().collect(),
483            "directory_config",
484            &tenant_id.to_string(),
485        )?;
486        // The row exists; if the guard held it back, its connection still
487        // differs from the one asked for. The message names the rule and none
488        // of the values.
489        if let Some((url, start_tls, bind_dn, anchors)) = moved
490            && (updated.url != url
491                || updated.start_tls != start_tls
492                || updated.bind_dn != bind_dn
493                || updated.trust_anchors_pem != anchors)
494        {
495            return Err(AxiamError::Validation {
496                message: CONNECTION_MOVED_WITHOUT_SECRET.into(),
497            });
498        }
499        Ok(updated)
500    }

506    async fn decrypt_bind_secret(&self, tenant_id: Uuid) -> AxiamResult<Zeroizing<String>> {
507        let key = self
508            .key
509            .as_ref()
510            .ok_or_else(|| AxiamError::ServiceUnavailable(Self::missing_key_message()))?;
511
512        let mut result = self
513            .db
514            .current()
515            .query(
516                "SELECT bind_secret_ciphertext, bind_secret_nonce \
517                 FROM directory_config WHERE tenant_id = $tenant_id",
518            )
519            .bind(("tenant_id", tenant_id.to_string()))
520            .await
521            .map_err(DbError::from)?;
522        let rows: Vec<SecretRow> = result.take(0).map_err(DbError::from)?;
523        let row = take_first_or_not_found(rows, "directory_config", &tenant_id.to_string())?;
524
525        let plaintext = decrypt_separate(key, &row.bind_secret_nonce, &row.bind_secret_ciphertext)
526            .map_err(|_| {
527                // No detail on purpose: "wrong key" and "corrupt row" are both
528                // an operator problem, and neither belongs next to a secret.
529                AxiamError::Crypto(
530                    "the directory bind secret could not be decrypted: the stored value \
531                     does not match the configured directory_encryption_key"
532                        .into(),
533                )
534            })?;
535        let text = String::from_utf8(plaintext).map_err(|_| {
536            AxiamError::Crypto("the decrypted directory bind secret is not valid UTF-8".into())
537        })?;
538        Ok(Zeroizing::new(text))
539    }

556            .current()

559                 WHERE enabled = true ORDER BY created_at ASC"
560            ))
561            .await
562            .map_err(DbError::from)?;
563        let rows: Vec<DirectoryRow> = result.take(0).map_err(DbError::from)?;
564        rows.into_iter()
