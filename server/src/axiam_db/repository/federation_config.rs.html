196/// Hydrate the schema-v52 columns from a row.
197///
198/// Every absent or unparseable value resolves **towards today's behaviour**,
199/// which is also the more restrictive one: no inheritance, no PKCE beyond what
200/// the protocol forces, no issuer templating, and the per-kind default scopes.
201/// A `provider_kind` this build does not recognise reads as the generic kind of
202/// the row's own protocol rather than as an error, so a downgrade cannot make a
203/// tenant's federation configs unreadable — but it can never widen what a row
204/// permits, because the generic kinds carry no extra permission.
205fn login_provider_fields_from_columns(
206    cols: LoginProviderColumns,
207    protocol: FederationProtocol,
208) -> LoginProviderFields {
209    let (
210        kind,
211        slug,
212        inherit,
213        scopes,
214        authorization_endpoint,
215        token_endpoint,
216        userinfo_endpoint,
217        allowed_issuer_tenants,
218        apple_team_id,
219        apple_key_id,
220        require_pkce,
221        button_icon,
222    ) = cols;
223    LoginProviderFields {
224        provider_kind: kind
225            .as_deref()
226            .and_then(ProviderKind::from_wire)
227            .unwrap_or_else(|| ProviderKind::from_legacy_protocol(protocol)),
228        provider_slug: slug,
229        allow_tenant_inheritance: inherit.unwrap_or(false),
230        scopes: scopes.unwrap_or_default(),
231        authorization_endpoint,
232        token_endpoint,
233        userinfo_endpoint,
234        allowed_issuer_tenants: allowed_issuer_tenants.unwrap_or_default(),
235        apple_team_id,
236        apple_key_id,
237        require_pkce: require_pkce.unwrap_or(false),
238        button_icon,
239    }
240}

242/// The six X4 columns, as they were read back, in row order.
243type TrustColumns = (
244    Option<bool>,
245    Option<Vec<String>>,
246    Option<String>,
247    Option<String>,
248    Option<i64>,
249    Option<i64>,
250);
251
252/// Hydrate the X4 trust block from its columns.
253///
254/// Every failure mode here resolves **towards the default**, which is
255/// `enabled: false` — a row whose `scope_map_json` no longer parses, or whose
256/// `subject_mapping` holds a value this build does not know, must not become a
257/// row that trusts *more* than intended. The one thing that is never inferred
258/// is `enabled`: it is read straight from its own column, so a corrupt
259/// neighbouring column can never switch external exchange on.
260fn trust_from_columns(cols: TrustColumns) -> TokenExchangeTrust {
261    let (enabled, audiences, mapping, scope_map_json, max_age, max_lifetime) = cols;
262    let default = TokenExchangeTrust::default();
263    TokenExchangeTrust {
264        enabled: enabled.unwrap_or(false),
265        accepted_audiences: audiences.unwrap_or_default(),
266        subject_mapping: mapping
267            .as_deref()
268            .and_then(SubjectMapping::from_wire)
269            .unwrap_or(default.subject_mapping),
270        scope_map: scope_map_json
271            .as_deref()
272            .and_then(|raw| serde_json::from_str(raw).ok())
273            .unwrap_or_default(),
274        max_token_age_secs: max_age.unwrap_or(DEFAULT_MAX_TOKEN_AGE_SECS),
275        max_lifetime_secs: max_lifetime,
276    }
277}
278
279/// Serialize the scope map for storage.
280///
281/// Infallible by construction — the map is `BTreeMap<String, Vec<String>>`,
282/// which has no un-serializable inhabitant — so a failure here would be a bug
283/// in serde_json rather than bad data, and `{}` (trust nothing) is the safe
284/// answer to a bug.
285fn scope_map_json(trust: &TokenExchangeTrust) -> String {
286    serde_json::to_string(&trust.scope_map).unwrap_or_else(|_| "{}".to_string())
287}

497impl<C: Connection> SurrealFederationConfigRepository<C> {
498    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
499        let db = db.into();
500        Self { db }
501    }
502
503    /// Body of `list_all`.
504    ///
505    /// The projection is the narrowed, secret-free one (SECHRD-09) — the same
506    /// one `list()` uses. A second copy of it is how one of them ends up
507    /// hydrating the encrypted columns after a later edit.
508    async fn list_all_inner(&self, tenant_id: Uuid) -> AxiamResult<Vec<FederationConfig>> {
509        // `created_at ASC` is not cosmetic: it is the order the sign-in buttons
510        // render in, and an unordered query would reshuffle a login page
511        // between requests.
512        let sql = "SELECT meta::id(id) AS record_id, tenant_id, provider, protocol, \
513             metadata_url, client_id, attribute_map, enabled, allowed_algorithms, \
514             idp_signing_cert_pem, token_exchange_enabled, \
515             token_exchange_accepted_audiences, token_exchange_subject_mapping, \
516             token_exchange_scope_map_json, token_exchange_max_token_age_secs, \
517             token_exchange_max_lifetime_secs, provider_kind, provider_slug, \
518             allow_tenant_inheritance, scopes, authorization_endpoint, \
519             token_endpoint, userinfo_endpoint, allowed_issuer_tenants, \
520             apple_team_id, apple_key_id, require_pkce, button_icon, \
521             created_at, updated_at \
522             FROM federation_config \
523             WHERE tenant_id = $tenant_id \
524             ORDER BY created_at ASC";
525
526        let result = self
527            .db
528            .current()
529            .query(sql)
530            .bind(("tenant_id", tenant_id.to_string()))
531            .await
532            .map_err(DbError::from)?;
533
534        let mut result = result
535            .check()
536            .map_err(|e| DbError::Migration(e.to_string()))?;
537        let rows: Vec<FederationConfigListRow> = result.take(0).map_err(DbError::from)?;
538        rows.into_iter()
539            .map(|r| r.try_into_entry().map_err(Into::into))
540            .collect()
541    }
542}

544impl<C: Connection> FederationConfigRepository for SurrealFederationConfigRepository<C> {
545    async fn create(&self, input: CreateFederationConfig) -> AxiamResult<FederationConfig> {
546        let id = new_id();
547        let protocol = protocol_to_string(&input.protocol);
548        let attribute_map = input.attribute_map.unwrap_or_else(|| serde_json::json!({}));
549
550        // The kind decides the algorithm default, because "RS256" is wrong for
551        // Apple (ES256-signed client secret, RS256 ID tokens — but the field is
552        // per-kind on purpose) and meaningless for the OAuth2 variant, which has
553        // no signature at all. Omitted kind ⇒ derived from the protocol, which
554        // reproduces the previous unconditional `["RS256"]` for OIDC.
555        let provider_kind = input
556            .provider_kind
557            .unwrap_or_else(|| ProviderKind::from_legacy_protocol(input.protocol));
558        let allowed_algorithms = input
559            .allowed_algorithms
560            .unwrap_or_else(|| provider_kind.default_allowed_algorithms());
561
562        // X4: absent means "no external token exchange", which is also what
563        // `TokenExchangeTrust::default()` means. The caller (the REST handler)
564        // has already validated it; re-validating here would duplicate the
565        // rule in a place that cannot report it usefully.
566        let trust = input.token_exchange.unwrap_or_default();
567
568        let result = self
569            .db
570            .current()
571            .query(
572                "CREATE type::record('federation_config', $id) SET \
573                 tenant_id = $tenant_id, \
574                 provider = $provider, \
575                 protocol = $protocol, \
576                 metadata_url = $metadata_url, \
577                 client_id = $client_id, \
578                 client_secret = $client_secret, \
579                 attribute_map = $attribute_map, \
580                 idp_signing_cert_pem = $idp_signing_cert_pem, \
581                 allowed_algorithms = $allowed_algorithms, \
582                 token_exchange_enabled = $tx_enabled, \
583                 token_exchange_accepted_audiences = $tx_audiences, \
584                 token_exchange_subject_mapping = $tx_mapping, \
585                 token_exchange_scope_map_json = $tx_scope_map, \
586                 token_exchange_max_token_age_secs = $tx_max_age, \
587                 token_exchange_max_lifetime_secs = $tx_max_lifetime, \
588                 provider_kind = $provider_kind, \
589                 provider_slug = $provider_slug, \
590                 allow_tenant_inheritance = $allow_inherit, \
591                 scopes = $scopes, \
592                 authorization_endpoint = $authorization_endpoint, \

677    async fn update(
678        &self,
679        tenant_id: Uuid,
680        id: Uuid,
681        input: UpdateFederationConfig,
682    ) -> AxiamResult<FederationConfig> {
683        let mut set_clauses = vec!["updated_at = time::now()".to_string()];
684        let mut binds: Vec<(String, serde_json::Value)> = Vec::new();
685
686        if let Some(ref provider) = input.provider {
687            set_clauses.push("provider = $provider".into());
688            binds.push(("provider".into(), serde_json::json!(provider)));
689        }
690        if let Some(ref metadata_url) = input.metadata_url {
691            set_clauses.push("metadata_url = $metadata_url".into());
692            binds.push(("metadata_url".into(), serde_json::json!(metadata_url)));
693        }
694        if let Some(ref client_id) = input.client_id {
695            set_clauses.push("client_id = $client_id".into());
696            binds.push(("client_id".into(), serde_json::json!(client_id)));
697        }
698        if let Some(ref client_secret) = input.client_secret {
699            // TODO(T19.8): encrypt client_secret before storage (see create()).
700            set_clauses.push("client_secret = $client_secret".into());
701            binds.push(("client_secret".into(), serde_json::json!(client_secret)));
702        }
703        if let Some(ref attribute_map) = input.attribute_map {
704            set_clauses.push("attribute_map = $attribute_map".into());
705            binds.push(("attribute_map".into(), serde_json::json!(attribute_map)));
706        }
707        if let Some(enabled) = input.enabled {
708            set_clauses.push("enabled = $enabled".into());
709            binds.push(("enabled".into(), serde_json::json!(enabled)));
710        }
711        if let Some(ref idp_signing_cert_pem) = input.idp_signing_cert_pem {
712            set_clauses.push("idp_signing_cert_pem = $idp_signing_cert_pem".into());
713            binds.push((
714                "idp_signing_cert_pem".into(),
715                serde_json::json!(idp_signing_cert_pem),
716            ));
717        }
718        if let Some(ref allowed_algorithms) = input.allowed_algorithms {
719            set_clauses.push("allowed_algorithms = $allowed_algorithms".into());
720            binds.push((
721                "allowed_algorithms".into(),
722                serde_json::json!(allowed_algorithms),
723            ));
724        }
725        // Replaced wholesale, never merged field-by-field: a partial merge of
726        // a trust configuration is how an operator ends up keeping an
727        // `accepted_audiences` entry they believed they had removed.
728        if let Some(ref trust) = input.token_exchange {
729            set_clauses.push("token_exchange_enabled = $tx_enabled".into());
730            binds.push(("tx_enabled".into(), serde_json::json!(trust.enabled)));
731            set_clauses.push("token_exchange_accepted_audiences = $tx_audiences".into());
732            binds.push((
733                "tx_audiences".into(),
734                serde_json::json!(trust.accepted_audiences),
735            ));
736            set_clauses.push("token_exchange_subject_mapping = $tx_mapping".into());
737            binds.push((
738                "tx_mapping".into(),
739                serde_json::json!(trust.subject_mapping.as_str()),
740            ));
741            set_clauses.push("token_exchange_scope_map_json = $tx_scope_map".into());
742            binds.push((
743                "tx_scope_map".into(),
744                serde_json::json!(scope_map_json(trust)),
745            ));
746            set_clauses.push("token_exchange_max_token_age_secs = $tx_max_age".into());
747            binds.push((
748                "tx_max_age".into(),
749                serde_json::json!(trust.max_token_age_secs),
750            ));
751            set_clauses.push("token_exchange_max_lifetime_secs = $tx_max_lifetime".into());
752            binds.push((
753                "tx_max_lifetime".into(),
754                serde_json::json!(trust.max_lifetime_secs),
755            ));
756        }
757
758        // Login-provider columns (schema v52). `provider_kind` is deliberately
759        // absent: it selects the protocol and the override key, and changing it
760        // on a live config would silently re-point which inherited provider a
761        // tenant is shadowing.
762        if let Some(ref provider_slug) = input.provider_slug {
763            set_clauses.push("provider_slug = $provider_slug".into());
764            binds.push(("provider_slug".into(), serde_json::json!(provider_slug)));
765        }
766        if let Some(allow) = input.allow_tenant_inheritance {
767            set_clauses.push("allow_tenant_inheritance = $allow_inherit".into());
768            binds.push(("allow_inherit".into(), serde_json::json!(allow)));
769        }
770        if let Some(ref scopes) = input.scopes {
771            set_clauses.push("scopes = $scopes".into());
772            binds.push(("scopes".into(), serde_json::json!(scopes)));
773        }
774        if let Some(ref v) = input.authorization_endpoint {
775            set_clauses.push("authorization_endpoint = $authorization_endpoint".into());
776            binds.push(("authorization_endpoint".into(), serde_json::json!(v)));
777        }
778        if let Some(ref v) = input.token_endpoint {
779            set_clauses.push("token_endpoint = $token_endpoint".into());
780            binds.push(("token_endpoint".into(), serde_json::json!(v)));
781        }
782        if let Some(ref v) = input.userinfo_endpoint {
783            set_clauses.push("userinfo_endpoint = $userinfo_endpoint".into());
784            binds.push(("userinfo_endpoint".into(), serde_json::json!(v)));
785        }
786        if let Some(ref v) = input.allowed_issuer_tenants {
787            set_clauses.push("allowed_issuer_tenants = $allowed_issuer_tenants".into());
788            binds.push(("allowed_issuer_tenants".into(), serde_json::json!(v)));
789        }
790        if let Some(ref v) = input.apple_team_id {
791            set_clauses.push("apple_team_id = $apple_team_id".into());
792            binds.push(("apple_team_id".into(), serde_json::json!(v)));
793        }
794        if let Some(ref v) = input.apple_key_id {
795            set_clauses.push("apple_key_id = $apple_key_id".into());
796            binds.push(("apple_key_id".into(), serde_json::json!(v)));
797        }
798        if let Some(v) = input.require_pkce {
799            set_clauses.push("require_pkce = $require_pkce".into());
800            binds.push(("require_pkce".into(), serde_json::json!(v)));
801        }
802        if let Some(ref v) = input.button_icon {
803            set_clauses.push("button_icon = $button_icon".into());
804            binds.push(("button_icon".into(), serde_json::json!(v)));
805        }
806
807        let sql = format!(
808            "UPDATE type::record('federation_config', $id) SET {} \
809             WHERE tenant_id = $tenant_id",
810            set_clauses.join(", ")
811        );
812
813        let db = self.db.current();
814        let mut query = db.query(&sql);
815        query = query
816            .bind(("id", id.to_string()))
817            .bind(("tenant_id", tenant_id.to_string()));
818
819        for (key, val) in binds {
820            query = query.bind((key, val));
821        }
822
823        let result = query.await.map_err(DbError::from)?;
824        let mut result = result
825            .check()
826            .map_err(|e| DbError::Migration(e.to_string()))?;
827        let rows: Vec<FederationConfigRow> = result.take(0).map_err(DbError::from)?;
828        let row = take_first_or_not_found(rows, "federation_config", &id.to_string())?;
829        row.try_into_entry(id).map_err(Into::into)
830    }

865
866        let count_result = self
867            .db
868            .current()

870                "SELECT count() AS total FROM federation_config \

880
881        // Narrowed projection (SECHRD-09 / D-06): explicitly excludes
882        // client_secret / client_secret_ciphertext / client_secret_nonce /

901                 ORDER BY created_at DESC \
902                 LIMIT $limit START $offset",
903            )
904            .bind(("tenant_id", tid))
905            .bind(("limit", pagination.limit))
906            .bind(("offset", pagination.offset))
907            .await
908            .map_err(DbError::from)?;
909        let mut data_result = data_result
910            .check()
911            .map_err(|e| DbError::Migration(e.to_string()))?;
912        let rows: Vec<FederationConfigListRow> = data_result.take(0).map_err(DbError::from)?;
913
914        let items: Vec<FederationConfig> = rows
915            .into_iter()
916            .map(|r| r.try_into_entry())
917            .collect::<Result<_, _>>()?;
918
919        Ok(paginate(items, count_rows, &pagination))
920    }
