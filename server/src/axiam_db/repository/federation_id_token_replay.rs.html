1//! SurrealDB implementation of [`ProviderIdTokenReplayRepository`] (G-A,
2//! #816, D-N10).
3//!
4//! The provider-issued ID tokens a native sign-in consumed, by hash. The same
5//! mechanism as [`saml_replay`](super::saml_replay) and
6//! [`oauth2_proof_replay`](super::oauth2_proof_replay): `CREATE` against a
7//! table with a `UNIQUE` index, and the index violation **is** the "already
8//! seen" answer. No `SELECT` in the path, so two concurrent presentations of
9//! one token cannot both see "not seen".
10//!
11//! The key is `(issuer, token_hash)`, deployment-wide (D-N19, F4-W2-06): a
12//! token is spent once whichever tenant or provider configuration it is
13//! presented through. The rows carry no tenant, are not in the tenant purge,
14//! and leave with the expiry sweep.
15

25/// SurrealDB implementation of the provider ID-token replay repository.
26pub struct SurrealProviderIdTokenReplayRepository<C: Connection> {
27    db: DbHandle<C>,
28}

30// Manual Clone impl (not derive): avoids the spurious `C: Clone` bound, as
31// the other replay repositories do.
32impl<C: Connection> Clone for SurrealProviderIdTokenReplayRepository<C> {
33    fn clone(&self) -> Self {
34        Self {
35            db: self.db.clone(),
36        }
37    }
38}

40impl<C: Connection> SurrealProviderIdTokenReplayRepository<C> {
41    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
42        Self { db: db.into() }
43    }
44
45    /// One `CREATE`; the error is the engine's, unclassified.
46    async fn try_insert(
47        &self,
48        issuer: &str,
49        token_hash: &str,
50        expires_at: DateTime<Utc>,
51    ) -> Result<(), surrealdb::Error> {
52        self.db
53            .current()
54            .query(
55                "CREATE type::record('federation_id_token_replay', $row_id) SET \
56                 issuer = $issuer, \
57                 token_hash = $token_hash, \
58                 expires_at = $expires_at",
59            )
60            .bind(("row_id", new_id().to_string()))
61            .bind(("issuer", issuer.to_owned()))
62            .bind(("token_hash", token_hash.to_owned()))
63            .bind(("expires_at", expires_at))
64            .await?
65            .check()
66            .map(|_| ())
67    }
68}

70impl<C: Connection> ProviderIdTokenReplayRepository for SurrealProviderIdTokenReplayRepository<C> {
71    async fn insert_id_token(
72        &self,
73        issuer: &str,
74        token_hash: &str,
75        expires_at: DateTime<Utc>,
76    ) -> AxiamResult<()> {
77        // On the engine production runs (`surrealkv`), concurrent `CREATE`s
78        // under one UNIQUE index can abort with a retryable write conflict
79        // instead of the violation. An aborted `CREATE` wrote nothing, so it is
80        // tried again: the retry either records the token or meets the
81        // winner's row as the violation, i.e. `ReplayDetected`. Bounded, and a
82        // conflict that outlasts the bound is a `Database` error — which the
83        // caller treats as "cannot record", and refuses.
84        const ATTEMPTS: usize = 4;
85        let mut attempt = 1;
86        loop {
87            match self.try_insert(issuer, token_hash, expires_at).await {
88                Err(e) if attempt < ATTEMPTS && is_transaction_conflict(&e) => attempt += 1,
89                // The UNIQUE violation is the answer, classified by the one
90                // shared definition every replay guard uses (D-09).
91                other => return other.map_err(classify_replay_write_error),
92            }
93        }
94    }
95
96    async fn cleanup_expired(&self) -> AxiamResult<u64> {
97        cleanup_expired_rows(&self.db, "federation_id_token_replay").await
98    }
