30/// DB-side row struct that includes the record ID via `meta::id(id)`.

42impl GroupRowWithId {

44        let id = Uuid::parse_str(&self.record_id)

48        Ok(Group {
49            id,

51            name: self.name,
52            description: self.description,
53            metadata: self.metadata,
54            created_at: self.created_at,
55            updated_at: self.updated_at,

80fn parse_status(s: &str) -> Result<UserStatus, DbError> {
81    match s {
82        "Active" => Ok(UserStatus::Active),
83        "Inactive" => Ok(UserStatus::Inactive),
84        "Locked" => Ok(UserStatus::Locked),
85        "PendingVerification" => Ok(UserStatus::PendingVerification),
86        "Anonymized" => Ok(UserStatus::Anonymized),
87        "Deleted" => Ok(UserStatus::Deleted),
88        other => Err(DbError::Migration(format!("unknown user status: {other}"))),
89    }
90}

137    async fn create(&self, input: CreateGroup) -> AxiamResult<Group> {
138        let id = new_id();
139        let id_str = id.to_string();
140        let tenant_id_str = input.tenant_id.to_string();
141
142        let metadata = input
143            .metadata
144            .unwrap_or(serde_json::Value::Object(Default::default()));
145
146        let result = self
147            .db
148            .current()
149            .query(
150                "CREATE type::record('group', $id) SET \
151                 tenant_id = $tenant_id, \
152                 name = $name, description = $description, \
153                 metadata = $metadata",
154            )
155            .bind(("id", id_str.clone()))
156            .bind(("tenant_id", tenant_id_str.clone()))
157            .bind(("name", input.name))
158            .bind(("description", input.description))
159            .bind(("metadata", metadata))
160            .await
161            .map_err(DbError::from)?;
162
163        let mut result = result
164            .check()
165            .map_err(|e| DbError::Migration(e.to_string()))?;
166
167        let rows: Vec<GroupRow> = result.take(0).map_err(DbError::from)?;
168        let row = take_first_or_not_found(rows, "group", &id_str)?;
169
170        let tenant_id = Uuid::parse_str(&tenant_id_str)
171            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
172
173        Ok(Group {
174            id,
175            tenant_id,
176            name: row.name,
177            description: row.description,
178            metadata: row.metadata,
179            created_at: row.created_at,
180            updated_at: row.updated_at,
181        })
182    }

217    async fn update(&self, tenant_id: Uuid, id: Uuid, input: UpdateGroup) -> AxiamResult<Group> {
218        let id_str = id.to_string();
219        let tenant_id_str = tenant_id.to_string();
220
221        let mut sets = Vec::new();
222        if input.name.is_some() {
223            sets.push("name = $name");
224        }
225        if input.description.is_some() {
226            sets.push("description = $description");
227        }
228        if input.metadata.is_some() {
229            sets.push("metadata = $metadata");
230        }
231        sets.push("updated_at = time::now()");
232
233        let query = format!(
234            "UPDATE type::record('group', $id) SET {} \
235             WHERE tenant_id = $tenant_id",
236            sets.join(", ")
237        );
238
239        let db = self.db.current();
240        let mut builder = db
241            .query(&query)
242            .bind(("id", id_str.clone()))
243            .bind(("tenant_id", tenant_id_str));
244
245        if let Some(name) = input.name {
246            builder = builder.bind(("name", name));
247        }
248        if let Some(description) = input.description {
249            builder = builder.bind(("description", description));
250        }
251        if let Some(metadata) = input.metadata {
252            builder = builder.bind(("metadata", metadata));
253        }
254
255        let result = builder.await.map_err(DbError::from)?;
256        let mut result = result
257            .check()
258            .map_err(|e| DbError::Migration(e.to_string()))?;
259
260        let rows: Vec<GroupRow> = result.take(0).map_err(DbError::from)?;
261        let row = take_first_or_not_found(rows, "group", &id_str)?;
262
263        let tenant_id_parsed = Uuid::parse_str(&row.tenant_id)
264            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
265
266        Ok(Group {
267            id,
268            tenant_id: tenant_id_parsed,
269            name: row.name,
270            description: row.description,
271            metadata: row.metadata,
272            created_at: row.created_at,
273            updated_at: row.updated_at,
274        })
275    }

277    async fn delete(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
278        let id_str = id.to_string();
279        let tenant_id_str = tenant_id.to_string();
280
281        // Delete associated membership edges first, then the group record —
282        // inside one transaction so a concurrent reader never observes a
283        // partially-deleted group. The member_of edge carries no flat
284        // tenant_id column, so tenant scoping is a node-tenant guard on the
285        // edge's `out` endpoint (out.tenant_id), mirroring role.delete;
286        // without it a caller with a foreign-tenant group id could strip
287        // another tenant's memberships. `.check()` surfaces per-statement
288        // failures instead of swallowing them as Ok.
289        // SEC-104: the existence guard runs FIRST, inside the transaction, so
290        // a delete against a foreign or unknown id aborts with `NotFound`
291        // instead of reporting success — see `helpers::delete_existence_guard`.
292        let guard = crate::helpers::delete_existence_guard("group");
293        let query = format!(
294            "BEGIN TRANSACTION; \
295             {guard} \
296             DELETE member_of WHERE out = group:`{id_str}` AND out.tenant_id = $tenant_id; \
297             DELETE type::record('group', $id) WHERE tenant_id = $tenant_id; \
298             COMMIT TRANSACTION"
299        );
300
301        let mut result = self
302            .db
303            .current()
304            .query(query)
305            .bind(("id", id_str.clone()))
306            .bind(("tenant_id", tenant_id_str))
307            .await
308            .map_err(DbError::from)?;
309
310        crate::helpers::map_delete_errors(result.take_errors(), "group", &id_str)?;
311
312        Ok(())
313    }

315    async fn list(
316        &self,
317        tenant_id: Uuid,
318        pagination: Pagination,
319    ) -> AxiamResult<PaginatedResult<Group>> {
320        let tenant_id_str = tenant_id.to_string();
321
322        // Free-text filter, applied to BOTH queries below so the
323        // total counts matches rather than rows — a pager whose page
324        // count belongs to a different result set than the page it
325        // shows is worse than no pager. Empty when unsearched, so an
326        // unfiltered list runs exactly the query it always ran.
327        let search = search_filter(&pagination, &["name", "description"]);
328        let search_term = search_bind(&pagination);
329
330        let mut count_result = self
331            .db
332            .current()
333            .query(format!(
334                "SELECT count() AS total FROM group \
335                 WHERE tenant_id = $tenant_id{search} GROUP ALL"
336            ))
337            .bind(("tenant_id", tenant_id_str.clone()))
338            .bind(("search", search_term.clone()))
339            .await
340            .map_err(DbError::from)?;
341        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
342
343        let mut result = self
344            .db
345            .current()
346            .query(format!(
347                "SELECT meta::id(id) AS record_id, * FROM group \
348                 WHERE tenant_id = $tenant_id{search} \
349                 ORDER BY created_at ASC \
350                 LIMIT $limit START $offset"
351            ))
352            .bind(("tenant_id", tenant_id_str))
353            .bind(("search", search_term))
354            .bind(("limit", pagination.limit))
355            .bind(("offset", pagination.offset))
356            .await
357            .map_err(DbError::from)?;
358
359        let rows: Vec<GroupRowWithId> = result.take(0).map_err(DbError::from)?;
360
361        let items = rows
362            .into_iter()
363            .map(|row| row.try_into_group())
364            .collect::<Result<Vec<_>, DbError>>()?;
365
366        Ok(paginate(items, count_rows, &pagination))
367    }

369    async fn add_member(&self, tenant_id: Uuid, user_id: Uuid, group_id: Uuid) -> AxiamResult<()> {
370        let user_id_str = user_id.to_string();
371        let group_id_str = group_id.to_string();
372        let tenant_id_str = tenant_id.to_string();
373
374        // Verify both user and group belong to the same tenant.
375        let mut check = self
376            .db
377            .current()
378            .query(
379                "SELECT count() AS total FROM user \
380                 WHERE id = type::record('user', $user_id) \
381                 AND tenant_id = $tenant_id GROUP ALL; \
382                 SELECT count() AS total FROM group \
383                 WHERE id = type::record('group', $group_id) \
384                 AND tenant_id = $tenant_id GROUP ALL;",
385            )
386            .bind(("user_id", user_id_str.clone()))
387            .bind(("group_id", group_id_str.clone()))
388            .bind(("tenant_id", tenant_id_str))
389            .await
390            .map_err(DbError::from)?;
391
392        let user_count: Vec<CountRow> = check.take(0).map_err(DbError::from)?;
393        if user_count.first().map(|r| r.total).unwrap_or(0) == 0 {
394            return Err(DbError::NotFound {
395                entity: "user".into(),
396                id: user_id_str,
397            }
398            .into());
399        }
400
401        let group_count: Vec<CountRow> = check.take(1).map_err(DbError::from)?;
402        if group_count.first().map(|r| r.total).unwrap_or(0) == 0 {
403            return Err(DbError::NotFound {
404                entity: "group".into(),
405                id: group_id_str,
406            }
407            .into());
408        }
409
410        // Create the membership edge.
411        let query = format!("RELATE user:`{user_id_str}` -> member_of -> group:`{group_id_str}`;");
412
413        let result = self
414            .db
415            .current()
416            .query(query)
417            .await
418            .map_err(DbError::from)?;
419
420        // QUAL-03/D-09: a RELATE's per-statement failure (e.g. a duplicate
421        // membership violating idx_member_of_unique) only surfaces via
422        // `.check()` — without it, a duplicate add_member call silently
423        // "succeeded" while the edge was never (re-)created. classify_write_error
424        // routes a genuine duplicate to 409; anything else (e.g. a DB outage)
425        // still falls through to Migration (5xx).
426        result
427            .check()
428            .map_err(|e| classify_write_error(e.to_string(), "group_membership"))?;
429
430        Ok(())
431    }

433    async fn remove_member(
434        &self,
435        tenant_id: Uuid,
436        user_id: Uuid,
437        group_id: Uuid,
438    ) -> AxiamResult<()> {
439        let user_id_str = user_id.to_string();
440        let group_id_str = group_id.to_string();
441
442        // Tenant scoping: the member_of edge carries no flat tenant_id column,
443        // so scope the DELETE to the caller's tenant via node-tenant guards on
444        // both endpoints (in.tenant_id / out.tenant_id). Without this a caller
445        // with a foreign user+group id pair could sever another tenant's
446        // membership. `.check()` surfaces per-statement failures.
447        self.db
448            .current()
449            .query(
450                "DELETE member_of WHERE \
451                 in = type::record('user', $user_id) AND \
452                 out = type::record('group', $group_id) AND \
453                 in.tenant_id = $tenant_id AND out.tenant_id = $tenant_id",
454            )
455            .bind(("user_id", user_id_str))
456            .bind(("group_id", group_id_str))
457            .bind(("tenant_id", tenant_id.to_string()))
458            .await
459            .map_err(DbError::from)?
460            .check()
461            .map_err(|e| DbError::Migration(e.to_string()))?;
462
463        Ok(())
464    }

466    async fn get_members(
467        &self,
468        tenant_id: Uuid,
469        group_id: Uuid,
470        pagination: Pagination,
471    ) -> AxiamResult<PaginatedResult<User>> {
472        let tenant_id_str = tenant_id.to_string();
473        let group_id_str = group_id.to_string();
474
475        // Count total members.
476        let mut count_result = self
477            .db
478            .current()
479            .query(
480                "SELECT count() AS total FROM member_of \
481                 WHERE out = type::record('group', $group_id) GROUP ALL",
482            )
483            .bind(("group_id", group_id_str.clone()))
484            .await
485            .map_err(DbError::from)?;
486        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
487
488        // Fetch member users via the edge.
489        let mut result = self
490            .db
491            .current()
492            .query(
493                "SELECT meta::id(id) AS record_id, * FROM user \
494                 WHERE tenant_id = $tenant_id \
495                 AND id IN (\
496                     SELECT VALUE in FROM member_of \
497                     WHERE out = type::record('group', $group_id)\
498                 ) \
499                 ORDER BY created_at ASC \
500                 LIMIT $limit START $offset",
501            )
502            .bind(("tenant_id", tenant_id_str))
503            .bind(("group_id", group_id_str))
504            .bind(("limit", pagination.limit))
505            .bind(("offset", pagination.offset))
506            .await
507            .map_err(DbError::from)?;
508
509        let rows: Vec<MemberRow> = result.take(0).map_err(DbError::from)?;
510
511        let items = rows
512            .into_iter()
513            .map(|row| row.try_into_user())
514            .collect::<Result<Vec<_>, DbError>>()?;
515
516        Ok(paginate(items, count_rows, &pagination))
517    }
