1//! The daily cap on mailed tokens, counted inside the mint (item 7 of the
2//! 1.1.0 atomicity audit).
3//!
4//! A password reset link and a resent verification link are each capped per
5//! account per UTC day (`axiam_auth::password_reset`, `axiam_auth::verification`).
6//! The cap was a count made before the mint, in a statement of its own: N
7//! requests sent together each read a count under the cap and all minted, so
8//! the cap was exceeded and — for a reset, which supersedes the account's
9//! earlier links — more than one link was live.
10//!
11//! A capped mint is now one transaction: it counts the account's tokens of
12//! that kind minted today, `THROW`s [`MAIL_DAILY_CAP_REACHED`] at the cap, and
13//! writes `user.mail_epoch` (schema v104) before its own writes. SurrealDB
14//! 3.3.4 is snapshot-isolated with write-write conflict detection only, so the
15//! count alone would not do: N transactions that each counted under the cap
16//! and each created a different token row would all commit. The shared write
17//! on the user row is what makes two overlapping mints conflict; the loser is
18//! retried on a fresh snapshot ([`crate::helpers::retry_hot_row`]), where its
19//! count sees the winner's token. A refusal writes nothing, so it conflicts
20//! with nothing. The epoch's next value is a `LET` of its own and the `UPDATE`
21//! only assigns it, for the stack-depth reason `role::bump_statement` gives.
22
23use chrono::{DateTime, Utc};
24
25use crate::error::DbError;
26use crate::helpers::classify_write_error;
27
28/// `THROW`n by a capped mint when the account already has the day's cap of
29/// tokens of that kind.
30const MAIL_DAILY_CAP_REACHED: &str = "axiam:mail_daily_cap_reached";
31
32/// `THROW`n by a capped mint whose account is not the tenant's (or no longer
33/// exists): there is no row to claim, so nothing is minted.
34const MAIL_CAP_NO_USER: &str = "axiam:mail_cap_no_user";
35
36/// How many result slots [`claim`] fills.
37pub(crate) const CLAIM_SLOTS: usize = 6;
38
39/// The statements that open a capped mint of a `table` row (a UUID-free table
40/// name this crate chose, never caller text). They read `$tenant_id`,
41/// `$user_id`, `$day_start` and `$daily_cap` ([`day_start`]), and fill
42/// [`CLAIM_SLOTS`] result slots.
43pub(crate) fn claim(table: &str) -> String {
44    format!(
45        "LET $mail_epoch = (SELECT VALUE mail_epoch ?? 0 \
46             FROM type::record('user', $user_id) WHERE tenant_id = $tenant_id); \
47         IF array::len($mail_epoch) = 0 {{ THROW '{MAIL_CAP_NO_USER}'; }}; \
48         LET $minted = (SELECT count() AS total FROM {table} \
49             WHERE tenant_id = $tenant_id AND user_id = $user_id \
50             AND created_at >= $day_start GROUP ALL); \
51         IF ($minted[0].total ?? 0) >= $daily_cap {{ THROW '{MAIL_DAILY_CAP_REACHED}'; }}; \
52         LET $next_mail_epoch = $mail_epoch[0] + 1; \
53         UPDATE type::record('user', $user_id) SET mail_epoch = $next_mail_epoch; "
54    )
55}
56
57/// Midnight UTC today: the start of the window the cap counts, the one each
58/// repository's `count_today` counts.
59pub(crate) fn day_start() -> DateTime<Utc> {
60    Utc::now()
61        .date_naive()
62        .and_hms_opt(0, 0, 0)
63        .expect("midnight is always valid")
64        .and_utc()
65}
66
67/// A capped mint's statement errors, read. `Ok(true)`: the cap was reached.
68/// `Ok(false)`: nothing failed. `Err`: `NotFound` for an account that is not
69/// the tenant's, otherwise what is left, classified — an engine conflict is
70/// retried by the caller.
71///
72/// Every slot is read: a `THROW` fires on its own statement while the rest
73/// report "not executed due to a failed transaction", which is also a
74/// write-conflict marker — so the refusals are matched first.
75pub(crate) fn outcome(
76    errors: std::collections::HashMap<usize, surrealdb::Error>,
77    entity: &str,
78    user_id: &str,
79) -> Result<bool, DbError> {
80    if errors.is_empty() {
81        return Ok(false);
82    }
83    let combined = errors
84        .into_values()
85        .map(|e| e.to_string())
86        .collect::<Vec<_>>()
87        .join("; ");
88    if combined.contains(MAIL_DAILY_CAP_REACHED) {
89        return Ok(true);
90    }
91    if combined.contains(MAIL_CAP_NO_USER) {
92        return Err(DbError::NotFound {
93            entity: "user".into(),
94            id: user_id.to_owned(),
95        });
96    }
97    Err(classify_write_error(combined, entity))
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103
104    /// The claim's slot count is what the callers index the writes after it
105    /// by; a statement added without moving it would read the wrong slot.
106    #[test]
107    fn the_claim_fills_the_slots_it_declares() {
108        let claim = claim("password_reset_token");
109        let statements: usize = ["LET $", "IF ", "UPDATE "]
110            .iter()
111            .map(|keyword| claim.matches(keyword).count())
112            .sum();
113        assert_eq!(statements, CLAIM_SLOTS);
114    }
