1//! The minimal profile's singleton lease (T23.8.1, G-8, D-59).
2//!
3//! With `AXIAM__AMQP__ENABLED=false` there is no broker to carry a cache
4//! invalidation or an authorization decision to a second instance, so the
5//! profile is **single-instance by definition**. Nothing in a process's
6//! configuration can prove it is the only one; a row in the datastore both
7//! instances share can. This is that row: `minimal_profile_lease:instance`.
8//!
9//! * **Claim** — `CREATE` the row; if it exists, a conditional `UPDATE ... WHERE
10//!   holder = $me OR expires_at <= $now` takes it over when it has expired (or
11//!   is already ours). Both are single-record writes: of two simultaneous
12//!   claimants exactly one is acknowledged, and a write that loses the race to
13//!   the datastore's optimistic concurrency is retried and then sees the
14//!   winner's row.
15//! * **Renew** — `UPDATE ... WHERE holder = $me`. An empty result means the row
16//!   is no longer ours: another instance took it over, or it was removed.
17//! * **Release** — `DELETE ... WHERE holder = $me`, on an orderly stop, so a
18//!   successor (a rolling update's next pod) does not wait out the TTL.
19//!
20//! The caller passes `now` and the TTL, so the whole protocol is testable
21//! without sleeping, and the production constants live with the caller
22//! (`axiam-server`'s profile module). Times are the caller's clock: the
23//! protocol assumes the clocks of two instances sharing a datastore agree to
24//! well within the TTL (30 s in production), which any NTP-disciplined host
25//! does.
26

36/// The row's fixed record id.
37const LEASE_ID: &str = "instance";
38
39#[derive(Debug, SurrealValue)]
40struct LeaseRow {
41    holder: String,
42    acquired_at: DateTime<Utc>,
43    renewed_at: DateTime<Utc>,
44    expires_at: DateTime<Utc>,
45}
46
47/// What the lease row says.
48#[derive(Debug, Clone, PartialEq, Eq)]
49pub struct LeaseRecord {
50    /// The instance id that holds the lease.
51    pub holder: String,
52    /// When the current holder first took it.
53    pub acquired_at: DateTime<Utc>,
54    /// When the holder last renewed it.
55    pub renewed_at: DateTime<Utc>,
56    /// After this instant the lease may be taken over.
57    pub expires_at: DateTime<Utc>,
58}
59
60impl From<LeaseRow> for LeaseRecord {
61    fn from(row: LeaseRow) -> Self {
62        Self {
63            holder: row.holder,
64            acquired_at: row.acquired_at,
65            renewed_at: row.renewed_at,
66            expires_at: row.expires_at,
67        }
68    }
69}

71/// The result of [`SurrealMinimalProfileLeaseRepository::claim`].
72#[derive(Debug, Clone, PartialEq, Eq)]
73pub enum LeaseClaim {
74    /// The caller holds the lease now (created, taken over after expiry, or
75    /// already its own).
76    Acquired,
77    /// Another instance holds a live lease.
78    Held {
79        /// That instance's id.
80        holder: String,
81        /// When its lease expires unless renewed.
82        expires_at: DateTime<Utc>,
83    },
84}

86/// SurrealDB repository for the singleton lease. Not generic over a port: the
87/// only consumer is the composition root's profile guard.
88pub struct SurrealMinimalProfileLeaseRepository<C: Connection> {
89    db: DbHandle<C>,

100impl<C: Connection> SurrealMinimalProfileLeaseRepository<C> {
101    /// Bind to a datastore handle.
102    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
103        Self { db: db.into() }
104    }
105
106    /// The row as it stands, if there is one.
107    pub async fn current(&self) -> AxiamResult<Option<LeaseRecord>> {
108        let mut result = self
109            .db
110            .current()
111            .query("SELECT holder, acquired_at, renewed_at, expires_at FROM type::record('minimal_profile_lease', $id)")
112            .bind(("id", LEASE_ID.to_owned()))
113            .await
114            .map_err(DbError::from)?
115            .check()
116            .map_err(|e| DbError::Migration(e.to_string()))?;
117        let rows: Vec<LeaseRow> = result.take(0).map_err(DbError::from)?;
118        Ok(rows.into_iter().next().map(Into::into))
119    }
120
121    /// Try to take the lease for `holder` until `now + ttl`.
122    pub async fn claim(
123        &self,
124        holder: &str,
125        now: DateTime<Utc>,
126        ttl: Duration,
127    ) -> AxiamResult<LeaseClaim> {
128        retry_on_write_conflict(|| self.claim_once(holder, now, ttl)).await
129    }
130
131    async fn claim_once(
132        &self,
133        holder: &str,
134        now: DateTime<Utc>,
135        ttl: Duration,
136    ) -> AxiamResult<LeaseClaim> {
137        let expires_at = now + ttl;
138
139        // 1. First claim ever (or after a release): create the row. A second
140        //    simultaneous creator hits the existing record.
141        let created = self
142            .db
143            .current()
144            .query(
145                "CREATE type::record('minimal_profile_lease', $id) SET \
146                 holder = $holder, acquired_at = $now, renewed_at = $now, \
147                 expires_at = $expires_at",
148            )
149            .bind(("id", LEASE_ID.to_owned()))
150            .bind(("holder", holder.to_owned()))
151            .bind(("now", now))
152            .bind(("expires_at", expires_at))
153            .await
154            .map_err(DbError::from)?
155            .check();
156        match created {
157            Ok(_) => return Ok(LeaseClaim::Acquired),
158            Err(e) if is_unique_violation(&e.to_string()) => {}
159            Err(e) => return Err(DbError::Migration(e.to_string()).into()),
160        }
161
162        // 2. The row exists: take it over if it is ours or has expired. The
163        //    WHERE makes it a conditional write; an empty result is "not yours".
164        let mut taken = self
165            .db
166            .current()
167            .query(
168                "UPDATE type::record('minimal_profile_lease', $id) SET \
169                 holder = $holder, acquired_at = $now, renewed_at = $now, \
170                 expires_at = $expires_at \
171                 WHERE holder = $holder OR expires_at <= $now \
172                 RETURN holder, acquired_at, renewed_at, expires_at",
173            )
174            .bind(("id", LEASE_ID.to_owned()))
175            .bind(("holder", holder.to_owned()))
176            .bind(("now", now))
177            .bind(("expires_at", expires_at))
178            .await
179            .map_err(DbError::from)?
180            .check()
181            .map_err(|e| DbError::Migration(e.to_string()))?;
182        let rows: Vec<LeaseRow> = taken.take(0).map_err(DbError::from)?;
183        if !rows.is_empty() {
184            return Ok(LeaseClaim::Acquired);
185        }
186
187        // 3. Someone else holds a live lease. Report who and until when.
188        match self.current().await? {
189            Some(record) => Ok(LeaseClaim::Held {
190                holder: record.holder,
191                expires_at: record.expires_at,
192            }),
193            // Released between the two statements: the next attempt creates it.
194            None => Err(DbError::Conflict("the lease was released during the claim".into()).into()),
195        }
196    }
197
198    /// Extend `holder`'s lease to `now + ttl`. `Ok(false)` means the row is not
199    /// `holder`'s any more — taken over by another instance, or gone.
200    pub async fn renew(
201        &self,
202        holder: &str,
203        now: DateTime<Utc>,
204        ttl: Duration,
205    ) -> AxiamResult<bool> {
206        retry_on_write_conflict(|| async {
207            let mut result = self
208                .db
209                .current()
210                .query(
211                    "UPDATE type::record('minimal_profile_lease', $id) SET \
212                     renewed_at = $now, expires_at = $expires_at \
213                     WHERE holder = $holder \
214                     RETURN holder, acquired_at, renewed_at, expires_at",
215                )
216                .bind(("id", LEASE_ID.to_owned()))
217                .bind(("holder", holder.to_owned()))
218                .bind(("now", now))
219                .bind(("expires_at", now + ttl))
220                .await
221                .map_err(DbError::from)?
222                .check()
223                .map_err(|e| DbError::Migration(e.to_string()))?;
224            let rows: Vec<LeaseRow> = result.take(0).map_err(DbError::from)?;
225            Ok::<bool, axiam_core::error::AxiamError>(!rows.is_empty())
226        })
227        .await
228    }
229
230    /// Give the lease up, if `holder` still holds it. Idempotent.
231    pub async fn release(&self, holder: &str) -> AxiamResult<()> {
232        retry_on_write_conflict(|| async {
233            self.db
234                .current()
235                .query(
236                    "DELETE type::record('minimal_profile_lease', $id) \
237                     WHERE holder = $holder",
238                )
239                .bind(("id", LEASE_ID.to_owned()))
240                .bind(("holder", holder.to_owned()))
241                .await
242                .map_err(DbError::from)?
243                .check()
244                .map_err(|e| DbError::Migration(e.to_string()))?;
245            Ok::<(), axiam_core::error::AxiamError>(())
246        })
247        .await
248    }
249}
