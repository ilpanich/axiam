1//! SurrealDB implementations of [`NetworkSegmentRepository`],
2//! [`ReplyProfileRepository`] and [`NetworkPolicyRepository`] (RADIUS Live profile, W6.1,
3//! schema v100; `claude_dev/radius-reply-profile-design.md` §3.5 and §4).
4//!
5//! # Segments are resources
6//!
7//! A segment is a `resource` row in the reserved `/network` subtree plus a
8//! `network_segment` sidecar that shares its id. The two are written in **one
9//! transaction**, together with whichever of the `/network`, branch and site
10//! nodes do not exist yet. Those reserved nodes have **deterministic ids**
11//! (a v5 UUID of the tenant and the node's path), so two writers that both find
12//! a node missing both try to create the same record, and the engine's key
13//! conflict decides the race — instead of each making a node of its own and the
14//! tree growing two `/network` roots.
15//!
16//! Nothing in this module writes outside the reserved subtree, and a path that
17//! is not `/network/{zones|quarantine}/<site>/<segment>` is refused before the
18//! datastore is touched ([`NetworkSegmentPath::parse`]).
19//!
20//! # Two writers, one version
21//!
22//! Every write names the version it read and is conditional on it **inside the
23//! write's own transaction**: the check runs against the row the write replaces.
24//!
25//! A profile is also written under the **segment's** version, and the profile
26//! transaction writes the segment row (it bumps `profile_epoch`, a counter whose
27//! value is never read). That is what makes the two writers conflict: the
28//! engine detects two transactions writing the same key, not a read that a later
29//! write invalidates, so without the shared write a segment's `filter_ids` could
30//! change between the profile's envelope check and its commit (the T-416
31//! lesson, design note §3.5). For the same reason the segment's own update
32//! re-reads the stored profile inside its transaction — both for the stranding
33//! rule and for the placement rule of R110W6-03: a replace the gate cleared
34//! without `network:grant` (its clearance is
35//! [frozen](NetworkWriteClearance::freeze_profiled_placement)) may not change
36//! `vlan_id`, `rank` or `enabled` of a segment that has a profile when the
37//! transaction runs, so a profile created after the gate's read is not missed.
38//! The segment's delete re-reads it too, for the decision of 2026-10-16: a
39//! delete cleared without `network:grant` may not remove a segment that has a
40//! profile when the transaction runs. The shared write follows
41//! `resource`'s `claim_parent` (#308); the racing-writers test pins the invariant
42//! on the engine production deploys, and did not distinguish this write from a
43//! plain read of the version on that engine, so it is the precedent's
44//! belt-and-braces rather than a measured necessity.
45//!
46//! # Clamped at read
47//!
48//! [`ReplyProfileRepository`] returns the **effective** profile: its
49//! `Session-Timeout` is lowered to the tenant's *current* cap on every read
50//! (design note §3.4), so lowering the cap narrows every stored profile at once
51//! and nothing has to be rewritten.

88// The messages the write transactions `THROW`. `refusal_of` turns each into its
89// typed `NetworkConflict` on the way out, so the caller sees a typed refusal
90// rather than an opaque database error. It reads one back only in the form the
91// engine renders a `THROW` — `An error occurred: <marker>`, then the end of the
92// error or `;` — never as a substring, so a value a user can put in a field the
93// engine echoes (a segment named like a marker) is not taken for one (#730 d).
94const STALE_SEGMENT: &str = "axiam:network_segment_version_stale";
95const STALE_PROFILE: &str = "axiam:reply_profile_version_stale";
96const STRANDS_PROFILE: &str = "axiam:segment_update_strands_profile";
97// The gate cleared the replace or delete on a read that found no profile (or no
98// placement change); the row the write replaces or deletes says otherwise. A
99// read again goes back through the gate, which now asks `network:grant`. The
100// `axiam:` prefix keeps the marker from colliding with any text a user can put
101// in a field the transaction echoes.
102const PLACEMENT_RACED: &str = "axiam:profiled_placement_raced";
103const HAS_CHILDREN: &str = "axiam:network_segment_has_children";
104const SITE_GONE: &str = "axiam:network_site_gone";
105
106// The uniqueness indexes a clash names.
107const IDX_VLAN: &str = "idx_network_segment_site_vlan";
108const IDX_RANK: &str = "idx_network_segment_site_rank";
109const IDX_NAME: &str = "idx_network_segment_site_name";

115fn conflict(reason: NetworkConflict) -> TxFail {
116    TxFail::Refused(AxiamError::Network(reason))
117}
118
119/// The typed refusal a transaction's error text carries, if any: the one place
120/// the `THROW` markers (and the uniqueness indexes) written by this module are
121/// read back.
122fn refusal_of(combined: &str) -> Option<NetworkConflict> {
123    let hit = |marker: &str| thrown(combined, marker);
124    let index = violated_index(combined);
125    Some(if hit(STALE_SEGMENT) {
126        NetworkConflict::StaleSegment
127    } else if hit(STALE_PROFILE) {
128        NetworkConflict::StaleProfile
129    } else if hit(PLACEMENT_RACED) {
130        NetworkConflict::PlacementRaced
131    } else if hit(STRANDS_PROFILE) {
132        NetworkConflict::StrandsProfile {
133            segment_id: None,
134            filter_id: None,
135        }
136    } else if hit(HAS_CHILDREN) {
137        NetworkConflict::HasChildren
138    } else if hit(SITE_GONE) {
139        NetworkConflict::SiteGone
140    } else if index == Some(IDX_VLAN) {
141        NetworkConflict::DuplicateVlan
142    } else if index == Some(IDX_RANK) {
143        NetworkConflict::DuplicateRank
144    } else if index == Some(IDX_NAME) {
145        NetworkConflict::DuplicateName
146    } else {
147        return None;
148    })
149}

151/// Whether `combined` carries `marker` exactly as the engine renders a `THROW`
152/// of it: `An error occurred: <marker>`, ended by the end of the text or `;`.
153fn thrown(combined: &str, marker: &str) -> bool {
154    let head = format!("An error occurred: {marker}");
155    combined
156        .match_indices(&head)
157        .any(|(at, _)| matches!(combined[at + head.len()..].chars().next(), None | Some(';')))
158}
159
160/// The name of the unique index a clash reports, read where SurrealDB 3.x puts
161/// it: ``Database index `<name>` already contains [<values>]…``. Only the first
162/// such head counts: everything after `already contains` echoes the clashing
163/// values (site key, name), which may themselves spell an index's name
164/// (R112W4-04).
165fn violated_index(combined: &str) -> Option<&str> {
166    const HEAD: &str = "Database index `";
167    let rest = &combined[combined.find(HEAD)? + HEAD.len()..];
168    let (name, tail) = rest.split_once('`')?;
169    tail.starts_with(" already contains").then_some(name)
170}

172/// Turn a transaction's statement errors into the right answer.
173///
174/// Every statement's error is scanned rather than only the first: a `THROW`
175/// fires on its own slot while the statements after it report the generic "not
176/// executed due to a failed transaction", and `check()` can surface either.
177fn tx_outcome(errors: HashMap<usize, surrealdb::Error>, entity: &str) -> Result<(), TxFail> {
178    if errors.is_empty() {
179        return Ok(());
180    }
181    let combined = errors
182        .into_values()
183        .map(|e| e.to_string())
184        .collect::<Vec<_>>()
185        .join("; ");
186    // A uniqueness index is named only by an error `classify_write_error`
187    // reads as `AlreadyExists`; a `THROW` marker wins over it, as before.
188    match refusal_of(&combined) {
189        Some(reason) if !reason.is_uniqueness() => return Err(conflict(reason)),
190        Some(reason) => match classify_write_error(&combined, entity) {
191            DbError::AlreadyExists { .. } => return Err(conflict(reason)),
192            other => return Err(TxFail::Db(other)),
193        },
194        None => {}
195    }
196    match classify_write_error(&combined, entity) {
197        DbError::AlreadyExists { .. } if entity == PROFILE_ENTITY => {
198            Err(conflict(NetworkConflict::ProfileExists))
199        }
200        DbError::AlreadyExists { .. } => Err(TxFail::Refused(AxiamError::AlreadyExists {
201            entity: entity.into(),
202        })),
203        other => Err(TxFail::Db(other)),
204    }
205}

281    /// The segment's own row, tenant-scoped; `NotFound` otherwise.
282    async fn fetch(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<NetworkSegment> {
283        let mut result = self
284            .db
285            .current()
286            .query(
287                "SELECT * FROM type::record('network_segment', $id) \
288                 WHERE tenant_id = $tenant_id",
289            )
290            .bind(("id", id.to_string()))
291            .bind(("tenant_id", tenant_id.to_string()))
292            .await
293            .map_err(DbError::from)?;
294        let rows: Vec<SegmentRow> = result.take(0).map_err(DbError::from)?;
295        let row = take_first_or_not_found(rows, SEGMENT_ENTITY, &id.to_string())?;
296        Ok(row.into_domain()?)
297    }

299    /// One attempt at the create transaction.
300    #[allow(clippy::too_many_arguments)]
301    async fn create_tx(
302        &self,
303        input: &NewNetworkSegment,
304        path: &NetworkSegmentPath,
305        seg_id: Uuid,
306    ) -> Result<(), TxFail> {
307        let tenant = input.tenant_id;
308        let root = node_id(tenant, "/network");
309        let branch = node_id(tenant, &format!("/network/{}", path.kind.branch()));
310        let site = node_id(
311            tenant,
312            &format!("/network/{}/{}", path.kind.branch(), path.site_key),
313        );
314
315        // The three reserved nodes are made only when missing, each with its
316        // `child_of` edge; the segment's parent (the site) is then claimed, as
317        // `resource`'s own child creation does (#308), so a concurrent delete
318        // of it has a key to conflict on.
319        let query = format!(
320            "BEGIN TRANSACTION; \
321             IF array::len((SELECT VALUE id FROM type::record('resource', $root_id))) = 0 {{ \
322                 CREATE type::record('resource', $root_id) SET tenant_id = $tenant_id, \
323                     name = $root_name, resource_type = $root_type; \
324             }}; \
325             IF array::len((SELECT VALUE id FROM type::record('resource', $branch_id))) = 0 {{ \
326                 CREATE type::record('resource', $branch_id) SET tenant_id = $tenant_id, \
327                     name = $branch_name, resource_type = $branch_type, parent_id = $root_id; \
328                 RELATE resource:`{branch}` -> child_of -> resource:`{root}`; \
329             }}; \
330             IF array::len((SELECT VALUE id FROM type::record('resource', $site_id))) = 0 {{ \
331                 CREATE type::record('resource', $site_id) SET tenant_id = $tenant_id, \
332                     name = $site_key, resource_type = $site_type, parent_id = $branch_id; \
333                 RELATE resource:`{site}` -> child_of -> resource:`{branch}`; \
334             }}; \
335             LET $claimed = (UPDATE type::record('resource', $site_id) \
336                 SET child_epoch = (child_epoch ?? 0) + 1 \
337                 WHERE tenant_id = $tenant_id RETURN VALUE id); \
338             IF array::len($claimed) = 0 {{ THROW '{SITE_GONE}'; }}; \
339             CREATE type::record('resource', $seg_id) SET tenant_id = $tenant_id, \
340                 name = $name, resource_type = $segment_type, parent_id = $site_id; \
341             RELATE resource:`{seg_id}` -> child_of -> resource:`{site}`; \
342             CREATE type::record('network_segment', $seg_id) SET tenant_id = $tenant_id, \
343                 resource_id = $seg_id, name = $name, site_key = $site_key, kind = $kind, \
344                 vlan_id = $vlan_id, filter_ids = $filter_ids, rank = $rank, \
345                 enabled = $enabled, version = 1, created_by = $actor, updated_by = $actor; \
346             COMMIT TRANSACTION"
347        );
348
349        let mut result = self
350            .db
351            .current()
352            .query(query)
353            .bind(("tenant_id", tenant.to_string()))
354            .bind(("root_id", root.to_string()))
355            .bind(("branch_id", branch.to_string()))
356            .bind(("site_id", site.to_string()))
357            .bind(("seg_id", seg_id.to_string()))
358            .bind(("root_name", NETWORK_ROOT_NAME.to_owned()))
359            .bind(("root_type", RESOURCE_TYPE_NETWORK_ROOT.to_owned()))
360            .bind(("branch_name", path.kind.branch().to_owned()))
361            .bind(("branch_type", RESOURCE_TYPE_NETWORK_BRANCH.to_owned()))
362            .bind(("site_key", path.site_key.clone()))
363            .bind(("site_type", RESOURCE_TYPE_NETWORK_SITE.to_owned()))
364            .bind(("name", path.name.clone()))
365            .bind(("segment_type", RESOURCE_TYPE_NETWORK_SEGMENT.to_owned()))
366            .bind(("kind", path.kind.as_str().to_owned()))
367            .bind(("vlan_id", i64::from(input.vlan_id)))
368            .bind(("filter_ids", input.filter_ids.clone()))
369            .bind(("rank", input.rank.map(i64::from)))
370            .bind(("enabled", input.enabled))
371            .bind(("actor", input.actor.to_string()))
372            .await?;
373        tx_outcome(result.take_errors(), SEGMENT_ENTITY)
374    }

376    /// One attempt at the update transaction.
377    async fn update_tx(
378        &self,
379        tenant_id: Uuid,
380        id: Uuid,
381        update: &NetworkSegmentUpdate,
382        profiled_placement_frozen: bool,
383    ) -> Result<(), TxFail> {
384        // The profile is read inside the transaction, and the segment row is
385        // written by it: a profile writer bumps the same row (`profile_epoch`),
386        // so the two cannot both commit (module documentation). The same read
387        // backs the gate's placement rule (R110W6-03): when the gate cleared
388        // the replace without `network:grant`, a change of `vlan_id`, `rank` or
389        // `enabled` against the row replaced here is refused if a profile
390        // exists now. `??` folds NONE and NULL, the two shapes an absent rank
391        // takes between the stored row and the bound value.
392        let query = format!(
393            "BEGIN TRANSACTION; \
394             LET $current = (SELECT vlan_id, rank, enabled \
395                 FROM type::record('network_segment', $id) \
396                 WHERE tenant_id = $tenant_id AND version = $expected); \
397             IF array::len($current) = 0 {{ THROW '{STALE_SEGMENT}'; }}; \
398             LET $profiled = array::len((SELECT VALUE id FROM type::record('reply_profile', $id) \
399                 WHERE tenant_id = $tenant_id)) > 0; \
400             LET $moves = $current[0].vlan_id != $vlan_id \
401                 OR ($current[0].rank ?? -1) != ($rank ?? -1) \
402                 OR $current[0].enabled != $enabled; \
403             IF $frozen AND $profiled AND $moves {{ THROW '{PLACEMENT_RACED}'; }}; \
404             LET $used = (SELECT VALUE filter_id FROM type::record('reply_profile', $id) \
405                 WHERE tenant_id = $tenant_id); \
406             IF array::len($used) > 0 AND $used[0] != NONE \
407                 AND !($used[0] IN $filter_ids) {{ THROW '{STRANDS_PROFILE}'; }}; \
408             UPDATE type::record('network_segment', $id) SET \
409                 vlan_id = $vlan_id, filter_ids = $filter_ids, rank = $rank, \
410                 enabled = $enabled, version = version + 1, updated_by = $actor, \
411                 updated_at = time::now() \
412                 WHERE tenant_id = $tenant_id AND version = $expected; \
413             COMMIT TRANSACTION"
414        );
415        let mut result = self
416            .db
417            .current()
418            .query(query)
419            .bind(("id", id.to_string()))
420            .bind(("tenant_id", tenant_id.to_string()))
421            .bind((
422                "expected",
423                i64::try_from(update.expected_version).unwrap_or(i64::MAX),
424            ))
425            .bind(("vlan_id", i64::from(update.vlan_id)))
426            .bind(("filter_ids", update.filter_ids.clone()))
427            .bind(("rank", update.rank.map(i64::from)))
428            .bind(("enabled", update.enabled))
429            .bind(("frozen", profiled_placement_frozen))
430            .bind(("actor", update.actor.to_string()))
431            .await?;
432        tx_outcome(result.take_errors(), SEGMENT_ENTITY)
433    }

435    /// One attempt at the delete transaction.
436    async fn delete_tx(
437        &self,
438        tenant_id: Uuid,
439        id: Uuid,
440        expected_version: u64,
441        profiled_placement_frozen: bool,
442    ) -> Result<(), TxFail> {
443        // Edges are matched by the exact record id: the segment was shown to be
444        // this tenant's by `$current`, in the same transaction. The profile is
445        // read here for the decision of 2026-10-16: a delete the gate cleared
446        // without `network:grant` (it read no profile) is refused if one exists
447        // now. The profile writer bumps the segment row this deletes, so the
448        // two cannot both commit (module documentation).
449        let query = format!(
450            "BEGIN TRANSACTION; \
451             LET $current = (SELECT VALUE version FROM type::record('network_segment', $id) \
452                 WHERE tenant_id = $tenant_id AND version = $expected); \
453             IF array::len($current) = 0 {{ THROW '{STALE_SEGMENT}'; }}; \
454             LET $profiled = array::len((SELECT VALUE id FROM type::record('reply_profile', $id) \
455                 WHERE tenant_id = $tenant_id)) > 0; \
456             IF $frozen AND $profiled {{ THROW '{PLACEMENT_RACED}'; }}; \
457             LET $children = (SELECT VALUE id FROM child_of WHERE out = resource:`{id}`); \
458             IF array::len($children) > 0 {{ THROW '{HAS_CHILDREN}'; }}; \
459             DELETE reply_profile WHERE segment_id = $id AND tenant_id = $tenant_id; \
460             DELETE child_of WHERE in = resource:`{id}`; \
461             DELETE on_resource WHERE out = resource:`{id}`; \
462             DELETE scope WHERE resource_id = $id AND tenant_id = $tenant_id; \
463             DELETE type::record('resource', $id) WHERE tenant_id = $tenant_id; \
464             DELETE type::record('network_segment', $id) \
465                 WHERE tenant_id = $tenant_id AND version = $expected; \
466             COMMIT TRANSACTION"
467        );
468        let mut result = self
469            .db
470            .current()
471            .query(query)
472            .bind(("id", id.to_string()))
473            .bind(("tenant_id", tenant_id.to_string()))
474            .bind((
475                "expected",
476                i64::try_from(expected_version).unwrap_or(i64::MAX),
477            ))
478            .bind(("frozen", profiled_placement_frozen))
479            .await?;
480        tx_outcome(result.take_errors(), SEGMENT_ENTITY)
481    }

484impl<C: Connection> NetworkSegmentRepository for SurrealNetworkSegmentRepository<C> {
485    async fn create(
486        &self,
487        input: NewNetworkSegment,
488        clearance: &NetworkWriteClearance,
489    ) -> AxiamResult<NetworkSegment> {
490        // Cleared by the authorization function for exactly this write, or
491        // nothing happens (design note §5.1).
492        clearance.check(
493            input.tenant_id,
494            input.actor,
495            &NetworkWriteTarget::SegmentCreate {
496                path: input.path.clone(),
497            },
498        )?;
499        // Everything the data decides, before the datastore is touched.
500        let path = NetworkSegmentPath::parse(&input.path)?;
501        if path.kind != input.kind {
502            return Err(AxiamError::Validation {
503                message: format!(
504                    "a {} segment lives under /network/{}, not under /network/{}",
505                    input.kind.as_str(),
506                    input.kind.branch(),
507                    path.kind.branch()
508                ),
509            });
510        }
511        validate_segment_fields(path.kind, input.vlan_id, &input.filter_ids, input.rank)?;
512
513        let seg_id = new_id();
514        retry_on_write_conflict(|| self.create_tx(&input, &path, seg_id))
515            .await
516            .map_err(AxiamError::from)?;
517        self.fetch(input.tenant_id, seg_id).await
518    }
519
520    async fn get(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<NetworkSegment> {
521        self.fetch(tenant_id, id).await
522    }

679struct PolicyRow {

688    fn into_domain(self) -> Result<NetworkPolicy, DbError> {

691        let cap = u32::try_from(self.session_timeout_max_secs)
692            .unwrap_or(DEFAULT_SESSION_TIMEOUT_MAX_SECS)
693            .clamp(SESSION_TIMEOUT_FLOOR_SECS, SESSION_TIMEOUT_CAP_CEILING_SECS);
694        Ok(NetworkPolicy {

696            session_timeout_max_secs: cap,
697            version: to_u64(self.version, "version")?,
698            updated_by: Some(parse_uuid(&self.updated_by, "actor")?),
699            updated_at: Some(self.updated_at),
700        })
701    }

704/// SurrealDB implementation of the network policy repository.
705#[derive(Clone)]
706pub struct SurrealNetworkPolicyRepository<C: Connection> {
707    db: DbHandle<C>,
708}
709
710impl<C: Connection> SurrealNetworkPolicyRepository<C> {
711    /// A repository over `db`.
712    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
713        Self { db: db.into() }
714    }
715
716    async fn write_tx(&self, tenant_id: Uuid, input: &SetNetworkPolicy) -> Result<(), TxFail> {
717        let cap = i64::from(input.session_timeout_max_secs);
718        let mut result = if input.expected_version == 0 {
719            // First write: the record must not exist yet.
720            self.db
721                .current()
722                .query(
723                    "CREATE type::record('network_policy', $tenant_id) SET \
724                     tenant_id = $tenant_id, session_timeout_max_secs = $cap, version = 1, \
725                     updated_by = $actor, updated_at = time::now()",
726                )
727                .bind(("tenant_id", tenant_id.to_string()))
728                .bind(("cap", cap))
729                .bind(("actor", input.actor.to_string()))
730                .await?
731        } else {
732            self.db
733                .current()
734                .query(format!(
735                    "BEGIN TRANSACTION; \
736                     LET $written = (UPDATE type::record('network_policy', $tenant_id) SET \
737                         session_timeout_max_secs = $cap, version = version + 1, \
738                         updated_by = $actor, updated_at = time::now() \
739                         WHERE tenant_id = $tenant_id AND version = $expected \
740                         RETURN VALUE version); \
741                     IF array::len($written) = 0 {{ THROW '{STALE_SEGMENT}'; }}; \
742                     COMMIT TRANSACTION"
743                ))
744                .bind(("tenant_id", tenant_id.to_string()))
745                .bind(("cap", cap))
746                .bind(("actor", input.actor.to_string()))
747                .bind((
748                    "expected",
749                    i64::try_from(input.expected_version).unwrap_or(i64::MAX),
750                ))
751                .await?
752        };
753        match tx_outcome(result.take_errors(), POLICY_ENTITY) {
754            // A first write that finds the record there is a stale read.
755            Err(TxFail::Refused(AxiamError::AlreadyExists { .. } | AxiamError::Network(_))) => {
756                Err(conflict(NetworkConflict::StalePolicy))
757            }
758            other => other,
759        }
760    }
761}

769            .bind(("tenant_id", tenant_id.to_string()))
770            .await

772        let rows: Vec<PolicyRow> = result.take(0).map_err(DbError::from)?;
773        match rows.into_iter().next() {

775            None => Ok(NetworkPolicy::default_for(tenant_id)),

778
779    async fn set(
780        &self,
781        tenant_id: Uuid,
782        input: SetNetworkPolicy,
783        clearance: &NetworkWriteClearance,
784    ) -> AxiamResult<NetworkPolicy> {
785        // Cleared by the authorization function for exactly this write, or
786        // nothing happens (design note §5.1).
787        clearance.check(tenant_id, input.actor, &NetworkWriteTarget::PolicyWrite)?;
788        input.validate()?;
789        retry_on_write_conflict(|| self.write_tx(tenant_id, &input))
790            .await
791            .map_err(AxiamError::from)?;
792        self.get(tenant_id).await
793    }

801struct ProfileRow {
802    tenant_id: String,
803    segment_id: String,
804    dictionary: String,
805    session_timeout_secs: i64,
806    termination_action: String,
807    #[surreal(default)]
808    idle_timeout_secs: Option<i64>,
809    #[surreal(default)]
810    filter_id: Option<String>,
811    #[surreal(default)]
812    service_type: Option<String>,
813    version: i64,
814    created_by: String,
815    updated_by: String,
816    created_at: DateTime<Utc>,
817    updated_at: DateTime<Utc>,
818}
819
820impl ProfileRow {
821    /// The row as stored, for the decision path to re-validate.
822    fn into_stored(self) -> Result<StoredReplyProfile, DbError> {
823        Ok(StoredReplyProfile {
824            segment_id: parse_uuid(&self.segment_id, SEGMENT_ENTITY)?,
825            tenant_id: parse_uuid(&self.tenant_id, "tenant")?,
826            dictionary: self.dictionary,
827            session_timeout_secs: self.session_timeout_secs,
828            termination_action: self.termination_action,
829            idle_timeout_secs: self.idle_timeout_secs,
830            filter_id: self.filter_id,
831            service_type: self.service_type,
832            version: to_u64(self.version, "version")?,
833        })
834    }
835
836    /// The effective profile under `cap`: `Session-Timeout` lowered to it, and
837    /// `Idle-Timeout` kept within the result.
838    fn into_effective(self, cap: u32) -> Result<ReplyProfile, DbError> {
839        let stored = to_u32(self.session_timeout_secs, "session_timeout_secs")?;
840        let effective = clamp_session_timeout(stored, cap);
841        let idle = self
842            .idle_timeout_secs
843            .map(|i| to_u32(i, "idle_timeout_secs"))
844            .transpose()?
845            .map(|i| i.min(effective));
846        Ok(ReplyProfile {
847            segment_id: parse_uuid(&self.segment_id, SEGMENT_ENTITY)?,
848            tenant_id: parse_uuid(&self.tenant_id, "tenant")?,
849            dictionary: self.dictionary,
850            session_timeout_secs: effective,
851            session_timeout_clamped: effective != stored,
852            termination_action: TerminationAction::parse(&self.termination_action)
853                .ok_or_else(|| DbError::Serialization("unknown termination_action".into()))?,
854            idle_timeout_secs: idle,
855            filter_id: self.filter_id,
856            service_type: self
857                .service_type
858                .map(|s| {
859                    ServiceType::parse(&s)
860                        .ok_or_else(|| DbError::Serialization("unknown service_type".into()))
861                })
862                .transpose()?,
863            version: to_u64(self.version, "version")?,
864            created_by: parse_uuid(&self.created_by, "actor")?,
865            updated_by: parse_uuid(&self.updated_by, "actor")?,
866            created_at: self.created_at,
867            updated_at: self.updated_at,
868        })
869    }
870}

872/// SurrealDB implementation of the reply profile repository.
873#[derive(Clone)]
874pub struct SurrealReplyProfileRepository<C: Connection> {
875    db: DbHandle<C>,
876    segments: SurrealNetworkSegmentRepository<C>,
877    policy: SurrealNetworkPolicyRepository<C>,
878}
879
880impl<C: Connection> SurrealReplyProfileRepository<C> {
881    /// A repository over `db`.
882    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
883        let db = db.into();
884        Self {
885            segments: SurrealNetworkSegmentRepository::new(db.clone()),
886            policy: SurrealNetworkPolicyRepository::new(db.clone()),
887            db,
888        }
889    }
890
891    /// The segment the write is about, checked against the version the writer
892    /// read and the profile against that segment's envelope and the tenant cap.
893    /// Returns nothing: the transaction re-checks the version against the row it
894    /// writes, this is the value half and the early, precise answer.
895    async fn check_write(
896        &self,
897        tenant_id: Uuid,
898        segment_id: Uuid,
899        expected_segment_version: u64,
900        profile: &ReplyProfileInput,
901    ) -> AxiamResult<()> {
902        let segment = self.segments.fetch(tenant_id, segment_id).await?;
903        if segment.version != expected_segment_version {
904            return Err(AxiamError::Network(NetworkConflict::StaleSegment));
905        }
906        let cap = self.policy.get(tenant_id).await?.session_timeout_max_secs;
907        profile.validate(&segment.filter_ids, cap)
908    }
909
910    async fn fetch_stored(&self, tenant_id: Uuid, segment_id: Uuid) -> Result<ProfileRow, DbError> {
911        let mut result = self
912            .db
913            .current()
914            .query(
915                "SELECT * FROM type::record('reply_profile', $id) \
916                 WHERE tenant_id = $tenant_id",
917            )
918            .bind(("id", segment_id.to_string()))
919            .bind(("tenant_id", tenant_id.to_string()))
920            .await?;
921        let rows: Vec<ProfileRow> = result.take(0)?;
922        take_first_or_not_found(rows, PROFILE_ENTITY, &segment_id.to_string())
923    }
924
925    /// One attempt at a profile write. `expected_version` is `None` for a
926    /// create.
927    #[allow(clippy::too_many_arguments)]
928    async fn write_tx(
929        &self,
930        tenant_id: Uuid,
931        segment_id: Uuid,
932        expected_segment_version: u64,
933        expected_version: Option<u64>,
934        profile: &ReplyProfileInput,
935        actor: Uuid,
936    ) -> Result<(), TxFail> {
937        // The segment row is written (and its version checked in the same
938        // statement) so that a concurrent segment write conflicts with this one.
939        let write = if expected_version.is_some() {
940            format!(
941                "LET $written = (UPDATE type::record('reply_profile', $segment_id) SET \
942                     dictionary = $dictionary, session_timeout_secs = $session_timeout_secs, \
943                     termination_action = $termination_action, \
944                     idle_timeout_secs = $idle_timeout_secs, filter_id = $filter_id, \
945                     service_type = $service_type, version = version + 1, \
946                     updated_by = $actor, updated_at = time::now() \
947                     WHERE tenant_id = $tenant_id AND version = $expected_version \
948                     RETURN VALUE version); \
949                 IF array::len($written) = 0 {{ THROW '{STALE_PROFILE}'; }};"
950            )
951        } else {
952            "CREATE type::record('reply_profile', $segment_id) SET \
953                 tenant_id = $tenant_id, segment_id = $segment_id, dictionary = $dictionary, \
954                 session_timeout_secs = $session_timeout_secs, \
955                 termination_action = $termination_action, \
956                 idle_timeout_secs = $idle_timeout_secs, filter_id = $filter_id, \
957                 service_type = $service_type, version = 1, \
958                 created_by = $actor, updated_by = $actor;"
959                .to_owned()
960        };
961        let query = format!(
962            "BEGIN TRANSACTION; \
963             LET $claimed = (UPDATE type::record('network_segment', $segment_id) \
964                 SET profile_epoch = (profile_epoch ?? 0) + 1 \
965                 WHERE tenant_id = $tenant_id AND version = $segment_version \
966                 RETURN VALUE id); \
967             IF array::len($claimed) = 0 {{ THROW '{STALE_SEGMENT}'; }}; \
968             {write} \
969             COMMIT TRANSACTION"
970        );
971        let mut result = self
972            .db
973            .current()
974            .query(query)
975            .bind(("tenant_id", tenant_id.to_string()))
976            .bind(("segment_id", segment_id.to_string()))
977            .bind((
978                "segment_version",
979                i64::try_from(expected_segment_version).unwrap_or(i64::MAX),
980            ))
981            .bind((
982                "expected_version",
983                expected_version.map(|v| i64::try_from(v).unwrap_or(i64::MAX)),
984            ))
985            .bind(("dictionary", profile.dictionary.clone()))
986            .bind((
987                "session_timeout_secs",
988                i64::from(profile.session_timeout_secs),
989            ))
990            .bind((
991                "termination_action",
992                profile.termination_action.as_str().to_owned(),
993            ))
994            .bind((
995                "idle_timeout_secs",
996                profile.idle_timeout_secs.map(i64::from),
997            ))
998            .bind(("filter_id", profile.filter_id.clone()))
999            .bind((
1000                "service_type",
1001                profile.service_type.map(|s| s.as_str().to_owned()),
1002            ))
1003            .bind(("actor", actor.to_string()))
1004            .await?;
1005        tx_outcome(result.take_errors(), PROFILE_ENTITY)
1006    }
1007
1008    async fn effective(&self, tenant_id: Uuid, segment_id: Uuid) -> AxiamResult<ReplyProfile> {
1009        let row = self.fetch_stored(tenant_id, segment_id).await?;
1010        let cap = self.policy.get(tenant_id).await?.session_timeout_max_secs;
1011        Ok(row.into_effective(cap)?)
1012    }
1013}

1128            .collect::<AxiamResult<Vec<_>>>()?;
1129        Ok(paginate(items, count_rows, &pagination))
1130    }

1132    async fn replace(
1133        &self,
1134        tenant_id: Uuid,
1135        segment_id: Uuid,
1136        input: ReplyProfileReplace,
1137        clearance: &NetworkWriteClearance,
1138    ) -> AxiamResult<ReplyProfile> {
1139        clearance.check(
1140            tenant_id,
1141            input.actor,
1142            &NetworkWriteTarget::ProfileWrite { segment_id },
1143        )?;
1144        // `NotFound` for a segment without a profile, before anything else.
1145        self.fetch_stored(tenant_id, segment_id).await?;
1146        self.check_write(
1147            tenant_id,
1148            segment_id,
1149            input.expected_segment_version,
1150            &input.profile,
1151        )
1152        .await?;
1153        retry_on_write_conflict(|| {
1154            self.write_tx(
1155                tenant_id,
1156                segment_id,
1157                input.expected_segment_version,
1158                Some(input.expected_version),
1159                &input.profile,
1160                input.actor,
1161            )
1162        })
1163        .await
1164        .map_err(AxiamError::from)?;
1165        self.effective(tenant_id, segment_id).await
1166    }

1168    async fn delete(
1169        &self,
1170        tenant_id: Uuid,
1171        segment_id: Uuid,
1172        expected_version: u64,
1173        clearance: &NetworkWriteClearance,
1174    ) -> AxiamResult<()> {
1175        clearance.check(
1176            tenant_id,
1177            clearance.actor(),
1178            &NetworkWriteTarget::ProfileDelete { segment_id },
1179        )?;
1180        self.fetch_stored(tenant_id, segment_id).await?;
1181        retry_on_write_conflict(|| async {
1182            let mut result = self
1183                .db
1184                .current()
1185                .query(format!(
1186                    "BEGIN TRANSACTION; \
1187                     LET $deleted = (DELETE type::record('reply_profile', $segment_id) \
1188                         WHERE tenant_id = $tenant_id AND version = $expected \
1189                         RETURN BEFORE); \
1190                     IF array::len($deleted) = 0 {{ THROW '{STALE_PROFILE}'; }}; \
1191                     COMMIT TRANSACTION"
1192                ))
1193                .bind(("tenant_id", tenant_id.to_string()))
1194                .bind(("segment_id", segment_id.to_string()))
1195                .bind((
1196                    "expected",
1197                    i64::try_from(expected_version).unwrap_or(i64::MAX),
1198                ))
1199                .await?;
1200            tx_outcome(result.take_errors(), PROFILE_ENTITY)
1201        })
1202        .await
1203        .map_err(AxiamError::from)
1204    }
