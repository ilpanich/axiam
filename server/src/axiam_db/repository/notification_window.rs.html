1//! SurrealDB implementation of [`NotificationWindowRepository`] (#551, T-117,
2//! schema v85's `notification_window`).
3//!
4//! One row per `(tenant, rule, event)`, its record id derived from the triple,
5//! so the first claims of a window on two replicas write the same record and
6//! the datastore orders them. A claim is one `UPSERT` whose every assignment is
7//! conditional on the window having expired — the precondition of the write,
8//! as `claim_failure_notification` (D-73) makes the hour its precondition — and
9//! it is retried on a write conflict: of two concurrent claimants one commits
10//! and the other re-reads the winner's window.
11//!
12//! Retried a few times, not with a hot row's patience (R1W2-01). A claim that
13//! still conflicts after [`crate::helpers::MAX_WRITE_ATTEMPTS`] is
14//! [`NotificationWindowClaim::Contended`]: another claimant has just written
15//! the record, so a window is open, and the caller counts the event in memory
16//! and hands the count back through `add_uncounted`. Waiting out every
17//! replica's claims on one record is what serialized the fleet.
18//!
19//! The row holds ids, an event name and two counts: no address, no request.
20

33const ENTITY: &str = "notification_window";

35#[derive(SurrealValue)]
36struct WindowRow {
37    opened_by: String,
38    carried: i64,
39    opened_at: DateTime<Utc>,
40}

62impl<C: Connection> SurrealNotificationWindowRepository<C> {
63    /// Build the repository.
64    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
65        Self { db: db.into() }
66    }
67
68    /// The record id of a `(tenant, rule, event)` triple. Event names are
69    /// `snake_case` identifiers, so the id needs no escaping beyond what
70    /// `type::record` does.
71    fn record_id(tenant_id: Uuid, rule_id: Uuid, event: &str) -> String {
72        format!("{}_{}_{event}", tenant_id.simple(), rule_id.simple())
73    }
74}
75
76impl<C: Connection> NotificationWindowRepository for SurrealNotificationWindowRepository<C> {
77    async fn claim(
78        &self,
79        tenant_id: Uuid,
80        rule_id: Uuid,
81        event: &str,
82        now: DateTime<Utc>,
83        window_secs: i64,
84    ) -> AxiamResult<NotificationWindowClaim> {
85        let cutoff = now - Duration::seconds(window_secs);
86        let record_id = Self::record_id(tenant_id, rule_id, event);
87        // Every assignment reads the row as it was before this statement:
88        // `opened_at` is written last, and each condition reads it, so the
89        // statement means the same whether SET is applied at once or in order.
90        // An expired (or absent) window is reopened by this claimant — its
91        // `opened_by` — with the old count carried for the mail; an open one
92        // counts the event and keeps its claimant. The claimant compares the
93        // `opened_by` it gets back with its own id.
94        let claimant = Uuid::new_v4().to_string();
95        let claimed = retry_on_write_conflict(|| async {
96            let mut result = self
97                .db
98                .current()
99                .query(
100                    "UPSERT type::record('notification_window', $id) SET \
101                       tenant_id = $tenant_id, rule_id = $rule_id, event = $event, \
102                       carried = IF opened_at = NONE OR opened_at <= $cutoff \
103                         THEN suppressed ?? 0 ELSE carried END, \
104                       suppressed = IF opened_at = NONE OR opened_at <= $cutoff \
105                         THEN 0 ELSE suppressed + 1 END, \
106                       opened_by = IF opened_at = NONE OR opened_at <= $cutoff \
107                         THEN $claimant ELSE opened_by END, \
108                       opened_at = IF opened_at = NONE OR opened_at <= $cutoff \
109                         THEN $now ELSE opened_at END \
110                     RETURN opened_by, carried, opened_at",
111                )
112                .bind(("id", record_id.clone()))
113                .bind(("tenant_id", tenant_id.to_string()))
114                .bind(("rule_id", rule_id.to_string()))
115                .bind(("event", event.to_string()))
116                .bind(("claimant", claimant.clone()))
117                .bind(("now", now))
118                .bind(("cutoff", cutoff))
119                .await
120                .map_err(DbError::from)?
121                .check()
122                .map_err(|e| classify_write_error(e, ENTITY))?;
123            let rows: Vec<WindowRow> = result.take(0).map_err(DbError::from)?;
124            rows.into_iter().next().ok_or_else(|| {
125                DbError::Migration("notification window claim returned no row".into())
126            })
127        })
128        .await;
129        let row = match claimed {
130            Ok(row) => row,
131            // Every attempt lost to another claimant's write: a window is open.
132            Err(e) if is_write_conflict(&e.to_string()) => {
133                return Ok(NotificationWindowClaim::Contended);
134            }
135            Err(e) => return Err(e.into()),
136        };
137
138        Ok(if row.opened_by == claimant {
139            NotificationWindowClaim::Opened {
140                suppressed: u64::try_from(row.carried).unwrap_or(0),
141            }
142        } else {
143            NotificationWindowClaim::Counted {
144                open_until: row.opened_at + Duration::seconds(window_secs),
145            }
146        })
147    }
148
149    async fn add_uncounted(
150        &self,
151        tenant_id: Uuid,
152        rule_id: Uuid,
153        event: &str,
154        count: u64,
155    ) -> AxiamResult<()> {
156        if count == 0 {
157            return Ok(());
158        }
159        let record_id = Self::record_id(tenant_id, rule_id, event);
160        let count = i64::try_from(count).unwrap_or(i64::MAX);
161        // `UPDATE` of one record id creates nothing: a window whose rule was
162        // deleted (and its rows with it) takes no count. Whether the window is
163        // still the one the events were counted in or a later one, the count
164        // reaches the next mail that opens a window.
165        retry_on_write_conflict(|| async {
166            self.db
167                .current()
168                .query(
169                    "UPDATE type::record('notification_window', $id) \
170                     SET suppressed = (suppressed ?? 0) + $count RETURN NONE",
171                )
172                .bind(("id", record_id.clone()))
173                .bind(("count", count))
174                .await
175                .map_err(DbError::from)?
176                .check()
177                .map_err(|e| classify_write_error(e, ENTITY))?;
178            Ok::<_, DbError>(())
179        })
180        .await?;
181        Ok(())
182    }
183}
