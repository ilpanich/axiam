1//! SurrealDB implementation of [`AuthorizationCodeRepository`].
2//!
3//! # Why `consume` layers a transaction under a nonce
4//!
5//! An authorization code is single-use: redeeming it mints one token pair, and
6//! a code redeemed twice is two token pairs from one authorization. The
7//! precondition — unused, unexpired, right tenant, right client, right
8//! `redirect_uri` — lives in the `WHERE` clause of the redeeming `UPDATE`,
9//! which is what makes a later, non-concurrent replay match nothing.
10//!
11//! Single-use is a **guarantee, conditional on an attested persistent storage
12//! engine**, carried by the same two layers as `permission_ticket.consume`,
13//! `device_grant.redeem` and `pushed_auth_request.consume` (#302, T-164):
14//!
15//!   1. The guarded `UPDATE` runs inside an explicit `BEGIN`/`COMMIT`, so two
16//!      concurrent redemptions conflict on one key and the deployed engine
17//!      aborts the loser. This path had that layer implicitly before — its
18//!      redemption was a single statement, which runs in the engine's own
19//!      transaction — and writing it out changes nothing about the
20//!      arbitration. It is written out anyway so all four consumes read the
21//!      same and none of them depends on a reader knowing that a lone
22//!      statement is atomic.
23//!   2. A per-attempt nonce is read back **after** the commit, in a query of
24//!      its own; only the caller whose `redemption_id` survived reports a
25//!      redemption. This is what this path did not have before v37. It asks
26//!      the engine for nothing, so it catches a conflict the engine silently
27//!      missed — and conflict detection is not a documented SurrealDB
28//!      guarantee, so a version bump could take layer 1 away without notice.
29//!
30//! The read-back **must** stay outside the transaction. Inside one, snapshot
31//! isolation shows every racer its own write and every racer believes it won,
32//! turning an occasional double redemption into a certain one — see
33//! `SCHEMA_V31`, and `SCHEMA_V37` for why this path joined them.
34//!
35//! Losing either race answers exactly as an unknown code does: `NotFound`. A
36//! caller that could distinguish "someone else just redeemed this" from "no
37//! such code" could probe for live codes, and the token endpoint answers
38//! `invalid_grant` for both regardless.

54/// Projected explicitly rather than `SELECT *` because the redeeming statement
55/// now also writes `redemption_id`, which [`AuthCodeRowWithId`] does not carry
56/// — naming the columns keeps the row type and the query in step instead of
57/// relying on how the deserializer treats an unexpected key.
58const CONSUME_FIELDS: &str = "meta::id(id) AS record_id, tenant_id, client_id, user_id, \
59     code_hash, redirect_uri, scopes, code_challenge, code_challenge_method, nonce, \
60     session_id, expires_at, used, created_at";

62/// Parse an optional stored UUID.
63///
64/// Empty string and absent both mean "no session": SurrealDB writes `NONE` as
65/// absent, but a row migrated in from an earlier shape can carry `""`, and
66/// treating that as a parse failure would make an old authorization code
67/// unreadable rather than merely session-less.
68fn parse_opt_uuid(raw: Option<&str>) -> Result<Option<Uuid>, DbError> {
69    match raw {
70        None | Some("") => Ok(None),
71        Some(v) => Uuid::parse_str(v)
72            .map(Some)
73            .map_err(|e| DbError::Migration(format!("invalid session UUID: {e}"))),
74    }
75}

114impl AuthCodeRowWithId {
115    fn try_into_auth_code(self) -> Result<AuthorizationCode, DbError> {
116        let id = Uuid::parse_str(&self.record_id)
117            .map_err(|e| DbError::Migration(format!("invalid UUID: {e}")))?;
118        let tenant_id = Uuid::parse_str(&self.tenant_id)
119            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
120        let user_id = Uuid::parse_str(&self.user_id)
121            .map_err(|e| DbError::Migration(format!("invalid user UUID: {e}")))?;
122        Ok(AuthorizationCode {
123            id,
124            tenant_id,
125            client_id: self.client_id,
126            user_id,
127            code_hash: self.code_hash,
128            redirect_uri: self.redirect_uri,
129            scopes: self.scopes,
130            code_challenge: self.code_challenge,
131            code_challenge_method: self.code_challenge_method,
132            nonce: self.nonce,
133            session_id: parse_opt_uuid(self.session_id.as_deref())?,
134            expires_at: self.expires_at,
135            used: self.used,
136            created_at: self.created_at,
137        })
138    }
139}

154impl<C: Connection> AuthorizationCodeRepository for SurrealAuthorizationCodeRepository<C> {
155    async fn create(&self, input: CreateAuthorizationCode) -> AxiamResult<AuthorizationCode> {
156        let id = new_id();
157        let id_str = id.to_string();
158
159        let result = self
160            .db
161            .current()
162            .query(
163                "CREATE type::record('oauth2_auth_code', $id) SET \
164                 tenant_id = $tenant_id, \
165                 client_id = $client_id, \
166                 user_id = $user_id, \
167                 code_hash = $code_hash, \
168                 redirect_uri = $redirect_uri, \
169                 scopes = $scopes, \
170                 code_challenge = $code_challenge, \
171                 code_challenge_method = $code_challenge_method, \
172                 nonce = $nonce, \
173                 session_id = $session_id, \
174                 expires_at = $expires_at, \
175                 used = false",
176            )
177            .bind(("id", id_str.clone()))
178            .bind(("tenant_id", input.tenant_id.to_string()))
179            .bind(("client_id", input.client_id))
180            .bind(("user_id", input.user_id.to_string()))
181            .bind(("code_hash", input.code_hash))
182            .bind(("redirect_uri", input.redirect_uri))
183            .bind(("scopes", input.scopes))
184            .bind(("code_challenge", input.code_challenge))
185            .bind(("code_challenge_method", input.code_challenge_method))
186            .bind(("nonce", input.nonce))
187            .bind(("session_id", input.session_id.map(|id| id.to_string())))
188            .bind(("expires_at", input.expires_at))
189            .await
190            .map_err(DbError::from)?;
191
192        let mut result = result
193            .check()
194            .map_err(|e| DbError::Migration(e.to_string()))?;
195
196        let rows: Vec<AuthCodeRow> = result.take(0).map_err(DbError::from)?;
197        let row = take_first_or_not_found(rows, "oauth2_auth_code", &id_str)?;
198
199        let tenant_id = Uuid::parse_str(&row.tenant_id)
200            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
201        let user_id = Uuid::parse_str(&row.user_id)
202            .map_err(|e| DbError::Migration(format!("invalid user UUID: {e}")))?;
203
204        Ok(AuthorizationCode {
205            id,
206            tenant_id,
207            client_id: row.client_id,
208            user_id,
209            code_hash: row.code_hash,
210            redirect_uri: row.redirect_uri,
211            scopes: row.scopes,
212            code_challenge: row.code_challenge,
213            code_challenge_method: row.code_challenge_method,
214            nonce: row.nonce,
215            session_id: parse_opt_uuid(row.session_id.as_deref())?,
216            expires_at: row.expires_at,
217            used: row.used,
218            created_at: row.created_at,
219        })
220    }

222    async fn get_by_hash(
223        &self,
224        tenant_id: Uuid,
225        code_hash: &str,
226        client_id: &str,
227        redirect_uri: &str,
228    ) -> AxiamResult<AuthorizationCode> {
229        let code_hash_owned = code_hash.to_string();
230        let tenant_id_str = tenant_id.to_string();
231
232        let result = self
233            .db
234            .current()
235            .query(
236                "SELECT meta::id(id) AS record_id, * \
237                 FROM oauth2_auth_code \
238                 WHERE tenant_id = $tenant_id \
239                   AND code_hash = $code_hash \
240                   AND client_id = $client_id \
241                   AND redirect_uri = $redirect_uri \
242                   AND used = false \
243                   AND expires_at > time::now()",
244            )
245            .bind(("tenant_id", tenant_id_str))
246            .bind(("code_hash", code_hash_owned.clone()))
247            .bind(("client_id", client_id.to_string()))
248            .bind(("redirect_uri", redirect_uri.to_string()))
249            .await
250            .map_err(DbError::from)?;
251
252        let mut result = result
253            .check()
254            .map_err(|e| DbError::Migration(e.to_string()))?;
255
256        let rows: Vec<AuthCodeRowWithId> = result.take(0).map_err(DbError::from)?;
257        let row = take_first_or_not_found(
258            rows,
259            "oauth2_auth_code",
260            &format!("code_hash={code_hash_owned}"),
261        )?;
262
263        row.try_into_auth_code().map_err(Into::into)
264    }

266    async fn consume(
267        &self,
268        tenant_id: Uuid,
269        code_hash: &str,
270        client_id: &str,
271        redirect_uri: &str,
272    ) -> AxiamResult<AuthorizationCode> {
273        let code_hash_owned = code_hash.to_string();
274        let tenant_id_str = tenant_id.to_string();
275
276        // Two layers, in two queries (#302 / T-164 — see the module header).
277        //
278        // Layer 1, here: the guarded `UPDATE` inside an explicit transaction.
279        // `used = false` is what makes a later, non-concurrent replay match
280        // nothing; `client_id` and `redirect_uri` are matched inside the
281        // statement rather than checked on the returned row, so a redemption
282        // attempt with the wrong pair changes nothing and cannot burn a code
283        // its holder is entitled to.
284        //
285        // Layer 2, in the SEPARATE query below: the per-attempt nonce read
286        // back after this transaction has committed. It must not be folded in
287        // here — under snapshot isolation every racer would see its own write
288        // and believe it won (`SCHEMA_V31`, `SCHEMA_V37`).
289        let redemption = new_id().to_string();
290        let result = self
291            .db
292            .current()
293            .query(format!(
294                "BEGIN TRANSACTION; \
295                 LET $claimed = (UPDATE oauth2_auth_code \
296                     SET used = true, redemption_id = $redemption \
297                  WHERE tenant_id = $tenant_id \
298                    AND code_hash = $code_hash \
299                    AND client_id = $client_id \
300                    AND redirect_uri = $redirect_uri \
301                    AND used = false \
302                    AND expires_at > time::now()); \
303                 SELECT {CONSUME_FIELDS} FROM $claimed; \
304                 COMMIT TRANSACTION"
305            ))
306            .bind(("tenant_id", tenant_id_str.clone()))
307            .bind(("code_hash", code_hash_owned.clone()))
308            .bind(("client_id", client_id.to_string()))
309            .bind(("redirect_uri", redirect_uri.to_string()))
310            .bind(("redemption", redemption.clone()))
311            .await;
312
313        // A loser's abort is "someone else redeemed first", which is this
314        // method's `NotFound` answer rather than a server fault; propagating
315        // it would turn a correctly-refused replay into a 500.
316        let not_found = || {
317            DbError::NotFound {
318                entity: "oauth2_auth_code".to_string(),
319                id: format!("code_hash={code_hash_owned}"),
320            }
321            .into()
322        };
323
324        let mut result = match result {
325            Ok(r) => r,
326            Err(e) if is_transaction_conflict(&e) => return Err(not_found()),
327            Err(e) => return Err(DbError::from(e).into()),
328        };
329
330        // BEGIN=0, LET=1, SELECT=2, COMMIT=3.
331        let rows: Vec<AuthCodeRowWithId> = match result.take::<Vec<AuthCodeRowWithId>>(2) {
332            Ok(rows) => rows,
333            Err(e) if is_transaction_conflict(&e) => return Err(not_found()),
334            Err(e) => return Err(DbError::from(e).into()),
335        };
336        let row = take_first_or_not_found(
337            rows,
338            "oauth2_auth_code",
339            &format!("code_hash={code_hash_owned}"),
340        )?;
341
342        // Layer 2: outside, and after, the transaction above.
343        let stored = self
344            .db
345            .current()
346            .query(
347                "SELECT VALUE redemption_id FROM oauth2_auth_code \
348                 WHERE tenant_id = $tenant_id AND code_hash = $code_hash LIMIT 1",
349            )
350            .bind(("tenant_id", tenant_id_str))
351            .bind(("code_hash", code_hash_owned.clone()))
352            .await;
353        let mut stored = match stored {
354            Ok(r) => r,
355            Err(e) if is_transaction_conflict(&e) => return Err(not_found()),
356            Err(e) => return Err(DbError::from(e).into()),
357        };
358        let stored: Vec<Option<String>> = match stored.take::<Vec<Option<String>>>(0) {
359            Ok(v) => v,
360            Err(e) if is_transaction_conflict(&e) => return Err(not_found()),
361            Err(e) => return Err(DbError::from(e).into()),
362        };
363        if stored.into_iter().flatten().next().as_deref() != Some(redemption.as_str()) {
364            // Our write landed but another redemption's landed after it. That
365            // caller holds the code; this one must not also mint a token pair.
366            // The code is spent either way, so `used` staying true is correct.
367            return Err(not_found());
368        }
369
370        row.try_into_auth_code().map_err(Into::into)
371    }

373    async fn delete_expired(&self) -> AxiamResult<u64> {
374        let mut result = self
375            .db
376            .current()
377            .query(
378                "SELECT count() AS total FROM oauth2_auth_code \
379                 WHERE expires_at < time::now() OR used = true GROUP ALL",
380            )
381            .await
382            .map_err(DbError::from)?;
383
384        let count_rows: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
385        let count = count_rows.first().map(|r| r.total).unwrap_or(0);
386
387        self.db
388            .current()
389            .query(
390                "DELETE FROM oauth2_auth_code \
391                 WHERE expires_at < time::now() OR used = true",
392            )
393            .await
394            .map_err(DbError::from)?;
395
396        Ok(count)
397    }
