25/// Generate a random client ID with the `oa_` prefix (32 hex chars).
26fn generate_client_id() -> String {
27    let mut rng = rand::rng();
28    let bytes: [u8; 16] = rng.random();
29    format!("oa_{}", hex::encode(bytes))
30}

32/// Collapse a blank string to `None` before it is stored.
33///
34/// An empty or whitespace-only `tls_client_auth_*` value must never reach the
35/// matcher: `""` is not a DN anybody holds, but a matcher that compared it
36/// literally would authenticate any certificate whose corresponding field is
37/// also absent. Storing `None` makes the "registered nothing" case
38/// unambiguous, and `mtls_client_auth` refuses to authenticate a client with
39/// no registered expectation at all.
40fn normalise_optional(value: Option<String>) -> Option<String> {
41    value.map(|v| v.trim().to_owned()).filter(|v| !v.is_empty())
42}

48    hex::encode(bytes)

199/// Decode the stored `profile` string.
200///
201/// An **absent** value is a pre-v38 row and correctly reads as `Standard`. An
202/// **unrecognised** value is not: it means this binary is older than the row
203/// it is reading, and guessing `Standard` there would silently strip a client
204/// of the constraint bundle it was registered under — a downgrade performed by
205/// a rollback. That is refused, loudly, as a migration error.
206fn decode_profile(raw: Option<&str>) -> Result<ClientProfile, DbError> {
207    match raw {
208        None => Ok(ClientProfile::default()),
209        Some(s) => ClientProfile::from_wire(s).ok_or_else(|| {
210            DbError::Migration(format!(
211                "oauth2_client.profile holds an unrecognised value {s:?}; this binary cannot \
212                 safely serve a client registered under a profile it does not implement"
213            ))
214        }),
215    }
216}

218/// Decode the stored `token_endpoint_auth_method`. Fails closed for the same
219/// reason [`decode_profile`] does — resolving an unknown method to
220/// `client_secret_post` would let a certificate-authenticated client be

222fn decode_auth_method(raw: Option<&str>) -> Result<ClientAuthMethod, DbError> {

227                "oauth2_client.token_endpoint_auth_method holds an unrecognised value {s:?}; \

229            ))
230        }),
231    }
232}

234/// Decode the stored `authn_request_params` mode.
235///
236/// Fails closed on an unrecognised value for the same reason
237/// [`decode_profile`] does, with the argument running the other way round: a
238/// value this binary does not implement must not resolve to `Honour`, which
239/// would act on authentication-request parameters the operator never opted
240/// into, nor be quietly downgraded, which would hide a rollback. An **absent**
241/// value is a pre-v54 row and correctly reads as `Ignore`.
242fn decode_authn_request_params(raw: Option<&str>) -> Result<AuthnRequestParamsMode, DbError> {
243    match raw {
244        None => Ok(AuthnRequestParamsMode::default()),
245        Some(s) => AuthnRequestParamsMode::from_wire(s).ok_or_else(|| {
246            DbError::Migration(format!(
247                "oauth2_client.authn_request_params holds an unrecognised value {s:?}; this \
248                 binary cannot serve a client under a parameter policy it does not implement"
249            ))
250        }),
251    }
252}

254/// Decode the stored `managed_by` discriminator (T21.4 / D5).
255///
256/// An **absent** value is a pre-v64 row and correctly reads as `Admin`: every
257/// client that existed before this migration was created by an administrator
258/// through `POST /oauth2-clients`, so that is the fact rather than a guess.
259///
260/// An **unrecognised** value fails closed, and this is the decoder where that
261/// matters most. `Admin` is the permissive answer in three independent places
262/// — it is the only provenance that may carry a FAPI profile (I5), the only
263/// one exempt from the forced consent hop (D4), and the only one the sweeper
264/// will not touch — so a rollback reading a `cimd` row it does not implement
265/// must refuse it rather than promote it to an administrator's client.
266fn decode_managed_by(raw: Option<&str>) -> Result<ManagedBy, DbError> {
267    match raw {
268        None => Ok(ManagedBy::default()),
269        Some(s) => ManagedBy::from_wire(s).ok_or_else(|| {
270            DbError::Migration(format!(
271                "oauth2_client.managed_by holds an unrecognised value {s:?}; this binary \
272                 cannot serve a client whose provenance it does not implement, because \
273                 every gate that reads this field treats 'admin' as the trusted answer"
274            ))
275        }),
276    }
277}

279/// Decode the stored CIBA metadata (G-7).
280///
281/// An absent mode is a client without the grant. An unrecognised one fails
282/// closed, for the reason [`decode_profile`] gives: a binary older than the row
283/// must not guess how a client it cannot serve wants to be told.
284fn decode_ciba(
285    mode: Option<&str>,
286    endpoint: Option<String>,
287    signing_alg: Option<&str>,
288) -> Result<CibaClientMetadata, DbError> {
289    let backchannel_token_delivery_mode = match mode {
290        None => None,
291        Some(raw) => Some(CibaDeliveryMode::from_wire(raw).ok_or_else(|| {
292            DbError::Migration(format!(
293                "oauth2_client.backchannel_token_delivery_mode holds an unrecognised value \
294                 {raw:?}; this binary cannot serve a CIBA client in a mode it does not implement"
295            ))
296        })?),
297    };
298    // Fails closed for the same reason: a row naming an algorithm this binary
299    // does not verify must not be served as if it named none, which would
300    // quietly accept the unsigned requests the client registered against.
301    let backchannel_authentication_request_signing_alg = match signing_alg {
302        None => None,
303        Some(raw) => Some(CibaRequestSigningAlg::from_wire(raw).ok_or_else(|| {
304            DbError::Migration(format!(
305                "oauth2_client.backchannel_authentication_request_signing_alg holds an \
306                 unrecognised value {raw:?}; this binary cannot verify a CIBA request signed \
307                 with an algorithm it does not implement"
308            ))
309        })?),
310    };
311    Ok(CibaClientMetadata {
312        backchannel_token_delivery_mode,
313        backchannel_client_notification_endpoint: normalise_optional(endpoint),
314        backchannel_authentication_request_signing_alg,
315    })
316}

431impl<C: Connection> SurrealOAuth2ClientRepository<C> {
432    /// [`OAuth2ClientRepository::create`], optionally with the digest of an
433    /// RFC 7592 management token written in the same statement (T23.4.1).
434    ///
435    /// One `CREATE` for both, so a `dcr` client never exists without the token
436    /// its registrant was given. `None` writes `NONE`, which is what every
437    /// client that is not self-registered carries.
438    async fn create_row(
439        &self,
440        input: CreateOAuth2Client,
441        registration_access_token_hash: Option<String>,
442    ) -> AxiamResult<(OAuth2Client, String)> {
443        let id = new_id();
444        let id_str = id.to_string();
445        let tenant_id_str = input.tenant_id.to_string();
446
447        let client_id = generate_client_id();
448        // T21.2 — a public client is created with no secret at all.
449        //
450        // Not "a secret nobody tells the operator about": none. Minting one
451        // unconditionally is what SEC-093 found had given every `tls_client_auth`
452        // client a live password-equivalent credential its registration said it
453        // did not use, and a public client is the sharper case — its whole
454        // registration is the statement that it cannot keep one. An empty
455        // `client_secret_hash` is therefore the truth about the row rather than
456        // a placeholder, and the token endpoint never consults it: the public
457        // arm of `authenticate_client_credential` returns before any hash is
458        // read, so no comparison against the empty string can ever be reached.
459        let (raw_secret, secret_hash) = if input.token_endpoint_auth_method.is_public() {
460            (String::new(), String::new())
461        } else {
462            let raw = generate_client_secret();
463            let hash = client_secret::global()?.hash(&raw);
464            (raw, hash)
465        };
466
467        let result = self
468            .db
469            .current()
470            .query(
471                "CREATE type::record('oauth2_client', $id) SET \
472                 tenant_id = $tenant_id, \
473                 client_id = $client_id, \
474                 client_secret_hash = $secret_hash, \
475                 name = $name, \
476                 redirect_uris = $redirect_uris, \
477                 grant_types = $grant_types, \
478                 scopes = $scopes, \
479                 post_logout_redirect_uris = $post_logout_redirect_uris, \
480                 backchannel_logout_uri = $backchannel_logout_uri, \
481                 require_par = $require_par, \
482                 profile = $profile, \
483                 token_endpoint_auth_method = $token_endpoint_auth_method, \
484                 tls_client_auth_subject_dn = $tls_client_auth_subject_dn, \
485                 tls_client_auth_san_dns = $tls_client_auth_san_dns, \
486                 tls_client_auth_san_uri = $tls_client_auth_san_uri, \
487                 self_signed_tls_client_auth_thumbprints = $self_signed_thumbprints, \
488                 tls_client_certificate_bound_access_tokens = $cert_bound_tokens, \
489                 jwks = $jwks, \
490                 jwks_uri = $jwks_uri, \
491                 dpop_bound_access_tokens = $dpop_bound_tokens, \
492                 dpop_require_nonce = $dpop_require_nonce, \
493                 authn_request_params = $authn_request_params, \
494                 browser_sso = $browser_sso, \
495                 allowed_resources = $allowed_resources, \
496                 managed_by = $managed_by, \
497                 registration_access_token_hash = $registration_access_token_hash, \
498                 backchannel_token_delivery_mode = $ciba_mode, \
499                 backchannel_client_notification_endpoint = $ciba_endpoint, \
500                 backchannel_authentication_request_signing_alg = $ciba_signing_alg, \
501                 last_authorized_at = NONE",
502            )
503            .bind(("id", id_str.clone()))
504            .bind(("tenant_id", tenant_id_str))
505            .bind(("client_id", client_id))
506            .bind(("secret_hash", secret_hash))
507            .bind(("name", input.name))
508            .bind(("redirect_uris", input.redirect_uris))
509            .bind(("grant_types", input.grant_types))
510            .bind(("scopes", input.scopes))
511            .bind(("post_logout_redirect_uris", input.post_logout_redirect_uris))
512            .bind(("backchannel_logout_uri", input.backchannel_logout_uri))
513            .bind(("require_par", input.require_par))
514            .bind(("profile", input.profile.as_str()))
515            .bind((
516                "token_endpoint_auth_method",
517                input.token_endpoint_auth_method.as_str(),
518            ))
519            .bind((
520                "tls_client_auth_subject_dn",
521                normalise_optional(input.tls_client_auth_subject_dn),
522            ))
523            .bind((
524                "tls_client_auth_san_dns",
525                normalise_optional(input.tls_client_auth_san_dns),
526            ))
527            .bind((

529                normalise_optional(input.tls_client_auth_san_uri),

594    /// T21.5 — create or refresh a CIMD shadow row. See the trait's
595    /// documentation for the three guarantees; this is how each is kept.
596    ///
597    /// **A row with another provenance is never modified** because the
598    /// `UPDATE` carries `AND managed_by = 'cimd'` in its `WHERE`, and because
599    /// the `CREATE` that follows an update matching nothing is guarded by a
600    /// re-read: a row that exists but did not match the update is an
601    /// administrator's (or a DCR client's) and is returned as a conflict
602    /// rather than written over. The unique index on
603    /// `(tenant_id, client_id)` is the backstop for the race — two concurrent
604    /// first authorizations for the same document — and the loser of that race
605    /// re-reads rather than failing the request, because the row it wanted now
606    /// exists and is the one it would have written.
607    ///
608    /// **No secret is minted**: `client_secret_hash` is written as the empty
609    /// string on create and is not in the update's `SET` list at all, so a row
610    /// that somehow held one keeps it invisible to the only arm that could
611    /// read it (`authenticate_client_credential`'s public arm returns before
612    /// any hash is consulted, and `private_key_jwt` never looks at one).
613    ///

1250    /// T23.4.1 — replacement and rotation as one compare-and-swap, in the two
1251    /// layers X6 uses for every single-use credential (see
1252    /// `repository::device_grant`'s module header for the measurements).
1253    ///
1254    /// 1. The guarded `UPDATE` runs inside `BEGIN`/`COMMIT`, so two concurrent
1255    ///    replacements on one row conflict and the deployed engine aborts the
1256    ///    loser, which is answered `None`.
1257    /// 2. The new digest is the per-attempt nonce: it is read back **after**
1258    ///    the commit, in a query of its own, and only the caller whose digest
1259    ///    survived reports success. A loser whose write the engine failed to
1260    ///    abort finds the winner's digest and is answered `None` — its token
1261    ///    is dead, which is exactly what it would have been had it lost
1262    ///    cleanly. Outside the transaction for the reason `SCHEMA_V31` gives.
1263    async fn replace_dcr_registration(
1264        &self,
1265        tenant_id: Uuid,
1266        client_id: &str,
1267        expected_hash: &str,
1268        new_hash: &str,
1269        replacement: DcrRegistrationReplacement,
1270    ) -> AxiamResult<Option<OAuth2Client>> {
1271        if expected_hash.is_empty() || new_hash.is_empty() || expected_hash == new_hash {
1272            return Ok(None);
1273        }
1274        let result = self
1275            .db
1276            .current()
1277            .query(
1278                "BEGIN TRANSACTION; \
1279                 LET $after = (UPDATE oauth2_client SET \
1280                     name = $name, \
1281                     redirect_uris = $redirect_uris, \
1282                     grant_types = $grant_types, \
1283                     scopes = $scopes, \
1284                     token_endpoint_auth_method = $token_endpoint_auth_method, \
1285                     jwks = $jwks, \
1286                     jwks_uri = $jwks_uri, \
1287                     allowed_resources = $allowed_resources, \
1288                     backchannel_token_delivery_mode = $ciba_mode, \
1289                     backchannel_client_notification_endpoint = $ciba_endpoint, \
1290                     backchannel_authentication_request_signing_alg = $ciba_signing_alg, \
1291                     registration_access_token_hash = $new_hash, \
1292                     updated_at = time::now() \
1293                     WHERE tenant_id = $tenant_id AND client_id = $client_id \
1294                     AND managed_by = 'dcr' \
1295                     AND registration_access_token_hash = $expected_hash \
1296                     RETURN AFTER); \
1297                 SELECT meta::id(id) AS record_id, * FROM $after; \
1298                 COMMIT TRANSACTION",
1299            )
1300            .bind(("tenant_id", tenant_id.to_string()))
1301            .bind(("client_id", client_id.to_owned()))
1302            .bind(("expected_hash", expected_hash.to_owned()))
1303            .bind(("new_hash", new_hash.to_owned()))
1304            .bind(("name", replacement.name))
1305            .bind(("redirect_uris", replacement.redirect_uris))
1306            .bind(("grant_types", replacement.grant_types))
1307            .bind(("scopes", replacement.scopes))
1308            .bind((
1309                "token_endpoint_auth_method",
1310                replacement.token_endpoint_auth_method.as_str(),
1311            ))
1312            .bind(("jwks", normalise_optional(replacement.jwks)))
1313            .bind(("jwks_uri", normalise_optional(replacement.jwks_uri)))
1314            .bind(("allowed_resources", replacement.allowed_resources))
1315            .bind((
1316                "ciba_mode",
1317                replacement
1318                    .ciba
1319                    .backchannel_token_delivery_mode
1320                    .map(|m| m.as_str().to_owned()),
1321            ))
1322            .bind((
1323                "ciba_signing_alg",
1324                replacement
1325                    .ciba
1326                    .backchannel_authentication_request_signing_alg
1327                    .map(|a| a.as_str().to_owned()),
1328            ))
1329            .bind((
1330                "ciba_endpoint",
1331                normalise_optional(replacement.ciba.backchannel_client_notification_endpoint),
1332            ))
1333            .await;
1334        let mut result = match result {
1335            Ok(r) => r,
1336            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1337            Err(e) => return Err(DbError::from(e).into()),
1338        };
1339        // BEGIN=0, LET=1, SELECT=2, COMMIT=3.
1340        let rows: Vec<OAuth2ClientRowWithId> = match result.take(2) {
1341            Ok(rows) => rows,
1342            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1343            Err(e) => return Err(DbError::from(e).into()),
1344        };
1345        let Some(row) = rows.into_iter().next() else {
1346            // Unknown client, wrong token, a token already rotated away, or a
1347            // client with none. Nothing was written.
1348            return Ok(None);
1349        };
1350
1351        // Layer 2: outside, and after, the transaction above.
1352        let stored = self
1353            .db
1354            .current()
1355            .query(
1356                "SELECT VALUE registration_access_token_hash FROM oauth2_client \
1357                 WHERE tenant_id = $tenant_id AND client_id = $client_id LIMIT 1",
1358            )
1359            .bind(("tenant_id", tenant_id.to_string()))
1360            .bind(("client_id", client_id.to_owned()))
1361            .await;
1362        let mut stored = match stored {
1363            Ok(r) => r,
1364            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1365            Err(e) => return Err(DbError::from(e).into()),
1366        };
1367        let stored: Vec<Option<String>> = match stored.take(0) {
1368            Ok(v) => v,
1369            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1370            Err(e) => return Err(DbError::from(e).into()),
1371        };
1372        if stored.into_iter().flatten().next().as_deref() != Some(new_hash) {
1373            // Our write landed and another replacement's landed after it. That
1374            // one holds the registration; this caller's new token is not the
1375            // stored one and must not be handed out as though it were.
1376            return Ok(None);
1377        }
1378
1379        Ok(Some(row.try_into_client()?))
1380    }
