47/// Collapse a blank string to `None` before it is stored.
48///
49/// An empty or whitespace-only `tls_client_auth_*` value must never reach the
50/// matcher: `""` is not a DN anybody holds, but a matcher that compared it
51/// literally would authenticate any certificate whose corresponding field is
52/// also absent. Storing `None` makes the "registered nothing" case
53/// unambiguous, and `mtls_client_auth` refuses to authenticate a client with
54/// no registered expectation at all.
55fn normalise_optional(value: Option<String>) -> Option<String> {
56    value.map(|v| v.trim().to_owned()).filter(|v| !v.is_empty())
57}

214/// Decode the stored `profile` string.
215///
216/// An **absent** value is a pre-v38 row and correctly reads as `Standard`. An
217/// **unrecognised** value is not: it means this binary is older than the row
218/// it is reading, and guessing `Standard` there would silently strip a client
219/// of the constraint bundle it was registered under — a downgrade performed by
220/// a rollback. That is refused, loudly, as a migration error.
221fn decode_profile(raw: Option<&str>) -> Result<ClientProfile, DbError> {
222    match raw {
223        None => Ok(ClientProfile::default()),
224        Some(s) => ClientProfile::from_wire(s).ok_or_else(|| {
225            DbError::Migration(format!(
226                "oauth2_client.profile holds an unrecognised value {s:?}; this binary cannot \
227                 safely serve a client registered under a profile it does not implement"
228            ))
229        }),
230    }
231}

269/// Decode the stored `managed_by` discriminator (T21.4 / D5).
270///
271/// An **absent** value is a pre-v64 row and correctly reads as `Admin`: every
272/// client that existed before this migration was created by an administrator
273/// through `POST /oauth2-clients`, so that is the fact rather than a guess.
274///
275/// An **unrecognised** value fails closed, and this is the decoder where that
276/// matters most. `Admin` is the permissive answer in three independent places
277/// — it is the only provenance that may carry a FAPI profile (I5), the only
278/// one exempt from the forced consent hop (D4), and the only one the sweeper
279/// will not touch — so a rollback reading a `cimd` row it does not implement
280/// must refuse it rather than promote it to an administrator's client.
281fn decode_managed_by(raw: Option<&str>) -> Result<ManagedBy, DbError> {
282    match raw {
283        None => Ok(ManagedBy::default()),
284        Some(s) => ManagedBy::from_wire(s).ok_or_else(|| {
285            DbError::Migration(format!(
286                "oauth2_client.managed_by holds an unrecognised value {s:?}; this binary \
287                 cannot serve a client whose provenance it does not implement, because \
288                 every gate that reads this field treats 'admin' as the trusted answer"
289            ))
290        }),
291    }
292}

294/// Decode the stored CIBA metadata (G-7).
295///
296/// An absent mode is a client without the grant. An unrecognised one fails
297/// closed, for the reason [`decode_profile`] gives: a binary older than the row
298/// must not guess how a client it cannot serve wants to be told.
299fn decode_ciba(
300    mode: Option<&str>,
301    endpoint: Option<String>,
302    signing_alg: Option<&str>,
303) -> Result<CibaClientMetadata, DbError> {
304    let backchannel_token_delivery_mode = match mode {
305        None => None,
306        Some(raw) => Some(CibaDeliveryMode::from_wire(raw).ok_or_else(|| {
307            DbError::Migration(format!(
308                "oauth2_client.backchannel_token_delivery_mode holds an unrecognised value \
309                 {raw:?}; this binary cannot serve a CIBA client in a mode it does not implement"
310            ))
311        })?),
312    };
313    // Fails closed for the same reason: a row naming an algorithm this binary
314    // does not verify must not be served as if it named none, which would
315    // quietly accept the unsigned requests the client registered against.
316    let backchannel_authentication_request_signing_alg = match signing_alg {
317        None => None,
318        Some(raw) => Some(CibaRequestSigningAlg::from_wire(raw).ok_or_else(|| {
319            DbError::Migration(format!(
320                "oauth2_client.backchannel_authentication_request_signing_alg holds an \
321                 unrecognised value {raw:?}; this binary cannot verify a CIBA request signed \
322                 with an algorithm it does not implement"
323            ))
324        })?),
325    };
326    Ok(CibaClientMetadata {
327        backchannel_token_delivery_mode,
328        backchannel_client_notification_endpoint: normalise_optional(endpoint),
329        backchannel_authentication_request_signing_alg,
330    })
331}

480/// `THROW`n by a quota-claimed create when the tenant already holds
481/// `$max_clients` rows of the provenance being created (item 6 of the 1.1.0
482/// atomicity audit).
483const CLIENT_QUOTA_REACHED: &str = "axiam:client_quota_reached";
484
485/// `THROW`n by a quota-claimed create whose tenant row does not exist: there
486/// is no row to claim, so nothing is created.
487const CLIENT_QUOTA_NO_TENANT: &str = "axiam:client_quota_no_tenant";
488
489/// The statements that make a create of a `dcr` or `cimd` row one transaction
490/// with its quota (item 6 of the 1.1.0 atomicity audit). They read
491/// `$tenant_id`, `$managed_by` and `$max_clients`, and fill
492/// [`CLIENT_QUOTA_CLAIM_SLOTS`] result slots.
493///
494/// The count is made **inside** the create's transaction, and refused at the
495/// ceiling. That alone would not do: SurrealDB 3.3.4 is snapshot-isolated
496/// with write-write conflict detection only, so N transactions that each
497/// counted `max - 1` and each created a different row would all commit. So
498/// every claimed create also writes one shared key, `tenant.client_quota_epoch`
499/// (schema v103): two overlapping creates for one tenant conflict on it, the
500/// loser is retried on a fresh snapshot ([`retry_hot_row`]: a burst of
501/// registrations for one tenant is a hot row by design, one winner per
502/// round), and its count then sees the winner's row. A refusal writes
503/// nothing, so it conflicts with nothing. The epoch's next value is a `LET` of
504/// its own and the `UPDATE` only assigns it, for the stack-depth reason
505/// `role::bump_statement` gives.
506fn client_quota_claim() -> String {
507    format!(
508        "LET $quota_epoch = (SELECT VALUE client_quota_epoch ?? 0 \
509             FROM type::record('tenant', $tenant_id)); \
510         IF array::len($quota_epoch) = 0 {{ THROW '{CLIENT_QUOTA_NO_TENANT}'; }}; \
511         LET $held = (SELECT count() AS total FROM oauth2_client \
512             WHERE tenant_id = $tenant_id AND managed_by = $managed_by GROUP ALL); \
513         IF ($held[0].total ?? 0) >= $max_clients {{ THROW '{CLIENT_QUOTA_REACHED}'; }}; \
514         LET $next_quota_epoch = $quota_epoch[0] + 1; \
515         UPDATE type::record('tenant', $tenant_id) \
516             SET client_quota_epoch = $next_quota_epoch; "
517    )
518}
519
520/// How many result slots [`client_quota_claim`] fills.
521const CLIENT_QUOTA_CLAIM_SLOTS: usize = 6;
522
523/// A quota-claimed create's statement errors, read. `Ok(true)`: the tenant
524/// was at its ceiling. `Ok(false)`: nothing failed. `Err`: what is left,
525/// classified — an engine conflict is retried by the caller's
526/// [`retry_hot_row`], a duplicate is `AlreadyExists`.
527///
528/// Every slot is read: a `THROW` fires on its own statement while the rest
529/// report "not executed due to a failed transaction", which is also a
530/// write-conflict marker — so the refusals are matched first.
531fn client_quota_outcome(
532    errors: std::collections::HashMap<usize, surrealdb::Error>,
533    tenant_id: &str,
534) -> Result<bool, DbError> {
535    if errors.is_empty() {
536        return Ok(false);
537    }
538    let combined = errors
539        .into_values()
540        .map(|e| e.to_string())
541        .collect::<Vec<_>>()
542        .join("; ");
543    if combined.contains(CLIENT_QUOTA_REACHED) {
544        return Ok(true);
545    }
546    if combined.contains(CLIENT_QUOTA_NO_TENANT) {
547        return Err(DbError::NotFound {
548            entity: "tenant".into(),
549            id: tenant_id.to_owned(),
550        });
551    }
552    Err(classify_write_error(combined, "oauth2_client"))
553}

571        max_clients: Option<u32>,

573        let id = new_id();

576        let client_id = generate_client_id();
577        // T21.2 — a public client is created with no secret at all.
578        //
579        // Not "a secret nobody tells the operator about": none. Minting one
580        // unconditionally is what SEC-093 found had given every `tls_client_auth`

587        // read, so no comparison against the empty string can ever be reached.
588        let (raw_secret, secret_hash) = if input.token_endpoint_auth_method.is_public() {
589            (String::new(), String::new())
590        } else {
591            let raw = generate_client_secret();
592            let hash = client_secret::global()?.hash(&raw);
593            (raw, hash)
594        };
595
596        let Some(max_clients) = max_clients else {
597            let mut result = self
598                .send_create(
599                    CREATE_CLIENT,
600                    &input,
601                    &id_str,
602                    &client_id,
603                    &secret_hash,
604                    registration_access_token_hash,
605                    0,
606                )
607                .await?
608                .check()
609                .map_err(|e| DbError::Migration(e.to_string()))?;
610            let rows: Vec<OAuth2ClientRow> = result.take(0).map_err(DbError::from)?;
611            let row = take_first_or_not_found(rows, "oauth2_client", &id_str)?;
612            return Ok(Some((row.try_into_client(id)?, raw_secret)));
613        };
614
615        let query = format!(
616            "BEGIN TRANSACTION; {claim}{CREATE_CLIENT}; COMMIT TRANSACTION",
617            claim = client_quota_claim()
618        );
619        // Result slots: BEGIN, the claim's, then the CREATE.
620        let create_slot = 1 + CLIENT_QUOTA_CLAIM_SLOTS;
621        let tenant_id_str = input.tenant_id.to_string();
622        let max_clients = i64::from(max_clients);
623        let outcome = retry_hot_row(|| async {
624            let mut result = self
625                .send_create(
626                    &query,
627                    &input,
628                    &id_str,
629                    &client_id,
630                    &secret_hash,
631                    registration_access_token_hash,
632                    max_clients,
633                )
634                .await?;
635            if client_quota_outcome(result.take_errors(), &tenant_id_str)? {
636                return Ok(None);
637            }
638            let rows: Vec<OAuth2ClientRow> = result.take(create_slot).map_err(DbError::from)?;
639            Ok::<_, DbError>(Some(rows))
640        })
641        .await
642        .map_err(AxiamError::from)?;
643        let Some(rows) = outcome else {
644            return Ok(None);
645        };
646        let row = take_first_or_not_found(rows, "oauth2_client", &id_str)?;
647        Ok(Some((row.try_into_client(id)?, raw_secret)))
648    }

1011    /// T21.5 — create or refresh a CIMD shadow row. See the trait's
1012    /// documentation for the three guarantees; this is how each is kept.
1013    ///
1014    /// **A row with another provenance is never modified** because the
1015    /// `UPDATE` carries `AND managed_by = 'cimd'` in its `WHERE`, and because
1016    /// the `CREATE` that follows an update matching nothing is guarded by a
1017    /// re-read: a row that exists but did not match the update is an
1018    /// administrator's (or a DCR client's) and is returned as a conflict
1019    /// rather than written over. The unique index on
1020    /// `(tenant_id, client_id)` is the backstop for the race — two concurrent
1021    /// first authorizations for the same document — and the loser of that race
1022    /// re-reads rather than failing the request, because the row it wanted now
1023    /// exists and is the one it would have written.
1024    ///
1025    /// **No secret is minted**: `client_secret_hash` is written as the empty
1026    /// string on create and is not in the update's `SET` list at all, so a row
1027    /// that somehow held one keeps it invisible to the only arm that could
1028    /// read it (`authenticate_client_credential`'s public arm returns before
1029    /// any hash is consulted, and `private_key_jwt` never looks at one).
1030    ///
1031    /// **`created_at` survives** because the update does not set it; the
1032    /// table's `updated_at` moves on every write.

1037        max_clients: u32,

1044            .db

1048                 name = $name, \
1049                 redirect_uris = $redirect_uris, \
1050                 grant_types = $grant_types, \
1051                 scopes = $scopes, \
1052                 token_endpoint_auth_method = $token_endpoint_auth_method, \
1053                 jwks = $jwks, \
1054                 jwks_uri = $jwks_uri, \
1055                 allowed_resources = $allowed_resources, \
1056                 updated_at = time::now() \
1057                 WHERE tenant_id = $tenant_id AND client_id = $client_id \
1058                 AND managed_by = 'cimd' \
1059                 RETURN meta::id(id) AS record_id, *",
1060            )

1570    /// `repository::device_grant`'s module header for the measurements).
1571    ///
1572    /// 1. The guarded `UPDATE` runs inside `BEGIN`/`COMMIT`, so two concurrent
1573    ///    replacements on one row conflict and the deployed engine aborts the
1574    ///    loser, which is answered `None`.
1575    /// 2. The new digest is the per-attempt nonce: it is read back **after**
1576    ///    the commit, in a query of its own, and only the caller whose digest
1577    ///    survived reports success. A loser whose write the engine failed to
1578    ///    abort finds the winner's digest and is answered `None` — its token
1579    ///    is dead, which is exactly what it would have been had it lost
1580    ///    cleanly. Outside the transaction for the reason `SCHEMA_V31` gives.
1581    async fn replace_dcr_registration(
1582        &self,
1583        tenant_id: Uuid,
1584        client_id: &str,

1586        new_hash: &str,
1587        replacement: DcrRegistrationReplacement,
1588    ) -> AxiamResult<Option<OAuth2Client>> {
1589        if expected_hash.is_empty() || new_hash.is_empty() || expected_hash == new_hash {
1590            return Ok(None);
1591        }
1592        let result = self
1593            .db
1594            .current()
1595            .query(
1596                "BEGIN TRANSACTION; \
1597                 LET $after = (UPDATE oauth2_client SET \
1598                     name = $name, \
1599                     redirect_uris = $redirect_uris, \
1600                     grant_types = $grant_types, \
1601                     scopes = $scopes, \
1602                     token_endpoint_auth_method = $token_endpoint_auth_method, \
1603                     jwks = $jwks, \
1604                     jwks_uri = $jwks_uri, \
1605                     allowed_resources = $allowed_resources, \
1606                     backchannel_token_delivery_mode = $ciba_mode, \
1607                     backchannel_client_notification_endpoint = $ciba_endpoint, \
1608                     backchannel_authentication_request_signing_alg = $ciba_signing_alg, \
1609                     registration_access_token_hash = $new_hash, \
1610                     updated_at = time::now() \
1611                     WHERE tenant_id = $tenant_id AND client_id = $client_id \
1612                     AND managed_by = 'dcr' \
1613                     AND registration_access_token_hash = $expected_hash \
1614                     RETURN AFTER); \
1615                 SELECT meta::id(id) AS record_id, * FROM $after; \
1616                 COMMIT TRANSACTION",
1617            )
1618            .bind(("tenant_id", tenant_id.to_string()))
1619            .bind(("client_id", client_id.to_owned()))
1620            .bind(("expected_hash", expected_hash.to_owned()))
1621            .bind(("new_hash", new_hash.to_owned()))
1622            .bind(("name", replacement.name))
1623            .bind(("redirect_uris", replacement.redirect_uris))
1624            .bind(("grant_types", replacement.grant_types))
1625            .bind(("scopes", replacement.scopes))
1626            .bind((
1627                "token_endpoint_auth_method",
1628                replacement.token_endpoint_auth_method.as_str(),
1629            ))
1630            .bind(("jwks", normalise_optional(replacement.jwks)))
1631            .bind(("jwks_uri", normalise_optional(replacement.jwks_uri)))
1632            .bind(("allowed_resources", replacement.allowed_resources))
1633            .bind((
1634                "ciba_mode",
1635                replacement
1636                    .ciba
1637                    .backchannel_token_delivery_mode
1638                    .map(|m| m.as_str().to_owned()),
1639            ))
1640            .bind((
1641                "ciba_signing_alg",
1642                replacement
1643                    .ciba
1644                    .backchannel_authentication_request_signing_alg
1645                    .map(|a| a.as_str().to_owned()),
1646            ))
1647            .bind((
1648                "ciba_endpoint",
1649                normalise_optional(replacement.ciba.backchannel_client_notification_endpoint),
1650            ))
1651            .await;
1652        let mut result = match result {
1653            Ok(r) => r,
1654            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1655            Err(e) => return Err(DbError::from(e).into()),
1656        };
1657        // BEGIN=0, LET=1, SELECT=2, COMMIT=3.
1658        let rows: Vec<OAuth2ClientRowWithId> = match result.take(2) {
1659            Ok(rows) => rows,
1660            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1661            Err(e) => return Err(DbError::from(e).into()),
1662        };
1663        let Some(row) = rows.into_iter().next() else {
1664            // Unknown client, wrong token, a token already rotated away, or a
1665            // client with none. Nothing was written.
1666            return Ok(None);
1667        };
1668
1669        // Layer 2: outside, and after, the transaction above.
1670        let stored = self
1671            .db
1672            .current()
1673            .query(
1674                "SELECT VALUE registration_access_token_hash FROM oauth2_client \
1675                 WHERE tenant_id = $tenant_id AND client_id = $client_id LIMIT 1",
1676            )
1677            .bind(("tenant_id", tenant_id.to_string()))
1678            .bind(("client_id", client_id.to_owned()))
1679            .await;
1680        let mut stored = match stored {
1681            Ok(r) => r,
1682            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1683            Err(e) => return Err(DbError::from(e).into()),
1684        };
1685        let stored: Vec<Option<String>> = match stored.take(0) {
1686            Ok(v) => v,
1687            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1688            Err(e) => return Err(DbError::from(e).into()),
1689        };
1690        if stored.into_iter().flatten().next().as_deref() != Some(new_hash) {
1691            // Our write landed and another replacement's landed after it. That
1692            // one holds the registration; this caller's new token is not the
1693            // stored one and must not be handed out as though it were.
1694            return Ok(None);
1695        }
1696
1697        Ok(Some(row.try_into_client()?))
1698    }

1700    /// T23.4.1 — deletion, conditional on the digest. Inside a transaction so
1701    /// that a concurrent `PUT` presenting the same token conflicts with it
1702    /// rather than interleaving; `RETURN BEFORE` is what tells the caller
1703    /// which client it deleted, so the handler can revoke that client's
1704    /// refresh tokens by its `client_id` and audit it.
1705    async fn delete_by_registration_access_token(
1706        &self,
1707        tenant_id: Uuid,
1708        client_id: &str,
1709        registration_access_token_hash: &str,
1710    ) -> AxiamResult<Option<OAuth2Client>> {
1711        if registration_access_token_hash.is_empty() {
1712            return Ok(None);
1713        }
1714        let result = self
1715            .db
1716            .current()
1717            .query(
1718                "BEGIN TRANSACTION; \
1719                 LET $before = (DELETE oauth2_client \
1720                     WHERE tenant_id = $tenant_id AND client_id = $client_id \
1721                     AND managed_by = 'dcr' \
1722                     AND registration_access_token_hash = $hash \
1723                     RETURN BEFORE); \
1724                 SELECT meta::id(id) AS record_id, * FROM $before; \
1725                 COMMIT TRANSACTION",
1726            )
1727            .bind(("tenant_id", tenant_id.to_string()))
1728            .bind(("client_id", client_id.to_owned()))
1729            .bind(("hash", registration_access_token_hash.to_owned()))
1730            .await;
1731        let mut result = match result {
1732            Ok(r) => r,
1733            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1734            Err(e) => return Err(DbError::from(e).into()),
1735        };
1736        let rows: Vec<OAuth2ClientRowWithId> = match result.take(2) {
1737            Ok(rows) => rows,
1738            Err(e) if is_transaction_conflict(&e) => return Ok(None),
1739            Err(e) => return Err(DbError::from(e).into()),
1740        };
1741        rows.into_iter()
1742            .next()
1743            .map(OAuth2ClientRowWithId::try_into_client)
1744            .transpose()
1745            .map_err(Into::into)
1746    }
