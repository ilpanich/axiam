1//! SurrealDB implementation of [`ProofReplayRepository`] (X5.1).
2//!
3//! Insert-or-conflict semantics for the `jti` values of RFC 7523 client
4//! assertions and RFC 9449 DPoP proofs. This is the same mechanism
5//! [`saml_replay`](super::saml_replay) and
6//! [`amqp_nonce_replay`](super::amqp_nonce_replay) use, deliberately: `CREATE`
7//! against a table with a `UNIQUE` index, and the index violation **is** the
8//! "already seen" answer.
9//!
10//! There is no `SELECT` in this file, and that absence is the design. A
11//! read-then-write replay check leaves a window between the read and the write
12//! in which two concurrent copies of the same proof both see "not seen" — the
13//! exact race X6 (#316) and #318 closed for authorization codes. A
14//! proof-of-possession credential that survives one concurrent replay is not a
15//! proof of possession.
16

28/// SurrealDB implementation of the OAuth2 proof-replay repository.
29pub struct SurrealProofReplayRepository<C: Connection> {
30    db: DbHandle<C>,
31}
32
33// Manual Clone impl (not derive): avoids the spurious `C: Clone` bound that
34// blocks cloning under generic `C: Connection` callers. Matches
35// SurrealAssertionReplayRepository.
36impl<C: Connection> Clone for SurrealProofReplayRepository<C> {
37    fn clone(&self) -> Self {
38        Self {
39            db: self.db.clone(),
40        }
41    }
42}
43
44impl<C: Connection> SurrealProofReplayRepository<C> {
45    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
46        Self { db: db.into() }
47    }
48}
49
50impl<C: Connection> ProofReplayRepository for SurrealProofReplayRepository<C> {
51    async fn insert_proof_jti(
52        &self,
53        tenant_id: Uuid,
54        kind: ProofKind,
55        scope: &str,
56        jti: &str,
57        expires_at: DateTime<Utc>,
58    ) -> AxiamResult<()> {
59        let row_id = new_id().to_string();
60
61        let result = self
62            .db
63            .current()
64            .query(
65                "CREATE type::record('oauth2_proof_replay', $row_id) SET \
66                 tenant_id = $tenant_id, \
67                 kind = $kind, \
68                 scope = $scope, \
69                 jti = $jti, \
70                 expires_at = $expires_at",
71            )
72            .bind(("row_id", row_id))
73            .bind(("tenant_id", tenant_id.to_string()))
74            .bind(("kind", kind.as_str().to_owned()))
75            .bind(("scope", scope.to_owned()))
76            .bind(("jti", jti.to_owned()))
77            .bind(("expires_at", expires_at))
78            .await
79            .map_err(DbError::from)?;
80
81        // The UNIQUE violation IS the answer: this `jti` has been presented
82        // before. The three replay guards used to carry a hand-copied marker
83        // set each, kept identical by hand so they "cannot drift into
84        // disagreeing about what a conflict looks like" -- the right worry,
85        // and a mechanism that would have let a security fix land in two of
86        // three files. They now share one definition (D-09).
87        result
88            .check()
89            .map_err(classify_replay_write_error)
90            .map(|_| ())
91    }
92
93    async fn cleanup_expired_proofs(&self) -> AxiamResult<u64> {
94        cleanup_expired_rows(&self.db, "oauth2_proof_replay").await
95    }
96}
