18#[derive(Debug, SurrealValue)]
19struct TokenRowWithId {
20    record_id: String,
21    tenant_id: String,
22    name: String,
23    token_hash: String,
24    created_by: String,
25    expires_at: DateTime<Utc>,
26    used_at: Option<DateTime<Utc>>,
27    used_by_client_id: Option<String>,
28    created_at: DateTime<Utc>,

31impl TokenRowWithId {
32    fn try_into_token(self) -> Result<OAuth2RegistrationToken, DbError> {
33        let parse = |raw: &str, what: &str| {
34            Uuid::parse_str(raw)
35                .map_err(|e| DbError::Migration(format!("invalid {what} UUID: {e}")))
36        };
37        Ok(OAuth2RegistrationToken {
38            id: parse(&self.record_id, "oauth2_registration_token")?,
39            tenant_id: parse(&self.tenant_id, "tenant")?,
40            name: self.name,
41            token_hash: self.token_hash,
42            created_by: parse(&self.created_by, "created_by")?,
43            expires_at: self.expires_at,
44            used_at: self.used_at,
45            used_by_client_id: self.used_by_client_id,
46            created_at: self.created_at,
47        })
48    }
49}

71impl<C: Connection> OAuth2RegistrationTokenRepository
72    for SurrealOAuth2RegistrationTokenRepository<C>
73{
74    async fn create(
75        &self,
76        input: CreateOAuth2RegistrationToken,
77    ) -> AxiamResult<OAuth2RegistrationToken> {
78        let id = new_id();
79        let id_str = id.to_string();
80
81        let result = self
82            .db
83            .current()
84            .query(
85                "CREATE type::record('oauth2_registration_token', $id) SET \
86                 tenant_id = $tenant_id, \
87                 name = $name, \
88                 token_hash = $token_hash, \
89                 created_by = $created_by, \
90                 expires_at = $expires_at, \
91                 used_at = NONE, \
92                 used_by_client_id = NONE, \
93                 created_at = time::now(); \
94                 SELECT meta::id(id) AS record_id, * FROM \
95                 type::record('oauth2_registration_token', $id);",
96            )
97            .bind(("id", id_str.clone()))
98            .bind(("tenant_id", input.tenant_id.to_string()))
99            .bind(("name", input.name))
100            .bind(("token_hash", input.token_hash))
101            .bind(("created_by", input.created_by.to_string()))
102            .bind(("expires_at", input.expires_at))
103            .await
104            .map_err(DbError::from)?;
105
106        let mut result = result
107            .check()
108            .map_err(|e| DbError::Migration(e.to_string()))?;
109
110        let rows: Vec<TokenRowWithId> = result.take(1).map_err(DbError::from)?;
111        let row = take_first_or_not_found(rows, "oauth2_registration_token", &id_str)?;
112        Ok(row.try_into_token()?)
113    }
114
115    /// The single-use guarantee, as one statement.
116    ///
117    /// `used_at IS NONE` in the `WHERE` clause is the compare-and-swap: two
118    /// registrations presenting the same handle at the same moment both run
119    /// this `UPDATE`, exactly one matches a row, and the other receives an
120    /// empty result and is refused. Doing it as a read followed by a write —
121    /// which is what "check the token, then mark it" means — would let both
122    /// pass the check before either performed the write, on an endpoint that
123    /// is by design reachable without a credential.
124    ///
125    /// `tenant_id` and `expires_at` are conditions rather than post-checks for
126    /// the same reason: a handle minted for one tenant must not register a
127    /// client in another even for the instant between two statements, and a
128    /// token that expires mid-request must not be spendable by whichever half
129    /// of the race read it first.
130    ///
131    /// The `UPDATE` returns the rows it wrote, so a non-empty result *is* the
132    /// proof that this call is the one that spent the token.
133    async fn consume_by_token_hash(
134        &self,
135        tenant_id: Uuid,
136        token_hash: &str,
137        now: DateTime<Utc>,
138    ) -> AxiamResult<Option<OAuth2RegistrationToken>> {
139        let result = self
140            .db
141            .current()
142            .query(
143                "UPDATE oauth2_registration_token SET used_at = $now \
144                 WHERE token_hash = $token_hash \
145                   AND tenant_id = $tenant_id \
146                   AND used_at IS NONE \
147                   AND expires_at > $now \
148                 RETURN meta::id(id) AS record_id, *",
149            )
150            .bind(("token_hash", token_hash.to_string()))
151            .bind(("tenant_id", tenant_id.to_string()))
152            .bind(("now", now))
153            .await
154            .map_err(DbError::from)?;
155
156        let mut result = result
157            .check()
158            .map_err(|e| DbError::Migration(e.to_string()))?;
159
160        let rows: Vec<TokenRowWithId> = result.take(0).map_err(DbError::from)?;
161        match rows.into_iter().next() {
162            Some(row) => Ok(Some(row.try_into_token()?)),
163            None => Ok(None),
164        }
165    }

167    async fn list_for_tenant(&self, tenant_id: Uuid) -> AxiamResult<Vec<OAuth2RegistrationToken>> {
168        let result = self
169            .db
170            .current()
171            .query(
172                "SELECT meta::id(id) AS record_id, * FROM oauth2_registration_token \
173                 WHERE tenant_id = $tenant_id ORDER BY created_at DESC",
174            )
175            .bind(("tenant_id", tenant_id.to_string()))
176            .await
177            .map_err(DbError::from)?;
178
179        let mut result = result
180            .check()
181            .map_err(|e| DbError::Migration(e.to_string()))?;
182
183        let rows: Vec<TokenRowWithId> = result.take(0).map_err(DbError::from)?;
184        rows.into_iter()
185            .map(|r| r.try_into_token().map_err(Into::into))
186            .collect()
187    }

189    async fn prune_expired(&self, before: DateTime<Utc>) -> AxiamResult<u64> {
190        let result = self
191            .db
192            .current()
193            .query(
194                "DELETE oauth2_registration_token WHERE expires_at < $before \
195                 RETURN BEFORE",
196            )
197            .bind(("before", before))
198            .await
199            .map_err(DbError::from)?;
200
201        let mut result = result
202            .check()
203            .map_err(|e| DbError::Migration(e.to_string()))?;
204
205        let rows: Vec<TokenRowWithId> = result.take(0).map_err(DbError::from)?;
206        Ok(rows.len() as u64)
207    }
