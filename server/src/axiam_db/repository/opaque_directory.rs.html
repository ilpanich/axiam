1//! An enabled directory and `opaque_mode = required` never coexist, under
2//! concurrency too (item 10 of the 1.1.0 atomicity audit).
3//!
4//! Under `required` a tenant refuses password sign-in before reading a
5//! password, so a directory account — which has no OPAQUE record and never can
6//! — could not sign in at all (G-3, CONTRACT §30.3 rule 3). Each side refuses
7//! the other: a directory save that would leave an enabled directory reads the
8//! tenant's effective settings, and a settings write that would make the mode
9//! `required` reads the tenant's directory. Both reads were made before the
10//! write, in statements of their own, and the two writes touch different rows
11//! (`directory_config`, `security_settings`), so an enable and a `required`
12//! sent together each passed on the other's "before" state and both landed —
13//! leaving every directory account locked out.
14//!
15//! Both writes now claim `tenant.opaque_directory_epoch` (schema v106) the way
16//! the network gate claims a role's epochs (#701): the handler reads the epoch
17//! **before** the read its rule decides on ([`OpaqueDirectoryClaim`]), and the
18//! write's own transaction compares it and moves it ([`claim_statement`]). A
19//! claim that no longer matches — the other side committed since the read —
20//! `THROW`s, and the caller answers `409` ("changed since it was read"). Two
21//! overlapping claimed writes also write the same tenant row, so the engine
22//! aborts one; it is retried on a fresh snapshot and its compare fails. The
23//! next value is computed by the caller and only assigned in the `UPDATE`
24//! (`role::bump_statement` gives the stack-depth reason).
25//!
26//! Only the writes that could break the rule claim: a directory save whose
27//! result is enabled, and a settings write whose result is an effective
28//! `required` — for an organization baseline, on every tenant of the
29//! organization, and refused as stale when a tenant appeared since its read.
30

38/// `THROW`n by a claimed write whose tenant's epoch moved since the read.
39const OPAQUE_DIRECTORY_STALE: &str = "axiam:opaque_directory_stale";

41/// What a claimed write answers when it lost: the other side committed since
42/// the caller's read.
43pub const CHANGED_SINCE_READ: &str = "the tenant's directory or its opaque_mode changed since \
44     this request read them; reload and retry";

46/// The epoch a handler read for one tenant, before the read its rule decides
47/// on. A claimed write compares it inside its own transaction.
48#[derive(Debug, Clone, Copy, PartialEq, Eq)]
49pub struct OpaqueDirectoryClaim {
50    /// The tenant claimed.
51    pub tenant_id: Uuid,
52    /// `tenant.opaque_directory_epoch` as read (`NONE` reads as 0).
53    pub epoch: i64,
54}
55
56#[derive(Debug, SurrealValue)]
57pub(crate) struct ClaimRow {
58    pub(crate) tenant_id: String,
59    pub(crate) epoch: i64,
60}

62impl ClaimRow {
63    pub(crate) fn into_claim(self) -> Result<OpaqueDirectoryClaim, DbError> {
64        Ok(OpaqueDirectoryClaim {
65            tenant_id: crate::helpers::parse_uuid(&self.tenant_id, "tenant")?,
66            epoch: self.epoch,
67        })
68    }
69}

71/// The read a claim starts from, for the tenants a statement's `WHERE` names
72/// (`$tenant_id` or `$org_id`). Live tenants only, as the tenant listing reads.
73pub(crate) const READ_ONE: &str = "SELECT meta::id(id) AS tenant_id, \
74     opaque_directory_epoch ?? 0 AS epoch FROM type::record('tenant', $tenant_id)";
75pub(crate) const READ_ORG: &str = "SELECT meta::id(id) AS tenant_id, \
76     opaque_directory_epoch ?? 0 AS epoch FROM tenant \
77     WHERE organization_id = $org_id AND deleted_at = NONE";

79/// The bound form of a set of claims: `[{tenant, epoch, next}]`.
80pub(crate) fn bind_claims(claims: &[OpaqueDirectoryClaim]) -> serde_json::Value {
81    serde_json::Value::Array(
82        claims
83            .iter()
84            .map(|c| {
85                serde_json::json!({
86                    "tenant": c.tenant_id.to_string(),
87                    "epoch": c.epoch,
88                    "next": c.epoch + 1,
89                })
90            })
91            .collect(),
92    )
93}

95/// The statement that opens a claimed write: for each of `$opaque_claims`
96/// ([`bind_claims`]), compare the tenant's epoch and move it. One result slot.
97/// With `org_scoped`, a second statement pair (two more slots) refuses a write
98/// whose organization holds a live tenant it did not claim — one created since
99/// the read.
100pub(crate) fn claim_statement(org_scoped: bool) -> String {
101    let mut sql = format!(
102        "FOR $claim IN $opaque_claims {{ \
103             LET $held = (SELECT VALUE opaque_directory_epoch ?? 0 \
104                 FROM type::record('tenant', $claim.tenant)); \
105             IF array::len($held) = 0 OR $held[0] != $claim.epoch \
106                 {{ THROW '{OPAQUE_DIRECTORY_STALE}'; }}; \
107             UPDATE type::record('tenant', $claim.tenant) \
108                 SET opaque_directory_epoch = $claim.next; \
109         }}; "
110    );
111    if org_scoped {
112        sql.push_str(&format!(
113            "LET $unclaimed = (SELECT VALUE meta::id(id) FROM tenant \
114                 WHERE organization_id = $org_id AND deleted_at = NONE \
115                 AND meta::id(id) NOTINSIDE $opaque_claims.tenant); \
116             IF array::len($unclaimed) > 0 {{ THROW '{OPAQUE_DIRECTORY_STALE}'; }}; "
117        ));
118    }
119    sql
120}
121
122/// How many result slots [`claim_statement`] fills.
123pub(crate) fn claim_slots(org_scoped: bool) -> usize {
124    if org_scoped { 3 } else { 1 }
125}
126
127/// A claimed write's statement errors, read: the stale refusal first (see
128/// [`TxFail`] for why the order matters), then what is left, classified.
129pub(crate) fn outcome(
130    errors: std::collections::HashMap<usize, surrealdb::Error>,
131    entity: &str,
132) -> Result<(), TxFail> {
133    if errors.is_empty() {
134        return Ok(());
135    }
136    let combined = errors
137        .into_values()
138        .map(|e| e.to_string())
139        .collect::<Vec<_>>()
140        .join("; ");
141    if combined.contains(OPAQUE_DIRECTORY_STALE) {
142        return Err(TxFail::Refused(AxiamError::Conflict {
143            reason: CHANGED_SINCE_READ.into(),
144        }));
145    }
146    Err(TxFail::Db(classify_write_error(combined, entity)))
147}

149#[cfg(test)]
150mod tests {
151    use super::*;
152
153    /// The slot count is what callers index the write after the claim by:
154    /// the `FOR` is one statement whatever its body holds, and the organization
155    /// check adds a `LET` and an `IF`.
156    #[test]
157    fn the_claim_fills_the_slots_it_declares() {
158        for org_scoped in [false, true] {
159            let sql = claim_statement(org_scoped);
160            assert_eq!(sql.matches("FOR $claim").count(), 1);
161            let org_statements = sql.matches("LET $unclaimed").count()
162                + sql.matches("IF array::len($unclaimed)").count();
163            assert_eq!(claim_slots(org_scoped), 1 + org_statements);
164        }
165        assert_eq!(claim_slots(true), 3);
166    }
167
168    #[test]
169    fn a_claim_binds_its_next_epoch() {
170        let tenant_id = Uuid::new_v4();
171        let bound = bind_claims(&[OpaqueDirectoryClaim {
172            tenant_id,
173            epoch: 4,
174        }]);
175        assert_eq!(bound[0]["tenant"], tenant_id.to_string());
176        assert_eq!(bound[0]["epoch"], 4);
177        assert_eq!(bound[0]["next"], 5);
178    }
