1//! SurrealDB implementation of [`OpaqueServerSetupRepository`].
2//!
3//! One row per `(tenant, suite)`, holding the tenant's OPAQUE server key
4//! material encrypted at rest.
5//!
6//! # Why there is no update and no delete
7//!
8//! The stored `ServerSetup` carries the OPRF seed every one of the tenant's
9//! registration records was sealed against. Replacing it invalidates all of
10//! them at once; removing it locks the tenant out. Both are recoverable only
11//! by a tenant-wide password reset, which is not an operation that should be
12//! one repository call away. An operator who genuinely needs it reaches for
13//! the database directly, having read the runbook.
14
15use axiam_core::error::AxiamResult;
16use axiam_core::models::opaque::{OpaqueServerSetup, OpaqueSuite};
17use axiam_core::repository::OpaqueServerSetupRepository;
18use chrono::{DateTime, Utc};
19use surrealdb::Connection;
20use surrealdb_types::SurrealValue;
21use uuid::Uuid;

23use crate::error::DbError;
24use crate::handle::DbHandle;
25use crate::helpers::take_first_or_not_found;

27#[derive(Debug, SurrealValue)]
28struct SetupRow {
29    tenant_id: String,
30    suite: String,
31    sealed_setup: String,
32    sealed_decoy_key: String,
33    created_at: DateTime<Utc>,
34}
35
36impl SetupRow {
37    fn try_into_setup(self) -> Result<OpaqueServerSetup, DbError> {
38        Ok(OpaqueServerSetup {
39            tenant_id: Uuid::parse_str(&self.tenant_id)
40                .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?,
41            suite: self.suite.parse().map_err(DbError::Migration)?,
42            sealed_setup: self.sealed_setup,
43            sealed_decoy_key: self.sealed_decoy_key,
44            created_at: self.created_at,
45        })
46    }
47}

49/// SurrealDB implementation of the OPAQUE server setup repository.
50pub struct SurrealOpaqueServerSetupRepository<C: Connection> {
51    db: DbHandle<C>,
52}
53
54impl<C: Connection> Clone for SurrealOpaqueServerSetupRepository<C> {
55    fn clone(&self) -> Self {
56        Self {
57            db: self.db.clone(),
58        }
59    }
60}
61
62impl<C: Connection> SurrealOpaqueServerSetupRepository<C> {
63    /// Build a repository over `db`.
64    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
65        Self { db: db.into() }
66    }
67
68    async fn select(&self, tenant_id: Uuid, suite: OpaqueSuite) -> AxiamResult<Vec<SetupRow>> {
69        let result = self
70            .db
71            .current()
72            .query(
73                "SELECT * FROM opaque_server_setup \
74                 WHERE tenant_id = $tenant_id AND suite = $suite",
75            )
76            .bind(("tenant_id", tenant_id.to_string()))
77            .bind(("suite", suite.to_string()))
78            .await
79            .map_err(DbError::from)?;
80
81        let mut result = result
82            .check()
83            .map_err(|e| DbError::Migration(e.to_string()))?;
84        Ok(result.take(0).map_err(DbError::from)?)
85    }
86}
87
88impl<C: Connection> OpaqueServerSetupRepository for SurrealOpaqueServerSetupRepository<C> {
89    async fn get(&self, tenant_id: Uuid, suite: OpaqueSuite) -> AxiamResult<OpaqueServerSetup> {
90        let rows = self.select(tenant_id, suite).await?;
91        let row = take_first_or_not_found(rows, "opaque_server_setup", &tenant_id.to_string())?;
92        Ok(row.try_into_setup()?)
93    }
94
95    async fn get_or_create(&self, input: OpaqueServerSetup) -> AxiamResult<OpaqueServerSetup> {
96        // Read first, so the common path — every login after the first — does
97        // no write at all.
98        if let Some(row) = self
99            .select(input.tenant_id, input.suite)
100            .await?
101            .into_iter()
102            .next()
103        {
104            return Ok(row.try_into_setup()?);
105        }
106
107        // The record id is derived from `(tenant, suite)` rather than random,
108        // which is what makes this idempotent: two concurrent first logins
109        // address the same row, so the loser of the race gets a uniqueness
110        // failure instead of writing a second seed. A tenant whose seed
111        // depended on which request won would have records that only
112        // sometimes open — the kind of defect that presents as an
113        // intermittent wrong password and takes weeks to find.
114        let row_id = format!("{}:{}", input.tenant_id, input.suite);
115        let created = self
116            .db
117            .current()
118            .query(
119                "CREATE type::record('opaque_server_setup', $id) SET \
120                 tenant_id = $tenant_id, \
121                 suite = $suite, \
122                 sealed_setup = $sealed_setup, \
123                 sealed_decoy_key = $sealed_decoy_key \
124                 RETURN *",
125            )
126            .bind(("id", row_id))
127            .bind(("tenant_id", input.tenant_id.to_string()))
128            .bind(("suite", input.suite.to_string()))
129            .bind(("sealed_setup", input.sealed_setup.clone()))
130            .bind(("sealed_decoy_key", input.sealed_decoy_key.clone()))
131            .await;
132
133        match created {
134            Ok(response) => match response.check() {
135                Ok(mut checked) => {
136                    let rows: Vec<SetupRow> = checked.take(0).map_err(DbError::from)?;
137                    if let Some(row) = rows.into_iter().next() {
138                        return Ok(row.try_into_setup()?);
139                    }
140                    // CREATE returned nothing: somebody else won. Re-read.
141                    self.get(input.tenant_id, input.suite).await
142                }
143                // A uniqueness violation means the race was lost, which is a
144                // success for this method's contract — the caller wanted key
145                // material to exist, and it does.
146                Err(_) => self.get(input.tenant_id, input.suite).await,
147            },
148            Err(_) => self.get(input.tenant_id, input.suite).await,
149        }
150    }
151}
