92/// `get_role_permission_grants_for_roles`' statement, over the role records in
93/// `$role_records` and the tenant in `$tenant_id` — shared with the
94/// one-round-trip authorization read (`SurrealRoleRepository::authorization_inputs`)
95/// so the two cannot drift. See that method for why `in` is compared raw.
96pub(crate) const GRANTS_FOR_ROLES_SQL: &str = "SELECT \
97         meta::id(in) AS role_id, \
98         meta::id(out.id) AS record_id, \
99         out.tenant_id AS tenant_id, \
100         out.action AS action, \
101         out.description AS description, \
102         out.created_at AS created_at, \
103         out.updated_at AS updated_at, \
104         scope_ids, \
105         effect \
106     FROM grants \
107     WHERE in IN $role_records \
108     AND out.tenant_id = $tenant_id";

110/// The grant rows of several roles, grouped per role, each role keeping its own
111/// outcome: a row that cannot be read fails its role and no other.
112///
113/// `get_role_permission_grants_for_roles` fails the whole call on such a row;
114/// that call only ever reads the roles that apply. The one-round-trip
115/// authorization read fetches every assigned role's grants, so a broken row on
116/// a role that does not apply must not fail a check the three-read path would
117/// have answered — the engine surfaces a role's error only if it uses the role.
118/// A row whose role id itself cannot be read is attributed to no role and fails
119/// the whole read, as it does on the three-read path.
120pub(crate) fn grants_by_role_isolated(
121    rows: Vec<RolePermissionGrantRow>,
122) -> Result<HashMap<Uuid, AxiamResult<Vec<PermissionGrant>>>, DbError> {
123    let mut grouped: HashMap<Uuid, AxiamResult<Vec<PermissionGrant>>> = HashMap::new();
124    for row in rows {
125        let role_id = Uuid::parse_str(&row.role_id)
126            .map_err(|e| DbError::Migration(format!("invalid role UUID: {e}")))?;
127        let outcome = row.try_into_role_grant();
128        match outcome {
129            // SEC-092: a dropped grant contributes no entry at all.
130            Ok(None) => {}
131            Ok(Some((_, grant))) => {
132                if let Ok(grants) = grouped.entry(role_id).or_insert_with(|| Ok(Vec::new())) {
133                    grants.push(grant);
134                }
135            }
136            Err(e) => {
137                let slot = grouped.entry(role_id).or_insert_with(|| Ok(Vec::new()));
138                if slot.is_ok() {
139                    *slot = Err(e.into());
140                }
141            }
142        }
143    }
144    Ok(grouped)
145}

168impl PermissionGrantRow {
169    /// `Ok(None)` means "this grant edge is malformed and contributes nothing"
170    /// — not an error, and deliberately not an allow. See the `effect` match
171    /// below (SEC-092).
172    fn try_into_grant(self) -> Result<Option<PermissionGrant>, DbError> {
173        let id = Uuid::parse_str(&self.record_id)
174            .map_err(|e| DbError::Migration(format!("invalid UUID: {e}")))?;
175        let tenant_id = Uuid::parse_str(&self.tenant_id)
176            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
177
178        let scope_ids = self
179            .scope_ids
180            .unwrap_or_default()
181            .into_iter()
182            .map(|s| {
183                Uuid::parse_str(&s)
184                    .map_err(|e| DbError::Migration(format!("invalid scope UUID: {e}")))
185            })
186            .collect::<Result<Vec<_>, _>>()?;
187
188        // B1: an ABSENT `effect` reads back as Allow; an UNRECOGNISED one
189        // drops the grant entirely (`Ok(None)`).
190        //
191        // Absent is the backward-compatibility case and is simply correct:
192        // every edge written before deny-override existed means "allow".
193        //
194        // Unrecognised is the security-relevant one, and it has three possible
195        // answers, not two:
196        //
197        // 1. **Fail the read.** Takes the whole authorization path down for the
198        //    tenant over a single malformed row. Rejected.
199        // 2. **Default to Allow** — what this did until SEC-092. A grant edge
200        //    nobody can interpret became a *permissive* one. `from_wire` trims
201        //    and lowercases, so casing and whitespace are not how this happens;
202        //    the realistic paths are a truncated or partially-written value, a
203        //    row edited outside the API, and — the one worth planning for — a
204        //    **rolling upgrade in which a newer node writes a third effect** a
205        //    node still on this version does not know. Under the old default,
206        //    every one of those grants read back as allow on the old node,
207        //    which is deny-override defeated by a version skew.
208        // 3. **Default to Deny.** Tempting, and wrong for a different reason:
209        //    `PermissionEffect::Deny` is not "ignore this grant", it is an
210        //    active override that masks the action for *everyone* holding the
211        //    role. One malformed row would revoke access tenant-wide.
212        //
213        // Dropping the grant is the only answer that fails closed without
214        // being weaponisable: the row contributes neither an allow nor a deny,
215        // so the decision falls through to the other grants and ultimately to
216        // default-deny. It is logged at `error` — the write path validates
217        // `effect` (`PermissionEffect::from_wire`) and the v25 schema ASSERT
218        // rejects anything else, so reaching this branch means either a
219        // datastore modified outside both, or exactly the version skew above.
220        let effect = match self.effect.as_deref() {
221            None => PermissionEffect::Allow,
222            Some(raw) => match PermissionEffect::from_wire(raw) {
223                Some(effect) => effect,
224                None => {
225                    tracing::error!(
226                        effect = %raw,
227                        permission_id = %id,
228                        "grant edge carries an unrecognised effect; DROPPING the \
229                         grant (SEC-092). It contributes neither an allow nor a \
230                         deny — treating it as 'allow' would let a malformed \
231                         value defeat deny-override"
232                    );
233                    return Ok(None);
234                }
235            },
236        };
237
238        Ok(Some(PermissionGrant {
239            permission: Permission {
240                id,
241                tenant_id,
242                action: self.action,
243                description: self.description,
244                created_at: self.created_at,
245                updated_at: self.updated_at,
246            },
247            scope_ids,
248            effect,
249        }))
250    }
251}

265    /// The single `RELATE` behind every grant entry point (B1).
266    ///
267    /// `effect` is written as a plain string on the SAME `grants` edge the
268    /// allow path already used. That is what keeps the authorization hot path
269    /// at one batched query: denies arrive in the fetch that was already
270    /// happening, so there is no deny-specific round trip and no per-tenant
271    /// "has denies" flag that could go stale.
272    ///
273    /// It is also what keeps the `EXPLAIN` guard holding. `effect` appears
274    /// only in the SELECT *projection*, never in a predicate, so it cannot
275    /// make the query opaque to the planner — the I7(a) failure mode, where
276    /// wrapping an indexed field in a function call turned an `IndexScan` into
277    /// a full `TableScan` over every tenant's grants.
278    async fn relate_grant(
279        &self,
280        tenant_id: Uuid,
281        role_id: Uuid,
282        permission_id: Uuid,
283        scope_ids: Vec<Uuid>,
284        effect: PermissionEffect,
285        claim: Option<&RoleClaim>,
286    ) -> AxiamResult<()> {
287        let role_id_str = role_id.to_string();
288        let perm_id_str = permission_id.to_string();
289
290        // SEC-058/SECFIX-02: mirror grant_to_role's tenant guard on BOTH branches —
291        // this method is the REST-reachable path (POST /api/v1/roles/{id}/permissions).
292        // Also verify every scope_id belongs to the caller's tenant before RELATE.
293        let (scope_guard, scope_check, scope_value) = if scope_ids.is_empty() {
294            // Wildcard — same as grant_to_role
295            ("", "", "NONE")
296        } else {
297            (
298                "LET $sc = (SELECT id FROM scope WHERE tenant_id = $tid AND meta::id(id) IN $scope_ids); ",
299                " OR array::len($sc) != array::len($scope_ids)",
300                "$scope_ids",
301            )
302        };
303        let network = network_grant_statements(&role_id_str);
304        let query = format!(
305            "BEGIN TRANSACTION; \
306             LET $ro = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid); \
307             LET $pe = (SELECT VALUE action FROM permission:`{perm_id_str}` WHERE tenant_id = $tid); \
308             {scope_guard}\
309             IF array::len($ro) = 0 OR array::len($pe) = 0{scope_check} {{ \
310                 THROW 'cross-tenant edge denied'; \
311             }}; \
312             {network}\
313             RELATE role:`{role_id_str}` -> grants -> \
314             permission:`{perm_id_str}` SET scope_ids = {scope_value}, effect = $effect; \
315             COMMIT TRANSACTION"
316        );
317        let scope_strs: Vec<String> = scope_ids.iter().map(|id| id.to_string()).collect();
318        let (claimed, claim_grants, claim_holders) = claim_binds(claim);
319
320        let outcome = retry_on_write_conflict(|| async {
321            let mut result = self
322                .db
323                .current()
324                .query(query.as_str())
325                .bind(("tid", tenant_id.to_string()))
326                .bind(("scope_ids", scope_strs.clone()))
327                .bind(("effect", effect.as_str().to_string()))
328                .bind(("claimed", claimed))
329                .bind(("claim_grants", claim_grants))
330                .bind(("claim_holders", claim_holders))
331                .await
332                .map_err(DbError::from)?;
333            // This is the REST-reachable path
334            // (POST /api/v1/roles/{id}/permissions) — mirror grant_to_role's
335            // classify_write_error so a duplicate grant surfaces as 409, not 500;
336            // an engine conflict is retried, and past the attempts
337            // `WriteContention`.
338            role_tx_outcome(
339                result.take_errors(),
340                "permission_grant",
341                &role_id_str,
342                "cross-tenant permission grant denied",
343            )
344        })
345        .await;
346        match outcome.map_err(AxiamError::from)? {
347            None => Ok(()),
348            Some(refusal) => Err(refusal),
349        }
350    }
351
352    /// [`PermissionRepository::grant_to_role_with_effect`], claiming `claim`
353    /// (R110W6-04; the notes above `role::ROLE_EPOCH_STALE`).
354    ///
355    /// # Errors
356    ///
357    /// As the trait method, plus `Conflict` ("the role changed since it was
358    /// read") when the claim's epochs have moved, or when a group holds the
359    /// role and the permission is in `network:`.
360    pub async fn grant_to_role_claimed(
361        &self,
362        tenant_id: Uuid,
363        role_id: Uuid,
364        permission_id: Uuid,
365        scope_ids: Vec<Uuid>,
366        effect: PermissionEffect,
367        claim: Option<RoleClaim>,
368    ) -> AxiamResult<()> {
369        self.relate_grant(
370            tenant_id,
371            role_id,
372            permission_id,
373            scope_ids,
374            effect,
375            claim.as_ref(),
376        )
377        .await
378    }

380    /// [`PermissionRepository::revoke_from_role`], claiming `claim`
381    /// (R110W6-04).
382    ///
383    /// A `network:` revoke compares both epochs, moves `grant_epoch`, and —
384    /// with a claim — deletes only the grant the gate read: the edge must
385    /// still carry the effect the gate read (`claim.revoked_effect`), or be
386    /// absent if it read none. Otherwise the revoke could delete a deny an
387    /// administrator wrote after the gate looked, releasing a quarantine
388    /// without `network:grant` (H6 read "allow" or "none"). A revoke outside
389    /// the namespace is unchanged.
390    ///
391    /// # Errors
392    ///
393    /// As the trait method, plus `NetworkConflict::StaleRole` ("the role changed since it was
394    /// read") when the claim no longer holds.
395    pub async fn revoke_from_role_claimed(
396        &self,
397        tenant_id: Uuid,
398        role_id: Uuid,
399        permission_id: Uuid,
400        claim: Option<RoleClaim>,
401    ) -> AxiamResult<()> {
402        let role_id_str = role_id.to_string();
403        let perm_id_str = permission_id.to_string();
404        let claim_check = claim_statements(
405            &role_id_str,
406            Claims {
407                grants: true,
408                holders: true,
409            },
410        );
411        let bump = bump_statement(&role_id_str, Epoch::Grants);
412
413        // CQ-B07: verify both endpoints belong to the same tenant before DELETE.
414        let query = format!(
415            "BEGIN TRANSACTION; \
416             LET $ro = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid); \
417             LET $pe = (SELECT VALUE action FROM permission:`{perm_id_str}` WHERE tenant_id = $tid); \
418             IF array::len($ro) = 0 OR array::len($pe) = 0 {{ \
419                 THROW 'cross-tenant edge denied'; \
420             }}; \
421             IF string::matches($pe[0], {NETWORK_ACTION_REGEX}) {{ \
422                 {claim_check}\
423                 LET $now = (SELECT VALUE (effect ?? 'allow') FROM grants \
424                     WHERE in = role:`{role_id_str}` AND out = permission:`{perm_id_str}`); \
425                 IF $check_effect AND $now != $expect_effect {{ THROW '{ROLE_EPOCH_STALE}'; }}; \
426                 {bump}\
427             }}; \
428             DELETE grants WHERE \
429             in = role:`{role_id_str}` AND \
430             out = permission:`{perm_id_str}`; \
431             COMMIT TRANSACTION"
432        );
433        let (claimed, claim_grants, claim_holders) = claim_binds(claim.as_ref());
434        let (check_effect, expect_effect): (bool, Vec<String>) =
435            match claim.and_then(|c| c.revoked_effect) {
436                None => (false, Vec::new()),
437                Some(read) => (
438                    true,
439                    read.map(|e| e.as_str().to_owned()).into_iter().collect(),
440                ),
441            };
442
443        let outcome = retry_on_write_conflict(|| async {
444            let mut result = self
445                .db
446                .current()
447                .query(query.as_str())
448                .bind(("tid", tenant_id.to_string()))
449                .bind(("claimed", claimed))
450                .bind(("claim_grants", claim_grants))
451                .bind(("claim_holders", claim_holders))
452                .bind(("check_effect", check_effect))
453                .bind(("expect_effect", expect_effect.clone()))
454                .await
455                .map_err(DbError::from)?;
456            role_tx_outcome(
457                result.take_errors(),
458                "permission_grant",
459                &role_id_str,
460                "cross-tenant permission revocation denied",
461            )
462        })
463        .await;
464        match outcome.map_err(AxiamError::from)? {
465            None => Ok(()),
466            Some(refusal) => Err(refusal),
467        }
468    }
469}

471/// The statements a grant runs after its tenant guard, when the permission
472/// (`$pe[0]`, its action) is in `network:` (R110W6-04; the notes above
473/// `role::ROLE_EPOCH_STALE`): the claim on both epochs, §4.2 rule 2 re-read
474/// (a group holding the role refuses the grant, whoever asks — the seeder's
475/// reconcile included), and the `grant_epoch` move. A grant outside the
476/// namespace runs none of them.
477fn network_grant_statements(role_id: &str) -> String {
478    let claim_check = claim_statements(
479        role_id,
480        Claims {
481            grants: true,
482            holders: true,
483        },
484    );
485    let bump = bump_statement(role_id, Epoch::Grants);
486    format!(
487        "IF string::matches($pe[0], {NETWORK_ACTION_REGEX}) {{ \
488             {claim_check}\
489             LET $held = (SELECT VALUE id FROM has_role \
490                 WHERE out = role:`{role_id}` AND record::tb(in) = 'group' LIMIT 1); \
491             IF array::len($held) > 0 {{ THROW '{NETWORK_GRANT_TO_GROUP_HELD_ROLE}'; }}; \
492             {bump}\
493         }}; "
494    )
495}

569    async fn update(
570        &self,
571        tenant_id: Uuid,
572        id: Uuid,
573        input: UpdatePermission,
574    ) -> AxiamResult<Permission> {
575        let id_str = id.to_string();
576        let tenant_id_str = tenant_id.to_string();
577
578        let mut sets = Vec::new();
579        if input.action.is_some() {
580            sets.push("action = $action");
581        }
582        if input.description.is_some() {
583            sets.push("description = $description");
584        }
585        sets.push("updated_at = time::now()");
586
587        let query = format!(
588            "UPDATE type::record('permission', $id) SET {} \
589             WHERE tenant_id = $tenant_id",
590            sets.join(", ")
591        );
592
593        let db = self.db.current();
594        let mut builder = db
595            .query(&query)
596            .bind(("id", id_str.clone()))
597            .bind(("tenant_id", tenant_id_str));
598
599        if let Some(action) = input.action {
600            builder = builder.bind(("action", action));
601        }
602        if let Some(description) = input.description {
603            builder = builder.bind(("description", description));
604        }
605
606        let result = builder.await.map_err(DbError::from)?;
607        let mut result = result
608            .check()
609            .map_err(|e| classify_write_error(e.to_string(), "permission"))?;
610
611        let rows: Vec<PermissionRow> = result.take(0).map_err(DbError::from)?;
612        let row = take_first_or_not_found(rows, "permission", &id_str)?;
613
614        let tenant_id = Uuid::parse_str(&row.tenant_id)
615            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
616
617        Ok(Permission {
618            id,
619            tenant_id,
620            action: row.action,
621            description: row.description,
622            created_at: row.created_at,
623            updated_at: row.updated_at,
624        })
625    }

627    async fn delete(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
628        let id_str = id.to_string();
629
630        // Delete associated grants edges first, then the permission record —
631        // all inside one transaction so a concurrent reader never observes a
632        // partially-deleted permission. The grants edge carries no flat
633        // tenant_id column, so tenant scoping is expressed as a node-tenant
634        // guard on the edge's `out` endpoint (out.tenant_id) — mirroring
635        // role.delete. Without this guard a caller supplying a foreign-tenant
636        // permission id could strip another tenant's grants edges. `.check()`
637        // surfaces per-statement failures instead of swallowing them as Ok.
638        // SEC-104 — see `helpers::delete_existence_guard`.
639        let guard = crate::helpers::delete_existence_guard("permission");
640        let query = format!(
641            "BEGIN TRANSACTION; \
642             {guard} \
643             DELETE grants WHERE out = permission:`{id_str}` AND out.tenant_id = $tenant_id; \
644             DELETE type::record('permission', $id) WHERE tenant_id = $tenant_id; \
645             COMMIT TRANSACTION"
646        );
647
648        let mut result = self
649            .db
650            .current()
651            .query(query)
652            .bind(("id", id_str.clone()))
653            .bind(("tenant_id", tenant_id.to_string()))
654            .await
655            .map_err(DbError::from)?;
656
657        crate::helpers::map_delete_errors(result.take_errors(), "permission", &id_str)?;
658
659        Ok(())
660    }

662    async fn list(
663        &self,
664        tenant_id: Uuid,
665        pagination: Pagination,
666    ) -> AxiamResult<PaginatedResult<Permission>> {
667        let tenant_id_str = tenant_id.to_string();
668
669        // Free-text filter, applied to BOTH queries below so the
670        // total counts matches rather than rows — a pager whose page
671        // count belongs to a different result set than the page it
672        // shows is worse than no pager. Empty when unsearched, so an
673        // unfiltered list runs exactly the query it always ran.
674        let search = search_filter(&pagination, &["action", "description"]);
675        let search_term = search_bind(&pagination);
676
677        let mut count_result = self
678            .db
679            .current()
680            .query(format!(
681                "SELECT count() AS total FROM permission \
682                 WHERE tenant_id = $tenant_id{search} GROUP ALL"
683            ))
684            .bind(("tenant_id", tenant_id_str.clone()))
685            .bind(("search", search_term.clone()))
686            .await
687            .map_err(DbError::from)?;
688        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
689
690        let mut result = self
691            .db
692            .current()
693            .query(format!(
694                "SELECT meta::id(id) AS record_id, * FROM permission \
695                 WHERE tenant_id = $tenant_id{search} \
696                 ORDER BY created_at ASC \
697                 LIMIT $limit START $offset"
698            ))
699            .bind(("tenant_id", tenant_id_str))
700            .bind(("search", search_term))
701            .bind(("limit", pagination.limit))
702            .bind(("offset", pagination.offset))
703            .await
704            .map_err(DbError::from)?;
705
706        let rows: Vec<PermissionRowWithId> = result.take(0).map_err(DbError::from)?;
707
708        let items = rows
709            .into_iter()
710            .map(|row| row.try_into_permission())
711            .collect::<Result<Vec<_>, DbError>>()?;
712
713        Ok(paginate(items, count_rows, &pagination))
714    }

716    async fn grant_to_role(
717        &self,
718        tenant_id: Uuid,
719        role_id: Uuid,
720        permission_id: Uuid,
721    ) -> AxiamResult<()> {
722        let role_id_str = role_id.to_string();
723        let perm_id_str = permission_id.to_string();
724        // The seeder's path (`seed_default_roles`, `reconcile_default_role_grants`).
725        // A `network:` grant here runs the same in-transaction rule 2 and
726        // `grant_epoch` move as the REST path (R110W6-04), with no claim: a
727        // group holding the role refuses it, and a claimed write racing it
728        // loses.
729        let network = network_grant_statements(&role_id_str);
730
731        // CQ-B07: verify both endpoints belong to the same tenant before RELATE.
732        let query = format!(
733            "BEGIN TRANSACTION; \
734             LET $ro = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid); \
735             LET $pe = (SELECT VALUE action FROM permission:`{perm_id_str}` WHERE tenant_id = $tid); \
736             IF array::len($ro) = 0 OR array::len($pe) = 0 {{ \
737                 THROW 'cross-tenant edge denied'; \
738             }}; \
739             {network}\
740             RELATE role:`{role_id_str}` -> grants -> \
741             permission:`{perm_id_str}` SET scope_ids = NONE; \
742             COMMIT TRANSACTION"
743        );
744        let (claimed, claim_grants, claim_holders) = claim_binds(None);
745
746        let outcome = retry_on_write_conflict(|| async {
747            let mut result = self
748                .db
749                .current()
750                .query(query.as_str())
751                .bind(("tid", tenant_id.to_string()))
752                .bind(("claimed", claimed))
753                .bind(("claim_grants", claim_grants))
754                .bind(("claim_holders", claim_holders))
755                .await
756                .map_err(DbError::from)?;
757            role_tx_outcome(
758                result.take_errors(),
759                "permission_grant",
760                &role_id_str,
761                "cross-tenant permission grant denied",
762            )
763        })
764        .await;
765        match outcome.map_err(AxiamError::from)? {
766            None => Ok(()),
767            Some(refusal) => Err(refusal),
768        }
769    }

890    async fn get_role_permission_grants_for_roles(
891        &self,
892        tenant_id: Uuid,
893        role_ids: &[Uuid],
894    ) -> AxiamResult<HashMap<Uuid, Vec<PermissionGrant>>> {
895        if role_ids.is_empty() {
896            return Ok(HashMap::new());
897        }
898
899        let tenant_id_str = tenant_id.to_string();
900
901        // I7(a) — the predicate MUST compare the raw `in` field against record
902        // ids, never `meta::id(in)`.
903        //
904        // Wrapping the indexed field in a function call makes the predicate
905        // opaque to the planner: SurrealDB reports
906        // `TableScan { pre_decode_filter: "no (unsupported predicate)" }` and
907        // walks **every** `grants` edge in the database — i.e. every
908        // role→permission grant of every tenant — on every authorization check.
909        // Comparing `in` directly lets the `(in, out)` composite index
910        // `idx_grants_unique` (schema v19) serve it as an `IndexScan`.
911        // `crates/axiam-db/tests/authz_query_plan_test.rs` pins both plans.
912        let role_records: Vec<RecordId> = role_ids
913            .iter()
914            .map(|id| RecordId::new("role", id.to_string()))
915            .collect();
916
917        // Batched mirror of get_role_permission_grants: one query for every
918        // applicable role. `meta::id(in)` in the *projection* is fine (it is not
919        // a filter) and yields the owning role's UUID so the flattened rows can
920        // be regrouped per role. Tenant scoping is enforced on the permission
921        // endpoint (out.tenant_id), identical to the single-role query.
922        let mut result = self
923            .db
924            .current()
925            .query(GRANTS_FOR_ROLES_SQL)
926            .bind(("role_records", role_records))
927            .bind(("tenant_id", tenant_id_str))
928            .await
929            .map_err(DbError::from)?;
930
931        let rows: Vec<RolePermissionGrantRow> = result.take(0).map_err(DbError::from)?;
932
933        let mut grouped: HashMap<Uuid, Vec<PermissionGrant>> = HashMap::new();
934        for row in rows {
935            // A dropped grant (SEC-092) contributes no entry at all, so a role
936            // whose ONLY grant is malformed is absent from the map rather than
937            // present-and-empty. The engine's `grants_by_role.get(role_id)`
938            // miss and an empty Vec are already equivalent, so both spellings
939            // reach the same decision.
940            let Some((role_id, grant)) = row.try_into_role_grant()? else {
941                continue;
942            };
943            grouped.entry(role_id).or_default().push(grant);
944        }
945
946        Ok(grouped)
947    }

1001    /// **SEC-092.** An unrecognised effect drops the grant. It must be neither
1002    /// an allow (which would let an uninterpretable value read back as
1003    /// permission) nor a deny (which would let one malformed row mask the
1004    /// action for every holder of the role), and it must not be an error
1005    /// (which would take the whole authorization path down for the tenant).
1006    ///
1007    /// `"restrict"` stands in for the case worth planning for: a **third
1008    /// effect written by a newer node** during a rolling upgrade. Before
1009    /// SEC-092 every such grant read back as **allow** on the old node.
1010    #[test]
1011    fn an_unrecognised_effect_drops_the_grant() {
1012        for raw in ["restrict", "audit", "den", "", "true"] {
1013            let outcome = row(Some(raw)).try_into_grant().unwrap();
1014            assert!(
1015                outcome.is_none(),
1016                "effect {raw:?} must drop the grant, not resolve to an effect"
1017            );
1018        }
1019    }
