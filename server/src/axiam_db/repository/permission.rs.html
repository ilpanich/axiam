113impl PermissionGrantRow {
114    /// `Ok(None)` means "this grant edge is malformed and contributes nothing"
115    /// — not an error, and deliberately not an allow. See the `effect` match
116    /// below (SEC-092).
117    fn try_into_grant(self) -> Result<Option<PermissionGrant>, DbError> {
118        let id = Uuid::parse_str(&self.record_id)
119            .map_err(|e| DbError::Migration(format!("invalid UUID: {e}")))?;
120        let tenant_id = Uuid::parse_str(&self.tenant_id)
121            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
122
123        let scope_ids = self
124            .scope_ids
125            .unwrap_or_default()
126            .into_iter()
127            .map(|s| {
128                Uuid::parse_str(&s)
129                    .map_err(|e| DbError::Migration(format!("invalid scope UUID: {e}")))
130            })
131            .collect::<Result<Vec<_>, _>>()?;
132
133        // B1: an ABSENT `effect` reads back as Allow; an UNRECOGNISED one
134        // drops the grant entirely (`Ok(None)`).
135        //
136        // Absent is the backward-compatibility case and is simply correct:
137        // every edge written before deny-override existed means "allow".
138        //
139        // Unrecognised is the security-relevant one, and it has three possible
140        // answers, not two:
141        //
142        // 1. **Fail the read.** Takes the whole authorization path down for the
143        //    tenant over a single malformed row. Rejected.
144        // 2. **Default to Allow** — what this did until SEC-092. A grant edge
145        //    nobody can interpret became a *permissive* one. `from_wire` trims
146        //    and lowercases, so casing and whitespace are not how this happens;
147        //    the realistic paths are a truncated or partially-written value, a
148        //    row edited outside the API, and — the one worth planning for — a
149        //    **rolling upgrade in which a newer node writes a third effect** a
150        //    node still on this version does not know. Under the old default,
151        //    every one of those grants read back as allow on the old node,
152        //    which is deny-override defeated by a version skew.
153        // 3. **Default to Deny.** Tempting, and wrong for a different reason:
154        //    `PermissionEffect::Deny` is not "ignore this grant", it is an
155        //    active override that masks the action for *everyone* holding the
156        //    role. One malformed row would revoke access tenant-wide.
157        //
158        // Dropping the grant is the only answer that fails closed without
159        // being weaponisable: the row contributes neither an allow nor a deny,
160        // so the decision falls through to the other grants and ultimately to
161        // default-deny. It is logged at `error` — the write path validates
162        // `effect` (`PermissionEffect::from_wire`) and the v25 schema ASSERT
163        // rejects anything else, so reaching this branch means either a
164        // datastore modified outside both, or exactly the version skew above.
165        let effect = match self.effect.as_deref() {
166            None => PermissionEffect::Allow,
167            Some(raw) => match PermissionEffect::from_wire(raw) {
168                Some(effect) => effect,
169                None => {
170                    tracing::error!(
171                        effect = %raw,
172                        permission_id = %id,
173                        "grant edge carries an unrecognised effect; DROPPING the \
174                         grant (SEC-092). It contributes neither an allow nor a \
175                         deny — treating it as 'allow' would let a malformed \
176                         value defeat deny-override"
177                    );
178                    return Ok(None);
179                }
180            },
181        };
182
183        Ok(Some(PermissionGrant {
184            permission: Permission {
185                id,
186                tenant_id,
187                action: self.action,
188                description: self.description,
189                created_at: self.created_at,
190                updated_at: self.updated_at,
191            },
192            scope_ids,
193            effect,
194        }))
195    }
196}

210    /// The single `RELATE` behind every grant entry point (B1).
211    ///
212    /// `effect` is written as a plain string on the SAME `grants` edge the
213    /// allow path already used. That is what keeps the authorization hot path
214    /// at one batched query: denies arrive in the fetch that was already
215    /// happening, so there is no deny-specific round trip and no per-tenant
216    /// "has denies" flag that could go stale.
217    ///
218    /// It is also what keeps the `EXPLAIN` guard holding. `effect` appears
219    /// only in the SELECT *projection*, never in a predicate, so it cannot
220    /// make the query opaque to the planner — the I7(a) failure mode, where
221    /// wrapping an indexed field in a function call turned an `IndexScan` into
222    /// a full `TableScan` over every tenant's grants.
223    async fn relate_grant(
224        &self,
225        tenant_id: Uuid,
226        role_id: Uuid,
227        permission_id: Uuid,
228        scope_ids: Vec<Uuid>,
229        effect: PermissionEffect,
230        claim: Option<&RoleClaim>,
231    ) -> AxiamResult<()> {
232        let role_id_str = role_id.to_string();
233        let perm_id_str = permission_id.to_string();
234
235        // SEC-058/SECFIX-02: mirror grant_to_role's tenant guard on BOTH branches —
236        // this method is the REST-reachable path (POST /api/v1/roles/{id}/permissions).
237        // Also verify every scope_id belongs to the caller's tenant before RELATE.
238        let (scope_guard, scope_check, scope_value) = if scope_ids.is_empty() {
239            // Wildcard — same as grant_to_role
240            ("", "", "NONE")
241        } else {
242            (
243                "LET $sc = (SELECT id FROM scope WHERE tenant_id = $tid AND meta::id(id) IN $scope_ids); ",
244                " OR array::len($sc) != array::len($scope_ids)",
245                "$scope_ids",
246            )
247        };
248        let network = network_grant_statements(&role_id_str);
249        let query = format!(
250            "BEGIN TRANSACTION; \
251             LET $ro = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid); \
252             LET $pe = (SELECT VALUE action FROM permission:`{perm_id_str}` WHERE tenant_id = $tid); \
253             {scope_guard}\
254             IF array::len($ro) = 0 OR array::len($pe) = 0{scope_check} {{ \
255                 THROW 'cross-tenant edge denied'; \
256             }}; \
257             {network}\
258             RELATE role:`{role_id_str}` -> grants -> \
259             permission:`{perm_id_str}` SET scope_ids = {scope_value}, effect = $effect; \
260             COMMIT TRANSACTION"
261        );
262        let scope_strs: Vec<String> = scope_ids.iter().map(|id| id.to_string()).collect();
263        let (claimed, claim_grants, claim_holders) = claim_binds(claim);
264
265        let outcome = retry_on_write_conflict(|| async {
266            let mut result = self
267                .db
268                .current()
269                .query(query.as_str())
270                .bind(("tid", tenant_id.to_string()))
271                .bind(("scope_ids", scope_strs.clone()))
272                .bind(("effect", effect.as_str().to_string()))
273                .bind(("claimed", claimed))
274                .bind(("claim_grants", claim_grants))
275                .bind(("claim_holders", claim_holders))
276                .await
277                .map_err(DbError::from)?;
278            // This is the REST-reachable path
279            // (POST /api/v1/roles/{id}/permissions) — mirror grant_to_role's
280            // classify_write_error so a duplicate grant surfaces as 409, not 500;
281            // an engine conflict is retried, and past the attempts
282            // `WriteContention`.
283            role_tx_outcome(
284                result.take_errors(),
285                "permission_grant",
286                &role_id_str,
287                "cross-tenant permission grant denied",
288            )
289        })
290        .await;
291        match outcome.map_err(AxiamError::from)? {
292            None => Ok(()),
293            Some(refusal) => Err(refusal),
294        }
295    }
296
297    /// [`PermissionRepository::grant_to_role_with_effect`], claiming `claim`
298    /// (R110W6-04; the notes above `role::ROLE_EPOCH_STALE`).
299    ///
300    /// # Errors
301    ///
302    /// As the trait method, plus `Conflict` ("the role changed since it was
303    /// read") when the claim's epochs have moved, or when a group holds the
304    /// role and the permission is in `network:`.
305    pub async fn grant_to_role_claimed(
306        &self,
307        tenant_id: Uuid,
308        role_id: Uuid,
309        permission_id: Uuid,
310        scope_ids: Vec<Uuid>,
311        effect: PermissionEffect,
312        claim: Option<RoleClaim>,
313    ) -> AxiamResult<()> {
314        self.relate_grant(
315            tenant_id,
316            role_id,
317            permission_id,
318            scope_ids,
319            effect,
320            claim.as_ref(),
321        )
322        .await
323    }

325    /// [`PermissionRepository::revoke_from_role`], claiming `claim`
326    /// (R110W6-04).
327    ///
328    /// A `network:` revoke compares both epochs, moves `grant_epoch`, and —
329    /// with a claim — deletes only the grant the gate read: the edge must
330    /// still carry the effect the gate read (`claim.revoked_effect`), or be
331    /// absent if it read none. Otherwise the revoke could delete a deny an
332    /// administrator wrote after the gate looked, releasing a quarantine
333    /// without `network:grant` (H6 read "allow" or "none"). A revoke outside
334    /// the namespace is unchanged.
335    ///
336    /// # Errors
337    ///
338    /// As the trait method, plus `Conflict` ("the role changed since it was
339    /// read") when the claim no longer holds.
340    pub async fn revoke_from_role_claimed(
341        &self,
342        tenant_id: Uuid,
343        role_id: Uuid,
344        permission_id: Uuid,
345        claim: Option<RoleClaim>,
346    ) -> AxiamResult<()> {
347        let role_id_str = role_id.to_string();
348        let perm_id_str = permission_id.to_string();
349        let claim_check = claim_statements(
350            &role_id_str,
351            Claims {
352                grants: true,
353                holders: true,
354            },
355        );
356        let bump = bump_statement(&role_id_str, Epoch::Grants);
357
358        // CQ-B07: verify both endpoints belong to the same tenant before DELETE.
359        let query = format!(
360            "BEGIN TRANSACTION; \
361             LET $ro = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid); \
362             LET $pe = (SELECT VALUE action FROM permission:`{perm_id_str}` WHERE tenant_id = $tid); \
363             IF array::len($ro) = 0 OR array::len($pe) = 0 {{ \
364                 THROW 'cross-tenant edge denied'; \
365             }}; \
366             IF string::matches($pe[0], {NETWORK_ACTION_REGEX}) {{ \
367                 {claim_check}\
368                 LET $now = (SELECT VALUE (effect ?? 'allow') FROM grants \
369                     WHERE in = role:`{role_id_str}` AND out = permission:`{perm_id_str}`); \
370                 IF $check_effect AND $now != $expect_effect {{ THROW '{ROLE_EPOCH_STALE}'; }}; \
371                 {bump}\
372             }}; \
373             DELETE grants WHERE \
374             in = role:`{role_id_str}` AND \
375             out = permission:`{perm_id_str}`; \
376             COMMIT TRANSACTION"
377        );
378        let (claimed, claim_grants, claim_holders) = claim_binds(claim.as_ref());
379        let (check_effect, expect_effect): (bool, Vec<String>) =
380            match claim.and_then(|c| c.revoked_effect) {
381                None => (false, Vec::new()),
382                Some(read) => (
383                    true,
384                    read.map(|e| e.as_str().to_owned()).into_iter().collect(),
385                ),
386            };
387
388        let outcome = retry_on_write_conflict(|| async {
389            let mut result = self
390                .db
391                .current()
392                .query(query.as_str())
393                .bind(("tid", tenant_id.to_string()))
394                .bind(("claimed", claimed))
395                .bind(("claim_grants", claim_grants))
396                .bind(("claim_holders", claim_holders))
397                .bind(("check_effect", check_effect))
398                .bind(("expect_effect", expect_effect.clone()))
399                .await
400                .map_err(DbError::from)?;
401            role_tx_outcome(
402                result.take_errors(),
403                "permission_grant",
404                &role_id_str,
405                "cross-tenant permission revocation denied",
406            )
407        })
408        .await;
409        match outcome.map_err(AxiamError::from)? {
410            None => Ok(()),
411            Some(refusal) => Err(refusal),
412        }
413    }
414}

416/// The statements a grant runs after its tenant guard, when the permission
417/// (`$pe[0]`, its action) is in `network:` (R110W6-04; the notes above
418/// `role::ROLE_EPOCH_STALE`): the claim on both epochs, §4.2 rule 2 re-read
419/// (a group holding the role refuses the grant, whoever asks — the seeder's
420/// reconcile included), and the `grant_epoch` move. A grant outside the
421/// namespace runs none of them.
422fn network_grant_statements(role_id: &str) -> String {
423    let claim_check = claim_statements(
424        role_id,
425        Claims {
426            grants: true,
427            holders: true,
428        },
429    );
430    let bump = bump_statement(role_id, Epoch::Grants);
431    format!(
432        "IF string::matches($pe[0], {NETWORK_ACTION_REGEX}) {{ \
433             {claim_check}\
434             LET $held = (SELECT VALUE id FROM has_role \
435                 WHERE out = role:`{role_id}` AND record::tb(in) = 'group' LIMIT 1); \
436             IF array::len($held) > 0 {{ THROW '{NETWORK_GRANT_TO_GROUP_HELD_ROLE}'; }}; \
437             {bump}\
438         }}; "
439    )
440}

444        let id = new_id();
445        let id_str = id.to_string();
446        let tenant_id_str = input.tenant_id.to_string();

448        let result = self
449            .db
450            .current()
451            .query(

454                 action = $action, description = $description",

456            .bind(("id", id_str.clone()))

459            .bind(("description", input.description))

461            .map_err(DbError::from)?;
462
463        let mut result = result

466
467        let rows: Vec<PermissionRow> = result.take(0).map_err(DbError::from)?;
468        let row = take_first_or_not_found(rows, "permission", &id_str)?;
469

471            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;

473        Ok(Permission {
474            id,
475            tenant_id,
476            action: row.action,
477            description: row.description,
478            created_at: row.created_at,
479            updated_at: row.updated_at,

514    async fn update(
515        &self,
516        tenant_id: Uuid,
517        id: Uuid,
518        input: UpdatePermission,
519    ) -> AxiamResult<Permission> {
520        let id_str = id.to_string();
521        let tenant_id_str = tenant_id.to_string();
522
523        let mut sets = Vec::new();
524        if input.action.is_some() {
525            sets.push("action = $action");
526        }
527        if input.description.is_some() {
528            sets.push("description = $description");
529        }
530        sets.push("updated_at = time::now()");
531
532        let query = format!(
533            "UPDATE type::record('permission', $id) SET {} \
534             WHERE tenant_id = $tenant_id",
535            sets.join(", ")
536        );
537
538        let db = self.db.current();
539        let mut builder = db
540            .query(&query)
541            .bind(("id", id_str.clone()))
542            .bind(("tenant_id", tenant_id_str));
543
544        if let Some(action) = input.action {
545            builder = builder.bind(("action", action));
546        }
547        if let Some(description) = input.description {
548            builder = builder.bind(("description", description));
549        }
550
551        let result = builder.await.map_err(DbError::from)?;
552        let mut result = result
553            .check()
554            .map_err(|e| classify_write_error(e.to_string(), "permission"))?;
555
556        let rows: Vec<PermissionRow> = result.take(0).map_err(DbError::from)?;
557        let row = take_first_or_not_found(rows, "permission", &id_str)?;
558
559        let tenant_id = Uuid::parse_str(&row.tenant_id)
560            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
561
562        Ok(Permission {
563            id,
564            tenant_id,
565            action: row.action,
566            description: row.description,
567            created_at: row.created_at,
568            updated_at: row.updated_at,

572    async fn delete(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
573        let id_str = id.to_string();
574
575        // Delete associated grants edges first, then the permission record —
576        // all inside one transaction so a concurrent reader never observes a
577        // partially-deleted permission. The grants edge carries no flat
578        // tenant_id column, so tenant scoping is expressed as a node-tenant
579        // guard on the edge's `out` endpoint (out.tenant_id) — mirroring
580        // role.delete. Without this guard a caller supplying a foreign-tenant
581        // permission id could strip another tenant's grants edges. `.check()`
582        // surfaces per-statement failures instead of swallowing them as Ok.
583        // SEC-104 — see `helpers::delete_existence_guard`.
584        let guard = crate::helpers::delete_existence_guard("permission");
585        let query = format!(
586            "BEGIN TRANSACTION; \
587             {guard} \
588             DELETE grants WHERE out = permission:`{id_str}` AND out.tenant_id = $tenant_id; \
589             DELETE type::record('permission', $id) WHERE tenant_id = $tenant_id; \
590             COMMIT TRANSACTION"
591        );
592
593        let mut result = self
594            .db
595            .current()
596            .query(query)
597            .bind(("id", id_str.clone()))
598            .bind(("tenant_id", tenant_id.to_string()))
599            .await
600            .map_err(DbError::from)?;
601
602        crate::helpers::map_delete_errors(result.take_errors(), "permission", &id_str)?;
603
604        Ok(())
605    }

607    async fn list(
608        &self,
609        tenant_id: Uuid,
610        pagination: Pagination,
611    ) -> AxiamResult<PaginatedResult<Permission>> {
612        let tenant_id_str = tenant_id.to_string();
613
614        // Free-text filter, applied to BOTH queries below so the
615        // total counts matches rather than rows — a pager whose page
616        // count belongs to a different result set than the page it
617        // shows is worse than no pager. Empty when unsearched, so an
618        // unfiltered list runs exactly the query it always ran.
619        let search = search_filter(&pagination, &["action", "description"]);
620        let search_term = search_bind(&pagination);
621
622        let mut count_result = self
623            .db
624            .current()
625            .query(format!(
626                "SELECT count() AS total FROM permission \
627                 WHERE tenant_id = $tenant_id{search} GROUP ALL"
628            ))
629            .bind(("tenant_id", tenant_id_str.clone()))
630            .bind(("search", search_term.clone()))
631            .await
632            .map_err(DbError::from)?;
633        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
634
635        let mut result = self
636            .db
637            .current()
638            .query(format!(
639                "SELECT meta::id(id) AS record_id, * FROM permission \
640                 WHERE tenant_id = $tenant_id{search} \
641                 ORDER BY created_at ASC \
642                 LIMIT $limit START $offset"
643            ))
644            .bind(("tenant_id", tenant_id_str))
645            .bind(("search", search_term))
646            .bind(("limit", pagination.limit))
647            .bind(("offset", pagination.offset))
648            .await
649            .map_err(DbError::from)?;
650
651        let rows: Vec<PermissionRowWithId> = result.take(0).map_err(DbError::from)?;
652
653        let items = rows
654            .into_iter()
655            .map(|row| row.try_into_permission())
656            .collect::<Result<Vec<_>, DbError>>()?;
657
658        Ok(paginate(items, count_rows, &pagination))
659    }

661    async fn grant_to_role(
662        &self,
663        tenant_id: Uuid,
664        role_id: Uuid,
665        permission_id: Uuid,
666    ) -> AxiamResult<()> {
667        let role_id_str = role_id.to_string();
668        let perm_id_str = permission_id.to_string();
669        // The seeder's path (`seed_default_roles`, `reconcile_default_role_grants`).
670        // A `network:` grant here runs the same in-transaction rule 2 and
671        // `grant_epoch` move as the REST path (R110W6-04), with no claim: a
672        // group holding the role refuses it, and a claimed write racing it
673        // loses.
674        let network = network_grant_statements(&role_id_str);
675
676        // CQ-B07: verify both endpoints belong to the same tenant before RELATE.
677        let query = format!(
678            "BEGIN TRANSACTION; \
679             LET $ro = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid); \
680             LET $pe = (SELECT VALUE action FROM permission:`{perm_id_str}` WHERE tenant_id = $tid); \
681             IF array::len($ro) = 0 OR array::len($pe) = 0 {{ \
682                 THROW 'cross-tenant edge denied'; \
683             }}; \
684             {network}\
685             RELATE role:`{role_id_str}` -> grants -> \
686             permission:`{perm_id_str}` SET scope_ids = NONE; \
687             COMMIT TRANSACTION"
688        );
689        let (claimed, claim_grants, claim_holders) = claim_binds(None);
690
691        let outcome = retry_on_write_conflict(|| async {
692            let mut result = self
693                .db
694                .current()
695                .query(query.as_str())
696                .bind(("tid", tenant_id.to_string()))
697                .bind(("claimed", claimed))
698                .bind(("claim_grants", claim_grants))
699                .bind(("claim_holders", claim_holders))
700                .await
701                .map_err(DbError::from)?;
702            role_tx_outcome(
703                result.take_errors(),
704                "permission_grant",
705                &role_id_str,
706                "cross-tenant permission grant denied",
707            )
708        })
709        .await;
710        match outcome.map_err(AxiamError::from)? {
711            None => Ok(()),
712            Some(refusal) => Err(refusal),
713        }
714    }

835    async fn get_role_permission_grants_for_roles(
836        &self,
837        tenant_id: Uuid,
838        role_ids: &[Uuid],
839    ) -> AxiamResult<HashMap<Uuid, Vec<PermissionGrant>>> {
840        if role_ids.is_empty() {
841            return Ok(HashMap::new());
842        }
843
844        let tenant_id_str = tenant_id.to_string();
845
846        // I7(a) — the predicate MUST compare the raw `in` field against record
847        // ids, never `meta::id(in)`.
848        //
849        // Wrapping the indexed field in a function call makes the predicate
850        // opaque to the planner: SurrealDB reports
851        // `TableScan { pre_decode_filter: "no (unsupported predicate)" }` and
852        // walks **every** `grants` edge in the database — i.e. every
853        // role→permission grant of every tenant — on every authorization check.
854        // Comparing `in` directly lets the `(in, out)` composite index
855        // `idx_grants_unique` (schema v19) serve it as an `IndexScan`.
856        // `crates/axiam-db/tests/authz_query_plan_test.rs` pins both plans.
857        let role_records: Vec<RecordId> = role_ids
858            .iter()
859            .map(|id| RecordId::new("role", id.to_string()))
860            .collect();
861
862        // Batched mirror of get_role_permission_grants: one query for every
863        // applicable role. `meta::id(in)` in the *projection* is fine (it is not
864        // a filter) and yields the owning role's UUID so the flattened rows can
865        // be regrouped per role. Tenant scoping is enforced on the permission
866        // endpoint (out.tenant_id), identical to the single-role query.
867        let mut result = self
868            .db
869            .current()
870            .query(
871                "SELECT \
872                     meta::id(in) AS role_id, \
873                     meta::id(out.id) AS record_id, \
874                     out.tenant_id AS tenant_id, \
875                     out.action AS action, \
876                     out.description AS description, \
877                     out.created_at AS created_at, \
878                     out.updated_at AS updated_at, \
879                     scope_ids, \
880                     effect \
881                 FROM grants \
882                 WHERE in IN $role_records \
883                 AND out.tenant_id = $tenant_id",
884            )
885            .bind(("role_records", role_records))
886            .bind(("tenant_id", tenant_id_str))
887            .await
888            .map_err(DbError::from)?;
889
890        let rows: Vec<RolePermissionGrantRow> = result.take(0).map_err(DbError::from)?;
891
892        let mut grouped: HashMap<Uuid, Vec<PermissionGrant>> = HashMap::new();
893        for row in rows {
894            // A dropped grant (SEC-092) contributes no entry at all, so a role
895            // whose ONLY grant is malformed is absent from the map rather than
896            // present-and-empty. The engine's `grants_by_role.get(role_id)`
897            // miss and an empty Vec are already equivalent, so both spellings
898            // reach the same decision.
899            let Some((role_id, grant)) = row.try_into_role_grant()? else {
900                continue;
901            };
902            grouped.entry(role_id).or_default().push(grant);
903        }
904
905        Ok(grouped)
906    }

960    /// **SEC-092.** An unrecognised effect drops the grant. It must be neither
961    /// an allow (which would let an uninterpretable value read back as
962    /// permission) nor a deny (which would let one malformed row mask the
963    /// action for every holder of the role), and it must not be an error
964    /// (which would take the whole authorization path down for the tenant).
965    ///
966    /// `"restrict"` stands in for the case worth planning for: a **third
967    /// effect written by a newer node** during a rolling upgrade. Before
968    /// SEC-092 every such grant read back as **allow** on the old node.
969    #[test]
970    fn an_unrecognised_effect_drops_the_grant() {
971        for raw in ["restrict", "audit", "den", "", "true"] {
972            let outcome = row(Some(raw)).try_into_grant().unwrap();
973            assert!(
974                outcome.is_none(),
975                "effect {raw:?} must drop the grant, not resolve to an effect"
976            );
977        }
978    }

981    /// the role id of the rows that survive — a bug here would silently empty a
982    /// role that has perfectly good grants alongside a malformed one.

984    fn the_batched_row_drops_malformed_grants_and_keeps_the_rest() {

988            record_id: Uuid::new_v4().to_string(),
989            tenant_id: Uuid::new_v4().to_string(),
990            action: "read".into(),
991            description: "read".into(),
992            created_at: Utc::now(),
993            updated_at: Utc::now(),
994            scope_ids: None,
995            effect: effect.map(str::to_string),
996        };
997
998        assert!(
999            batched(Some("nonsense"))
1000                .try_into_role_grant()
1001                .unwrap()
1002                .is_none()
1003        );
