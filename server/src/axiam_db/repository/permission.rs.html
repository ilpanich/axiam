107impl PermissionGrantRow {
108    /// `Ok(None)` means "this grant edge is malformed and contributes nothing"
109    /// — not an error, and deliberately not an allow. See the `effect` match
110    /// below (SEC-092).
111    fn try_into_grant(self) -> Result<Option<PermissionGrant>, DbError> {
112        let id = Uuid::parse_str(&self.record_id)
113            .map_err(|e| DbError::Migration(format!("invalid UUID: {e}")))?;
114        let tenant_id = Uuid::parse_str(&self.tenant_id)
115            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
116
117        let scope_ids = self
118            .scope_ids
119            .unwrap_or_default()
120            .into_iter()
121            .map(|s| {
122                Uuid::parse_str(&s)
123                    .map_err(|e| DbError::Migration(format!("invalid scope UUID: {e}")))
124            })
125            .collect::<Result<Vec<_>, _>>()?;
126
127        // B1: an ABSENT `effect` reads back as Allow; an UNRECOGNISED one
128        // drops the grant entirely (`Ok(None)`).
129        //
130        // Absent is the backward-compatibility case and is simply correct:
131        // every edge written before deny-override existed means "allow".
132        //
133        // Unrecognised is the security-relevant one, and it has three possible
134        // answers, not two:
135        //
136        // 1. **Fail the read.** Takes the whole authorization path down for the
137        //    tenant over a single malformed row. Rejected.
138        // 2. **Default to Allow** — what this did until SEC-092. A grant edge
139        //    nobody can interpret became a *permissive* one. `from_wire` trims
140        //    and lowercases, so casing and whitespace are not how this happens;
141        //    the realistic paths are a truncated or partially-written value, a
142        //    row edited outside the API, and — the one worth planning for — a
143        //    **rolling upgrade in which a newer node writes a third effect** a
144        //    node still on this version does not know. Under the old default,
145        //    every one of those grants read back as allow on the old node,
146        //    which is deny-override defeated by a version skew.
147        // 3. **Default to Deny.** Tempting, and wrong for a different reason:
148        //    `PermissionEffect::Deny` is not "ignore this grant", it is an
149        //    active override that masks the action for *everyone* holding the
150        //    role. One malformed row would revoke access tenant-wide.
151        //
152        // Dropping the grant is the only answer that fails closed without
153        // being weaponisable: the row contributes neither an allow nor a deny,
154        // so the decision falls through to the other grants and ultimately to
155        // default-deny. It is logged at `error` — the write path validates
156        // `effect` (`PermissionEffect::from_wire`) and the v25 schema ASSERT
157        // rejects anything else, so reaching this branch means either a
158        // datastore modified outside both, or exactly the version skew above.
159        let effect = match self.effect.as_deref() {
160            None => PermissionEffect::Allow,
161            Some(raw) => match PermissionEffect::from_wire(raw) {
162                Some(effect) => effect,
163                None => {
164                    tracing::error!(
165                        effect = %raw,
166                        permission_id = %id,
167                        "grant edge carries an unrecognised effect; DROPPING the \
168                         grant (SEC-092). It contributes neither an allow nor a \
169                         deny — treating it as 'allow' would let a malformed \
170                         value defeat deny-override"
171                    );
172                    return Ok(None);
173                }
174            },
175        };
176
177        Ok(Some(PermissionGrant {
178            permission: Permission {
179                id,
180                tenant_id,
181                action: self.action,
182                description: self.description,
183                created_at: self.created_at,
184                updated_at: self.updated_at,
185            },
186            scope_ids,
187            effect,
188        }))
189    }
190}

198impl<C: Connection> SurrealPermissionRepository<C> {
199    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
200        let db = db.into();
201        Self { db }
202    }
203
204    /// The single `RELATE` behind every grant entry point (B1).
205    ///
206    /// `effect` is written as a plain string on the SAME `grants` edge the
207    /// allow path already used. That is what keeps the authorization hot path
208    /// at one batched query: denies arrive in the fetch that was already
209    /// happening, so there is no deny-specific round trip and no per-tenant
210    /// "has denies" flag that could go stale.
211    ///
212    /// It is also what keeps the `EXPLAIN` guard holding. `effect` appears
213    /// only in the SELECT *projection*, never in a predicate, so it cannot
214    /// make the query opaque to the planner — the I7(a) failure mode, where
215    /// wrapping an indexed field in a function call turned an `IndexScan` into
216    /// a full `TableScan` over every tenant's grants.
217    async fn relate_grant(
218        &self,
219        tenant_id: Uuid,
220        role_id: Uuid,
221        permission_id: Uuid,
222        scope_ids: Vec<Uuid>,
223        effect: PermissionEffect,
224    ) -> AxiamResult<()> {
225        let role_id_str = role_id.to_string();
226        let perm_id_str = permission_id.to_string();
227
228        // SEC-058/SECFIX-02: mirror grant_to_role's tenant guard on BOTH branches —
229        // this method is the REST-reachable path (POST /api/v1/roles/{id}/permissions).
230        let result = if scope_ids.is_empty() {
231            // Wildcard — same as grant_to_role
232            let query = format!(
233                "LET $ro = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid);\
234                 LET $pe = (SELECT id FROM permission:`{perm_id_str}` WHERE tenant_id = $tid);\
235                 IF array::len($ro) = 0 OR array::len($pe) = 0 {{\
236                     THROW 'cross-tenant edge denied';\
237                 }};\
238                 RELATE role:`{role_id_str}` -> grants -> \
239                 permission:`{perm_id_str}` SET scope_ids = NONE, effect = $effect;"
240            );
241            self.db
242                .current()
243                .query(query)
244                .bind(("tid", tenant_id.to_string()))
245                .bind(("effect", effect.as_str().to_string()))
246                .await
247                .map_err(DbError::from)?
248        } else {
249            let scope_strs: Vec<String> = scope_ids.iter().map(|id| id.to_string()).collect();
250            // Also verify every scope_id belongs to the caller's tenant before RELATE.
251            let query = format!(
252                "LET $ro = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid);\
253                 LET $pe = (SELECT id FROM permission:`{perm_id_str}` WHERE tenant_id = $tid);\
254                 LET $sc = (SELECT id FROM scope WHERE tenant_id = $tid AND meta::id(id) IN $scope_ids);\
255                 IF array::len($ro) = 0 OR array::len($pe) = 0 \
256                    OR array::len($sc) != array::len($scope_ids) {{\
257                     THROW 'cross-tenant edge denied';\
258                 }};\
259                 RELATE role:`{role_id_str}` -> grants -> \
260                 permission:`{perm_id_str}` SET scope_ids = $scope_ids, effect = $effect;"
261            );
262            self.db
263                .current()
264                .query(query)
265                .bind(("tid", tenant_id.to_string()))
266                .bind(("scope_ids", scope_strs))
267                .bind(("effect", effect.as_str().to_string()))
268                .await
269                .map_err(DbError::from)?
270        };
271
272        if let Err(e) = result.check() {
273            let msg = e.to_string();
274            if msg.contains("cross-tenant edge denied") {
275                return Err(AxiamError::AuthorizationDenied {
276                    reason: "cross-tenant permission grant denied".into(),
277                    action: None,
278                    resource_id: None,
279                });
280            }
281            // This is the REST-reachable path
282            // (POST /api/v1/roles/{id}/permissions) — mirror grant_to_role's
283            // classify_write_error so a duplicate grant surfaces as 409, not 500.
284            return Err(classify_write_error(msg, "permission_grant").into());
285        }
286
287        Ok(())
288    }
289}

293        let id = new_id();

295        let tenant_id_str = input.tenant_id.to_string();
296
297        let result = self
298            .db

303                 action = $action, description = $description",

305            .bind(("id", id_str.clone()))

315

317        let row = take_first_or_not_found(rows, "permission", &id_str)?;

322        Ok(Permission {

324            tenant_id,

326            description: row.description,

364        &self,
365        tenant_id: Uuid,
366        id: Uuid,
367        input: UpdatePermission,
368    ) -> AxiamResult<Permission> {
369        let id_str = id.to_string();
370        let tenant_id_str = tenant_id.to_string();
371
372        let mut sets = Vec::new();
373        if input.action.is_some() {
374            sets.push("action = $action");
375        }
376        if input.description.is_some() {
377            sets.push("description = $description");
378        }
379        sets.push("updated_at = time::now()");
380
381        let query = format!(
382            "UPDATE type::record('permission', $id) SET {} \
383             WHERE tenant_id = $tenant_id",
384            sets.join(", ")
385        );
386
387        let db = self.db.current();
388        let mut builder = db
389            .query(&query)
390            .bind(("id", id_str.clone()))
391            .bind(("tenant_id", tenant_id_str));
392
393        if let Some(action) = input.action {
394            builder = builder.bind(("action", action));
395        }
396        if let Some(description) = input.description {
397            builder = builder.bind(("description", description));
398        }
399
400        let result = builder.await.map_err(DbError::from)?;
401        let mut result = result
402            .check()
403            .map_err(|e| classify_write_error(e.to_string(), "permission"))?;
404
405        let rows: Vec<PermissionRow> = result.take(0).map_err(DbError::from)?;
406        let row = take_first_or_not_found(rows, "permission", &id_str)?;
407
408        let tenant_id = Uuid::parse_str(&row.tenant_id)
409            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
410
411        Ok(Permission {
412            id,
413            tenant_id,
414            action: row.action,
415            description: row.description,
416            created_at: row.created_at,
417            updated_at: row.updated_at,

421    async fn delete(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
422        let id_str = id.to_string();
423
424        // Delete associated grants edges first, then the permission record —
425        // all inside one transaction so a concurrent reader never observes a
426        // partially-deleted permission. The grants edge carries no flat
427        // tenant_id column, so tenant scoping is expressed as a node-tenant
428        // guard on the edge's `out` endpoint (out.tenant_id) — mirroring
429        // role.delete. Without this guard a caller supplying a foreign-tenant
430        // permission id could strip another tenant's grants edges. `.check()`
431        // surfaces per-statement failures instead of swallowing them as Ok.
432        // SEC-104 — see `helpers::delete_existence_guard`.
433        let guard = crate::helpers::delete_existence_guard("permission");
434        let query = format!(
435            "BEGIN TRANSACTION; \
436             {guard} \
437             DELETE grants WHERE out = permission:`{id_str}` AND out.tenant_id = $tenant_id; \
438             DELETE type::record('permission', $id) WHERE tenant_id = $tenant_id; \
439             COMMIT TRANSACTION"
440        );
441
442        let mut result = self
443            .db
444            .current()
445            .query(query)
446            .bind(("id", id_str.clone()))
447            .bind(("tenant_id", tenant_id.to_string()))
448            .await
449            .map_err(DbError::from)?;
450
451        crate::helpers::map_delete_errors(result.take_errors(), "permission", &id_str)?;
452
453        Ok(())
454    }

456    async fn list(
457        &self,
458        tenant_id: Uuid,
459        pagination: Pagination,
460    ) -> AxiamResult<PaginatedResult<Permission>> {
461        let tenant_id_str = tenant_id.to_string();
462
463        // Free-text filter, applied to BOTH queries below so the
464        // total counts matches rather than rows — a pager whose page
465        // count belongs to a different result set than the page it
466        // shows is worse than no pager. Empty when unsearched, so an
467        // unfiltered list runs exactly the query it always ran.
468        let search = search_filter(&pagination, &["action", "description"]);
469        let search_term = search_bind(&pagination);
470
471        let mut count_result = self
472            .db
473            .current()
474            .query(format!(
475                "SELECT count() AS total FROM permission \
476                 WHERE tenant_id = $tenant_id{search} GROUP ALL"
477            ))
478            .bind(("tenant_id", tenant_id_str.clone()))
479            .bind(("search", search_term.clone()))
480            .await
481            .map_err(DbError::from)?;
482        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
483
484        let mut result = self
485            .db
486            .current()
487            .query(format!(
488                "SELECT meta::id(id) AS record_id, * FROM permission \
489                 WHERE tenant_id = $tenant_id{search} \
490                 ORDER BY created_at ASC \
491                 LIMIT $limit START $offset"
492            ))
493            .bind(("tenant_id", tenant_id_str))
494            .bind(("search", search_term))
495            .bind(("limit", pagination.limit))
496            .bind(("offset", pagination.offset))
497            .await
498            .map_err(DbError::from)?;
499
500        let rows: Vec<PermissionRowWithId> = result.take(0).map_err(DbError::from)?;
501
502        let items = rows
503            .into_iter()
504            .map(|row| row.try_into_permission())
505            .collect::<Result<Vec<_>, DbError>>()?;
506
507        Ok(paginate(items, count_rows, &pagination))
508    }

631    async fn grant_to_role_with_effect(

662    async fn get_role_permission_grants(
663        &self,
664        tenant_id: Uuid,
665        role_id: Uuid,
666    ) -> AxiamResult<Vec<PermissionGrant>> {
667        let tenant_id_str = tenant_id.to_string();
668        let role_id_str = role_id.to_string();
669
670        let mut result = self
671            .db
672            .current()
673            .query(
674                "SELECT \
675                     meta::id(out.id) AS record_id, \
676                     out.tenant_id AS tenant_id, \
677                     out.action AS action, \
678                     out.description AS description, \
679                     out.created_at AS created_at, \
680                     out.updated_at AS updated_at, \
681                     scope_ids, \
682                     effect \
683                 FROM grants \
684                 WHERE in = type::record('role', $role_id) \
685                 AND out.tenant_id = $tenant_id",
686            )
687            .bind(("role_id", role_id_str))
688            .bind(("tenant_id", tenant_id_str))
689            .await
690            .map_err(DbError::from)?;
691
692        let rows: Vec<PermissionGrantRow> = result.take(0).map_err(DbError::from)?;
693
694        let grants = rows
695            .into_iter()
696            .map(|row| row.try_into_grant())
697            .collect::<Result<Vec<_>, DbError>>()?
698            .into_iter()
699            .flatten()
700            .collect();
701
702        Ok(grants)
703    }
704
705    async fn get_role_permission_grants_for_roles(
706        &self,
707        tenant_id: Uuid,
708        role_ids: &[Uuid],
709    ) -> AxiamResult<HashMap<Uuid, Vec<PermissionGrant>>> {
710        if role_ids.is_empty() {
711            return Ok(HashMap::new());
712        }
713
714        let tenant_id_str = tenant_id.to_string();
715
716        // I7(a) — the predicate MUST compare the raw `in` field against record
717        // ids, never `meta::id(in)`.
718        //
719        // Wrapping the indexed field in a function call makes the predicate
720        // opaque to the planner: SurrealDB reports
721        // `TableScan { pre_decode_filter: "no (unsupported predicate)" }` and
722        // walks **every** `grants` edge in the database — i.e. every
723        // role→permission grant of every tenant — on every authorization check.
724        // Comparing `in` directly lets the `(in, out)` composite index
725        // `idx_grants_unique` (schema v19) serve it as an `IndexScan`.
726        // `crates/axiam-db/tests/authz_query_plan_test.rs` pins both plans.
727        let role_records: Vec<RecordId> = role_ids
728            .iter()
729            .map(|id| RecordId::new("role", id.to_string()))
730            .collect();
731
732        // Batched mirror of get_role_permission_grants: one query for every
733        // applicable role. `meta::id(in)` in the *projection* is fine (it is not
734        // a filter) and yields the owning role's UUID so the flattened rows can
735        // be regrouped per role. Tenant scoping is enforced on the permission
736        // endpoint (out.tenant_id), identical to the single-role query.
737        let mut result = self
738            .db
739            .current()
740            .query(
741                "SELECT \
742                     meta::id(in) AS role_id, \
743                     meta::id(out.id) AS record_id, \
744                     out.tenant_id AS tenant_id, \
745                     out.action AS action, \
746                     out.description AS description, \
747                     out.created_at AS created_at, \
748                     out.updated_at AS updated_at, \
749                     scope_ids, \
750                     effect \
751                 FROM grants \
752                 WHERE in IN $role_records \
753                 AND out.tenant_id = $tenant_id",
754            )
755            .bind(("role_records", role_records))
756            .bind(("tenant_id", tenant_id_str))
757            .await
758            .map_err(DbError::from)?;
759
760        let rows: Vec<RolePermissionGrantRow> = result.take(0).map_err(DbError::from)?;
761
762        let mut grouped: HashMap<Uuid, Vec<PermissionGrant>> = HashMap::new();
763        for row in rows {
764            // A dropped grant (SEC-092) contributes no entry at all, so a role
765            // whose ONLY grant is malformed is absent from the map rather than
766            // present-and-empty. The engine's `grants_by_role.get(role_id)`
767            // miss and an empty Vec are already equivalent, so both spellings
768            // reach the same decision.
769            let Some((role_id, grant)) = row.try_into_role_grant()? else {
770                continue;
771            };
772            grouped.entry(role_id).or_default().push(grant);
773        }
774
775        Ok(grouped)
776    }

780mod tests {
781    use super::*;
782    use chrono::Utc;
783
784    fn row(effect: Option<&str>) -> PermissionGrantRow {
785        PermissionGrantRow {
786            record_id: Uuid::new_v4().to_string(),
787            tenant_id: Uuid::new_v4().to_string(),
788            action: "read".into(),
789            description: "read".into(),
790            created_at: Utc::now(),
791            updated_at: Utc::now(),
792            scope_ids: None,
793            effect: effect.map(str::to_string),
794        }
795    }
796
797    /// The backward-compatibility case: every grant edge written before B1 has
798    /// no `effect` at all, and every one of them means allow.
799    #[test]
800    fn an_absent_effect_reads_back_as_allow() {
801        let grant = row(None).try_into_grant().unwrap().expect("kept");
802        assert_eq!(grant.effect, PermissionEffect::Allow);
803    }
804
805    #[test]
806    fn the_two_valid_effects_round_trip() {
807        assert_eq!(
808            row(Some("allow")).try_into_grant().unwrap().unwrap().effect,
809            PermissionEffect::Allow
810        );
811        assert_eq!(
812            row(Some("deny")).try_into_grant().unwrap().unwrap().effect,
813            PermissionEffect::Deny
814        );
815    }
816
817    /// Casing and surrounding whitespace are *not* malformed — `from_wire`
818    /// trims and lowercases. Pinned here so the SEC-092 drop below is not
819    /// mistaken for strictness it does not have.
820    #[test]
821    fn casing_and_whitespace_still_parse() {
822        for raw in ["DENY", "Deny", " deny ", "ALLOW"] {
823            assert!(
824                row(Some(raw)).try_into_grant().unwrap().is_some(),
825                "effect {raw:?} is well-formed and must not be dropped"
826            );
827        }
828    }
829
830    /// **SEC-092.** An unrecognised effect drops the grant. It must be neither
831    /// an allow (which would let an uninterpretable value read back as
832    /// permission) nor a deny (which would let one malformed row mask the
833    /// action for every holder of the role), and it must not be an error
834    /// (which would take the whole authorization path down for the tenant).
835    ///
836    /// `"restrict"` stands in for the case worth planning for: a **third
837    /// effect written by a newer node** during a rolling upgrade. Before
838    /// SEC-092 every such grant read back as **allow** on the old node.
839    #[test]
840    fn an_unrecognised_effect_drops_the_grant() {
841        for raw in ["restrict", "audit", "den", "", "true"] {
842            let outcome = row(Some(raw)).try_into_grant().unwrap();
843            assert!(
844                outcome.is_none(),
845                "effect {raw:?} must drop the grant, not resolve to an effect"
846            );
847        }
848    }
849
850    /// The batched shape drops the same rows, and drops them *without* losing
851    /// the role id of the rows that survive — a bug here would silently empty a
852    /// role that has perfectly good grants alongside a malformed one.
853    #[test]
854    fn the_batched_row_drops_malformed_grants_and_keeps_the_rest() {
855        let role_id = Uuid::new_v4();
856        let batched = |effect: Option<&str>| RolePermissionGrantRow {
857            role_id: role_id.to_string(),
858            record_id: Uuid::new_v4().to_string(),
859            tenant_id: Uuid::new_v4().to_string(),
860            action: "read".into(),
861            description: "read".into(),
862            created_at: Utc::now(),
863            updated_at: Utc::now(),
864            scope_ids: None,
865            effect: effect.map(str::to_string),
866        };
867
868        assert!(
869            batched(Some("nonsense"))
870                .try_into_role_grant()
871                .unwrap()
872                .is_none()
873        );
874
875        let (kept_role, kept) = batched(Some("deny"))
876            .try_into_role_grant()
877            .unwrap()
878            .expect("kept");
879        assert_eq!(kept_role, role_id);
880        assert_eq!(kept.effect, PermissionEffect::Deny);
881    }
