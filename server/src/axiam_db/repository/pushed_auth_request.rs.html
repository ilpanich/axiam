1//! SurrealDB implementation of [`PushedAuthRequestRepository`] (RFC 9126 / B5).
2//!
3//! # Why `consume` layers a transaction under a nonce
4//!
5//! Same reason as the device grant's `redeem` and refresh-token rotation: a
6//! `request_uri` is single-use by RFC 9126 §2.2, and a read-then-write would
7//! let two concurrent authorize requests both spend one pushed request. The
8//! precondition — unconsumed, unexpired, right tenant — lives in the `WHERE`
9//! clause, which is what makes a later, non-concurrent replay match nothing.
10//!
11//! Expiry is part of that same clause rather than a check in the service, so a
12//! request cannot be consumed in the window between the service reading it and
13//! writing it back.
14//!
15//! Single-use is a **guarantee, conditional on an attested persistent storage
16//! engine** (X6, #302), carried by the same two layers as
17//! `permission_ticket.consume` and `device_grant.redeem`:
18//!
19//!   1. The guarded `UPDATE` runs inside an explicit `BEGIN`/`COMMIT`, so two
20//!      concurrent authorize requests conflict on one key and the deployed
21//!      engine aborts the loser — 0 double consumes in 5000 rounds of 8 racers
22//!      on `surrealkv` and 0 in 1200 on `rocksdb` (`tools/surreal-race-probe`),
23//!      against 23 in 1200 on `kv-mem`, which arbitrates at the same 54% rate
24//!      and then silently misses.
25//!   2. A per-attempt nonce is read back **after** the commit, in a query of
26//!      its own; only the caller whose nonce survived reports a consume. It
27//!      asks the engine for nothing, so it catches a conflict the engine
28//!      missed — unless the two commits interleave around the read-back (A
29//!      commits, A reads back and wins, *then* B commits and B's read-back also
30//!      wins), so the claim this layer supports is "a double consume needs two
31//!      independent failures", not "the nonce catches any conflict the engine
32//!      misses" (SEC-105). It must stay outside the transaction — inside one,
33//!      every racer sees its own write and believes it won (`SCHEMA_V31`).
34//!
35//! RFC 9126 §2.2 makes `request_uri` one-time-use precisely because a
36//! replayable one is a replayable authorization request. On `kv-mem` the
37//! guarantee does not hold and is not claimed; `axiam-server` attests the
38//! engine at startup (`axiam_db::engine_attestation`).
39

55/// Projected explicitly rather than `SELECT *` so that `meta::id(id)` lands in
56/// `record_id`; a bare `*` yields SurrealDB's `Thing`, which does not
57/// deserialize into a `String`.
58const SELECT_FIELDS: &str = "meta::id(id) AS record_id, tenant_id, client_id, \
59     request_uri_hash, params, consumed, expires_at, created_at";

117impl PushedAuthRequestRow {
118    fn try_into_request(self) -> Result<PushedAuthRequest, DbError> {
119        let id = Uuid::parse_str(&self.record_id)
120            .map_err(|e| DbError::Migration(format!("invalid pushed request UUID: {e}")))?;
121        let tenant_id = Uuid::parse_str(&self.tenant_id)
122            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
123        Ok(PushedAuthRequest {
124            id,
125            tenant_id,
126            client_id: self.client_id,
127            request_uri_hash: self.request_uri_hash,
128            params: self.params.into(),
129            consumed: self.consumed,
130            expires_at: self.expires_at,
131            created_at: self.created_at,
132        })
133    }
134}

136/// SurrealDB implementation of the pushed-authorization-request repository.
137#[derive(Clone)]
138pub struct SurrealPushedAuthRequestRepository<C: Connection> {
139    db: DbHandle<C>,
140}
141
142impl<C: Connection> SurrealPushedAuthRequestRepository<C> {
143    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
144        Self { db: db.into() }
145    }
146}
147
148impl<C: Connection> PushedAuthRequestRepository for SurrealPushedAuthRequestRepository<C> {
149    async fn create(&self, input: CreatePushedAuthRequest) -> AxiamResult<PushedAuthRequest> {
150        let id = new_id();
151        let params: PushedAuthParamsRow = input.params.into();
152
153        let mut result = self
154            .db
155            .current()
156            .query(format!(
157                "LET $created = (CREATE type::record('pushed_auth_request', $id) SET \
158                     tenant_id = $tenant_id, \
159                     client_id = $client_id, \
160                     request_uri_hash = $hash, \
161                     params = $params, \
162                     consumed = false, \
163                     expires_at = $expires_at); \
164                 SELECT {SELECT_FIELDS} FROM $created"
165            ))
166            .bind(("id", id.to_string()))
167            .bind(("tenant_id", input.tenant_id.to_string()))
168            .bind(("client_id", input.client_id))
169            .bind(("hash", input.request_uri_hash))
170            .bind(("params", params))
171            .bind(("expires_at", input.expires_at))
172            .await
173            .map_err(DbError::from)?;
174
175        let rows: Vec<PushedAuthRequestRow> = result.take(1).map_err(DbError::from)?;
176        rows.into_iter()
177            .next()
178            .ok_or_else(|| DbError::Migration("pushed_auth_request insert returned no row".into()))?
179            .try_into_request()
180            .map_err(Into::into)
181    }
182
183    async fn consume(
184        &self,
185        tenant_id: Uuid,
186        request_uri_hash: &str,
187    ) -> AxiamResult<Option<PushedAuthRequest>> {
188        // `RETURN BEFORE` yields the pre-transition row. Marking `consumed`
189        // rather than deleting keeps "already used" distinguishable from
190        // "never existed" in an audit trail, even though both answer
191        // `invalid_request` on the wire.
192        //
193        // Two layers, in two queries (X6/#302 — see the module header).
194        //
195        // Layer 1, here: the guarded `UPDATE` inside an explicit transaction,
196        // so two concurrent authorize requests conflict on one key and the
197        // deployed engine aborts the loser. `consumed = false` stays in the
198        // clause — it is what makes a later, non-concurrent replay match
199        // nothing.
200        //
201        // Layer 2, in the SEPARATE query below: the per-attempt nonce read back
202        // after this transaction has committed, never folded into it (inside a
203        // transaction every racer sees its own write and believes it won —
204        // `SCHEMA_V31`).
205        let nonce = new_id().to_string();
206        let result = self
207            .db
208            .current()
209            .query(format!(
210                "BEGIN TRANSACTION; \
211                 LET $before = (UPDATE pushed_auth_request \
212                     SET consumed = true, redemption_id = $nonce \
213                     WHERE tenant_id = $tenant_id AND request_uri_hash = $hash \
214                     AND consumed = false AND expires_at > time::now() \
215                     RETURN BEFORE); \
216                 SELECT {SELECT_FIELDS} FROM $before; \
217                 COMMIT TRANSACTION"
218            ))
219            .bind(("tenant_id", tenant_id.to_string()))
220            .bind(("hash", request_uri_hash.to_string()))
221            .bind(("nonce", nonce.clone()))
222            .await;
223
224        // A loser's abort is this method's "someone else consumed it" answer,
225        // not a server fault.
226        let mut result = match result {
227            Ok(r) => r,
228            Err(e) if is_transaction_conflict(&e) => return Ok(None),
229            Err(e) => return Err(DbError::from(e).into()),
230        };
231
232        // BEGIN=0, LET=1, SELECT=2, COMMIT=3.
233        let rows: Vec<PushedAuthRequestRow> = match result.take::<Vec<PushedAuthRequestRow>>(2) {
234            Ok(rows) => rows,
235            Err(e) if is_transaction_conflict(&e) => return Ok(None),
236            Err(e) => return Err(DbError::from(e).into()),
237        };
238        if rows.is_empty() {
239            // Unknown, already consumed, or expired. Nothing was written.
240            return Ok(None);
241        }
242
243        // Layer 2: outside, and after, the transaction above.
244        let stored = self
245            .db
246            .current()
247            .query(
248                "SELECT VALUE redemption_id FROM pushed_auth_request \
249                 WHERE tenant_id = $tenant_id AND request_uri_hash = $hash LIMIT 1",
250            )
251            .bind(("tenant_id", tenant_id.to_string()))
252            .bind(("hash", request_uri_hash.to_string()))
253            .await;
254        let mut stored = match stored {
255            Ok(r) => r,
256            Err(e) if is_transaction_conflict(&e) => return Ok(None),
257            Err(e) => return Err(DbError::from(e).into()),
258        };
259        let stored: Vec<Option<String>> = match stored.take::<Vec<Option<String>>>(0) {
260            Ok(v) => v,
261            Err(e) if is_transaction_conflict(&e) => return Ok(None),
262            Err(e) => return Err(DbError::from(e).into()),
263        };
264        if stored.into_iter().flatten().next().as_deref() != Some(nonce.as_str()) {
265            // Our write landed but another authorize request's landed after it.
266            // That one holds the consumption; this one must not replay it.
267            return Ok(None);
268        }
269
270        rows.into_iter()
271            .next()
272            .map(PushedAuthRequestRow::try_into_request)
273            .transpose()
274            .map_err(Into::into)
275    }
276
277    async fn cleanup_expired(&self, tenant_id: Uuid) -> AxiamResult<u64> {
278        let mut result = self
279            .db
280            .current()
281            .query(
282                "SELECT count() AS c FROM (DELETE pushed_auth_request \
283                 WHERE tenant_id = $tenant_id AND expires_at <= time::now() \
284                 RETURN BEFORE) GROUP ALL",
285            )
286            .bind(("tenant_id", tenant_id.to_string()))
287            .await
288            .map_err(DbError::from)?;
289
290        #[derive(Debug, SurrealValue)]
291        struct CountRow {
292            c: i64,
293        }
294        let rows: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
295        Ok(rows.first().map(|r| r.c.max(0) as u64).unwrap_or(0))
296    }
