1//! SurrealDB-backed shared rate-limit bucket counter (SECHRD-03 / D-01a).
2//!
3//! Backs the REST shared-store rate-limit pre-check middleware
4//! (`axiam-api-rest::middleware::rate_limit_shared`), which runs BEFORE the
5//! existing per-replica in-memory `Governor`/`GovernorLayer` and closes the
6//! multi-replica HPA gap (per-replica buckets otherwise multiply the
7//! effective limit by the replica count).
8//!
9//! `key` MUST already encode the endpoint (e.g. `"{endpoint}:{ip}"`) so
10//! per-endpoint limits are preserved — this repository has no notion of
11//! "endpoint" itself and never collapses distinct endpoints into one
12//! global bucket.
13
14use chrono::{DateTime, Utc};
15use surrealdb::{Connection, Surreal};
16use surrealdb_types::SurrealValue;
17
18use crate::error::DbError;
19
20/// Row shape returned by the windowed-CAS `UPSERT ... RETURN AFTER` query —
21/// only `count` is needed by the caller.
22#[derive(Debug, SurrealValue)]
23struct RateLimitBucketRow {
24    count: u64,
25}
26
27/// SurrealDB-backed shared rate-limit bucket counter.
28#[derive(Clone)]
29pub struct SurrealRateLimitBucketRepository<C: Connection> {
30    db: Surreal<C>,
31}
32
33impl<C: Connection> SurrealRateLimitBucketRepository<C> {
34    pub fn new(db: Surreal<C>) -> Self {
35        Self { db }
36    }
37
38    /// Increments the shared bucket identified by `key` for the fixed
39    /// window starting at `window_start`, returning the POST-update count.
40    ///
41    /// Windowed compare-and-set semantics (follows `increment_failed_logins`'s
42    /// read-before-write-in-one-statement pattern, `user.rs` — the RHS
43    /// `count`/`window_start` refer to the PRE-update document):
44    ///
45    /// - First hit on a fresh key: `count` is unset (`NONE`) — sets
46    ///   `count = 1`, `window_start = $window_start`.
47    /// - A hit whose `window_start` matches (or is older than requested but
48    ///   the stored value is not older than `$window_start`) the current
49    ///   window: `count = count + 1` (the SAME window — increments).
50    /// - A hit whose stored `window_start` is OLDER than `$window_start`
51    ///   (a NEW window has begun): resets `count = 1` and advances
52    ///   `window_start = $window_start`.
53    ///
54    /// All error paths `?`-propagate `DbError` — no `.unwrap()`/`.expect()`
55    /// on this counter path.
56    ///
57    /// Thin delegation to [`Self::increment_by`] with `delta = 1`; the two
58    /// share one UPSERT statement so the windowed-CAS semantics can never
59    /// drift between the single-hit and batched-delta callers.
60    pub async fn increment(&self, key: &str, window_start: DateTime<Utc>) -> Result<u64, DbError> {
61        self.increment_by(key, window_start, 1).await
62    }
63
64    /// Adds `delta` to the shared bucket identified by `key` for the fixed
65    /// window starting at `window_start`, returning the POST-update count.
66    ///
67    /// This is the batched generalization of [`Self::increment`] and the
68    /// write primitive used by
69    /// [`crate::rate_limit_counter::SharedRateLimitCounter`]'s write-behind
70    /// flusher: instead of one UPSERT per request, the flusher accumulates
71    /// `delta` local increments in memory and issues ONE UPSERT per
72    /// `(key, window)` per sync interval. That is what removes the
73    /// per-request synchronous datastore round-trip that capped the six
74    /// hottest endpoints at ~20 ops/s (see the module docs of
75    /// [`crate::rate_limit_counter`] for the measurements).
76    ///
77    /// Windowed compare-and-set semantics are identical to
78    /// [`Self::increment`], with `1` generalized to `$delta`:
79    ///
80    /// - First hit on a fresh key (`count` is `NONE`): `count = $delta`.
81    /// - Same window: `count = count + $delta`.
82    /// - Stored `window_start` OLDER than `$window_start` (a new window
83    ///   began): `count = $delta`, `window_start = $window_start`.
84    ///
85    /// A `delta` of `0` is a legal no-op write (it still refreshes
86    /// `updated_at` and returns the current count), but the flusher never
87    /// issues one — it skips buckets with nothing pending.
88    pub async fn increment_by(
89        &self,
90        key: &str,
91        window_start: DateTime<Utc>,
92        delta: u64,
93    ) -> Result<u64, DbError> {
94        let result = self
95            .db
96            .query(
97                "UPSERT type::record('rate_limit_bucket', $key) SET \
98                 count = IF window_start = NONE OR window_start < $window_start \
99                          THEN $delta ELSE count + $delta END, \
100                 window_start = IF window_start = NONE OR window_start < $window_start \
101                          THEN $window_start ELSE window_start END, \
102                 updated_at = time::now() \
103                 RETURN AFTER",
104            )
105            .bind(("key", key.to_string()))
106            .bind(("window_start", window_start))
107            .bind(("delta", delta))
108            .await
109            .map_err(DbError::from)?;
110
111        let mut result = result
112            .check()
113            .map_err(|e| DbError::Migration(e.to_string()))?;
114
115        let rows: Vec<RateLimitBucketRow> = result.take(0).map_err(DbError::from)?;
116        let row = rows
117            .into_iter()
118            .next()
119            .ok_or_else(|| DbError::Migration("rate_limit_bucket UPSERT returned no row".into()))?;
120
121        Ok(row.count)
122    }
123}

125#[cfg(test)]
126mod tests {
127    use super::*;
128    use chrono::Duration;
129    use surrealdb::Surreal;
130    use surrealdb::engine::local::Mem;
131
132    #[tokio::test]
133    async fn rate_limit_bucket_increment_sequence_and_window_reset() {
134        let db = Surreal::new::<Mem>(()).await.unwrap();
135        db.use_ns("test").use_db("test").await.unwrap();
136        crate::schema::run_migrations(&db).await.unwrap();
137
138        let repo = SurrealRateLimitBucketRepository::new(db);
139
140        let key = "login:203.0.113.5";
141        let window1 = Utc::now();
142
143        // First hit in a fresh window sets count = 1.
144        assert_eq!(repo.increment(key, window1).await.unwrap(), 1);
145
146        // Subsequent hits in the SAME window increment.
147        assert_eq!(repo.increment(key, window1).await.unwrap(), 2);
148        assert_eq!(repo.increment(key, window1).await.unwrap(), 3);
149
150        // A hit with a NEWER window_start resets count to 1 (new window).
151        let window2 = window1 + Duration::minutes(1);
152        assert_eq!(repo.increment(key, window2).await.unwrap(), 1);
153
154        // Per-endpoint granularity: a DIFFERENT key starts its own bucket at
155        // 1, independent of `key`'s already-advanced count.
156        let other_key = "mfa_verify:203.0.113.5";
157        assert_eq!(repo.increment(other_key, window2).await.unwrap(), 1);
158
159        // The original key's bucket is untouched by the other key's
160        // increment — no cross-endpoint bucket collapsing.
161        assert_eq!(repo.increment(key, window2).await.unwrap(), 2);
162    }
163
164    /// `increment_by` is the write primitive behind the write-behind flusher
165    /// (`crate::rate_limit_counter`): a batched delta must land exactly as
166    /// `delta` single increments would have, including the window reset.
167    #[tokio::test]
168    async fn rate_limit_bucket_increment_by_batches_deltas_and_resets_on_new_window() {
169        let db = Surreal::new::<Mem>(()).await.unwrap();
170        db.use_ns("test").use_db("test").await.unwrap();
171        crate::schema::run_migrations(&db).await.unwrap();
172
173        let repo = SurrealRateLimitBucketRepository::new(db);
174
175        let key = "authz_check:203.0.113.7";
176        let window1 = Utc::now();
177
178        // First flush on a fresh key sets count = delta (not 1).
179        assert_eq!(repo.increment_by(key, window1, 7).await.unwrap(), 7);
180
181        // A second flush in the SAME window ADDS the delta.
182        assert_eq!(repo.increment_by(key, window1, 5).await.unwrap(), 12);
183
184        // A single increment interleaves correctly with batched deltas.
185        assert_eq!(repo.increment(key, window1).await.unwrap(), 13);
186
187        // A NEWER window resets to exactly the delta (not delta + old count).
188        let window2 = window1 + Duration::minutes(1);
189        assert_eq!(repo.increment_by(key, window2, 4).await.unwrap(), 4);
190
191        // delta = 0 is a legal no-op that returns the unchanged count.
192        assert_eq!(repo.increment_by(key, window2, 0).await.unwrap(), 4);
193
194        // Per-key isolation holds for batched deltas too.
195        let other = "login:203.0.113.7";
196        assert_eq!(repo.increment_by(other, window2, 2).await.unwrap(), 2);
197        assert_eq!(repo.increment_by(key, window2, 1).await.unwrap(), 5);
198    }
