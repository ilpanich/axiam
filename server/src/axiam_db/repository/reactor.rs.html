60/// Parse `mode`, defaulting an unrecognised value to [`ReactorMode::Listen`].
61///
62/// Listen is the safe side: a listener is fire-and-forget and cannot veto,
63/// mutate or delay anything. Defaulting the other way would let a row written
64/// outside the API turn itself into an interceptor on the token path.
65///
66/// Erroring instead was considered and rejected for the same reason SEC-092
67/// rejected it on grants — one malformed row must not take a tenant's token
68/// issuance down. Unlike SEC-092 there is a genuinely inert value available
69/// here, so demotion is possible where dropping was needed there.
70fn mode_or_safe_default(raw: &str, id: Uuid) -> ReactorMode {
71    ReactorMode::from_wire(raw).unwrap_or_else(|| {
72        tracing::error!(
73            mode = %raw,
74            reactor_id = %id,
75            "reactor row carries an unrecognised mode; demoting it to 'listen' \
76             so it cannot intercept"
77        );
78        ReactorMode::Listen
79    })
80}
81
82/// Parse `failure_policy`, defaulting an unrecognised value to
83/// [`FailurePolicy::FailClosed`] — an unreachable veto has not passed.
84fn failure_policy_or_safe_default(raw: &str, id: Uuid) -> FailurePolicy {
85    FailurePolicy::from_wire(raw).unwrap_or_else(|| {
86        tracing::error!(
87            failure_policy = %raw,
88            reactor_id = %id,
89            "reactor row carries an unrecognised failure_policy; defaulting to \
90             'fail_closed'"
91        );
92        FailurePolicy::FailClosed
93    })
94}
95
96/// Clamp a stored `timeout_ms` into the range the API would have accepted.
97///
98/// The v29 schema ASSERT already enforces this on write, so a value outside it
99/// means the row was written outside the API. Clamping rather than trusting
100/// keeps the dispatcher's wait bounded by [`MAX_TIMEOUT_MS`] no matter what is
101/// in the column — the ASSERT and this clamp are the same rule stated in the
102/// two places that can enforce it.
103fn clamp_timeout(raw: i64, id: Uuid) -> u32 {
104    let clamped = raw.clamp(1, i64::from(MAX_TIMEOUT_MS)) as u32;
105    if i64::from(clamped) != raw {
106        tracing::error!(
107            timeout_ms = raw,
108            clamped_to = clamped,
109            reactor_id = %id,
110            "reactor row carries an out-of-range timeout_ms; clamping"
111        );
112    }
113    clamped
114}

116impl ReactorRow {
117    /// The `CREATE`/`UPDATE` shape has no `record_id` projection, so the id
118    /// comes from the caller. Rather than a second converter that could drift
119    /// from the read path's, it borrows the id and delegates — one place
120    /// decides what a row means.
121    fn try_into_entry(self, id: Uuid) -> Result<Reactor, DbError> {
122        ReactorRowWithId {
123            record_id: id.to_string(),
124            tenant_id: self.tenant_id,
125            name: self.name,
126            description: self.description,
127            events: self.events,
128            mode: self.mode,
129            priority: self.priority,
130            timeout_ms: self.timeout_ms,
131            failure_policy: self.failure_policy,
132            enabled: self.enabled,
133            created_at: self.created_at,
134            updated_at: self.updated_at,
135            last_seen_at: self.last_seen_at,
136        }
137        .try_into_entry()
138    }
139}

141impl ReactorRowWithId {
142    fn try_into_entry(self) -> Result<Reactor, DbError> {
143        let id = Uuid::parse_str(&self.record_id).map_err(|e| DbError::Migration(e.to_string()))?;
144        Ok(Reactor {
145            id,
146            tenant_id: Uuid::parse_str(&self.tenant_id)
147                .map_err(|e| DbError::Migration(e.to_string()))?,
148            name: self.name,
149            description: self.description,
150            events: self.events,
151            mode: mode_or_safe_default(&self.mode, id),
152            priority: self.priority as i32,
153            timeout_ms: clamp_timeout(self.timeout_ms, id),
154            failure_policy: failure_policy_or_safe_default(&self.failure_policy, id),
155            enabled: self.enabled,
156            created_at: self.created_at,
157            updated_at: self.updated_at,
158            last_seen_at: self.last_seen_at,
159        })
160    }
161}

179impl<C: Connection> ReactorRepository for SurrealReactorRepository<C> {
180    async fn create(&self, input: CreateReactor) -> AxiamResult<Reactor> {
181        let id = new_id();
182        // An omitted failure_policy takes the STRICTEST default among the
183        // subscribed events, not a blanket one — see
184        // `default_failure_policy_for`. That is how a caller registering for
185        // `login.post_auth` gets fail_closed without having to know it should.
186        let failure_policy = input
187            .failure_policy
188            .unwrap_or_else(|| default_failure_policy_for(&input.events));
189        let timeout_ms = input.timeout_ms.unwrap_or(DEFAULT_TIMEOUT_MS);
190
191        let result = self
192            .db
193            .current()
194            .query(
195                "CREATE type::record('reactor', $id) SET \
196                 tenant_id = $tenant_id, \
197                 name = $name, \
198                 description = $description, \
199                 events = $events, \
200                 mode = $mode, \
201                 priority = $priority, \
202                 timeout_ms = $timeout_ms, \
203                 failure_policy = $failure_policy, \
204                 enabled = $enabled, \
205                 created_at = time::now(), \
206                 updated_at = time::now(), \
207                 last_seen_at = NONE",
208            )
209            .bind(("id", id.to_string()))
210            .bind(("tenant_id", input.tenant_id.to_string()))
211            .bind(("name", input.name))
212            .bind(("description", input.description))
213            .bind(("events", input.events))
214            .bind(("mode", input.mode.as_str().to_string()))
215            .bind(("priority", i64::from(input.priority)))
216            .bind(("timeout_ms", i64::from(timeout_ms)))
217            .bind(("failure_policy", failure_policy.as_str().to_string()))
218            .bind(("enabled", input.enabled))
219            .await
220            .map_err(|e| classify_write_error(e, "reactor"))?;
221
222        let mut result = result
223            .check()
224            .map_err(|e| classify_write_error(e, "reactor"))?;
225        let rows: Vec<ReactorRow> = result.take(0).map_err(DbError::from)?;
226        let row = take_first_or_not_found(rows, "reactor", &id.to_string())?;
227        row.try_into_entry(id).map_err(Into::into)
228    }

251    async fn update(
252        &self,
253        tenant_id: Uuid,
254        id: Uuid,
255        input: UpdateReactor,
256    ) -> AxiamResult<Reactor> {
257        let mut set_clauses = vec!["updated_at = time::now()".to_string()];
258        let mut binds: Vec<(String, serde_json::Value)> = Vec::new();
259
260        if let Some(ref name) = input.name {
261            set_clauses.push("name = $name".into());
262            binds.push(("name".into(), serde_json::json!(name)));
263        }
264        if let Some(ref description) = input.description {
265            set_clauses.push("description = $description".into());
266            binds.push(("description".into(), serde_json::json!(description)));
267        }
268        if let Some(ref events) = input.events {
269            set_clauses.push("events = $events".into());
270            binds.push(("events".into(), serde_json::json!(events)));
271        }
272        if let Some(mode) = input.mode {
273            set_clauses.push("mode = $mode".into());
274            binds.push(("mode".into(), serde_json::json!(mode.as_str())));
275        }
276        if let Some(priority) = input.priority {
277            set_clauses.push("priority = $priority".into());
278            binds.push(("priority".into(), serde_json::json!(priority)));
279        }
280        if let Some(timeout_ms) = input.timeout_ms {
281            set_clauses.push("timeout_ms = $timeout_ms".into());
282            binds.push(("timeout_ms".into(), serde_json::json!(timeout_ms)));
283        }
284        if let Some(failure_policy) = input.failure_policy {
285            set_clauses.push("failure_policy = $failure_policy".into());
286            binds.push((
287                "failure_policy".into(),
288                serde_json::json!(failure_policy.as_str()),
289            ));
290        }
291        if let Some(enabled) = input.enabled {
292            set_clauses.push("enabled = $enabled".into());
293            binds.push(("enabled".into(), serde_json::json!(enabled)));
294        }
295
296        // `last_seen_at` is deliberately absent from this list. It is the
297        // consumer's heartbeat, not a setting, and `touch_last_seen` writes it
298        // without bumping `updated_at` — an operator reading "last modified"
299        // wants the last configuration change, not the last heartbeat.
300
301        let sql = format!(
302            "UPDATE type::record('reactor', $id) SET {} WHERE tenant_id = $tenant_id",
303            set_clauses.join(", ")
304        );
305
306        let db = self.db.current();
307        let mut query = db.query(&sql);
308        query = query
309            .bind(("id", id.to_string()))
310            .bind(("tenant_id", tenant_id.to_string()));
311        for (key, val) in binds {
312            query = query.bind((key, val));
313        }
314
315        let result = query
316            .await
317            .map_err(|e| classify_write_error(e, "reactor"))?;
318        let mut result = result
319            .check()
320            .map_err(|e| classify_write_error(e, "reactor"))?;
321        let rows: Vec<ReactorRow> = result.take(0).map_err(DbError::from)?;
322        let row = take_first_or_not_found(rows, "reactor", &id.to_string())?;
323        row.try_into_entry(id).map_err(Into::into)
324    }

326    async fn delete(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
327        let result = self
328            .db
329            .current()
330            .query(
331                "DELETE type::record('reactor', $id) \
332                 WHERE tenant_id = $tenant_id RETURN BEFORE",
333            )
334            .bind(("id", id.to_string()))
335            .bind(("tenant_id", tenant_id.to_string()))
336            .await
337            .map_err(DbError::from)?;
338
339        let mut result = result
340            .check()
341            .map_err(|e| DbError::Migration(e.to_string()))?;
342        let rows: Vec<ReactorRow> = result.take(0).map_err(DbError::from)?;
343        if rows.is_empty() {
344            return Err(DbError::NotFound {
345                entity: "reactor".into(),
346                id: id.to_string(),
347            }
348            .into());
349        }
350        Ok(())
351    }

353    async fn list(
354        &self,
355        tenant_id: Uuid,
356        pagination: Pagination,
357    ) -> AxiamResult<PaginatedResult<Reactor>> {
358        let tid = tenant_id.to_string();
359
360        let count_result = self
361            .db
362            .current()
363            .query(
364                "SELECT count() AS total FROM reactor \
365                 WHERE tenant_id = $tenant_id GROUP ALL",
366            )
367            .bind(("tenant_id", tid.clone()))
368            .await
369            .map_err(DbError::from)?;
370        let mut count_result = count_result
371            .check()
372            .map_err(|e| DbError::Migration(e.to_string()))?;
373        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
374
375        let data_result = self
376            .db
377            .current()
378            .query(
379                "SELECT meta::id(id) AS record_id, * FROM reactor \
380                 WHERE tenant_id = $tenant_id \
381                 ORDER BY priority ASC, record_id ASC \
382                 LIMIT $limit START $offset",
383            )
384            .bind(("tenant_id", tid))
385            .bind(("limit", pagination.limit))
386            .bind(("offset", pagination.offset))
387            .await
388            .map_err(DbError::from)?;
389        let mut data_result = data_result
390            .check()
391            .map_err(|e| DbError::Migration(e.to_string()))?;
392        let rows: Vec<ReactorRowWithId> = data_result.take(0).map_err(DbError::from)?;
393
394        let items = rows
395            .into_iter()
396            .map(ReactorRowWithId::try_into_entry)
397            .collect::<Result<Vec<_>, _>>()?;
398
399        Ok(paginate(items, count_rows, &pagination))
400    }

402    async fn get_enabled_by_event(
403        &self,
404        tenant_id: Uuid,
405        event: &str,
406    ) -> AxiamResult<Vec<Reactor>> {
407        // `enabled = true` is in the WHERE so it uses idx_reactor_tenant_enabled;
408        // the events membership is a post-filter on a set that is already
409        // per-tenant and small. Ordering is (priority, id) so the interceptor
410        // chain is a total order and therefore reproducible across replicas.
411        let result = self
412            .db
413            .current()
414            .query(
415                "SELECT meta::id(id) AS record_id, * FROM reactor \
416                 WHERE tenant_id = $tenant_id AND enabled = true \
417                 AND $event IN events \
418                 ORDER BY priority ASC, record_id ASC",
419            )
420            .bind(("tenant_id", tenant_id.to_string()))
421            .bind(("event", event.to_string()))
422            .await
423            .map_err(DbError::from)?;
424
425        let mut result = result
426            .check()
427            .map_err(|e| DbError::Migration(e.to_string()))?;
428        let rows: Vec<ReactorRowWithId> = result.take(0).map_err(DbError::from)?;
429        rows.into_iter()
430            .map(ReactorRowWithId::try_into_entry)
431            .collect::<Result<Vec<_>, _>>()
432            .map_err(Into::into)
433    }

435    async fn touch_last_seen(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
436        self.db
437            .current()
438            .query(
439                "UPDATE type::record('reactor', $id) SET last_seen_at = time::now() \
440                 WHERE tenant_id = $tenant_id",
441            )
442            .bind(("id", id.to_string()))
443            .bind(("tenant_id", tenant_id.to_string()))
444            .await
445            .map_err(DbError::from)?
446            .check()
447            .map_err(|e| DbError::Migration(e.to_string()))?;
448        // Deliberately does NOT 404 on a missing row. This is called from the
449        // consumer's heartbeat path, where the reactor may have been deleted
450        // between the consume and the touch; failing there would turn an
451        // ordinary race into a logged error on every subsequent message.
452        Ok(())
453    }

456#[cfg(test)]
457mod tests {
458    use super::*;
459
460    const ANY: Uuid = Uuid::nil();
461
462    /// A row written outside the API cannot promote itself to an interceptor.
463    #[test]
464    fn an_unrecognised_mode_is_demoted_to_listen() {
465        for raw in ["", "INTERCEPTOR", "veto", "block"] {
466            assert_eq!(
467                mode_or_safe_default(raw, ANY),
468                ReactorMode::Listen,
469                "mode {raw:?} must not resolve to intercept"
470            );
471        }
472        // The two real values still parse, casing and padding included.
473        assert_eq!(
474            mode_or_safe_default("intercept", ANY),
475            ReactorMode::Intercept
476        );
477        assert_eq!(mode_or_safe_default(" LISTEN ", ANY), ReactorMode::Listen);
478    }
479
480    /// An unreachable veto has not passed, so a malformed policy fails closed.
481    #[test]
482    fn an_unrecognised_failure_policy_fails_closed() {
483        for raw in ["", "open", "fail", "fail_sometimes"] {
484            assert_eq!(
485                failure_policy_or_safe_default(raw, ANY),
486                FailurePolicy::FailClosed,
487                "policy {raw:?} must not resolve to fail_open"
488            );
489        }
490        assert_eq!(
491            failure_policy_or_safe_default("fail_open", ANY),
492            FailurePolicy::FailOpen
493        );
494    }
495
496    /// The v29 ASSERT and this clamp are the same rule in the two places that
497    /// can enforce it. If `MAX_TIMEOUT_MS` and the schema ever disagree, the
498    /// dispatcher's wait is still bounded by this.
499    #[test]
500    fn a_stored_timeout_is_clamped_into_the_api_range() {
501        assert_eq!(clamp_timeout(0, ANY), 1);
502        assert_eq!(clamp_timeout(-5, ANY), 1);
503        assert_eq!(clamp_timeout(500, ANY), 500);
504        assert_eq!(
505            clamp_timeout(i64::from(MAX_TIMEOUT_MS), ANY),
506            MAX_TIMEOUT_MS
507        );
508        assert_eq!(clamp_timeout(600_000, ANY), MAX_TIMEOUT_MS);
509        assert_eq!(clamp_timeout(i64::MAX, ANY), MAX_TIMEOUT_MS);
510    }
