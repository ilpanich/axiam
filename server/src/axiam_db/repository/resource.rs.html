103/// The message `claim_parent` throws when the parent is gone. Matched on the
104/// way back out, so the caller sees `NotFound` rather than an opaque error.
105const PARENT_GONE: &str = "parent resource not found";
106
107/// SQL that claims `parent_id` for a child about to be attached to it (#308).
108///
109/// # Why a write, and why it must be this write
110///
111/// `resource.delete` guards itself by reading the `child_of` range for the
112/// resource being deleted and refusing if it is non-empty. That read cannot see
113/// an edge a concurrent transaction has not written yet, and snapshot isolation
114/// does not turn a range read into a conflict against a later insert — that is
115/// a phantom, and it needs serialisable isolation. So the guard alone loses the
116/// race, reproducibly (#308: trial 0 of every run on surrealkv).
117///
118/// What these engines DO detect reliably is two transactions writing the same
119/// key; `tools/surreal-race-probe` measures them doing it 21613 times in 40000
120/// contended attempts. `delete` already writes the parent's key — it deletes
121/// the row. So every path that attaches a child writes it too, and the engine's
122/// existing conflict detection decides the race instead of a query that cannot.
123///
124/// Two things here are load-bearing and easy to strip by accident:
125///
126/// 1. **The `THROW` is not an error-message nicety.** If the parent has already
127///    been deleted, the `UPDATE` matches nothing, and *nothing matching is not
128///    a conflict* — the create would sail past and write an edge to a row that
129///    no longer exists, which is precisely the orphan being prevented. Losing
130///    the race has to be an error, so it throws.
131/// 2. **It must be in the same transaction as the `RELATE`.** Split across two
132///    transactions, the parent could be deleted in the gap.
133///
134/// The counter's value is never read. Bumping it is the entire point.
135///
136/// Reads the already-bound `$parent_id` and `$tenant_id`, so callers add no
137/// bindings; the tenant guard is what stops a child being attached under
138/// another tenant's node.
139fn claim_parent() -> String {
140    format!(
141        "LET $parent = (UPDATE type::record('resource', $parent_id) \
142             SET child_epoch = (child_epoch ?? 0) + 1 \
143             WHERE tenant_id = $tenant_id RETURN VALUE id); \
144         IF array::len($parent) = 0 {{ THROW '{PARENT_GONE}'; }};"
145    )
146}

148/// The message a re-parent throws when its new parent's ancestors include the
149/// resource being moved. Matched on the way back out as the cycle refusal.
150const HIERARCHY_CYCLE: &str = "axiam:resource_hierarchy_cycle";
151
152/// The message a re-parent throws when the new parent's ancestor chain is at
153/// the depth cap — the same refusal `get_ancestors` gives such a chain.
154const HIERARCHY_TOO_DEEP: &str = "axiam:resource_hierarchy_too_deep";
155
156/// The message a re-parent throws when the tenant row it claims is not there.
157const HIERARCHY_NO_TENANT: &str = "axiam:resource_hierarchy_no_tenant";
158
159/// The cycle refusal a re-parent answers, whether the Rust self-parent check or
160/// the transaction's ancestor walk found it.
161fn cycle_refusal() -> AxiamError {
162    AxiamError::Validation {
163        message: "cycle detected in resource hierarchy".into(),
164    }
165}
166
167/// SQL that claims the tenant's resource hierarchy for a re-parent and walks
168/// the new parent's ancestors inside the same transaction (item 8 of the 1.1.0
169/// atomicity audit).
170///
171/// # Why the walk alone is not enough
172///
173/// The cycle check was a walk of the new parent's ancestors made before the
174/// write, in statements of its own; `claim_parent` writes only the new parent.
175/// Two re-parents X→Y and Z→W, where Z is an ancestor of Y and X an ancestor of
176/// W, each walked a hierarchy in which the other's edge did not exist yet, each
177/// wrote a different `child_of` edge and a different parent, and both
178/// committed: X → Y ⇝ Z → W ⇝ X, a cycle that `get_ancestors` then reports as a
179/// depth overflow on every authorization check under it. Moving the walk into
180/// the transaction does not change that by itself: SurrealDB 3.3.4 is
181/// snapshot-isolated with write-write conflict detection only, so each
182/// transaction still walks its own snapshot and nothing it read is checked at
183/// commit.
184///
185/// So every re-parent writes one row the tenant's other re-parents also write:
186/// `tenant.hierarchy_epoch` (schema v105). Two overlapping re-parents in one
187/// tenant conflict on it; the loser is retried on a fresh snapshot
188/// ([`retry_on_write_conflict`]), where its walk sees the winner's edge
189/// and refuses the cycle. The epoch's next value is a `LET` of its own and the
190/// `UPDATE` only assigns it (`role::bump_statement` gives the stack-depth
191/// reason). Its value is never read beyond that.
192///
193/// A create never needs this: a new resource has no descendants, so no edge to
194/// it can close a cycle. The same holds for the reserved `/network` subtree,
195/// whose writes only ever create nodes (`network.rs`), and for a detach, which
196/// only removes an edge.
197///
198/// Reads the bound `$tenant_id`, `$id` (the resource moved) and `$parent_id`
199/// (its new parent), and fills [`HIERARCHY_CLAIM_SLOTS`] result slots.
200fn claim_hierarchy() -> String {
201    format!(
202        "LET $hierarchy = (SELECT VALUE hierarchy_epoch ?? 0 \
203             FROM type::record('tenant', $tenant_id)); \
204         IF array::len($hierarchy) = 0 {{ THROW '{HIERARCHY_NO_TENANT}'; }}; \
205         LET $next_hierarchy_epoch = $hierarchy[0] + 1; \
206         UPDATE type::record('tenant', $tenant_id) SET hierarchy_epoch = $next_hierarchy_epoch; \
207         LET $ancestors = (SELECT VALUE meta::id(id) FROM array::flatten(\
208             type::record('resource', $parent_id).{{..{depth}+path}}(\
209                 ->child_of->(resource WHERE tenant_id = $tenant_id)))); \
210         IF array::len($ancestors) >= {depth} {{ THROW '{HIERARCHY_TOO_DEEP}'; }}; \
211         IF $id IN $ancestors {{ THROW '{HIERARCHY_CYCLE}'; }};",
212        depth = MAX_ANCESTOR_DEPTH,
213    )
214}
215
216/// How many result slots [`claim_hierarchy`] fills.
217const HIERARCHY_CLAIM_SLOTS: usize = 7;
218
219/// Turns a `claim_parent` throw into `NotFound`, and anything else into a
220/// database error.
221///
222/// Mirrors `delete`'s handling and for the same reason: a `THROW` fires on its
223/// own statement slot while the trailing statements report the generic "not
224/// executed due to a failed transaction", and `Response::check()` may surface
225/// either — so every statement error is scanned rather than just the first.
226fn parent_claim_error(errors: Vec<String>, parent_id: &str) -> AxiamError {
227    let combined = errors.join("; ");
228    if combined.contains(PARENT_GONE) {
229        return AxiamError::NotFound {
230            entity: "resource".into(),
231            id: parent_id.to_string(),
232        };
233    }
234    DbError::Migration(combined).into()
235}
236
237/// A re-parent's statement errors, read: the refusals its `THROW`s name first
238/// — a `THROW` fires on its own slot while the trailing statements report the
239/// generic "not executed due to a failed transaction", which is also a
240/// write-conflict marker — then what is left, classified, so an engine
241/// conflict is retried.
242fn reparent_error(errors: Vec<String>, tenant_id: Uuid, parent_id: &str) -> TxFail {
243    let combined = errors.join("; ");
244    if combined.contains(HIERARCHY_CYCLE) {
245        return TxFail::Refused(cycle_refusal());
246    }
247    if combined.contains(HIERARCHY_TOO_DEEP) {
248        // What `get_ancestors` answers for the same chain.
249        return TxFail::Refused(
250            DbError::Migration("resource hierarchy exceeds maximum depth — possible cycle".into())
251                .into(),
252        );
253    }
254    if combined.contains(HIERARCHY_NO_TENANT) {
255        return TxFail::Refused(AxiamError::NotFound {
256            entity: "tenant".into(),
257            id: tenant_id.to_string(),
258        });
259    }
260    if combined.contains(PARENT_GONE) {
261        return TxFail::Refused(AxiamError::NotFound {
262            entity: "resource".into(),
263            id: parent_id.to_string(),
264        });
265    }
266    TxFail::Db(classify_write_error(combined, "resource"))
267}

275    /// One attempt at [`ResourceRepository::update`]: the write, and for a
276    /// re-parent the hierarchy claim, the ancestor walk and the parent claim,
277    /// as one transaction.
278    async fn update_once(
279        &self,
280        tenant_id: Uuid,
281        id: Uuid,
282        input: UpdateResource,
283    ) -> Result<Resource, TxFail> {
284        let id_str = id.to_string();
285        let tenant_id_str = tenant_id.to_string();
286
287        let mut sets = Vec::new();
288        if input.name.is_some() {
289            sets.push("name = $name");
290        }
291        if input.resource_type.is_some() {
292            sets.push("resource_type = $resource_type");
293        }
294        if input.parent_id.is_some() {
295            sets.push("parent_id = $parent_id");
296        }
297        if input.metadata.is_some() {
298            sets.push("metadata = $metadata");
299        }
300        sets.push("updated_at = time::now()");
301
302        let mut query = format!(
303            "UPDATE type::record('resource', $id) SET {} \
304             WHERE tenant_id = $tenant_id",
305            sets.join(", ")
306        );
307
308        // If parent_id is being changed, update the child_of edge.
309        //
310        // Re-parenting mutates two-to-three statements (DELETE the old
311        // child_of edge, UPDATE the record, optionally RELATE the new edge).
312        // They are one transaction so a failure after the DELETE (e.g. the
313        // RELATE hitting a UNIQUE violation) cannot leave the resource orphaned
314        // with no rollback — the same class the resource.delete transaction
315        // closed. The child_of edge carries no flat tenant_id column, so the
316        // DELETE is tenant-scoped via a node-tenant guard on its `in` endpoint
317        // (in.tenant_id); without it a foreign-tenant resource id could strip
318        // another tenant's parent edge.
319        //
320        // Attaching to a new parent opens with two claims:
321        //
322        // - CQ-B08 and item 8 of the 1.1.0 atomicity audit: the tenant's
323        //   hierarchy (`claim_hierarchy`), and the walk of the new parent's
324        //   ancestors that refuses a cycle, inside this transaction.
325        // - #308: the new parent itself (`claim_parent`) — re-parenting inserts
326        //   into the NEW parent's `child_of` range, the phantom the `create`
327        //   path had — which also refuses a parent that is gone or another
328        //   tenant's. Only the new one: detaching a child removes an edge, and
329        //   a delete racing that either sees the child and refuses or does not
330        //   and has nothing to orphan.
331        //
332        // Result slots become: BEGIN=0, DELETE child_of=1, then either
333        // [hierarchy claim=2..=8, LET $parent=9, IF/THROW=10, UPDATE=11,
334        // RELATE=12] when a new parent is claimed, or [UPDATE=2] when the
335        // child is being detached.
336        let parent_id_changed = input.parent_id.is_some();
337        if parent_id_changed {
338            let claim = match &input.parent_id {
339                Some(Some(_)) => format!("{} {}", claim_hierarchy(), claim_parent()),
340                _ => String::new(),
341            };
342            // Both edge endpoints are bound record ids (`$child`, `$new_parent`),
343            // never text formatted into the statement.
344            query = format!(
345                "BEGIN TRANSACTION; \
346                 DELETE child_of WHERE in = $child AND in.tenant_id = $tenant_id; \
347                 {claim} {query}"
348            );
349
350            // If new parent is Some, create new child_of edge.
351            if matches!(input.parent_id, Some(Some(_))) {
352                query = format!("{query}; RELATE $child -> child_of -> $new_parent");
353            }
354
355            query = format!("{query}; COMMIT TRANSACTION");
356        }
357
358        let parent_id_str: Option<String> = input
359            .parent_id
360            .map(|opt| opt.map(|u| u.to_string()))
361            .unwrap_or(None);
362
363        let db = self.db.current();
364        let mut builder = db
365            .query(&query)
366            .bind(("id", id_str.clone()))
367            .bind(("tenant_id", tenant_id_str));
368        if parent_id_changed {
369            builder = builder.bind(("child", RecordId::new("resource", id_str.clone())));
370        }
371        if let Some(Some(new_parent)) = &input.parent_id {
372            builder = builder.bind((
373                "new_parent",
374                RecordId::new("resource", new_parent.to_string()),
375            ));
376        }
377
378        if let Some(name) = input.name {
379            builder = builder.bind(("name", name));
380        }
381        if let Some(resource_type) = input.resource_type {
382            builder = builder.bind(("resource_type", resource_type));
383        }
384        if input.parent_id.is_some() {
385            builder = builder.bind(("parent_id", parent_id_str));
386        }
387        if let Some(metadata) = input.metadata {
388            builder = builder.bind(("metadata", metadata));
389        }
390
391        let mut result = builder.await.map_err(DbError::from)?;
392
393        let errors = result.take_errors();
394        if !errors.is_empty() {
395            let msgs: Vec<String> = errors.into_values().map(|e| e.to_string()).collect();
396            return Err(match &input.parent_id {
397                Some(Some(new_parent)) => reparent_error(msgs, tenant_id, &new_parent.to_string()),
398                _ => TxFail::Db(classify_write_error(msgs.join("; "), "resource")),
399            });
400        }
401
402        // The UPDATE statement index depends on whether we wrapped the
403        // re-parent in a transaction, and on whether that re-parent claimed a
404        // new parent (see the slot map above).
405        let stmt_idx = match (parent_id_changed, &input.parent_id) {
406            (true, Some(Some(_))) => 2 + HIERARCHY_CLAIM_SLOTS + 2,
407            (true, _) => 2,
408            (false, _) => 0,
409        };
410        let rows: Vec<ResourceRow> = result.take(stmt_idx).map_err(DbError::from)?;
411        let row = take_first_or_not_found(rows, "resource", &id_str)?;
412
413        Ok(row_to_resource(row, id)?)
414    }

416    /// Create a resource and stamp the UMA client that registered it (X2).
417    ///
418    /// # Why this is an inherent method and not on `ResourceRepository`
419    ///
420    /// Provenance has exactly one writer — the resource-registration handler —
421    /// and putting it on the trait would offer it to every caller that holds
422    /// any `impl ResourceRepository`, which is the opposite of the point. The
423    /// handler reaches this repository concretely through `AppState`, so the
424    /// narrower home costs nothing and keeps `uma_registered_by` unreachable
425    /// from `create`/`update`.
426    ///
427    /// One statement rather than create-then-mark: a crash between two writes
428    /// would leave a resource that the Protection API registered but that
429    /// claims it did not, and an unmarked resource is indistinguishable from
430    /// one an administrator made by hand.
431    pub async fn create_uma_registered(
432        &self,
433        input: CreateResource,
434        client_id: &str,
435    ) -> AxiamResult<Resource> {
436        let id = new_id();
437        let id_str = id.to_string();
438        let tenant_id_str = input.tenant_id.to_string();
439        let parent_id_str = input.parent_id.map(|p| p.to_string());
440        let metadata = input
441            .metadata
442            .unwrap_or(serde_json::Value::Object(Default::default()));
443
444        let base = "CREATE type::record('resource', $id) SET \
445                    tenant_id = $tenant_id, \
446                    name = $name, resource_type = $resource_type, \
447                    parent_id = $parent_id, \
448                    metadata = $metadata, \
449                    uma_registered_by = $uma_registered_by";
450
451        // Same #308 parent claim as `create` — the Protection API attaches
452        // children through this path too, and an orphan made here is
453        // indistinguishable from one made there.
454        let (query, row_slot) = match parent_id_str.as_deref() {
455            Some(_) => (
456                format!(
457                    "BEGIN TRANSACTION; {claim} {base}; \
458                     RELATE $child -> child_of -> $parent_record; \
459                     COMMIT TRANSACTION",
460                    claim = claim_parent(),
461                ),
462                3,
463            ),
464            None => (base.to_string(), 0),
465        };
466
467        let mut result = self
468            .db
469            .current()
470            .query(query)
471            .bind(("id", id_str.clone()))
472            .bind(("tenant_id", tenant_id_str))
473            .bind(("name", input.name))
474            .bind(("resource_type", input.resource_type))
475            .bind(("parent_id", parent_id_str.clone()))
476            // The edge's endpoints, bound as record ids rather than formatted
477            // into the statement.
478            .bind(("child", RecordId::new("resource", id_str.clone())))
479            .bind((
480                "parent_record",
481                parent_id_str
482                    .clone()
483                    .map(|pid| RecordId::new("resource", pid)),
484            ))
485            .bind(("metadata", metadata))
486            .bind(("uma_registered_by", client_id.to_string()))
487            .await
488            .map_err(DbError::from)?;
489
490        let errors = result.take_errors();
491        if !errors.is_empty() {
492            let msgs: Vec<String> = errors.into_values().map(|e| e.to_string()).collect();
493            return Err(match parent_id_str.as_deref() {
494                Some(pid) => parent_claim_error(msgs, pid),
495                None => DbError::Migration(msgs.join("; ")).into(),
496            });
497        }
498
499        let rows: Vec<ResourceRow> = result.take(row_slot).map_err(DbError::from)?;
500        let row = take_first_or_not_found(rows, "resource", &id_str)?;
501
502        row_to_resource(row, id).map_err(Into::into)
503    }
504}

506impl<C: Connection> ResourceRepository for SurrealResourceRepository<C> {
507    async fn create(&self, input: CreateResource) -> AxiamResult<Resource> {
508        let id = new_id();
509        let id_str = id.to_string();
510        let tenant_id_str = input.tenant_id.to_string();
511        let parent_id_str = input.parent_id.map(|p| p.to_string());
512
513        let metadata = input
514            .metadata
515            .unwrap_or(serde_json::Value::Object(Default::default()));
516
517        // Create resource + optional child_of edge in one query.
518        let base = "CREATE type::record('resource', $id) SET \
519                    tenant_id = $tenant_id, \
520                    name = $name, resource_type = $resource_type, \
521                    parent_id = $parent_id, \
522                    metadata = $metadata";
523
524        // #308: attaching a child claims the parent inside the same
525        // transaction as the RELATE, so a concurrent `delete` of that parent
526        // has a key to conflict on. See `claim_parent` for why the range guard
527        // in `delete` cannot do this on its own. Without a parent there is no
528        // edge, nothing to race, and no transaction needed.
529        //
530        // Slots with a parent: BEGIN=0, LET $parent=1, IF/THROW=2, CREATE=3,
531        // RELATE=4, COMMIT=5. Without: CREATE=0.
532        let (query, row_slot) = match parent_id_str.as_deref() {
533            Some(_) => (
534                format!(
535                    "BEGIN TRANSACTION; {claim} {base}; \
536                     RELATE $child -> child_of -> $parent_record; \
537                     COMMIT TRANSACTION",
538                    claim = claim_parent(),
539                ),
540                3,
541            ),
542            None => (base.to_string(), 0),
543        };
544
545        let mut result = self
546            .db
547            .current()
548            .query(query)
549            .bind(("id", id_str.clone()))
550            .bind(("tenant_id", tenant_id_str))
551            .bind(("name", input.name))
552            .bind(("resource_type", input.resource_type))
553            .bind(("parent_id", parent_id_str.clone()))
554            // The edge's endpoints, bound as record ids rather than formatted
555            // into the statement.
556            .bind(("child", RecordId::new("resource", id_str.clone())))
557            .bind((
558                "parent_record",
559                parent_id_str
560                    .clone()
561                    .map(|pid| RecordId::new("resource", pid)),
562            ))
563            .bind(("metadata", metadata))
564            .await
565            .map_err(DbError::from)?;
566
567        // Every statement's error is scanned rather than only the first: a
568        // THROW fires on its own slot while the statements after it report the
569        // generic "not executed due to a failed transaction", and `check()` can
570        // surface either. Same reasoning as `delete`.
571        let errors = result.take_errors();
572        if !errors.is_empty() {
573            let msgs: Vec<String> = errors.into_values().map(|e| e.to_string()).collect();
574            return Err(match parent_id_str.as_deref() {
575                Some(pid) => parent_claim_error(msgs, pid),
576                None => DbError::Migration(msgs.join("; ")).into(),
577            });
578        }
579
580        let rows: Vec<ResourceRow> = result.take(row_slot).map_err(DbError::from)?;
581        let row = take_first_or_not_found(rows, "resource", &id_str)?;
582
583        row_to_resource(row, id).map_err(Into::into)
584    }

607    async fn update(
608        &self,
609        tenant_id: Uuid,
610        id: Uuid,
611        input: UpdateResource,
612    ) -> AxiamResult<Resource> {
613        // CQ-B08: self-parent is always a cycle, and needs no datastore to say
614        // so. Every other cycle is found by the walk inside the transaction.
615        if input.parent_id == Some(Some(id)) {
616            return Err(cycle_refusal());
617        }
618        // A re-parent onto a new parent claims the tenant's hierarchy row
619        // (`claim_hierarchy`), so two overlapping ones conflict; the loser runs
620        // again on a fresh snapshot, where its walk sees the winner's edge. A
621        // conflicted transaction committed nothing, so the replay cannot
622        // double-apply.
623        retry_on_write_conflict(|| self.update_once(tenant_id, id, input.clone()))
624            .await
625            .map_err(AxiamError::from)
626    }

628    async fn delete(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
629        let id_str = id.to_string();
630        let tenant_id_str = tenant_id.to_string();
631
632        // D-13/CQ-B46: fold the child-count guard into the SAME
633        // transaction as the deletes via a LET-capture (mirroring
634        // federation_login_state.rs's `LET $row = (...)` idiom). Before
635        // this fix, the guard ran as a separate `.query()` round-trip
636        // BEFORE the deletes — a classic TOCTOU: a concurrent child-create
637        // could commit in the gap between the count-check and the delete,
638        // leaving both a successful parent delete AND an orphaned child.
639        // Now the read-then-decide-then-delete happens atomically: either
640        // the transaction observes children and aborts before any DELETE
641        // runs, or it observes none and every DELETE below commits
642        // together.
643        //
644        // D-13/CQ-B07/SEC-058: child_of/on_resource edge tables carry no
645        // tenant_id field of their own (schema v19 defines
646        // idx_child_of_unique / idx_on_resource_unique on (in, out) only),
647        // so tenant scoping on those two DELETEs is expressed as a
648        // node-tenant subquery guard on the edge's in/out endpoint
649        // (in.tenant_id / out.tenant_id via graph-traversal dereference),
650        // not a flat WHERE clause. `scope` and the resource record itself
651        // DO carry their own tenant_id field, so those two keep their
652        // pre-existing flat predicates.
653        //
654        // Result slots: BEGIN=0, LET children=1, IF/THROW=2,
655        // DELETE child_of=3, DELETE on_resource=4, DELETE scope=5,
656        // DELETE resource=6, COMMIT=7. delete() returns Ok(()) — no row
657        // data to extract, .check() alone proves the transaction
658        // committed (or surfaces the child-guard THROW).
659        // SEC-104: the existence guard runs before the child-count guard, so a
660        // delete against a foreign or unknown id answers `NotFound` rather
661        // than "no children, deleted nothing, 204".
662        let guard = crate::helpers::delete_existence_guard("resource");
663        let query = format!(
664            "BEGIN TRANSACTION; \
665             {guard} \
666             LET $children = (SELECT VALUE id FROM child_of WHERE out = $node); \
667             IF array::len($children) > 0 {{ \
668                 THROW 'cannot delete resource with children'; \
669             }}; \
670             DELETE child_of WHERE \
671                 (in = $node AND in.tenant_id = $tenant_id) OR \
672                 (out = $node AND out.tenant_id = $tenant_id); \
673             DELETE on_resource WHERE out = $node AND out.tenant_id = $tenant_id; \
674             DELETE scope WHERE resource_id = $resource_id AND tenant_id = $tenant_id; \
675             DELETE type::record('resource', $id) WHERE tenant_id = $tenant_id; \
676             COMMIT TRANSACTION"
677        );
678
679        let id_str_for_error = id_str.clone();
680        let mut result = self
681            .db
682            .current()
683            .query(query)
684            .bind(("id", id_str.clone()))
685            // The node the edges hang off, bound as a record id.
686            .bind(("node", RecordId::new("resource", id_str.clone())))
687            .bind(("resource_id", id_str))
688            .bind(("tenant_id", tenant_id_str))
689            .await
690            .map_err(DbError::from)?;
691
692        // The child-guard THROW fires on its own statement slot, but the
693        // trailing DELETE/COMMIT slots report the generic "not executed due
694        // to a failed transaction" error, and Response::check() can surface
695        // one of those instead of the THROW text — which silently downgraded
696        // the child-guard rejection to an opaque Migration error
697        // (req14_tenant_isolation_test::resource_delete_with_children_rejected
698        // regressed when 29-02 moved the guard from a Rust pre-check into the
699        // transaction). Scan every statement error so the THROW message is
700        // reliably detected regardless of which slot check() would return.
701        let errors = result.take_errors();
702        if !errors.is_empty() {
703            let combined = errors
704                .into_values()
705                .map(|e| e.to_string())
706                .collect::<Vec<_>>()
707                .join("; ");
708            if combined.contains("cannot delete resource with children") {
709                return Err(AxiamError::Validation {
710                    message: "cannot delete resource with children".into(),
711                });
712            }
713            // SEC-104: checked after the child guard, because a resource that
714            // both exists and has children must keep answering with the more
715            // specific refusal.
716            if combined.contains(crate::helpers::DELETE_TARGET_MISSING) {
717                return Err(DbError::NotFound {
718                    entity: "resource".into(),
719                    id: id_str_for_error,
720                }
721                .into());
722            }
723            return Err(DbError::Migration(combined).into());
724        }
725
726        Ok(())
727    }

729    async fn list(
730        &self,
731        tenant_id: Uuid,
732        pagination: Pagination,
733    ) -> AxiamResult<PaginatedResult<Resource>> {
734        let tenant_id_str = tenant_id.to_string();
735
736        // Free-text filter, applied to BOTH queries below so the
737        // total counts matches rather than rows — a pager whose page
738        // count belongs to a different result set than the page it
739        // shows is worse than no pager. Empty when unsearched, so an
740        // unfiltered list runs exactly the query it always ran.
741        let search = search_filter(&pagination, &["name", "resource_type"]);
742        let search_term = search_bind(&pagination);
743
744        let mut count_result = self
745            .db
746            .current()
747            .query(format!(
748                "SELECT count() AS total FROM resource \
749                 WHERE tenant_id = $tenant_id{search} GROUP ALL"
750            ))
751            .bind(("tenant_id", tenant_id_str.clone()))
752            .bind(("search", search_term.clone()))
753            .await
754            .map_err(DbError::from)?;
755        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
756
757        let mut result = self
758            .db
759            .current()
760            .query(format!(
761                "SELECT meta::id(id) AS record_id, * FROM resource \
762                 WHERE tenant_id = $tenant_id{search} \
763                 ORDER BY created_at ASC \
764                 LIMIT $limit START $offset"
765            ))
766            .bind(("tenant_id", tenant_id_str))
767            .bind(("search", search_term))
768            .bind(("limit", pagination.limit))
769            .bind(("offset", pagination.offset))
770            .await
771            .map_err(DbError::from)?;
772
773        let rows: Vec<ResourceRowWithId> = result.take(0).map_err(DbError::from)?;
774
775        let items = rows
776            .into_iter()
777            .map(|row| row.try_into_resource())
778            .collect::<Result<Vec<_>, DbError>>()?;
779
780        Ok(paginate(items, count_rows, &pagination))
781    }

807    async fn get_ancestors(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<Vec<Resource>> {
808        let tenant_id_str = tenant_id.to_string();
809        let id_str = id.to_string();
810
811        // CQ-B13: walk the hierarchy in a SINGLE recursive graph query instead of
812        // one SELECT per level. The `child_of` edge points child -> parent, so
813        // `->child_of->resource` climbs one level; `.{..N+path}` follows that edge
814        // up to N times and yields the ordered path of ancestor records
815        // (nearest-first, EXCLUDING the starting resource — matching the previous
816        // loop's output). The `(resource WHERE tenant_id = $tenant_id)` node filter
817        // enforces tenant scoping: an ancestor in another tenant terminates the
818        // walk, exactly like the old per-level `WHERE tenant_id` guard.
819        //
820        // CQ-B08 preserved: `+path` does NOT deduplicate, so both a
821        // pathologically deep chain AND a cycle traverse until the depth cap and
822        // return exactly MAX_ANCESTOR_DEPTH records. The original loop errored as
823        // soon as it would step onto the MAX_ANCESTOR_DEPTH-th ancestor (i.e. when
824        // there are >= MAX_ANCESTOR_DEPTH ancestors), so we recurse to
825        // MAX_ANCESTOR_DEPTH and treat a result of that length as an
826        // overflow/cycle — surfacing the same error rather than silently
827        // truncating.
828        let query = format!(
829            "SELECT meta::id(id) AS record_id, tenant_id, name, resource_type, \
830                    parent_id, metadata, created_at, updated_at \
831             FROM array::flatten(\
832                 type::record('resource', $id).{{..{depth}+path}}(\
833                     ->child_of->(resource WHERE tenant_id = $tenant_id)));",
834            depth = MAX_ANCESTOR_DEPTH,
835        );
836
837        let mut result = self
838            .db
839            .current()
840            .query(query)
841            .bind(("id", id_str))
842            .bind(("tenant_id", tenant_id_str))
843            .await
844            .map_err(DbError::from)?;
845
846        let rows: Vec<ResourceRowWithId> = result.take(0).map_err(DbError::from)?;
847
848        if rows.len() >= MAX_ANCESTOR_DEPTH {
849            return Err(DbError::Migration(
850                "resource hierarchy exceeds maximum depth — possible cycle".into(),
851            )
852            .into());
853        }
854
855        rows.into_iter()
856            .map(|row| row.try_into_resource())
857            .collect::<Result<Vec<_>, DbError>>()
858            .map_err(Into::into)
859    }
