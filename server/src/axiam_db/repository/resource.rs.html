48impl ResourceRowWithId {
49    fn try_into_resource(self) -> Result<Resource, DbError> {
50        let id = Uuid::parse_str(&self.record_id)
51            .map_err(|e| DbError::Migration(format!("invalid UUID: {e}")))?;
52        let tenant_id = Uuid::parse_str(&self.tenant_id)
53            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
54        let parent_id = self
55            .parent_id
56            .map(|p| Uuid::parse_str(&p))
57            .transpose()
58            .map_err(|e| DbError::Migration(format!("invalid parent UUID: {e}")))?;
59        Ok(Resource {
60            id,
61            tenant_id,
62            name: self.name,
63            resource_type: self.resource_type,
64            parent_id,
65            metadata: self.metadata,
66            uma_registered_by: self.uma_registered_by,
67            created_at: self.created_at,
68            updated_at: self.updated_at,
69        })
70    }

94/// Maximum depth for ancestor traversal to prevent infinite loops.
95const MAX_ANCESTOR_DEPTH: usize = 50;
96
97/// `get_ancestors`' statement, over the resource id in `id_param` and the
98/// tenant in `tenant_param` — shared with the one-round-trip authorization read
99/// (`SurrealRoleRepository::authorization_inputs`) so the two cannot drift.
100/// See [`SurrealResourceRepository::get_ancestors`] for the shape and why it is
101/// one recursive query.
102pub(crate) fn ancestors_sql(id_param: &str, tenant_param: &str) -> String {
103    format!(
104        "SELECT meta::id(id) AS record_id, tenant_id, name, resource_type, \
105                parent_id, metadata, created_at, updated_at \
106         FROM array::flatten(\
107             type::record('resource', {id_param}).{{..{depth}+path}}(\
108                 ->child_of->(resource WHERE tenant_id = {tenant_param})));",
109        depth = MAX_ANCESTOR_DEPTH,
110    )
111}
112
113/// `get_ancestors`' answer from its rows: nearest-first, or the possible-cycle
114/// error when the walk reached the depth cap (CQ-B08).
115pub(crate) fn ancestors_from_rows(rows: Vec<ResourceRowWithId>) -> Result<Vec<Resource>, DbError> {
116    if rows.len() >= MAX_ANCESTOR_DEPTH {
117        return Err(DbError::Migration(
118            "resource hierarchy exceeds maximum depth — possible cycle".into(),
119        ));
120    }
121    rows.into_iter()
122        .map(|row| row.try_into_resource())
123        .collect()
124}

132/// The message `claim_parent` throws when the parent is gone. Matched on the
133/// way back out, so the caller sees `NotFound` rather than an opaque error.
134const PARENT_GONE: &str = "parent resource not found";
135
136/// SQL that claims `parent_id` for a child about to be attached to it (#308).
137///
138/// # Why a write, and why it must be this write
139///
140/// `resource.delete` guards itself by reading the `child_of` range for the
141/// resource being deleted and refusing if it is non-empty. That read cannot see
142/// an edge a concurrent transaction has not written yet, and snapshot isolation
143/// does not turn a range read into a conflict against a later insert — that is
144/// a phantom, and it needs serialisable isolation. So the guard alone loses the
145/// race, reproducibly (#308: trial 0 of every run on surrealkv).
146///
147/// What these engines DO detect reliably is two transactions writing the same
148/// key; `tools/surreal-race-probe` measures them doing it 21613 times in 40000
149/// contended attempts. `delete` already writes the parent's key — it deletes
150/// the row. So every path that attaches a child writes it too, and the engine's
151/// existing conflict detection decides the race instead of a query that cannot.
152///
153/// Two things here are load-bearing and easy to strip by accident:
154///
155/// 1. **The `THROW` is not an error-message nicety.** If the parent has already
156///    been deleted, the `UPDATE` matches nothing, and *nothing matching is not
157///    a conflict* — the create would sail past and write an edge to a row that
158///    no longer exists, which is precisely the orphan being prevented. Losing
159///    the race has to be an error, so it throws.
160/// 2. **It must be in the same transaction as the `RELATE`.** Split across two
161///    transactions, the parent could be deleted in the gap.
162///
163/// The counter's value is never read. Bumping it is the entire point.
164///
165/// Reads the already-bound `$parent_id` and `$tenant_id`, so callers add no
166/// bindings; the tenant guard is what stops a child being attached under
167/// another tenant's node.
168fn claim_parent() -> String {
169    format!(
170        "LET $parent = (UPDATE type::record('resource', $parent_id) \
171             SET child_epoch = (child_epoch ?? 0) + 1 \
172             WHERE tenant_id = $tenant_id RETURN VALUE id); \
173         IF array::len($parent) = 0 {{ THROW '{PARENT_GONE}'; }};"
174    )
175}

177/// The message a re-parent throws when its new parent's ancestors include the
178/// resource being moved. Matched on the way back out as the cycle refusal.
179const HIERARCHY_CYCLE: &str = "axiam:resource_hierarchy_cycle";
180
181/// The message a re-parent throws when the new parent's ancestor chain is at
182/// the depth cap — the same refusal `get_ancestors` gives such a chain.
183const HIERARCHY_TOO_DEEP: &str = "axiam:resource_hierarchy_too_deep";
184
185/// The message a re-parent throws when the tenant row it claims is not there.
186const HIERARCHY_NO_TENANT: &str = "axiam:resource_hierarchy_no_tenant";
187
188/// The cycle refusal a re-parent answers, whether the Rust self-parent check or
189/// the transaction's ancestor walk found it.
190fn cycle_refusal() -> AxiamError {
191    AxiamError::Validation {
192        message: "cycle detected in resource hierarchy".into(),
193    }
194}
195
196/// SQL that claims the tenant's resource hierarchy for a re-parent and walks
197/// the new parent's ancestors inside the same transaction (item 8 of the 1.1.0
198/// atomicity audit).
199///
200/// # Why the walk alone is not enough
201///
202/// The cycle check was a walk of the new parent's ancestors made before the
203/// write, in statements of its own; `claim_parent` writes only the new parent.
204/// Two re-parents X→Y and Z→W, where Z is an ancestor of Y and X an ancestor of
205/// W, each walked a hierarchy in which the other's edge did not exist yet, each
206/// wrote a different `child_of` edge and a different parent, and both
207/// committed: X → Y ⇝ Z → W ⇝ X, a cycle that `get_ancestors` then reports as a
208/// depth overflow on every authorization check under it. Moving the walk into
209/// the transaction does not change that by itself: SurrealDB 3.3.4 is
210/// snapshot-isolated with write-write conflict detection only, so each
211/// transaction still walks its own snapshot and nothing it read is checked at
212/// commit.
213///
214/// So every re-parent writes one row the tenant's other re-parents also write:
215/// `tenant.hierarchy_epoch` (schema v105). Two overlapping re-parents in one
216/// tenant conflict on it; the loser is retried on a fresh snapshot
217/// ([`retry_on_write_conflict`]), where its walk sees the winner's edge
218/// and refuses the cycle. The epoch's next value is a `LET` of its own and the
219/// `UPDATE` only assigns it (`role::bump_statement` gives the stack-depth
220/// reason). Its value is never read beyond that.
221///
222/// A create never needs this: a new resource has no descendants, so no edge to
223/// it can close a cycle. The same holds for the reserved `/network` subtree,
224/// whose writes only ever create nodes (`network.rs`), and for a detach, which
225/// only removes an edge.
226///
227/// Reads the bound `$tenant_id`, `$id` (the resource moved) and `$parent_id`
228/// (its new parent), and fills [`HIERARCHY_CLAIM_SLOTS`] result slots.
229fn claim_hierarchy() -> String {
230    format!(
231        "LET $hierarchy = (SELECT VALUE hierarchy_epoch ?? 0 \
232             FROM type::record('tenant', $tenant_id)); \
233         IF array::len($hierarchy) = 0 {{ THROW '{HIERARCHY_NO_TENANT}'; }}; \
234         LET $next_hierarchy_epoch = $hierarchy[0] + 1; \
235         UPDATE type::record('tenant', $tenant_id) SET hierarchy_epoch = $next_hierarchy_epoch; \
236         LET $ancestors = (SELECT VALUE meta::id(id) FROM array::flatten(\
237             type::record('resource', $parent_id).{{..{depth}+path}}(\
238                 ->child_of->(resource WHERE tenant_id = $tenant_id)))); \
239         IF array::len($ancestors) >= {depth} {{ THROW '{HIERARCHY_TOO_DEEP}'; }}; \
240         IF $id IN $ancestors {{ THROW '{HIERARCHY_CYCLE}'; }};",
241        depth = MAX_ANCESTOR_DEPTH,
242    )
243}
244
245/// How many result slots [`claim_hierarchy`] fills.
246const HIERARCHY_CLAIM_SLOTS: usize = 7;
247
248/// Turns a `claim_parent` throw into `NotFound`, and anything else into a
249/// database error.
250///
251/// Mirrors `delete`'s handling and for the same reason: a `THROW` fires on its
252/// own statement slot while the trailing statements report the generic "not
253/// executed due to a failed transaction", and `Response::check()` may surface
254/// either — so every statement error is scanned rather than just the first.
255fn parent_claim_error(errors: Vec<String>, parent_id: &str) -> AxiamError {
256    let combined = errors.join("; ");
257    if combined.contains(PARENT_GONE) {
258        return AxiamError::NotFound {
259            entity: "resource".into(),
260            id: parent_id.to_string(),
261        };
262    }
263    DbError::Migration(combined).into()
264}
265
266/// A re-parent's statement errors, read: the refusals its `THROW`s name first
267/// — a `THROW` fires on its own slot while the trailing statements report the
268/// generic "not executed due to a failed transaction", which is also a
269/// write-conflict marker — then what is left, classified, so an engine
270/// conflict is retried.
271fn reparent_error(errors: Vec<String>, tenant_id: Uuid, parent_id: &str) -> TxFail {
272    let combined = errors.join("; ");
273    if combined.contains(HIERARCHY_CYCLE) {
274        return TxFail::Refused(cycle_refusal());
275    }
276    if combined.contains(HIERARCHY_TOO_DEEP) {
277        // What `get_ancestors` answers for the same chain.
278        return TxFail::Refused(
279            DbError::Migration("resource hierarchy exceeds maximum depth — possible cycle".into())
280                .into(),
281        );
282    }
283    if combined.contains(HIERARCHY_NO_TENANT) {
284        return TxFail::Refused(AxiamError::NotFound {
285            entity: "tenant".into(),
286            id: tenant_id.to_string(),
287        });
288    }
289    if combined.contains(PARENT_GONE) {
290        return TxFail::Refused(AxiamError::NotFound {
291            entity: "resource".into(),
292            id: parent_id.to_string(),
293        });
294    }
295    TxFail::Db(classify_write_error(combined, "resource"))
296}

304    /// One attempt at [`ResourceRepository::update`]: the write, and for a
305    /// re-parent the hierarchy claim, the ancestor walk and the parent claim,
306    /// as one transaction.
307    async fn update_once(
308        &self,
309        tenant_id: Uuid,
310        id: Uuid,
311        input: UpdateResource,
312    ) -> Result<Resource, TxFail> {
313        let id_str = id.to_string();
314        let tenant_id_str = tenant_id.to_string();
315
316        let mut sets = Vec::new();
317        if input.name.is_some() {
318            sets.push("name = $name");
319        }
320        if input.resource_type.is_some() {
321            sets.push("resource_type = $resource_type");
322        }
323        if input.parent_id.is_some() {
324            sets.push("parent_id = $parent_id");
325        }
326        if input.metadata.is_some() {
327            sets.push("metadata = $metadata");
328        }
329        sets.push("updated_at = time::now()");
330
331        let mut query = format!(
332            "UPDATE type::record('resource', $id) SET {} \
333             WHERE tenant_id = $tenant_id",
334            sets.join(", ")
335        );
336
337        // If parent_id is being changed, update the child_of edge.
338        //
339        // Re-parenting mutates two-to-three statements (DELETE the old
340        // child_of edge, UPDATE the record, optionally RELATE the new edge).
341        // They are one transaction so a failure after the DELETE (e.g. the
342        // RELATE hitting a UNIQUE violation) cannot leave the resource orphaned
343        // with no rollback — the same class the resource.delete transaction
344        // closed. The child_of edge carries no flat tenant_id column, so the
345        // DELETE is tenant-scoped via a node-tenant guard on its `in` endpoint
346        // (in.tenant_id); without it a foreign-tenant resource id could strip
347        // another tenant's parent edge.
348        //
349        // Attaching to a new parent opens with two claims:
350        //
351        // - CQ-B08 and item 8 of the 1.1.0 atomicity audit: the tenant's
352        //   hierarchy (`claim_hierarchy`), and the walk of the new parent's
353        //   ancestors that refuses a cycle, inside this transaction.
354        // - #308: the new parent itself (`claim_parent`) — re-parenting inserts
355        //   into the NEW parent's `child_of` range, the phantom the `create`
356        //   path had — which also refuses a parent that is gone or another
357        //   tenant's. Only the new one: detaching a child removes an edge, and
358        //   a delete racing that either sees the child and refuses or does not
359        //   and has nothing to orphan.
360        //
361        // Result slots become: BEGIN=0, DELETE child_of=1, then either
362        // [hierarchy claim=2..=8, LET $parent=9, IF/THROW=10, UPDATE=11,
363        // RELATE=12] when a new parent is claimed, or [UPDATE=2] when the
364        // child is being detached.
365        let parent_id_changed = input.parent_id.is_some();
366        if parent_id_changed {
367            let claim = match &input.parent_id {
368                Some(Some(_)) => format!("{} {}", claim_hierarchy(), claim_parent()),
369                _ => String::new(),
370            };
371            // Both edge endpoints are bound record ids (`$child`, `$new_parent`),
372            // never text formatted into the statement.
373            query = format!(
374                "BEGIN TRANSACTION; \
375                 DELETE child_of WHERE in = $child AND in.tenant_id = $tenant_id; \
376                 {claim} {query}"
377            );
378
379            // If new parent is Some, create new child_of edge.
380            if matches!(input.parent_id, Some(Some(_))) {
381                query = format!("{query}; RELATE $child -> child_of -> $new_parent");
382            }
383
384            query = format!("{query}; COMMIT TRANSACTION");
385        }
386
387        let parent_id_str: Option<String> = input
388            .parent_id
389            .map(|opt| opt.map(|u| u.to_string()))
390            .unwrap_or(None);
391
392        let db = self.db.current();
393        let mut builder = db
394            .query(&query)
395            .bind(("id", id_str.clone()))
396            .bind(("tenant_id", tenant_id_str));
397        if parent_id_changed {
398            builder = builder.bind(("child", RecordId::new("resource", id_str.clone())));
399        }
400        if let Some(Some(new_parent)) = &input.parent_id {
401            builder = builder.bind((
402                "new_parent",
403                RecordId::new("resource", new_parent.to_string()),
404            ));
405        }
406
407        if let Some(name) = input.name {
408            builder = builder.bind(("name", name));
409        }
410        if let Some(resource_type) = input.resource_type {
411            builder = builder.bind(("resource_type", resource_type));
412        }
413        if input.parent_id.is_some() {
414            builder = builder.bind(("parent_id", parent_id_str));
415        }
416        if let Some(metadata) = input.metadata {
417            builder = builder.bind(("metadata", metadata));
418        }
419
420        let mut result = builder.await.map_err(DbError::from)?;
421
422        let errors = result.take_errors();
423        if !errors.is_empty() {
424            let msgs: Vec<String> = errors.into_values().map(|e| e.to_string()).collect();
425            return Err(match &input.parent_id {
426                Some(Some(new_parent)) => reparent_error(msgs, tenant_id, &new_parent.to_string()),
427                _ => TxFail::Db(classify_write_error(msgs.join("; "), "resource")),
428            });
429        }
430
431        // The UPDATE statement index depends on whether we wrapped the
432        // re-parent in a transaction, and on whether that re-parent claimed a
433        // new parent (see the slot map above).
434        let stmt_idx = match (parent_id_changed, &input.parent_id) {
435            (true, Some(Some(_))) => 2 + HIERARCHY_CLAIM_SLOTS + 2,
436            (true, _) => 2,
437            (false, _) => 0,
438        };
439        let rows: Vec<ResourceRow> = result.take(stmt_idx).map_err(DbError::from)?;
440        let row = take_first_or_not_found(rows, "resource", &id_str)?;
441
442        Ok(row_to_resource(row, id)?)
443    }

445    /// Create a resource and stamp the UMA client that registered it (X2).
446    ///
447    /// # Why this is an inherent method and not on `ResourceRepository`
448    ///
449    /// Provenance has exactly one writer — the resource-registration handler —
450    /// and putting it on the trait would offer it to every caller that holds
451    /// any `impl ResourceRepository`, which is the opposite of the point. The
452    /// handler reaches this repository concretely through `AppState`, so the
453    /// narrower home costs nothing and keeps `uma_registered_by` unreachable
454    /// from `create`/`update`.
455    ///
456    /// One statement rather than create-then-mark: a crash between two writes
457    /// would leave a resource that the Protection API registered but that
458    /// claims it did not, and an unmarked resource is indistinguishable from
459    /// one an administrator made by hand.
460    pub async fn create_uma_registered(
461        &self,
462        input: CreateResource,
463        client_id: &str,
464    ) -> AxiamResult<Resource> {
465        let id = new_id();
466        let id_str = id.to_string();
467        let tenant_id_str = input.tenant_id.to_string();
468        let parent_id_str = input.parent_id.map(|p| p.to_string());
469        let metadata = input
470            .metadata
471            .unwrap_or(serde_json::Value::Object(Default::default()));
472
473        let base = "CREATE type::record('resource', $id) SET \
474                    tenant_id = $tenant_id, \
475                    name = $name, resource_type = $resource_type, \
476                    parent_id = $parent_id, \
477                    metadata = $metadata, \
478                    uma_registered_by = $uma_registered_by";
479
480        // Same #308 parent claim as `create` — the Protection API attaches
481        // children through this path too, and an orphan made here is
482        // indistinguishable from one made there.
483        let (query, row_slot) = match parent_id_str.as_deref() {
484            Some(_) => (
485                format!(
486                    "BEGIN TRANSACTION; {claim} {base}; \
487                     RELATE $child -> child_of -> $parent_record; \
488                     COMMIT TRANSACTION",
489                    claim = claim_parent(),
490                ),
491                3,
492            ),
493            None => (base.to_string(), 0),
494        };
495
496        let mut result = self
497            .db
498            .current()
499            .query(query)
500            .bind(("id", id_str.clone()))
501            .bind(("tenant_id", tenant_id_str))
502            .bind(("name", input.name))
503            .bind(("resource_type", input.resource_type))
504            .bind(("parent_id", parent_id_str.clone()))
505            // The edge's endpoints, bound as record ids rather than formatted
506            // into the statement.
507            .bind(("child", RecordId::new("resource", id_str.clone())))
508            .bind((
509                "parent_record",
510                parent_id_str
511                    .clone()
512                    .map(|pid| RecordId::new("resource", pid)),
513            ))
514            .bind(("metadata", metadata))
515            .bind(("uma_registered_by", client_id.to_string()))
516            .await
517            .map_err(DbError::from)?;
518
519        let errors = result.take_errors();
520        if !errors.is_empty() {
521            let msgs: Vec<String> = errors.into_values().map(|e| e.to_string()).collect();
522            return Err(match parent_id_str.as_deref() {
523                Some(pid) => parent_claim_error(msgs, pid),
524                None => DbError::Migration(msgs.join("; ")).into(),
525            });
526        }
527
528        let rows: Vec<ResourceRow> = result.take(row_slot).map_err(DbError::from)?;
529        let row = take_first_or_not_found(rows, "resource", &id_str)?;
530
531        row_to_resource(row, id).map_err(Into::into)
532    }
533}

535impl<C: Connection> ResourceRepository for SurrealResourceRepository<C> {
536    async fn create(&self, input: CreateResource) -> AxiamResult<Resource> {
537        let id = new_id();
538        let id_str = id.to_string();
539        let tenant_id_str = input.tenant_id.to_string();
540        let parent_id_str = input.parent_id.map(|p| p.to_string());
541
542        let metadata = input
543            .metadata
544            .unwrap_or(serde_json::Value::Object(Default::default()));
545
546        // Create resource + optional child_of edge in one query.
547        let base = "CREATE type::record('resource', $id) SET \
548                    tenant_id = $tenant_id, \
549                    name = $name, resource_type = $resource_type, \
550                    parent_id = $parent_id, \
551                    metadata = $metadata";
552
553        // #308: attaching a child claims the parent inside the same
554        // transaction as the RELATE, so a concurrent `delete` of that parent
555        // has a key to conflict on. See `claim_parent` for why the range guard
556        // in `delete` cannot do this on its own. Without a parent there is no
557        // edge, nothing to race, and no transaction needed.
558        //
559        // Slots with a parent: BEGIN=0, LET $parent=1, IF/THROW=2, CREATE=3,
560        // RELATE=4, COMMIT=5. Without: CREATE=0.
561        let (query, row_slot) = match parent_id_str.as_deref() {
562            Some(_) => (
563                format!(
564                    "BEGIN TRANSACTION; {claim} {base}; \
565                     RELATE $child -> child_of -> $parent_record; \
566                     COMMIT TRANSACTION",
567                    claim = claim_parent(),
568                ),
569                3,
570            ),
571            None => (base.to_string(), 0),
572        };
573
574        let mut result = self
575            .db
576            .current()
577            .query(query)
578            .bind(("id", id_str.clone()))
579            .bind(("tenant_id", tenant_id_str))
580            .bind(("name", input.name))
581            .bind(("resource_type", input.resource_type))
582            .bind(("parent_id", parent_id_str.clone()))
583            // The edge's endpoints, bound as record ids rather than formatted
584            // into the statement.
585            .bind(("child", RecordId::new("resource", id_str.clone())))
586            .bind((
587                "parent_record",
588                parent_id_str
589                    .clone()
590                    .map(|pid| RecordId::new("resource", pid)),
591            ))
592            .bind(("metadata", metadata))
593            .await
594            .map_err(DbError::from)?;
595
596        // Every statement's error is scanned rather than only the first: a
597        // THROW fires on its own slot while the statements after it report the
598        // generic "not executed due to a failed transaction", and `check()` can
599        // surface either. Same reasoning as `delete`.
600        let errors = result.take_errors();
601        if !errors.is_empty() {
602            let msgs: Vec<String> = errors.into_values().map(|e| e.to_string()).collect();
603            return Err(match parent_id_str.as_deref() {
604                Some(pid) => parent_claim_error(msgs, pid),
605                None => DbError::Migration(msgs.join("; ")).into(),
606            });
607        }
608
609        let rows: Vec<ResourceRow> = result.take(row_slot).map_err(DbError::from)?;
610        let row = take_first_or_not_found(rows, "resource", &id_str)?;
611
612        row_to_resource(row, id).map_err(Into::into)
613    }

636    async fn update(
637        &self,
638        tenant_id: Uuid,
639        id: Uuid,
640        input: UpdateResource,
641    ) -> AxiamResult<Resource> {
642        // CQ-B08: self-parent is always a cycle, and needs no datastore to say
643        // so. Every other cycle is found by the walk inside the transaction.
644        if input.parent_id == Some(Some(id)) {
645            return Err(cycle_refusal());
646        }
647        // A re-parent onto a new parent claims the tenant's hierarchy row
648        // (`claim_hierarchy`), so two overlapping ones conflict; the loser runs
649        // again on a fresh snapshot, where its walk sees the winner's edge. A
650        // conflicted transaction committed nothing, so the replay cannot
651        // double-apply.
652        retry_on_write_conflict(|| self.update_once(tenant_id, id, input.clone()))
653            .await
654            .map_err(AxiamError::from)
655    }

657    async fn delete(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
658        let id_str = id.to_string();
659        let tenant_id_str = tenant_id.to_string();
660
661        // D-13/CQ-B46: fold the child-count guard into the SAME
662        // transaction as the deletes via a LET-capture (mirroring
663        // federation_login_state.rs's `LET $row = (...)` idiom). Before
664        // this fix, the guard ran as a separate `.query()` round-trip
665        // BEFORE the deletes — a classic TOCTOU: a concurrent child-create
666        // could commit in the gap between the count-check and the delete,
667        // leaving both a successful parent delete AND an orphaned child.
668        // Now the read-then-decide-then-delete happens atomically: either
669        // the transaction observes children and aborts before any DELETE
670        // runs, or it observes none and every DELETE below commits
671        // together.
672        //
673        // D-13/CQ-B07/SEC-058: child_of/on_resource edge tables carry no
674        // tenant_id field of their own (schema v19 defines
675        // idx_child_of_unique / idx_on_resource_unique on (in, out) only),
676        // so tenant scoping on those two DELETEs is expressed as a
677        // node-tenant subquery guard on the edge's in/out endpoint
678        // (in.tenant_id / out.tenant_id via graph-traversal dereference),
679        // not a flat WHERE clause. `scope` and the resource record itself
680        // DO carry their own tenant_id field, so those two keep their
681        // pre-existing flat predicates.
682        //
683        // Result slots: BEGIN=0, LET children=1, IF/THROW=2,
684        // DELETE child_of=3, DELETE on_resource=4, DELETE scope=5,
685        // DELETE resource=6, COMMIT=7. delete() returns Ok(()) — no row
686        // data to extract, .check() alone proves the transaction
687        // committed (or surfaces the child-guard THROW).
688        // SEC-104: the existence guard runs before the child-count guard, so a
689        // delete against a foreign or unknown id answers `NotFound` rather
690        // than "no children, deleted nothing, 204".
691        let guard = crate::helpers::delete_existence_guard("resource");
692        let query = format!(
693            "BEGIN TRANSACTION; \
694             {guard} \
695             LET $children = (SELECT VALUE id FROM child_of WHERE out = $node); \
696             IF array::len($children) > 0 {{ \
697                 THROW 'cannot delete resource with children'; \
698             }}; \
699             DELETE child_of WHERE \
700                 (in = $node AND in.tenant_id = $tenant_id) OR \
701                 (out = $node AND out.tenant_id = $tenant_id); \
702             DELETE on_resource WHERE out = $node AND out.tenant_id = $tenant_id; \
703             DELETE scope WHERE resource_id = $resource_id AND tenant_id = $tenant_id; \
704             DELETE type::record('resource', $id) WHERE tenant_id = $tenant_id; \
705             COMMIT TRANSACTION"
706        );
707
708        let id_str_for_error = id_str.clone();
709        let mut result = self
710            .db
711            .current()
712            .query(query)
713            .bind(("id", id_str.clone()))
714            // The node the edges hang off, bound as a record id.
715            .bind(("node", RecordId::new("resource", id_str.clone())))
716            .bind(("resource_id", id_str))
717            .bind(("tenant_id", tenant_id_str))
718            .await
719            .map_err(DbError::from)?;
720
721        // The child-guard THROW fires on its own statement slot, but the
722        // trailing DELETE/COMMIT slots report the generic "not executed due
723        // to a failed transaction" error, and Response::check() can surface
724        // one of those instead of the THROW text — which silently downgraded
725        // the child-guard rejection to an opaque Migration error
726        // (req14_tenant_isolation_test::resource_delete_with_children_rejected
727        // regressed when 29-02 moved the guard from a Rust pre-check into the
728        // transaction). Scan every statement error so the THROW message is
729        // reliably detected regardless of which slot check() would return.
730        let errors = result.take_errors();
731        if !errors.is_empty() {
732            let combined = errors
733                .into_values()
734                .map(|e| e.to_string())
735                .collect::<Vec<_>>()
736                .join("; ");
737            if combined.contains("cannot delete resource with children") {
738                return Err(AxiamError::Validation {
739                    message: "cannot delete resource with children".into(),
740                });
741            }
742            // SEC-104: checked after the child guard, because a resource that
743            // both exists and has children must keep answering with the more
744            // specific refusal.
745            if combined.contains(crate::helpers::DELETE_TARGET_MISSING) {
746                return Err(DbError::NotFound {
747                    entity: "resource".into(),
748                    id: id_str_for_error,
749                }
750                .into());
751            }
752            return Err(DbError::Migration(combined).into());
753        }
754
755        Ok(())
756    }

758    async fn list(
759        &self,
760        tenant_id: Uuid,
761        pagination: Pagination,
762    ) -> AxiamResult<PaginatedResult<Resource>> {
763        let tenant_id_str = tenant_id.to_string();
764
765        // Free-text filter, applied to BOTH queries below so the
766        // total counts matches rather than rows — a pager whose page
767        // count belongs to a different result set than the page it
768        // shows is worse than no pager. Empty when unsearched, so an
769        // unfiltered list runs exactly the query it always ran.
770        let search = search_filter(&pagination, &["name", "resource_type"]);
771        let search_term = search_bind(&pagination);
772
773        let mut count_result = self
774            .db
775            .current()
776            .query(format!(
777                "SELECT count() AS total FROM resource \
778                 WHERE tenant_id = $tenant_id{search} GROUP ALL"
779            ))
780            .bind(("tenant_id", tenant_id_str.clone()))
781            .bind(("search", search_term.clone()))
782            .await
783            .map_err(DbError::from)?;
784        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
785
786        let mut result = self
787            .db
788            .current()
789            .query(format!(
790                "SELECT meta::id(id) AS record_id, * FROM resource \
791                 WHERE tenant_id = $tenant_id{search} \
792                 ORDER BY created_at ASC \
793                 LIMIT $limit START $offset"
794            ))
795            .bind(("tenant_id", tenant_id_str))
796            .bind(("search", search_term))
797            .bind(("limit", pagination.limit))
798            .bind(("offset", pagination.offset))
799            .await
800            .map_err(DbError::from)?;
801
802        let rows: Vec<ResourceRowWithId> = result.take(0).map_err(DbError::from)?;
803
804        let items = rows
805            .into_iter()
806            .map(|row| row.try_into_resource())
807            .collect::<Result<Vec<_>, DbError>>()?;
808
809        Ok(paginate(items, count_rows, &pagination))
810    }

836    async fn get_ancestors(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<Vec<Resource>> {
837        let tenant_id_str = tenant_id.to_string();
838        let id_str = id.to_string();
839
840        // CQ-B13: walk the hierarchy in a SINGLE recursive graph query instead of
841        // one SELECT per level. The `child_of` edge points child -> parent, so
842        // `->child_of->resource` climbs one level; `.{..N+path}` follows that edge
843        // up to N times and yields the ordered path of ancestor records
844        // (nearest-first, EXCLUDING the starting resource — matching the previous
845        // loop's output). The `(resource WHERE tenant_id = $tenant_id)` node filter
846        // enforces tenant scoping: an ancestor in another tenant terminates the
847        // walk, exactly like the old per-level `WHERE tenant_id` guard.
848        //
849        // CQ-B08 preserved: `+path` does NOT deduplicate, so both a
850        // pathologically deep chain AND a cycle traverse until the depth cap and
851        // return exactly MAX_ANCESTOR_DEPTH records. The original loop errored as
852        // soon as it would step onto the MAX_ANCESTOR_DEPTH-th ancestor (i.e. when
853        // there are >= MAX_ANCESTOR_DEPTH ancestors), so we recurse to
854        // MAX_ANCESTOR_DEPTH and treat a result of that length as an
855        // overflow/cycle — surfacing the same error rather than silently
856        // truncating.
857        let mut result = self
858            .db
859            .current()
860            .query(ancestors_sql("$id", "$tenant_id"))
861            .bind(("id", id_str))
862            .bind(("tenant_id", tenant_id_str))
863            .await
864            .map_err(DbError::from)?;
865
866        let rows: Vec<ResourceRowWithId> = result.take(0).map_err(DbError::from)?;
867        ancestors_from_rows(rows).map_err(Into::into)
868    }
