1//! The revocation feed's read and prune side (T-39, T-143).
2//!
3//! The write side lives on [`crate::repository::SurrealSessionRepository`],
4//! because a revocation is published by the same call that performs it and a
5//! second repository in that path would be a second thing to forget.
6//!
7//! What is here is what the public handler and the background sweep need: the
8//! live entries, and a way to drop the dead ones. Both are deliberately
9//! tenant-**less**. The feed is a deployment-wide document served without
10//! authentication, and scoping it by tenant would mean either asking the caller
11//! which tenant it is interested in — turning the feed into the enumeration
12//! oracle T-244 spent an entire design avoiding — or storing a tenant id on
13//! rows whose whole property is that they identify nothing.
14
15use axiam_core::error::AxiamResult;
16use chrono::{DateTime, Utc};
17use surrealdb::Connection;
18use surrealdb_types::SurrealValue;
19
20use crate::error::DbError;
21use crate::handle::DbHandle;
22
23#[derive(Debug, SurrealValue)]
24struct RevokedSessionRow {
25    sid_hash: String,
26    /// Selected because the `ORDER BY` needs it, decoded because
27    /// `SurrealValue` is exhaustive, and used by nothing: the document
28    /// publishes hashes and a single deployment-wide `ttl`, never a per-entry
29    /// expiry. A per-entry expiry would say *when* each session was revoked,
30    /// which is one more thing an unauthenticated document would disclose.
31    #[allow(dead_code)]
32    expires_at: DateTime<Utc>,
33}
34
35/// Reads and prunes `revoked_session`.
36pub struct SurrealRevokedSessionRepository<C: Connection> {
37    db: DbHandle<C>,
38}
39
40impl<C: Connection> Clone for SurrealRevokedSessionRepository<C> {
41    fn clone(&self) -> Self {
42        Self {
43            db: self.db.clone(),
44        }
45    }
46}
47
48impl<C: Connection> SurrealRevokedSessionRepository<C> {
49    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
50        Self { db: db.into() }
51    }
52
53    /// Every entry that has not yet expired, as of `now`.
54    ///
55    /// Filtered on read rather than relying on the sweep, because the two
56    /// answer different questions: the sweep keeps the table small, this keeps
57    /// the **document** truthful. A deployment whose sweep is late must not
58    /// publish an entry for a session whose tokens have all expired — that is
59    /// a guard rejecting nothing, at the cost of disclosing one more
60    /// revocation than it needed to.
61    ///
62    /// Ordered by `expires_at` so the document is stable between polls that see
63    /// the same set, which is what lets an `ETag` mean anything.
64    pub async fn list_live(&self, now: DateTime<Utc>) -> AxiamResult<Vec<String>> {
65        let mut result = self
66            .db
67            .current()
68            .query(
69                // `expires_at` is in the projection because SurrealDB
70                // requires an `ORDER BY` idiom to be selected; the row struct
71                // reads only `sid_hash`, so nothing else escapes.
72                "SELECT sid_hash, expires_at FROM revoked_session \
73                 WHERE expires_at > $now \
74                 ORDER BY expires_at ASC",
75            )
76            .bind(("now", now))
77            .await
78            .map_err(DbError::from)?
79            .check()
80            .map_err(|e| DbError::Migration(e.to_string()))?;
81
82        let rows: Vec<RevokedSessionRow> = result.take(0).map_err(DbError::from)?;
83        Ok(rows.into_iter().map(|r| r.sid_hash).collect())
84    }
85
86    /// Delete every entry whose expiry has passed. Returns how many went.
87    ///
88    /// Called by the background sweep. This is the only deletion path on the
89    /// table, and unlike the audit log's it needs no ceremony: an expired
90    /// revocation entry is not a record of anything — every token it described
91    /// has expired on its own `exp` — and keeping it would be disclosure with
92    /// no remaining purpose.
93    pub async fn prune_expired(&self, now: DateTime<Utc>) -> AxiamResult<u64> {
94        let mut counted = self
95            .db
96            .current()
97            .query(
98                "SELECT count() AS total FROM revoked_session WHERE expires_at <= $now GROUP ALL",
99            )
100            .bind(("now", now))
101            .await
102            .map_err(DbError::from)?
103            .check()
104            .map_err(|e| DbError::Migration(e.to_string()))?;
105        let counts: Vec<crate::helpers::CountRow> = counted.take(0).map_err(DbError::from)?;
106        let count = counts.first().map_or(0, |c| c.total);
107
108        self.db
109            .current()
110            .query("DELETE revoked_session WHERE expires_at <= $now")
111            .bind(("now", now))
112            .await
113            .map_err(DbError::from)?
114            .check()
115            .map_err(|e| DbError::Migration(e.to_string()))?;
116
117        Ok(count)
118    }
119}
