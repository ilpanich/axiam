77/// Parse a `tenant_scope` list off an edge row.
78///
79/// Whether the list is absent and whether it is empty are different facts and
80/// stay different: `None` is an unrestricted assignment, and an empty `Some`
81/// would be one that reaches nothing. The write path refuses to create the
82/// second, but a row that somehow holds one must read back as itself rather
83/// than be silently widened to "everywhere".
84fn parse_tenant_scope(raw: Option<Vec<String>>) -> Result<Option<Vec<Uuid>>, DbError> {
85    raw.map(|ids| {
86        ids.iter()
87            .map(|s| parse_uuid(s, "tenant_scope"))
88            .collect::<Result<Vec<_>, _>>()
89    })
90    .transpose()
91}

147// ---------------------------------------------------------------------------
148// The role row's epochs: the network gate's reads and the write as one
149// transaction (R110W6-04, #701; schema v101)
150// ---------------------------------------------------------------------------
151//
152// The network gate (`axiam_authz::network_gate`) decides a role write on reads
153// it makes before the write: the role's `network:` grants, who holds it, its
154// `is_global`. Two writes to one role can each pass the gate on a state the
155// other is about to change, and the engine will not notice — SurrealDB 3.3.4
156// is snapshot-isolated with **write-write** conflict detection only (on
157// `surrealkv`, the engine production runs, a plain read is never validated at
158// commit), so two transactions that write disjoint keys — one `grants` edge,
159// one `has_role` edge — both commit. That is how, before this, a `network:`
160// grant and a group assignment of one role could both land (rule 2), a writer
161// could grant `network:join` to a role while assigning the role to itself
162// (H6), a role could be made global while gaining a `network:` grant (rule 1),
163// assigned with `inherit: false` while gaining a deny (rule 3), or a revoke
164// could delete a deny committed after its gate read none (§5.3 row 5).
165//
166// So every write that can change what those decisions read moves an epoch on
167// the role row, and every write the gate cleared **claims** the epochs the gate
168// read before anything else, inside its own transaction:
169//
170// * `grant_epoch` moves with a `network:` grant or revoke and an `is_global`
171//   write; `holder_epoch` with every assignment and unassignment.
172// * An assignment or unassignment claims `grant_epoch` (its rules read the
173//   role's grants; two assignments of one role do not invalidate each other).
174//   A `network:` grant or revoke, and a role delete, claim both. An `is_global`
175//   update claims `grant_epoch`.
176// * The claim is a compare inside the transaction (`THROW` → `409`, "the role
177//   changed since it was read"), and every claiming write also writes the row,
178//   so two overlapping transactions conflict on the key and the loser is
179//   retried on a fresh snapshot ([`retry_on_write_conflict`]), where the compare
180//   fails. A write that starts after the other committed fails the compare at
181//   once.
182//
183// A write the gate did not claim for (the seeder, a role or permission that is
184// not the tenant's) still moves its epoch, so a claimed write racing it loses.
185// Rule 2 is also re-asserted inside the grant and group-assignment
186// transactions themselves, for the writers that make no claim (the seeder's
187// reconcile, `seeder.rs`).
188
189/// `THROW`n when an epoch the write claimed has moved since the gate read it.
190pub(crate) const ROLE_EPOCH_STALE: &str = "axiam:role_epoch_stale";
191
192/// `THROW`n by a `network:` grant whose role a group holds when the
193/// transaction runs.
194pub(crate) const NETWORK_GRANT_TO_GROUP_HELD_ROLE: &str =
195    "axiam:network_grant_to_a_group_held_role";
196
197/// `THROW`n by a group assignment of a role that carries a `network:`
198/// permission when the transaction runs.
199pub(crate) const GROUP_ASSIGNMENT_OF_NETWORK_ROLE: &str =
200    "axiam:group_assignment_of_a_network_role";
201
202/// The SurrealQL regex of an action in the reserved `network:` namespace:
203/// exactly `axiam_core::models::network::is_network_action` — leading
204/// whitespace skipped, the prefix compared ASCII case-insensitively (`-u`, so
205/// that U+212A KELVIN SIGN is not a `k`, as it is not for the gate).
206pub(crate) const NETWORK_ACTION_REGEX: &str = r"/^\s*(?i-u:network:)/";
207
208/// The `409` a write refused by its claim or by the in-transaction rule 2
209/// answers.
210///
211/// "changed since it was read" is the sentence `NetworkConflictReason::of`
212/// reads as `stale_version`: the gate read the role before the other write
213/// landed, and a request sent again goes back through the gate, which decides
214/// on the role as it now is.
215pub(crate) fn network_role_raced(what_landed: &str) -> AxiamError {
216    AxiamError::Conflict {
217        reason: format!(
218            "the role changed since it was read ({what_landed}); read it again and retry"
219        ),
220    }
221}

223/// Which epochs a write compares against its claim.
224#[derive(Clone, Copy)]
225pub(crate) struct Claims {
226    pub(crate) grants: bool,
227    pub(crate) holders: bool,
228}
229
230/// The statements that check `claim` inside a transaction, for the role whose
231/// id is `role_id` (a UUID the caller rendered, never caller text). They read
232/// `$tid`, `$claimed`, `$claim_grants` and `$claim_holders` ([`claim_binds`]).
233///
234/// A role that is not the tenant's is not refused here: the write's own guard
235/// answers it as it always has.
236pub(crate) fn claim_statements(role_id: &str, claims: Claims) -> String {
237    let mut out = format!(
238        "LET $epochs = (SELECT VALUE [grant_epoch ?? 0, holder_epoch ?? 0] \
239             FROM role:`{role_id}` WHERE tenant_id = $tid); "
240    );
241    if claims.grants {
242        out.push_str(&format!(
243            "IF $claimed AND array::len($epochs) > 0 AND $epochs[0][0] != $claim_grants \
244                 {{ THROW '{ROLE_EPOCH_STALE}'; }}; "
245        ));
246    }
247    if claims.holders {
248        out.push_str(&format!(
249            "IF $claimed AND array::len($epochs) > 0 AND $epochs[0][1] != $claim_holders \
250                 {{ THROW '{ROLE_EPOCH_STALE}'; }}; "
251        ));
252    }
253    out
254}
255
256/// The values [`claim_statements`] reads: `(claimed, grants, holders)`.
257pub(crate) fn claim_binds(claim: Option<&RoleClaim>) -> (bool, i64, i64) {
258    claim.map_or((false, 0, 0), |c| {
259        (
260            true,
261            i64::try_from(c.epochs.grants).unwrap_or(i64::MAX),
262            i64::try_from(c.epochs.holders).unwrap_or(i64::MAX),
263        )
264    })
265}

267/// One of the role row's two epochs.
268#[derive(Clone, Copy)]
269pub(crate) enum Epoch {
270    /// `grant_epoch`: `network:` grants and revokes, `is_global` writes.
271    Grants,
272    /// `holder_epoch`: assignments and unassignments.
273    Holders,
274}
275
276impl Epoch {
277    /// The role row's field.
278    pub(crate) fn field(self) -> &'static str {
279        match self {
280            Self::Grants => "grant_epoch",
281            Self::Holders => "holder_epoch",
282        }
283    }
284
285    /// The statement that leaves the epoch's next value in `$next_epoch`,
286    /// read from the `$epochs` pair [`claim_statements`] read earlier in the
287    /// same transaction (and block): the same snapshot an in-place
288    /// `x = (x ?? 0) + 1` would read. A role that is not the tenant's reads no
289    /// pair, and its UPDATE matches nothing.
290    pub(crate) fn next_statement(self) -> &'static str {
291        match self {
292            Self::Grants => "LET $next_epoch = ($epochs[0][0] ?? 0) + 1; ",
293            Self::Holders => "LET $next_epoch = ($epochs[0][1] ?? 0) + 1; ",
294        }
295    }
296}
297
298/// The statements that move `epoch` on the role row. Must follow
299/// [`claim_statements`] in the same block, whose `$epochs` they read.
300///
301/// The arithmetic is a `LET` of its own, so the `UPDATE` only assigns a
302/// parameter: the embedded engine evaluates an expression in a `SET` inside
303/// its per-document write pipeline, nested several frames below the statement
304/// that runs it, and in a debug build that nesting alone took the seeder's
305/// grant past a 2 MiB thread stack (`directory_e2e`, PR #716). The values
306/// written are the same.
307pub(crate) fn bump_statement(role_id: &str, epoch: Epoch) -> String {
308    format!(
309        "{next}UPDATE role:`{role_id}` SET {field} = $next_epoch WHERE tenant_id = $tid; ",
310        next = epoch.next_statement(),
311        field = epoch.field(),
312    )
313}
314
315/// A transaction's statement errors, read for the refusals every role write
316/// shares. `Ok(None)`: nothing failed. `Ok(Some(e))`: a refusal to answer with.
317/// `Err`: what is left, classified (a duplicate is `AlreadyExists`, an engine
318/// conflict is retried by the caller's [`retry_on_write_conflict`]).
319///
320/// Every slot is read: a `THROW` fires on its own statement while the rest
321/// report "not executed due to a failed transaction", which is also a
322/// write-conflict marker — so the refusals are matched first.
323pub(crate) fn role_tx_outcome(
324    errors: std::collections::HashMap<usize, surrealdb::Error>,
325    entity: &str,
326    id: &str,
327    cross_tenant: &str,
328) -> Result<Option<AxiamError>, DbError> {
329    if errors.is_empty() {
330        return Ok(None);
331    }
332    let combined = errors
333        .into_values()
334        .map(|e| e.to_string())
335        .collect::<Vec<_>>()
336        .join("; ");
337    if combined.contains(ROLE_EPOCH_STALE) {
338        return Ok(Some(network_role_raced(
339            "another write to its grants, holders or scope landed first",
340        )));
341    }
342    if combined.contains(NETWORK_GRANT_TO_GROUP_HELD_ROLE) {
343        return Ok(Some(network_role_raced("a group was assigned it")));
344    }
345    if combined.contains(GROUP_ASSIGNMENT_OF_NETWORK_ROLE) {
346        return Ok(Some(network_role_raced(
347            "a network: permission was granted to it",
348        )));
349    }
350    if combined.contains("cross-tenant edge denied") {
351        return Ok(Some(AxiamError::AuthorizationDenied {
352            reason: cross_tenant.to_owned(),
353            action: None,
354            resource_id: None,
355        }));
356    }
357    if combined.contains(crate::helpers::DELETE_TARGET_MISSING) {
358        return Err(DbError::NotFound {
359            entity: entity.to_owned(),
360            id: id.to_owned(),
361        });
362    }
363    Err(classify_write_error(combined, entity))
364}

378    /// Look up a single role by name within a tenant (CQ-B42: replaces list()+scan pattern).
379    pub async fn get_by_name(&self, tenant_id: Uuid, name: &str) -> AxiamResult<Option<Role>> {
380        let tenant_str = tenant_id.to_string();
381        let mut result = self
382            .db
383            .current()
384            .query(
385                "SELECT meta::id(id) AS record_id, * FROM role \
386                 WHERE tenant_id = $tenant_id AND name = $name LIMIT 1",
387            )
388            .bind(("tenant_id", tenant_str))
389            .bind(("name", name.to_string()))
390            .await
391            .map_err(DbError::from)?;
392
393        let rows: Vec<RoleRowWithId> = result.take(0).map_err(DbError::from)?;
394        rows.into_iter()
395            .next()
396            .map(|row| row.try_into_role().map_err(AxiamError::from))
397            .transpose()
398    }

400    /// [`RoleRepository::update`], claiming `claim`.
401    ///
402    /// A write of `is_global` moves `grant_epoch` and, with a claim, compares
403    /// it (rule 1 reads the role's grants; a `network:` grant reads
404    /// `is_global`). A write of `name` or `description` alone concerns no
405    /// network decision and compares nothing — but it writes the row, so a
406    /// claimed write racing it is retried, never mistaken.
407    ///
408    /// # Errors
409    ///
410    /// As the trait method, plus `Conflict` ("the role changed since it was
411    /// read") when the claim's `grant_epoch` has moved.
412    pub async fn update_claimed(
413        &self,
414        tenant_id: Uuid,
415        id: Uuid,
416        input: UpdateRole,
417        claim: Option<RoleClaim>,
418    ) -> AxiamResult<Role> {
419        let id_str = id.to_string();
420        let tenant_id_str = tenant_id.to_string();
421
422        let mut sets = Vec::new();
423        if input.name.is_some() {
424            sets.push("name = $name");
425        }
426        if input.description.is_some() {
427            sets.push("description = $description");
428        }
429        let claim_check = if input.is_global.is_some() {
430            sets.push("is_global = $is_global");
431            // The next value is a `LET` of its own (`Epoch::next_statement`),
432            // for the reason `bump_statement` gives.
433            sets.push("grant_epoch = $next_epoch");
434            let mut statements = claim_statements(
435                &id_str,
436                Claims {
437                    grants: true,
438                    holders: false,
439                },
440            );
441            statements.push_str(Epoch::Grants.next_statement());
442            statements
443        } else {
444            String::new()
445        };
446        sets.push("updated_at = time::now()");
447        // Result slots: BEGIN, then — when `is_global` is written — the
448        // claim's LET and IF and the next epoch's LET, then the UPDATE.
449        let update_slot = if claim_check.is_empty() { 1 } else { 4 };
450
451        let query = format!(
452            "BEGIN TRANSACTION; \
453             {claim_check}\
454             UPDATE type::record('role', $id) SET {} \
455             WHERE tenant_id = $tenant_id; \
456             COMMIT TRANSACTION",
457            sets.join(", ")
458        );
459        let (claimed, claim_grants, claim_holders) = claim_binds(claim.as_ref());
460
461        let outcome = retry_on_write_conflict(|| async {
462            let db = self.db.current();
463            let mut builder = db
464                .query(query.as_str())
465                .bind(("id", id_str.clone()))
466                .bind(("tenant_id", tenant_id_str.clone()))
467                .bind(("tid", tenant_id_str.clone()))
468                .bind(("claimed", claimed))
469                .bind(("claim_grants", claim_grants))
470                .bind(("claim_holders", claim_holders));
471            if let Some(name) = &input.name {
472                builder = builder.bind(("name", name.clone()));
473            }
474            if let Some(description) = &input.description {
475                builder = builder.bind(("description", description.clone()));
476            }
477            if let Some(is_global) = input.is_global {
478                builder = builder.bind(("is_global", is_global));
479            }
480            let mut result = builder.await.map_err(DbError::from)?;
481            if let Some(refusal) = role_tx_outcome(result.take_errors(), "role", &id_str, "")? {
482                return Ok(Err(refusal));
483            }
484            let rows: Vec<RoleRow> = result.take(update_slot).map_err(DbError::from)?;
485            Ok::<_, DbError>(Ok(rows))
486        })
487        .await;
488        let rows = outcome.map_err(AxiamError::from)??;
489
490        let row = rows.into_iter().next().ok_or_else(|| DbError::NotFound {
491            entity: "role".into(),
492            id: id_str,
493        })?;
494
495        let tenant_id = Uuid::parse_str(&row.tenant_id)
496            .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
497
498        Ok(Role {
499            id,
500            tenant_id,
501            name: row.name,
502            description: row.description,
503            is_global: row.is_global,
504            created_at: row.created_at,
505            updated_at: row.updated_at,
506        })
507    }

509    /// [`RoleRepository::delete`], claiming `claim` (both epochs: the gate's
510    /// rule reads the role's grants and whether the writer holds it).
511    ///
512    /// # Errors
513    ///
514    /// As the trait method, plus `Conflict` ("the role changed since it was
515    /// read") when an epoch the claim names has moved.
516    pub async fn delete_claimed(
517        &self,
518        tenant_id: Uuid,
519        id: Uuid,
520        claim: Option<RoleClaim>,
521    ) -> AxiamResult<()> {
522        let id_str = id.to_string();
523        let tenant_id_str = tenant_id.to_string();
524
525        // D-13/CQ-B07/SEC-058: has_role/grants edge tables carry no
526        // tenant_id field of their own (schema v19 defines
527        // idx_has_role_unique / idx_grants_unique on (in, out) only), so
528        // tenant scoping on these two DELETEs is expressed as a
529        // node-tenant subquery guard on the edge's out/in endpoint record
530        // (out.tenant_id / in.tenant_id — dereferencing the linked
531        // record's field via graph traversal, the same syntax already used
532        // by get_user_role_assignments above) rather than a flat WHERE
533        // clause. Without this guard, a caller supplying a foreign-tenant
534        // role id could strip another tenant's has_role/grants edges even
535        // though the final record DELETE (which does carry a flat
536        // tenant_id predicate) would simply no-op for not matching the
537        // caller's tenant.
538        //
539        // All three deletes run inside one transaction so a concurrent
540        // reader never observes a partially-deleted role; the claim is
541        // compared in the same transaction, and the role row it deletes is
542        // the row every other role write moves an epoch on.
543        // SEC-104 — see `helpers::delete_existence_guard`.
544        let guard = crate::helpers::delete_existence_guard("role");
545        let claim_check = claim_statements(
546            &id_str,
547            Claims {
548                grants: true,
549                holders: true,
550            },
551        );
552        let query = format!(
553            "BEGIN TRANSACTION; \
554             {guard} \
555             {claim_check}\
556             DELETE has_role WHERE out = role:`{id_str}` AND out.tenant_id = $tenant_id; \
557             DELETE grants WHERE in = role:`{id_str}` AND in.tenant_id = $tenant_id; \
558             DELETE type::record('role', $id) WHERE tenant_id = $tenant_id; \
559             COMMIT TRANSACTION"
560        );
561        let (claimed, claim_grants, claim_holders) = claim_binds(claim.as_ref());
562
563        let outcome = retry_on_write_conflict(|| async {
564            let mut result = self
565                .db
566                .current()
567                .query(query.as_str())
568                .bind(("id", id_str.clone()))
569                .bind(("tenant_id", tenant_id_str.clone()))
570                .bind(("tid", tenant_id_str.clone()))
571                .bind(("claimed", claimed))
572                .bind(("claim_grants", claim_grants))
573                .bind(("claim_holders", claim_holders))
574                .await
575                .map_err(DbError::from)?;
576            role_tx_outcome(result.take_errors(), "role", &id_str, "")
577        })
578        .await;
579        match outcome.map_err(AxiamError::from)? {
580            None => Ok(()),
581            Some(refusal) => Err(refusal),
582        }
583    }

585    /// The role's two epochs, `None` when it is not the tenant's (the
586    /// [`RoleRepository::get_role_epochs`] override).
587    async fn read_role_epochs(
588        &self,
589        tenant_id: Uuid,
590        role_id: Uuid,
591    ) -> AxiamResult<Option<RoleEpochs>> {
592        let mut result = self
593            .db
594            .current()

776    /// Shared body of `get_role_user_assignments` / `get_role_group_assignments`.
777    ///
778    /// `subject_table` is one of the two literals those methods pass — it is
779    /// never caller input, which is why it can be interpolated into the query
780    /// text. It has to be: a table name is not a bindable value in SurrealQL,
781    /// and `type::table()` cannot appear in a `FROM` clause here.
782    ///
783    /// Anchoring the subject set on its own table is what discriminates a user
784    /// edge from a group edge — `has_role` carries both, and `in` is the only
785    /// thing that tells them apart. That is the same idiom `get_role_user_ids`
786    /// uses; the difference is that the second statement reads `resource_id`
787    /// off the edge instead of throwing it away.
788    ///
789    /// The `LET` matters for the same reason it does in
790    /// `get_user_role_assignments`: evaluated inline, the planner cannot see a
791    /// constant on the right-hand side of `in IN (...)` and walks every
792    /// `has_role` edge in the database. Bound first, the set is a constant and
793    /// the predicate is an index lookup. Note the subject set is bounded by the
794    /// role's own assignees, not by the tenant's whole user table.
795    ///
796    /// Statement indices: 0 = LET, 1 = SELECT.
797    /// Every role a principal holds — assigned directly, or inherited through a
798    /// group it belongs to.
799    ///
800    /// `subject_table` is `user` or `service_account`. Both traverse
801    /// `member_of` the same way, because a group is a collection of principals
802    /// and a machine identity in one inherits its roles exactly as a person
803    /// does. That is the whole reason to put a fleet of devices in a group: it
804    /// is granted and revoked as one thing.
805    async fn subject_roles(
806        &self,
807        subject_table: &str,
808        tenant_id: Uuid,
809        subject_id: Uuid,
810    ) -> AxiamResult<Vec<Role>> {
811        let tenant_id_str = tenant_id.to_string();
812        let subject_id_str = subject_id.to_string();
813
814        // Two queries: direct roles + roles inherited through group membership.
815        let mut result = self
816            .db
817            .current()
818            .query(format!(
819                "SELECT meta::id(id) AS record_id, * FROM role \
820                 WHERE tenant_id = $tenant_id \
821                 AND id IN (\
822                     SELECT VALUE out FROM has_role \
823                     WHERE in = type::record('{subject_table}', $subject_id)\
824                 ); \
825                 SELECT meta::id(id) AS record_id, * FROM role \
826                 WHERE tenant_id = $tenant_id \
827                 AND id IN (\
828                     SELECT VALUE out FROM has_role \
829                     WHERE in IN (\
830                         SELECT VALUE out FROM member_of \
831                         WHERE in = type::record('{subject_table}', $subject_id)\
832                     )\
833                 );"
834            ))
835            .bind(("tenant_id", tenant_id_str))
836            .bind(("subject_id", subject_id_str))
837            .await
838            .map_err(DbError::from)?;
839
840        let direct: Vec<RoleRowWithId> = result.take(0).map_err(DbError::from)?;
841        let inherited: Vec<RoleRowWithId> = result.take(1).map_err(DbError::from)?;
842
843        // Merge and deduplicate by record_id.
844        let mut seen = std::collections::HashSet::new();
845        let mut roles = Vec::new();
846        for row in direct.into_iter().chain(inherited) {
847            if seen.insert(row.record_id.clone()) {
848                roles.push(row.try_into_role()?);
849            }
850        }
851
852        Ok(roles)
853    }

855    /// Create a `has_role` edge from any principal table.
856    ///
857    /// `subject_table` is `user`, `group` or `service_account`. The three were
858    /// three near-identical copies of this query, and adding service accounts
859    /// would have made a fourth — which is how the tenant guard, the duplicate
860    /// classification and the resource scoping drift apart between principal
861    /// kinds. `has_role` has always been declared `User/ServiceAccount/Group ->
862    /// Role`; nothing about the edge itself distinguishes them.
863    ///
864    /// One transaction (R110W6-04, the notes above [`ROLE_EPOCH_STALE`]): the
865    /// tenant guard, the `claim` on `grant_epoch` when the gate made one, for a
866    /// group the rule-2 re-check, the `holder_epoch` move and the edge.
867    ///
868    /// `subject_label` only names the subject in the denial message.
869    #[allow(clippy::too_many_arguments)]
870    async fn relate_subject_to_role(
871        &self,
872        subject_table: &str,
873        subject_label: &str,
874        tenant_id: Uuid,
875        subject_id: Uuid,
876        role_id: Uuid,
877        scope: AssignmentScope,
878        claim: Option<&RoleClaim>,
879    ) -> AxiamResult<()> {
880        let subject_id_str = subject_id.to_string();
881        let role_id_str = role_id.to_string();
882        let resource_id_str = scope.resource_id.map(|r| r.to_string());
883        // Bound as `Option<Vec<String>>` so an absent scope lands as SurrealDB
884        // `NONE` and not as an empty array: those are the two different answers
885        // "reaches everywhere the role does" and "reaches nothing", and the
886        // read path keeps them apart too (see `parse_tenant_scope`).
887        let tenant_scope_strs = scope
888            .tenant_scope
889            .map(|ids| ids.iter().map(Uuid::to_string).collect::<Vec<_>>());
890        // §4.2 rule 2, re-asserted for a group whatever the claim: a role
891        // carrying a `network:` permission is never held by a group.
892        let rule_2 = if subject_table == "group" {
893            format!(
894                "LET $network = (SELECT VALUE id FROM grants \
895                     WHERE in = role:`{role_id_str}` \
896                     AND string::matches(out.action ?? '', {NETWORK_ACTION_REGEX}) LIMIT 1); \
897                 IF array::len($network) > 0 {{ THROW '{GROUP_ASSIGNMENT_OF_NETWORK_ROLE}'; }}; "
898            )
899        } else {
900            String::new()
901        };
902        let claim_check = claim_statements(
903            &role_id_str,
904            Claims {
905                grants: true,
906                holders: false,
907            },
908        );
909        let bump = bump_statement(&role_id_str, Epoch::Holders);
910
911        // CQ-B07: verify both endpoints belong to the same tenant before RELATE.
912        let query = format!(
913            "BEGIN TRANSACTION; \
914             LET $s = (SELECT id FROM {subject_table}:`{subject_id_str}` WHERE tenant_id = $tid); \
915             LET $r = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid); \
916             IF array::len($s) = 0 OR array::len($r) = 0 {{ \
917                 THROW 'cross-tenant edge denied'; \
918             }}; \
919             {claim_check}{rule_2}{bump}\
920             RELATE {subject_table}:`{subject_id_str}` -> has_role -> role:`{role_id_str}` \
921             SET resource_id = $resource_id, tenant_scope = $tenant_scope, inherit = $inherit; \
922             COMMIT TRANSACTION"
923        );
924        let (claimed, claim_grants, claim_holders) = claim_binds(claim);
925        let cross_tenant = format!("cross-tenant {subject_label} role assignment denied");
926
927        let outcome = retry_on_write_conflict(|| async {
928            let mut result = self
929                .db
930                .current()
931                .query(query.as_str())
932                .bind(("tid", tenant_id.to_string()))
933                .bind(("resource_id", resource_id_str.clone()))
934                .bind(("tenant_scope", tenant_scope_strs.clone()))
935                // Written only when it departs from the default, so an ordinary
936                // assignment lands exactly as it did before v66 — NONE, read back
937                // as `true` — and a pre-v66 edge and a post-v66 one are
938                // indistinguishable unless the caller asked for the difference.
939                .bind(("inherit", (!scope.inherit).then_some(false)))
940                .bind(("claimed", claimed))
941                .bind(("claim_grants", claim_grants))
942                .bind(("claim_holders", claim_holders))
943                .await
944                .map_err(DbError::from)?;
945            // QUAL-03/D-09: a duplicate has_role edge violates the
946            // idx_has_role_unique UNIQUE(in,out) index — classified (and only
947            // that) as 409.
948            role_tx_outcome(
949                result.take_errors(),
950                "role_assignment",
951                &role_id_str,
952                &cross_tenant,
953            )
954        })
955        .await;
956        match outcome.map_err(AxiamError::from)? {
957            None => Ok(()),
958            Some(refusal) => Err(refusal),
959        }
960    }
961
962    /// Remove a `has_role` edge from any principal table.
963    ///
964    /// The inverse of [`Self::relate_subject_to_role`], and scoped the same way:
965    /// a `None` resource removes the *global* grant specifically, not every
966    /// grant of that role, so revoking one resource-scoped assignment cannot
967    /// silently take the others with it. In one transaction with the `claim` on
968    /// `grant_epoch` and the `holder_epoch` move, as the assignment is.
969    #[allow(clippy::too_many_arguments)]
970    async fn unrelate_subject_from_role(
971        &self,
972        subject_table: &str,
973        subject_label: &str,
974        tenant_id: Uuid,
975        subject_id: Uuid,
976        role_id: Uuid,
977        resource_id: Option<Uuid>,
978        claim: Option<&RoleClaim>,
979    ) -> AxiamResult<()> {
980        let subject_id_str = subject_id.to_string();
981        let role_id_str = role_id.to_string();
982        let resource_id_str = resource_id.map(|r| r.to_string());
983
984        let scope_clause = if resource_id_str.is_some() {
985            "resource_id = $resource_id"
986        } else {
987            "resource_id = NONE"
988        };
989        let claim_check = claim_statements(
990            &role_id_str,
991            Claims {
992                grants: true,
993                holders: false,
994            },
995        );
996        let bump = bump_statement(&role_id_str, Epoch::Holders);
997
998        // CQ-B07: verify both endpoints belong to the same tenant before DELETE.
999        let query = format!(
1000            "BEGIN TRANSACTION; \
1001             LET $s = (SELECT id FROM {subject_table}:`{subject_id_str}` WHERE tenant_id = $tid); \
1002             LET $r = (SELECT id FROM role:`{role_id_str}` WHERE tenant_id = $tid); \
1003             IF array::len($s) = 0 OR array::len($r) = 0 {{ \
1004                 THROW 'cross-tenant edge denied'; \
1005             }}; \
1006             {claim_check}{bump}\
1007             DELETE has_role WHERE \
1008             in = {subject_table}:`{subject_id_str}` AND \
1009             out = role:`{role_id_str}` AND \
1010             {scope_clause}; \
1011             COMMIT TRANSACTION"
1012        );
1013        let (claimed, claim_grants, claim_holders) = claim_binds(claim);
1014        let cross_tenant = format!("cross-tenant {subject_label} role unassignment denied");
1015
1016        let outcome = retry_on_write_conflict(|| async {
1017            let mut result = self
1018                .db
1019                .current()
1020                .query(query.as_str())
1021                .bind(("tid", tenant_id.to_string()))
1022                .bind(("resource_id", resource_id_str.clone()))
1023                .bind(("claimed", claimed))
1024                .bind(("claim_grants", claim_grants))
1025                .bind(("claim_holders", claim_holders))
1026                .await
1027                .map_err(DbError::from)?;
1028            role_tx_outcome(
1029                result.take_errors(),
1030                "role_assignment",
1031                &role_id_str,
1032                &cross_tenant,
1033            )
1034        })
1035        .await;
1036        match outcome.map_err(AxiamError::from)? {
1037            None => Ok(()),
1038            Some(refusal) => Err(refusal),
1039        }
1040    }

1255    async fn get_user_role_assignments(
1256        &self,
1257        tenant_id: Uuid,
1258        user_id: Uuid,
1259    ) -> AxiamResult<Vec<RoleAssignment>> {
1260        let tenant_id_str = tenant_id.to_string();
1261        let user_id_str = user_id.to_string();
1262
1263        // Query has_role edges (direct + via groups) and join with role data.
1264        // Each result row contains role fields + the resource_id from the edge.
1265        //
1266        // I7(a) — the group-inherited half MUST hoist the membership sub-select
1267        // into a `LET` binding.
1268        //
1269        // Written inline as `WHERE in IN (SELECT VALUE out FROM member_of …)`
1270        // the planner cannot see a constant on the right-hand side; it reports
1271        // `TableScan { predicate: "in INSIDE (ProjectValue) …",
1272        // pre_decode_filter: "no (unsupported predicate)" }` and walks **every**
1273        // `has_role` edge in the database — every role assignment of every user
1274        // of every tenant — on every authorization check. Evaluating the
1275        // membership set first and binding it to `$group_records` turns the same
1276        // predicate into an `IndexScan` over the `(in, out)` composite index
1277        // `idx_has_role_unique` (schema v19). Rows returned are identical; only
1278        // the plan changes. Pinned by
1279        // `crates/axiam-db/tests/authz_query_plan_test.rs`.
1280        //
1281        // Defence in depth (security re-verification 2026-08-03, residual 1):
1282        // the `member_of` traversal carries its own `out.tenant_id =
1283        // $tenant_id` predicate. It is not reachable today — `group.rs`
1284        // validates BOTH endpoints against the tenant before writing a
1285        // `member_of` edge, and the outer `has_role` predicate still confines
1286        // the resulting role to the tenant — but without it this was the one
1287        // authorization edge with no read-time tenant check, so a migration or
1288        // bulk import writing `member_of` directly would have bypassed it.
1289        // `out` is a `group` record and `idx_member_of_unique` still serves the
1290        // `in =` half, so the added predicate is a post-filter on an already
1291        // index-selected row set: no plan change (pinned by
1292        // `authz_query_plan_test::group_membership_lookup_is_index_satisfied`).
1293        //
1294        // A subject id is looked up in BOTH principal tables. `AccessRequest`
1295        // carries no principal kind on purpose — `RequirePermission::check_subject`
1296        // states that RBAC applies identically to a service account and a user —
1297        // so the query resolves whichever record exists. Ids are v4 UUIDs, so a
1298        // subject is never both.
1299        //
1300        // Without this a service account resolved zero grants no matter what was
1301        // assigned to it, which is why a machine identity could hold no
1302        // permissions at all.
1303        let mut result = self
1304            .db
1305            .current()
1306            .query(
1307                "LET $subject_records = [\
1308                     type::record('user', $user_id), \
1309                     type::record('service_account', $user_id)\
1310                 ]; \
1311                 SELECT meta::id(out.id) AS record_id, \
1312                        out.tenant_id AS tenant_id, \
1313                        out.name AS name, \
1314                        out.description AS description, \
1315                        out.is_global AS is_global, \
1316                        out.created_at AS created_at, \
1317                        out.updated_at AS updated_at, \
1318                        resource_id, \
1319                        tenant_scope, \
1320                        inherit \
1321                 FROM has_role \
1322                 WHERE in IN $subject_records \
1323                 AND out.tenant_id = $tenant_id; \
1324                 LET $group_records = (\
1325                     SELECT VALUE out FROM member_of \
1326                     WHERE in IN $subject_records \
1327                     AND out.tenant_id = $tenant_id\
1328                 ); \
1329                 SELECT meta::id(out.id) AS record_id, \
1330                        out.tenant_id AS tenant_id, \
1331                        out.name AS name, \
1332                        out.description AS description, \
1333                        out.is_global AS is_global, \
1334                        out.created_at AS created_at, \
1335                        out.updated_at AS updated_at, \
1336                        resource_id, \
1337                        tenant_scope, \
1338                        inherit \
1339                 FROM has_role \
1340                 WHERE in IN $group_records \
1341                 AND out.tenant_id = $tenant_id;",
1342            )
1343            .bind(("tenant_id", tenant_id_str))
1344            .bind(("user_id", user_id_str))
1345            .await
1346            .map_err(DbError::from)?;
1347
1348        // Statement indices: 0 = LET $subject_records, 1 = direct SELECT,
1349        // 2 = LET $group_records, 3 = group-inherited SELECT.
1350        let direct: Vec<RoleAssignmentRow> = result.take(1).map_err(DbError::from)?;
1351        let inherited: Vec<RoleAssignmentRow> = result.take(3).map_err(DbError::from)?;
1352
1353        let mut assignments = Vec::new();
1354        for row in direct.into_iter().chain(inherited) {
1355            assignments.push(row.try_into_assignment()?);
1356        }
1357
1358        Ok(assignments)
1359    }
