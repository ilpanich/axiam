1//! SurrealDB implementation of [`SamlIdpCredentialRepository`] (G-2, D-21,
2//! T23.2.1).
3//!
4//! One row per credential, tenant-scoped, **not** a `certificate` row. Two
5//! properties are the datastore's, not this file's:
6//!
7//! * *At most one `active` and one `next` per tenant.* The `slot` column is a
8//!   `VALUE` expression evaluated by the database on every write and
9//!   `idx_saml_idp_credential_slot` is UNIQUE over it, so a second occupant is an
10//!   index violation whichever caller wrote it. This file only maps that
11//!   violation to `AlreadyExists`.
12//! * *The key is ciphertext.* The column is `option<bytes>` and the value written
13//!   is what `axiam_pki` sealed under `pki_encryption_key`; there is no column a
14//!   plaintext key could be written to.
15//!
16//! Key material is read by exactly one method, [`get_active_sealed`], through a
17//! separate projection ([`SEALED_COLUMNS`]); every other read selects
18//! [`COLUMNS`], which does not name the key columns, so no list or get can return
19//! one by accident. Errors and messages here never format a row.
20//!
21//! [`get_active_sealed`]: SamlIdpCredentialRepository::get_active_sealed
22

42const ENTITY: &str = "saml_idp_credential";
43
44/// `THROW` text: the credential is not the tenant's current `next`.
45const PROMOTE_NOT_NEXT: &str = "axiam:saml_idp_promote_not_next";
46/// `THROW` text: the credential's validity window does not contain now.
47const PROMOTE_OUTSIDE_WINDOW: &str = "axiam:saml_idp_promote_outside_window";
48
49/// Every column **except** the key columns.
50const COLUMNS: &str = "meta::id(id) AS record_id, tenant_id, issuer_ca_id, certificate_pem, \
51    serial, fingerprint, not_before, not_after, status, key_custody, created_at, retired_at";
52
53/// [`COLUMNS`] plus the sealed key. Used by `get_active_sealed` and nothing else.
54const SEALED_COLUMNS: &str = "meta::id(id) AS record_id, tenant_id, issuer_ca_id, \
55    certificate_pem, serial, fingerprint, not_before, not_after, status, key_custody, \
56    created_at, retired_at, key_locator, encrypted_private_key";

103fn custody_from(raw: &str) -> Result<CaKeyCustody, DbError> {
104    raw.parse::<CaKeyCustody>().map_err(|_| {
105        DbError::Serialization("saml_idp_credential has an unknown key_custody".into())
106    })

109impl CredentialRow {
110    fn into_domain(self) -> Result<SamlIdpCredential, DbError> {
111        Ok(SamlIdpCredential {
112            id: parse_uuid(&self.record_id, ENTITY)?,
113            tenant_id: parse_uuid(&self.tenant_id, "tenant")?,
114            issuer_ca_id: parse_uuid(&self.issuer_ca_id, "issuing CA")?,
115            certificate_pem: self.certificate_pem,
116            serial: self.serial,
117            fingerprint: self.fingerprint,
118            not_before: self.not_before,
119            not_after: self.not_after,
120            status: SamlIdpCredentialStatus::from_wire(&self.status).ok_or_else(|| {
121                DbError::Serialization("saml_idp_credential has an unknown status".into())
122            })?,
123            key_custody: custody_from(&self.key_custody)?,
124            created_at: self.created_at,
125            retired_at: self.retired_at,
126        })
127    }
128}

130impl SealedRow {
131    fn into_domain(self) -> Result<SealedSamlIdpCredential, DbError> {
132        let key = SealedSamlIdpKey {
133            custody: custody_from(&self.key_custody)?,
134            locator: self.key_locator,
135            ciphertext: self.encrypted_private_key.map(|b| b.into_inner().to_vec()),
136        };
137        let credential = CredentialRow {
138            record_id: self.record_id,
139            tenant_id: self.tenant_id,
140            issuer_ca_id: self.issuer_ca_id,
141            certificate_pem: self.certificate_pem,
142            serial: self.serial,
143            fingerprint: self.fingerprint,
144            not_before: self.not_before,
145            not_after: self.not_after,
146            status: self.status,
147            key_custody: self.key_custody,
148            created_at: self.created_at,
149            retired_at: self.retired_at,
150        }
151        .into_domain()?;
152        Ok(SealedSamlIdpCredential { credential, key })
153    }

156/// Why one attempt of [`SamlIdpCredentialRepository::promote`]'s transaction
157/// failed. Its `Display` is the engine's own text for a datastore failure, which
158/// is what the write-conflict retry reads.
159enum PromoteFailure {
160    NotFound,
161    NotNext,
162    OutsideWindow,
163    Db(DbError),
164}
165
166impl std::fmt::Display for PromoteFailure {
167    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
168        match self {
169            Self::NotFound => f.write_str("the credential does not exist"),
170            Self::NotNext => f.write_str("the credential is not the current next credential"),
171            Self::OutsideWindow => f.write_str("the credential is outside its validity window"),
172            Self::Db(error) => write!(f, "{error}"),
173        }
174    }
175}

177/// SurrealDB implementation of [`SamlIdpCredentialRepository`].
178#[derive(Clone)]
179pub struct SurrealSamlIdpCredentialRepository<C: Connection> {
180    db: DbHandle<C>,
181}

183impl<C: Connection> SurrealSamlIdpCredentialRepository<C> {
184    /// Build the repository over a database handle.
185    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
186        Self { db: db.into() }
187    }
188
189    async fn fetch(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<Option<SamlIdpCredential>> {
190        let mut result = self
191            .db
192            .current()
193            .query(format!(
194                "SELECT {COLUMNS} FROM saml_idp_credential \
195                 WHERE meta::id(id) = $id AND tenant_id = $tenant_id"
196            ))
197            .bind(("id", id.to_string()))
198            .bind(("tenant_id", tenant_id.to_string()))
199            .await
200            .map_err(DbError::from)?;
201        let rows: Vec<CredentialRow> = result.take(0).map_err(DbError::from)?;
202        rows.into_iter()
203            .next()
204            .map(|row| row.into_domain().map_err(AxiamError::from))
205            .transpose()
206    }
207}

209impl<C: Connection> SamlIdpCredentialRepository for SurrealSamlIdpCredentialRepository<C> {
210    async fn create(&self, input: StoreSamlIdpCredential) -> AxiamResult<SamlIdpCredential> {
211        if input.status == SamlIdpCredentialStatus::Retired {
212            return Err(AxiamError::Validation {
213                message: "a SAML IdP credential cannot be created retired".into(),
214            });
215        }
216        let id = input.id;
217        let tenant_id = input.tenant_id;
218        let result = self
219            .db
220            .current()
221            .query(
222                "CREATE type::record('saml_idp_credential', $id) SET \
223                 tenant_id = $tenant_id, issuer_ca_id = $issuer_ca_id, \
224                 certificate_pem = $certificate_pem, serial = $serial, \
225                 fingerprint = $fingerprint, not_before = $not_before, \
226                 not_after = $not_after, status = $status, \
227                 key_custody = $key_custody, key_locator = $key_locator, \
228                 encrypted_private_key = $encrypted_private_key, \
229                 created_at = time::now(), updated_at = time::now()",
230            )
231            .bind(("id", id.to_string()))
232            .bind(("tenant_id", tenant_id.to_string()))
233            .bind(("issuer_ca_id", input.issuer_ca_id.to_string()))
234            .bind(("certificate_pem", input.certificate_pem))
235            .bind(("serial", input.serial))
236            .bind(("fingerprint", input.fingerprint))
237            .bind(("not_before", input.not_before))
238            .bind(("not_after", input.not_after))
239            .bind(("status", input.status.as_str().to_owned()))
240            .bind(("key_custody", input.key.custody.to_string()))
241            .bind(("key_locator", input.key.locator))
242            .bind((
243                "encrypted_private_key",
244                input.key.ciphertext.map(surrealdb_types::Bytes::from),
245            ))
246            .await
247            .map_err(DbError::from)?;
248        // The unique index over `slot` is what refuses a second active or next
249        // credential for the tenant.
250        result
251            .check()
252            .map_err(|e| classify_write_error(e, ENTITY))?;
253
254        let created = self.fetch(tenant_id, id).await?;
255        Ok(take_first_or_not_found(
256            created.into_iter().collect(),
257            ENTITY,
258            &id.to_string(),
259        )?)
260    }

327    async fn retire(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<SamlIdpCredential> {
328        // The tenant guard is in the `WHERE`: a row of another tenant matches
329        // nothing and reads back as `NotFound`. The key columns are cleared in
330        // the same statement that takes the row out of its slot, so there is no
331        // moment at which a retired credential still holds a key. A row that is
332        // already retired is left exactly as it is.
333        self.db
334            .current()
335            .query(
336                "UPDATE type::record('saml_idp_credential', $id) SET \
337                 status = 'retired', encrypted_private_key = NONE, key_locator = NONE, \
338                 retired_at = time::now(), updated_at = time::now() \
339                 WHERE tenant_id = $tenant_id AND status != 'retired'",
340            )
341            .bind(("id", id.to_string()))
342            .bind(("tenant_id", tenant_id.to_string()))
343            .await
344            .map_err(DbError::from)?
345            .check()
346            .map_err(|e| classify_write_error(e, ENTITY))?;
347
348        self.get(tenant_id, id).await
349    }

351    async fn promote(
352        &self,
353        tenant_id: Uuid,
354        id: Uuid,
355        now: DateTime<Utc>,
356    ) -> AxiamResult<SamlIdpCredentialPromotion> {
357        #[derive(Debug, SurrealValue)]
358        struct RetiredId {
359            record_id: String,
360        }
361
362        // The whole rotation is one transaction. The checks come first and
363        // `THROW`, so a promotion that cannot happen has written nothing; then
364        // the old `active` leaves its slot (and its key is destroyed in the same
365        // write, as `retire` does) *before* the `next` row takes the slot, which
366        // is what keeps the UNIQUE `slot` index satisfied at every statement and
367        // leaves no state with two signers or none. Statement numbers, counting
368        // BEGIN: 0 BEGIN, 1 LET $row, 2-4 the three checks, 5 LET $retired,
369        // 6 UPDATE, 7 SELECT, 8 COMMIT.
370        //
371        // A contended attempt aborts and commits nothing, so it is replayed
372        // (`retry_on_write_conflict`); the replay sees the winner's state and
373        // answers `Conflict`, which is how the loser of two concurrent
374        // promotions is told.
375        let attempt = || async {
376            let mut result = self
377                .db
378                .current()
379                .query(format!(
380                    "BEGIN TRANSACTION; \
381                     LET $row = (SELECT status, not_before, not_after FROM saml_idp_credential \
382                         WHERE meta::id(id) = $id AND tenant_id = $tenant_id); \
383                     IF array::len($row) == 0 {{ THROW '{DELETE_TARGET_MISSING}'; }}; \
384                     IF $row[0].status != 'next' {{ THROW '{PROMOTE_NOT_NEXT}'; }}; \
385                     IF $row[0].not_before > $now OR $row[0].not_after <= $now \
386                         {{ THROW '{PROMOTE_OUTSIDE_WINDOW}'; }}; \
387                     LET $retired = (UPDATE saml_idp_credential SET \
388                         status = 'retired', encrypted_private_key = NONE, key_locator = NONE, \
389                         retired_at = time::now(), updated_at = time::now() \
390                         WHERE tenant_id = $tenant_id AND status = 'active'); \
391                     UPDATE type::record('saml_idp_credential', $id) SET \
392                         status = 'active', updated_at = time::now(); \
393                     SELECT meta::id(id) AS record_id FROM $retired; \
394                     COMMIT TRANSACTION"
395                ))
396                .bind(("id", id.to_string()))
397                .bind(("tenant_id", tenant_id.to_string()))
398                .bind(("now", now))
399                .await
400                .map_err(|e| PromoteFailure::Db(DbError::from(e)))?;
401
402            let errors = result.take_errors();
403            if !errors.is_empty() {
404                // Every statement's error, not `check()`'s pick: a `THROW`
405                // fires on its own slot while the rest report the generic
406                // "not executed due to a failed transaction".
407                let combined = errors
408                    .into_values()
409                    .map(|e| e.to_string())
410                    .collect::<Vec<_>>()
411                    .join("; ");
412                return Err(if combined.contains(DELETE_TARGET_MISSING) {
413                    PromoteFailure::NotFound
414                } else if combined.contains(PROMOTE_NOT_NEXT) {
415                    PromoteFailure::NotNext
416                } else if combined.contains(PROMOTE_OUTSIDE_WINDOW) {
417                    PromoteFailure::OutsideWindow
418                } else if is_write_conflict(&combined) {
419                    PromoteFailure::Db(DbError::Conflict(combined))
420                } else {
421                    PromoteFailure::Db(classify_write_error(combined, ENTITY))
422                });
423            }
424            result
425                .take::<Vec<RetiredId>>(7)
426                .map_err(|e| PromoteFailure::Db(DbError::from(e)))
427        };
428        // Replayed while the engine reports a write conflict. The retry reads the
429        // *engine's* text, so the attempt keeps its failure as a `PromoteFailure`
430        // (whose `Display` carries it) and only the settled outcome becomes an
431        // `AxiamError` — converted earlier, a conflict would already be the
432        // payload-free `WriteContention` and never be retried.
433        let retired = retry_on_write_conflict(attempt)
434            .await
435            .map_err(|failure| match failure {
436                PromoteFailure::NotFound => AxiamError::from(DbError::NotFound {
437                    entity: ENTITY.into(),
438                    id: id.to_string(),
439                }),
440                PromoteFailure::NotNext => AxiamError::Conflict {
441                    reason: "the credential is not the tenant's current `next` credential".into(),
442                },
443                PromoteFailure::OutsideWindow => AxiamError::Conflict {
444                    reason: "the `next` credential is outside its validity window".into(),
445                },
446                PromoteFailure::Db(error) => AxiamError::from(error),
447            })?;
448
449        let active = self.get(tenant_id, id).await?;
450        let retired = match retired.into_iter().next() {
451            Some(row) => {
452                let old = parse_uuid(&row.record_id, ENTITY).map_err(AxiamError::from)?;
453                Some(self.get(tenant_id, old).await?)
454            }
455            None => None,
456        };
457        Ok(SamlIdpCredentialPromotion { active, retired })
458    }
