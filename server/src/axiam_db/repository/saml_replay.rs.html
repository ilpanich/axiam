1//! SurrealDB implementation of [`AssertionReplayRepository`].
2//!
3//! Provides insert-or-conflict semantics for SAML assertion IDs. Inserting a
4//! `(tenant_id, assertion_id)` pair that already exists returns
5//! `Err(AxiamError::ReplayDetected)` so the SAML ACS handler can reject
6//! replayed assertions (D-09).
7
8use axiam_core::error::AxiamResult;
9use axiam_core::id::new_id;
10use axiam_core::repository::AssertionReplayRepository;
11use chrono::{DateTime, Utc};
12use surrealdb::Connection;
13use uuid::Uuid;
14
15use crate::error::DbError;
16use crate::handle::DbHandle;
17use crate::helpers::{classify_replay_write_error, cleanup_expired_rows};

23/// SurrealDB implementation of the SAML assertion replay repository.
24pub struct SurrealAssertionReplayRepository<C: Connection> {
25    db: DbHandle<C>,
26}
27
28// Manual Clone impl (not derive): avoids the spurious `C: Clone` bound that
29// blocks cloning under generic `C: Connection` callers. Matches SurrealUserRepository.
30impl<C: Connection> Clone for SurrealAssertionReplayRepository<C> {
31    fn clone(&self) -> Self {
32        Self {
33            db: self.db.clone(),
34        }
35    }
36}
37
38impl<C: Connection> SurrealAssertionReplayRepository<C> {
39    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
40        let db = db.into();
41        Self { db }
42    }
43}
44
45impl<C: Connection> AssertionReplayRepository for SurrealAssertionReplayRepository<C> {
46    async fn insert_assertion(
47        &self,
48        tenant_id: Uuid,
49        assertion_id: &str,
50        expires_at: DateTime<Utc>,
51    ) -> AxiamResult<()> {
52        let row_id = new_id().to_string();
53        let assertion_id_owned = assertion_id.to_string();
54
55        let result = self
56            .db
57            .current()
58            .query(
59                "CREATE type::record('saml_assertion_replay', $row_id) SET \
60                 tenant_id = $tenant_id, \
61                 assertion_id = $assertion_id, \
62                 expires_at = $expires_at",
63            )
64            .bind(("row_id", row_id))
65            .bind(("tenant_id", tenant_id.to_string()))
66            .bind(("assertion_id", assertion_id_owned))
67            .bind(("expires_at", expires_at))
68            .await
69            .map_err(DbError::from)?;
70
71        // The UNIQUE violation IS the answer: this assertion ID has been seen.
72        // What counts as one is decided in exactly one place (D-09) so the
73        // three replay guards cannot drift apart -- see
74        // `helpers::is_unique_violation`.
75        result
76            .check()
77            .map_err(classify_replay_write_error)
78            .map(|_| ())
79    }
80
81    async fn cleanup_expired(&self) -> AxiamResult<u64> {
82        cleanup_expired_rows(&self.db, "saml_assertion_replay").await
83    }
84}
