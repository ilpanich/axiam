42fn row_to_session(row: SessionRow, id: Uuid) -> Result<Session, DbError> {
43    let tenant_id = Uuid::parse_str(&row.tenant_id)
44        .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
45    let user_id = Uuid::parse_str(&row.user_id)
46        .map_err(|e| DbError::Migration(format!("invalid user UUID: {e}")))?;
47    Ok(Session {
48        id,
49        tenant_id,
50        user_id,
51        token_hash: row.token_hash,
52        ip_address: row.ip_address,
53        user_agent: row.user_agent,
54        expires_at: row.expires_at,
55        created_at: row.created_at,
56    })
57}

80/// SurrealDB implementation of the Session repository.
81pub struct SurrealSessionRepository<C: Connection> {
82    db: DbHandle<C>,
83    /// I6: optional short-TTL validity cache. `None` (the default) makes every
84    /// method below byte-identical to the pre-I6 behaviour.
85    validation_cache: Option<Arc<SessionValidationCache>>,
86}
87
88// Manual Clone impl (not derive): `#[derive(Clone)]` would add a `C: Clone`
89// bound that generic `C: Connection` callers (e.g. REST handlers) cannot
90// satisfy, silently cloning a `&Self` instead. Matches SurrealUserRepository.
91impl<C: Connection> Clone for SurrealSessionRepository<C> {
92    fn clone(&self) -> Self {
93        Self {
94            db: self.db.clone(),
95            // Clones share the same cache: `axiam-server` clones this repository
96            // into several services, and they must agree on what has been
97            // revoked.
98            validation_cache: self.validation_cache.clone(),
99        }
100    }
101}
102
103impl<C: Connection> SurrealSessionRepository<C> {
104    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
105        let db = db.into();
106        Self {
107            db,
108            validation_cache: None,
109        }
110    }
111
112    /// Attach a [`SessionValidationCache`] (I6).
113    ///
114    /// Only [`Self::is_session_active_checked`] reads the cache; every delete path in
115    /// this file writes to it. Attaching a cache therefore cannot desynchronise
116    /// it from the rows this repository owns — see the invalidation contract in
117    /// [`crate::session_validation_cache`].
118    pub fn with_validation_cache(mut self, cache: Arc<SessionValidationCache>) -> Self {
119        self.validation_cache = Some(cache);
120        self
121    }
122
123    /// The attached validity cache, if any.
124    pub fn validation_cache(&self) -> Option<&Arc<SessionValidationCache>> {
125        self.validation_cache.as_ref()
126    }
127
128    /// Is the session behind an access token's `jti` still usable? (D-15 /
129    /// REQ-7.)
130    ///
131    /// Named `_checked` rather than `is_session_active` so it can never be
132    /// confused with `axiam_api_rest::SessionValidator::is_session_active`,
133    /// which delegates here — an accidental name collision there would recurse
134    /// forever.
135    ///
136    /// This is the per-request session-revocation check every authenticated
137    /// REST request performs. Without a cache attached it is exactly
138    /// `get_by_id(..).is_ok() && expires_at > now` — one SurrealDB read per
139    /// request. With one attached, a repeat check inside the TTL is answered in
140    /// process.
141    ///
142    /// Negative answers are never cached (see the module docs), so a revoked or
143    /// unknown session always costs a read and can never be resurrected.
144    pub async fn is_session_active_checked(&self, tenant_id: Uuid, session_id: Uuid) -> bool {
145        if let Some(cache) = &self.validation_cache
146            && cache.get(tenant_id, session_id) == Some(true)
147        {
148            return true;
149        }
150
151        match self.get_by_id(tenant_id, session_id).await {
152            Ok(session) if session.expires_at > Utc::now() => {
153                if let Some(cache) = &self.validation_cache {
154                    cache.insert_valid(tenant_id, session_id, session.user_id, session.expires_at);
155                }
156                true
157            }
158            _ => false,
159        }
160    }
161}

164    async fn create(&self, input: CreateSession) -> AxiamResult<Session> {
165        let id = new_id();
166        let id_str = id.to_string();
167
168        let result = self
169            .db
170            .current()
171            .query(
172                "CREATE type::record('session', $id) SET \
173                 tenant_id = $tenant_id, \
174                 user_id = $user_id, \
175                 token_hash = $token_hash, \
176                 ip_address = $ip_address, \
177                 user_agent = $user_agent, \
178                 expires_at = $expires_at",
179            )
180            .bind(("id", id_str.clone()))
181            .bind(("tenant_id", input.tenant_id.to_string()))
182            .bind(("user_id", input.user_id.to_string()))
183            .bind(("token_hash", input.token_hash))
184            .bind(("ip_address", input.ip_address))
185            .bind(("user_agent", input.user_agent))
186            .bind(("expires_at", input.expires_at))
187            .await
188            .map_err(DbError::from)?;
189
190        let mut result = result
191            .check()
192            .map_err(|e| DbError::Migration(e.to_string()))?;
193
194        let rows: Vec<SessionRow> = result.take(0).map_err(DbError::from)?;
195        let row = take_first_or_not_found(rows, "session", &id_str)?;
196
197        row_to_session(row, id).map_err(Into::into)
198    }

243    async fn invalidate(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
244        let result = self
245            .db
246            .current()
247            .query(
248                "DELETE type::record('session', $id) \
249                 WHERE tenant_id = $tenant_id",
250            )
251            .bind(("id", id.to_string()))
252            .bind(("tenant_id", tenant_id.to_string()))
253            .await
254            .map_err(DbError::from)?;
255
256        // I6: drop the validity entry in the same call that removes the row, so
257        // a logout can never be outlived by a cached "still valid" answer on
258        // this replica.
259        //
260        // ORDERING (residual 2, security re-verification 2026-08-03): this MUST
261        // stay above the `.check()` below. Invalidation is infallible and
262        // idempotent, and dropping a cache entry for a row that turned out NOT
263        // to be deleted only costs one avoidable re-read — whereas the reverse
264        // order would let a newly-fallible step strand a *positive* cache entry
265        // for a session the DELETE did remove.
266        if let Some(cache) = &self.validation_cache {
267            cache.invalidate(tenant_id, id);
268        }
269
270        // OBS-3 (security re-verification 2026-08-03): a `DELETE` that fails at
271        // the *statement* level still returns `Ok` from the `.await` — only
272        // `.check()` surfaces it. Without this, logout / password-reset session
273        // revocation / MFA reset were told "revoked" when nothing was deleted.
274        // No `take()` here, so no deserialization is involved: this can only
275        // fail when the statement itself did.
276        result
277            .check()
278            .map_err(|e| DbError::Migration(e.to_string()))?;
279
280        Ok(())
281    }

283    async fn consume(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<bool> {
284        // NEW-3: `RETURN BEFORE` yields the pre-delete row; a non-empty result
285        // means THIS statement removed the session. SurrealDB serializes two
286        // concurrent DELETEs on the same record, so exactly one caller sees a
287        // non-empty BEFORE image — the single-use gate for refresh rotation.
288        let result = self
289            .db
290            .current()
291            .query(
292                "DELETE type::record('session', $id) \
293                 WHERE tenant_id = $tenant_id RETURN BEFORE",
294            )
295            .bind(("id", id.to_string()))
296            .bind(("tenant_id", tenant_id.to_string()))
297            .await
298            .map_err(DbError::from)?;
299
300        // I6: invalidate unconditionally — whether or not this caller won the
301        // single-use race, the row is gone once any caller consumed it.
302        //
303        // This MUST come before `take(0)` (security re-verification
304        // 2026-08-03, residual 2). The DELETE has already committed by the
305        // time the `.await` returns `Ok`; if the BEFORE image then failed to
306        // deserialize, an early `?` would leave a *positive* cache entry live
307        // for up to the TTL — i.e. a deleted session would keep validating.
308        // Invalidation is infallible and idempotent, so running it first is
309        // free.
310        if let Some(cache) = &self.validation_cache {
311            cache.invalidate(tenant_id, id);
312        }
313
314        // OBS-3: `.check()` before `take(0)` so a statement-level failure is
315        // reported as such rather than as a "no rows" (`won == false`) answer.
316        // `take(0)` alone would already surface it for this single-statement
317        // query; the explicit check keeps all five delete paths uniform.
318        let mut result = result
319            .check()
320            .map_err(|e| DbError::Migration(e.to_string()))?;
321
322        let deleted: Vec<SessionRow> = result.take(0).map_err(DbError::from)?;
323        Ok(!deleted.is_empty())
324    }

326    async fn consume_by_token_hash(
327        &self,
328        tenant_id: Uuid,
329        token_hash: &str,
330    ) -> AxiamResult<Option<Session>> {
331        // A2/J2: the SELECT-then-DELETE pair collapsed into one statement.
332        // Same `RETURN BEFORE` single-use gate as `consume` (SurrealDB
333        // serializes concurrent DELETEs touching the same row, so exactly one
334        // caller receives a non-empty before-image), except that the row is
335        // also *found* by this statement — so there is no window between
336        // finding it and removing it, and the rotation path pays one round
337        // trip here instead of two.
338        //
339        // Two statements, ONE round trip. `RETURN BEFORE` does not accept a
340        // projection list, and the caller needs `session.id`, so the
341        // before-image is captured into `$before` and projected by a second
342        // statement in the same query — `meta::id(id) AS record_id`, the same
343        // shape `get_by_token_hash` produces and `SessionRowWithId` expects.
344        //
345        // Atomicity is unaffected by the split: the DELETE alone decides who
346        // wins, and `$before` is that decision's own output, not a re-read.
347        let result = self
348            .db
349            .current()
350            .query(
351                "LET $before = (DELETE session \
352                     WHERE tenant_id = $tenant_id AND token_hash = $token_hash \
353                     RETURN BEFORE); \
354                 SELECT meta::id(id) AS record_id, * FROM $before",
355            )
356            .bind(("tenant_id", tenant_id.to_string()))
357            .bind(("token_hash", token_hash.to_string()))
358            .await
359            .map_err(DbError::from)?;
360
361        // OBS-3 / residual 2 ordering, same as `consume`: `.check()` the
362        // statement before deserializing, and drop any cached validity entry
363        // before an error can short-circuit us past it. The row id is only
364        // known after deserialization here, so the cache drop moves below —
365        // which is sound because a *failed* deserialization of a committed
366        // DELETE is exactly the case the unconditional `invalidate` below
367        // handles, and there is no id to invalidate before that point.
368        let mut result = result
369            .check()
370            .map_err(|e| DbError::Migration(e.to_string()))?;
371
372        // Statement index 1: the projected SELECT (index 0 is the `LET`).
373        let rows: Vec<SessionRowWithId> = result.take(1).map_err(DbError::from)?;
374        let Some(row) = rows.into_iter().next() else {
375            return Ok(None);
376        };
377
378        let session: Session = row.try_into_session()?;
379
380        // I6: the row is gone, so no cached "still valid" answer may outlive
381        // it on this replica.
382        if let Some(cache) = &self.validation_cache {
383            cache.invalidate(tenant_id, session.id);
384        }
385
386        Ok(Some(session))
387    }

389    async fn invalidate_user_sessions(&self, tenant_id: Uuid, user_id: Uuid) -> AxiamResult<()> {
390        let result = self
391            .db
392            .current()
393            .query("DELETE session WHERE tenant_id = $tenant_id AND user_id = $user_id")
394            .bind(("tenant_id", tenant_id.to_string()))
395            .bind(("user_id", user_id.to_string()))
396            .await
397            .map_err(DbError::from)?;
398
399        // I6: every session of this user just disappeared.
400        //
401        // ORDERING: above `.check()` for the same reason as `invalidate` —
402        // see the note there (residual 2).
403        if let Some(cache) = &self.validation_cache {
404            cache.invalidate_user(tenant_id, user_id, None);
405        }
406
407        // OBS-3: propagate a statement-level DELETE failure instead of
408        // reporting a revocation that may not have happened.
409        result
410            .check()
411            .map_err(|e| DbError::Migration(e.to_string()))?;
412
413        Ok(())
414    }

416    async fn invalidate_user_sessions_except(
417        &self,
418        tenant_id: Uuid,
419        user_id: Uuid,
420        current_session_id: Uuid,
421    ) -> AxiamResult<u64> {
422        // DELETE all sessions for this user in this tenant EXCEPT the
423        // current one (identified by its record ID). RETURN BEFORE gives us
424        // the deleted rows so we can count them.
425        let result = self
426            .db
427            .current()
428            .query(
429                "DELETE session \
430                 WHERE tenant_id = $tenant_id \
431                   AND user_id = $user_id \
432                   AND id != type::record('session', $current_session_id) \
433                 RETURN BEFORE",
434            )
435            .bind(("tenant_id", tenant_id.to_string()))
436            .bind(("user_id", user_id.to_string()))
437            .bind(("current_session_id", current_session_id.to_string()))
438            .await
439            .map_err(DbError::from)?;
440
441        // I6: same set, minus the session the caller deliberately kept.
442        //
443        // Ordered before `take(0)` for the same reason as `consume` — the
444        // DELETE is already committed once the `.await` returns, so a
445        // deserialize failure of the BEFORE image must not be able to strand a
446        // positive cache entry for a session that no longer exists
447        // (residual 2).
448        if let Some(cache) = &self.validation_cache {
449            cache.invalidate_user(tenant_id, user_id, Some(current_session_id));
450        }
451
452        // OBS-3: uniform with the other four delete paths — a statement-level
453        // failure must never be reported as "0 sessions deleted".
454        let mut result = result
455            .check()
456            .map_err(|e| DbError::Migration(e.to_string()))?;
457
458        let deleted: Vec<SessionRow> = result.take(0).map_err(DbError::from)?;
459        Ok(deleted.len() as u64)
460    }

481    async fn cleanup_expired(&self, tenant_id: Uuid) -> AxiamResult<u64> {
482        // Count expired sessions first, then delete.
483        let mut count_result = self
484            .db
485            .current()
486            .query(
487                "SELECT count() AS total FROM session \
488                 WHERE tenant_id = $tenant_id AND expires_at < time::now() \
489                 GROUP ALL",
490            )
491            .bind(("tenant_id", tenant_id.to_string()))
492            .await
493            .map_err(DbError::from)?;
494        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
495        let total = count_rows.first().map(|r| r.total).unwrap_or(0);
496
497        let result = self
498            .db
499            .current()
500            .query("DELETE session WHERE tenant_id = $tenant_id AND expires_at < time::now()")
501            .bind(("tenant_id", tenant_id.to_string()))
502            .await
503            .map_err(DbError::from)?;
504
505        // I6: this deletes by predicate, not by id, so the cache cannot know
506        // which entries went. Dropping the whole tenant is the conservative
507        // choice; entries for expired sessions could not have been served
508        // anyway (`get` re-checks `expires_at`), so this only costs a few
509        // avoidable re-reads on a periodic janitor run.
510        //
511        // ORDERING: above `.check()`, per residual 2.
512        if let Some(cache) = &self.validation_cache {
513            cache.invalidate_tenant(tenant_id);
514        }
515
516        // OBS-3: the janitor must not log "N sessions reaped" when the DELETE
517        // statement failed. `total` is a pre-delete count, so returning it
518        // after a failed DELETE would be doubly misleading.
519        result
520            .check()
521            .map_err(|e| DbError::Migration(e.to_string()))?;
522
523        Ok(total)
524    }

609    /// Forces every `DELETE session` in this database to fail at the
610    /// **statement** level, which is the exact failure mode OBS-3 is about:
611    /// the `.await` still returns `Ok(Response)`, and only `.check()` (or a
612    /// `take()`) surfaces the error.
613    ///
614    /// §10.7/residual 2 recorded that "the repository takes a concrete
615    /// `Surreal<C>` rather than a mockable trait, so there is no seam". That
616    /// is true of the *connection*, but SurrealDB provides one inside the
617    /// database: a `DEFINE EVENT … THEN THROW` on the DELETE event makes the
618    /// server itself fail the statement. No mock, no fake — the real embedded
619    /// engine returns the real error shape.
620    async fn poison_session_deletes(db: &Surreal<surrealdb::engine::local::Db>) {
621        db.query(
622            "DEFINE EVENT session_delete_poison ON TABLE session \
623             WHEN $event = 'DELETE' THEN { THROW 'poisoned delete' }",
624        )
625        .await
626        .expect("define poison event")
627        .check()
628        .expect("poison event definition must itself succeed");
629    }

649    /// OBS-3: all five session-deleting methods must return `Err` when the
650    /// DELETE statement fails, instead of reporting a revocation that never
651    /// happened. Before the fix, `invalidate`, `invalidate_user_sessions` and
652    /// `cleanup_expired` returned `Ok` here.
653    #[tokio::test]
654    async fn every_delete_path_propagates_a_statement_level_failure() {
655        let tenant_id = Uuid::new_v4();
656        let user_id = Uuid::new_v4();
657
658        // `invalidate`
659        let db = setup_db().await;
660        let repo = SurrealSessionRepository::new(db.clone());
661        let s = seed_session(&repo, tenant_id, user_id, "a").await;
662        poison_session_deletes(&db).await;
663        assert!(
664            repo.invalidate(tenant_id, s.id).await.is_err(),
665            "invalidate must not report success when the DELETE statement failed"
666        );
667
668        // `consume`
669        let db = setup_db().await;
670        let repo = SurrealSessionRepository::new(db.clone());
671        let s = seed_session(&repo, tenant_id, user_id, "b").await;
672        poison_session_deletes(&db).await;
673        assert!(
674            repo.consume(tenant_id, s.id).await.is_err(),
675            "consume must not report `false` (lost the race) for a failed statement"
676        );
677
678        // `invalidate_user_sessions`
679        let db = setup_db().await;
680        let repo = SurrealSessionRepository::new(db.clone());
681        seed_session(&repo, tenant_id, user_id, "c").await;
682        poison_session_deletes(&db).await;
683        assert!(
684            repo.invalidate_user_sessions(tenant_id, user_id)
685                .await
686                .is_err(),
687            "invalidate_user_sessions must not report success when the DELETE failed"
688        );
689
690        // `invalidate_user_sessions_except`
691        let db = setup_db().await;
692        let repo = SurrealSessionRepository::new(db.clone());
693        let keep = seed_session(&repo, tenant_id, user_id, "d").await;
694        seed_session(&repo, tenant_id, user_id, "e").await;
695        poison_session_deletes(&db).await;
696        assert!(
697            repo.invalidate_user_sessions_except(tenant_id, user_id, keep.id)
698                .await
699                .is_err(),
700            "invalidate_user_sessions_except must not report `0 deleted` for a failed statement"
701        );
702
703        // `cleanup_expired`
704        let db = setup_db().await;
705        let repo = SurrealSessionRepository::new(db.clone());
706        repo.create(CreateSession {
707            tenant_id,
708            user_id,
709            token_hash: "hash-expired".into(),
710            ip_address: None,
711            user_agent: None,
712            expires_at: Utc::now() - Duration::hours(1),
713        })
714        .await
715        .unwrap();
716        poison_session_deletes(&db).await;
717        assert!(
718            repo.cleanup_expired(tenant_id).await.is_err(),
719            "cleanup_expired must not return a pre-delete count when the DELETE failed"
720        );
721    }

723    /// Residual 2 (cache-invalidation ordering) must survive OBS-3 making
724    /// these paths fallible: the cache entry is dropped **before** the newly
725    /// added `.check()`, so an erroring DELETE can never leave a positive
726    /// "still valid" entry behind.
727    ///
728    /// This is the regression test §10.7 could not write — the poison event
729    /// above supplies the failing step that did not previously exist.
730    #[tokio::test]
731    async fn cache_is_invalidated_before_the_fallible_step_on_every_path() {
732        let tenant_id = Uuid::new_v4();
733        let user_id = Uuid::new_v4();
734
735        /// Builds a repo whose cache holds a live positive entry for each of
736        /// `n` fresh sessions, optionally adds `expired` already-expired rows
737        /// (which `cleanup_expired` needs in order to delete anything at all,
738        /// and which are never cached because the cache is positive-only),
739        /// then poisons DELETE.
740        async fn armed(
741            tenant_id: Uuid,
742            user_id: Uuid,
743            n: usize,
744            expired: usize,
745        ) -> (
746            SurrealSessionRepository<surrealdb::engine::local::Db>,
747            Arc<SessionValidationCache>,
748            Vec<Session>,
749        ) {
750            let db = setup_db().await;
751            let cache = Arc::new(SessionValidationCache::new(std::time::Duration::from_secs(
752                300,
753            )));
754            let repo =
755                SurrealSessionRepository::new(db.clone()).with_validation_cache(cache.clone());
756            let mut sessions = Vec::new();
757            for i in 0..n {
758                let s = seed_session(&repo, tenant_id, user_id, &format!("{i}")).await;
759                assert!(
760                    repo.is_session_active_checked(tenant_id, s.id).await,
761                    "session must warm the cache before the delete path runs"
762                );
763                sessions.push(s);
764            }
765            for i in 0..expired {
766                repo.create(CreateSession {
767                    tenant_id,
768                    user_id,
769                    token_hash: format!("hash-expired-{i}"),
770                    ip_address: None,
771                    user_agent: None,
772                    expires_at: Utc::now() - Duration::hours(1),
773                })
774                .await
775                .unwrap();
776            }
777            assert_eq!(cache.len(), n, "cache must be warm");
778            poison_session_deletes(&db).await;
779            (repo, cache, sessions)
780        }
781
782        // `invalidate`
783        let (repo, cache, sessions) = armed(tenant_id, user_id, 1, 0).await;
784        assert!(repo.invalidate(tenant_id, sessions[0].id).await.is_err());
785        assert_eq!(
786            cache.len(),
787            0,
788            "invalidate: cache entry must be dropped even though the DELETE failed"
789        );
790
791        // `consume`
792        let (repo, cache, sessions) = armed(tenant_id, user_id, 1, 0).await;
793        assert!(repo.consume(tenant_id, sessions[0].id).await.is_err());
794        assert_eq!(cache.len(), 0, "consume: cache entry must be dropped");
795
796        // `invalidate_user_sessions`
797        let (repo, cache, _) = armed(tenant_id, user_id, 2, 0).await;
798        assert!(
799            repo.invalidate_user_sessions(tenant_id, user_id)
800                .await
801                .is_err()
802        );
803        assert_eq!(
804            cache.len(),
805            0,
806            "invalidate_user_sessions: all of the user's entries must be dropped"
807        );
808
809        // `invalidate_user_sessions_except` — the kept session stays cached by
810        // design; every other entry must go.
811        let (repo, cache, sessions) = armed(tenant_id, user_id, 2, 0).await;
812        assert!(
813            repo.invalidate_user_sessions_except(tenant_id, user_id, sessions[0].id)
814                .await
815                .is_err()
816        );
817        assert_eq!(
818            cache.len(),
819            1,
820            "invalidate_user_sessions_except: only the deliberately kept session may remain"
821        );
822
823        // `cleanup_expired`
824        let (repo, cache, _) = armed(tenant_id, user_id, 2, 1).await;
825        assert!(repo.cleanup_expired(tenant_id).await.is_err());
826        assert_eq!(
827            cache.len(),
828            0,
829            "cleanup_expired: the tenant's entries must be dropped"
830        );
831    }

971    /// The happy path is unchanged by OBS-3: every delete path still returns
972    /// `Ok`, still removes the rows, and still clears the cache. Guards
973    /// against a `.check()` that rejects an ordinary successful DELETE.
974    #[tokio::test]
975    async fn delete_paths_still_succeed_and_clear_the_cache() {
976        let tenant_id = Uuid::new_v4();
977        let user_id = Uuid::new_v4();
978        let db = setup_db().await;
979        let cache = Arc::new(SessionValidationCache::new(std::time::Duration::from_secs(
980            300,
981        )));
982        let repo = SurrealSessionRepository::new(db).with_validation_cache(cache.clone());
983
984        let a = seed_session(&repo, tenant_id, user_id, "1").await;
985        let b = seed_session(&repo, tenant_id, user_id, "2").await;
986        let c = seed_session(&repo, tenant_id, user_id, "3").await;
987        for s in [&a, &b, &c] {
988            assert!(repo.is_session_active_checked(tenant_id, s.id).await);
989        }
990        assert_eq!(cache.len(), 3);
991
992        repo.invalidate(tenant_id, a.id).await.unwrap();
993        assert!(repo.get_by_id(tenant_id, a.id).await.is_err());
994        assert_eq!(cache.len(), 2);
995
996        assert!(repo.consume(tenant_id, b.id).await.unwrap());
997        assert!(
998            !repo.consume(tenant_id, b.id).await.unwrap(),
999            "second consume of the same session must lose the single-use race"
1000        );
1001        assert_eq!(cache.len(), 1);
1002
1003        assert_eq!(
1004            repo.invalidate_user_sessions_except(tenant_id, user_id, c.id)
1005                .await
1006                .unwrap(),
1007            0,
1008            "c is the only session left and it is the one being kept"
1009        );
1010        repo.invalidate_user_sessions(tenant_id, user_id)
1011            .await
1012            .unwrap();
1013        assert!(repo.get_by_id(tenant_id, c.id).await.is_err());
1014        assert_eq!(cache.len(), 0);
1015
1016        // `cleanup_expired` over an empty/all-fresh table still succeeds.
1017        assert_eq!(repo.cleanup_expired(tenant_id).await.unwrap(), 0);
1018    }
