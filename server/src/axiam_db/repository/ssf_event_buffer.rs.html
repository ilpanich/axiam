1//! SurrealDB implementation of [`SsfEventBufferRepository`] (G-5, T23.5.3,
2//! schema v77's `ssf_event_buffer`).
3//!
4//! One row per `(tenant, stream, jti)` — the datastore's unique index decides —
5//! holding an **unsigned** [`SsfPendingEvent`] as JSON. At most
6//! [`POLL_BUFFER_MAX_EVENTS`] rows per stream, **oldest dropped** to admit the
7//! newest (D-48; SSF 1.0 §8.1.2 permits a transmitter to drop held events and
8//! the newest state is the one a receiver can act on), each for at most
9//! [`POLL_BUFFER_RETENTION_DAYS`] days (`expires_at`, swept by
10//! [`SsfEventBufferRepository::delete_expired`], a job on `/health/jobs`).
11//!
12//! # What the row holds, and what it must not
13//!
14//! The row is an event waiting for a receiver: `jti`, `iat`, the event URI and
15//! object, the subject member (an address only on an `email` stream for an
16//! address something vouched for) and the `txn`. Never a signed token and never
17//! a credential. `pending_json` is read back only through
18//! [`SsfPendingEvent`], whose `Debug` omits the subject.
19

72impl<C: Connection> SurrealSsfEventBufferRepository<C> {
73    /// Build the repository.
74    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
75        Self { db: db.into() }
76    }
77
78    /// Drop the oldest rows of the stream beyond the bound.
79    async fn trim(&self, tenant_id: Uuid, stream_id: Uuid) -> AxiamResult<()> {
80        let held = self.count(tenant_id, stream_id).await? as usize;
81        if held <= POLL_BUFFER_MAX_EVENTS {
82            return Ok(());
83        }
84        let excess = (held - POLL_BUFFER_MAX_EVENTS) as i64;
85        let mut result = self
86            .db
87            .current()
88            .query(
89                "SELECT jti, created_at FROM ssf_event_buffer \
90                 WHERE tenant_id = $tenant_id AND stream_id = $stream_id \
91                 ORDER BY created_at ASC LIMIT $excess",
92            )
93            .bind(("tenant_id", tenant_id.to_string()))
94            .bind(("stream_id", stream_id.to_string()))
95            .bind(("excess", excess))
96            .await
97            .map_err(DbError::from)?;
98        let oldest: Vec<JtiRow> = result.take(0).map_err(DbError::from)?;
99        let jtis: Vec<String> = oldest.into_iter().map(|r| r.jti).collect();
100        self.delete_by_jti(tenant_id, stream_id, &jtis).await?;
101        Ok(())
102    }
103}

105impl<C: Connection> SsfEventBufferRepository for SurrealSsfEventBufferRepository<C> {
106    async fn push(
107        &self,
108        tenant_id: Uuid,
109        stream_id: Uuid,
110        event: &SsfPendingEvent,
111        now: DateTime<Utc>,
112    ) -> AxiamResult<()> {
113        let pending_json = serde_json::to_string(event).map_err(|_| {
114            AxiamError::Internal("an SSF event could not be encoded for the buffer".into())
115        })?;
116        let result = self
117            .db
118            .current()
119            .query(
120                "CREATE ssf_event_buffer SET tenant_id = $tenant_id, stream_id = $stream_id, \
121                 jti = $jti, event_uri = $event_uri, pending_json = $pending_json, \
122                 created_at = $now, expires_at = $expires_at",
123            )
124            .bind(("tenant_id", tenant_id.to_string()))
125            .bind(("stream_id", stream_id.to_string()))
126            .bind(("jti", event.jti.clone()))
127            .bind(("event_uri", event.event_uri.clone()))
128            .bind(("pending_json", pending_json))
129            .bind(("now", now))
130            .bind((
131                "expires_at",
132                now + Duration::days(POLL_BUFFER_RETENTION_DAYS),
133            ))
134            .await
135            .map_err(DbError::from)?;
136        match result
137            .check()
138            .map_err(|e| AxiamError::from(crate::helpers::classify_write_error(e, ENTITY)))
139        {
140            Ok(_) => {}
141            // One row per jti: an event already buffered is buffered.
142            Err(AxiamError::AlreadyExists { .. }) => return Ok(()),
143            Err(other) => return Err(other),
144        }
145        self.trim(tenant_id, stream_id).await
146    }
147
148    async fn list_oldest(
149        &self,
150        tenant_id: Uuid,
151        stream_id: Uuid,
152        limit: usize,
153        now: DateTime<Utc>,
154    ) -> AxiamResult<Vec<SsfPendingEvent>> {
155        let mut result = self
156            .db
157            .current()
158            .query(
159                "SELECT pending_json, created_at FROM ssf_event_buffer \
160                 WHERE tenant_id = $tenant_id AND stream_id = $stream_id \
161                   AND expires_at > $now \
162                 ORDER BY created_at ASC LIMIT $limit",
163            )
164            .bind(("tenant_id", tenant_id.to_string()))
165            .bind(("stream_id", stream_id.to_string()))
166            .bind(("now", now))
167            .bind(("limit", limit as i64))
168            .await
169            .map_err(DbError::from)?;
170        let rows: Vec<EventRow> = result.take(0).map_err(DbError::from)?;
171        rows.into_iter()
172            .map(|row| {
173                serde_json::from_str::<SsfPendingEvent>(&row.pending_json).map_err(|_| {
174                    AxiamError::Internal("a buffered SSF event could not be decoded".into())
175                })
176            })
177            .collect()
178    }
179
180    async fn delete_by_jti(
181        &self,
182        tenant_id: Uuid,
183        stream_id: Uuid,
184        jtis: &[String],
185    ) -> AxiamResult<u64> {
186        if jtis.is_empty() {
187            return Ok(0);
188        }
189        let mut result = self
190            .db
191            .current()
192            .query(
193                "LET $removed = (DELETE ssf_event_buffer \
194                     WHERE tenant_id = $tenant_id AND stream_id = $stream_id \
195                       AND jti IN $jtis RETURN BEFORE); \
196                 SELECT meta::id(id) AS record_id FROM $removed",
197            )
198            .bind(("tenant_id", tenant_id.to_string()))
199            .bind(("stream_id", stream_id.to_string()))
200            .bind(("jtis", jtis.to_vec()))
201            .await
202            .map_err(DbError::from)?;
203        let removed: Vec<RemovedRow> = result.take(1).map_err(DbError::from)?;
204        Ok(removed.len() as u64)
205    }
206
207    async fn count(&self, tenant_id: Uuid, stream_id: Uuid) -> AxiamResult<u64> {
208        let mut result = self
209            .db
210            .current()
211            .query(
212                "SELECT count() AS total FROM ssf_event_buffer \
213                 WHERE tenant_id = $tenant_id AND stream_id = $stream_id GROUP ALL",
214            )
215            .bind(("tenant_id", tenant_id.to_string()))
216            .bind(("stream_id", stream_id.to_string()))
217            .await
218            .map_err(DbError::from)?;
219        let rows: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
220        Ok(rows.first().map(|r| r.total).unwrap_or(0))
221    }
222
223    async fn delete_expired(&self, now: DateTime<Utc>) -> AxiamResult<u64> {
224        let mut result = self
225            .db
226            .current()
227            .query(
228                "LET $removed = (DELETE ssf_event_buffer WHERE expires_at <= $now \
229                     RETURN BEFORE); \
230                 SELECT meta::id(id) AS record_id FROM $removed",
231            )
232            .bind(("now", now))
233            .await
234            .map_err(DbError::from)?;
235        let removed: Vec<RemovedRow> = result.take(1).map_err(DbError::from)?;
236        Ok(removed.len() as u64)
237    }
238}
