1//! SurrealDB implementation of [`SsfStepUpRepository`] (G-5, T23.5.3, D-53 (1),
2//! schema v78's `ssf_step_up`).
3//!
4//! One row per `(tenant, user)` — the record id is derived from the pair and a
5//! unique index backs it — so the latest step-up replaces the earlier one, and
6//! a row lives [`STEP_UP_RECORD_TTL_MINUTES`] minutes. `take` deletes in the
7//! same statement that returns the row, so a record is consumed exactly once
8//! however many return legs race for it, and never by a return leg made in the
9//! session the step-up was asked of (F4 W4 P23W4-02).
10//!
11//! The row holds ids and one `acr` URN: no credential, no address.
12
13use axiam_core::error::AxiamResult;
14use axiam_core::models::ssf::{STEP_UP_RECORD_TTL_MINUTES, SsfStepUp};
15use axiam_core::repository::SsfStepUpRepository;
16use chrono::{DateTime, Duration, Utc};
17use surrealdb::Connection;
18use surrealdb_types::SurrealValue;
19use uuid::Uuid;

21use crate::error::DbError;
22use crate::handle::DbHandle;
23use crate::helpers::{CountRow, classify_write_error};

25const ENTITY: &str = "ssf_step_up";
26
27#[derive(SurrealValue)]
28struct StepUpRow {
29    previous_session_id: String,
30    previous_acr: String,
31    expires_at: DateTime<Utc>,
32}
33
34#[derive(SurrealValue)]
35struct RemovedRow {
36    #[allow(dead_code)] // counted, never read
37    record_id: String,
38}

60impl<C: Connection> SurrealSsfStepUpRepository<C> {
61    /// Build the repository.
62    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
63        Self { db: db.into() }
64    }
65
66    /// The record id of a `(tenant, user)` pair.
67    fn record_id(tenant_id: Uuid, user_id: Uuid) -> String {
68        format!("{}_{}", tenant_id.simple(), user_id.simple())
69    }
70}

72impl<C: Connection> SsfStepUpRepository for SurrealSsfStepUpRepository<C> {
73    async fn put(&self, record: &SsfStepUp, now: DateTime<Utc>) -> AxiamResult<()> {
74        self.db
75            .current()
76            .query(
77                "UPSERT type::record('ssf_step_up', $record_id) SET \
78                 tenant_id = $tenant_id, user_id = $user_id, \
79                 previous_session_id = $previous_session_id, \
80                 previous_acr = $previous_acr, created_at = $now, \
81                 expires_at = $expires_at",
82            )
83            .bind((
84                "record_id",
85                Self::record_id(record.tenant_id, record.user_id),
86            ))
87            .bind(("tenant_id", record.tenant_id.to_string()))
88            .bind(("user_id", record.user_id.to_string()))
89            .bind((
90                "previous_session_id",
91                record.previous_session_id.to_string(),
92            ))
93            .bind(("previous_acr", record.previous_acr.clone()))
94            .bind(("now", now))
95            .bind((
96                "expires_at",
97                now + Duration::minutes(STEP_UP_RECORD_TTL_MINUTES),
98            ))
99            .await
100            .map_err(DbError::from)?
101            .check()
102            .map_err(|e| classify_write_error(e.to_string(), ENTITY))?;
103        Ok(())
104    }
105
106    async fn take(
107        &self,
108        tenant_id: Uuid,
109        user_id: Uuid,
110        current_session_id: Uuid,
111        now: DateTime<Utc>,
112    ) -> AxiamResult<Option<SsfStepUp>> {
113        // A return leg in the session the step-up was asked of leaves the record
114        // for the real one (F4 W4 P23W4-02): the guard is in the one statement.
115        let mut result = self
116            .db
117            .current()
118            .query(
119                "DELETE ssf_step_up WHERE tenant_id = $tenant_id AND user_id = $user_id \
120                 AND previous_session_id != $current_session_id RETURN BEFORE",
121            )
122            .bind(("tenant_id", tenant_id.to_string()))
123            .bind(("user_id", user_id.to_string()))
124            .bind(("current_session_id", current_session_id.to_string()))
125            .await
126            .map_err(DbError::from)?;
127        let rows: Vec<StepUpRow> = result.take(0).map_err(DbError::from)?;
128        let Some(row) = rows.into_iter().next() else {
129            return Ok(None);
130        };
131        // Consumed either way; only an unexpired record is a record.
132        if row.expires_at <= now {
133            return Ok(None);
134        }
135        let Ok(previous_session_id) = Uuid::parse_str(&row.previous_session_id) else {
136            return Ok(None);
137        };
138        Ok(Some(SsfStepUp {
139            tenant_id,
140            user_id,
141            previous_session_id,
142            previous_acr: row.previous_acr,
143        }))
144    }
145
146    async fn count_for_tenant(&self, tenant_id: Uuid) -> AxiamResult<u64> {
147        let mut result = self
148            .db
149            .current()
150            .query(
151                "SELECT count() AS total FROM ssf_step_up \
152                 WHERE tenant_id = $tenant_id GROUP ALL",
153            )
154            .bind(("tenant_id", tenant_id.to_string()))
155            .await
156            .map_err(DbError::from)?;
157        let rows: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
158        Ok(rows.first().map(|r| r.total).unwrap_or(0))
159    }
160
161    async fn delete_expired(&self, now: DateTime<Utc>) -> AxiamResult<u64> {
162        let mut result = self
163            .db
164            .current()
165            .query(
166                "LET $removed = (DELETE ssf_step_up WHERE expires_at <= $now RETURN BEFORE); \
167                 SELECT meta::id(id) AS record_id FROM $removed",
168            )
169            .bind(("now", now))
170            .await
171            .map_err(DbError::from)?;
172        let removed: Vec<RemovedRow> = result.take(1).map_err(DbError::from)?;
173        Ok(removed.len() as u64)
174    }
175}
