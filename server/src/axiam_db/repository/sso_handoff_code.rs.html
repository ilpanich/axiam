1//! SurrealDB implementation of [`SsoHandoffCodeRepository`].
2//!
3//! Single-use, 60-second codes that convert a cross-site SSO return (SAML, and
4//! Apple's `response_mode=form_post`) into a same-site session issuance without
5//! weakening `SameSite=Strict` on the session cookies. See
6//! [`axiam_core::repository::SsoHandoffCode`] and
7//! `claude_dev/federation-sso-login-design.md` §5.2.
8//!
9//! Only the SHA-256 hash of a code is ever written here. That is the difference
10//! between this table and `federation_login_state`, which stores its `state`
11//! raw: a `state` is a correlation value, while a handoff code is a bearer
12//! credential for a session, and a database read must not yield one.
13
14use axiam_core::error::{AxiamError, AxiamResult};
15use axiam_core::id::new_id;
16use axiam_core::repository::{SsoHandoffCode, SsoHandoffCodeRepository};
17use chrono::{DateTime, Utc};
18use surrealdb::Connection;
19use surrealdb_types::SurrealValue;
20use uuid::Uuid;
21
22use crate::error::DbError;
23use crate::handle::DbHandle;
24use crate::helpers::{CountRow, classify_conflict_write_error};
25
26#[derive(Debug, SurrealValue)]
27struct SsoHandoffCodeRow {
28    code_hash: String,
29    tenant_id: String,
30    user_id: String,
31    redirect_uri: String,
32    expires_at: DateTime<Utc>,
33}
34
35/// SurrealDB implementation of the SSO handoff-code repository.
36#[derive(Clone)]
37pub struct SurrealSsoHandoffCodeRepository<C: Connection> {
38    db: DbHandle<C>,
39}
40
41impl<C: Connection> SurrealSsoHandoffCodeRepository<C> {
42    /// Construct a repository over a database handle.
43    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
44        let db = db.into();
45        Self { db }
46    }
47}
48
49impl<C: Connection> SsoHandoffCodeRepository for SurrealSsoHandoffCodeRepository<C> {
50    async fn insert(&self, row: &SsoHandoffCode) -> AxiamResult<()> {
51        let row_id = new_id().to_string();
52
53        let result = self
54            .db
55            .current()
56            .query(
57                "CREATE type::record('sso_handoff_code', $id) SET \
58                 code_hash = $code_hash, \
59                 tenant_id = $tenant_id, \
60                 user_id = $user_id, \
61                 redirect_uri = $redirect_uri, \
62                 expires_at = $expires_at",
63            )
64            .bind(("id", row_id))
65            .bind(("code_hash", row.code_hash.clone()))
66            .bind(("tenant_id", row.tenant_id.to_string()))
67            .bind(("user_id", row.user_id.to_string()))
68            .bind(("redirect_uri", row.redirect_uri.clone()))
69            .bind(("expires_at", row.expires_at))
70            .await
71            .map_err(DbError::from)?;
72
73        result
74            .check()
75            // A UNIQUE violation is a hash collision on 256 bits of entropy,
76            // i.e. never in practice — but it is classified rather than
77            // swallowed, because the alternative reading is "somebody replayed
78            // a code", and that must not look like a success.
79            .map_err(|e| classify_conflict_write_error(e, "sso_handoff_code.code_hash"))
80            .map(|_| ())
81    }
82
83    async fn consume_by_hash(&self, code_hash: &str) -> AxiamResult<Option<SsoHandoffCode>> {
84        let hash_owned = code_hash.to_string();
85
86        // SELECT + DELETE in one transaction, exactly as `consume_by_state`
87        // does. The row is deleted whenever it is found — expired or not — so a
88        // second redemption of the same code cannot succeed even if the first
89        // one was rejected for expiry.
90        let mut result = self
91            .db
92            .current()
93            .query(
94                "BEGIN TRANSACTION; \
95                 LET $row = (SELECT code_hash, tenant_id, user_id, redirect_uri, expires_at \
96                             FROM sso_handoff_code \
97                             WHERE code_hash = $code_hash LIMIT 1); \
98                 DELETE sso_handoff_code WHERE code_hash = $code_hash; \
99                 RETURN $row; \
100                 COMMIT TRANSACTION",
101            )
102            .bind(("code_hash", hash_owned))
103            .await
104            .map_err(DbError::from)?;
105
106        // BEGIN=0, LET=1, DELETE=2, RETURN=3.
107        let rows: Vec<SsoHandoffCodeRow> = result
108            .take(3)
109            .map_err(|e| AxiamError::Database(e.to_string()))?;
110
111        let Some(row) = rows.into_iter().next() else {
112            return Ok(None);
113        };
114
115        let tenant_id = Uuid::parse_str(&row.tenant_id)
116            .map_err(|_| AxiamError::Database("invalid tenant_id in handoff row".into()))?;
117        let user_id = Uuid::parse_str(&row.user_id)
118            .map_err(|_| AxiamError::Database("invalid user_id in handoff row".into()))?;
119
120        // Expiry is checked after the delete, and both outcomes are reported to
121        // the caller as the same `None`: "expired" and "never existed" must be
122        // indistinguishable, for the same reason they are on the state row.
123        if row.expires_at <= Utc::now() {
124            return Ok(None);
125        }
126
127        Ok(Some(SsoHandoffCode {
128            code_hash: row.code_hash,
129            tenant_id,
130            user_id,
131            redirect_uri: row.redirect_uri,
132            expires_at: row.expires_at,
133        }))
134    }
135
136    async fn cleanup_expired(&self) -> AxiamResult<u64> {
137        let mut count_result = self
138            .db
139            .current()
140            .query(
141                "SELECT count() AS total FROM sso_handoff_code \
142                 WHERE expires_at < time::now() GROUP ALL",
143            )
144            .await
145            .map_err(DbError::from)?;
146
147        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
148        let total = count_rows.first().map(|r| r.total).unwrap_or(0);
149
150        self.db
151            .current()
152            .query("DELETE sso_handoff_code WHERE expires_at < time::now()")
153            .await
154            .map_err(DbError::from)?;
155
156        Ok(total)
157    }
158}
