18/// The `metadata` key of the tenant row an orphan purge's claim writes
19/// (R101W1-09 (a); [`SurrealTenantRepository::claim_orphan_purge`]).

22/// D-55: bumped whenever this process creates or deletes a tenant or an
23/// organization, **after** the write commits, so a tenant count cached at an
24/// earlier generation is known to be stale at once (the SSF shared-issuer gate,
25/// `axiam_oauth2::ssf::SsfIssuerGate`). Process-wide on purpose: every repository
26/// handle of the process writes the same datastore.
27static TENANT_GENERATION: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
28
29/// Record that the set of tenants or organizations changed (D-55).
30pub(crate) fn bump_tenant_generation() {
31    TENANT_GENERATION.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
32}
33
34/// The current tenant generation (D-55).
35#[must_use]
36pub fn tenant_generation() -> u64 {
37    TENANT_GENERATION.load(std::sync::atomic::Ordering::SeqCst)
38}

40fn parse_status(s: &str) -> Result<TenantStatus, DbError> {
41    match s {
42        "Active" => Ok(TenantStatus::Active),
43        "Suspended" => Ok(TenantStatus::Suspended),
44        other => Err(DbError::Migration(format!(
45            "unknown tenant status: {other}"
46        ))),
47    }
48}

57/// An absent or empty `kind` is [`TenantKind::Standard`].
58///
59/// Every tenant row written before organization scope existed has no `kind` at
60/// all, and every one of them is an ordinary tenant — so the absent case is not
61/// a parse failure, it is the answer. An *unrecognised* value is still an
62/// error: that is a row from a future version, and guessing at it would be
63/// guessing about an authorization boundary.
64fn parse_kind(s: Option<&str>) -> Result<TenantKind, DbError> {
65    match s.unwrap_or("").trim() {
66        "" | "standard" => Ok(TenantKind::Standard),
67        "organization" => Ok(TenantKind::Organization),
68        other => Err(DbError::Migration(format!("unknown tenant kind: {other}"))),
69    }
70}
71
72fn kind_to_str(k: TenantKind) -> &'static str {
73    match k {
74        TenantKind::Standard => "standard",
75        TenantKind::Organization => "organization",
76    }
77}

159/// The tenant purge (#523, D-4): what the cleanup job's `tenant_purge` sweep
160/// calls. See [`super::tenant_purge`] for the order and why it is that order.
161impl<C: Connection> SurrealTenantRepository<C> {
162    /// The tenants a deletion has tombstoned and the purge has not yet removed,
163    /// oldest deletion first.
164    ///
165    /// # Errors
166    ///
167    /// A datastore failure.
168    pub async fn list_tombstoned(&self) -> AxiamResult<Vec<Uuid>> {
169        #[derive(Debug, SurrealValue)]
170        struct IdRow {
171            record_id: String,
172        }
173        let mut result = self
174            .db
175            .current()
176            .query(
177                "SELECT meta::id(id) AS record_id, deleted_at FROM tenant \
178                 WHERE deleted_at != NONE ORDER BY deleted_at ASC",
179            )
180            .await
181            .map_err(DbError::from)?
182            .check()
183            .map_err(DbError::from)?;
184        let rows: Vec<IdRow> = result.take(0).map_err(DbError::from)?;
185        Ok(rows
186            .into_iter()
187            .filter_map(|r| Uuid::parse_str(&r.record_id).ok())
188            .collect())
189    }

191    /// Claim a tombstoned tenant's purge for this replica (#610): `true` when
192    /// this call won it, `false` when the tenant is not tombstoned (live, or
193    /// already purged) or another replica's claim, stamped after
194    /// `now - lease`, still holds.
195    ///
196    /// The cleanup job runs on every replica, so every replica's
197    /// `tenant_purge` sweep sees the same tombstoned tenant. The lease is the
198    /// precondition of the write (the SCIM `claim_reconciliation` pattern):
199    /// of two concurrent claimants, one write matches and stamps
200    /// `purge_claimed_at`; the other either matches nothing or loses the
201    /// optimistic write, retries, and then finds the winner's stamp. A claim
202    /// older than the lease is taken over, so a replica that claimed and died
203    /// holds the tenant back for one lease at most. A purge still running when
204    /// its lease runs out may then overlap another replica's; every step of
205    /// it is idempotent and only one of them removes the tenant row (see
206    /// [`Self::purge_tombstoned`]), so that costs work, not correctness.
207    ///
208    /// # Errors
209    ///
210    /// A datastore failure.
211    pub async fn claim_purge(
212        &self,
213        id: Uuid,
214        now: DateTime<Utc>,
215        lease: chrono::TimeDelta,
216    ) -> AxiamResult<bool> {
217        let cutoff = now - lease;
218        retry_hot_row(|| async {
219            let mut result = self
220                .db
221                .current()
222                .query(
223                    "UPDATE type::record('tenant', $id) SET purge_claimed_at = $now \
224                     WHERE deleted_at != NONE \
225                       AND (purge_claimed_at = NONE OR purge_claimed_at <= $cutoff) \
226                     RETURN VALUE meta::id(id)",
227                )
228                .bind(("id", id.to_string()))
229                .bind(("now", now))
230                .bind(("cutoff", cutoff))
231                .await
232                .map_err(DbError::from)?
233                .check()
234                .map_err(|e| classify_write_error(e, "tenant"))?;
235            let claimed: Vec<String> = result.take(0).map_err(DbError::from)?;
236            Ok::<_, DbError>(!claimed.is_empty())
237        })
238        .await
239        .map_err(Into::into)
240    }
241
242    /// Give up a claim [`Self::claim_purge`] stamped at `claimed_at`, so the
243    /// next sweep — this replica's or another's — retries a purge that failed
244    /// without waiting out the lease. A claim another replica has since taken
245    /// over carries a different stamp and is left alone.
246    ///
247    /// # Errors
248    ///
249    /// A datastore failure.
250    pub async fn release_purge_claim(
251        &self,
252        id: Uuid,
253        claimed_at: DateTime<Utc>,
254    ) -> AxiamResult<()> {
255        retry_hot_row(|| async {
256            self.db
257                .current()
258                .query(
259                    "UPDATE type::record('tenant', $id) SET purge_claimed_at = NONE \
260                     WHERE purge_claimed_at = $claimed_at",
261                )
262                .bind(("id", id.to_string()))
263                .bind(("claimed_at", claimed_at))
264                .await
265                .map_err(DbError::from)?
266                .check()
267                .map_err(|e| classify_write_error(e, "tenant"))?;
268            Ok::<_, DbError>(())
269        })
270        .await
271        .map_err(Into::into)
272    }

274    /// Remove every tenant-scoped row of a **tombstoned** tenant, then the
275    /// tenant row. Of its audit trail, only the rows the export its deletion
276    /// required holds go — those at or before the export time the
277    /// `tenants.deleted` record names, never later than the receipt it names
278    /// (R1W1-04, R101W3-03); the rest, or all of it when no export time can be
279    /// read, stays for the retention sweep. The report says which.
280    ///
281    /// Each table is one idempotent `DELETE`; a failure stops the purge with
282    /// the tenant still tombstoned, and calling this again resumes it. A live
283    /// tenant is refused before anything is touched — the purge never decides
284    /// on its own that a tenant is gone.
285    ///
286    /// `Ok` means **this call** removed the tenant row (#610): when two purges
287    /// of one tenant overlap — a claim whose lease ran out mid-purge, taken
288    /// over by another replica — both run the same idempotent steps, but only
289    /// one of their final `DELETE`s finds the row, and the other answers
290    /// `NotFound`, as for a tenant purged before the call. The caller records
291    /// the purge on `Ok` only, so it is recorded once.
292    ///
293    /// The row may be the claim marker of an orphan purge
294    /// ([`Self::claim_orphan_purge`]) whose replica died holding it: the id is
295    /// then purged as the orphan it is (its audit trail kept, as
296    /// [`Self::purge_orphan`] keeps it) and the report says so
297    /// (R101W1-09 (a)).
298    ///
299    /// # Errors
300    ///
301    /// `NotFound` for a tenant that is not tombstoned (live, or already
302    /// purged — before this call or while it ran); a datastore failure
303    /// otherwise.
304    pub async fn purge_tombstoned(
305        &self,
306        id: Uuid,
307    ) -> AxiamResult<super::tenant_purge::TenantPurgeReport> {
308        let report = self.purge_tombstoned_rows(id).await?;
309        self.remove_purged_tenant(id).await?;
310        Ok(report)
311    }

313    /// The first part of [`Self::purge_tombstoned`]: every tenant-scoped row of
314    /// a **tombstoned** tenant, not its tenant row (#655).
315    ///
316    /// The tenant stays tombstoned, so a purge stopped anywhere in here — a
317    /// failure, or the cleanup task giving the purge up because the process is
318    /// stopping — is resumed by the next call: every step is an idempotent
319    /// `DELETE`. Nothing here is a point of no return.
320    ///
321    /// The row may be the claim marker of an orphan purge
322    /// ([`Self::claim_orphan_purge`]): the id is then purged as the orphan it
323    /// is, its audit trail kept, and the report says so (R101W1-09 (a)).
324    ///
325    /// # Errors
326    ///
327    /// `NotFound` for a tenant that is not tombstoned (live, or already
328    /// purged); a datastore failure otherwise.
329    pub async fn purge_tombstoned_rows(
330        &self,
331        id: Uuid,
332    ) -> AxiamResult<super::tenant_purge::TenantPurgeReport> {
333        let Some(orphan_marker) = self.tombstoned_row_is_orphan_marker(id).await? else {
334            return Err(DbError::NotFound {
335                entity: "tombstoned tenant".into(),
336                id: id.to_string(),
337            }
338            .into());
339        };
340        let scope = if orphan_marker {
341            super::tenant_purge::PurgeScope::Orphan
342        } else {
343            super::tenant_purge::PurgeScope::Tombstoned
344        };
345        Ok(super::tenant_purge::purge_rows(&self.db, id, scope).await?)
346    }

348    /// `None` when `id` has no tombstoned tenant row; otherwise whether that
349    /// row is an orphan purge's claim marker ([`ORPHAN_PURGE_MARKER`]).
350    async fn tombstoned_row_is_orphan_marker(&self, id: Uuid) -> AxiamResult<Option<bool>> {
351        let mut result = self
352            .db
353            .current()
354            .query(format!(
355                "SELECT VALUE metadata.{ORPHAN_PURGE_MARKER} = true \
356                 FROM type::record('tenant', $id) WHERE deleted_at != NONE"
357            ))
358            .bind(("id", id.to_string()))
359            .await
360            .map_err(DbError::from)?
361            .check()
362            .map_err(DbError::from)?;
363        let rows: Vec<bool> = result.take(0).map_err(DbError::from)?;
364        Ok(rows.first().copied())
365    }

367    /// The last part of [`Self::purge_tombstoned`]: remove the tenant row, only
368    /// while it is still tombstoned (#655). Call it only after
369    /// [`Self::purge_tombstoned_rows`] has succeeded for the tenant; a tenant
370    /// row removed before its rows leaves them for the orphan scan. For an
371    /// orphaned id the row is its purge's claim marker, removed after
372    /// [`Self::purge_orphan_rows`] (R101W1-09 (a)).
373    ///
374    /// This is the purge's **point of no return**: once the row is gone, no
375    /// later sweep finds the tenant, so whoever removed it is the only one that
376    /// can write `tenants.purged`. `Ok` means this call removed it.
377    ///
378    /// # Errors
379    ///
380    /// `NotFound` when the row is not there or not tombstoned — an overlapping
381    /// purge removed it first (#610); a datastore failure otherwise.
382    pub async fn remove_purged_tenant(&self, id: Uuid) -> AxiamResult<()> {
383        // Only while still tombstoned: the guard is in the statement too, so
384        // nothing here can remove a live tenant's row.
385        if !self.delete_tombstoned_row(id).await? {
386            return Err(DbError::NotFound {
387                entity: "tombstoned tenant".into(),
388                id: id.to_string(),
389            }
390            .into());
391        }
392        Ok(())
393    }

395    /// Delete `id`'s tenant row if it is tombstoned: `true` when **this call**
396    /// removed it (`RETURN BEFORE`), `false` when it was already gone — the
397    /// one fact that keeps two overlapping purges from both recording.
398    async fn delete_tombstoned_row(&self, id: Uuid) -> AxiamResult<bool> {
399        let mut result = self
400            .db
401            .current()
402            .query(
403                "SELECT count() AS total FROM (DELETE type::record('tenant', $id) \
404                 WHERE deleted_at != NONE RETURN BEFORE) GROUP ALL",
405            )
406            .bind(("id", id.to_string()))
407            .await
408            .map_err(DbError::from)?
409            .check()
410            .map_err(DbError::from)?;
411        let removed: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
412        Ok(removed.first().is_some_and(|r| r.total > 0))
413    }

430    /// Claim an orphaned tenant id's purge for this replica (R101W1-09 (a)):
431    /// `true` when this call created the id's claim marker, `false` when the id
432    /// already has a tenant row — another replica's marker, or a tenant of its
433    /// own.
434    ///
435    /// An orphan has no tenant row for [`Self::claim_purge`] to stamp, so the
436    /// claim **is** a row: a tombstoned tenant row under the orphan's own id,
437    /// with `metadata.orphan_purge = true` and `purge_claimed_at = now`. A
438    /// record id is unique, so of replicas whose orphan scans coincide exactly
439    /// one `CREATE` succeeds (the `bootstrap_lock` and `organization_scope`
440    /// pattern); the others fail on the existing id, or lose the optimistic
441    /// write, retry, and then fail on it. While the marker exists the id is no
442    /// orphan to any scan (the scan reads the tenant table), it is out of every
443    /// tenant read (each filters `deleted_at`), and the tombstoned-tenant sweep
444    /// leaves it alone for [`Self::claim_purge`]'s lease — then takes it over,
445    /// as an orphan, if the claimant died. No `has_tenant` edge is written and
446    /// the organization id is the nil id, so no organization's listing reaches
447    /// it.
448    ///
449    /// # Errors
450    ///
451    /// A datastore failure.
452    pub async fn claim_orphan_purge(&self, id: Uuid, now: DateTime<Utc>) -> AxiamResult<bool> {
453        let created = retry_hot_row(|| async {
454            let mut result = self
455                .db
456                .current()
457                .query(format!(
458                    "CREATE type::record('tenant', $id) SET \
459                       organization_id = $nil, \
460                       name = 'orphaned tenant id', \
461                       slug = $slug, \
462                       metadata = {{ {ORPHAN_PURGE_MARKER}: true }}, \
463                       deleted_at = $now, \
464                       purge_claimed_at = $now \
465                     RETURN VALUE meta::id(id)"
466                ))
467                .bind(("id", id.to_string()))
468                .bind(("nil", Uuid::nil().to_string()))
469                .bind(("slug", format!("orphan-purge-{id}")))
470                .bind(("now", now))
471                .await
472                .map_err(DbError::from)?
473                .check()
474                .map_err(|e| classify_write_error(e, "tenant"))?;
475            let created: Vec<String> = result.take(0).map_err(DbError::from)?;
476            Ok::<_, DbError>(!created.is_empty())
477        })
478        .await;
479        match created {
480            Ok(created) => Ok(created),
481            Err(DbError::AlreadyExists { .. }) => Ok(false),
482            Err(e) => Err(e.into()),
483        }
484    }

486    /// Remove the claim marker [`Self::claim_orphan_purge`] wrote at
487    /// `claimed_at`: the orphan purge found nothing to remove, or failed and
488    /// leaves the id to the next scan. A marker another replica has since
489    /// taken over carries a different stamp and is left alone.
490    async fn drop_orphan_claim(&self, id: Uuid, claimed_at: DateTime<Utc>) -> AxiamResult<()> {
491        retry_hot_row(|| async {
492            self.db
493                .current()
494                .query(format!(
495                    "DELETE type::record('tenant', $id) \
496                     WHERE metadata.{ORPHAN_PURGE_MARKER} = true \
497                       AND purge_claimed_at = $claimed_at"
498                ))
499                .bind(("id", id.to_string()))
500                .bind(("claimed_at", claimed_at))
501                .await
502                .map_err(DbError::from)?
503                .check()
504                .map_err(|e| classify_write_error(e, "tenant"))?;
505            Ok::<_, DbError>(())
506        })
507        .await
508        .map_err(Into::into)
509    }

511    /// Remove an orphaned tenant id's rows, as [`Self::purge_tombstoned`] does
512    /// for a tombstoned tenant, except its audit trail, which no export receipt
513    /// covers and which the audit retention sweep governs (the report counts
514    /// it as kept).
515    ///
516    /// `Ok(Some(report))` means **this call** purged the id and the caller
517    /// records it; `Ok(None)` that there is nothing for this call to record
518    /// (R101W1-09 (a)). Every replica scans for orphans, so two scans may name
519    /// one id. The purge is claimed first ([`Self::claim_orphan_purge`]): while
520    /// one replica holds the id, another's call answers `None` without
521    /// touching a row. A scan that predates a finished purge of the id claims
522    /// it afresh, finds no row left to remove, drops its marker and answers
523    /// `None`. The purge's last step deletes the marker, and only the call
524    /// whose `DELETE` removed it answers `Some` — should a purge outlive the
525    /// claim's lease and the tombstoned-tenant sweep take the marker over, only
526    /// one of the two records the purge.
527    ///
528    /// The composition of [`Self::begin_orphan_purge`],
529    /// [`Self::purge_orphan_rows`] and [`Self::remove_purged_tenant`] (the
530    /// point of no return), which the cleanup task runs step by step so that
531    /// a stop gives the purge up before that point (#655).
532    ///
533    /// # Errors
534    ///
535    /// `Conflict` when a tenant row of its own exists for this id (live, or
536    /// tombstoned by a deletion); a datastore failure otherwise, after which
537    /// the marker is dropped so that the next scan finds the id again.
538    pub async fn purge_orphan(
539        &self,
540        id: Uuid,
541    ) -> AxiamResult<Option<super::tenant_purge::TenantPurgeReport>> {
542        let Some(claimed_at) = self.begin_orphan_purge(id).await? else {
543            return Ok(None);
544        };
545        let report = self.purge_orphan_rows(id, claimed_at).await?;
546        match self.remove_purged_tenant(id).await {
547            Ok(()) => Ok(Some(report)),
548            Err(axiam_core::error::AxiamError::NotFound { .. }) => Ok(None),
549            Err(e) => Err(e),
550        }
551    }

553    /// The first part of [`Self::purge_orphan`]: claim the orphaned id
554    /// ([`Self::claim_orphan_purge`]) and check, under the claim, that it
555    /// still owns a row to purge (R101W1-09 (a)).
556    ///
557    /// `Ok(Some(claimed_at))` when this call holds the claim of an id with
558    /// rows to purge: the caller goes on with [`Self::purge_orphan_rows`], or,
559    /// giving the purge up, releases the claim
560    /// ([`Self::release_purge_claim`] with `claimed_at`), which leaves the
561    /// marker to the tombstoned-tenant sweep to purge as the orphan it is.
562    /// `Ok(None)` when another replica holds the id, or when a scan that
563    /// predates a finished purge finds nothing left (its marker is dropped):
564    /// nothing for this call to record.
565    ///
566    /// # Errors
567    ///
568    /// `Conflict` when a tenant row of its own exists for this id; a datastore
569    /// failure otherwise, after which the marker is dropped.
570    pub async fn begin_orphan_purge(&self, id: Uuid) -> AxiamResult<Option<DateTime<Utc>>> {
571        let claimed_at = Utc::now();
572        if !self.claim_orphan_purge(id, claimed_at).await? {
573            let mut result = self
574                .db
575                .current()
576                .query(format!(
577                    "SELECT VALUE metadata.{ORPHAN_PURGE_MARKER} = true \
578                     FROM type::record('tenant', $id)"
579                ))
580                .bind(("id", id.to_string()))
581                .await
582                .map_err(DbError::from)?
583                .check()
584                .map_err(DbError::from)?;
585            let marker: Vec<bool> = result.take(0).map_err(DbError::from)?;
586            if marker.first() == Some(&false) {
587                return Err(axiam_core::error::AxiamError::Conflict {
588                    reason: "a tenant row exists for this id; it is not an orphan".into(),
589                });
590            }
591            // Another replica's claim (or one dropped again since): not ours.
592            return Ok(None);
593        }
594
595        match super::tenant_purge::has_purgeable_rows(&self.db, id).await {
596            Ok(true) => Ok(Some(claimed_at)),
597            Ok(false) => {
598                self.drop_orphan_claim(id, claimed_at).await?;
599                Ok(None)
600            }
601            Err(e) => {
602                self.drop_orphan_claim_or_warn(id, claimed_at).await;
603                Err(e.into())
604            }
605        }
606    }
607
608    /// The second part of [`Self::purge_orphan`]: the orphaned id's rows, with
609    /// [`super::tenant_purge::PurgeScope::Orphan`], under the claim
610    /// [`Self::begin_orphan_purge`] stamped at `claimed_at` (#655).
611    ///
612    /// Every step is an idempotent `DELETE` and the marker stays, so nothing
613    /// here is a point of no return: a purge stopped in here is resumed by
614    /// whoever purges the marker next. The point of no return is
615    /// [`Self::remove_purged_tenant`], and only the call that removed the
616    /// marker records the purge.
617    ///
618    /// # Errors
619    ///
620    /// A datastore failure, after which the marker is dropped so that the next
621    /// scan finds the id again.
622    pub async fn purge_orphan_rows(
623        &self,
624        id: Uuid,
625        claimed_at: DateTime<Utc>,
626    ) -> AxiamResult<super::tenant_purge::TenantPurgeReport> {
627        match super::tenant_purge::purge_rows(&self.db, id, super::tenant_purge::PurgeScope::Orphan)
628            .await
629        {
630            Ok(report) => Ok(report),
631            Err(e) => {
632                self.drop_orphan_claim_or_warn(id, claimed_at).await;
633                Err(e.into())
634            }
635        }
636    }
637
638    /// [`Self::drop_orphan_claim`] after a failed purge: a drop that fails too
639    /// is logged, and the marker lapses with its lease.
640    async fn drop_orphan_claim_or_warn(&self, id: Uuid, claimed_at: DateTime<Utc>) {
641        if let Err(release) = self.drop_orphan_claim(id, claimed_at).await {
642            tracing::warn!(
643                error = %release,
644                tenant_id = %id,
645                "could not drop the orphan purge's claim; it lapses with its lease"
646            );
647        }
648    }

652    async fn create(&self, input: CreateTenant) -> AxiamResult<Tenant> {
653        let id = new_id();
654        let id_str = id.to_string();
655        let org_id_str = input.organization_id.to_string();
656        let metadata = input
657            .metadata
658            .unwrap_or(serde_json::Value::Object(Default::default()));
659
660        // Create tenant record and relate to organization in one query.
661        // RELATE requires literal record-id syntax, so we embed UUIDs
662        // directly in the RELATE portion (they are safe — UUID format).
663        //
664        // An organization tenant also claims `organization_scope:<org_id>`,
665        // whose record id IS the constraint that there is only one of them. A
666        // second attempt fails on that CREATE rather than quietly producing a
667        // second place organization-level principals could live — see
668        // `SCHEMA_V50` for why this is a marker row and not a partial unique
669        // index.
670        let claim_scope = if input.kind.is_organization() {
671            " CREATE type::record('organization_scope', $org_id) \
672              SET tenant_id = $id;"
673        } else {
674            ""
675        };
676        let query = format!(
677            "CREATE type::record('tenant', $id) SET \
678             organization_id = $org_id, \
679             name = $name, slug = $slug, \
680             status = 'Active', \
681             kind = $kind, \
682             metadata = $metadata; \
683             RELATE organization:`{org_id_str}` \
684             -> has_tenant -> tenant:`{id_str}`;{claim_scope}"
685        );
686
687        let result = self
688            .db
689            .current()
690            .query(query)
691            .bind(("id", id_str.clone()))
692            .bind(("org_id", org_id_str))
693            .bind(("name", input.name))
694            .bind(("slug", input.slug))
695            .bind(("kind", kind_to_str(input.kind)))
696            .bind(("metadata", metadata))
697            .await
698            .map_err(DbError::from)?;
699
700        // A duplicate (organization_id, slug) trips the `idx_tenant_org_slug`
701        // unique index. Left unclassified it becomes `DbError::Migration`, which
702        // the API layer maps to a bare 500 `{"error":"internal_error"}` — so
703        // re-running any provisioning script answers "something broke" for a
704        // system that is in fact perfectly fine, and the documented 409 branch
705        // of `scripts/e2e-bootstrap.sh` was dead code.
706        //
707        // Per D-09 the marker matching lives in `helpers::classify_write_error`
708        // and nowhere else, so that a marker added there reaches every call
709        // site at once.
710        let mut result = result
711            .check()
712            .map_err(|e| classify_write_error(e, "tenant"))?;
713
714        // Statement 0 is the CREATE, statement 1 is the RELATE.
715        let rows: Vec<TenantRow> = result.take(0).map_err(DbError::from)?;
716        let row = take_first_or_not_found(rows, "tenant", &id_str)?;
717        bump_tenant_generation();
718
719        Ok(row.into_tenant(id)?)
720    }

793    async fn update(&self, id: Uuid, input: UpdateTenant) -> AxiamResult<Tenant> {
794        let id_str = id.to_string();
795
796        let mut sets = Vec::new();
797        if input.name.is_some() {
798            sets.push("name = $name");
799        }
800        if input.slug.is_some() {
801            sets.push("slug = $slug");
802        }
803        if input.status.is_some() {
804            sets.push("status = $status");
805        }
806        if input.metadata.is_some() {
807            sets.push("metadata = $metadata");
808        }
809        sets.push("updated_at = time::now()");
810
811        // #523: a tombstoned tenant is not there to update — `NotFound`, as
812        // every read answers.
813        let query = format!(
814            "UPDATE type::record('tenant', $id) SET {} WHERE deleted_at = NONE",
815            sets.join(", ")
816        );
817
818        let db = self.db.current();
819        let mut builder = db.query(&query).bind(("id", id_str.clone()));
820
821        if let Some(name) = input.name {
822            builder = builder.bind(("name", name));
823        }
824        if let Some(slug) = input.slug {
825            builder = builder.bind(("slug", slug));
826        }
827        if let Some(status) = input.status {
828            builder = builder.bind(("status", status_to_str(&status).to_string()));
829        }
830        if let Some(metadata) = input.metadata {
831            builder = builder.bind(("metadata", metadata));
832        }
833
834        let result = builder.await.map_err(DbError::from)?;
835        let mut result = result
836            .check()
837            .map_err(|e| DbError::Migration(e.to_string()))?;
838
839        let rows: Vec<TenantRow> = result.take(0).map_err(DbError::from)?;
840        let row = take_first_or_not_found(rows, "tenant", &id_str)?;
841
842        Ok(row.into_tenant(id)?)
843    }

845    async fn delete(&self, id: Uuid) -> AxiamResult<()> {
846        // #523 (P23W2-04, D-4): deleting a tenant TOMBSTONES it. The row keeps
847        // its data and gains `deleted_at`, which takes it out of every read in
848        // this file (and the settings repository's tenant lookup): sign-in,
849        // token issuance, refresh and every handler that resolves the tenant
850        // answer as for a tenant that does not exist, from this commit on. The
851        // cleanup job's `tenant_purge` sweep then removes every tenant-scoped
852        // row in the order user erasure uses (`tenant_purge`), and the tenant
853        // row last. The slug stays claimed by the tombstone until then, so a
854        // new tenant cannot take it while the old one's rows are still there.
855        //
856        // In the same transaction, and so in the request:
857        //
858        // * Its sessions are deleted and its refresh tokens revoked — the
859        //   erasure's first step, so the tenant's last session cannot refresh
860        //   and its access tokens fail the per-request session check. The
861        //   handler revoked them once already, through the repositories that
862        //   publish to the revocation feed and drop the validity cache; this
863        //   pass catches a sign-in that raced it, atomically with the stamp.
864        //
865        // * The rows that make AXIAM act for the tenant, or hold its credentials
866        //   for systems outside AXIAM, go now rather than at the purge, so no
867        //   background job — the directory sync, the SCIM delivery and
868        //   reconciliation, the SSF transmitter — works for a tenant that no
869        //   longer exists: its directory configuration and sync state (T23.3.1,
870        //   T23.3.5: an encrypted bind credential, account ids), its SAML
871        //   service providers, IdP signing credential, pending `AuthnRequest`s
872        //   and single-logout state (T23.2.1–T23.2.4: a sealed private key, the
873        //   `NameID` each SP was given), its SSF streams, step-up records and
874        //   buffered events (T23.5.2: a sealed receiver credential, subjects'
875        //   addresses), its CIBA requests and device grants (#641: an approved
876        //   grant is a bearer for an access token, and a user code is the
877        //   tenant's own), and its SCIM targets with their links and delivery
878        //   state.
879        //
880        // * R1W1-01: its certificates and signing CAs are revoked. The handler
881        //   revoked them once already, through the services that forward a
882        //   `vault_pki` leaf to Vault and release a CA's key; this pass catches
883        //   one issued while it ran, atomically with the stamp. A deleted
884        //   tenant's certificates are disowned, so they go on their issuers'
885        //   revocation lists — and the purge keeps each revoked row until
886        //   eight days after it expires, so they stay there (`tenant_purge`).
887        //
888        // F4 P23W2-02: and a transaction that rolled back is an error. The
889        // driver reports a failed statement inside the response, not from
890        // `.await`, so without `check` a cancelled delete answered `Ok` — the
891        // handler then answered `204` and wrote a "tenant deleted" record for
892        // a tenant that still existed, with its encrypted bind secret.
893        self.db
894            .current()
895            .query(
896                "BEGIN TRANSACTION; \
897                 DELETE session WHERE tenant_id = $id; \
898                 UPDATE oauth2_refresh_token SET revoked = true \
899                     WHERE tenant_id = $id AND revoked = false; \
900                 DELETE directory_config WHERE tenant_id = $id; \
901                 DELETE directory_sync_state WHERE tenant_id = $id; \
902                 DELETE saml_service_provider WHERE tenant_id = $id; \
903                 DELETE saml_idp_credential WHERE tenant_id = $id; \
904                 DELETE saml_authn_request WHERE tenant_id = $id; \
905                 DELETE saml_sp_session WHERE tenant_id = $id; \
906                 DELETE saml_logout_run WHERE tenant_id = $id; \
907                 DELETE ssf_event_buffer WHERE tenant_id = $id; \
908                 DELETE ssf_step_up WHERE tenant_id = $id; \
909                 DELETE ciba_request WHERE tenant_id = $id; \
910                 DELETE device_grant WHERE tenant_id = $id; \
911                 DELETE ssf_stream WHERE tenant_id = $id; \
912                 DELETE scim_target_link WHERE tenant_id = $id; \
913                 DELETE scim_target_state WHERE tenant_id = $id; \
914                 DELETE scim_target WHERE tenant_id = $id; \
915                 UPDATE certificate SET status = 'Revoked', \
916                     revoked_at = revoked_at ?? time::now() \
917                     WHERE tenant_id = $id AND status != 'Revoked' \
918                       AND not_after > time::now(); \
919                 UPDATE ca_certificate SET status = 'Revoked', \
920                     revoked_at = revoked_at ?? time::now() \
921                     WHERE tenant_id = $id AND status != 'Revoked' \
922                       AND not_after > time::now(); \
923                 UPDATE type::record('tenant', $id) \
924                     SET deleted_at = time::now(), updated_at = time::now() \
925                     WHERE deleted_at = NONE; \
926                 COMMIT TRANSACTION;",
927            )
928            .bind(("id", id.to_string()))
929            .await
930            .map_err(DbError::from)?
931            .check()
932            .map_err(DbError::from)?;
933
934        bump_tenant_generation();
935        Ok(())
936    }

994                .current()
995                .query(
996                    "SELECT count() AS total FROM tenant \

998                           AND deleted_at = NONE GROUP ALL; \
999                     SELECT count() AS total FROM organization GROUP ALL",
1000                )
1001                .await
1002                .map_err(DbError::from)?;
1003            let tenants: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
1004            let organizations: Vec<CountRow> = result.take(1).map_err(DbError::from)?;
1005            let tenants = tenants.first().map_or(0, |r| r.total);
1006            let organizations = organizations.first().map_or(0, |r| r.total);
1007            Ok(tenants.max(organizations))
1008        })
