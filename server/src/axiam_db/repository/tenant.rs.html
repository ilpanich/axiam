20/// The condition, on a `tenant` row selected by record id (`$id`, `$nil`),
21/// that makes it an orphan purge's claim marker
22/// ([`SurrealTenantRepository::claim_orphan_purge`], R101W1-09 (a)): the nil
23/// organization id and the `orphan-purge-<id>` slug. A tenant is created under
24/// a real organization, so no API writes the pair, whereas `metadata` is
25/// caller-supplied and no key in it can mark a row (#664).
26const ORPHAN_MARKER_CONDITION: &str =
27    "organization_id = $nil AND slug = string::concat('orphan-purge-', $id)";

29/// A `tenant_purge_pending` marker (#689, schema v108) that
30/// [`SurrealTenantRepository::claim_pending_purge_records`] claimed: a purge
31/// past its point of no return whose `tenants.purged` record its unit may not
32/// have written.
33#[derive(Debug, Clone, Copy, PartialEq, Eq)]
34pub struct PendingPurgeRecord {
35    /// The purged tenant's id.
36    pub tenant_id: Uuid,
37    /// What the record says, as the purge reported it.
38    pub report: super::tenant_purge::TenantPurgeReport,
39    /// When the tenant row was removed: a `tenants.purged` for the id written
40    /// since then is this purge's.
41    pub purged_at: DateTime<Utc>,
42    /// The claim this call holds on the marker.
43    pub claimed_at: DateTime<Utc>,
44}

46/// D-55: bumped whenever this process creates or deletes a tenant or an
47/// organization, **after** the write commits, so a tenant count cached at an
48/// earlier generation is known to be stale at once (the SSF shared-issuer gate,
49/// `axiam_oauth2::ssf::SsfIssuerGate`). Process-wide on purpose: every repository
50/// handle of the process writes the same datastore.
51static TENANT_GENERATION: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
52
53/// Record that the set of tenants or organizations changed (D-55).
54pub(crate) fn bump_tenant_generation() {
55    TENANT_GENERATION.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
56}

58/// The current tenant generation (D-55).
59#[must_use]
60pub fn tenant_generation() -> u64 {
61    TENANT_GENERATION.load(std::sync::atomic::Ordering::SeqCst)
62}

64fn parse_status(s: &str) -> Result<TenantStatus, DbError> {
65    match s {
66        "Active" => Ok(TenantStatus::Active),
67        "Suspended" => Ok(TenantStatus::Suspended),
68        other => Err(DbError::Migration(format!(
69            "unknown tenant status: {other}"
70        ))),
71    }

81/// An absent or empty `kind` is [`TenantKind::Standard`].
82///
83/// Every tenant row written before organization scope existed has no `kind` at
84/// all, and every one of them is an ordinary tenant — so the absent case is not
85/// a parse failure, it is the answer. An *unrecognised* value is still an
86/// error: that is a row from a future version, and guessing at it would be
87/// guessing about an authorization boundary.
88fn parse_kind(s: Option<&str>) -> Result<TenantKind, DbError> {
89    match s.unwrap_or("").trim() {
90        "" | "standard" => Ok(TenantKind::Standard),
91        "organization" => Ok(TenantKind::Organization),
92        other => Err(DbError::Migration(format!("unknown tenant kind: {other}"))),
93    }
94}
95
96fn kind_to_str(k: TenantKind) -> &'static str {
97    match k {
98        TenantKind::Standard => "standard",
99        TenantKind::Organization => "organization",
100    }

195/// The tenant purge (#523, D-4): what the cleanup job's `tenant_purge` sweep
196/// calls. See [`super::tenant_purge`] for the order and why it is that order.
197impl<C: Connection> SurrealTenantRepository<C> {
198    /// The tenants a deletion has tombstoned and the purge has not yet removed,
199    /// oldest deletion first.
200    ///
201    /// # Errors
202    ///
203    /// A datastore failure.
204    pub async fn list_tombstoned(&self) -> AxiamResult<Vec<Uuid>> {
205        #[derive(Debug, SurrealValue)]
206        struct IdRow {
207            record_id: String,
208        }
209        let mut result = self
210            .db
211            .current()
212            .query(
213                "SELECT meta::id(id) AS record_id, deleted_at FROM tenant \
214                 WHERE deleted_at != NONE ORDER BY deleted_at ASC",
215            )
216            .await
217            .map_err(DbError::from)?
218            .check()
219            .map_err(DbError::from)?;
220        let rows: Vec<IdRow> = result.take(0).map_err(DbError::from)?;
221        Ok(rows
222            .into_iter()
223            .filter_map(|r| Uuid::parse_str(&r.record_id).ok())
224            .collect())
225    }

227    /// Claim a tombstoned tenant's purge for this replica (#610): `true` when
228    /// this call won it, `false` when the tenant is not tombstoned (live, or
229    /// already purged) or another replica's claim, stamped after
230    /// `now - lease`, still holds.
231    ///
232    /// The cleanup job runs on every replica, so every replica's
233    /// `tenant_purge` sweep sees the same tombstoned tenant. The lease is the
234    /// precondition of the write (the SCIM `claim_reconciliation` pattern):
235    /// of two concurrent claimants, one write matches and stamps
236    /// `purge_claimed_at`; the other either matches nothing or loses the
237    /// optimistic write, retries, and then finds the winner's stamp. A claim
238    /// older than the lease is taken over, so a replica that claimed and died
239    /// holds the tenant back for one lease at most. A purge still running when
240    /// its lease runs out may then overlap another replica's; every step of
241    /// it is idempotent and only one of them removes the tenant row (see
242    /// [`Self::purge_tombstoned`]), so that costs work, not correctness.
243    ///
244    /// # Errors
245    ///
246    /// A datastore failure.
247    pub async fn claim_purge(
248        &self,
249        id: Uuid,
250        now: DateTime<Utc>,
251        lease: chrono::TimeDelta,
252    ) -> AxiamResult<bool> {
253        let cutoff = now - lease;
254        retry_hot_row(|| async {
255            let mut result = self
256                .db
257                .current()
258                .query(
259                    "UPDATE type::record('tenant', $id) SET purge_claimed_at = $now \
260                     WHERE deleted_at != NONE \
261                       AND (purge_claimed_at = NONE OR purge_claimed_at <= $cutoff) \
262                     RETURN VALUE meta::id(id)",
263                )
264                .bind(("id", id.to_string()))
265                .bind(("now", now))
266                .bind(("cutoff", cutoff))
267                .await
268                .map_err(DbError::from)?
269                .check()
270                .map_err(|e| classify_write_error(e, "tenant"))?;
271            let claimed: Vec<String> = result.take(0).map_err(DbError::from)?;
272            Ok::<_, DbError>(!claimed.is_empty())
273        })
274        .await
275        .map_err(Into::into)
276    }
277
278    /// Give up a claim [`Self::claim_purge`] stamped at `claimed_at`, so the
279    /// next sweep — this replica's or another's — retries a purge that failed
280    /// without waiting out the lease. A claim another replica has since taken
281    /// over carries a different stamp and is left alone.
282    ///
283    /// # Errors
284    ///
285    /// A datastore failure.
286    pub async fn release_purge_claim(
287        &self,
288        id: Uuid,
289        claimed_at: DateTime<Utc>,
290    ) -> AxiamResult<()> {
291        retry_hot_row(|| async {
292            self.db
293                .current()
294                .query(
295                    "UPDATE type::record('tenant', $id) SET purge_claimed_at = NONE \
296                     WHERE purge_claimed_at = $claimed_at",
297                )
298                .bind(("id", id.to_string()))
299                .bind(("claimed_at", claimed_at))
300                .await
301                .map_err(DbError::from)?
302                .check()
303                .map_err(|e| classify_write_error(e, "tenant"))?;
304            Ok::<_, DbError>(())
305        })
306        .await
307        .map_err(Into::into)
308    }

310    /// Remove every tenant-scoped row of a **tombstoned** tenant, then the
311    /// tenant row. Of its audit trail, only the rows the export its deletion
312    /// required holds go — those at or before the export time the
313    /// `tenants.deleted` record names, never later than the receipt it names
314    /// (R1W1-04, R101W3-03); the rest, or all of it when no export time can be
315    /// read, stays for the retention sweep. The report says which.
316    ///
317    /// Each table is one idempotent `DELETE`; a failure stops the purge with
318    /// the tenant still tombstoned, and calling this again resumes it. A live
319    /// tenant is refused before anything is touched — the purge never decides
320    /// on its own that a tenant is gone.
321    ///
322    /// `Ok` means **this call** removed the tenant row (#610): when two purges
323    /// of one tenant overlap — a claim whose lease ran out mid-purge, taken
324    /// over by another replica — both run the same idempotent steps, but only
325    /// one of their final `DELETE`s finds the row, and the other answers
326    /// `NotFound`, as for a tenant purged before the call. The caller records
327    /// the purge on `Ok` only, so it is recorded once.
328    ///
329    /// The row may be the claim marker of an orphan purge
330    /// ([`Self::claim_orphan_purge`]) whose replica died holding it, or gave it
331    /// up: the id is then purged as the orphan it is (its audit trail kept, as
332    /// [`Self::purge_orphan`] keeps it) and the report says so
333    /// (R101W1-09 (a)).
334    ///
335    /// # Errors
336    ///
337    /// `NotFound` for a tenant that is not tombstoned (live, or already
338    /// purged — before this call or while it ran); a datastore failure
339    /// otherwise.
340    pub async fn purge_tombstoned(
341        &self,
342        id: Uuid,
343    ) -> AxiamResult<super::tenant_purge::TenantPurgeReport> {
344        let report = self.purge_tombstoned_rows(id).await?;
345        let removed_at = Utc::now();
346        self.remove_purged_tenant(id, &report, removed_at).await?;
347        // The caller records the purge (#689: no marker is left for it).
348        self.settle_purge_record(id, removed_at).await?;
349        Ok(report)
350    }

352    /// The first part of [`Self::purge_tombstoned`]: every tenant-scoped row of
353    /// a **tombstoned** tenant, not its tenant row (#655).
354    ///
355    /// The tenant stays tombstoned, so a purge stopped anywhere in here — a
356    /// failure, or the cleanup task giving the purge up because the process is
357    /// stopping — is resumed by the next call: every step is an idempotent
358    /// `DELETE`. Nothing here is a point of no return.
359    ///
360    /// The row may be the claim marker of an orphan purge
361    /// ([`Self::claim_orphan_purge`]): the id is then purged as the orphan it
362    /// is, its audit trail kept, and the report says so (R101W1-09 (a)).
363    ///
364    /// # Errors
365    ///
366    /// `NotFound` for a tenant that is not tombstoned (live, or already
367    /// purged); a datastore failure otherwise.
368    pub async fn purge_tombstoned_rows(
369        &self,
370        id: Uuid,
371    ) -> AxiamResult<super::tenant_purge::TenantPurgeReport> {
372        let Some(orphan_marker) = self.tombstoned_row_is_orphan_marker(id).await? else {
373            return Err(DbError::NotFound {
374                entity: "tombstoned tenant".into(),
375                id: id.to_string(),
376            }
377            .into());
378        };
379        let scope = if orphan_marker {
380            super::tenant_purge::PurgeScope::Orphan
381        } else {
382            super::tenant_purge::PurgeScope::Tombstoned
383        };
384        Ok(super::tenant_purge::purge_rows(&self.db, id, scope).await?)
385    }

387    /// `None` when `id` has no tombstoned tenant row; otherwise whether that
388    /// row is an orphan purge's claim marker ([`ORPHAN_MARKER_CONDITION`]).
389    async fn tombstoned_row_is_orphan_marker(&self, id: Uuid) -> AxiamResult<Option<bool>> {
390        let mut result = self
391            .db
392            .current()
393            .query(format!(
394                "SELECT VALUE ({ORPHAN_MARKER_CONDITION}) \
395                 FROM type::record('tenant', $id) WHERE deleted_at != NONE"
396            ))
397            .bind(("id", id.to_string()))
398            .bind(("nil", Uuid::nil().to_string()))
399            .await
400            .map_err(DbError::from)?
401            .check()
402            .map_err(DbError::from)?;
403        let rows: Vec<bool> = result.take(0).map_err(DbError::from)?;
404        Ok(rows.first().copied())
405    }

407    /// The last part of [`Self::purge_tombstoned`]: remove the tenant row, only
408    /// while it is still tombstoned (#655). Call it only after
409    /// [`Self::purge_tombstoned_rows`] has succeeded for the tenant; a tenant
410    /// row removed before its rows leaves them for the orphan scan. For an
411    /// orphaned id the row is its purge's claim marker, removed after
412    /// [`Self::purge_orphan_rows`] (R101W1-09 (a)).
413    ///
414    /// This is the purge's **point of no return**: once the row is gone, no
415    /// later sweep finds the tenant. So the same transaction writes the
416    /// `tenant_purge_pending` marker (#689, schema v108), keyed by the id, with
417    /// what `tenants.purged` says — `report`'s export time and counts, and
418    /// `orphan` when the report says so or the removed row was an orphan's
419    /// claim marker ([`ORPHAN_MARKER_CONDITION`]) — and `claimed_at`, the
420    /// caller's claim on it. The caller records the purge and then clears the
421    /// marker ([`Self::settle_purge_record`]); a caller cut in between leaves
422    /// it for [`Self::claim_pending_purge_records`]. `Ok` means this call
423    /// removed the row and wrote the marker.
424    ///
425    /// # Errors
426    ///
427    /// `NotFound` when the row is not there or not tombstoned — an overlapping
428    /// purge removed it first (#610); a datastore failure otherwise, in which
429    /// case neither the removal nor the marker happened.
430    pub async fn remove_purged_tenant(
431        &self,
432        id: Uuid,
433        report: &super::tenant_purge::TenantPurgeReport,
434        claimed_at: DateTime<Utc>,
435    ) -> AxiamResult<()> {
436        // Only while still tombstoned: the guard is in the statement too, so
437        // nothing here can remove a live tenant's row.
438        let statement = format!(
439            "BEGIN TRANSACTION; \
440             LET $gone = (DELETE type::record('tenant', $id) WHERE deleted_at != NONE \
441                 RETURN BEFORE); \
442             IF array::len($gone) > 0 {{ \
443                 CREATE type::record('tenant_purge_pending', $id) SET \
444                     orphan = $orphan OR array::len((SELECT VALUE id FROM $gone \
445                         WHERE {ORPHAN_MARKER_CONDITION})) > 0, \
446                     audit_exported_through = $exported_through, \
447                     audit_rows_deleted = $rows_deleted, \
448                     audit_rows_kept = $rows_kept, \
449                     purged_at = time::now(), \
450                     claimed_at = $claimed_at \
451                 RETURN NONE; \
452             }}; \
453             RETURN array::len($gone); \
454             COMMIT TRANSACTION;"
455        );
456        let mut result = self
457            .db
458            .current()
459            .query(statement)
460            .bind(("id", id.to_string()))
461            .bind(("nil", Uuid::nil().to_string()))
462            .bind(("orphan", report.orphan))
463            .bind(("exported_through", report.audit_exported_through))
464            .bind(("rows_deleted", report.audit_rows_deleted))
465            .bind(("rows_kept", report.audit_rows_kept))
466            .bind(("claimed_at", claimed_at))
467            .await
468            .map_err(DbError::from)?;
469        let errors = result.take_errors();
470        if !errors.is_empty() {
471            let combined = errors
472                .into_values()
473                .map(|e| e.to_string())
474                .collect::<Vec<_>>()
475                .join("; ");
476            return Err(classify_write_error(combined, "tenant_purge_pending").into());
477        }
478        let removed: Option<i64> = result.take(3).map_err(DbError::from)?;
479        if removed.unwrap_or(0) == 0 {
480            return Err(DbError::NotFound {
481                entity: "tombstoned tenant".into(),
482                id: id.to_string(),
483            }
484            .into());
485        }
486        Ok(())
487    }

489    /// Clear `id`'s `tenant_purge_pending` marker once its `tenants.purged`
490    /// record is written (#689): `true` when this call removed it. Conditional
491    /// on `claimed_at`, the claim the caller holds on it, so a marker another
492    /// replica has since taken over is left to that replica.
493    ///
494    /// # Errors
495    ///
496    /// A datastore failure; the marker stays, and the next sweep finds the
497    /// record already written and clears it.
498    pub async fn settle_purge_record(
499        &self,
500        id: Uuid,
501        claimed_at: DateTime<Utc>,
502    ) -> AxiamResult<bool> {
503        retry_hot_row(|| async {
504            let mut result = self
505                .db
506                .current()
507                .query(
508                    "SELECT count() AS total FROM (DELETE type::record('tenant_purge_pending', $id) \
509                     WHERE claimed_at = $claimed_at RETURN BEFORE) GROUP ALL",
510                )
511                .bind(("id", id.to_string()))
512                .bind(("claimed_at", claimed_at))
513                .await
514                .map_err(DbError::from)?
515                .check()
516                .map_err(|e| classify_write_error(e, "tenant_purge_pending"))?;
517            let removed: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
518            Ok::<_, DbError>(removed.first().is_some_and(|r| r.total > 0))
519        })
520        .await
521        .map_err(Into::into)
522    }

556    /// Claim, at `now`, every `tenant_purge_pending` marker whose purge nobody
557    /// is recording (#689): unclaimed (released by a unit that was cut), or
558    /// claimed at or before `now - lease` (a unit, or a sweep, that died
559    /// holding it). Each claim is a write conditional on the claim this call
560    /// read, so of two sweeps that read one marker exactly one takes it; the
561    /// other's write matches nothing. What is returned is this call's to
562    /// record, under the claim `now`, then to clear
563    /// ([`Self::settle_purge_record`]) or give up
564    /// ([`Self::release_purge_record`]).
565    ///
566    /// # Errors
567    ///
568    /// A datastore failure.
569    pub async fn claim_pending_purge_records(
570        &self,
571        now: DateTime<Utc>,
572        lease: chrono::TimeDelta,
573    ) -> AxiamResult<Vec<PendingPurgeRecord>> {
574        #[derive(Debug, SurrealValue)]
575        struct MarkerRow {
576            record_id: String,
577            claimed_at: Option<DateTime<Utc>>,
578        }
579        #[derive(Debug, SurrealValue)]
580        struct ClaimedRow {
581            orphan: bool,
582            audit_exported_through: Option<DateTime<Utc>>,
583            audit_rows_deleted: i64,
584            audit_rows_kept: i64,
585            purged_at: DateTime<Utc>,
586        }
587        let mut result = self
588            .db
589            .current()
590            .query(
591                "SELECT meta::id(id) AS record_id, claimed_at FROM tenant_purge_pending \
592                 WHERE claimed_at = NONE OR claimed_at <= $cutoff",
593            )
594            .bind(("cutoff", now - lease))
595            .await
596            .map_err(DbError::from)?
597            .check()
598            .map_err(DbError::from)?;
599        let markers: Vec<MarkerRow> = result.take(0).map_err(DbError::from)?;
600        let mut claimed = Vec::new();
601        for marker in markers {
602            let Ok(tenant_id) = Uuid::parse_str(&marker.record_id) else {
603                continue;
604            };
605            // Conditional on the claim read: NONE, or the stale stamp.
606            let condition = if marker.claimed_at.is_some() {
607                "claimed_at = $read"
608            } else {
609                "claimed_at = NONE"
610            };
611            let statement = format!(
612                "UPDATE type::record('tenant_purge_pending', $id) SET claimed_at = $now \
613                 WHERE {condition} \
614                 RETURN orphan, audit_exported_through, audit_rows_deleted, audit_rows_kept, \
615                     purged_at"
616            );
617            let rows = retry_hot_row(|| async {
618                let mut result = self
619                    .db
620                    .current()
621                    .query(statement.as_str())
622                    .bind(("id", marker.record_id.clone()))
623                    .bind(("now", now))
624                    .bind(("read", marker.claimed_at))
625                    .await
626                    .map_err(DbError::from)?
627                    .check()
628                    .map_err(|e| classify_write_error(e, "tenant_purge_pending"))?;
629                let rows: Vec<ClaimedRow> = result.take(0).map_err(DbError::from)?;
630                Ok::<_, DbError>(rows)
631            })
632            .await?;
633            if let Some(row) = rows.into_iter().next() {
634                claimed.push(PendingPurgeRecord {
635                    tenant_id,
636                    report: super::tenant_purge::TenantPurgeReport {
637                        audit_exported_through: row.audit_exported_through,
638                        audit_rows_deleted: u64::try_from(row.audit_rows_deleted).unwrap_or(0),
639                        audit_rows_kept: u64::try_from(row.audit_rows_kept).unwrap_or(0),
640                        orphan: row.orphan,
641                    },
642                    purged_at: row.purged_at,
643                    claimed_at: now,
644                });
645            }
646        }
647        Ok(claimed)
648    }

741    /// Claim an orphaned tenant id's purge for this replica (R101W1-09 (a)):
742    /// `true` when this call created the id's claim marker, `false` when it
743    /// did not — the id already has a tenant row (another replica's marker, or
744    /// a tenant of its own; [`Self::orphan_claim_refusal`] tells which), or it
745    /// owns no row left to purge.
746    ///
747    /// An orphan has no tenant row for [`Self::claim_purge`] to stamp, so the
748    /// claim **is** a row: a tombstoned tenant row under the orphan's own id,
749    /// with the nil organization id, the slug `orphan-purge-<id>` and
750    /// `purge_claimed_at = now`. A record id is unique, so of replicas whose
751    /// orphan scans coincide exactly one `CREATE` succeeds (the
752    /// `bootstrap_lock` and `organization_scope` pattern); the others fail on
753    /// the existing id, or lose the optimistic write, retry, and then fail on
754    /// it. While the marker exists the id is no orphan to any scan (the scan
755    /// reads the tenant table), it is out of every tenant read (each filters
756    /// `deleted_at`), and the tombstoned-tenant sweep leaves it alone for
757    /// [`Self::claim_purge`]'s lease — then takes it over, as an orphan, if the
758    /// claimant died or released it. No `has_tenant` edge is written and the
759    /// organization id is the nil id, so no organization's listing reaches it.
760    ///
761    /// The `CREATE` is conditional, in the same statement, on the id still
762    /// owning a row the orphan scan counts (#714): a scan that predates
763    /// another replica's finished purge writes no marker at all. A marker
764    /// therefore always stands for a purge nobody has recorded yet, so whoever
765    /// removes it — the claimant, or the tombstoned-tenant sweep taking it over
766    /// — records the purge, and it is recorded once. There is no window in
767    /// which the marker exists unchecked: the cleanup task's unit arms its
768    /// claim guard before this call, keyed by `now`, so a marker this call
769    /// writes is guarded from the instant it exists.
770    ///
771    /// # Errors
772    ///
773    /// A datastore failure; the marker may then exist or not, and releasing
774    /// the claim ([`Self::release_purge_claim`] with `now`) is right either way.
775    pub async fn claim_orphan_purge(&self, id: Uuid, now: DateTime<Utc>) -> AxiamResult<bool> {
776        let statement = format!(
777            "IF {} {{ \
778               CREATE type::record('tenant', $id) SET \
779                 organization_id = $nil, \
780                 name = 'orphaned tenant id', \
781                 slug = $slug, \
782                 metadata = {{}}, \
783                 deleted_at = $now, \
784                 purge_claimed_at = $now \
785               RETURN VALUE meta::id(id) \
786             }} ELSE {{ [] }}",
787            super::tenant_purge::purgeable_rows_condition()
788        );
789        let created = retry_hot_row(|| async {
790            let mut result = self
791                .db
792                .current()
793                .query(statement.as_str())
794                .bind(("id", id.to_string()))
795                .bind(("nil", Uuid::nil().to_string()))
796                .bind(("slug", format!("orphan-purge-{id}")))
797                .bind(("now", now))
798                .await
799                .map_err(DbError::from)?
800                .check()
801                .map_err(|e| classify_write_error(e, "tenant"))?;
802            let created: Vec<String> = result.take(0).map_err(DbError::from)?;
803            Ok::<_, DbError>(!created.is_empty())
804        })
805        .await;
806        match created {
807            Ok(created) => Ok(created),
808            Err(DbError::AlreadyExists { .. }) => Ok(false),
809            Err(e) => Err(e.into()),
810        }
811    }

813    /// Why [`Self::claim_orphan_purge`] answered `false`: `Conflict` when the
814    /// id has a tenant row of its own (live, or tombstoned by a deletion), so
815    /// it is no orphan; `Ok` when the row is another replica's marker, or when
816    /// there is no row and nothing was left to purge.
817    ///
818    /// # Errors
819    ///
820    /// `Conflict` as above; a datastore failure.
821    pub async fn orphan_claim_refusal(&self, id: Uuid) -> AxiamResult<()> {
822        let mut result = self
823            .db
824            .current()
825            .query(format!(
826                "SELECT VALUE ({ORPHAN_MARKER_CONDITION}) \
827                 FROM type::record('tenant', $id)"
828            ))
829            .bind(("id", id.to_string()))
830            .bind(("nil", Uuid::nil().to_string()))
831            .await
832            .map_err(DbError::from)?
833            .check()
834            .map_err(DbError::from)?;
835        let marker: Vec<bool> = result.take(0).map_err(DbError::from)?;
836        if marker.first() == Some(&false) {
837            return Err(axiam_core::error::AxiamError::Conflict {
838                reason: "a tenant row exists for this id; it is not an orphan".into(),
839            });
840        }
841        Ok(())
842    }

844    /// Remove an orphaned tenant id's rows, as [`Self::purge_tombstoned`] does
845    /// for a tombstoned tenant, except its audit trail, which no export receipt
846    /// covers and which the audit retention sweep governs (the report counts
847    /// it as kept).
848    ///
849    /// `Ok(Some(report))` means **this call** purged the id and the caller
850    /// records it; `Ok(None)` that there is nothing for this call to record
851    /// (R101W1-09 (a)). Every replica scans for orphans, so two scans may name
852    /// one id. The purge is claimed first ([`Self::claim_orphan_purge`]): while
853    /// one replica holds the id, another's call answers `None` without
854    /// touching a row, and a scan that predates a finished purge finds nothing
855    /// left, writes no marker and answers `None`. The purge's last step deletes
856    /// the marker, and only the call whose `DELETE` removed it answers `Some` —
857    /// should a purge outlive the claim's lease and the tombstoned-tenant sweep
858    /// take the marker over, only one of the two records the purge.
859    ///
860    /// The composition of [`Self::claim_orphan_purge`],
861    /// [`Self::purge_orphan_rows`] and [`Self::remove_purged_tenant`] (the
862    /// point of no return), which the cleanup task runs step by step, under
863    /// its claim guard, so that a stop gives the purge up before that point
864    /// (#655, #714).
865    ///
866    /// # Errors
867    ///
868    /// `Conflict` when a tenant row of its own exists for this id (live, or
869    /// tombstoned by a deletion); a datastore failure otherwise, after which
870    /// the claim is released, so that the tombstoned-tenant sweep resumes the
871    /// purge at once.
872    pub async fn purge_orphan(
873        &self,
874        id: Uuid,
875    ) -> AxiamResult<Option<super::tenant_purge::TenantPurgeReport>> {
876        let claimed_at = Utc::now();
877        match self.claim_orphan_purge(id, claimed_at).await {
878            Ok(true) => {}
879            Ok(false) => {
880                self.orphan_claim_refusal(id).await?;
881                return Ok(None);
882            }
883            Err(e) => {
884                self.release_orphan_claim_or_warn(id, claimed_at).await;
885                return Err(e);
886            }
887        }
888        let report = match self.purge_orphan_rows(id).await {
889            Ok(report) => report,
890            Err(e) => {
891                self.release_orphan_claim_or_warn(id, claimed_at).await;
892                return Err(e);
893            }
894        };
895        match self.remove_purged_tenant(id, &report, claimed_at).await {
896            Ok(()) => {
897                // The caller records the purge (#689: no marker is left for it).
898                self.settle_purge_record(id, claimed_at).await?;
899                Ok(Some(report))
900            }
901            Err(axiam_core::error::AxiamError::NotFound { .. }) => Ok(None),
902            Err(e) => {
903                self.release_orphan_claim_or_warn(id, claimed_at).await;
904                Err(e)
905            }
906        }
907    }

909    /// The second part of [`Self::purge_orphan`]: the orphaned id's rows, with
910    /// [`super::tenant_purge::PurgeScope::Orphan`], under the claim
911    /// [`Self::claim_orphan_purge`] wrote (#655).
912    ///
913    /// Every step is an idempotent `DELETE` and the marker stays, so nothing
914    /// here is a point of no return: a purge stopped in here, or failed, is
915    /// resumed by whoever purges the marker next once its claim is released.
916    /// The point of no return is [`Self::remove_purged_tenant`], and only the
917    /// call that removed the marker records the purge.
918    ///
919    /// # Errors
920    ///
921    /// A datastore failure; the marker stays, and the caller releases its
922    /// claim.
923    pub async fn purge_orphan_rows(
924        &self,
925        id: Uuid,
926    ) -> AxiamResult<super::tenant_purge::TenantPurgeReport> {
927        Ok(
928            super::tenant_purge::purge_rows(&self.db, id, super::tenant_purge::PurgeScope::Orphan)
929                .await?,
930        )
931    }

942        }
943    }

960    async fn create(&self, input: CreateTenant) -> AxiamResult<Tenant> {
961        let id = new_id();
962        let id_str = id.to_string();

964        let metadata = input
965            .metadata
966            .unwrap_or(serde_json::Value::Object(Default::default()));
967
968        // Create tenant record and relate to organization in one query.
969        // RELATE requires literal record-id syntax, so we embed UUIDs
970        // directly in the RELATE portion (they are safe — UUID format).
971        //
972        // An organization tenant also claims `organization_scope:<org_id>`,
973        // whose record id IS the constraint that there is only one of them. A
974        // second attempt fails on that CREATE rather than quietly producing a
975        // second place organization-level principals could live — see
976        // `SCHEMA_V50` for why this is a marker row and not a partial unique
977        // index.
978        let claim_scope = if input.kind.is_organization() {
979            " CREATE type::record('organization_scope', $org_id) \

981        } else {
982            ""
983        };

985            "CREATE type::record('tenant', $id) SET \
986             organization_id = $org_id, \
987             name = $name, slug = $slug, \
988             status = 'Active', \
989             kind = $kind, \
990             metadata = $metadata; \
991             RELATE organization:`{org_id_str}` \
992             -> has_tenant -> tenant:`{id_str}`;{claim_scope}"
993        );
994
995        let result = self
996            .db
997            .current()
998            .query(query)
999            .bind(("id", id_str.clone()))
1000            .bind(("org_id", org_id_str))
1001            .bind(("name", input.name))
1002            .bind(("slug", input.slug))
1003            .bind(("kind", kind_to_str(input.kind)))

1010        // the API layer maps to a bare 500 `{"error":"internal_error"}` — so
1011        // re-running any provisioning script answers "something broke" for a
1012        // system that is in fact perfectly fine, and the documented 409 branch
1013        // of `scripts/e2e-bootstrap.sh` was dead code.
1014        //
1015        // Per D-09 the marker matching lives in `helpers::classify_write_error`
1016        // and nowhere else, so that a marker added there reaches every call

1019            .check()

1023        let rows: Vec<TenantRow> = result.take(0).map_err(DbError::from)?;

1028    }

1109    async fn update(&self, id: Uuid, input: UpdateTenant) -> AxiamResult<Tenant> {
1110        let _forget = RecordCaches::tenant_write(self.record_caches.as_deref(), id);
1111        let id_str = id.to_string();
1112
1113        let mut sets = Vec::new();
1114        if input.name.is_some() {
1115            sets.push("name = $name");
1116        }
1117        if input.slug.is_some() {
1118            sets.push("slug = $slug");
1119        }
1120        if input.status.is_some() {
1121            sets.push("status = $status");
1122        }
1123        if input.metadata.is_some() {
1124            sets.push("metadata = $metadata");
1125        }
1126        sets.push("updated_at = time::now()");
1127
1128        // #523: a tombstoned tenant is not there to update — `NotFound`, as
1129        // every read answers.
1130        let update = format!(
1131            "UPDATE type::record('tenant', $id) SET {} WHERE deleted_at = NONE",
1132            sets.join(", ")
1133        );
1134        // SA-13 (N16, D-3): suspending a tenant revokes, in the same
1135        // transaction as the status write, every session and OAuth2 refresh
1136        // token it holds — the tombstone's first step (`delete` below). Its
1137        // access tokens then fail the per-request session check, its last
1138        // session cannot refresh, and un-suspending restores sign-in only
1139        // through a new session. The data stays. The handler revoked them once
1140        // already, through the repositories that publish to the revocation
1141        // feed and drop the validity cache; this pass catches a sign-in that
1142        // raced it, atomically with the status.
1143        let suspends = input.status == Some(TenantStatus::Suspended);
1144        // SA-13 (OD-13a): the organization's own tenant holds every
1145        // organization principal — the principals who lift a suspension — so it
1146        // is never suspended. Decided here, where the status write is, so no
1147        // caller can reach around it; refused before anything is written or
1148        // revoked. A read rather than a guard inside the transaction is enough:
1149        // `kind` is fixed at creation (`UpdateTenant` cannot change it), so
1150        // nothing can race it. A tombstoned tenant answers `NotFound` here, as
1151        // the write itself would.
1152        if suspends && self.get_by_id(id).await?.is_organization_scope() {
1153            return Err(axiam_core::error::AxiamError::OrganizationTenantNotSuspendable);
1154        }
1155        let (query, slot) = if suspends {
1156            (
1157                // BEGIN=0, DELETE=1, UPDATE refresh tokens=2, UPDATE tenant=3.
1158                format!(
1159                    "BEGIN TRANSACTION; \
1160                     DELETE session WHERE tenant_id = $id; \
1161                     UPDATE oauth2_refresh_token SET revoked = true \
1162                         WHERE tenant_id = $id AND revoked = false; \
1163                     {update}; \
1164                     COMMIT TRANSACTION"
1165                ),
1166                3,
1167            )
1168        } else {
1169            (update, 0)
1170        };
1171
1172        let db = self.db.current();
1173        let mut builder = db.query(&query).bind(("id", id_str.clone()));
1174
1175        if let Some(name) = input.name {
1176            builder = builder.bind(("name", name));
1177        }
1178        if let Some(slug) = input.slug {
1179            builder = builder.bind(("slug", slug));
1180        }
1181        if let Some(status) = input.status {
1182            builder = builder.bind(("status", status_to_str(&status).to_string()));
1183        }
1184        if let Some(metadata) = input.metadata {
1185            builder = builder.bind(("metadata", metadata));
1186        }
1187
1188        let result = builder.await.map_err(DbError::from)?;
1189        let mut result = result
1190            .check()
1191            .map_err(|e| DbError::Migration(e.to_string()))?;
1192
1193        let rows: Vec<TenantRow> = result.take(slot).map_err(DbError::from)?;
1194        let row = take_first_or_not_found(rows, "tenant", &id_str)?;
1195
1196        Ok(row.into_tenant(id)?)
1197    }

1199    async fn delete(&self, id: Uuid) -> AxiamResult<()> {
1200        let _forget = RecordCaches::tenant_write(self.record_caches.as_deref(), id);
1201        // #523 (P23W2-04, D-4): deleting a tenant TOMBSTONES it. The row keeps
1202        // its data and gains `deleted_at`, which takes it out of every read in
1203        // this file (and the settings repository's tenant lookup): sign-in,
1204        // token issuance, refresh and every handler that resolves the tenant
1205        // answer as for a tenant that does not exist, from this commit on. The
1206        // cleanup job's `tenant_purge` sweep then removes every tenant-scoped
1207        // row in the order user erasure uses (`tenant_purge`), and the tenant
1208        // row last. The slug stays claimed by the tombstone until then, so a
1209        // new tenant cannot take it while the old one's rows are still there.
1210        //
1211        // In the same transaction, and so in the request:
1212        //
1213        // * Its sessions are deleted and its refresh tokens revoked — the
1214        //   erasure's first step, so the tenant's last session cannot refresh
1215        //   and its access tokens fail the per-request session check. The
1216        //   handler revoked them once already, through the repositories that
1217        //   publish to the revocation feed and drop the validity cache; this
1218        //   pass catches a sign-in that raced it, atomically with the stamp.
1219        //
1220        // * The rows that make AXIAM act for the tenant, or hold its credentials
1221        //   for systems outside AXIAM, go now rather than at the purge, so no
1222        //   background job — the directory sync, the SCIM delivery and
1223        //   reconciliation, the SSF transmitter — works for a tenant that no
1224        //   longer exists: its directory configuration and sync state (T23.3.1,
1225        //   T23.3.5: an encrypted bind credential, account ids), its SAML
1226        //   service providers, IdP signing credential, pending `AuthnRequest`s
1227        //   and single-logout state (T23.2.1–T23.2.4: a sealed private key, the
1228        //   `NameID` each SP was given), its SSF streams, step-up records and
1229        //   buffered events (T23.5.2: a sealed receiver credential, subjects'
1230        //   addresses), its CIBA requests and device grants (#641: an approved
1231        //   grant is a bearer for an access token, and a user code is the
1232        //   tenant's own), and its SCIM targets with their links and delivery
1233        //   state.
1234        //
1235        // * SA-13 (N16): its SCIM provisioning tokens are revoked, so a
1236        //   provisioner's long-lived handle stops at the tombstone rather than
1237        //   at the purge (`SurrealScimTokenResolver::resolve` refuses a
1238        //   tombstoned or suspended tenant's tokens too).
1239        //
1240        // * R1W1-01: its certificates and signing CAs are revoked. The handler
1241        //   revoked them once already, through the services that forward a
1242        //   `vault_pki` leaf to Vault and release a CA's key; this pass catches
1243        //   one issued while it ran, atomically with the stamp. A deleted
1244        //   tenant's certificates are disowned, so they go on their issuers'
1245        //   revocation lists — and the purge keeps each revoked row until
1246        //   eight days after it expires, so they stay there (`tenant_purge`).
1247        //
1248        // F4 P23W2-02: and a transaction that rolled back is an error. The
1249        // driver reports a failed statement inside the response, not from
1250        // `.await`, so without `check` a cancelled delete answered `Ok` — the
1251        // handler then answered `204` and wrote a "tenant deleted" record for
1252        // a tenant that still existed, with its encrypted bind secret.
1253        self.db
1254            .current()
1255            .query(
1256                "BEGIN TRANSACTION; \
1257                 DELETE session WHERE tenant_id = $id; \
1258                 UPDATE oauth2_refresh_token SET revoked = true \
1259                     WHERE tenant_id = $id AND revoked = false; \
1260                 DELETE directory_config WHERE tenant_id = $id; \
1261                 DELETE directory_sync_state WHERE tenant_id = $id; \
1262                 DELETE saml_service_provider WHERE tenant_id = $id; \
1263                 DELETE saml_idp_credential WHERE tenant_id = $id; \
1264                 DELETE saml_authn_request WHERE tenant_id = $id; \
1265                 DELETE saml_sp_session WHERE tenant_id = $id; \
1266                 DELETE saml_logout_run WHERE tenant_id = $id; \
1267                 DELETE ssf_event_buffer WHERE tenant_id = $id; \
1268                 DELETE ssf_step_up WHERE tenant_id = $id; \
1269                 DELETE ciba_request WHERE tenant_id = $id; \
1270                 DELETE device_grant WHERE tenant_id = $id; \
1271                 DELETE ssf_stream WHERE tenant_id = $id; \
1272                 DELETE scim_target_link WHERE tenant_id = $id; \
1273                 DELETE scim_target_state WHERE tenant_id = $id; \
1274                 DELETE scim_target WHERE tenant_id = $id; \
1275                 UPDATE scim_token SET revoked_at = time::now() \
1276                     WHERE tenant_id = $id AND revoked_at = NONE; \
1277                 UPDATE certificate SET status = 'Revoked', \
1278                     revoked_at = revoked_at ?? time::now() \
1279                     WHERE tenant_id = $id AND status != 'Revoked' \
1280                       AND not_after > time::now(); \
1281                 UPDATE ca_certificate SET status = 'Revoked', \
1282                     revoked_at = revoked_at ?? time::now() \
1283                     WHERE tenant_id = $id AND status != 'Revoked' \
1284                       AND not_after > time::now(); \
1285                 UPDATE type::record('tenant', $id) \
1286                     SET deleted_at = time::now(), updated_at = time::now() \
1287                     WHERE deleted_at = NONE; \
1288                 COMMIT TRANSACTION;",
1289            )
1290            .bind(("id", id.to_string()))
1291            .await
1292            .map_err(DbError::from)?
1293            .check()
1294            .map_err(DbError::from)?;
1295
1296        bump_tenant_generation();
1297        Ok(())
1298    }
