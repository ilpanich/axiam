18/// The condition, on a `tenant` row selected by record id (`$id`, `$nil`),
19/// that makes it an orphan purge's claim marker
20/// ([`SurrealTenantRepository::claim_orphan_purge`], R101W1-09 (a)): the nil
21/// organization id and the `orphan-purge-<id>` slug. A tenant is created under
22/// a real organization, so no API writes the pair, whereas `metadata` is
23/// caller-supplied and no key in it can mark a row (#664).
24const ORPHAN_MARKER_CONDITION: &str =
25    "organization_id = $nil AND slug = string::concat('orphan-purge-', $id)";

27/// A `tenant_purge_pending` marker (#689, schema v108) that
28/// [`SurrealTenantRepository::claim_pending_purge_records`] claimed: a purge
29/// past its point of no return whose `tenants.purged` record its unit may not
30/// have written.
31#[derive(Debug, Clone, Copy, PartialEq, Eq)]
32pub struct PendingPurgeRecord {
33    /// The purged tenant's id.
34    pub tenant_id: Uuid,
35    /// What the record says, as the purge reported it.
36    pub report: super::tenant_purge::TenantPurgeReport,
37    /// When the tenant row was removed: a `tenants.purged` for the id written
38    /// since then is this purge's.
39    pub purged_at: DateTime<Utc>,
40    /// The claim this call holds on the marker.
41    pub claimed_at: DateTime<Utc>,
42}

44/// D-55: bumped whenever this process creates or deletes a tenant or an
45/// organization, **after** the write commits, so a tenant count cached at an
46/// earlier generation is known to be stale at once (the SSF shared-issuer gate,
47/// `axiam_oauth2::ssf::SsfIssuerGate`). Process-wide on purpose: every repository
48/// handle of the process writes the same datastore.
49static TENANT_GENERATION: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
50
51/// Record that the set of tenants or organizations changed (D-55).
52pub(crate) fn bump_tenant_generation() {
53    TENANT_GENERATION.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
54}

56/// The current tenant generation (D-55).
57#[must_use]
58pub fn tenant_generation() -> u64 {
59    TENANT_GENERATION.load(std::sync::atomic::Ordering::SeqCst)
60}

62fn parse_status(s: &str) -> Result<TenantStatus, DbError> {
63    match s {
64        "Active" => Ok(TenantStatus::Active),
65        "Suspended" => Ok(TenantStatus::Suspended),
66        other => Err(DbError::Migration(format!(
67            "unknown tenant status: {other}"
68        ))),
69    }

79/// An absent or empty `kind` is [`TenantKind::Standard`].
80///
81/// Every tenant row written before organization scope existed has no `kind` at
82/// all, and every one of them is an ordinary tenant — so the absent case is not
83/// a parse failure, it is the answer. An *unrecognised* value is still an
84/// error: that is a row from a future version, and guessing at it would be
85/// guessing about an authorization boundary.
86fn parse_kind(s: Option<&str>) -> Result<TenantKind, DbError> {
87    match s.unwrap_or("").trim() {
88        "" | "standard" => Ok(TenantKind::Standard),
89        "organization" => Ok(TenantKind::Organization),
90        other => Err(DbError::Migration(format!("unknown tenant kind: {other}"))),
91    }
92}
93
94fn kind_to_str(k: TenantKind) -> &'static str {
95    match k {
96        TenantKind::Standard => "standard",
97        TenantKind::Organization => "organization",
98    }

181/// The tenant purge (#523, D-4): what the cleanup job's `tenant_purge` sweep

184    /// The tenants a deletion has tombstoned and the purge has not yet removed,
185    /// oldest deletion first.

187    /// # Errors

189    /// A datastore failure.
190    pub async fn list_tombstoned(&self) -> AxiamResult<Vec<Uuid>> {
191        #[derive(Debug, SurrealValue)]
192        struct IdRow {
193            record_id: String,
194        }
195        let mut result = self
196            .db
197            .current()
198            .query(

200                 WHERE deleted_at != NONE ORDER BY deleted_at ASC",
201            )
202            .await
203            .map_err(DbError::from)?
204            .check()
205            .map_err(DbError::from)?;
206        let rows: Vec<IdRow> = result.take(0).map_err(DbError::from)?;
207        Ok(rows
208            .into_iter()
209            .filter_map(|r| Uuid::parse_str(&r.record_id).ok())
210            .collect())
211    }

213    /// Claim a tombstoned tenant's purge for this replica (#610): `true` when
214    /// this call won it, `false` when the tenant is not tombstoned (live, or
215    /// already purged) or another replica's claim, stamped after
216    /// `now - lease`, still holds.
217    ///
218    /// The cleanup job runs on every replica, so every replica's
219    /// `tenant_purge` sweep sees the same tombstoned tenant. The lease is the
220    /// precondition of the write (the SCIM `claim_reconciliation` pattern):
221    /// of two concurrent claimants, one write matches and stamps
222    /// `purge_claimed_at`; the other either matches nothing or loses the
223    /// optimistic write, retries, and then finds the winner's stamp. A claim
224    /// older than the lease is taken over, so a replica that claimed and died
225    /// holds the tenant back for one lease at most. A purge still running when
226    /// its lease runs out may then overlap another replica's; every step of
227    /// it is idempotent and only one of them removes the tenant row (see
228    /// [`Self::purge_tombstoned`]), so that costs work, not correctness.
229    ///
230    /// # Errors
231    ///
232    /// A datastore failure.
233    pub async fn claim_purge(
234        &self,
235        id: Uuid,
236        now: DateTime<Utc>,
237        lease: chrono::TimeDelta,
238    ) -> AxiamResult<bool> {
239        let cutoff = now - lease;
240        retry_hot_row(|| async {
241            let mut result = self
242                .db
243                .current()
244                .query(
245                    "UPDATE type::record('tenant', $id) SET purge_claimed_at = $now \
246                     WHERE deleted_at != NONE \
247                       AND (purge_claimed_at = NONE OR purge_claimed_at <= $cutoff) \
248                     RETURN VALUE meta::id(id)",
249                )
250                .bind(("id", id.to_string()))
251                .bind(("now", now))
252                .bind(("cutoff", cutoff))
253                .await
254                .map_err(DbError::from)?
255                .check()
256                .map_err(|e| classify_write_error(e, "tenant"))?;
257            let claimed: Vec<String> = result.take(0).map_err(DbError::from)?;
258            Ok::<_, DbError>(!claimed.is_empty())
259        })
260        .await
261        .map_err(Into::into)
262    }

291        })

294    }

296    /// Remove every tenant-scoped row of a **tombstoned** tenant, then the
297    /// tenant row. Of its audit trail, only the rows the export its deletion
298    /// required holds go — those at or before the export time the
299    /// `tenants.deleted` record names, never later than the receipt it names
300    /// (R1W1-04, R101W3-03); the rest, or all of it when no export time can be
301    /// read, stays for the retention sweep. The report says which.
302    ///
303    /// Each table is one idempotent `DELETE`; a failure stops the purge with
304    /// the tenant still tombstoned, and calling this again resumes it. A live
305    /// tenant is refused before anything is touched — the purge never decides
306    /// on its own that a tenant is gone.
307    ///
308    /// `Ok` means **this call** removed the tenant row (#610): when two purges
309    /// of one tenant overlap — a claim whose lease ran out mid-purge, taken
310    /// over by another replica — both run the same idempotent steps, but only
311    /// one of their final `DELETE`s finds the row, and the other answers
312    /// `NotFound`, as for a tenant purged before the call. The caller records
313    /// the purge on `Ok` only, so it is recorded once.
314    ///
315    /// The row may be the claim marker of an orphan purge
316    /// ([`Self::claim_orphan_purge`]) whose replica died holding it, or gave it
317    /// up: the id is then purged as the orphan it is (its audit trail kept, as
318    /// [`Self::purge_orphan`] keeps it) and the report says so
319    /// (R101W1-09 (a)).
320    ///
321    /// # Errors
322    ///
323    /// `NotFound` for a tenant that is not tombstoned (live, or already
324    /// purged — before this call or while it ran); a datastore failure
325    /// otherwise.
326    pub async fn purge_tombstoned(
327        &self,
328        id: Uuid,
329    ) -> AxiamResult<super::tenant_purge::TenantPurgeReport> {
330        let report = self.purge_tombstoned_rows(id).await?;
331        let removed_at = Utc::now();
332        self.remove_purged_tenant(id, &report, removed_at).await?;
333        // The caller records the purge (#689: no marker is left for it).
334        self.settle_purge_record(id, removed_at).await?;
335        Ok(report)
336    }

338    /// The first part of [`Self::purge_tombstoned`]: every tenant-scoped row of
339    /// a **tombstoned** tenant, not its tenant row (#655).
340    ///
341    /// The tenant stays tombstoned, so a purge stopped anywhere in here — a
342    /// failure, or the cleanup task giving the purge up because the process is
343    /// stopping — is resumed by the next call: every step is an idempotent
344    /// `DELETE`. Nothing here is a point of no return.
345    ///
346    /// The row may be the claim marker of an orphan purge
347    /// ([`Self::claim_orphan_purge`]): the id is then purged as the orphan it
348    /// is, its audit trail kept, and the report says so (R101W1-09 (a)).
349    ///
350    /// # Errors
351    ///
352    /// `NotFound` for a tenant that is not tombstoned (live, or already
353    /// purged); a datastore failure otherwise.
354    pub async fn purge_tombstoned_rows(
355        &self,
356        id: Uuid,
357    ) -> AxiamResult<super::tenant_purge::TenantPurgeReport> {
358        let Some(orphan_marker) = self.tombstoned_row_is_orphan_marker(id).await? else {
359            return Err(DbError::NotFound {
360                entity: "tombstoned tenant".into(),
361                id: id.to_string(),
362            }
363            .into());
364        };
365        let scope = if orphan_marker {
366            super::tenant_purge::PurgeScope::Orphan
367        } else {
368            super::tenant_purge::PurgeScope::Tombstoned
369        };
370        Ok(super::tenant_purge::purge_rows(&self.db, id, scope).await?)
371    }

373    /// `None` when `id` has no tombstoned tenant row; otherwise whether that
374    /// row is an orphan purge's claim marker ([`ORPHAN_MARKER_CONDITION`]).
375    async fn tombstoned_row_is_orphan_marker(&self, id: Uuid) -> AxiamResult<Option<bool>> {
376        let mut result = self
377            .db
378            .current()
379            .query(format!(
380                "SELECT VALUE ({ORPHAN_MARKER_CONDITION}) \
381                 FROM type::record('tenant', $id) WHERE deleted_at != NONE"
382            ))
383            .bind(("id", id.to_string()))
384            .bind(("nil", Uuid::nil().to_string()))
385            .await
386            .map_err(DbError::from)?
387            .check()
388            .map_err(DbError::from)?;
389        let rows: Vec<bool> = result.take(0).map_err(DbError::from)?;
390        Ok(rows.first().copied())
391    }

393    /// The last part of [`Self::purge_tombstoned`]: remove the tenant row, only
394    /// while it is still tombstoned (#655). Call it only after
395    /// [`Self::purge_tombstoned_rows`] has succeeded for the tenant; a tenant
396    /// row removed before its rows leaves them for the orphan scan. For an
397    /// orphaned id the row is its purge's claim marker, removed after
398    /// [`Self::purge_orphan_rows`] (R101W1-09 (a)).
399    ///
400    /// This is the purge's **point of no return**: once the row is gone, no
401    /// later sweep finds the tenant. So the same transaction writes the
402    /// `tenant_purge_pending` marker (#689, schema v108), keyed by the id, with
403    /// what `tenants.purged` says — `report`'s export time and counts, and
404    /// `orphan` when the report says so or the removed row was an orphan's
405    /// claim marker ([`ORPHAN_MARKER_CONDITION`]) — and `claimed_at`, the
406    /// caller's claim on it. The caller records the purge and then clears the
407    /// marker ([`Self::settle_purge_record`]); a caller cut in between leaves
408    /// it for [`Self::claim_pending_purge_records`]. `Ok` means this call
409    /// removed the row and wrote the marker.
410    ///
411    /// # Errors
412    ///
413    /// `NotFound` when the row is not there or not tombstoned — an overlapping
414    /// purge removed it first (#610); a datastore failure otherwise, in which
415    /// case neither the removal nor the marker happened.
416    pub async fn remove_purged_tenant(
417        &self,
418        id: Uuid,
419        report: &super::tenant_purge::TenantPurgeReport,
420        claimed_at: DateTime<Utc>,
421    ) -> AxiamResult<()> {
422        // Only while still tombstoned: the guard is in the statement too, so
423        // nothing here can remove a live tenant's row.
424        let statement = format!(
425            "BEGIN TRANSACTION; \
426             LET $gone = (DELETE type::record('tenant', $id) WHERE deleted_at != NONE \
427                 RETURN BEFORE); \
428             IF array::len($gone) > 0 {{ \
429                 CREATE type::record('tenant_purge_pending', $id) SET \
430                     orphan = $orphan OR array::len((SELECT VALUE id FROM $gone \
431                         WHERE {ORPHAN_MARKER_CONDITION})) > 0, \
432                     audit_exported_through = $exported_through, \
433                     audit_rows_deleted = $rows_deleted, \
434                     audit_rows_kept = $rows_kept, \
435                     purged_at = time::now(), \
436                     claimed_at = $claimed_at \
437                 RETURN NONE; \
438             }}; \
439             RETURN array::len($gone); \
440             COMMIT TRANSACTION;"
441        );
442        let mut result = self
443            .db
444            .current()
445            .query(statement)
446            .bind(("id", id.to_string()))
447            .bind(("nil", Uuid::nil().to_string()))
448            .bind(("orphan", report.orphan))
449            .bind(("exported_through", report.audit_exported_through))
450            .bind(("rows_deleted", report.audit_rows_deleted))
451            .bind(("rows_kept", report.audit_rows_kept))
452            .bind(("claimed_at", claimed_at))
453            .await
454            .map_err(DbError::from)?;
455        let errors = result.take_errors();
456        if !errors.is_empty() {
457            let combined = errors
458                .into_values()
459                .map(|e| e.to_string())
460                .collect::<Vec<_>>()
461                .join("; ");
462            return Err(classify_write_error(combined, "tenant_purge_pending").into());
463        }
464        let removed: Option<i64> = result.take(3).map_err(DbError::from)?;
465        if removed.unwrap_or(0) == 0 {
466            return Err(DbError::NotFound {
467                entity: "tombstoned tenant".into(),
468                id: id.to_string(),
469            }
470            .into());
471        }
472        Ok(())
473    }

475    /// Clear `id`'s `tenant_purge_pending` marker once its `tenants.purged`
476    /// record is written (#689): `true` when this call removed it. Conditional
477    /// on `claimed_at`, the claim the caller holds on it, so a marker another
478    /// replica has since taken over is left to that replica.
479    ///
480    /// # Errors
481    ///
482    /// A datastore failure; the marker stays, and the next sweep finds the
483    /// record already written and clears it.
484    pub async fn settle_purge_record(
485        &self,
486        id: Uuid,
487        claimed_at: DateTime<Utc>,
488    ) -> AxiamResult<bool> {
489        retry_hot_row(|| async {
490            let mut result = self
491                .db
492                .current()
493                .query(
494                    "SELECT count() AS total FROM (DELETE type::record('tenant_purge_pending', $id) \
495                     WHERE claimed_at = $claimed_at RETURN BEFORE) GROUP ALL",
496                )
497                .bind(("id", id.to_string()))
498                .bind(("claimed_at", claimed_at))
499                .await
500                .map_err(DbError::from)?
501                .check()
502                .map_err(|e| classify_write_error(e, "tenant_purge_pending"))?;
503            let removed: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
504            Ok::<_, DbError>(removed.first().is_some_and(|r| r.total > 0))
505        })
506        .await
507        .map_err(Into::into)
508    }

542    /// Claim, at `now`, every `tenant_purge_pending` marker whose purge nobody
543    /// is recording (#689): unclaimed (released by a unit that was cut), or
544    /// claimed at or before `now - lease` (a unit, or a sweep, that died
545    /// holding it). Each claim is a write conditional on the claim this call
546    /// read, so of two sweeps that read one marker exactly one takes it; the
547    /// other's write matches nothing. What is returned is this call's to
548    /// record, under the claim `now`, then to clear
549    /// ([`Self::settle_purge_record`]) or give up
550    /// ([`Self::release_purge_record`]).
551    ///
552    /// # Errors
553    ///
554    /// A datastore failure.
555    pub async fn claim_pending_purge_records(
556        &self,
557        now: DateTime<Utc>,
558        lease: chrono::TimeDelta,
559    ) -> AxiamResult<Vec<PendingPurgeRecord>> {
560        #[derive(Debug, SurrealValue)]
561        struct MarkerRow {
562            record_id: String,
563            claimed_at: Option<DateTime<Utc>>,
564        }
565        #[derive(Debug, SurrealValue)]
566        struct ClaimedRow {
567            orphan: bool,
568            audit_exported_through: Option<DateTime<Utc>>,
569            audit_rows_deleted: i64,
570            audit_rows_kept: i64,
571            purged_at: DateTime<Utc>,
572        }
573        let mut result = self
574            .db
575            .current()
576            .query(
577                "SELECT meta::id(id) AS record_id, claimed_at FROM tenant_purge_pending \
578                 WHERE claimed_at = NONE OR claimed_at <= $cutoff",
579            )
580            .bind(("cutoff", now - lease))
581            .await
582            .map_err(DbError::from)?
583            .check()
584            .map_err(DbError::from)?;
585        let markers: Vec<MarkerRow> = result.take(0).map_err(DbError::from)?;
586        let mut claimed = Vec::new();
587        for marker in markers {
588            let Ok(tenant_id) = Uuid::parse_str(&marker.record_id) else {
589                continue;
590            };
591            // Conditional on the claim read: NONE, or the stale stamp.
592            let condition = if marker.claimed_at.is_some() {
593                "claimed_at = $read"
594            } else {
595                "claimed_at = NONE"
596            };
597            let statement = format!(
598                "UPDATE type::record('tenant_purge_pending', $id) SET claimed_at = $now \
599                 WHERE {condition} \
600                 RETURN orphan, audit_exported_through, audit_rows_deleted, audit_rows_kept, \
601                     purged_at"
602            );
603            let rows = retry_hot_row(|| async {
604                let mut result = self
605                    .db
606                    .current()
607                    .query(statement.as_str())
608                    .bind(("id", marker.record_id.clone()))
609                    .bind(("now", now))
610                    .bind(("read", marker.claimed_at))
611                    .await
612                    .map_err(DbError::from)?
613                    .check()
614                    .map_err(|e| classify_write_error(e, "tenant_purge_pending"))?;
615                let rows: Vec<ClaimedRow> = result.take(0).map_err(DbError::from)?;
616                Ok::<_, DbError>(rows)
617            })
618            .await?;
619            if let Some(row) = rows.into_iter().next() {
620                claimed.push(PendingPurgeRecord {
621                    tenant_id,
622                    report: super::tenant_purge::TenantPurgeReport {
623                        audit_exported_through: row.audit_exported_through,
624                        audit_rows_deleted: u64::try_from(row.audit_rows_deleted).unwrap_or(0),
625                        audit_rows_kept: u64::try_from(row.audit_rows_kept).unwrap_or(0),
626                        orphan: row.orphan,
627                    },
628                    purged_at: row.purged_at,
629                    claimed_at: now,
630                });
631            }
632        }
633        Ok(claimed)
634    }

685    /// Claim an orphaned tenant id's purge for this replica (R101W1-09 (a)):
686    /// `true` when this call created the id's claim marker, `false` when it
687    /// did not — the id already has a tenant row (another replica's marker, or
688    /// a tenant of its own; [`Self::orphan_claim_refusal`] tells which), or it
689    /// owns no row left to purge.
690    ///
691    /// An orphan has no tenant row for [`Self::claim_purge`] to stamp, so the
692    /// claim **is** a row: a tombstoned tenant row under the orphan's own id,
693    /// with the nil organization id, the slug `orphan-purge-<id>` and
694    /// `purge_claimed_at = now`. A record id is unique, so of replicas whose
695    /// orphan scans coincide exactly one `CREATE` succeeds (the
696    /// `bootstrap_lock` and `organization_scope` pattern); the others fail on
697    /// the existing id, or lose the optimistic write, retry, and then fail on
698    /// it. While the marker exists the id is no orphan to any scan (the scan
699    /// reads the tenant table), it is out of every tenant read (each filters
700    /// `deleted_at`), and the tombstoned-tenant sweep leaves it alone for
701    /// [`Self::claim_purge`]'s lease — then takes it over, as an orphan, if the
702    /// claimant died or released it. No `has_tenant` edge is written and the
703    /// organization id is the nil id, so no organization's listing reaches it.
704    ///
705    /// The `CREATE` is conditional, in the same statement, on the id still
706    /// owning a row the orphan scan counts (#714): a scan that predates
707    /// another replica's finished purge writes no marker at all. A marker
708    /// therefore always stands for a purge nobody has recorded yet, so whoever
709    /// removes it — the claimant, or the tombstoned-tenant sweep taking it over
710    /// — records the purge, and it is recorded once. There is no window in
711    /// which the marker exists unchecked: the cleanup task's unit arms its
712    /// claim guard before this call, keyed by `now`, so a marker this call
713    /// writes is guarded from the instant it exists.
714    ///
715    /// # Errors
716    ///
717    /// A datastore failure; the marker may then exist or not, and releasing
718    /// the claim ([`Self::release_purge_claim`] with `now`) is right either way.
719    pub async fn claim_orphan_purge(&self, id: Uuid, now: DateTime<Utc>) -> AxiamResult<bool> {
720        let statement = format!(
721            "IF {} {{ \
722               CREATE type::record('tenant', $id) SET \
723                 organization_id = $nil, \
724                 name = 'orphaned tenant id', \
725                 slug = $slug, \
726                 metadata = {{}}, \
727                 deleted_at = $now, \
728                 purge_claimed_at = $now \
729               RETURN VALUE meta::id(id) \
730             }} ELSE {{ [] }}",
731            super::tenant_purge::purgeable_rows_condition()
732        );
733        let created = retry_hot_row(|| async {
734            let mut result = self
735                .db
736                .current()
737                .query(statement.as_str())
738                .bind(("id", id.to_string()))
739                .bind(("nil", Uuid::nil().to_string()))
740                .bind(("slug", format!("orphan-purge-{id}")))
741                .bind(("now", now))
742                .await
743                .map_err(DbError::from)?
744                .check()
745                .map_err(|e| classify_write_error(e, "tenant"))?;
746            let created: Vec<String> = result.take(0).map_err(DbError::from)?;
747            Ok::<_, DbError>(!created.is_empty())
748        })
749        .await;
750        match created {
751            Ok(created) => Ok(created),
752            Err(DbError::AlreadyExists { .. }) => Ok(false),
753            Err(e) => Err(e.into()),
754        }
755    }

757    /// Why [`Self::claim_orphan_purge`] answered `false`: `Conflict` when the
758    /// id has a tenant row of its own (live, or tombstoned by a deletion), so
759    /// it is no orphan; `Ok` when the row is another replica's marker, or when
760    /// there is no row and nothing was left to purge.
761    ///
762    /// # Errors
763    ///
764    /// `Conflict` as above; a datastore failure.
765    pub async fn orphan_claim_refusal(&self, id: Uuid) -> AxiamResult<()> {
766        let mut result = self
767            .db
768            .current()
769            .query(format!(
770                "SELECT VALUE ({ORPHAN_MARKER_CONDITION}) \
771                 FROM type::record('tenant', $id)"
772            ))
773            .bind(("id", id.to_string()))
774            .bind(("nil", Uuid::nil().to_string()))
775            .await
776            .map_err(DbError::from)?
777            .check()
778            .map_err(DbError::from)?;
779        let marker: Vec<bool> = result.take(0).map_err(DbError::from)?;
780        if marker.first() == Some(&false) {
781            return Err(axiam_core::error::AxiamError::Conflict {
782                reason: "a tenant row exists for this id; it is not an orphan".into(),
783            });
784        }
785        Ok(())
786    }

788    /// Remove an orphaned tenant id's rows, as [`Self::purge_tombstoned`] does
789    /// for a tombstoned tenant, except its audit trail, which no export receipt
790    /// covers and which the audit retention sweep governs (the report counts
791    /// it as kept).
792    ///
793    /// `Ok(Some(report))` means **this call** purged the id and the caller
794    /// records it; `Ok(None)` that there is nothing for this call to record
795    /// (R101W1-09 (a)). Every replica scans for orphans, so two scans may name
796    /// one id. The purge is claimed first ([`Self::claim_orphan_purge`]): while
797    /// one replica holds the id, another's call answers `None` without
798    /// touching a row, and a scan that predates a finished purge finds nothing
799    /// left, writes no marker and answers `None`. The purge's last step deletes
800    /// the marker, and only the call whose `DELETE` removed it answers `Some` —
801    /// should a purge outlive the claim's lease and the tombstoned-tenant sweep
802    /// take the marker over, only one of the two records the purge.
803    ///
804    /// The composition of [`Self::claim_orphan_purge`],
805    /// [`Self::purge_orphan_rows`] and [`Self::remove_purged_tenant`] (the
806    /// point of no return), which the cleanup task runs step by step, under
807    /// its claim guard, so that a stop gives the purge up before that point
808    /// (#655, #714).
809    ///
810    /// # Errors
811    ///
812    /// `Conflict` when a tenant row of its own exists for this id (live, or
813    /// tombstoned by a deletion); a datastore failure otherwise, after which
814    /// the claim is released, so that the tombstoned-tenant sweep resumes the
815    /// purge at once.
816    pub async fn purge_orphan(
817        &self,
818        id: Uuid,
819    ) -> AxiamResult<Option<super::tenant_purge::TenantPurgeReport>> {
820        let claimed_at = Utc::now();
821        match self.claim_orphan_purge(id, claimed_at).await {
822            Ok(true) => {}
823            Ok(false) => {
824                self.orphan_claim_refusal(id).await?;
825                return Ok(None);
826            }
827            Err(e) => {
828                self.release_orphan_claim_or_warn(id, claimed_at).await;
829                return Err(e);
830            }
831        }
832        let report = match self.purge_orphan_rows(id).await {
833            Ok(report) => report,
834            Err(e) => {
835                self.release_orphan_claim_or_warn(id, claimed_at).await;
836                return Err(e);
837            }
838        };
839        match self.remove_purged_tenant(id, &report, claimed_at).await {
840            Ok(()) => {
841                // The caller records the purge (#689: no marker is left for it).
842                self.settle_purge_record(id, claimed_at).await?;
843                Ok(Some(report))
844            }
845            Err(axiam_core::error::AxiamError::NotFound { .. }) => Ok(None),
846            Err(e) => {
847                self.release_orphan_claim_or_warn(id, claimed_at).await;
848                Err(e)
849            }
850        }
851    }
852
853    /// The second part of [`Self::purge_orphan`]: the orphaned id's rows, with
854    /// [`super::tenant_purge::PurgeScope::Orphan`], under the claim
855    /// [`Self::claim_orphan_purge`] wrote (#655).
856    ///
857    /// Every step is an idempotent `DELETE` and the marker stays, so nothing
858    /// here is a point of no return: a purge stopped in here, or failed, is
859    /// resumed by whoever purges the marker next once its claim is released.
860    /// The point of no return is [`Self::remove_purged_tenant`], and only the
861    /// call that removed the marker records the purge.
862    ///
863    /// # Errors
864    ///
865    /// A datastore failure; the marker stays, and the caller releases its
866    /// claim.
867    pub async fn purge_orphan_rows(
868        &self,
869        id: Uuid,
870    ) -> AxiamResult<super::tenant_purge::TenantPurgeReport> {
871        Ok(
872            super::tenant_purge::purge_rows(&self.db, id, super::tenant_purge::PurgeScope::Orphan)
873                .await?,
874        )
875    }
876
877    /// [`Self::release_purge_claim`] after a failed orphan purge: a release
878    /// that fails too is logged, and the marker lapses with its lease.
879    async fn release_orphan_claim_or_warn(&self, id: Uuid, claimed_at: DateTime<Utc>) {
880        if let Err(release) = self.release_purge_claim(id, claimed_at).await {
881            tracing::warn!(
882                error = %release,
883                tenant_id = %id,
884                "could not release the orphan purge's claim; it lapses with its lease"
885            );
886        }
887    }

891    async fn create(&self, input: CreateTenant) -> AxiamResult<Tenant> {
892        let id = new_id();
893        let id_str = id.to_string();
894        let org_id_str = input.organization_id.to_string();
895        let metadata = input
896            .metadata
897            .unwrap_or(serde_json::Value::Object(Default::default()));
898
899        // Create tenant record and relate to organization in one query.
900        // RELATE requires literal record-id syntax, so we embed UUIDs
901        // directly in the RELATE portion (they are safe — UUID format).
902        //
903        // An organization tenant also claims `organization_scope:<org_id>`,
904        // whose record id IS the constraint that there is only one of them. A
905        // second attempt fails on that CREATE rather than quietly producing a
906        // second place organization-level principals could live — see
907        // `SCHEMA_V50` for why this is a marker row and not a partial unique
908        // index.
909        let claim_scope = if input.kind.is_organization() {
910            " CREATE type::record('organization_scope', $org_id) \
911              SET tenant_id = $id;"
912        } else {
913            ""
914        };
915        let query = format!(
916            "CREATE type::record('tenant', $id) SET \
917             organization_id = $org_id, \
918             name = $name, slug = $slug, \
919             status = 'Active', \
920             kind = $kind, \
921             metadata = $metadata; \
922             RELATE organization:`{org_id_str}` \
923             -> has_tenant -> tenant:`{id_str}`;{claim_scope}"
924        );
925
926        let result = self
927            .db
928            .current()
929            .query(query)
930            .bind(("id", id_str.clone()))
931            .bind(("org_id", org_id_str))
932            .bind(("name", input.name))
933            .bind(("slug", input.slug))
934            .bind(("kind", kind_to_str(input.kind)))
935            .bind(("metadata", metadata))
936            .await
937            .map_err(DbError::from)?;
938
939        // A duplicate (organization_id, slug) trips the `idx_tenant_org_slug`
940        // unique index. Left unclassified it becomes `DbError::Migration`, which
941        // the API layer maps to a bare 500 `{"error":"internal_error"}` — so
942        // re-running any provisioning script answers "something broke" for a
943        // system that is in fact perfectly fine, and the documented 409 branch
944        // of `scripts/e2e-bootstrap.sh` was dead code.
945        //
946        // Per D-09 the marker matching lives in `helpers::classify_write_error`
947        // and nowhere else, so that a marker added there reaches every call
948        // site at once.
949        let mut result = result
950            .check()
951            .map_err(|e| classify_write_error(e, "tenant"))?;
952
953        // Statement 0 is the CREATE, statement 1 is the RELATE.
954        let rows: Vec<TenantRow> = result.take(0).map_err(DbError::from)?;
955        let row = take_first_or_not_found(rows, "tenant", &id_str)?;
956        bump_tenant_generation();
957
958        Ok(row.into_tenant(id)?)
959    }

1084    async fn delete(&self, id: Uuid) -> AxiamResult<()> {
1085        // #523 (P23W2-04, D-4): deleting a tenant TOMBSTONES it. The row keeps
1086        // its data and gains `deleted_at`, which takes it out of every read in
1087        // this file (and the settings repository's tenant lookup): sign-in,
1088        // token issuance, refresh and every handler that resolves the tenant
1089        // answer as for a tenant that does not exist, from this commit on. The
1090        // cleanup job's `tenant_purge` sweep then removes every tenant-scoped
1091        // row in the order user erasure uses (`tenant_purge`), and the tenant
1092        // row last. The slug stays claimed by the tombstone until then, so a
1093        // new tenant cannot take it while the old one's rows are still there.
1094        //
1095        // In the same transaction, and so in the request:
1096        //
1097        // * Its sessions are deleted and its refresh tokens revoked — the
1098        //   erasure's first step, so the tenant's last session cannot refresh
1099        //   and its access tokens fail the per-request session check. The
1100        //   handler revoked them once already, through the repositories that
1101        //   publish to the revocation feed and drop the validity cache; this
1102        //   pass catches a sign-in that raced it, atomically with the stamp.
1103        //
1104        // * The rows that make AXIAM act for the tenant, or hold its credentials
1105        //   for systems outside AXIAM, go now rather than at the purge, so no
1106        //   background job — the directory sync, the SCIM delivery and
1107        //   reconciliation, the SSF transmitter — works for a tenant that no
1108        //   longer exists: its directory configuration and sync state (T23.3.1,
1109        //   T23.3.5: an encrypted bind credential, account ids), its SAML
1110        //   service providers, IdP signing credential, pending `AuthnRequest`s
1111        //   and single-logout state (T23.2.1–T23.2.4: a sealed private key, the
1112        //   `NameID` each SP was given), its SSF streams, step-up records and
1113        //   buffered events (T23.5.2: a sealed receiver credential, subjects'
1114        //   addresses), its CIBA requests and device grants (#641: an approved
1115        //   grant is a bearer for an access token, and a user code is the
1116        //   tenant's own), and its SCIM targets with their links and delivery
1117        //   state.
1118        //
1119        // * R1W1-01: its certificates and signing CAs are revoked. The handler
1120        //   revoked them once already, through the services that forward a
1121        //   `vault_pki` leaf to Vault and release a CA's key; this pass catches
1122        //   one issued while it ran, atomically with the stamp. A deleted
1123        //   tenant's certificates are disowned, so they go on their issuers'
1124        //   revocation lists — and the purge keeps each revoked row until
1125        //   eight days after it expires, so they stay there (`tenant_purge`).
1126        //
1127        // F4 P23W2-02: and a transaction that rolled back is an error. The
1128        // driver reports a failed statement inside the response, not from
1129        // `.await`, so without `check` a cancelled delete answered `Ok` — the
1130        // handler then answered `204` and wrote a "tenant deleted" record for
1131        // a tenant that still existed, with its encrypted bind secret.
1132        self.db
1133            .current()
1134            .query(
1135                "BEGIN TRANSACTION; \
1136                 DELETE session WHERE tenant_id = $id; \
1137                 UPDATE oauth2_refresh_token SET revoked = true \
1138                     WHERE tenant_id = $id AND revoked = false; \

1227    fn count_for_shared_issuer(&self) -> axiam_core::models::ssf::SsfFuture<'_, AxiamResult<u64>> {
1228        Box::pin(async move {
1229            // An absent `kind` is a standard tenant (rows older than
1230            // organization scope), so the test is "not the organization's own".
1231            let mut result = self
1232                .db
1233                .current()
1234                .query(
1235                    "SELECT count() AS total FROM tenant \
1236                         WHERE (kind = NONE OR kind != 'organization') \
1237                           AND deleted_at = NONE GROUP ALL; \
1238                     SELECT count() AS total FROM organization GROUP ALL",
1239                )
1240                .await
1241                .map_err(DbError::from)?;
1242            let tenants: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
1243            let organizations: Vec<CountRow> = result.take(1).map_err(DbError::from)?;
1244            let tenants = tenants.first().map_or(0, |r| r.total);
1245            let organizations = organizations.first().map_or(0, |r| r.total);
1246            Ok(tenants.max(organizations))
1247        })
1248    }
