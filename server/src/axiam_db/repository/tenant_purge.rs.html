1//! The tenant purge (#523, P23W2-04, D-4).
2//!
3//! Deleting a tenant tombstones it ([`super::SurrealTenantRepository`]'s
4//! `delete`): the row stays, with `deleted_at` set, so the tenant leaves every
5//! read path at once while its data is still there. The cleanup job's
6//! `tenant_purge` sweep then removes the data — every row of every
7//! tenant-scoped table — and the tenant row last.
8//!
9//! # The order
10//!
11//! [`TENANT_PURGE_ORDER`] follows the order the GDPR user erasure uses
12//! (`axiam_server::cleanup`'s purge pipeline, steps (a) to (f)), widened from
13//! one account to every account and every configuration of the tenant:
14//!
15//! 1. [`PurgeStage::Grants`] — sessions, refresh tokens and every other grant or
16//!    in-flight credential, so nothing issued to the tenant can be redeemed
17//!    while the rest goes (erasure step (a)). The request that tombstoned the
18//!    tenant already revoked its sessions and refresh tokens; this removes the
19//!    rows.
20//! 2. [`PurgeStage::IdentityLinks`] — federation links (step (b)).
21//! 3. [`PurgeStage::Credentials`] — passkeys, OPAQUE records, password history
22//!    (step (b2)).
23//! 4. [`PurgeStage::Authorization`] — groups, roles, permissions, scopes,
24//!    resources and service accounts; their graph edges go with the records
25//!    (step (b3)).
26//! 5. [`PurgeStage::AuditTrail`] — the tenant's own audit entries and their
27//!    signatures (step (d)). The deletion was refused until the trail had been
28//!    exported (T-118), so the purge destroys nothing the operator does not hold
29//!    a copy of; the `tenant.deleted` record is written to the system log (the
30//!    nil tenant) and is never touched here.
31//! 6. [`PurgeStage::Records`] — consents, deletion requests, export jobs and
32//!    erasure proofs (steps (f) and (g)).
33//! 7. [`PurgeStage::Accounts`] — the user rows themselves (step (e); an erasure
34//!    anonymizes one account in place, a tenant purge removes them all).
35//! 8. [`PurgeStage::Configuration`] — everything the tenant configured,
36//!    including the encrypted credentials it held for systems outside AXIAM.
37//!
38//! Every step is one idempotent `DELETE … WHERE <tenant>`, so a purge that
39//! fails part-way stops there, leaves the tenant tombstoned, and is resumed by
40//! the next sweep; the tenant row goes only after every step succeeded.
41//!
42//! # Revocation evidence outlives the tenant (R1W1-01)
43//!
44//! Two tables are not emptied: `certificate` and `ca_certificate` keep every
45//! **revoked, unexpired** row ([`Retain::RevokedUntilExpiry`]). An issuing CA's
46//! revocation list (`axiam_pki::crl`, T-102) is read from those rows, across
47//! tenants, by issuer; deleting them took a revoked leaf off its organization
48//! CA's list and made it valid again to every relying party outside AXIAM. The
49//! deletion revoked every certificate and signing CA of the tenant already
50//! (the handler, and again the tombstone transaction), and the step revokes
51//! anything still unrevoked first, so after the purge each of the tenant's
52//! unexpired certificates is a revoked row its issuer's list names — and that
53//! every authentication path refuses, since each reads the row's status.
54//!
55//! What is kept is what the list needs and no more: the step clears a
56//! certificate's free-form `metadata` and a CA's sealed private key. The rows
57//! name a tenant id with no tenant row, which is what the orphan scan looks
58//! for; it ignores a kept row until it expires, and then the orphan purge
59//! removes it — the cleanup job deletes the evidence once no relying party can
60//! need it.
61//!
62//! # Completeness is pinned
63//!
64//! `schema::tests::every_tenant_scoped_table_is_purged` scans every migration
65//! for a table with a `tenant_id` field (or the `scope`/`scope_id` pair) and
66//! fails when one is missing here — a table added later without a purge step
67//! fails the build's tests rather than surviving its tenant.

78/// How a table names the tenant a row belongs to.
79#[derive(Debug, Clone, Copy, PartialEq, Eq)]
80pub enum TenantKey {
81    /// `tenant_id = <tenant>` — every table with a `tenant_id` field.
82    TenantId,
83    /// `scope = 'tenant' AND scope_id = <tenant>` — the tables whose rows
84    /// belong to an organization or to a tenant (`scope = 'org'` rows are never
85    /// touched).
86    Scope,
87}
88
89/// Which part of the user-erasure order a step belongs to (see the module
90/// documentation).
91#[derive(Debug, Clone, Copy, PartialEq, Eq)]
92pub enum PurgeStage {
93    /// Sessions, refresh tokens and every other grant or pending credential.
94    Grants,
95    /// Federation links.
96    IdentityLinks,
97    /// Stored credential material.
98    Credentials,
99    /// The authorization graph's records.
100    Authorization,
101    /// The tenant's own audit entries and their signatures.
102    AuditTrail,
103    /// GDPR records: consents, deletion requests, export jobs, erasure proofs.
104    Records,
105    /// The user rows.
106    Accounts,
107    /// The tenant's configuration and the credentials it holds for others.
108    Configuration,
109}
110
111/// Which of a table's rows the purge keeps.
112#[derive(Debug, Clone, Copy, PartialEq, Eq)]
113pub enum Retain {
114    /// None: every row of the tenant goes.
115    Nothing,
116    /// A revoked certificate whose `not_after` is still ahead — the evidence
117    /// its issuer's revocation list is read from (R1W1-01; see the module
118    /// documentation). Every other row goes; an unrevoked, unexpired row is
119    /// revoked first, and a kept row has `strip` applied (an `UPDATE … SET`
120    /// clause clearing what the list does not need).
121    RevokedUntilExpiry {
122        /// The `SET` clause applied to a kept row.
123        strip: &'static str,
124    },
125}
126
127/// The condition a row the purge keeps satisfies.
128const KEPT_REVOCATION: &str = "status = 'Revoked' AND not_after > time::now()";
129
130/// One table of the purge.
131#[derive(Debug, Clone, Copy)]
132pub struct PurgeStep {
133    /// The table.
134    pub table: &'static str,
135    /// How its rows name their tenant.
136    pub key: TenantKey,
137    /// Where in the erasure order it is purged.
138    pub stage: PurgeStage,
139    /// Which rows outlive the purge.
140    pub retain: Retain,
141}
142
143impl PurgeStep {
144    const fn by_tenant(table: &'static str, stage: PurgeStage) -> Self {
145        Self {
146            table,
147            key: TenantKey::TenantId,
148            stage,
149            retain: Retain::Nothing,
150        }
151    }
152
153    const fn by_scope(table: &'static str, stage: PurgeStage) -> Self {
154        Self {
155            table,
156            key: TenantKey::Scope,
157            stage,
158            retain: Retain::Nothing,
159        }
160    }
161
162    /// A certificate table: keyed by `tenant_id`, keeping revocation evidence
163    /// until it expires.
164    const fn keeping_revocations(
165        table: &'static str,
166        strip: &'static str,
167        stage: PurgeStage,
168    ) -> Self {
169        Self {
170            table,
171            key: TenantKey::TenantId,
172            stage,
173            retain: Retain::RevokedUntilExpiry { strip },
174        }
175    }
176
177    /// The statement that removes the tenant's rows (`$id` is the tenant).
178    ///
179    /// The table name and a retaining step's `strip` clause are compile-time
180    /// constants of [`TENANT_PURGE_ORDER`], never input, so formatting them
181    /// into the statement is safe.
182    fn delete_statement(&self) -> String {
183        let table = self.table;
184        match (self.key, self.retain) {
185            (TenantKey::TenantId, Retain::Nothing) => {
186                format!("DELETE {table} WHERE tenant_id = $id")
187            }
188            (TenantKey::TenantId, Retain::RevokedUntilExpiry { strip }) => format!(
189                "UPDATE {table} SET status = 'Revoked', revoked_at = revoked_at ?? time::now() \
190                     WHERE tenant_id = $id AND status != 'Revoked' AND not_after > time::now(); \
191                 UPDATE {table} SET {strip} WHERE tenant_id = $id AND {KEPT_REVOCATION}; \
192                 DELETE {table} WHERE tenant_id = $id AND NOT ({KEPT_REVOCATION});"
193            ),
194            (TenantKey::Scope, _) => {
195                format!("DELETE {table} WHERE scope = 'tenant' AND scope_id = $id")
196            }
197        }
198    }
199
200    /// The condition, beyond the tenant, a row must meet to be purged — what
201    /// the orphan scan counts.
202    fn purgeable(&self) -> String {
203        match self.retain {
204            Retain::Nothing => "true".into(),
205            Retain::RevokedUntilExpiry { .. } => format!("NOT ({KEPT_REVOCATION})"),
206        }
207    }
208}

298/// Whose rows a purge removes, and whether it may take the audit trail.
299#[derive(Debug, Clone, Copy, PartialEq, Eq)]
300pub(crate) enum PurgeScope {
301    /// A tombstoned tenant: everything, its audit trail included (it was
302    /// exported before the deletion was allowed, T-118).
303    Tombstoned,
304    /// A tenant id that has rows but no tenant row — left by a deletion made
305    /// before the tombstone existed. Everything **but** the audit trail: no
306    /// export receipt can be checked for it, and an audit entry naming a tenant
307    /// id that never existed (a refused request) is a record the retention
308    /// sweep, not this one, decides about.
309    Orphan,
310}
311
312/// Remove `tenant_id`'s rows from every table of [`TENANT_PURGE_ORDER`], in
313/// order, stopping at the first failure (every step is idempotent, so the next
314/// sweep resumes).
315pub(crate) async fn purge_rows<C: Connection>(
316    db: &DbHandle<C>,
317    tenant_id: Uuid,
318    scope: PurgeScope,
319) -> Result<(), DbError> {
320    let id = tenant_id.to_string();
321    for step in TENANT_PURGE_ORDER {
322        if scope == PurgeScope::Orphan && step.stage == AuditTrail {
323            continue;
324        }
325        db.current()
326            .query(step.delete_statement())
327            .bind(("id", id.clone()))
328            .await?
329            .check()
330            .map_err(|e| DbError::Migration(format!("purging {}: {e}", step.table)))?;
331    }
332    Ok(())
333}

335/// Tenant ids that own rows in a tenant-scoped table (the audit trail aside,
336/// see [`PurgeScope::Orphan`]) but have no tenant row — tombstoned or live.
337///
338/// A row a purge keeps ([`Retain::RevokedUntilExpiry`]) is not counted until
339/// it expires; then it is, and the orphan purge removes it.
340///
341/// The tables are read **before** the tenant set: a tenant row is written
342/// before any row naming it, so a tenant created while this runs is either
343/// absent from the tables read or present in the set read after them, and is
344/// never mistaken for an orphan. The nil id is the system log's and never an
345/// orphan.
346pub(crate) async fn orphaned_tenant_ids<C: Connection>(
347    db: &DbHandle<C>,
348) -> Result<Vec<Uuid>, DbError> {
349    #[derive(Debug, SurrealValue)]
350    struct ByTenant {
351        #[surreal(default)]
352        tenant_id: Option<String>,
353    }
354    #[derive(Debug, SurrealValue)]
355    struct ByScope {
356        #[surreal(default)]
357        scope_id: Option<String>,
358    }
359    #[derive(Debug, SurrealValue)]
360    struct TenantRow {
361        record_id: String,
362    }
363
364    let mut named: BTreeSet<String> = BTreeSet::new();
365    for step in TENANT_PURGE_ORDER {
366        if step.stage == AuditTrail {
367            continue;
368        }
369        // `GROUP BY` so the datastore returns one row per tenant, not one
370        // per row of the table.
371        match step.key {
372            TenantKey::TenantId => {
373                let mut result = db
374                    .current()
375                    .query(format!(
376                        "SELECT tenant_id FROM {} WHERE {} GROUP BY tenant_id",
377                        step.table,
378                        step.purgeable()
379                    ))
380                    .await?
381                    .check()?;
382                let rows: Vec<ByTenant> = result.take(0)?;
383                named.extend(rows.into_iter().filter_map(|r| r.tenant_id));
384            }
385            TenantKey::Scope => {
386                let mut result = db
387                    .current()
388                    .query(format!(
389                        "SELECT scope_id FROM {} WHERE scope = 'tenant' GROUP BY scope_id",
390                        step.table
391                    ))
392                    .await?
393                    .check()?;
394                let rows: Vec<ByScope> = result.take(0)?;
395                named.extend(rows.into_iter().filter_map(|r| r.scope_id));
396            }
397        }
398    }
399
400    let mut result = db
401        .current()
402        .query("SELECT meta::id(id) AS record_id FROM tenant")
403        .await?
404        .check()?;
405    let tenants: Vec<TenantRow> = result.take(0)?;
406    let known: BTreeSet<String> = tenants.into_iter().map(|t| t.record_id).collect();
407
408    Ok(named
409        .difference(&known)
410        .filter_map(|id| Uuid::parse_str(id).ok())
411        .filter(|id| !id.is_nil())
412        .collect())
413}
