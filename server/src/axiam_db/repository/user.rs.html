267/// Decode a stored address, dropping one that has no members.
268///

374impl<C: Connection> SurrealUserRepository<C> {
375    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
376        let db = db.into();
377        Self {
378            db,
379            pepper: None,
380            provisioning_sink: None,

384    pub fn with_pepper(db: impl Into<DbHandle<C>>, pepper: String) -> Self {
385        let db = db.into();
386        Self {
387            db,
388            pepper: Some(pepper),
389            provisioning_sink: None,
390        }
391    }
392
393    /// Report committed changes to `sink` (G-6, D-57): every mutating method
394    /// that can change what a downstream SCIM service provider is told calls
395    /// [`ProvisioningSink::user_changed`] **after** the write succeeded. An
396    /// `update` calls it only when it touches a provisioned field (username,
397    /// email, status, metadata); login bookkeeping (`increment_failed_logins`,
398    /// `update_totp_step`, lock and failed-login stamps) never does. The sink
399    /// cannot fail the write (it returns nothing).
400    ///
401    /// Clones of the repository carry the sink with them, so one binding at
402    /// the composition root reaches every service that clones the repository.
403    #[must_use]
404    pub fn with_provisioning_sink(mut self, sink: Arc<dyn ProvisioningSink>) -> Self {
405        self.provisioning_sink = Some(sink);
406        self
407    }

409    /// Tell the sink the user changed. A no-op when no sink is attached.
410    async fn notify_user_changed(&self, tenant_id: Uuid, user_id: Uuid) {
411        if let Some(sink) = &self.provisioning_sink {
412            sink.user_changed(tenant_id, user_id).await;
413        }
414    }

417impl<C: Connection> SurrealUserRepository<C> {
418    /// The shared body of the two collision probes: any account of the tenant
419    /// whose username or email equals one of `names` ignoring case, leaving
420    /// `exclude` out when given.
421    async fn identity_collision(
422        &self,
423        tenant_id: Uuid,
424        names: &[String],
425        exclude: Option<Uuid>,
426    ) -> AxiamResult<Option<IdentityCollision>> {
427        let folded: Vec<String> = names
428            .iter()
429            .map(|name| name.trim().to_lowercase())
430            .filter(|name| !name.is_empty())
431            .collect();
432        if folded.is_empty() {
433            return Ok(None);
434        }
435        // Every status, tombstones included. The scan is bounded by the
436        // tenant (the first column of both unique indexes) and runs once per
437        // first-ever directory login or per changed directory attribute, never
438        // per sign-in.
439        let mut result = self
440            .db
441            .current()
442            .query(
443                "SELECT meta::id(id) AS record_id, \
444                        string::lowercase(username) IN $names AS by_username \
445                 FROM user \
446                 WHERE tenant_id = $tenant_id \
447                   AND ($exclude = NONE OR meta::id(id) != $exclude) \
448                   AND (string::lowercase(username) IN $names \
449                        OR string::lowercase(email) IN $names) \
450                 LIMIT 1",
451            )
452            .bind(("tenant_id", tenant_id.to_string()))
453            .bind(("names", folded))
454            .bind(("exclude", exclude.map(|id| id.to_string())))
455            .await
456            .map_err(DbError::from)?;
457        let rows: Vec<CollisionRow> = result.take(0).map_err(DbError::from)?;
458        let Some(row) = rows.into_iter().next() else {
459            return Ok(None);
460        };
461        Ok(Some(IdentityCollision {
462            user_id: parse_uuid(&row.record_id, "user")?,
463            attribute: if row.by_username {
464                CollisionAttribute::Username
465            } else {
466                CollisionAttribute::Email
467            },
468        }))
469    }
470}

472impl<C: Connection> SurrealUserRepository<C> {
473    /// An unusable password: the Argon2id hash of 32 random bytes nobody
474    /// holds, so the column keeps its shape (a valid PHC string that every
475    /// verifier parses) without being a hash of anything guessable. Never
476    /// empty: the empty string is the erasure tombstone, and a verifier handed
477    /// one errors instead of answering "no".
478    fn unusable_password_hash(&self) -> AxiamResult<String> {
479        use rand::Rng;
480        let mut bytes = [0u8; 32];
481        rand::rng().fill_bytes(&mut bytes);
482        let secret = zeroize::Zeroizing::new(hex::encode(bytes));
483        password::hash_password(&secret, self.pepper.as_deref())
484            .map_err(|e| classify_write_error(e, "user").into())
485    }
486}

617    async fn update(&self, tenant_id: Uuid, id: Uuid, input: UpdateUser) -> AxiamResult<User> {
618        // Retried on an optimistic-concurrency loss, for the same reason
619        // `increment_failed_logins` below is: this statement targets ONE row, so
620        // two concurrent writers to the same user race by construction and the
621        // loser is aborted with a write the datastore itself labels retryable.
622        //
623        // Unlike that method, this one is reached by every administrative and
624        // provisioning write — including SCIM `PATCH /scim/v2/Users/{id}`, whose
625        // benchmark cell is a deliberate flood of `replace active` against a
626        // single user and which returned HTTP 500 on 2.2% of its operations
627        // before this loop existed. An IdP driving Okta/Entra-shaped
628        // provisioning sees those as failed syncs and retries the whole record.
629        //
630        // `input` is cloned per attempt because SurrealDB's `bind` takes owned
631        // values: the binding block below moves each field out of `input`, so a
632        // second attempt needs its own copy. The clone is paid only on the
633        // attempt itself, which the uncontended path never reaches past the
634        // first.
635        // G-6 (D-57): only a change to a field a downstream directory is told
636        // about notifies the provisioning sink. A login's bookkeeping (lock
637        // stamps, TOTP step, failed-login counters) and a credential or MFA
638        // change do not.
639        let touches_provisioned_field = input.username.is_some()
640            || input.email.is_some()
641            || input.status.is_some()
642            || input.metadata.is_some();
643        let updated = crate::helpers::retry_on_write_conflict(|| async {
644            let input = input.clone();
645            let id_str = id.to_string();
646            let tenant_id_str = tenant_id.to_string();
647
648            let mut sets = Vec::new();
649            if input.username.is_some() {
650                sets.push("username = $username");
651            }
652            if input.email.is_some() {
653                sets.push("email = $email");
654            }
655            if input.password_hash.is_some() {
656                sets.push("password_hash = $password_hash");
657            }
658            if input.status.is_some() {
659                sets.push("status = $status");
660            }
661            if input.metadata.is_some() {
662                sets.push("metadata = $metadata");
663            }
664            if input.mfa_enabled.is_some() {
665                sets.push("mfa_enabled = $mfa_enabled");
666            }
667            if input.mfa_secret.is_some() {
668                sets.push("mfa_secret = $mfa_secret");
669            }
670            if input.totp_last_used_step.is_some() {
671                sets.push("totp_last_used_step = $totp_last_used_step");
672            }
673            if input.failed_login_attempts.is_some() {
674                sets.push("failed_login_attempts = $failed_login_attempts");
675            }
676            if input.last_failed_login_at.is_some() {
677                sets.push("last_failed_login_at = $last_failed_login_at");
678            }
679            if input.locked_until.is_some() {
680                sets.push("locked_until = $locked_until");
681            }
682            if input.email_verified_at.is_some() {
683                sets.push("email_verified_at = $email_verified_at");
684            }
685            // W7 / X7 G8. Same `Option<Option<T>>` convention as the fields above:
686            // `Some(Some(v))` sets, `Some(None)` clears, `None` leaves alone. The
687            // clear direction is not decoration — it is how a data subject
688            // withdrawing a telephone number gets it removed rather than hidden.
689            if input.phone_number.is_some() {
690                sets.push("phone_number = $phone_number");
691            }
692            if input.phone_number_verified_at.is_some() {
693                sets.push("phone_number_verified_at = $phone_number_verified_at");
694            }
695            if input.address.is_some() {
696                sets.push("address = $address");
697            }
698            sets.push("updated_at = time::now()");
699
700            let query = format!(
701                "UPDATE type::record('user', $id) SET {} \
702             WHERE tenant_id = $tenant_id",
703                sets.join(", ")
704            );
705
706            let db = self.db.current();
707            let mut builder = db
708                .query(&query)
709                .bind(("id", id_str.clone()))
710                .bind(("tenant_id", tenant_id_str));
711
712            if let Some(username) = input.username {
713                builder = builder.bind(("username", username));
714            }
715            if let Some(email) = input.email {
716                builder = builder.bind(("email", email));
717            }
718            if let Some(password_hash) = input.password_hash {
719                builder = builder.bind(("password_hash", password_hash));
720            }
721            if let Some(ref status) = input.status {
722                builder = builder.bind(("status", status_to_string(status).to_string()));
723            }
724            if let Some(metadata) = input.metadata {
725                builder = builder.bind(("metadata", metadata));
726            }
727            if let Some(mfa_enabled) = input.mfa_enabled {
728                builder = builder.bind(("mfa_enabled", mfa_enabled));
729            }
730            if let Some(mfa_secret) = input.mfa_secret {
731                // mfa_secret is Option<Option<String>>: Some(Some(v)) = set, Some(None) = clear
732                builder = builder.bind(("mfa_secret", mfa_secret));
733            }
734            if let Some(totp_last_used_step) = input.totp_last_used_step {
735                // totp_last_used_step is Option<Option<u64>>: Some(Some(v)) = set, Some(None) = clear
736                builder = builder.bind(("totp_last_used_step", totp_last_used_step));
737            }
738            if let Some(failed_login_attempts) = input.failed_login_attempts {
739                builder = builder.bind(("failed_login_attempts", failed_login_attempts));
740            }
741            if let Some(last_failed_login_at) = input.last_failed_login_at {
742                builder = builder.bind(("last_failed_login_at", last_failed_login_at));
743            }
744            if let Some(locked_until) = input.locked_until {
745                builder = builder.bind(("locked_until", locked_until));
746            }
747            if let Some(email_verified_at) = input.email_verified_at {
748                builder = builder.bind(("email_verified_at", email_verified_at));
749            }
750            if let Some(phone_number) = input.phone_number {
751                builder = builder.bind(("phone_number", phone_number));
752            }
753            if let Some(phone_number_verified_at) = input.phone_number_verified_at {
754                builder = builder.bind(("phone_number_verified_at", phone_number_verified_at));
755            }
756            if let Some(address) = input.address {
757                // An address with no members is stored as absent, so that "has an
758                // address" and "the address says something" cannot disagree.
759                let row = address
760                    .as_ref()
761                    .filter(|a| !a.is_empty())
762                    .map(AddressRow::from);
763                builder = builder.bind(("address", row));
764            }
765
766            let result = builder.await.map_err(DbError::from)?;
767            let mut result = result
768                .check()
769                .map_err(|e| classify_write_error(e.to_string(), "user"))?;
770
771            let rows: Vec<UserRow> = result.take(0).map_err(DbError::from)?;
772            let row = rows.into_iter().next().ok_or_else(|| DbError::NotFound {
773                entity: "user".into(),
774                id: id_str,
775            })?;
776
777            Ok::<_, axiam_core::error::AxiamError>(row.into_user(id)?)
778        })
779        .await?;
780        if touches_provisioned_field {
781            self.notify_user_changed(tenant_id, id).await;
782        }
783        Ok(updated)
784    }

786    async fn delete(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
787        // Anonymise the row in place, then mark it `Deleted`.
788        //
789        // This used to set `status = 'Inactive'` and stop there — the same value
790        // the edit dialog's Active/Inactive toggle writes — so a "deleted" user
791        // stayed in the list, kept their password hash and MFA secret, and was
792        // indistinguishable from a suspended one. From the administrator's side,
793        // Delete did nothing.
794        //
795        // # Why the row survives, and why it holds no personal data
796        //
797        // Audit entries are append-only and reference their actor by id, so
798        // hard-deleting the row would leave every entry the user ever produced
799        // pointing at nothing — an audit trail that cannot be resolved to a
800        // person is not an audit trail. What must NOT survive is the personal
801        // data, so `username` and `email` are overwritten rather than kept: a
802        // tombstone holding someone's address indefinitely is not erasure, it is
803        // retention with the UI hidden.
804        //
805        // The replacements derive from the row's own id, which is an internal
806        // identifier and not personal data. They are deterministic (so this is
807        // idempotent) and unique per user (so two deleted accounts cannot
808        // collide on the unique indexes).
809        //
810        // # Re-registration
811        //
812        // Freeing the real `username` and `email` from
813        // `idx_user_tenant_username` / `idx_user_tenant_email` is the point of
814        // overwriting rather than merely hiding them. Erasure has to leave the
815        // person able to sign up again, and a unique index still holding their
816        // address would refuse — with a duplicate-account error that itself
817        // discloses the deleted account had existed.
818        //
819        // # Relationship to scheduled Art. 17 erasure
820        //
821        // That pipeline (`anonymize_user`, driven by `axiam-server`'s purge
822        // sweep) does everything this does AND pseudonymises the audit log's
823        // actor references with a keyed HMAC, then writes a signed erasure
824        // proof. This is the immediate operational removal an administrator
825        // performs; that is the certified one a data subject requests. Both
826        // leave the account unable to authenticate and holding no personal data;
827        // only the second produces evidence of it.
828        let id_str = id.to_string();
829        let tenant_id_str = tenant_id.to_string();
830
831        // `deleted-<id>` reads as what it is anywhere the row is still shown,
832        // and the address keeps a syntactically valid shape so nothing that
833        // assumes one chokes on it. `.invalid` is reserved by RFC 2606 precisely
834        // so it can never be routed.
835        let pseudonym = format!("deleted-{id_str}");
836        let email_replacement = format!("deleted-{id_str}@deleted.invalid");
837
838        // Every personal-data clause comes from `axiam_core::personal_data`,
839        // which is the single declaration the Art. 17 pipeline below shares
840        // with this path and the Art. 15 export is checked against (T-261). A
841        // column added to the `user` schema and to nothing else fails the
842        // introspection gate in `crates/axiam-db/tests/personal_data_gate.rs`
843        // rather than silently surviving erasure here.
844        //
845        // The clauses appended after it are this path's own, and deliberately
846        // so: `status = 'Deleted'` is what distinguishes the administrator's
847        // tombstone from the certified erasure, and the three resets after it
848        // are ones the Art. 17 pipeline does not perform. Those asymmetries
849        // are recorded on the columns they belong to, not flattened here.
850        //
851        // The SAML single-logout rows naming the person go in the same
852        // transaction (T23.2.4, schema v76, T-381), as on the Art. 17 path.
853        let statement = format!(
854            "BEGIN TRANSACTION; \
855             UPDATE type::record('user', $id) SET {}, \
856             status = 'Deleted', \
857             totp_last_used_step = NONE, \
858             failed_login_attempts = 0, \
859             email_verified_at = NONE, \
860             updated_at = time::now() \
861             WHERE tenant_id = $tenant_id RETURN BEFORE; \
862             {SAML_ERASURE_STATEMENTS} \
863             COMMIT TRANSACTION",
864            axiam_core::personal_data::shared_erasure_fragment()
865        );
866
867        let result = self
868            .db
869            .current()
870            .query(statement)
871            .bind(("id", id_str.clone()))
872            .bind(("tenant_id", tenant_id_str))
873            .bind(("pseudonym", pseudonym))
874            .bind(("email_replacement", email_replacement))
875            .await
876            .map_err(DbError::from)?;
877
878        let mut result = result
879            .check()
880            .map_err(|e| DbError::Migration(e.to_string()))?;
881        // BEGIN=0, UPDATE=1, then the SAML and SSF deletes and COMMIT.
882        let rows: Vec<UserRow> = result.take(1).map_err(DbError::from)?;
883        if rows.is_empty() {
884            return Err(DbError::NotFound {
885                entity: "user".into(),
886                id: id_str,
887            }
888            .into());
889        }
890        self.notify_user_changed(tenant_id, id).await;
891        Ok(())
892    }

1258    async fn increment_failed_logins(
1259        &self,
1260        tenant_id: Uuid,
1261        user_id: Uuid,
1262        lockout_threshold: u32,
1263        base_lockout_secs: i64,
1264        backoff_multiplier: f64,
1265        max_lockout_secs: i64,
1266    ) -> AxiamResult<()> {
1267        let user_id_str = user_id.to_string();
1268        let tenant_id_str = tenant_id.to_string();
1269        // Retried on an optimistic-concurrency loss — see
1270        // `helpers::retry_on_write_conflict`, which this method's own
1271        // hand-rolled loop became once a second caller (`update`, above) needed
1272        // the same thing.
1273        //
1274        // SurrealDB is optimistic, and this statement targets ONE row: every
1275        // failed credential check for a given user updates the SAME record, so
1276        // two concurrent failures race by construction. Before the retry existed
1277        // the loser was reported to the caller as a 5xx — `grpc_admin_validate`
1278        // reproduces it as gRPC INTERNAL on ~0.3-0.7 % of calls — for a write
1279        // the datastore itself labels "can be retried".
1280        //
1281        // Replaying is safe precisely because the loser commits NOTHING, so the
1282        // non-idempotent `failed_login_attempts += 1` cannot double-count. Only
1283        // the conflict class is retried; any other error returns immediately.
1284        crate::helpers::retry_on_write_conflict(|| async {
1285            self.db
1286                .current()
1287                .query(
1288                    // SurrealDB evaluates each RHS in this SET against the
1289                    // pre-update document, so `failed_login_attempts` inside the IF
1290                    // is the OLD count (before `+= 1`). Compare `+ 1` so the lock
1291                    // fires on the Nth failure (when the new count reaches the
1292                    // threshold), not the N+1th — otherwise an account gets one
1293                    // extra guess past the configured limit.
1294                    //
1295                    // Lockout duration grows exponentially with each lockout past
1296                    // the threshold and is capped at $max_secs (brute-force
1297                    // protection): d = min(base * mult^(new_count - threshold), max).
1298                    // `duration::from_secs` needs an int, so the float result of the
1299                    // math is cast with `<int>`. (SurrealDB v3 renamed the duration
1300                    // constructor; `duration::secs` is now an accessor, not a ctor.)
1301                    "UPDATE type::record('user', $id) \
1302                    SET \
1303                    failed_login_attempts += 1, \
1304                    last_failed_login_at = time::now(), \
1305                    locked_until = IF (failed_login_attempts + 1 >= $threshold) \
1306                    THEN time::now() + duration::from_secs(<int> math::min([ \
1307                    $base_secs * math::pow($mult, failed_login_attempts + 1 - $threshold), \
1308                    $max_secs \
1309                    ])) \
1310                    ELSE locked_until \
1311                    END, \
1312                    updated_at = time::now() \
1313                    WHERE tenant_id = $tenant_id",
1314                )
1315                // Cloned per attempt: `bind` takes owned values and this loop
1316                // may run more than once, so it cannot move these out.
1317                .bind(("id", user_id_str.clone()))
1318                .bind(("tenant_id", tenant_id_str.clone()))
1319                .bind(("threshold", lockout_threshold))
1320                .bind(("base_secs", base_lockout_secs))
1321                .bind(("mult", backoff_multiplier))
1322                .bind(("max_secs", max_lockout_secs))
1323                .await
1324                .map_err(DbError::from)
1325                .and_then(|r| {
1326                    r.check()
1327                        .map_err(|e| classify_write_error(e.to_string(), "user"))
1328                })
1329                .map(|_| ())
1330        })
1331        .await
1332        .map_err(Into::into)
1333    }

1335    /// Anonymize a user row in-place (D-05).
1336    ///
1337    /// Scrubs every PII column:
1338    /// - email → `email_hash` (SHA-256 hex of original email, passed by caller)
1339    /// - username → `pseudonym` (DELETED_USER_<hmac>)
1340    /// - password_hash → NULL (login permanently blocked)
1341    /// - mfa_secret → NULL
1342    /// - metadata → `{}`
1343    /// - locked_until / last_failed_login_at → NULL
1344    /// - status → `Anonymized`
1345    ///
1346    /// The row and its `id` are kept to preserve referential integrity for
1347    /// `created_by`/owner foreign-key references.
1348    async fn anonymize_user(
1349        &self,
1350        tenant_id: Uuid,
1351        user_id: Uuid,
1352        email_hash: &str,
1353        pseudonym: &str,
1354    ) -> AxiamResult<()> {
1355        // password_hash is TYPE string (not nullable) — use empty string as
1356        // tombstone value. Argon2 output is never empty, so login is permanently
1357        // blocked without needing to make the column nullable.
1358        // The shared fragment is the same declaration `delete` renders
1359        // (T-261); what follows it is this path's own. Clearing
1360        // `deletion_pending` is the step that marks the erasure done — it is
1361        // the only one that does — which is why it is here and not in the
1362        // shared set.
1363        //
1364        // What SAML single logout remembers about the person goes in the same
1365        // transaction (T23.2.4, schema v76, D-37, T-381): the participant rows
1366        // hold the `NameID` each SP was given — an email address at an
1367        // `emailAddress` SP — and the logout runs name the user whose sessions
1368        // ended. Erasure that left them would be retention.
1369        let statement = format!(
1370            "BEGIN TRANSACTION; \
1371             UPDATE type::record('user', $id) SET {}, \
1372             deletion_pending = false, \
1373             scheduled_purge_at = NONE, \
1374             status = 'Anonymized', \
1375             updated_at = time::now() \
1376             WHERE tenant_id = $tenant_id; \
1377             {SAML_ERASURE_STATEMENTS} \
1378             COMMIT TRANSACTION",
1379            axiam_core::personal_data::shared_erasure_fragment()
1380        );
1381
1382        self.db
1383            .current()
1384            .query(statement)
1385            .bind(("id", user_id.to_string()))
1386            .bind(("email_replacement", email_hash.to_string()))
1387            .bind(("pseudonym", pseudonym.to_string()))
1388            .bind(("tenant_id", tenant_id.to_string()))
1389            .await
1390            .map_err(DbError::from)?
1391            .check()
1392            .map_err(|e| classify_write_error(e.to_string(), "user"))?;
1393        // G-6 (D-57, D-58): the deliverer sees `Anonymized` and sends the
1394        // downstream `DELETE`, whatever the target's deprovision policy says.
1395        self.notify_user_changed(tenant_id, user_id).await;
1396        Ok(())
1397    }

1408    /// Both inserts run inside a single SurrealDB `BEGIN..COMMIT` transaction:
1409    /// if the consent insert fails, the user insert is rolled back. This makes
1410    /// the invariant *"a user never exists without proof-of-consent"* hold even
1411    /// on a partial DB failure (threat T-5-consent-gap). AXIAM never physically
1412    /// deletes user rows (anonymize-in-place preserves FK integrity), so a
1413    /// compensating delete could not satisfy this invariant — the transaction

1428        // through this write path, so their initial password must be seeded
1429        // into `password_history` in the SAME transaction — otherwise the
1430        // unauthenticated reset path's current-password-reuse check
1431        // (`PasswordResetService::confirm_reset`) has zero history rows to

1451                 CREATE type::record('user', $id) SET \
1452                 tenant_id = $tenant_id, \
1453                 username = $username, email = $email, \
1454                 password_hash = $password_hash, \

1462                 CREATE type::record('consent', $consent_id) SET \
1463                 tenant_id = $tenant_id, \
1464                 user_id = $id, \
1465                 consent_type = $consent_type, \
1466                 version = $consent_version, \
1467                 accepted_at = time::now(), \
1468                 ip_address = $ip_address, \
1469                 user_agent = $user_agent; \
1470                 CREATE type::record('password_history', $ph_id) SET \
1471                 tenant_id = $tenant_id, \
1472                 user_id = $id, \
1473                 password_hash = $password_hash; \
1474                 COMMIT TRANSACTION",
1475            )
1476            .bind(("id", id_str.clone()))
1477            .bind(("consent_id", consent_id))
1478            .bind(("ph_id", ph_id_str))
1479            .bind(("tenant_id", tenant_id_str))
1480            .bind(("username", input.username))
1481            .bind(("email", input.email))
1482            .bind(("password_hash", password_hash))

1484            .bind(("metadata", metadata))
