21/// DB-side row struct for queries where the UUID is already known.
22///
23/// NOTE: `Debug` is manually implemented below to redact `mfa_secret` and
24/// `totp_last_used_step` (SEC-043 — AES-256-GCM ciphertext must not leak into
25/// logs or debug output).
26#[derive(SurrealValue)]
27struct UserRow {
28    tenant_id: String,
29    username: String,
30    email: String,
31    password_hash: String,
32    status: String,
33    mfa_enabled: bool,
34    mfa_secret: Option<String>,
35    /// Last TOTP step that was successfully verified (SEC-008/REQ-14 AC-5).
36    totp_last_used_step: Option<u64>,
37    failed_login_attempts: u32,
38    last_failed_login_at: Option<DateTime<Utc>>,
39    locked_until: Option<DateTime<Utc>>,
40    email_verified_at: Option<DateTime<Utc>>,
41    /// GDPR Art. 17 — set when user requests account deletion (D-08).
42    deletion_pending: Option<bool>,
43    /// Scheduled purge date when deletion_pending is true (D-08).
44    scheduled_purge_at: Option<DateTime<Utc>>,
45    metadata: serde_json::Value,
46    created_at: DateTime<Utc>,
47    updated_at: DateTime<Utc>,
48}

142fn parse_status(s: &str) -> Result<UserStatus, DbError> {
143    match s {
144        "Active" => Ok(UserStatus::Active),
145        "Inactive" => Ok(UserStatus::Inactive),
146        "Locked" => Ok(UserStatus::Locked),
147        "PendingVerification" => Ok(UserStatus::PendingVerification),
148        "Anonymized" => Ok(UserStatus::Anonymized),
149        other => Err(DbError::Migration(format!("unknown user status: {other}"))),
150    }
151}
152
153fn status_to_string(s: &UserStatus) -> &'static str {
154    match s {
155        UserStatus::Active => "Active",
156        UserStatus::Inactive => "Inactive",
157        UserStatus::Locked => "Locked",
158        UserStatus::PendingVerification => "PendingVerification",
159        UserStatus::Anonymized => "Anonymized",
160    }
161}

250    async fn create(&self, input: CreateUser) -> AxiamResult<User> {
251        let id = new_id();
252        let id_str = id.to_string();
253        let tenant_id_str = input.tenant_id.to_string();
254
255        let password_hash = password::hash_password(&input.password, self.pepper.as_deref())
256            .map_err(|e| classify_write_error(e, "user"))?;
257
258        let metadata = input
259            .metadata
260            .unwrap_or(serde_json::Value::Object(Default::default()));
261
262        let result = self
263            .db
264            .current()
265            .query(
266                "CREATE type::record('user', $id) SET \
267                 tenant_id = $tenant_id, \
268                 username = $username, email = $email, \
269                 password_hash = $password_hash, \
270                 status = $status, \
271                 mfa_enabled = false, \
272                 failed_login_attempts = 0, \
273                 last_failed_login_at = NONE, \
274                 locked_until = NONE, \
275                 email_verified_at = NONE, \
276                 metadata = $metadata",
277            )
278            .bind(("id", id_str.clone()))
279            .bind(("tenant_id", tenant_id_str))
280            .bind(("username", input.username))
281            .bind(("email", input.email))
282            .bind(("password_hash", password_hash))
283            .bind(("status", "PendingVerification".to_string()))
284            .bind(("metadata", metadata))
285            .await
286            .map_err(DbError::from)?;
287
288        let mut result = result
289            .check()
290            .map_err(|e| classify_write_error(e.to_string(), "user"))?;
291
292        let rows: Vec<UserRow> = result.take(0).map_err(DbError::from)?;
293        let row = rows.into_iter().next().ok_or_else(|| DbError::NotFound {
294            entity: "user".into(),
295            id: id_str,
296        })?;
297
298        Ok(row.into_user(id)?)
299    }

374    async fn update(&self, tenant_id: Uuid, id: Uuid, input: UpdateUser) -> AxiamResult<User> {
375        let id_str = id.to_string();
376        let tenant_id_str = tenant_id.to_string();
377
378        let mut sets = Vec::new();
379        if input.username.is_some() {
380            sets.push("username = $username");
381        }
382        if input.email.is_some() {
383            sets.push("email = $email");
384        }
385        if input.password_hash.is_some() {
386            sets.push("password_hash = $password_hash");
387        }
388        if input.status.is_some() {
389            sets.push("status = $status");
390        }
391        if input.metadata.is_some() {
392            sets.push("metadata = $metadata");
393        }
394        if input.mfa_enabled.is_some() {
395            sets.push("mfa_enabled = $mfa_enabled");
396        }
397        if input.mfa_secret.is_some() {
398            sets.push("mfa_secret = $mfa_secret");
399        }
400        if input.totp_last_used_step.is_some() {
401            sets.push("totp_last_used_step = $totp_last_used_step");
402        }
403        if input.failed_login_attempts.is_some() {
404            sets.push("failed_login_attempts = $failed_login_attempts");
405        }
406        if input.last_failed_login_at.is_some() {
407            sets.push("last_failed_login_at = $last_failed_login_at");
408        }
409        if input.locked_until.is_some() {
410            sets.push("locked_until = $locked_until");
411        }
412        if input.email_verified_at.is_some() {
413            sets.push("email_verified_at = $email_verified_at");
414        }
415        sets.push("updated_at = time::now()");
416
417        let query = format!(
418            "UPDATE type::record('user', $id) SET {} \
419             WHERE tenant_id = $tenant_id",
420            sets.join(", ")
421        );
422
423        let db = self.db.current();
424        let mut builder = db
425            .query(&query)
426            .bind(("id", id_str.clone()))
427            .bind(("tenant_id", tenant_id_str));
428
429        if let Some(username) = input.username {
430            builder = builder.bind(("username", username));
431        }
432        if let Some(email) = input.email {
433            builder = builder.bind(("email", email));
434        }
435        if let Some(password_hash) = input.password_hash {
436            builder = builder.bind(("password_hash", password_hash));
437        }
438        if let Some(ref status) = input.status {
439            builder = builder.bind(("status", status_to_string(status).to_string()));
440        }
441        if let Some(metadata) = input.metadata {
442            builder = builder.bind(("metadata", metadata));
443        }
444        if let Some(mfa_enabled) = input.mfa_enabled {
445            builder = builder.bind(("mfa_enabled", mfa_enabled));
446        }
447        if let Some(mfa_secret) = input.mfa_secret {
448            // mfa_secret is Option<Option<String>>: Some(Some(v)) = set, Some(None) = clear
449            builder = builder.bind(("mfa_secret", mfa_secret));
450        }
451        if let Some(totp_last_used_step) = input.totp_last_used_step {
452            // totp_last_used_step is Option<Option<u64>>: Some(Some(v)) = set, Some(None) = clear
453            builder = builder.bind(("totp_last_used_step", totp_last_used_step));
454        }
455        if let Some(failed_login_attempts) = input.failed_login_attempts {
456            builder = builder.bind(("failed_login_attempts", failed_login_attempts));
457        }
458        if let Some(last_failed_login_at) = input.last_failed_login_at {
459            builder = builder.bind(("last_failed_login_at", last_failed_login_at));
460        }
461        if let Some(locked_until) = input.locked_until {
462            builder = builder.bind(("locked_until", locked_until));
463        }
464        if let Some(email_verified_at) = input.email_verified_at {
465            builder = builder.bind(("email_verified_at", email_verified_at));
466        }
467
468        let result = builder.await.map_err(DbError::from)?;
469        let mut result = result
470            .check()
471            .map_err(|e| classify_write_error(e.to_string(), "user"))?;
472
473        let rows: Vec<UserRow> = result.take(0).map_err(DbError::from)?;
474        let row = rows.into_iter().next().ok_or_else(|| DbError::NotFound {
475            entity: "user".into(),
476            id: id_str,
477        })?;
478
479        Ok(row.into_user(id)?)
480    }

482    async fn delete(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<()> {
483        // Soft-delete: set status to Inactive.
484        let id_str = id.to_string();
485        let tenant_id_str = tenant_id.to_string();
486
487        let result = self
488            .db
489            .current()
490            .query(
491                "UPDATE type::record('user', $id) SET \
492                 status = 'Inactive', updated_at = time::now() \
493                 WHERE tenant_id = $tenant_id RETURN BEFORE",
494            )
495            .bind(("id", id_str.clone()))
496            .bind(("tenant_id", tenant_id_str))
497            .await
498            .map_err(DbError::from)?;

501            .check()
502            .map_err(|e| DbError::Migration(e.to_string()))?;
503        let rows: Vec<UserRow> = result.take(0).map_err(DbError::from)?;
504        if rows.is_empty() {
505            return Err(DbError::NotFound {
506                entity: "user".into(),
507                id: id_str,
508            }
509            .into());
510        }
511        Ok(())
512    }

514    async fn update_totp_step(&self, tenant_id: Uuid, id: Uuid, step: u64) -> AxiamResult<bool> {
515        // Atomic compare-and-set (SEC-008/SECHRD-01): the UPDATE only matches
516        // when the stored step is unset (NONE, first-ever verification —
517        // mirrors `increment_failed_logins`'s NONE-handling) or strictly less
518        // than the incoming step. Wrapping the UPDATE in a SELECT FROM (...)
519        // subquery (same shape as `oauth2_auth_code.rs::consume`) makes the
520        // read-of-affected-rows atomic with the write: exactly one concurrent
521        // caller observes a non-empty row set for a given step.
522        let result = self
523            .db
524            .current()
525            .query(
526                "SELECT meta::id(id) AS record_id FROM \
527                 (UPDATE type::record('user', $id) SET \
528                  totp_last_used_step = $step, updated_at = time::now() \
529                  WHERE tenant_id = $tenant_id \
530                    AND (totp_last_used_step = NONE OR totp_last_used_step < $step))",
531            )
532            .bind(("id", id.to_string()))
533            .bind(("tenant_id", tenant_id.to_string()))
534            .bind(("step", step))
535            .await
536            .map_err(DbError::from)?;
537
538        let mut result = result
539            .check()
540            .map_err(|e| classify_write_error(e.to_string(), "user"))?;
541
542        let rows: Vec<TotpStepCasRow> = result.take(0).map_err(DbError::from)?;
543        Ok(!rows.is_empty())
544    }

546    async fn list(
547        &self,
548        tenant_id: Uuid,
549        pagination: Pagination,
550    ) -> AxiamResult<PaginatedResult<User>> {
551        let tenant_id_str = tenant_id.to_string();
552
553        let mut count_result = self
554            .db
555            .current()
556            .query(
557                "SELECT count() AS total FROM user \
558                 WHERE tenant_id = $tenant_id GROUP ALL",
559            )
560            .bind(("tenant_id", tenant_id_str.clone()))
561            .await
562            .map_err(DbError::from)?;
563        let count_rows: Vec<CountRow> = count_result.take(0).map_err(DbError::from)?;
564        let total = count_rows.first().map(|r| r.total).unwrap_or(0);
565
566        let mut result = self
567            .db
568            .current()
569            .query(
570                // SEC-043: explicit column projection — mfa_secret and
571                // totp_last_used_step are intentionally excluded so they are
572                // never hydrated into list responses.
573                "SELECT meta::id(id) AS record_id, \
574                        tenant_id, username, email, password_hash, status, \
575                        mfa_enabled, \
576                        failed_login_attempts, last_failed_login_at, \
577                        locked_until, email_verified_at, \
578                        deletion_pending, scheduled_purge_at, \
579                        metadata, created_at, updated_at \
580                 FROM user \
581                 WHERE tenant_id = $tenant_id \
582                 ORDER BY created_at ASC \
583                 LIMIT $limit START $offset",
584            )
585            .bind(("tenant_id", tenant_id_str))
586            .bind(("limit", pagination.limit))
587            .bind(("offset", pagination.offset))
588            .await
589            .map_err(DbError::from)?;
590
591        let rows: Vec<UserRowWithId> = result.take(0).map_err(DbError::from)?;
592
593        let items = rows
594            .into_iter()
595            .map(|row| row.try_into_user())
596            .collect::<Result<Vec<_>, DbError>>()?;
597
598        Ok(PaginatedResult {
599            items,
600            total,
601            offset: pagination.offset,
602            limit: pagination.limit,
603        })
604    }

606    async fn increment_failed_logins(
607        &self,
608        tenant_id: Uuid,
609        user_id: Uuid,
610        lockout_threshold: u32,
611        base_lockout_secs: i64,
612        backoff_multiplier: f64,
613        max_lockout_secs: i64,
614    ) -> AxiamResult<()> {
615        let user_id_str = user_id.to_string();
616        let tenant_id_str = tenant_id.to_string();
617        // Retried on an optimistic-concurrency loss (`is_write_conflict`).
618        //
619        // SurrealDB is optimistic, and this statement targets ONE row: every
620        // failed credential check for a given user updates the SAME record, so
621        // two concurrent failures race by construction. Before this loop the
622        // loser was reported to the caller as a 5xx — `grpc_admin_validate`
623        // reproduces it as gRPC INTERNAL on ~0.3-0.7 % of calls — for a write
624        // the datastore itself labels "can be retried".
625        //
626        // Replaying is safe precisely because the loser commits NOTHING, so the
627        // non-idempotent `failed_login_attempts += 1` cannot double-count. Only
628        // the conflict class is retried; any other error returns immediately.
629        let mut attempt = 1;
630        loop {
631            let outcome = self
632                .db
633                .current()
634                .query(
635                    // SurrealDB evaluates each RHS in this SET against the
636                    // pre-update document, so `failed_login_attempts` inside the IF
637                    // is the OLD count (before `+= 1`). Compare `+ 1` so the lock
638                    // fires on the Nth failure (when the new count reaches the
639                    // threshold), not the N+1th — otherwise an account gets one
640                    // extra guess past the configured limit.
641                    //
642                    // Lockout duration grows exponentially with each lockout past
643                    // the threshold and is capped at $max_secs (brute-force
644                    // protection): d = min(base * mult^(new_count - threshold), max).
645                    // `duration::from_secs` needs an int, so the float result of the
646                    // math is cast with `<int>`. (SurrealDB v3 renamed the duration
647                    // constructor; `duration::secs` is now an accessor, not a ctor.)
648                    "UPDATE type::record('user', $id) \
649                    SET \
650                    failed_login_attempts += 1, \
651                    last_failed_login_at = time::now(), \
652                    locked_until = IF (failed_login_attempts + 1 >= $threshold) \
653                    THEN time::now() + duration::from_secs(<int> math::min([ \
654                    $base_secs * math::pow($mult, failed_login_attempts + 1 - $threshold), \
655                    $max_secs \
656                    ])) \
657                    ELSE locked_until \
658                    END, \
659                    updated_at = time::now() \
660                    WHERE tenant_id = $tenant_id",
661                )
662                // Cloned per attempt: `bind` takes owned values and this loop
663                // may run more than once, so it cannot move these out.
664                .bind(("id", user_id_str.clone()))
665                .bind(("tenant_id", tenant_id_str.clone()))
666                .bind(("threshold", lockout_threshold))
667                .bind(("base_secs", base_lockout_secs))
668                .bind(("mult", backoff_multiplier))
669                .bind(("max_secs", max_lockout_secs))
670                .await
671                .map_err(DbError::from)
672                .and_then(|r| {
673                    r.check()
674                        .map_err(|e| classify_write_error(e.to_string(), "user"))
675                });
676            match outcome {
677                Err(e)
678                    if attempt < crate::helpers::MAX_WRITE_ATTEMPTS
679                        && crate::helpers::is_write_conflict(&e.to_string()) =>
680                {
681                    tokio::time::sleep(crate::helpers::write_conflict_backoff(attempt)).await;
682                    attempt += 1;
683                }
684                Err(e) => return Err(e.into()),
685                Ok(_) => return Ok(()),
686            }
687        }
688    }

690    /// Anonymize a user row in-place (D-05).
691    ///
692    /// Scrubs every PII column:
693    /// - email → `email_hash` (SHA-256 hex of original email, passed by caller)
694    /// - username → `pseudonym` (DELETED_USER_<hmac>)
695    /// - password_hash → NULL (login permanently blocked)
696    /// - mfa_secret → NULL
697    /// - metadata → `{}`
698    /// - locked_until / last_failed_login_at → NULL
699    /// - status → `Anonymized`
700    ///
701    /// The row and its `id` are kept to preserve referential integrity for
702    /// `created_by`/owner foreign-key references.
703    async fn anonymize_user(
704        &self,
705        tenant_id: Uuid,
706        user_id: Uuid,
707        email_hash: &str,
708        pseudonym: &str,
709    ) -> AxiamResult<()> {
710        // password_hash is TYPE string (not nullable) — use empty string as
711        // tombstone value. Argon2 output is never empty, so login is permanently
712        // blocked without needing to make the column nullable.
713        self.db
714            .current()
715            .query(
716                "UPDATE type::record('user', $id) SET \
717                 email = $email_hash, \
718                 username = $pseudonym, \
719                 password_hash = '', \
720                 mfa_secret = NONE, \
721                 mfa_enabled = false, \
722                 metadata = {}, \
723                 locked_until = NONE, \
724                 last_failed_login_at = NONE, \
725                 deletion_pending = false, \
726                 scheduled_purge_at = NONE, \
727                 status = 'Anonymized', \
728                 updated_at = time::now() \
729                 WHERE tenant_id = $tenant_id",
730            )
731            .bind(("id", user_id.to_string()))
732            .bind(("email_hash", email_hash.to_string()))
733            .bind(("pseudonym", pseudonym.to_string()))
734            .bind(("tenant_id", tenant_id.to_string()))
735            .await
736            .map_err(DbError::from)?
737            .check()
738            .map_err(|e| classify_write_error(e.to_string(), "user"))?;
739        Ok(())
740    }

748    /// Atomically create a user together with its `terms_of_service` consent
749    /// row (REQ-8 / GDPR Art. 7 proof-of-consent).
750    ///
751    /// Both inserts run inside a single SurrealDB `BEGIN..COMMIT` transaction:
752    /// if the consent insert fails, the user insert is rolled back. This makes
753    /// the invariant *"a user never exists without proof-of-consent"* hold even
754    /// on a partial DB failure (threat T-5-consent-gap). AXIAM never physically
755    /// deletes user rows (anonymize-in-place preserves FK integrity), so a
756    /// compensating delete could not satisfy this invariant — the transaction
757    /// is required.
758    pub async fn create_with_consent(
759        &self,
760        input: CreateUser,
761        consent_type: &str,
762        consent_version: &str,
763        ip_address: Option<String>,
764        user_agent: Option<String>,
765    ) -> AxiamResult<User> {
766        let id = new_id();
767        let id_str = id.to_string();
768        let consent_id = new_id().to_string();
769        let tenant_id_str = input.tenant_id.to_string();
770        // SECHRD-12/T-24-93 (RESEARCH Pitfall 5): admin-created users go
771        // through this write path, so their initial password must be seeded
772        // into `password_history` in the SAME transaction — otherwise the
773        // unauthenticated reset path's current-password-reuse check
774        // (`PasswordResetService::confirm_reset`) has zero history rows to
775        // work against for these users. Federated-user creation paths
776        // (oidc.rs/saml.rs) have no local password and are intentionally
777        // left untouched.
778        let ph_id_str = new_id().to_string();
779
780        let password_hash = password::hash_password(&input.password, self.pepper.as_deref())
781            .map_err(|e| classify_write_error(e, "user"))?;
782
783        let metadata = input
784            .metadata
785            .unwrap_or(serde_json::Value::Object(Default::default()));
786
787        // Result slots: BEGIN=0, CREATE user=1, CREATE consent=2,
788        // CREATE password_history=3, COMMIT=4.
789        let result = self
790            .db
791            .current()
792            .query(
793                "BEGIN TRANSACTION; \
794                 CREATE type::record('user', $id) SET \
795                 tenant_id = $tenant_id, \
796                 username = $username, email = $email, \
797                 password_hash = $password_hash, \
798                 status = $status, \
799                 mfa_enabled = false, \
800                 failed_login_attempts = 0, \
801                 last_failed_login_at = NONE, \
802                 locked_until = NONE, \
803                 email_verified_at = NONE, \
804                 metadata = $metadata; \
805                 CREATE type::record('consent', $consent_id) SET \
806                 tenant_id = $tenant_id, \
807                 user_id = $id, \
808                 consent_type = $consent_type, \
809                 version = $consent_version, \
810                 accepted_at = time::now(), \
811                 ip_address = $ip_address, \
812                 user_agent = $user_agent; \
813                 CREATE type::record('password_history', $ph_id) SET \
814                 tenant_id = $tenant_id, \
815                 user_id = $id, \
816                 password_hash = $password_hash; \
817                 COMMIT TRANSACTION",
818            )
819            .bind(("id", id_str.clone()))
820            .bind(("consent_id", consent_id))
821            .bind(("ph_id", ph_id_str))
822            .bind(("tenant_id", tenant_id_str))
823            .bind(("username", input.username))
824            .bind(("email", input.email))
825            .bind(("password_hash", password_hash))
826            .bind(("status", "PendingVerification".to_string()))
827            .bind(("metadata", metadata))
828            .bind(("consent_type", consent_type.to_string()))
829            .bind(("consent_version", consent_version.to_string()))
830            .bind(("ip_address", ip_address))
831            .bind(("user_agent", user_agent))
832            .await
833            .map_err(DbError::from)?;
834
835        let mut result = result
836            .check()
837            .map_err(|e| classify_write_error(e.to_string(), "user"))?;
838
839        // The user CREATE result is at index 1 (BEGIN occupies slot 0).
840        let rows: Vec<UserRow> = result.take(1).map_err(DbError::from)?;
841        let row = rows.into_iter().next().ok_or_else(|| DbError::NotFound {
842            entity: "user".into(),
843            id: id_str,
844        })?;
845
846        Ok(row.into_user(id)?)
847    }
