59fn parse_credential_type(s: &str) -> Result<WebauthnCredentialType, DbError> {
60    match s {
61        "Passkey" => Ok(WebauthnCredentialType::Passkey),
62        "SecurityKey" => Ok(WebauthnCredentialType::SecurityKey),
63        other => Err(DbError::Migration(format!(
64            "invalid credential_type: {other}"
65        ))),
66    }
67}

69/// Parse an optional stored AAGUID string. A `None` column reads back as
70/// `None`; a corrupt (non-UUID) value is reported rather than silently
71/// dropped, matching `parse_uuid`'s D-10 rationale for other UUID columns.
72fn parse_optional_aaguid(s: Option<String>) -> Result<Option<Uuid>, DbError> {
73    s.map(|s| crate::helpers::parse_uuid(&s, "aaguid"))
74        .transpose()
75}

77fn row_to_credential(row: WebauthnCredentialRow, id: Uuid) -> Result<WebauthnCredential, DbError> {
78    let tenant_id = Uuid::parse_str(&row.tenant_id)
79        .map_err(|e| DbError::Migration(format!("invalid tenant UUID: {e}")))?;
80    let user_id = Uuid::parse_str(&row.user_id)
81        .map_err(|e| DbError::Migration(format!("invalid user UUID: {e}")))?;
82    let credential_type = parse_credential_type(&row.credential_type)?;
83    let aaguid = parse_optional_aaguid(row.aaguid)?;
84    Ok(WebauthnCredential {
85        id,
86        tenant_id,
87        user_id,
88        credential_id: row.credential_id,
89        name: row.name,
90        credential_type,
91        passkey_json: row.passkey_json,
92        created_at: row.created_at,
93        last_used_at: row.last_used_at,
94        // D6 (X3, wave 2): read back whatever was recorded at registration
95        // time. A pre-X3 row has no column value at all, which the schema
96        // (v35) reads as None/false — the same "today's behavior unchanged"
97        // guarantee the wave-1 model already documents.
98        aaguid,
99        attestation_format: row.attestation_format,
100        attested: row.attested,
101        authenticator_name: row.authenticator_name,
102    })
103}

134/// SurrealDB implementation of the WebAuthn credential repository.
135#[derive(Clone)]
136pub struct SurrealWebauthnCredentialRepository<C: Connection> {
137    db: DbHandle<C>,
138}

140impl<C: Connection> SurrealWebauthnCredentialRepository<C> {
141    /// Create a new repository backed by the given SurrealDB client.
142    pub fn new(db: impl Into<DbHandle<C>>) -> Self {
143        let db = db.into();
144        Self { db }
145    }
146}

148impl<C: Connection> WebauthnCredentialRepository for SurrealWebauthnCredentialRepository<C> {
149    async fn create(&self, input: CreateWebauthnCredential) -> AxiamResult<WebauthnCredential> {
150        let id = new_id();
151        let id_str = id.to_string();
152
153        let result = self
154            .db
155            .current()
156            .query(
157                "CREATE type::record('webauthn_credential', $id) SET \
158                 tenant_id = $tenant_id, \
159                 user_id = $user_id, \
160                 credential_id = $credential_id, \
161                 name = $name, \
162                 credential_type = $credential_type, \
163                 passkey_json = $passkey_json, \
164                 aaguid = $aaguid, \
165                 attestation_format = $attestation_format, \
166                 attested = $attested, \
167                 authenticator_name = $authenticator_name",
168            )
169            .bind(("id", id_str.clone()))
170            .bind(("tenant_id", input.tenant_id.to_string()))
171            .bind(("user_id", input.user_id.to_string()))
172            .bind(("credential_id", input.credential_id))
173            .bind(("name", input.name))
174            .bind((
175                "credential_type",
176                input.credential_type.as_str().to_string(),
177            ))
178            .bind(("passkey_json", input.passkey_json))
179            .bind(("aaguid", input.aaguid.map(|u| u.to_string())))
180            .bind(("attestation_format", input.attestation_format))
181            .bind(("attested", input.attested))
182            .bind(("authenticator_name", input.authenticator_name))
183            .await
184            .map_err(DbError::from)?;
185
186        let mut result = result
187            .check()
188            .map_err(|e| DbError::Migration(e.to_string()))?;
189
190        let rows: Vec<WebauthnCredentialRow> = result.take(0).map_err(DbError::from)?;
191        let row = take_first_or_not_found(rows, "webauthn_credential", &id_str)?;
192
193        row_to_credential(row, id).map_err(Into::into)
194    }

196    async fn get_by_id(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<WebauthnCredential> {
197        let id_str = id.to_string();
198
199        let mut result = self
200            .db
201            .current()
202            .query(
203                "SELECT * FROM type::record('webauthn_credential', $id) \
204                 WHERE tenant_id = $tenant_id",
205            )
206            .bind(("id", id_str.clone()))
207            .bind(("tenant_id", tenant_id.to_string()))
208            .await
209            .map_err(DbError::from)?;
210
211        let rows: Vec<WebauthnCredentialRow> = result.take(0).map_err(DbError::from)?;
212        let row = take_first_or_not_found(rows, "webauthn_credential", &id_str)?;
213
214        row_to_credential(row, id).map_err(Into::into)
215    }

217    async fn list_by_user(
218        &self,
219        tenant_id: Uuid,
220        user_id: Uuid,
221    ) -> AxiamResult<Vec<WebauthnCredential>> {
222        let mut result = self
223            .db
224            .current()
225            .query(
226                "SELECT meta::id(id) AS record_id, * \
227                 FROM webauthn_credential \
228                 WHERE tenant_id = $tenant_id \
229                 AND user_id = $user_id \
230                 ORDER BY created_at DESC",
231            )
232            .bind(("tenant_id", tenant_id.to_string()))
233            .bind(("user_id", user_id.to_string()))
234            .await
235            .map_err(DbError::from)?;
236
237        let rows: Vec<WebauthnCredentialRowWithId> = result.take(0).map_err(DbError::from)?;
238        rows.into_iter()
239            .map(|r| r.try_into_credential().map_err(Into::into))
240            .collect()
241    }

297    async fn count_by_user(&self, tenant_id: Uuid, user_id: Uuid) -> AxiamResult<u64> {
298        let mut result = self
299            .db
300            .current()
301            .query(
302                "SELECT count() AS total FROM webauthn_credential \
303                 WHERE tenant_id = $tenant_id \
304                 AND user_id = $user_id \
305                 GROUP ALL",
306            )
307            .bind(("tenant_id", tenant_id.to_string()))
308            .bind(("user_id", user_id.to_string()))
309            .await
310            .map_err(DbError::from)?;
311
312        let rows: Vec<CountRow> = result.take(0).map_err(DbError::from)?;
313        Ok(rows.first().map(|r| r.total).unwrap_or(0))
