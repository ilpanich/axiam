1//! Startup seeders for populating static data.
2//!
3//! The permission seeder uses raw SurrealQL `UPSERT` to ensure idempotency
4//! under concurrent startup (per D-07). A deterministic UUID derived from
5//! `namespace = tenant_id` + `name = action` ensures the same record is
6//! always targeted on subsequent restarts — true idempotency via UPSERT.
7//!
8//! CQ-B42: a sha256 hash of the registry is persisted in `seeder_state` per
9//! tenant. On startup, if the hash matches the stored value the UPSERT loop is
10//! skipped entirely, eliminating the O(n×95) boot storm on unchanged systems.

41/// Seed all permissions in `registry` for the given `tenant_id`.
42///
43/// # Idempotency
44///
45/// A deterministic UUID is generated for each `(tenant_id, action)` pair
46/// using [`Uuid::new_v5`] with `tenant_id` as the namespace.  The same pair
47/// always produces the same record ID, so the UPSERT targets the same row on
48/// every restart.
49///
50/// # SurrealQL
51///
52/// Uses raw `UPSERT` — NOT `list()` + conditional `create()`.  The hand-rolled
53/// pattern is race-prone under concurrent startup (per RESEARCH.md) and is
54/// explicitly prohibited by D-07.
55pub async fn seed_permissions<C: Connection>(
56    db: &Surreal<C>,
57    tenant_id: Uuid,
58    registry: &[(&str, &str)],
59) -> Result<(), DbError> {
60    // CQ-B42: compute sha256 hash of the registry to guard against UPSERT storm.
61    let registry_hash = {
62        let mut h = Sha256::new();
63        for (action, desc) in registry {
64            h.update(action.as_bytes());
65            h.update(b"|");
66            h.update(desc.as_bytes());
67        }
68        hex::encode(h.finalize())
69    };
70
71    // Derive a stable per-tenant record ID for seeder_state.
72    let state_id = Uuid::new_v5(&tenant_id, b"seeder_state").to_string();
73
74    // Check existing seeder_state; skip UPSERT loop if hash is unchanged.
75    let existing: Vec<SeederStateRow> = db
76        .query("SELECT hash FROM type::record('seeder_state', $id)")
77        .bind(("id", state_id.clone()))
78        .await
79        .map_err(|e| DbError::Migration(format!("seeder_state read failed: {e}")))?
80        .take(0)
81        .map_err(|e| DbError::Migration(format!("seeder_state take failed: {e}")))?;
82
83    if existing
84        .first()
85        .map(|r| r.hash.as_str())
86        .is_some_and(|h| h == registry_hash.as_str())
87    {
88        tracing::debug!(
89            %tenant_id,
90            "seed_permissions: registry hash unchanged — skipping UPSERT storm (CQ-B42)"
91        );
92        return Ok(());
93    }
94
95    for (action, description) in registry {
96        // Deterministic UUID: same tenant + action always produces same ID.
97        let id = Uuid::new_v5(&tenant_id, action.as_bytes());
98        let id_str = id.to_string();
99        let tenant_str = tenant_id.to_string();
100
101        db.query(
102            "UPSERT type::record('permission', $id) SET \
103             tenant_id = $tenant_id, \
104             action = $action, \
105             description = $description, \
106             created_at = IF (SELECT created_at FROM type::record('permission', $id))[0].created_at \
107               THEN (SELECT created_at FROM type::record('permission', $id))[0].created_at \
108               ELSE time::now() END, \
109             updated_at = time::now()",
110        )
111        .bind(("id", id_str))
112        .bind(("tenant_id", tenant_str))
113        .bind(("action", action.to_string()))
114        .bind(("description", description.to_string()))
115        .await
116        .map_err(|e| DbError::Migration(format!("seed_permissions UPSERT failed: {e}")))?
117        .check()
118        .map_err(|e| DbError::Migration(format!("seed_permissions UPSERT check failed: {e}")))?;
119    }
120
121    // Persist the new hash so subsequent restarts can skip this tenant.
122    let tenant_str = tenant_id.to_string();
123    db.query(
124        "UPSERT type::record('seeder_state', $id) SET \
125         tenant_id = $tenant_id, hash = $hash, updated_at = time::now()",
126    )
127    .bind(("id", state_id))
128    .bind(("tenant_id", tenant_str))
129    .bind(("hash", registry_hash))
130    .await
131    .map_err(|e| DbError::Migration(format!("seeder_state upsert failed: {e}")))?
132    .check()
133    .map_err(|e| DbError::Migration(format!("seeder_state upsert check: {e}")))?;
134
135    Ok(())
136}

138/// Mint a one-time first-run bootstrap setup token if the database has
139/// never been bootstrapped (SECHRD-04 / D-03b).
140///
141/// # Behavior
142///
143/// - On a fresh (never-bootstrapped) database — no `bootstrap_setup_token`
144///   row yet AND no `user` row anywhere — a cryptographically random token
145///   is generated, its sha256 hash is persisted as the
146///   `bootstrap_setup_token` record ID, and the returned `Some(token)`
147///   carries the PLAINTEXT token so the caller can log it exactly once
148///   (the one deliberate secret-log exception, per D-03b). Only the HASH
149///   is ever persisted to the database.
150/// - On any subsequent boot — a setup token already exists, or at least
151///   one user already exists — this is a no-op and returns `Ok(None)`;
152///   nothing is minted or re-logged.
153pub async fn mint_bootstrap_setup_token_if_needed<C: Connection>(
154    db: &Surreal<C>,
155) -> Result<Option<String>, DbError> {
156    // A setup token was already minted — no-op.
157    let existing_tokens: Vec<CountRow> = db
158        .query("SELECT count() AS total FROM bootstrap_setup_token GROUP ALL")
159        .await
160        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token count failed: {e}")))?
161        .take(0)
162        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token count take failed: {e}")))?;
163    if existing_tokens.first().map(|r| r.total).unwrap_or(0) > 0 {
164        return Ok(None);
165    }
166
167    // Bootstrap already completed somewhere — no-op.
168    let existing_users: Vec<CountRow> = db
169        .query("SELECT count() AS total FROM user GROUP ALL")
170        .await
171        .map_err(|e| DbError::Migration(format!("user count failed: {e}")))?
172        .take(0)
173        .map_err(|e| DbError::Migration(format!("user count take failed: {e}")))?;
174    if existing_users.first().map(|r| r.total).unwrap_or(0) > 0 {
175        return Ok(None);
176    }
177
178    // Generate a cryptographically random 32-byte token, base64url-encoded
179    // (same shape as `axiam_auth::token::generate_refresh_token`).
180    let mut rng = rand::rng();
181    let bytes: [u8; 32] = rand::RngExt::random(&mut rng);
182    let token = URL_SAFE_NO_PAD.encode(bytes);
183
184    let mut hasher = Sha256::new();
185    hasher.update(token.as_bytes());
186    let token_hash = hex::encode(hasher.finalize());
187
188    db.query("CREATE type::record('bootstrap_setup_token', $hash) SET created_at = time::now()")
189        .bind(("hash", token_hash))
190        .await
191        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token mint failed: {e}")))?
192        .check()
193        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token mint check failed: {e}")))?;
194
195    Ok(Some(token))
196}

198/// Whether the seeded roles of `tenant_id` may hold organization-level
199/// actions — true only for the organization's own scope.
200///
201/// Fails **closed**: a tenant that cannot be resolved is treated as ordinary,
202/// so the organization-level actions are withheld. Being wrong in that
203/// direction costs an operator a grant they can add back by hand; being wrong
204/// in the other seeds the escalation this filter exists to prevent.
205async fn tenant_may_hold_organization_actions<C: Connection>(
206    db: &Surreal<C>,
207    tenant_id: Uuid,
208) -> bool {
209    SurrealTenantRepository::new(db.clone())
210        .get_by_id(tenant_id)
211        .await
212        .map(|t| t.is_organization_scope())
213        .unwrap_or(false)
214}

216/// Seed the three default roles — super-admin, admin, and viewer — for the
217/// given tenant, assigning permissions according to D-12.
218///
219/// # Idempotency
220///
221/// If all three roles already exist the function returns their IDs without
222/// creating duplicates. If only some roles are missing they are created.
223///
224/// # Permission assignment rules
225///
226/// - **super-admin**: every permission in the tenant.
227/// - **admin**: every permission except `admin:bootstrap`.
228/// - **viewer**: permissions whose action ends with `:list` or `:get`.
229///
230/// # Organization-level actions
231///
232/// In an ordinary (`TenantKind::Standard`) tenant, the actions listed in
233/// [`ORGANIZATION_LEVEL_ACTIONS`](axiam_core::permission_scope::ORGANIZATION_LEVEL_ACTIONS)
234/// are withheld from super-admin and admin. A tenant administrator holding
235/// `ca_certificates:manage` could flag an organization-wide mTLS trust anchor
236/// and then mint certificates authenticating as principals in sibling tenants
237/// (B-04) — so the grant is not made in the first place. The organization's
238/// own scope is seeded with everything.
239///
240/// This is the second of two layers. `require_organization_principal` in
241/// `axiam-api-rest` is the load-bearing one and refuses such a caller whatever
242/// grants it holds; withholding the grant means the two layers agree instead of
243/// one quietly contradicting the other. The `viewer` role is untouched — every
244/// organization-level action is a mutation and viewer is seeded only with
245/// `:list` and `:get`.
246pub async fn seed_default_roles<C: Connection>(
247    db: &Surreal<C>,
248    tenant_id: Uuid,
249    _permission_registry: &[(&str, &str)],
250) -> Result<SeedRolesResult, DbError> {
251    let role_repo = SurrealRoleRepository::new(db.clone());
252    let perm_repo = SurrealPermissionRepository::new(db.clone());
253
254    // -----------------------------------------------------------------------
255    // 1. Ensure all three roles exist (idempotent)
256    // -----------------------------------------------------------------------
257
258    // Fetch existing roles for this tenant in a single list call.
259    let existing = role_repo
260        .list(
261            tenant_id,
262            Pagination {
263                offset: 0,
264                limit: 1000,
265                search: None,
266            },
267        )
268        .await
269        .map_err(|e| DbError::Migration(format!("seed_default_roles list roles failed: {e}")))?;
270
271    let find_role = |name: &str| existing.items.iter().find(|r| r.name == name).map(|r| r.id);
272
273    // SECHRD-04: role creation must tolerate a CONCURRENT racer creating the
274    // same default role for the same tenant — `find_or_create_role` treats
275    // the `idx_role_tenant_name` UNIQUE-index violation as "a concurrent
276    // caller already won" and re-fetches the winner's row instead of
277    // failing (see its doc comment for why this is now reachable).
278    let super_admin_id = find_or_create_role(
279        &role_repo,
280        tenant_id,
281        find_role("super-admin"),
282        "super-admin",
283        "Full system access — all permissions",
284    )
285    .await?;
286
287    let admin_id = find_or_create_role(
288        &role_repo,
289        tenant_id,
290        find_role("admin"),
291        "admin",
292        "Administrative access — all entity CRUD operations",
293    )
294    .await?;
295
296    let viewer_id = find_or_create_role(
297        &role_repo,
298        tenant_id,
299        find_role("viewer"),
300        "viewer",
301        "Read-only access — list and get operations",
302    )
303    .await?;
304
305    // -----------------------------------------------------------------------
306    // 2. Fetch all permissions for this tenant
307    // -----------------------------------------------------------------------
308
309    let permissions = perm_repo
310        .list(
311            tenant_id,
312            Pagination {
313                offset: 0,
314                limit: 10_000,
315                search: None,
316            },
317        )
318        .await
319        .map_err(|e| {
320            DbError::Migration(format!("seed_default_roles list permissions failed: {e}"))
321        })?;
322
323    // -----------------------------------------------------------------------
324    // 3. Grant permissions to roles (idempotent — skip already-granted pairs)
325    // -----------------------------------------------------------------------
326    //
327    // SECHRD-04: `grant_to_role` CREATEs a `grants` edge guarded by a UNIQUE
328    // (in, out) index (CQ-B17) — re-granting an already-granted permission
329    // is a hard error, not a silent no-op. Before this fix, `seed_default_roles`
330    // was only ever called once per tenant (the old bootstrap TOCTOU check
331    // disabled the endpoint before a second call could reach here). Now that
332    // bootstrap atomicity is enforced by the `bootstrap_lock` uniqueness
333    // invariant instead, this function can be reached again — including
334    // CONCURRENTLY, by two racing bootstrap requests — for a tenant whose
335    // roles+grants already exist. The pre-fetch below skips already-granted
336    // pairs in the common (sequential retry) case; `grant_to_role_idempotent`
337    // additionally tolerates the residual TOCTOU window between this
338    // pre-fetch and the CREATE, for true concurrent callers.
339
340    let super_admin_have = granted_permission_ids(&perm_repo, tenant_id, super_admin_id).await?;
341    let admin_have = granted_permission_ids(&perm_repo, tenant_id, admin_id).await?;
342    let viewer_have = granted_permission_ids(&perm_repo, tenant_id, viewer_id).await?;
343
344    // B-04 defence in depth: an ordinary tenant's roles are not seeded with
345    // organization-level actions. See this function's doc comment.
346    let may_hold_org_actions = tenant_may_hold_organization_actions(db, tenant_id).await;
347    let withheld = |action: &str| !may_hold_org_actions && is_organization_level_action(action);
348
349    for perm in &permissions.items {
350        if withheld(&perm.action) {
351            continue;
352        }
353
354        // super-admin gets ALL permissions.
355        if !super_admin_have.contains(&perm.id) {
356            grant_to_role_idempotent(&perm_repo, tenant_id, super_admin_id, perm.id)
357                .await
358                .map_err(|e| {
359                    DbError::Migration(format!(
360                        "seed_default_roles grant super-admin permission {} failed: {e}",
361                        perm.action
362                    ))
363                })?;
364        }
365
366        // admin gets every permission EXCEPT admin:bootstrap.
367        if perm.action != "admin:bootstrap" && !admin_have.contains(&perm.id) {
368            grant_to_role_idempotent(&perm_repo, tenant_id, admin_id, perm.id)
369                .await
370                .map_err(|e| {
371                    DbError::Migration(format!(
372                        "seed_default_roles grant admin permission {} failed: {e}",
373                        perm.action
374                    ))
375                })?;
376        }
377
378        // viewer gets only :list and :get permissions.
379        if (perm.action.ends_with(":list") || perm.action.ends_with(":get"))
380            && !viewer_have.contains(&perm.id)
381        {
382            grant_to_role_idempotent(&perm_repo, tenant_id, viewer_id, perm.id)
383                .await
384                .map_err(|e| {
385                    DbError::Migration(format!(
386                        "seed_default_roles grant viewer permission {} failed: {e}",
387                        perm.action
388                    ))
389                })?;
390        }
391    }
392
393    Ok(SeedRolesResult {
394        super_admin_role_id: super_admin_id,
395        admin_role_id: admin_id,
396        viewer_role_id: viewer_id,
397    })
398}

400/// Find an existing role by name, or create it — tolerating a CONCURRENT
401/// caller creating the SAME (tenant_id, name) role at the same time.
402///
403/// `role_repo.create()` CREATEs a fresh-UUID row guarded by the
404/// `idx_role_tenant_name` UNIQUE index; two racing bootstrap requests can
405/// both observe "role missing" and both attempt a create. The loser's
406/// UNIQUE-index violation is NOT a real failure here — a concurrent caller
407/// already won and the row exists — so it is treated as "re-fetch and use
408/// the winner's ID" rather than propagated (SECHRD-04: `seed_default_roles`
409/// is reachable from two concurrent bootstrap requests now that atomicity
410/// is enforced by `bootstrap_lock`, not the old TOCTOU check).
411async fn find_or_create_role<C: Connection>(
412    role_repo: &SurrealRoleRepository<C>,
413    tenant_id: Uuid,
414    existing_id: Option<Uuid>,
415    name: &str,
416    description: &str,
417) -> Result<Uuid, DbError> {
418    if let Some(id) = existing_id {
419        return Ok(id);
420    }
421
422    match role_repo
423        .create(CreateRole {
424            tenant_id,
425            name: name.to_string(),
426            description: description.to_string(),
427            is_global: true,
428        })
429        .await
430    {
431        Ok(role) => Ok(role.id),
432        Err(e) => {
433            let msg = e.to_string();
434            match classify_write_error(&msg, "role") {
435                DbError::AlreadyExists { .. } => {
436                    let refreshed = role_repo
437                        .list(
438                            tenant_id,
439                            Pagination {
440                                offset: 0,
441                                limit: 1000,
442                                search: None,
443                            },
444                        )
445                        .await
446                        .map_err(|e| {
447                            DbError::Migration(format!(
448                                "find_or_create_role re-list '{name}' failed: {e}"
449                            ))
450                        })?;
451                    refreshed
452                        .items
453                        .into_iter()
454                        .find(|r| r.name == name)
455                        .map(|r| r.id)
456                        .ok_or_else(|| {
457                            DbError::Migration(format!(
458                                "find_or_create_role: role '{name}' vanished after concurrent create race"
459                            ))
460                        })
461                }
462                _ => Err(DbError::Migration(format!(
463                    "find_or_create_role '{name}' failed: {msg}"
464                ))),
465            }
466        }
467    }
468}

470/// The set of permission IDs already granted to `role_id` (tenant-scoped).
471///
472/// Shared by [`seed_default_roles`] and [`reconcile_default_role_grants`] so
473/// neither re-attempts a `grant_to_role` CREATE for a pair that already
474/// exists — the `grants` edge table's UNIQUE (in, out) index (CQ-B17) makes
475/// a duplicate grant a hard error, not a no-op.
476async fn granted_permission_ids<C: Connection>(
477    perm_repo: &SurrealPermissionRepository<C>,
478    tenant_id: Uuid,
479    role_id: Uuid,
480) -> Result<std::collections::HashSet<Uuid>, DbError> {
481    perm_repo
482        .get_role_permissions(tenant_id, role_id)
483        .await
484        .map(|perms| perms.into_iter().map(|p| p.id).collect())
485        .map_err(|e| DbError::Migration(format!("granted_permission_ids failed: {e}")))
486}

488/// Grant `permission_id` to `role_id`, treating "already granted" (a
489/// concurrent caller won the same grant) as success rather than an error.
490///
491/// Closes the residual TOCTOU between [`granted_permission_ids`]'s pre-fetch
492/// and this CREATE for two truly concurrent callers (SECHRD-04).
493async fn grant_to_role_idempotent<C: Connection>(
494    perm_repo: &SurrealPermissionRepository<C>,
495    tenant_id: Uuid,
496    role_id: Uuid,
497    permission_id: Uuid,
498) -> Result<(), DbError> {
499    match perm_repo
500        .grant_to_role(tenant_id, role_id, permission_id)
501        .await
502    {
503        Ok(()) => Ok(()),
504        Err(e) => match classify_write_error(e.to_string(), "permission_grant") {
505            DbError::AlreadyExists { .. } => Ok(()),
506            other => Err(other),
507        },
508    }
509}

511/// What [`reconcile_default_role_grants`] changed for one tenant.
512///
513/// Two counters rather than one, because they mean opposite things to an
514/// operator reading the boot log: `granted` is a back-fill catching a role up
515/// with a registry that grew, while `revoked` is a **capability being taken
516/// away** from a role that already had it. The second deserves its own line in
517/// the log and its own number.
518#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
519pub struct ReconcileOutcome {
520    /// Grants created because a role was missing a permission it should hold.
521    pub granted: usize,
522    /// Grants removed because an ordinary tenant's role held an
523    /// organization-level action.
524    pub revoked: usize,
525}
526
527impl ReconcileOutcome {
528    /// Whether anything changed — the condition for logging at all.
529    pub const fn is_empty(&self) -> bool {
530        self.granted == 0 && self.revoked == 0
531    }
532}

534/// Reconcile the default roles' permission grants with the current permission
535/// set for a tenant: granting the permissions each role is MISSING, and
536/// revoking organization-level actions an ordinary tenant's roles should never
537/// have held.
538///
539/// [`seed_default_roles`] runs only at `/bootstrap`, which self-disables after
540/// the first admin is created. Server startup, by contrast, runs
541/// [`seed_permissions`] on every boot — so a permission ADDED to the registry
542/// after the initial bootstrap exists as a row but is granted to no role,
543/// causing spurious 403s for actions the super-admin/admin should be allowed
544/// to perform (e.g. a newly-added `oauth2_clients:create`). Calling this on
545/// startup back-fills those grants.
546///
547/// # Why this also revokes
548///
549/// Until B-04's follow-up, every tenant's `super-admin` was seeded with the
550/// *entire* permission registry, organization-level actions included. Those
551/// grants exist in every deployment bootstrapped before that change, and
552/// `seed_default_roles` no longer runs for them. A reconcile that could only
553/// ever grant would therefore leave the old, wrong state in place forever and
554/// the fix would apply to new tenants only.
555///
556/// Revocation is deliberately narrow. It touches **only**:
557///
558/// * the three seeded default roles, found by name — a role an operator
559///   created is theirs, and this function has no business editing it;
560/// * actions in
561///   [`ORGANIZATION_LEVEL_ACTIONS`](axiam_core::permission_scope::ORGANIZATION_LEVEL_ACTIONS);
562/// * tenants that are **not** the organization's own scope.
563///
564/// It is not a general "make the grants match the seeding rules" pass: an
565/// operator who deliberately granted `users:delete` to `viewer` keeps it. Only
566/// the specific escalation is undone.
567///
568/// # Idempotency
569///
570/// Existing correct `grants` edges are left untouched and only missing grants
571/// are created, so no duplicate edges accumulate across restarts. Once the
572/// organization-level grants are gone the revoke half finds nothing and returns
573/// zero, so a steady-state boot is a no-op. Tenants without the default roles
574/// (never bootstrapped) are skipped.
575pub async fn reconcile_default_role_grants<C: Connection>(
576    db: &Surreal<C>,
577    tenant_id: Uuid,
578) -> Result<ReconcileOutcome, DbError> {
579    let role_repo = SurrealRoleRepository::new(db.clone());
580    let perm_repo = SurrealPermissionRepository::new(db.clone());
581
582    let existing_roles = role_repo
583        .list(
584            tenant_id,
585            Pagination {
586                offset: 0,
587                limit: 1000,
588                search: None,
589            },
590        )
591        .await
592        .map_err(|e| DbError::Migration(format!("reconcile list roles failed: {e}")))?;
593    let find_role = |name: &str| {
594        existing_roles
595            .items
596            .iter()
597            .find(|r| r.name == name)
598            .map(|r| r.id)
599    };
600
601    // Only reconcile tenants bootstrapped with the three default roles.
602    let (Some(super_admin_id), Some(admin_id), Some(viewer_id)) = (
603        find_role("super-admin"),
604        find_role("admin"),
605        find_role("viewer"),
606    ) else {
607        return Ok(ReconcileOutcome::default());
608    };
609
610    let permissions = perm_repo
611        .list(
612            tenant_id,
613            Pagination {
614                offset: 0,
615                limit: 10_000,
616                search: None,
617            },
618        )
619        .await
620        .map_err(|e| DbError::Migration(format!("reconcile list permissions failed: {e}")))?;
621
622    let sa_have = granted_permission_ids(&perm_repo, tenant_id, super_admin_id).await?;
623    let admin_have = granted_permission_ids(&perm_repo, tenant_id, admin_id).await?;
624    let viewer_have = granted_permission_ids(&perm_repo, tenant_id, viewer_id).await?;
625
626    // B-04 defence in depth — see this function's doc comment. In an ordinary
627    // tenant the organization-level actions are neither granted below nor left
628    // in place if a previous version granted them.
629    let may_hold_org_actions = tenant_may_hold_organization_actions(db, tenant_id).await;
630
631    let mut outcome = ReconcileOutcome::default();
632    for perm in &permissions.items {
633        if !may_hold_org_actions && is_organization_level_action(&perm.action) {
634            // Withhold, and undo the grant if an earlier version made it.
635            // Only super-admin and admin are considered: viewer is seeded
636            // solely with `:list`/`:get` and no organization-level action is
637            // one, so a viewer holding one was not put there by this seeder.
638            for (role_id, have, role) in [
639                (super_admin_id, &sa_have, "super-admin"),
640                (admin_id, &admin_have, "admin"),
641            ] {
642                if have.contains(&perm.id) {
643                    perm_repo
644                        .revoke_from_role(tenant_id, role_id, perm.id)
645                        .await
646                        .map_err(|e| {
647                            DbError::Migration(format!(
648                                "reconcile revoke organization-level {} from {role} failed: {e}",
649                                perm.action
650                            ))
651                        })?;
652                    outcome.revoked += 1;
653                }
654            }
655            continue;
656        }
657
658        // super-admin gets every permission.
659        if !sa_have.contains(&perm.id) {
660            perm_repo
661                .grant_to_role(tenant_id, super_admin_id, perm.id)
662                .await
663                .map_err(|e| {
664                    DbError::Migration(format!("reconcile grant super-admin failed: {e}"))
665                })?;
666            outcome.granted += 1;
667        }
668        // admin gets every permission except admin:bootstrap.
669        if perm.action != "admin:bootstrap" && !admin_have.contains(&perm.id) {
670            perm_repo
671                .grant_to_role(tenant_id, admin_id, perm.id)
672                .await
673                .map_err(|e| DbError::Migration(format!("reconcile grant admin failed: {e}")))?;
674            outcome.granted += 1;
675        }
676        // viewer gets only :list and :get permissions.
677        if (perm.action.ends_with(":list") || perm.action.ends_with(":get"))
678            && !viewer_have.contains(&perm.id)
679        {
680            perm_repo
681                .grant_to_role(tenant_id, viewer_id, perm.id)
682                .await
683                .map_err(|e| DbError::Migration(format!("reconcile grant viewer failed: {e}")))?;
684            outcome.granted += 1;
685        }
686    }
687
688    Ok(outcome)
689}

731    /// where a concurrent caller creates the same (tenant_id, name) role

738
739        // Simulate "a concurrent caller already won": a role with this
740        // (tenant_id, name) already exists, but we call with existing_id =
741        // None (as if OUR pre-fetch ran before the winner's insert).
742        let winner = role_repo
743            .create(CreateRole {
744                tenant_id,

774        let perm_repo = SurrealPermissionRepository::new(db.clone());

776        let role = role_repo

778                tenant_id,
779                name: "grant-race-role".into(),
780                description: "role".into(),
781                is_global: true,
782            })
783            .await
784            .unwrap();
785        let perm = perm_repo

788                action: "widgets:list".into(),
789                description: "List widgets".into(),
