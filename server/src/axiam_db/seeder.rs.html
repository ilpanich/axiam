1//! Startup seeders for populating static data.
2//!
3//! The permission seeder uses raw SurrealQL `UPSERT` to ensure idempotency
4//! under concurrent startup (per D-07). A deterministic UUID derived from
5//! `namespace = tenant_id` + `name = action` ensures the same record is
6//! always targeted on subsequent restarts — true idempotency via UPSERT.
7//!
8//! CQ-B42: a sha256 hash of the registry is persisted in `seeder_state` per
9//! tenant. On startup, if the hash matches the stored value the UPSERT loop is
10//! skipped entirely, eliminating the O(n×95) boot storm on unchanged systems.

49/// Seed all permissions in `registry` for the given `tenant_id`.
50///
51/// # Idempotency
52///
53/// A deterministic UUID is generated for each `(tenant_id, action)` pair
54/// using [`Uuid::new_v5`] with `tenant_id` as the namespace.  The same pair
55/// always produces the same record ID, so the UPSERT targets the same row on
56/// every restart.
57///
58/// # SurrealQL
59///
60/// Uses raw `UPSERT` — NOT `list()` + conditional `create()`.  The hand-rolled
61/// pattern is race-prone under concurrent startup (per RESEARCH.md) and is
62/// explicitly prohibited by D-07.
63pub async fn seed_permissions<C: Connection>(
64    db: &Surreal<C>,
65    tenant_id: Uuid,
66    registry: &[(&str, &str)],
67) -> Result<(), DbError> {
68    // CQ-B42: compute sha256 hash of the registry to guard against UPSERT storm.
69    let registry_hash = {
70        let mut h = Sha256::new();
71        for (action, desc) in registry {
72            h.update(action.as_bytes());
73            h.update(b"|");
74            h.update(desc.as_bytes());
75        }
76        hex::encode(h.finalize())
77    };
78
79    // Derive a stable per-tenant record ID for seeder_state.
80    let state_id = Uuid::new_v5(&tenant_id, b"seeder_state").to_string();
81
82    // Check existing seeder_state; skip UPSERT loop if hash is unchanged.
83    let existing: Vec<SeederStateRow> = db
84        .query("SELECT hash FROM type::record('seeder_state', $id)")
85        .bind(("id", state_id.clone()))
86        .await
87        .map_err(|e| DbError::Migration(format!("seeder_state read failed: {e}")))?
88        .take(0)
89        .map_err(|e| DbError::Migration(format!("seeder_state take failed: {e}")))?;
90
91    if existing
92        .first()
93        .map(|r| r.hash.as_str())
94        .is_some_and(|h| h == registry_hash.as_str())
95    {
96        tracing::debug!(
97            %tenant_id,
98            "seed_permissions: registry hash unchanged — skipping UPSERT storm (CQ-B42)"
99        );
100        return Ok(());
101    }
102
103    // Two callers can seed the same tenant at once — two racing first-run
104    // bootstrap requests do exactly that — and they UPSERT the same rows. The
105    // engine aborts the loser's write with a retryable write conflict, which
106    // must not surface as a failed seed: both statements below are keyed
107    // UPSERTs of the same values, so replaying one cannot double-apply.
108    for (action, description) in registry {
109        // Deterministic UUID: same tenant + action always produces same ID.
110        let id = Uuid::new_v5(&tenant_id, action.as_bytes());
111
112        crate::helpers::retry_on_write_conflict(|| async {
113            db.query(
114                "UPSERT type::record('permission', $id) SET \
115                 tenant_id = $tenant_id, \
116                 action = $action, \
117                 description = $description, \
118                 created_at = IF (SELECT created_at FROM type::record('permission', $id))[0].created_at \
119                   THEN (SELECT created_at FROM type::record('permission', $id))[0].created_at \
120                   ELSE time::now() END, \
121                 updated_at = time::now()",
122            )
123            .bind(("id", id.to_string()))
124            .bind(("tenant_id", tenant_id.to_string()))
125            .bind(("action", action.to_string()))
126            .bind(("description", description.to_string()))
127            .await
128            .map_err(|e| DbError::Migration(format!("seed_permissions UPSERT failed: {e}")))?
129            .check()
130            .map_err(|e| DbError::Migration(format!("seed_permissions UPSERT check failed: {e}")))?;
131            Ok::<(), DbError>(())
132        })
133        .await?;
134    }
135
136    // Persist the new hash so subsequent restarts can skip this tenant.
137    crate::helpers::retry_on_write_conflict(|| async {
138        db.query(
139            "UPSERT type::record('seeder_state', $id) SET \
140             tenant_id = $tenant_id, hash = $hash, updated_at = time::now()",
141        )
142        .bind(("id", state_id.clone()))
143        .bind(("tenant_id", tenant_id.to_string()))
144        .bind(("hash", registry_hash.clone()))
145        .await
146        .map_err(|e| DbError::Migration(format!("seeder_state upsert failed: {e}")))?
147        .check()
148        .map_err(|e| DbError::Migration(format!("seeder_state upsert check: {e}")))?;
149        Ok::<(), DbError>(())
150    })
151    .await?;
152
153    Ok(())
154}

156/// Mint a one-time first-run bootstrap setup token if the database has
157/// never been bootstrapped (SECHRD-04 / D-03b).
158///
159/// # Behavior
160///
161/// - On a fresh (never-bootstrapped) database — no `bootstrap_setup_token`
162///   row yet AND no `user` row anywhere — a cryptographically random token
163///   is generated, its sha256 hash is persisted as the
164///   `bootstrap_setup_token` record ID, and the returned `Some(token)`
165///   carries the PLAINTEXT token so the caller can log it exactly once
166///   (the one deliberate secret-log exception, per D-03b). Only the HASH
167///   is ever persisted to the database.
168/// - On any subsequent boot — a setup token already exists, or at least
169///   one user already exists — this is a no-op and returns `Ok(None)`;
170///   nothing is minted or re-logged.
171pub async fn mint_bootstrap_setup_token_if_needed<C: Connection>(
172    db: &Surreal<C>,
173) -> Result<Option<String>, DbError> {
174    // A setup token was already minted — no-op.
175    let existing_tokens: Vec<CountRow> = db
176        .query("SELECT count() AS total FROM bootstrap_setup_token GROUP ALL")
177        .await
178        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token count failed: {e}")))?
179        .take(0)
180        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token count take failed: {e}")))?;
181    if existing_tokens.first().map(|r| r.total).unwrap_or(0) > 0 {
182        return Ok(None);
183    }
184
185    // Bootstrap already completed somewhere — no-op.
186    let existing_users: Vec<CountRow> = db
187        .query("SELECT count() AS total FROM user GROUP ALL")
188        .await
189        .map_err(|e| DbError::Migration(format!("user count failed: {e}")))?
190        .take(0)
191        .map_err(|e| DbError::Migration(format!("user count take failed: {e}")))?;
192    if existing_users.first().map(|r| r.total).unwrap_or(0) > 0 {
193        return Ok(None);
194    }
195
196    mint_setup_token(db).await.map(Some)
197}

199/// Why a re-mint was refused, or the token it produced.
200///
201/// Three outcomes rather than an error, because two of them are not failures:
202/// they are the security argument. A setup token is only ever re-mintable on a
203/// deployment that has **no administrator yet** — which is exactly the state an
204/// operator who lost the token at first boot is stuck in. Once anybody has
205/// bootstrapped, the token is a second way in and re-minting it would be a
206/// credential reset with no authentication in front of it.
207#[derive(Debug, Clone, PartialEq, Eq)]
208pub enum SetupTokenRemint {
209    /// A fresh token. The **plaintext**, returned once; only its hash is stored.
210    Minted(String),
211    /// At least one `user` row exists: this deployment has an administrator,
212    /// who can create others through the authenticated API.
213    RefusedUserExists,
214    /// A setup token was already redeemed. Bootstrap happened; whatever it
215    /// produced is the way in.
216    RefusedTokenConsumed,
217}

219/// Replace the bootstrap setup token, on a deployment nobody has bootstrapped.
220///
221/// The recovery path for DF-019: before this existed, an operator who lost the
222/// first-boot token had no way back except wiping the volume, because the token
223/// is stored only as a hash and the boot-time mint is a no-op once a row
224/// exists.
225///
226/// The gate is the whole security argument — see [`SetupTokenRemint`]. Both
227/// refusals are checked **before** anything is deleted, so a refused call
228/// leaves the existing token working.
229///
230/// The plaintext is returned, never logged: the caller prints it to stdout so
231/// it does not land in the container log a second time.
232pub async fn remint_bootstrap_setup_token<C: Connection>(
233    db: &Surreal<C>,
234) -> Result<SetupTokenRemint, DbError> {
235    let existing_users: Vec<CountRow> = db
236        .query("SELECT count() AS total FROM user GROUP ALL")
237        .await
238        .map_err(|e| DbError::Migration(format!("user count failed: {e}")))?
239        .take(0)
240        .map_err(|e| DbError::Migration(format!("user count take failed: {e}")))?;
241    if existing_users.first().map(|r| r.total).unwrap_or(0) > 0 {
242        return Ok(SetupTokenRemint::RefusedUserExists);
243    }
244
245    let consumed: Vec<CountRow> = db
246        .query("SELECT count() AS total FROM bootstrap_setup_token_consumed GROUP ALL")
247        .await
248        .map_err(|e| {
249            DbError::Migration(format!("bootstrap_setup_token_consumed count failed: {e}"))
250        })?
251        .take(0)
252        .map_err(|e| {
253            DbError::Migration(format!(
254                "bootstrap_setup_token_consumed count take failed: {e}"
255            ))
256        })?;
257    if consumed.first().map(|r| r.total).unwrap_or(0) > 0 {
258        return Ok(SetupTokenRemint::RefusedTokenConsumed);
259    }
260
261    // Delete then mint, in that order and only after both gates: the table
262    // holds at most one usable token, so leaving the old hash behind would
263    // give the deployment two valid tokens rather than a replacement.
264    db.query("DELETE bootstrap_setup_token")
265        .await
266        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token delete failed: {e}")))?
267        .check()
268        .map_err(|e| {
269            DbError::Migration(format!("bootstrap_setup_token delete check failed: {e}"))
270        })?;
271
272    mint_setup_token(db).await.map(SetupTokenRemint::Minted)
273}

275/// Generate a token, store its hash, and hand back the plaintext.
276///
277/// One function, shared by the first-boot mint and the re-mint, so the two
278/// cannot drift into producing differently-shaped tokens or storing them
279/// differently.
280async fn mint_setup_token<C: Connection>(db: &Surreal<C>) -> Result<String, DbError> {
281    // A cryptographically random 32-byte token, base64url-encoded (same shape
282    // as `axiam_auth::token::generate_refresh_token`).
283    let mut rng = rand::rng();
284    let bytes: [u8; 32] = rand::RngExt::random(&mut rng);
285    let token = URL_SAFE_NO_PAD.encode(bytes);
286
287    let mut hasher = Sha256::new();
288    hasher.update(token.as_bytes());
289    let token_hash = hex::encode(hasher.finalize());
290
291    db.query("CREATE type::record('bootstrap_setup_token', $hash) SET created_at = time::now()")
292        .bind(("hash", token_hash))
293        .await
294        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token mint failed: {e}")))?
295        .check()
296        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token mint check failed: {e}")))?;
297
298    Ok(token)
299}

301/// Whether `tenant_id` is the organization's own scope — the only tenant whose
302/// seeded roles may hold organization-level actions — or `None` when the tenant
303/// cannot be resolved.
304///
305/// The callers fail closed in opposite directions, so the unresolved case is
306/// theirs to settle: the organization-level actions are withheld from a tenant
307/// that is not known to be the organization scope (being wrong costs an
308/// operator a grant they can add back by hand; the other way seeds the
309/// escalation the filter exists to prevent), and the `network:` permissions
310/// are withheld from one that is not known to be an ordinary tenant (#702).
311async fn tenant_is_organization_scope<C: Connection>(
312    db: &Surreal<C>,
313    tenant_id: Uuid,
314) -> Option<bool> {
315    SurrealTenantRepository::new(db.clone())
316        .get_by_id(tenant_id)
317        .await
318        .ok()
319        .map(|t| t.is_organization_scope())
320}

322/// Seed the three default roles — super-admin, admin, and viewer — for the
323/// given tenant, assigning permissions according to D-12.
324///
325/// # Idempotency
326///
327/// If all three roles already exist the function returns their IDs without
328/// creating duplicates. If only some roles are missing they are created.
329///
330/// # Permission assignment rules
331///
332/// - **super-admin**: every permission in the tenant.
333/// - **admin**: every permission except `admin:bootstrap`.
334/// - **viewer**: permissions whose action ends with `:list` or `:get`.
335///
336/// # Organization-level actions
337///
338/// In an ordinary (`TenantKind::Standard`) tenant, the actions listed in
339/// [`ORGANIZATION_LEVEL_ACTIONS`](axiam_core::permission_scope::ORGANIZATION_LEVEL_ACTIONS)
340/// are withheld from super-admin and admin. A tenant administrator holding
341/// `ca_certificates:manage` could flag an organization-wide mTLS trust anchor
342/// and then mint certificates authenticating as principals in sibling tenants
343/// (B-04) — so the grant is not made in the first place. The organization's
344/// own scope is seeded with everything.
345///
346/// # The reserved `network:` namespace (RADIUS Live profile, W6.2)
347///
348/// The two global administrative roles are **not** given the whole `network:`
349/// namespace. `super-admin` gets
350/// [`SEEDED_SUPER_ADMIN_NETWORK_ACTIONS`] — `network:read` and the five meta
351/// permissions, never `network:join` nor the administrative class — and
352/// `admin` gets none. Design note §4.2 rule 1 forbids a network-sensitive role
353/// to be global; the seeded super-admin is the one exemption, because under the
354/// §5.3 delegation bound a network administrator can only be made by someone
355/// who already holds `network:grant` and every meta permission conferred, and a
356/// tenant whose bootstrap role held none of them could never administer its
357/// network. That constant's documentation gives the exemption's limits; the API
358/// refuses a `network:` grant to any global role, this one included.
359///
360/// This is the second of two layers. `require_organization_principal` in
361/// `axiam-api-rest` is the load-bearing one and refuses such a caller whatever
362/// grants it holds; withholding the grant means the two layers agree instead of
363/// one quietly contradicting the other. The `viewer` role is untouched — every
364/// organization-level action is a mutation and viewer is seeded only with
365/// `:list` and `:get`.
366pub async fn seed_default_roles<C: Connection>(
367    db: &Surreal<C>,
368    tenant_id: Uuid,
369    _permission_registry: &[(&str, &str)],
370) -> Result<SeedRolesResult, DbError> {
371    let role_repo = SurrealRoleRepository::new(db.clone());
372    let perm_repo = SurrealPermissionRepository::new(db.clone());
373
374    // -----------------------------------------------------------------------
375    // 1. Ensure all three roles exist (idempotent)
376    // -----------------------------------------------------------------------
377
378    // Fetch existing roles for this tenant in a single list call.
379    let existing = role_repo
380        .list(
381            tenant_id,
382            Pagination {
383                offset: 0,
384                limit: 1000,
385                search: None,
386            },
387        )
388        .await
389        .map_err(|e| DbError::Migration(format!("seed_default_roles list roles failed: {e}")))?;
390
391    let find_role = |name: &str| existing.items.iter().find(|r| r.name == name).map(|r| r.id);
392
393    // SECHRD-04: role creation must tolerate a CONCURRENT racer creating the
394    // same default role for the same tenant — `find_or_create_role` treats
395    // the `idx_role_tenant_name` UNIQUE-index violation as "a concurrent
396    // caller already won" and re-fetches the winner's row instead of
397    // failing (see its doc comment for why this is now reachable).
398    let super_admin_id = find_or_create_role(
399        &role_repo,
400        tenant_id,
401        find_role(SEEDED_SUPER_ADMIN_ROLE),
402        SEEDED_SUPER_ADMIN_ROLE,
403        "Full system access — all permissions",
404    )
405    .await?;
406
407    let admin_id = find_or_create_role(
408        &role_repo,
409        tenant_id,
410        find_role("admin"),
411        "admin",
412        "Administrative access — all entity CRUD operations",
413    )
414    .await?;
415
416    let viewer_id = find_or_create_role(
417        &role_repo,
418        tenant_id,
419        find_role("viewer"),
420        "viewer",
421        "Read-only access — list and get operations",
422    )
423    .await?;
424
425    // -----------------------------------------------------------------------
426    // 2. Fetch all permissions for this tenant
427    // -----------------------------------------------------------------------
428
429    let permissions = perm_repo
430        .list(
431            tenant_id,
432            Pagination {
433                offset: 0,
434                limit: 10_000,
435                search: None,
436            },
437        )
438        .await
439        .map_err(|e| {
440            DbError::Migration(format!("seed_default_roles list permissions failed: {e}"))
441        })?;
442
443    // -----------------------------------------------------------------------
444    // 3. Grant permissions to roles (idempotent — skip already-granted pairs)
445    // -----------------------------------------------------------------------
446    //
447    // SECHRD-04: `grant_to_role` CREATEs a `grants` edge guarded by a UNIQUE
448    // (in, out) index (CQ-B17) — re-granting an already-granted permission
449    // is a hard error, not a silent no-op. Before this fix, `seed_default_roles`
450    // was only ever called once per tenant (the old bootstrap TOCTOU check
451    // disabled the endpoint before a second call could reach here). Now that
452    // bootstrap atomicity is enforced by the `bootstrap_lock` uniqueness
453    // invariant instead, this function can be reached again — including
454    // CONCURRENTLY, by two racing bootstrap requests — for a tenant whose
455    // roles+grants already exist. The pre-fetch below skips already-granted
456    // pairs in the common (sequential retry) case; `grant_to_role_idempotent`
457    // additionally tolerates the residual TOCTOU window between this
458    // pre-fetch and the CREATE, for true concurrent callers.
459
460    let super_admin_have = granted_permission_ids(&perm_repo, tenant_id, super_admin_id).await?;
461    let admin_have = granted_permission_ids(&perm_repo, tenant_id, admin_id).await?;
462    let viewer_have = granted_permission_ids(&perm_repo, tenant_id, viewer_id).await?;
463
464    // B-04 defence in depth: an ordinary tenant's roles are not seeded with
465    // organization-level actions. See this function's doc comment.
466    let organization_scope = tenant_is_organization_scope(db, tenant_id).await;
467    let may_hold_org_actions = organization_scope == Some(true);
468    let network_super_admin = organization_scope == Some(false);
469    let withheld = |action: &str| !may_hold_org_actions && is_organization_level_action(action);
470
471    for perm in &permissions.items {
472        if withheld(&perm.action) {
473            continue;
474        }
475
476        // super-admin gets ALL permissions, except the parts of the reserved
477        // `network:` namespace a bootstrap role must not carry (see above).
478        if super_admin_may_hold(&perm.action, network_super_admin)
479            && !super_admin_have.contains(&perm.id)
480        {
481            grant_to_role_idempotent(&perm_repo, tenant_id, super_admin_id, perm.id)
482                .await
483                .map_err(|e| {
484                    DbError::Migration(format!(
485                        "seed_default_roles grant super-admin permission {} failed: {e}",
486                        perm.action
487                    ))
488                })?;
489        }
490
491        // admin gets every permission EXCEPT admin:bootstrap and the reserved
492        // `network:` namespace.
493        if admin_may_hold(&perm.action) && !admin_have.contains(&perm.id) {
494            grant_to_role_idempotent(&perm_repo, tenant_id, admin_id, perm.id)
495                .await
496                .map_err(|e| {
497                    DbError::Migration(format!(
498                        "seed_default_roles grant admin permission {} failed: {e}",
499                        perm.action
500                    ))
501                })?;
502        }
503
504        // viewer gets only :list and :get permissions.
505        if (perm.action.ends_with(":list") || perm.action.ends_with(":get"))
506            && !viewer_have.contains(&perm.id)
507        {
508            grant_to_role_idempotent(&perm_repo, tenant_id, viewer_id, perm.id)
509                .await
510                .map_err(|e| {
511                    DbError::Migration(format!(
512                        "seed_default_roles grant viewer permission {} failed: {e}",
513                        perm.action
514                    ))
515                })?;
516        }
517    }
518
519    Ok(SeedRolesResult {
520        super_admin_role_id: super_admin_id,
521        admin_role_id: admin_id,
522        viewer_role_id: viewer_id,
523    })
524}

526/// Whether the seeded `super-admin` may hold `action`: everything outside the
527/// reserved `network:` namespace, and inside it [`SEEDED_SUPER_ADMIN_NETWORK_ACTIONS`]
528/// in an ordinary tenant only.
529///
530/// `ordinary_tenant` is false for the organization's own scope, whose
531/// super-admin is global and so reaches every tenant of the organization: the
532/// exemption would carry the network meta permissions across tenants (#702,
533/// T-483), which the design note's §7 rules out. It is also false for a tenant
534/// that could not be resolved, so the question fails closed.
535fn super_admin_may_hold(action: &str, ordinary_tenant: bool) -> bool {
536    !is_network_action(action)
537        || (ordinary_tenant && SEEDED_SUPER_ADMIN_NETWORK_ACTIONS.contains(&action))
538}

540/// Whether the seeded `admin` may hold `action`: everything but
541/// `admin:bootstrap` and the reserved `network:` namespace. A tenant
542/// administrator is not a network administrator; the super-admin makes one
543/// (design note §5.3).
544fn admin_may_hold(action: &str) -> bool {
545    action != "admin:bootstrap" && !is_network_action(action)
546}

548/// Find an existing role by name, or create it — tolerating a CONCURRENT
549/// caller creating the SAME (tenant_id, name) role at the same time.
550///
551/// `role_repo.create()` CREATEs a fresh-UUID row guarded by the
552/// `idx_role_tenant_name` UNIQUE index; two racing bootstrap requests can
553/// both observe "role missing" and both attempt a create. The loser's
554/// UNIQUE-index violation is NOT a real failure here — a concurrent caller
555/// already won and the row exists — so it is treated as "re-fetch and use
556/// the winner's ID" rather than propagated (SECHRD-04: `seed_default_roles`
557/// is reachable from two concurrent bootstrap requests now that atomicity
558/// is enforced by `bootstrap_lock`, not the old TOCTOU check).
559async fn find_or_create_role<C: Connection>(
560    role_repo: &SurrealRoleRepository<C>,
561    tenant_id: Uuid,
562    existing_id: Option<Uuid>,
563    name: &str,
564    description: &str,
565) -> Result<Uuid, DbError> {
566    if let Some(id) = existing_id {
567        return Ok(id);
568    }
569
570    match role_repo
571        .create(CreateRole {
572            tenant_id,
573            name: name.to_string(),
574            description: description.to_string(),
575            is_global: true,
576        })
577        .await
578    {
579        Ok(role) => Ok(role.id),
580        Err(e) => {
581            let msg = e.to_string();
582            match classify_write_error(&msg, "role") {
583                DbError::AlreadyExists { .. } => {
584                    let refreshed = role_repo
585                        .list(
586                            tenant_id,
587                            Pagination {
588                                offset: 0,
589                                limit: 1000,
590                                search: None,
591                            },
592                        )
593                        .await
594                        .map_err(|e| {
595                            DbError::Migration(format!(
596                                "find_or_create_role re-list '{name}' failed: {e}"
597                            ))
598                        })?;
599                    refreshed
600                        .items
601                        .into_iter()
602                        .find(|r| r.name == name)
603                        .map(|r| r.id)
604                        .ok_or_else(|| {
605                            DbError::Migration(format!(
606                                "find_or_create_role: role '{name}' vanished after concurrent create race"
607                            ))
608                        })
609                }
610                _ => Err(DbError::Migration(format!(
611                    "find_or_create_role '{name}' failed: {msg}"
612                ))),
613            }
614        }
615    }
616}

618/// The set of permission IDs already granted to `role_id` (tenant-scoped).
619///
620/// Shared by [`seed_default_roles`] and [`reconcile_default_role_grants`] so
621/// neither re-attempts a `grant_to_role` CREATE for a pair that already
622/// exists — the `grants` edge table's UNIQUE (in, out) index (CQ-B17) makes
623/// a duplicate grant a hard error, not a no-op.
624async fn granted_permission_ids<C: Connection>(
625    perm_repo: &SurrealPermissionRepository<C>,
626    tenant_id: Uuid,
627    role_id: Uuid,
628) -> Result<std::collections::HashSet<Uuid>, DbError> {
629    perm_repo
630        .get_role_permissions(tenant_id, role_id)
631        .await
632        .map(|perms| perms.into_iter().map(|p| p.id).collect())
633        .map_err(|e| DbError::Migration(format!("granted_permission_ids failed: {e}")))
634}

636/// Grant `permission_id` to `role_id`, treating "already granted" (a
637/// concurrent caller won the same grant) as success rather than an error.
638///
639/// Closes the residual TOCTOU between [`granted_permission_ids`]'s pre-fetch
640/// and this CREATE for two truly concurrent callers (SECHRD-04).
641async fn grant_to_role_idempotent<C: Connection>(
642    perm_repo: &SurrealPermissionRepository<C>,
643    tenant_id: Uuid,
644    role_id: Uuid,
645    permission_id: Uuid,
646) -> Result<(), DbError> {
647    match perm_repo
648        .grant_to_role(tenant_id, role_id, permission_id)
649        .await
650    {
651        Ok(()) => Ok(()),
652        Err(e) => match classify_write_error(e.to_string(), "permission_grant") {
653            DbError::AlreadyExists { .. } => Ok(()),
654            other => Err(other),
655        },
656    }
657}

659/// Record one `network:` revocation of the boot reconcile as an audit row.
660///
661/// A capability taken away from a role is something a reviewer must be able to
662/// find afterwards, and the boot log line is not durable. The actor is the
663/// system; the row names the role, the permission and the action. A failed
664/// write is logged and does not stop the boot: the revocation has already
665/// happened, and refusing to start would leave the escalation in place.
666async fn audit_network_revocation<C: Connection>(
667    db: &Surreal<C>,
668    tenant_id: Uuid,
669    role_id: Uuid,
670    role: &str,
671    action: &str,
672    permission_id: Uuid,
673) {
674    let entry = CreateAuditLogEntry {
675        tenant_id,
676        actor_id: Uuid::nil(),
677        actor_type: ActorType::System,
678        action: "network_role.permission_revoked_by_reconcile".to_owned(),
679        resource_id: Some(role_id),
680        outcome: AuditOutcome::Success,
681        ip_address: None,
682        metadata: Some(serde_json::json!({
683            "role": role,
684            "permission": action,
685            "permission_id": permission_id.to_string(),
686            "reason": "the seeded role may not hold this network: permission in this tenant",
687        })),
688    };
689    if let Err(e) = SurrealAuditLogRepository::new(db.clone())
690        .append(entry)
691        .await
692    {
693        tracing::error!(
694            %tenant_id,
695            %role_id,
696            action,
697            error = %e,
698            "audit write for a network: permission revocation failed (non-fatal)"
699        );
700    }
701}

703/// What [`reconcile_default_role_grants`] changed for one tenant.
704///
705/// Two counters rather than one, because they mean opposite things to an
706/// operator reading the boot log: `granted` is a back-fill catching a role up
707/// with a registry that grew, while `revoked` is a **capability being taken
708/// away** from a role that already had it. The second deserves its own line in
709/// the log and its own number.
710#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
711pub struct ReconcileOutcome {
712    /// Grants created because a role was missing a permission it should hold.
713    pub granted: usize,
714    /// Grants removed because an ordinary tenant's role held an
715    /// organization-level action.
716    pub revoked: usize,
717}
718
719impl ReconcileOutcome {
720    /// Whether anything changed — the condition for logging at all.
721    pub const fn is_empty(&self) -> bool {
722        self.granted == 0 && self.revoked == 0
723    }
724}
725
726/// Reconcile the default roles' permission grants with the current permission
727/// set for a tenant: granting the permissions each role is MISSING, and
728/// revoking organization-level actions an ordinary tenant's roles should never
729/// have held.
730///
731/// [`seed_default_roles`] runs only at `/bootstrap`, which self-disables after
732/// the first admin is created. Server startup, by contrast, runs
733/// [`seed_permissions`] on every boot — so a permission ADDED to the registry
734/// after the initial bootstrap exists as a row but is granted to no role,
735/// causing spurious 403s for actions the super-admin/admin should be allowed
736/// to perform (e.g. a newly-added `oauth2_clients:create`). Calling this on
737/// startup back-fills those grants.
738///
739/// # Why this also revokes
740///
741/// Until B-04's follow-up, every tenant's `super-admin` was seeded with the
742/// *entire* permission registry, organization-level actions included. Those
743/// grants exist in every deployment bootstrapped before that change, and
744/// `seed_default_roles` no longer runs for them. A reconcile that could only
745/// ever grant would therefore leave the old, wrong state in place forever and
746/// the fix would apply to new tenants only.
747///
748/// Revocation is deliberately narrow. It touches **only**:
749///
750/// * the three seeded default roles, found by name — a role an operator
751///   created is theirs, and this function has no business editing it;
752/// * actions in
753///   [`ORGANIZATION_LEVEL_ACTIONS`](axiam_core::permission_scope::ORGANIZATION_LEVEL_ACTIONS),
754///   in tenants that are **not** the organization's own scope;
755/// * actions of the reserved `network:` namespace that
756///   [`seed_default_roles`] does not give the role (every one for `admin`,
757///   `network:join` and the administrative class for `super-admin`), in every
758///   tenant; and in the organization's own scope, where the super-admin is
759///   global and would carry them into every tenant of the organization, all of
760///   them (#702, T-483). One audit row is written per revocation.
761///
762/// It is not a general "make the grants match the seeding rules" pass: an
763/// operator who deliberately granted `users:delete` to `viewer` keeps it. Only
764/// the specific escalation is undone.
765///
766/// # Idempotency
767///
768/// Existing correct `grants` edges are left untouched and only missing grants
769/// are created, so no duplicate edges accumulate across restarts. Once the
770/// organization-level grants are gone the revoke half finds nothing and returns
771/// zero, so a steady-state boot is a no-op. Tenants without the default roles
772/// (never bootstrapped) are skipped.
773pub async fn reconcile_default_role_grants<C: Connection>(
774    db: &Surreal<C>,
775    tenant_id: Uuid,
776) -> Result<ReconcileOutcome, DbError> {
777    let role_repo = SurrealRoleRepository::new(db.clone());
778    let perm_repo = SurrealPermissionRepository::new(db.clone());
779
780    let existing_roles = role_repo
781        .list(
782            tenant_id,
783            Pagination {
784                offset: 0,
785                limit: 1000,
786                search: None,
787            },
788        )
789        .await
790        .map_err(|e| DbError::Migration(format!("reconcile list roles failed: {e}")))?;
791    let find_role = |name: &str| {
792        existing_roles
793            .items
794            .iter()
795            .find(|r| r.name == name)
796            .map(|r| r.id)
797    };
798
799    // Only reconcile tenants bootstrapped with the three default roles.
800    let (Some(super_admin_id), Some(admin_id), Some(viewer_id)) = (
801        find_role(SEEDED_SUPER_ADMIN_ROLE),
802        find_role("admin"),
803        find_role("viewer"),
804    ) else {
805        return Ok(ReconcileOutcome::default());
806    };
807
808    let permissions = perm_repo
809        .list(
810            tenant_id,
811            Pagination {
812                offset: 0,
813                limit: 10_000,
814                search: None,
815            },
816        )
817        .await
818        .map_err(|e| DbError::Migration(format!("reconcile list permissions failed: {e}")))?;
819
820    let sa_have = granted_permission_ids(&perm_repo, tenant_id, super_admin_id).await?;
821    let admin_have = granted_permission_ids(&perm_repo, tenant_id, admin_id).await?;
822    let viewer_have = granted_permission_ids(&perm_repo, tenant_id, viewer_id).await?;
823
824    // B-04 defence in depth — see this function's doc comment. In an ordinary
825    // tenant the organization-level actions are neither granted below nor left
826    // in place if a previous version granted them.
827    let organization_scope = tenant_is_organization_scope(db, tenant_id).await;
828    let may_hold_org_actions = organization_scope == Some(true);
829    let network_super_admin = organization_scope == Some(false);
830
831    let mut outcome = ReconcileOutcome::default();
832    for perm in &permissions.items {
833        // The reserved `network:` namespace: grant what the seeded roles may
834        // hold, and take back what an earlier build of this branch granted them
835        // (W6.1 gave both roles the whole namespace). Only the two seeded
836        // administrative roles are touched, as for the organization-level
837        // actions below.
838        if is_network_action(&perm.action)
839            && (!super_admin_may_hold(&perm.action, network_super_admin)
840                || !admin_may_hold(&perm.action))
841        {
842            for (role_id, have, role, may) in [
843                (
844                    super_admin_id,
845                    &sa_have,
846                    "super-admin",
847                    super_admin_may_hold(&perm.action, network_super_admin),
848                ),
849                (admin_id, &admin_have, "admin", admin_may_hold(&perm.action)),
850            ] {
851                if !may && have.contains(&perm.id) {
852                    perm_repo
853                        .revoke_from_role(tenant_id, role_id, perm.id)
854                        .await
855                        .map_err(|e| {
856                            DbError::Migration(format!(
857                                "reconcile revoke reserved {} from {role} failed: {e}",
858                                perm.action
859                            ))
860                        })?;
861                    outcome.revoked += 1;
862                    audit_network_revocation(db, tenant_id, role_id, role, &perm.action, perm.id)
863                        .await;
864                }
865            }
866            if super_admin_may_hold(&perm.action, network_super_admin)
867                && !sa_have.contains(&perm.id)
868            {
869                // §4.2 rule 2 binds the reconcile too (R110W6-04): the
870                // grant's own transaction refuses a `network:` permission to
871                // a super-admin a group holds — which a release before 1.1.0
872                // allowed — and a group assignment racing it. Withheld then,
873                // and said so, rather than failing the boot: the tenant keeps
874                // working, and its super-admin simply does not carry the
875                // network meta permissions until no group holds it (the next
876                // boot's reconcile grants them).
877                match perm_repo
878                    .grant_to_role(tenant_id, super_admin_id, perm.id)
879                    .await
880                {
881                    Ok(()) => outcome.granted += 1,
882                    Err(AxiamError::Network(NetworkConflict::StaleRole { what_landed })) => {
883                        tracing::warn!(
884                            %tenant_id,
885                            action = %perm.action,
886                            reason = %what_landed,
887                            "network permission withheld from super-admin: a group holds the \
888                             role, and a role carrying a network: permission is never group-held \
889                             (design note §4.2 rule 2); unassign the role from the group to let \
890                             the next boot grant it"
891                        );
892                    }
893                    Err(e) => {
894                        return Err(DbError::Migration(format!(
895                            "reconcile grant super-admin failed: {e}"
896                        )));
897                    }
898                }
899            }
900            continue;
901        }
902        if !may_hold_org_actions && is_organization_level_action(&perm.action) {
903            // Withhold, and undo the grant if an earlier version made it.
904            // Only super-admin and admin are considered: viewer is seeded
905            // solely with `:list`/`:get` and no organization-level action is
906            // one, so a viewer holding one was not put there by this seeder.
907            for (role_id, have, role) in [
908                (super_admin_id, &sa_have, "super-admin"),
909                (admin_id, &admin_have, "admin"),
910            ] {
911                if have.contains(&perm.id) {
912                    perm_repo
913                        .revoke_from_role(tenant_id, role_id, perm.id)
914                        .await
915                        .map_err(|e| {
916                            DbError::Migration(format!(
917                                "reconcile revoke organization-level {} from {role} failed: {e}",
918                                perm.action
919                            ))
920                        })?;
921                    outcome.revoked += 1;
922                }
923            }
924            continue;
925        }
926
927        // super-admin gets every permission.
928        if !sa_have.contains(&perm.id) {
929            perm_repo
930                .grant_to_role(tenant_id, super_admin_id, perm.id)
931                .await
932                .map_err(|e| {
933                    DbError::Migration(format!("reconcile grant super-admin failed: {e}"))
934                })?;
935            outcome.granted += 1;
936        }
937        // admin gets every permission except admin:bootstrap.
938        if perm.action != "admin:bootstrap" && !admin_have.contains(&perm.id) {
939            perm_repo
940                .grant_to_role(tenant_id, admin_id, perm.id)
941                .await
942                .map_err(|e| DbError::Migration(format!("reconcile grant admin failed: {e}")))?;
943            outcome.granted += 1;
944        }
945        // viewer gets only :list and :get permissions.
946        if (perm.action.ends_with(":list") || perm.action.ends_with(":get"))
947            && !viewer_have.contains(&perm.id)
948        {
949            perm_repo
950                .grant_to_role(tenant_id, viewer_id, perm.id)
951                .await
952                .map_err(|e| DbError::Migration(format!("reconcile grant viewer failed: {e}")))?;
953            outcome.granted += 1;
954        }
955    }
956
957    Ok(outcome)
958}
