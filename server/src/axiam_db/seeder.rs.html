1//! Startup seeders for populating static data.
2//!
3//! The permission seeder uses raw SurrealQL `UPSERT` to ensure idempotency
4//! under concurrent startup (per D-07). A deterministic UUID derived from
5//! `namespace = tenant_id` + `name = action` ensures the same record is
6//! always targeted on subsequent restarts — true idempotency via UPSERT.
7//!
8//! CQ-B42: a sha256 hash of the registry is persisted in `seeder_state` per
9//! tenant. On startup, if the hash matches the stored value the UPSERT loop is
10//! skipped entirely, eliminating the O(n×95) boot storm on unchanged systems.

49/// Seed all permissions in `registry` for the given `tenant_id`.
50///
51/// # Idempotency
52///
53/// A deterministic UUID is generated for each `(tenant_id, action)` pair
54/// using [`Uuid::new_v5`] with `tenant_id` as the namespace.  The same pair
55/// always produces the same record ID, so the UPSERT targets the same row on
56/// every restart.
57///
58/// # SurrealQL
59///
60/// Uses raw `UPSERT` — NOT `list()` + conditional `create()`.  The hand-rolled
61/// pattern is race-prone under concurrent startup (per RESEARCH.md) and is
62/// explicitly prohibited by D-07.
63pub async fn seed_permissions<C: Connection>(
64    db: &Surreal<C>,
65    tenant_id: Uuid,
66    registry: &[(&str, &str)],
67) -> Result<(), DbError> {
68    // CQ-B42: compute sha256 hash of the registry to guard against UPSERT storm.
69    let registry_hash = {
70        let mut h = Sha256::new();
71        for (action, desc) in registry {
72            h.update(action.as_bytes());
73            h.update(b"|");
74            h.update(desc.as_bytes());
75        }
76        hex::encode(h.finalize())
77    };
78
79    // Derive a stable per-tenant record ID for seeder_state.
80    let state_id = Uuid::new_v5(&tenant_id, b"seeder_state").to_string();
81
82    // Check existing seeder_state; skip UPSERT loop if hash is unchanged.
83    let existing: Vec<SeederStateRow> = db
84        .query("SELECT hash FROM type::record('seeder_state', $id)")
85        .bind(("id", state_id.clone()))
86        .await
87        .map_err(|e| DbError::Migration(format!("seeder_state read failed: {e}")))?
88        .take(0)
89        .map_err(|e| DbError::Migration(format!("seeder_state take failed: {e}")))?;
90
91    if existing
92        .first()
93        .map(|r| r.hash.as_str())
94        .is_some_and(|h| h == registry_hash.as_str())
95    {
96        tracing::debug!(
97            %tenant_id,
98            "seed_permissions: registry hash unchanged — skipping UPSERT storm (CQ-B42)"
99        );
100        return Ok(());
101    }
102
103    // Two callers can seed the same tenant at once — two racing first-run
104    // bootstrap requests do exactly that — and they UPSERT the same rows. The
105    // engine aborts the loser's write with a retryable write conflict, which
106    // must not surface as a failed seed: both statements below are keyed
107    // UPSERTs of the same values, so replaying one cannot double-apply.
108    for (action, description) in registry {
109        // Deterministic UUID: same tenant + action always produces same ID.
110        let id = Uuid::new_v5(&tenant_id, action.as_bytes());
111
112        crate::helpers::retry_on_write_conflict(|| async {
113            db.query(
114                "UPSERT type::record('permission', $id) SET \
115                 tenant_id = $tenant_id, \
116                 action = $action, \
117                 description = $description, \
118                 created_at = IF (SELECT created_at FROM type::record('permission', $id))[0].created_at \
119                   THEN (SELECT created_at FROM type::record('permission', $id))[0].created_at \
120                   ELSE time::now() END, \
121                 updated_at = time::now()",
122            )
123            .bind(("id", id.to_string()))
124            .bind(("tenant_id", tenant_id.to_string()))
125            .bind(("action", action.to_string()))
126            .bind(("description", description.to_string()))
127            .await
128            .map_err(|e| DbError::Migration(format!("seed_permissions UPSERT failed: {e}")))?
129            .check()
130            .map_err(|e| DbError::Migration(format!("seed_permissions UPSERT check failed: {e}")))?;
131            Ok::<(), DbError>(())
132        })
133        .await?;
134    }
135
136    // Persist the new hash so subsequent restarts can skip this tenant.
137    crate::helpers::retry_on_write_conflict(|| async {
138        db.query(
139            "UPSERT type::record('seeder_state', $id) SET \
140             tenant_id = $tenant_id, hash = $hash, updated_at = time::now()",
141        )
142        .bind(("id", state_id.clone()))
143        .bind(("tenant_id", tenant_id.to_string()))
144        .bind(("hash", registry_hash.clone()))
145        .await
146        .map_err(|e| DbError::Migration(format!("seeder_state upsert failed: {e}")))?
147        .check()
148        .map_err(|e| DbError::Migration(format!("seeder_state upsert check: {e}")))?;
149        Ok::<(), DbError>(())
150    })
151    .await?;
152
153    Ok(())
154}

156/// Mint a one-time first-run bootstrap setup token if the database has
157/// never been bootstrapped (SECHRD-04 / D-03b).
158///
159/// # Behavior
160///
161/// - On a fresh (never-bootstrapped) database — no `bootstrap_setup_token`
162///   row yet AND no `user` row anywhere — a cryptographically random token
163///   is generated, its sha256 hash is persisted as the
164///   `bootstrap_setup_token` record ID, and the returned `Some(token)`
165///   carries the PLAINTEXT token so the caller can log it exactly once
166///   (the one deliberate secret-log exception, per D-03b). Only the HASH
167///   is ever persisted to the database.
168/// - On any subsequent boot — a setup token already exists, or at least
169///   one user already exists — this is a no-op and returns `Ok(None)`;
170///   nothing is minted or re-logged.
171pub async fn mint_bootstrap_setup_token_if_needed<C: Connection>(
172    db: &Surreal<C>,
173) -> Result<Option<String>, DbError> {
174    // A setup token was already minted — no-op.
175    let existing_tokens: Vec<CountRow> = db
176        .query("SELECT count() AS total FROM bootstrap_setup_token GROUP ALL")
177        .await
178        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token count failed: {e}")))?
179        .take(0)
180        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token count take failed: {e}")))?;
181    if existing_tokens.first().map(|r| r.total).unwrap_or(0) > 0 {
182        return Ok(None);
183    }
184
185    // Bootstrap already completed somewhere — no-op.
186    let existing_users: Vec<CountRow> = db
187        .query("SELECT count() AS total FROM user GROUP ALL")
188        .await
189        .map_err(|e| DbError::Migration(format!("user count failed: {e}")))?
190        .take(0)
191        .map_err(|e| DbError::Migration(format!("user count take failed: {e}")))?;
192    if existing_users.first().map(|r| r.total).unwrap_or(0) > 0 {
193        return Ok(None);
194    }
195
196    mint_setup_token(db).await.map(Some)
197}

199/// Why a re-mint was refused, or the token it produced.
200///
201/// Three outcomes rather than an error, because two of them are not failures:
202/// they are the security argument. A setup token is only ever re-mintable on a
203/// deployment that has **no administrator yet** — which is exactly the state an
204/// operator who lost the token at first boot is stuck in. Once anybody has
205/// bootstrapped, the token is a second way in and re-minting it would be a
206/// credential reset with no authentication in front of it.
207#[derive(Debug, Clone, PartialEq, Eq)]
208pub enum SetupTokenRemint {
209    /// A fresh token. The **plaintext**, returned once; only its hash is stored.
210    Minted(String),
211    /// At least one `user` row exists: this deployment has an administrator,
212    /// who can create others through the authenticated API.
213    RefusedUserExists,
214    /// A setup token was already redeemed. Bootstrap happened; whatever it
215    /// produced is the way in.
216    RefusedTokenConsumed,
217}
218
219/// Replace the bootstrap setup token, on a deployment nobody has bootstrapped.
220///
221/// The recovery path for DF-019: before this existed, an operator who lost the
222/// first-boot token had no way back except wiping the volume, because the token
223/// is stored only as a hash and the boot-time mint is a no-op once a row
224/// exists.
225///
226/// The gate is the whole security argument — see [`SetupTokenRemint`]. Both
227/// refusals are checked **before** anything is deleted, so a refused call
228/// leaves the existing token working.
229///
230/// The plaintext is returned, never logged: the caller prints it to stdout so
231/// it does not land in the container log a second time.
232pub async fn remint_bootstrap_setup_token<C: Connection>(
233    db: &Surreal<C>,
234) -> Result<SetupTokenRemint, DbError> {
235    let existing_users: Vec<CountRow> = db
236        .query("SELECT count() AS total FROM user GROUP ALL")
237        .await
238        .map_err(|e| DbError::Migration(format!("user count failed: {e}")))?
239        .take(0)
240        .map_err(|e| DbError::Migration(format!("user count take failed: {e}")))?;
241    if existing_users.first().map(|r| r.total).unwrap_or(0) > 0 {
242        return Ok(SetupTokenRemint::RefusedUserExists);
243    }
244
245    let consumed: Vec<CountRow> = db
246        .query("SELECT count() AS total FROM bootstrap_setup_token_consumed GROUP ALL")
247        .await
248        .map_err(|e| {
249            DbError::Migration(format!("bootstrap_setup_token_consumed count failed: {e}"))
250        })?
251        .take(0)
252        .map_err(|e| {
253            DbError::Migration(format!(
254                "bootstrap_setup_token_consumed count take failed: {e}"
255            ))
256        })?;
257    if consumed.first().map(|r| r.total).unwrap_or(0) > 0 {
258        return Ok(SetupTokenRemint::RefusedTokenConsumed);
259    }
260
261    // Delete then mint, in that order and only after both gates: the table
262    // holds at most one usable token, so leaving the old hash behind would
263    // give the deployment two valid tokens rather than a replacement.
264    db.query("DELETE bootstrap_setup_token")
265        .await
266        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token delete failed: {e}")))?
267        .check()
268        .map_err(|e| {
269            DbError::Migration(format!("bootstrap_setup_token delete check failed: {e}"))
270        })?;
271
272    mint_setup_token(db).await.map(SetupTokenRemint::Minted)
273}

275/// Generate a token, store its hash, and hand back the plaintext.
276///
277/// One function, shared by the first-boot mint and the re-mint, so the two
278/// cannot drift into producing differently-shaped tokens or storing them
279/// differently.
280async fn mint_setup_token<C: Connection>(db: &Surreal<C>) -> Result<String, DbError> {
281    // A cryptographically random 32-byte token, base64url-encoded (same shape
282    // as `axiam_auth::token::generate_refresh_token`).
283    let mut rng = rand::rng();
284    let bytes: [u8; 32] = rand::RngExt::random(&mut rng);
285    let token = URL_SAFE_NO_PAD.encode(bytes);
286
287    let mut hasher = Sha256::new();
288    hasher.update(token.as_bytes());
289    let token_hash = hex::encode(hasher.finalize());
290
291    db.query("CREATE type::record('bootstrap_setup_token', $hash) SET created_at = time::now()")
292        .bind(("hash", token_hash))
293        .await
294        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token mint failed: {e}")))?
295        .check()
296        .map_err(|e| DbError::Migration(format!("bootstrap_setup_token mint check failed: {e}")))?;
297
298    Ok(token)
299}

301/// Whether `tenant_id` is the organization's own scope — the only tenant whose
302/// seeded roles may hold organization-level actions — or `None` when the tenant
303/// cannot be resolved.
304///
305/// The callers fail closed in opposite directions, so the unresolved case is
306/// theirs to settle: the organization-level actions are withheld from a tenant
307/// that is not known to be the organization scope (being wrong costs an
308/// operator a grant they can add back by hand; the other way seeds the
309/// escalation the filter exists to prevent), and the `network:` permissions
310/// are withheld from one that is not known to be an ordinary tenant (#702).
311async fn tenant_is_organization_scope<C: Connection>(
312    db: &Surreal<C>,
313    tenant_id: Uuid,
314) -> Option<bool> {
315    tenant_scope(db, tenant_id).await.ok()
316}

327/// What the reconcile does about the `network:` namespace in one tenant.
328#[derive(Debug, Clone, Copy, PartialEq, Eq)]
329enum NetworkHalf {
330    /// The tenant's kind is known: grant and revoke by the seeding rule.
331    /// `ordinary_tenant` is false for the organization's own scope.
332    Reconcile { ordinary_tenant: bool },
333    /// The tenant's kind could not be read: touch no `network:` grant this boot.
334    Skip,
335}

337/// What the reconcile does about the organization-level actions in one tenant.
338#[derive(Debug, Clone, Copy, PartialEq, Eq)]
339enum OrganizationHalf {
340    /// The tenant's kind is known: grant by the seeding rule and revoke what an
341    /// ordinary tenant's roles hold. `may_hold` is true for the organization's
342    /// own scope.
343    Reconcile { may_hold: bool },
344    /// The tenant's kind could not be read: touch no organization-level grant
345    /// this boot (R112W4-03).
346    Skip,
347}
348
349/// The reconcile's decision for the organization-level half, from the resolved
350/// scope. The same reasoning as [`network_half`]: the grants exist, so a failed
351/// read must not be taken as "not the organization scope" and strip that
352/// scope's super-admin of `organizations:*` until the next boot.
353const fn organization_half(scope: Option<bool>) -> OrganizationHalf {
354    match scope {
355        Some(organization_scope) => OrganizationHalf::Reconcile {
356            may_hold: organization_scope,
357        },
358        None => OrganizationHalf::Skip,
359    }
360}

362/// The reconcile's decision for the `network:` half, from the resolved scope.
363///
364/// The seeder withholds the namespace from a tenant it cannot classify, which
365/// costs nothing: nothing exists yet. The reconcile must not read the same
366/// answer as "the seeded role may not hold it", because there the grants
367/// *exist*: a transient read failure would strip an ordinary tenant's
368/// super-admin and write rows giving a false reason, and only the next boot
369/// would give them back (#728, R111W4-04). An unresolved tenant is left alone
370/// until a boot that can read it.
371const fn network_half(scope: Option<bool>) -> NetworkHalf {
372    match scope {
373        Some(organization_scope) => NetworkHalf::Reconcile {
374            ordinary_tenant: !organization_scope,
375        },
376        None => NetworkHalf::Skip,
377    }
378}

380/// Seed the three default roles — super-admin, admin, and viewer — for the
381/// given tenant, assigning permissions according to D-12.
382///
383/// # Idempotency
384///
385/// If all three roles already exist the function returns their IDs without
386/// creating duplicates. If only some roles are missing they are created.
387///
388/// # Permission assignment rules
389///
390/// - **super-admin**: every permission in the tenant.
391/// - **admin**: every permission except `admin:bootstrap`.
392/// - **viewer**: permissions whose action ends with `:list` or `:get`.
393///
394/// # Organization-level actions
395///
396/// In an ordinary (`TenantKind::Standard`) tenant, the actions listed in
397/// [`ORGANIZATION_LEVEL_ACTIONS`](axiam_core::permission_scope::ORGANIZATION_LEVEL_ACTIONS)
398/// are withheld from super-admin and admin. A tenant administrator holding
399/// `ca_certificates:manage` could flag an organization-wide mTLS trust anchor
400/// and then mint certificates authenticating as principals in sibling tenants
401/// (B-04) — so the grant is not made in the first place. The organization's
402/// own scope is seeded with everything.
403///
404/// # The reserved `network:` namespace (RADIUS Live profile, W6.2)
405///
406/// The two global administrative roles are **not** given the whole `network:`
407/// namespace. `super-admin` gets
408/// [`SEEDED_SUPER_ADMIN_NETWORK_ACTIONS`] — `network:read` and the five meta
409/// permissions, never `network:join` nor the administrative class — and
410/// `admin` gets none. Design note §4.2 rule 1 forbids a network-sensitive role
411/// to be global; the seeded super-admin is the one exemption, because under the
412/// §5.3 delegation bound a network administrator can only be made by someone
413/// who already holds `network:grant` and every meta permission conferred, and a
414/// tenant whose bootstrap role held none of them could never administer its
415/// network. That constant's documentation gives the exemption's limits; the API
416/// refuses a `network:` grant to any global role, this one included.
417///
418/// This is the second of two layers. `require_organization_principal` in
419/// `axiam-api-rest` is the load-bearing one and refuses such a caller whatever
420/// grants it holds; withholding the grant means the two layers agree instead of
421/// one quietly contradicting the other. The `viewer` role is untouched — every
422/// organization-level action is a mutation and viewer is seeded only with
423/// `:list` and `:get`.
424pub async fn seed_default_roles<C: Connection>(
425    db: &Surreal<C>,
426    tenant_id: Uuid,
427    _permission_registry: &[(&str, &str)],
428) -> Result<SeedRolesResult, DbError> {
429    let role_repo = SurrealRoleRepository::new(db.clone());
430    let perm_repo = SurrealPermissionRepository::new(db.clone());
431
432    // -----------------------------------------------------------------------
433    // 1. Ensure all three roles exist (idempotent)
434    // -----------------------------------------------------------------------
435
436    // Fetch existing roles for this tenant in a single list call.
437    let existing = role_repo
438        .list(
439            tenant_id,
440            Pagination {
441                offset: 0,
442                limit: 1000,
443                search: None,
444            },
445        )
446        .await
447        .map_err(|e| DbError::Migration(format!("seed_default_roles list roles failed: {e}")))?;
448
449    let find_role = |name: &str| existing.items.iter().find(|r| r.name == name).map(|r| r.id);
450
451    // SECHRD-04: role creation must tolerate a CONCURRENT racer creating the
452    // same default role for the same tenant — `find_or_create_role` treats
453    // the `idx_role_tenant_name` UNIQUE-index violation as "a concurrent
454    // caller already won" and re-fetches the winner's row instead of
455    // failing (see its doc comment for why this is now reachable).
456    let super_admin_id = find_or_create_role(
457        &role_repo,
458        tenant_id,
459        find_role(SEEDED_SUPER_ADMIN_ROLE),
460        SEEDED_SUPER_ADMIN_ROLE,
461        "Full system access — all permissions",
462    )
463    .await?;
464
465    let admin_id = find_or_create_role(
466        &role_repo,
467        tenant_id,
468        find_role("admin"),
469        "admin",
470        "Administrative access — all entity CRUD operations",
471    )
472    .await?;
473
474    let viewer_id = find_or_create_role(
475        &role_repo,
476        tenant_id,
477        find_role("viewer"),
478        "viewer",
479        "Read-only access — list and get operations",
480    )
481    .await?;
482
483    // -----------------------------------------------------------------------
484    // 2. Fetch all permissions for this tenant
485    // -----------------------------------------------------------------------
486
487    let permissions = perm_repo
488        .list(
489            tenant_id,
490            Pagination {
491                offset: 0,
492                limit: 10_000,
493                search: None,
494            },
495        )
496        .await
497        .map_err(|e| {
498            DbError::Migration(format!("seed_default_roles list permissions failed: {e}"))
499        })?;
500
501    // -----------------------------------------------------------------------
502    // 3. Grant permissions to roles (idempotent — skip already-granted pairs)
503    // -----------------------------------------------------------------------
504    //
505    // SECHRD-04: `grant_to_role` CREATEs a `grants` edge guarded by a UNIQUE
506    // (in, out) index (CQ-B17) — re-granting an already-granted permission
507    // is a hard error, not a silent no-op. Before this fix, `seed_default_roles`
508    // was only ever called once per tenant (the old bootstrap TOCTOU check
509    // disabled the endpoint before a second call could reach here). Now that
510    // bootstrap atomicity is enforced by the `bootstrap_lock` uniqueness
511    // invariant instead, this function can be reached again — including
512    // CONCURRENTLY, by two racing bootstrap requests — for a tenant whose
513    // roles+grants already exist. The pre-fetch below skips already-granted
514    // pairs in the common (sequential retry) case; `grant_to_role_idempotent`
515    // additionally tolerates the residual TOCTOU window between this
516    // pre-fetch and the CREATE, for true concurrent callers.
517
518    let super_admin_have = granted_permission_ids(&perm_repo, tenant_id, super_admin_id).await?;
519    let admin_have = granted_permission_ids(&perm_repo, tenant_id, admin_id).await?;
520    let viewer_have = granted_permission_ids(&perm_repo, tenant_id, viewer_id).await?;
521
522    // B-04 defence in depth: an ordinary tenant's roles are not seeded with
523    // organization-level actions. See this function's doc comment.
524    let organization_scope = tenant_is_organization_scope(db, tenant_id).await;
525    let may_hold_org_actions = organization_scope == Some(true);
526    let network_super_admin = organization_scope == Some(false);
527    let withheld = |action: &str| !may_hold_org_actions && is_organization_level_action(action);
528
529    for perm in &permissions.items {
530        if withheld(&perm.action) {
531            continue;
532        }
533
534        // super-admin gets ALL permissions, except the parts of the reserved
535        // `network:` namespace a bootstrap role must not carry (see above).
536        if super_admin_may_hold(&perm.action, network_super_admin)
537            && !super_admin_have.contains(&perm.id)
538        {
539            grant_to_role_idempotent(&perm_repo, tenant_id, super_admin_id, perm.id)
540                .await
541                .map_err(|e| {
542                    DbError::Migration(format!(
543                        "seed_default_roles grant super-admin permission {} failed: {e}",
544                        perm.action
545                    ))
546                })?;
547        }
548
549        // admin gets every permission EXCEPT admin:bootstrap and the reserved
550        // `network:` namespace.
551        if admin_may_hold(&perm.action) && !admin_have.contains(&perm.id) {
552            grant_to_role_idempotent(&perm_repo, tenant_id, admin_id, perm.id)
553                .await
554                .map_err(|e| {
555                    DbError::Migration(format!(
556                        "seed_default_roles grant admin permission {} failed: {e}",
557                        perm.action
558                    ))
559                })?;
560        }
561
562        // viewer gets only :list and :get permissions.
563        if (perm.action.ends_with(":list") || perm.action.ends_with(":get"))
564            && !viewer_have.contains(&perm.id)
565        {
566            grant_to_role_idempotent(&perm_repo, tenant_id, viewer_id, perm.id)
567                .await
568                .map_err(|e| {
569                    DbError::Migration(format!(
570                        "seed_default_roles grant viewer permission {} failed: {e}",
571                        perm.action
572                    ))
573                })?;
574        }
575    }
576
577    Ok(SeedRolesResult {

579        admin_role_id: admin_id,

581    })

584/// Whether the seeded `super-admin` may hold `action`: everything outside the
585/// reserved `network:` namespace, and inside it [`SEEDED_SUPER_ADMIN_NETWORK_ACTIONS`]
586/// in an ordinary tenant only.
587///
588/// `ordinary_tenant` is false for the organization's own scope, whose
589/// super-admin is global and so reaches every tenant of the organization: the
590/// exemption would carry the network meta permissions across tenants (#702,
591/// T-483), which the design note's §7 rules out. The seeder also passes false
592/// for a tenant it could not resolve, so the question fails closed; the
593/// reconcile does not ask it for one (see [`network_half`], #728).
594fn super_admin_may_hold(action: &str, ordinary_tenant: bool) -> bool {
595    !is_network_action(action)
596        || (ordinary_tenant && SEEDED_SUPER_ADMIN_NETWORK_ACTIONS.contains(&action))
597}

599/// Whether the seeded `admin` may hold `action`: everything but
600/// `admin:bootstrap` and the reserved `network:` namespace. A tenant
601/// administrator is not a network administrator; the super-admin makes one
602/// (design note §5.3).
603fn admin_may_hold(action: &str) -> bool {
604    action != "admin:bootstrap" && !is_network_action(action)
605}

607/// Find an existing role by name, or create it — tolerating a CONCURRENT
608/// caller creating the SAME (tenant_id, name) role at the same time.
609///
610/// `role_repo.create()` CREATEs a fresh-UUID row guarded by the
611/// `idx_role_tenant_name` UNIQUE index; two racing bootstrap requests can
612/// both observe "role missing" and both attempt a create. The loser's
613/// UNIQUE-index violation is NOT a real failure here — a concurrent caller
614/// already won and the row exists — so it is treated as "re-fetch and use
615/// the winner's ID" rather than propagated (SECHRD-04: `seed_default_roles`
616/// is reachable from two concurrent bootstrap requests now that atomicity
617/// is enforced by `bootstrap_lock`, not the old TOCTOU check).
618async fn find_or_create_role<C: Connection>(
619    role_repo: &SurrealRoleRepository<C>,
620    tenant_id: Uuid,
621    existing_id: Option<Uuid>,
622    name: &str,
623    description: &str,
624) -> Result<Uuid, DbError> {
625    if let Some(id) = existing_id {
626        return Ok(id);
627    }
628
629    match role_repo
630        .create(CreateRole {
631            tenant_id,
632            name: name.to_string(),
633            description: description.to_string(),
634            is_global: true,
635        })
636        .await
637    {
638        Ok(role) => Ok(role.id),
639        Err(e) => {
640            let msg = e.to_string();
641            match classify_write_error(&msg, "role") {
642                DbError::AlreadyExists { .. } => {
643                    let refreshed = role_repo
644                        .list(
645                            tenant_id,
646                            Pagination {
647                                offset: 0,
648                                limit: 1000,
649                                search: None,
650                            },
651                        )
652                        .await
653                        .map_err(|e| {
654                            DbError::Migration(format!(
655                                "find_or_create_role re-list '{name}' failed: {e}"
656                            ))
657                        })?;
658                    refreshed
659                        .items
660                        .into_iter()
661                        .find(|r| r.name == name)
662                        .map(|r| r.id)
663                        .ok_or_else(|| {
664                            DbError::Migration(format!(
665                                "find_or_create_role: role '{name}' vanished after concurrent create race"
666                            ))
667                        })
668                }
669                _ => Err(DbError::Migration(format!(
670                    "find_or_create_role '{name}' failed: {msg}"
671                ))),
672            }
673        }
674    }
675}

677/// The set of permission IDs already granted to `role_id` (tenant-scoped).
678///
679/// Shared by [`seed_default_roles`] and [`reconcile_default_role_grants`] so
680/// neither re-attempts a `grant_to_role` CREATE for a pair that already
681/// exists — the `grants` edge table's UNIQUE (in, out) index (CQ-B17) makes
682/// a duplicate grant a hard error, not a no-op.
683async fn granted_permission_ids<C: Connection>(
684    perm_repo: &SurrealPermissionRepository<C>,

686    role_id: Uuid,

689        .get_role_permissions(tenant_id, role_id)
690        .await
691        .map(|perms| perms.into_iter().map(|p| p.id).collect())
692        .map_err(|e| DbError::Migration(format!("granted_permission_ids failed: {e}")))
693}

696/// concurrent caller won the same grant) as success rather than an error.
697///
698/// Closes the residual TOCTOU between [`granted_permission_ids`]'s pre-fetch

762/// What [`reconcile_default_role_grants`] changed for one tenant.
763///
764/// Three counters rather than one, because they mean different things to an
765/// operator reading the boot log: `granted` is a back-fill catching a role up
766/// with a registry that grew, while `revoked` and `network_revoked` are a
767/// **capability being taken away** from a role that already had it — an
768/// organization-level action from an ordinary tenant's role, and a `network:`
769/// permission a seeded role may no longer hold (the organization scope's
770/// super-admin and, on any tenant, the roles the rule withholds it from). The
771/// last two deserve their own line in the log and their own numbers.
772#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
773pub struct ReconcileOutcome {
774    /// Grants created because a role was missing a permission it should hold.
775    pub granted: usize,
776    /// Grants removed because an ordinary tenant's role held an
777    /// organization-level action.
778    pub revoked: usize,
779    /// Grants removed because a seeded role held a `network:` permission the
780    /// seeding rule withholds from it (a count apart from `revoked`, whose
781    /// organization-level actions the operator reads differently).
782    pub network_revoked: usize,
783}
784
785impl ReconcileOutcome {
786    /// Whether anything changed — the condition for logging at all.
787    pub const fn is_empty(&self) -> bool {
788        self.granted == 0 && self.revoked == 0 && self.network_revoked == 0
789    }
790}

792/// Reconcile the default roles' permission grants with the current permission
793/// set for a tenant: granting the permissions each role is MISSING, and
794/// revoking organization-level actions an ordinary tenant's roles should never
795/// have held.
796///
797/// [`seed_default_roles`] runs only at `/bootstrap`, which self-disables after
798/// the first admin is created. Server startup, by contrast, runs
799/// [`seed_permissions`] on every boot — so a permission ADDED to the registry
800/// after the initial bootstrap exists as a row but is granted to no role,
801/// causing spurious 403s for actions the super-admin/admin should be allowed
802/// to perform (e.g. a newly-added `oauth2_clients:create`). Calling this on
803/// startup back-fills those grants.
804///
805/// # Why this also revokes
806///
807/// Until B-04's follow-up, every tenant's `super-admin` was seeded with the
808/// *entire* permission registry, organization-level actions included. Those
809/// grants exist in every deployment bootstrapped before that change, and
810/// `seed_default_roles` no longer runs for them. A reconcile that could only
811/// ever grant would therefore leave the old, wrong state in place forever and
812/// the fix would apply to new tenants only.
813///
814/// Revocation is deliberately narrow. It touches **only**:
815///
816/// * the three seeded default roles, found by name — a role an operator
817///   created is theirs, and this function has no business editing it;
818/// * actions in
819///   [`ORGANIZATION_LEVEL_ACTIONS`](axiam_core::permission_scope::ORGANIZATION_LEVEL_ACTIONS),
820///   in tenants that are **not** the organization's own scope;
821/// * actions of the reserved `network:` namespace that
822///   [`seed_default_roles`] does not give the role (every one for `admin`,
823///   `network:join` and the administrative class for `super-admin`), in every
824///   tenant; and in the organization's own scope, where the super-admin is
825///   global and would carry them into every tenant of the organization, all of
826///   them (#702, T-483). One audit row is written per revocation.
827///
828/// It is not a general "make the grants match the seeding rules" pass: an
829/// operator who deliberately granted `users:delete` to `viewer` keeps it. Only
830/// the specific escalation is undone.
831///
832/// # Idempotency
833///
834/// Existing correct `grants` edges are left untouched and only missing grants
835/// are created, so no duplicate edges accumulate across restarts. Once the
836/// organization-level grants are gone the revoke half finds nothing and returns
837/// zero, so a steady-state boot is a no-op. Tenants without the default roles
838/// (never bootstrapped) are skipped.
839pub async fn reconcile_default_role_grants<C: Connection>(
840    db: &Surreal<C>,
841    tenant_id: Uuid,
842) -> Result<ReconcileOutcome, DbError> {
843    let role_repo = SurrealRoleRepository::new(db.clone());
844    let perm_repo = SurrealPermissionRepository::new(db.clone());
845
846    let existing_roles = role_repo
847        .list(
848            tenant_id,
849            Pagination {
850                offset: 0,
851                limit: 1000,
852                search: None,
853            },
854        )
855        .await
856        .map_err(|e| DbError::Migration(format!("reconcile list roles failed: {e}")))?;
857    let find_role = |name: &str| {
858        existing_roles
859            .items
860            .iter()
861            .find(|r| r.name == name)
862            .map(|r| r.id)
863    };
864
865    // Only reconcile tenants bootstrapped with the three default roles.
866    let (Some(super_admin_id), Some(admin_id), Some(viewer_id)) = (
867        find_role(SEEDED_SUPER_ADMIN_ROLE),
868        find_role("admin"),
869        find_role("viewer"),
870    ) else {
871        return Ok(ReconcileOutcome::default());
872    };
873
874    let permissions = perm_repo
875        .list(
876            tenant_id,
877            Pagination {
878                offset: 0,
879                limit: 10_000,
880                search: None,
881            },
882        )
883        .await
884        .map_err(|e| DbError::Migration(format!("reconcile list permissions failed: {e}")))?;
885
886    let sa_have = granted_permission_ids(&perm_repo, tenant_id, super_admin_id).await?;
887    let admin_have = granted_permission_ids(&perm_repo, tenant_id, admin_id).await?;
888    let viewer_have = granted_permission_ids(&perm_repo, tenant_id, viewer_id).await?;
889
890    // B-04 defence in depth — see this function's doc comment. In an ordinary
891    // tenant the organization-level actions are neither granted below nor left
892    // in place if a previous version granted them.
893    let organization_scope = match tenant_scope(db, tenant_id).await {
894        Ok(organization_scope) => Some(organization_scope),
895        Err(e) => {
896            tracing::warn!(
897                %tenant_id,
898                error = %e,
899                "could not read the tenant's kind; the default-role reconcile is \
900                 skipped for it this boot, in both its network: and its organization-level \
901                 halves (nothing revoked, nothing granted)"
902            );
903            None
904        }
905    };
906    let organization_half = organization_half(organization_scope);
907    let may_hold_org_actions = organization_half == OrganizationHalf::Reconcile { may_hold: true };
908    let network_half = network_half(organization_scope);
909    let network_super_admin = network_half
910        == NetworkHalf::Reconcile {
911            ordinary_tenant: true,
912        };
913
914    let mut outcome = ReconcileOutcome::default();
915    for perm in &permissions.items {
916        if is_organization_level_action(&perm.action) && organization_half == OrganizationHalf::Skip
917        {
918            continue;
919        }
920        // The reserved `network:` namespace: grant what the seeded roles may
921        // hold, and take back what an earlier build of this branch granted them
922        // (W6.1 gave both roles the whole namespace). Only the two seeded
923        // administrative roles are touched, as for the organization-level
924        // actions below.
925        if is_network_action(&perm.action) && network_half == NetworkHalf::Skip {
926            continue;
927        }
928        if is_network_action(&perm.action)
929            && (!super_admin_may_hold(&perm.action, network_super_admin)
930                || !admin_may_hold(&perm.action))
931        {
932            for (role_id, have, role, may) in [
933                (
934                    super_admin_id,
935                    &sa_have,
936                    "super-admin",
937                    super_admin_may_hold(&perm.action, network_super_admin),
938                ),
939                (admin_id, &admin_have, "admin", admin_may_hold(&perm.action)),
940            ] {
941                if !may && have.contains(&perm.id) {
942                    perm_repo
943                        .revoke_from_role(tenant_id, role_id, perm.id)
944                        .await
945                        .map_err(|e| {
946                            DbError::Migration(format!(
947                                "reconcile revoke reserved {} from {role} failed: {e}",
948                                perm.action
949                            ))
950                        })?;
951                    outcome.network_revoked += 1;
952                    audit_network_revocation(db, tenant_id, role_id, role, &perm.action, perm.id)
953                        .await;
954                }
955            }
956            if super_admin_may_hold(&perm.action, network_super_admin)
957                && !sa_have.contains(&perm.id)
958            {
959                // §4.2 rule 2 binds the reconcile too (R110W6-04): the
960                // grant's own transaction refuses a `network:` permission to
961                // a super-admin a group holds — which a release before 1.1.0
962                // allowed — and a group assignment racing it. Withheld then,
963                // and said so, rather than failing the boot: the tenant keeps
964                // working, and its super-admin simply does not carry the
965                // network meta permissions until no group holds it (the next
966                // boot's reconcile grants them).
967                match perm_repo
968                    .grant_to_role(tenant_id, super_admin_id, perm.id)
969                    .await
970                {
971                    Ok(()) => outcome.granted += 1,
972                    Err(AxiamError::Network(NetworkConflict::StaleRole { what_landed })) => {
973                        tracing::warn!(
974                            %tenant_id,
975                            action = %perm.action,
976                            reason = %what_landed,
977                            "network permission withheld from super-admin: a group holds the \
978                             role, and a role carrying a network: permission is never group-held \
979                             (design note §4.2 rule 2); unassign the role from the group to let \
980                             the next boot grant it"
981                        );
982                    }
983                    Err(e) => {
984                        return Err(DbError::Migration(format!(
985                            "reconcile grant super-admin failed: {e}"
986                        )));
987                    }
988                }
989            }
990            continue;
991        }
992        if !may_hold_org_actions && is_organization_level_action(&perm.action) {
993            // Withhold, and undo the grant if an earlier version made it.
994            // Only super-admin and admin are considered: viewer is seeded
995            // solely with `:list`/`:get` and no organization-level action is
996            // one, so a viewer holding one was not put there by this seeder.
997            for (role_id, have, role) in [
998                (super_admin_id, &sa_have, "super-admin"),
999                (admin_id, &admin_have, "admin"),
1000            ] {
1001                if have.contains(&perm.id) {
1002                    perm_repo
1003                        .revoke_from_role(tenant_id, role_id, perm.id)
1004                        .await
1005                        .map_err(|e| {
1006                            DbError::Migration(format!(
1007                                "reconcile revoke organization-level {} from {role} failed: {e}",
1008                                perm.action
1009                            ))
1010                        })?;
1011                    outcome.revoked += 1;
1012                }
1013            }
1014            continue;
1015        }
1016
1017        // super-admin gets every permission.
1018        if !sa_have.contains(&perm.id) {
1019            perm_repo
1020                .grant_to_role(tenant_id, super_admin_id, perm.id)
1021                .await
1022                .map_err(|e| {
1023                    DbError::Migration(format!("reconcile grant super-admin failed: {e}"))
1024                })?;
1025            outcome.granted += 1;
1026        }
1027        // admin gets every permission except admin:bootstrap.
1028        if perm.action != "admin:bootstrap" && !admin_have.contains(&perm.id) {
1029            perm_repo
1030                .grant_to_role(tenant_id, admin_id, perm.id)
1031                .await
1032                .map_err(|e| DbError::Migration(format!("reconcile grant admin failed: {e}")))?;
1033            outcome.granted += 1;
1034        }
1035        // viewer gets only :list and :get permissions.
1036        if (perm.action.ends_with(":list") || perm.action.ends_with(":get"))
1037            && !viewer_have.contains(&perm.id)
1038        {
1039            perm_repo
1040                .grant_to_role(tenant_id, viewer_id, perm.id)
1041                .await
1042                .map_err(|e| DbError::Migration(format!("reconcile grant viewer failed: {e}")))?;
1043            outcome.granted += 1;
1044        }
1045    }
1046
1047    Ok(outcome)
1048}
