1//! The address guard: where a tenant's directory may be, and the one address a
2//! connection may use (T23.3.7, D-19, closes T-300).
3//!
4//! A tenant administrator chooses the directory URL. Unchecked, that turns
5//! every directory sign-in into a connection AXIAM opens to an address of the
6//! tenant's choosing — its own loopback services, the cloud metadata endpoint,
7//! the pod network — and the time a failure takes tells an open port from a
8//! closed one.
9//!
10//! # The rule ([`AddressPolicy::check`])
11//!
12//! The host is resolved (an IP-literal host is used as written) and **every**
13//! address it resolves to must pass, or the directory is refused:
14//!
15//! | Class ([`axiam_core::ip_class`]) | Answer |
16//! |---|---|
17//! | global unicast | admitted |
18//! | loopback, unspecified, link-local (incl. `169.254.169.254`, `fe80::/10`), multicast, special-purpose | **refused, always** |
19//! | private — RFC 1918, CGNAT `100.64/10`, ULA `fc00::/7` | refused **unless** inside a network the operator listed in [`ALLOWED_PRIVATE_NETWORKS_ENV`] |
20//! | a cloud metadata endpoint inside a private range (`fd00:ec2::254`, `100.100.100.200`) | refused, whatever the list says |
21//! | an address of this host on a port AXIAM itself listens on | refused |
22//!
23//! IPv4-mapped IPv6 addresses are judged as the IPv4 address they denote. The
24//! classification is the one `guarded_fetch` uses; only the policy differs —
25//! corporate directories live on private networks, so private ranges are
26//! admissible here where they are not for an outbound HTTP fetch, and only by
27//! the operator.
28//!
29//! An IPv6-literal URL (`ldaps://[2001:db8::1]`) is refused too
30//! ([`GuardError::Ipv6Literal`]): no certificate check can be done against it
31//! (T-293), so such a directory could never be used.
32//!
33//! # Pinning
34//!
35//! [`guard`] returns the vetted addresses, and the connector
36//! (`DirectoryClient::connect`) opens its TCP socket to one of **those**
37//! `SocketAddr`s and hands the stream to `ldap3`; nothing resolves the name a
38//! second time. The TLS server name stays the URL's host, so certificate
39//! verification is unchanged. DNS rebinding between the check and the connect
40//! is therefore impossible within one connection, and because the guard runs
41//! again at **every** connect — the pool, the user bind, group lookup, the sync
42//! job — a name that is re-pointed after the configuration was saved is caught
43//! at the next connection.
44//!
45//! # The allow-list is the operator's, and is a residual
46//!
47//! It is deployment configuration, never tenant-editable. Inside the networks
48//! it lists, any tenant administrator can point the connector at any host and
49//! port; what that buys is a TLS handshake (or a 31-byte StartTLS request)
50//! against a host that must then present a certificate chaining to the
51//! tenant's anchors before anything else is sent, answered to the user as the
52//! same generic failure. Keep it to the networks directories actually live in
53//! (decision D-32 says why it is a network list, not a host list as
54//! `AXIAM__PKI__SSRF_ALLOWED_HOSTS` is).
55

65/// The environment variable that lists the private networks a tenant's
66/// directory may be in: comma-separated CIDR blocks (`10.20.0.0/16`,

79/// Why an address, or a whole directory host, was refused.
80#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
81pub enum GuardError {
82    /// The URL could not be parsed or names no host.
83    #[error("the directory URL has no usable host")]
84    InvalidUrl,
85    /// The host is a bracketed IPv6 literal, which cannot be name-checked.
86    #[error(
87        "an IPv6-literal directory URL cannot be certificate-checked; name the directory by a \
88         host name"
89    )]
90    Ipv6Literal,
91    /// The name did not resolve, or resolved to nothing.
92    #[error("the directory host could not be resolved")]
93    Unresolvable,
94    /// The name resolved to more addresses than [`MAX_RESOLVED_ADDRESSES`].
95    #[error("the directory host resolves to too many addresses")]
96    TooManyAddresses,
97    /// An address is loopback.
98    #[error("the directory host resolves to a loopback address")]
99    Loopback,
100    /// An address is unspecified (`0.0.0.0/8`, `::`).
101    #[error("the directory host resolves to an unspecified address")]
102    Unspecified,
103    /// An address is link-local (`169.254/16`, `fe80::/10`).
104    #[error("the directory host resolves to a link-local address")]
105    LinkLocal,
106    /// An address is multicast.
107    #[error("the directory host resolves to a multicast address")]
108    Multicast,
109    /// An address is in a special-purpose block nothing legitimate lives in.
110    #[error("the directory host resolves to a special-purpose address")]
111    SpecialPurpose,
112    /// An address is a cloud metadata endpoint inside a private range.
113    #[error("the directory host resolves to a cloud metadata address")]
114    Metadata,
115    /// An address is private and outside every network the operator listed.
116    #[error(
117        "the directory host resolves to a private address outside the deployment's allowed \
118         networks"
119    )]
120    PrivateNotAllowed,
121    /// An address is this host's own, on a port AXIAM listens on.
122    #[error("the directory host resolves to one of AXIAM's own listeners")]
123    OwnListener,
124}
125
126impl GuardError {
127    /// A fixed sentence for the operator's log line (the connector's
128    /// `Failure` carries `&'static str` only).
129    #[must_use]
130    pub const fn reason(&self) -> &'static str {
131        match self {
132            Self::InvalidUrl => "the directory URL has no usable host",
133            Self::Ipv6Literal => "an IPv6-literal directory URL cannot be certificate-checked",
134            Self::Unresolvable => "the directory host could not be resolved",
135            Self::TooManyAddresses => "the directory host resolves to too many addresses",
136            Self::Loopback => "the directory host resolves to a loopback address (refused)",
137            Self::Unspecified => "the directory host resolves to an unspecified address (refused)",
138            Self::LinkLocal => {
139                "the directory host resolves to a link-local or metadata address (refused)"
140            }
141            Self::Multicast => "the directory host resolves to a multicast address (refused)",
142            Self::SpecialPurpose => {
143                "the directory host resolves to a special-purpose address (refused)"
144            }
145            Self::Metadata => "the directory host resolves to a cloud metadata address (refused)",
146            Self::PrivateNotAllowed => {
147                "the directory host resolves to a private address outside \
148                 AXIAM__DIRECTORY__ALLOWED_PRIVATE_NETWORKS (refused)"
149            }
150            Self::OwnListener => {
151                "the directory host resolves to one of AXIAM's own listeners (refused)"
152            }
153        }
154    }
155
156    /// Whether the refusal is the configuration's fault (`true`) rather than a
157    /// resolver that could not answer (`false`, worth a retry later).
158    #[must_use]
159    pub const fn is_policy(&self) -> bool {
160        !matches!(self, Self::Unresolvable)
161    }
162}

164/// The deployment's rule for directory addresses. Built once at composition
165/// from deployment configuration; tenants cannot change it.
166#[derive(Debug, Clone, Default)]
167pub struct AddressPolicy {
168    allowed_private: Vec<IpNetwork>,
169    listener_ports: Vec<u16>,
170    admit_loopback: bool,
171}
172
173impl AddressPolicy {
174    /// The strict policy: no private network admitted, no listener known.
175    #[must_use]
176    pub fn new() -> Self {
177        Self::default()
178    }
179
180    /// Admit private addresses inside `networks` (the operator's
181    /// [`ALLOWED_PRIVATE_NETWORKS_ENV`]). Only addresses of the private class
182    /// ever consult the list: a network that also covers loopback, link-local
183    /// or public space admits nothing more than its private part, and never a
184    /// metadata endpoint.
185    #[must_use]
186    pub fn with_allowed_private_networks(mut self, networks: Vec<IpNetwork>) -> Self {
187        self.allowed_private = networks;
188        self
189    }
190
191    /// The ports AXIAM's own listeners are bound to (REST, gRPC). An address
192    /// of this host on one of them is refused, so a directory cannot be
193    /// pointed back at AXIAM itself.
194    #[must_use]
195    pub fn with_listener_ports(mut self, ports: impl IntoIterator<Item = u16>) -> Self {
196        self.listener_ports = ports.into_iter().collect();
197        self.listener_ports.sort_unstable();
198        self.listener_ports.dedup();
199        self
200    }
201
202    /// **Tests only.** Admit loopback addresses, so the in-process test
203    /// directory on `127.0.0.1` can be reached; every other rule still applies
204    /// (link-local, metadata, private ranges, listener ports). Never called by
205    /// production code: the composition root builds its policy with
206    /// [`Self::new`], and a grep for this name finds every exception.
207    #[doc(hidden)]
208    #[must_use]
209    pub fn admitting_loopback_for_tests(mut self) -> Self {
210        self.admit_loopback = true;
211        self
212    }
213
214    /// The networks admitted.
215    #[must_use]
216    pub fn allowed_private_networks(&self) -> &[IpNetwork] {
217        &self.allowed_private
218    }
219
220    /// The listener ports refused on this host's addresses.
221    #[must_use]
222    pub fn listener_ports(&self) -> &[u16] {
223        &self.listener_ports
224    }
225
226    /// Judge one resolved address.
227    ///
228    /// # Errors
229    ///
230    /// The [`GuardError`] naming the rule `address` breaks.
231    pub fn check(&self, address: SocketAddr) -> Result<(), GuardError> {
232        let ip = canonical(address.ip());
233        match classify(ip) {
234            IpClass::Global => {}
235            IpClass::Loopback if self.admit_loopback => {}
236            IpClass::Loopback => return Err(GuardError::Loopback),
237            IpClass::Unspecified => return Err(GuardError::Unspecified),
238            IpClass::LinkLocal => return Err(GuardError::LinkLocal),
239            IpClass::Multicast => return Err(GuardError::Multicast),
240            IpClass::SpecialPurpose => return Err(GuardError::SpecialPurpose),
241            IpClass::Private => {
242                if is_never_allowed(ip) {
243                    return Err(GuardError::Metadata);
244                }
245                if !self.allowed_private.iter().any(|net| net.contains(ip)) {
246                    return Err(GuardError::PrivateNotAllowed);
247                }
248            }
249        }
250        if self.listener_ports.contains(&address.port()) && is_local_address(ip) {
251            return Err(GuardError::OwnListener);
252        }
253        Ok(())
254    }
255}

257/// Whether `ip` is an address of this host: loopback, or one an unprivileged
258/// socket can bind to (binding fails with `EADDRNOTAVAIL` for an address no
259/// local interface holds). No packet is sent. A host configured for non-local
260/// binds (`net.ipv4.ip_nonlocal_bind`) reads every address as local, which
261/// only widens the listener-port refusal — the safe direction.
262fn is_local_address(ip: IpAddr) -> bool {
263    ip.is_loopback() || std::net::UdpSocket::bind(SocketAddr::new(ip, 0)).is_ok()
264}

266/// Parse [`ALLOWED_PRIVATE_NETWORKS_ENV`]'s comma-separated form. Returns the
267/// blocks that parsed and the entries that did not (for the startup log; an
268/// entry that does not parse admits nothing, so a typo fails closed).
269#[must_use]
270pub fn parse_allowed_networks(raw: &str) -> (Vec<IpNetwork>, Vec<String>) {
271    let mut networks = Vec::new();
272    let mut rejected = Vec::new();
273    for entry in raw.split(',').map(str::trim).filter(|e| !e.is_empty()) {
274        match entry.parse::<IpNetwork>() {
275            Ok(network) if !networks.contains(&network) => networks.push(network),
276            Ok(_) => {}
277            Err(_) => rejected.push(entry.to_string()),
278        }
279    }
280    (networks, rejected)
281}

283/// A boxed resolution future.
284pub type ResolveFuture<'a> = Pin<Box<dyn Future<Output = io::Result<Vec<SocketAddr>>> + Send + 'a>>;
285
286/// Turns a host name into addresses. The production resolver is
287/// [`SystemResolver`]; tests script answers (including a name that answers
288/// differently each time, to prove the connector never resolves twice).
289pub trait Resolver: Send + Sync {
290    /// Resolve `host` (a domain name, never an IP literal), with `port` on
291    /// every address.
292    fn resolve<'a>(&'a self, host: &'a str, port: u16) -> ResolveFuture<'a>;
293}

295/// The operating system's resolver (`getaddrinfo`, through Tokio).
296#[derive(Debug, Clone, Copy, Default)]
297pub struct SystemResolver;
298
299impl Resolver for SystemResolver {
300    fn resolve<'a>(&'a self, host: &'a str, port: u16) -> ResolveFuture<'a> {
301        Box::pin(async move { Ok(tokio::net::lookup_host((host, port)).await?.collect()) })
302    }
303}

305/// A directory host that passed the guard: the name the certificate must carry
306/// and the addresses a connection may use.
307#[derive(Debug, Clone, PartialEq, Eq)]
308pub struct GuardedTarget {
309    /// The URL's host, as the TLS server name.
310    pub host: String,
311    /// The port.
312    pub port: u16,
313    /// Every address the host resolved to, each one vetted. A connection uses
314    /// one of these and nothing else.
315    pub addresses: Vec<SocketAddr>,
316}
317
318/// Resolve `url`'s host and judge every address under `policy`. Pure policy
319/// plus resolution: no LDAP traffic, no socket opened. The management routes
320/// call it before a configuration is saved; the connector calls it again at
321/// every connect.
322///
323/// # Errors
324///
325/// A [`GuardError`]: the URL has no usable host, is an IPv6 literal, the name
326/// does not resolve, or any address it resolves to is refused.
327pub async fn guard(
328    url: &str,
329    policy: &AddressPolicy,
330    resolver: &dyn Resolver,
331) -> Result<GuardedTarget, GuardError> {
332    let parsed = Url::parse(url).map_err(|_| GuardError::InvalidUrl)?;
333    let port = parsed.port().unwrap_or(if parsed.scheme() == "ldaps" {
334        LDAPS_PORT
335    } else {
336        LDAP_PORT
337    });
338    let (host, addresses) = match parsed.host() {
339        None => return Err(GuardError::InvalidUrl),
340        Some(Host::Domain("")) => return Err(GuardError::InvalidUrl),
341        Some(Host::Ipv6(_)) => return Err(GuardError::Ipv6Literal),
342        Some(Host::Ipv4(v4)) => (v4.to_string(), vec![SocketAddr::new(IpAddr::V4(v4), port)]),
343        // `url` parses IPv4 literals only for its "special" schemes (http and
344        // friends); under `ldap`/`ldaps` a dotted quad arrives as a domain. Use
345        // it as written rather than handing it to a resolver.
346        Some(Host::Domain(name)) if name.parse::<IpAddr>().is_ok() => {
347            let ip: IpAddr = name.parse().map_err(|_| GuardError::InvalidUrl)?;
348            if ip.is_ipv6() {
349                return Err(GuardError::Ipv6Literal);
350            }
351            (name.to_string(), vec![SocketAddr::new(ip, port)])
352        }
353        Some(Host::Domain(name)) => {
354            let addresses = resolver
355                .resolve(name, port)
356                .await
357                .map_err(|_| GuardError::Unresolvable)?;
358            (name.to_string(), addresses)
359        }
360    };
361    if addresses.is_empty() {
362        return Err(GuardError::Unresolvable);
363    }
364    if addresses.len() > MAX_RESOLVED_ADDRESSES {
365        return Err(GuardError::TooManyAddresses);
366    }
367    let mut vetted = Vec::with_capacity(addresses.len());
368    for address in addresses {
369        policy.check(address)?;
370        // Pin the canonical form: `::ffff:a.b.c.d` is dialled as `a.b.c.d`,
371        // the address that was judged.
372        let pinned = SocketAddr::new(canonical(address.ip()), port);
373        if !vetted.contains(&pinned) {
374            vetted.push(pinned);
375        }
376    }
377    Ok(GuardedTarget {
378        host,
379        port,
380        addresses: vetted,
381    })
382}

384/// The guard's inputs, shared by everything that opens a directory connection.
385#[derive(Clone)]
386pub struct NetworkGuard {
387    /// The deployment's policy.
388    pub policy: Arc<AddressPolicy>,
389    /// The resolver.
390    pub resolver: Arc<dyn Resolver>,
391}
392
393impl Default for NetworkGuard {
394    fn default() -> Self {
395        Self {
396            policy: Arc::new(AddressPolicy::new()),
397            resolver: Arc::new(SystemResolver),
398        }
399    }
400}
401
402impl NetworkGuard {
403    /// [`guard`] under this policy and resolver.
404    ///
405    /// # Errors
406    ///
407    /// See [`guard`].
408    pub async fn guard(&self, url: &str) -> Result<GuardedTarget, GuardError> {
409        guard(url, &self.policy, self.resolver.as_ref()).await
410    }
411}

426    /// Answers each name from a table; counts the questions.
427    struct Table {
428        answers: Vec<(&'static str, Vec<IpAddr>)>,
429        asked: Mutex<usize>,
430    }
431
432    impl Resolver for Table {
433        fn resolve<'a>(&'a self, host: &'a str, port: u16) -> ResolveFuture<'a> {
434            *self.asked.lock().unwrap() += 1;
435            let found = self
436                .answers
437                .iter()
438                .find(|(name, _)| *name == host)
439                .map(|(_, ips)| ips.iter().map(|ip| SocketAddr::new(*ip, port)).collect());
440            Box::pin(async move {
441                found.ok_or_else(|| io::Error::new(io::ErrorKind::NotFound, "no such name"))
442            })
443        }
444    }
445
446    fn table(answers: Vec<(&'static str, Vec<&str>)>) -> Table {
447        Table {
448            answers: answers
449                .into_iter()
450                .map(|(name, ips)| (name, ips.iter().map(|ip| ip.parse().unwrap()).collect()))
451                .collect(),
452            asked: Mutex::new(0),
453        }
454    }
455
456    fn strict() -> AddressPolicy {
457        AddressPolicy::new()
458    }

461        url: &str,
462        policy: &AddressPolicy,
463        resolver: &Table,
464    ) -> Result<(), GuardError> {
465        guard(url, policy, resolver).await.map(|_| ())
466    }

648    #[tokio::test]
649    async fn an_own_listener_port_on_a_local_address_is_refused_and_other_ports_are_not() {
650        // The loopback seam admits 127.0.0.1, so the listener rule is what is
651        // left to refuse AXIAM's own port on it.
652        let policy = AddressPolicy::new()
653            .admitting_loopback_for_tests()
654            .with_listener_ports([8090, 50051]);
655        let none = table(vec![]);
656        assert_eq!(
657            verdict("ldaps://127.0.0.1:8090", &policy, &none).await,
658            Err(GuardError::OwnListener)
659        );
660        assert_eq!(
661            verdict("ldap://127.0.0.1:50051", &policy, &none).await,
662            Err(GuardError::OwnListener)
663        );
664        assert_eq!(
665            verdict("ldaps://127.0.0.1:636", &policy, &none).await,
666            Ok(())
667        );
668        // A remote public address on the same port is somebody else's server.
669        assert_eq!(
670            verdict("ldaps://93.184.216.34:8090", &policy, &none).await,
671            Ok(())
672        );
673        assert_eq!(policy.listener_ports(), &[8090, 50051]);
674    }
