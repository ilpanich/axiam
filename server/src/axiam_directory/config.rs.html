1//! Configuration-time validation of a directory configuration.
2//!
3//! [`validate`] is a pure function over a [`NewDirectoryConfig`], so every
4//! refusal can be unit-tested without a directory and without a database. It
5//! is the single place the rules live: the REST handler (a later task) calls it
6//! before the repository is reached, and the repository, which sits a layer
7//! below this crate, deliberately does not repeat them.
8//!
9//! # What is refused, and why here
10//!
11//! * **Plaintext transport.** An `ldap://` URL without StartTLS, any other
12//!   scheme, and the contradictory `ldaps://` with StartTLS. A directory bind
13//!   carries the user's password, so a configuration that could ever send it in
14//!   the clear is refused when it is *saved*, not discovered at the first login.
15//! * **A URL that smuggles anything else:** userinfo (a credential in a URL is
16//!   a credential in a log), a path, a query, a fragment.
17//! * **A filter template that cannot be made safe.** The user filter must carry
18//!   exactly one `{username}` placeholder, in value position (directly after
19//!   `=`), inside a single balanced filter. The bind path substitutes the
20//!   RFC 4515-escaped login name there; the template is never formatted with raw
21//!   input. A placeholder in attribute-name position would make that escaping
22//!   meaningless, so it is refused.
23//! * **An empty bind secret.** A DN with an empty password is an RFC 4513
24//!   "unauthenticated bind", which many servers accept as success.
25//! * **Trust anchors that are not CA certificates.** A leaf pasted as an anchor
26//!   would pin one server certificate and fail at the next renewal, or, worse,
27//!   be mistaken for a CA by whoever reads the configuration.
28//!
29//! Error messages are fixed text. They never repeat the offending value: a URL
30//! may carry a password, and a refusal must not be the thing that logs it.
31//!
32//! # DN syntax is checked loosely on purpose
33//!
34//! A distinguished name is non-empty, free of control characters and bounded in
35//! length, and nothing more. In particular a DN is **not** required to contain
36//! `=`: Active Directory accepts a User Principal Name (`svc@corp.example.com`)
37//! as the bind identity, and a stricter check would refuse a common, correct
38//! configuration. Whether the server accepts the DN is learned by binding.

79/// Why a directory configuration was refused. The first defect found is
80/// reported; the messages are fixed text and never echo the offending value.
81#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
82pub enum ConfigError {
83    /// The server URL is unacceptable.
84    #[error("directory url: {0}")]
85    Url(UrlDefect),
86    /// A distinguished name is unacceptable.
87    #[error("directory {field}: {defect}")]
88    Dn {
89        /// The configuration field that carried the DN.
90        field: &'static str,
91        /// What is wrong with it.
92        defect: DnDefect,
93    },
94    /// The bind secret is unacceptable.
95    #[error("directory bind secret: {0}")]
96    BindSecret(SecretDefect),
97    /// A filter (the user filter template or the group filter) is unacceptable.
98    #[error("directory {field}: {defect}")]
99    Filter {
100        /// The configuration field that carried the filter.
101        field: &'static str,
102        /// What is wrong with it.
103        defect: FilterDefect,
104    },
105    /// An attribute name is unacceptable.
106    #[error("directory {field}: {defect}")]
107    Attribute {
108        /// The configuration field that carried the attribute name.
109        field: &'static str,
110        /// What is wrong with it.
111        defect: AttributeDefect,
112    },
113    /// The group nesting depth is outside `0..=GROUP_NESTING_DEPTH_MAX`.
114    #[error("directory group_nesting_depth must be between 0 and {GROUP_NESTING_DEPTH_MAX}")]
115    NestingDepth,
116    /// The sync interval is outside the permitted bounds.
117    #[error(
118        "directory sync_interval_secs must be between {SYNC_INTERVAL_MIN_SECS} and \
119         {SYNC_INTERVAL_MAX_SECS} seconds"
120    )]
121    SyncInterval,
122    /// More trust anchors than [`TRUST_ANCHORS_MAX`].
123    #[error("directory trust_anchors_pem may hold at most {TRUST_ANCHORS_MAX} certificates")]
124    TooManyTrustAnchors,
125    /// One trust anchor is unacceptable.
126    #[error("directory trust anchor #{index}: {defect}")]
127    TrustAnchor {
128        /// Zero-based position in `trust_anchors_pem`.
129        index: usize,
130        /// What is wrong with it.
131        defect: AnchorDefect,
132    },
133    /// The group-mapping table is unacceptable (T23.3.4, D-30).
134    #[error("directory group_mappings{}: {defect}", index.map(|i| format!(" #{i}")).unwrap_or_default())]
135    GroupMapping {
136        /// Zero-based position in `group_mappings`, when one row is at fault.
137        index: Option<usize>,
138        /// What is wrong with it.
139        defect: MappingDefect,
140    },
141}

143/// What is wrong with the group-mapping table. That every `group_id` is a
144/// group **of the tenant** is not here: [`validate`] is pure, so that rule
145/// lives in the repository's write path, and the management route (T23.3.8)
146/// calls both.
147#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
148pub enum MappingDefect {
149    /// More than [`GROUP_MAPPINGS_MAX`] rows.
150    #[error("may hold at most {GROUP_MAPPINGS_MAX} entries")]
151    TooMany,
152    /// The directory group's DN is empty, over-long or has a control character.
153    #[error("directory_group_dn is not acceptable (empty, too long or control characters)")]
154    BadDnText,
155    /// The directory group's DN is not an RFC 4514 distinguished name, so it
156    /// could never match anything and would only look like a mapping.
157    #[error("directory_group_dn is not a valid distinguished name: {0}")]
158    BadDn(crate::dn::DnError),
159    /// The same directory group (compared as the mapping compares) is mapped to
160    /// the same AXIAM group twice.
161    #[error("repeats an earlier entry (same directory group, same AXIAM group)")]
162    Duplicate,
163    /// A mapping needs `group_base_dn` where the directory is searched for the
164    /// groups a user is in (OpenLDAP's reverse `member` lookup): without it the
165    /// mapping could never be applied, and a sign-in would be refused.
166    #[error("needs group_base_dn for a directory that discovers groups by reverse member search")]
167    GroupBaseDnRequired,
168}

318/// Validate a directory configuration before it is stored.
319///
320/// Pure: no network, no database, no clock. Returns the first defect found.
321/// The secret is checked only when present, because an update that omits it
322/// keeps the stored one; that `create` requires a secret is the repository's
323/// rule.
324///
325/// # Errors
326///
327/// A [`ConfigError`] naming the field and the defect, never its value.
328pub fn validate(input: &NewDirectoryConfig) -> Result<(), ConfigError> {
329    validate_url(&input.url, input.start_tls).map_err(ConfigError::Url)?;
330    validate_dn("bind_dn", &input.bind_dn)?;
331    validate_dn("base_dn", &input.base_dn)?;
332    if let Some(secret) = &input.bind_secret {
333        if secret.is_empty() {
334            return Err(ConfigError::BindSecret(SecretDefect::Empty));
335        }
336        if secret.len() > BIND_SECRET_MAX_LEN {
337            return Err(ConfigError::BindSecret(SecretDefect::TooLong));
338        }
339    }
340    validate_user_filter(&input.user_filter)?;
341
342    let map = &input.user_attribute_map;
343    validate_attribute("user_attribute_map.username", &map.username)?;
344    validate_attribute("user_attribute_map.email", &map.email)?;
345    validate_attribute("user_attribute_map.display_name", &map.display_name)?;
346    validate_attribute("user_attribute_map.external_id", &map.external_id)?;
347
348    if let Some(group_base_dn) = &input.group_base_dn {
349        validate_dn("group_base_dn", group_base_dn)?;
350    }
351    if let Some(group_filter) = &input.group_filter {
352        validate_group_filter(group_filter)?;
353    }
354    validate_attribute("group_member_attribute", &input.group_member_attribute)?;
355    validate_group_mappings(input)?;
356
357    if input.group_nesting_depth > GROUP_NESTING_DEPTH_MAX {
358        return Err(ConfigError::NestingDepth);
359    }
360    if !(SYNC_INTERVAL_MIN_SECS..=SYNC_INTERVAL_MAX_SECS).contains(&input.sync_interval_secs) {
361        return Err(ConfigError::SyncInterval);
362    }
363
364    validate_trust_anchors(&input.trust_anchors_pem)
365}

367/// The mapping table's own rules (D-30): bounded, every DN an RFC 4514 DN, no
368/// redundant row, and — for a directory whose groups are found by searching —
369/// somewhere to search.
370fn validate_group_mappings(input: &NewDirectoryConfig) -> Result<(), ConfigError> {
371    let err = |index, defect| ConfigError::GroupMapping { index, defect };
372    let mappings = &input.group_mappings;
373    if mappings.len() > GROUP_MAPPINGS_MAX {
374        return Err(err(None, MappingDefect::TooMany));
375    }
376    if mappings.is_empty() {
377        return Ok(());
378    }
379    if input.kind.group_strategy() == GroupStrategy::ReverseMember && input.group_base_dn.is_none()
380    {
381        return Err(err(None, MappingDefect::GroupBaseDnRequired));
382    }
383    let mut seen = BTreeSet::new();
384    for (index, mapping) in mappings.iter().enumerate() {
385        let dn = &mapping.directory_group_dn;
386        if dn.trim().is_empty() || dn.len() > DN_MAX_LEN || dn.chars().any(char::is_control) {
387            return Err(err(Some(index), MappingDefect::BadDnText));
388        }
389        let key = crate::dn::normalize(dn)
390            .map_err(|defect| err(Some(index), MappingDefect::BadDn(defect)))?;
391        if !seen.insert((key, mapping.group_id)) {
392            return Err(err(Some(index), MappingDefect::Duplicate));
393        }
394    }
395    Ok(())
396}

398fn validate_url(raw: &str, start_tls: bool) -> Result<(), UrlDefect> {
399    if raw.len() > URL_MAX_LEN {
400        return Err(UrlDefect::TooLong);
401    }
402    // The URL parser strips surrounding whitespace and removes tabs and
403    // newlines anywhere. Accepting such text would store one string and
404    // connect to another, so it is refused rather than normalised.
405    if raw.chars().any(|c| c.is_whitespace() || c.is_control()) {
406        return Err(UrlDefect::WhitespaceOrControl);
407    }
408    let url = Url::parse(raw).map_err(|_| UrlDefect::Malformed)?;
409    let ldaps = match url.scheme() {
410        "ldaps" => true,
411        "ldap" => false,
412        _ => return Err(UrlDefect::UnsupportedScheme),
413    };
414    if url.host_str().is_none_or(str::is_empty) {
415        return Err(UrlDefect::MissingHost);
416    }
417    // Checked on the raw text as well as on the parsed URL: the parser drops an
418    // empty userinfo (`ldaps://@host`, `ldaps://:@host`), so the parsed form
419    // looks clean while the stored string, which is what a client would be
420    // handed, still carries the `@`.
421    let raw_authority = raw.split_once("://").map_or("", |(_, rest)| {
422        rest.split(['/', '?', '#']).next().unwrap_or("")
423    });
424    if raw_authority.contains('@') || url.authority().contains('@') {
425        return Err(UrlDefect::HasUserinfo);
426    }
427    if !matches!(url.path(), "" | "/") {
428        return Err(UrlDefect::HasPath);
429    }
430    if url.query().is_some() {
431        return Err(UrlDefect::HasQuery);
432    }
433    if url.fragment().is_some() {
434        return Err(UrlDefect::HasFragment);
435    }
436    match (ldaps, start_tls) {
437        (true, true) => Err(UrlDefect::StartTlsOnLdaps),
438        (false, false) => Err(UrlDefect::PlaintextWithoutStartTls),
439        _ => Ok(()),
440    }
441}

443fn validate_dn(field: &'static str, dn: &str) -> Result<(), ConfigError> {
444    let defect = if dn.trim().is_empty() {
445        DnDefect::Empty
446    } else if dn.chars().any(char::is_control) {
447        DnDefect::ControlCharacter
448    } else if dn.len() > DN_MAX_LEN {
449        DnDefect::TooLong
450    } else {
451        return Ok(());
452    };
453    Err(ConfigError::Dn { field, defect })
454}
455
456fn validate_attribute(field: &'static str, name: &str) -> Result<(), ConfigError> {
457    let defect = if name.is_empty() {
458        AttributeDefect::Empty
459    } else if name.len() > ATTRIBUTE_MAX_LEN {
460        AttributeDefect::TooLong
461    } else if !name.starts_with(|c: char| c.is_ascii_alphabetic())
462        || !name.chars().all(|c| c.is_ascii_alphanumeric() || c == '-')
463    {
464        AttributeDefect::InvalidCharacters
465    } else {
466        return Ok(());
467    };
468    Err(ConfigError::Attribute { field, defect })
469}

471/// The shape every filter must have: one parenthesised, balanced expression.
472fn check_filter_shape(filter: &str) -> Result<(), FilterDefect> {
473    if filter.is_empty() {
474        return Err(FilterDefect::Empty);
475    }
476    if filter.len() > FILTER_MAX_LEN {
477        return Err(FilterDefect::TooLong);
478    }
479    if filter.chars().any(char::is_control) {
480        return Err(FilterDefect::ControlCharacter);
481    }
482    if !filter.starts_with('(') || !filter.ends_with(')') {
483        return Err(FilterDefect::NotParenthesised);
484    }
485    // RFC 4515 writes a literal parenthesis inside a value as `\28` / `\29`,
486    // so every raw parenthesis is structural and a plain depth count is exact.
487    let last = filter.len() - 1;
488    let mut depth: usize = 0;
489    for (i, c) in filter.char_indices() {
490        match c {
491            '(' => depth += 1,
492            ')' => {
493                depth = depth.checked_sub(1).ok_or(FilterDefect::Unbalanced)?;
494                if depth == 0 && i != last {
495                    // Closed before the end: either a second filter follows or
496                    // the remainder is stray closing parentheses.
497                    return Err(if filter[i + 1..].contains('(') {
498                        FilterDefect::NotSingleFilter
499                    } else {
500                        FilterDefect::Unbalanced
501                    });
502                }
503            }
504            _ => {}
505        }
506    }
507    if depth != 0 {
508        return Err(FilterDefect::Unbalanced);
509    }
510    Ok(())
511}

513fn validate_user_filter(template: &str) -> Result<(), ConfigError> {
514    let err = |defect| ConfigError::Filter {
515        field: "user_filter",
516        defect,
517    };
518    check_filter_shape(template).map_err(err)?;
519
520    if template.matches(USERNAME_PLACEHOLDER).count() != 1 {
521        return Err(err(FilterDefect::PlaceholderCount));
522    }
523    let without = template.replacen(USERNAME_PLACEHOLDER, "", 1);
524    if without.contains(['{', '}']) {
525        return Err(err(FilterDefect::StrayBrace));
526    }
527    // Value position only. The one occurrence is guaranteed above, so `find`
528    // cannot miss; `unwrap_or(0)` keeps the function total without a panic.
529    let at = template.find(USERNAME_PLACEHOLDER).unwrap_or(0);
530    if !template[..at].ends_with('=') {
531        return Err(err(FilterDefect::PlaceholderPosition));
532    }
533    Ok(())
534}

536fn validate_group_filter(filter: &str) -> Result<(), ConfigError> {
537    let err = |defect| ConfigError::Filter {
538        field: "group_filter",
539        defect,
540    };
541    check_filter_shape(filter).map_err(err)?;
542    if filter.contains(['{', '}']) {
543        return Err(err(FilterDefect::PlaceholderNotAllowed));
544    }
545    Ok(())
546}

548fn validate_trust_anchors(anchors: &[String]) -> Result<(), ConfigError> {
549    if anchors.len() > TRUST_ANCHORS_MAX {
550        return Err(ConfigError::TooManyTrustAnchors);
551    }
552    for (index, pem) in anchors.iter().enumerate() {
553        validate_trust_anchor(pem).map_err(|defect| ConfigError::TrustAnchor { index, defect })?;
554    }
555    Ok(())
556}
557
558fn validate_trust_anchor(pem: &str) -> Result<(), AnchorDefect> {
559    if pem.len() > TRUST_ANCHOR_MAX_LEN {
560        return Err(AnchorDefect::TooLarge);
561    }
562    let (rest, block) =
563        x509_parser::pem::parse_x509_pem(pem.as_bytes()).map_err(|_| AnchorDefect::NotPem)?;
564    if block.label != "CERTIFICATE" {
565        return Err(AnchorDefect::NotCertificateBlock);
566    }
567    if !rest.iter().all(u8::is_ascii_whitespace) {
568        return Err(AnchorDefect::MoreThanOneBlock);
569    }
570    let (_, cert) =
571        X509Certificate::from_der(&block.contents).map_err(|_| AnchorDefect::Unparseable)?;
572    let is_ca = cert
573        .basic_constraints()
574        .ok()
575        .flatten()
576        .is_some_and(|bc| bc.value.ca);
577    if !is_ca {
578        return Err(AnchorDefect::NotCa);
579    }
580    Ok(())
581}

630    fn ca_pem() -> String {
631        let key = KeyPair::generate().unwrap();
632        let mut params = CertificateParams::new(Vec::<String>::new()).unwrap();
633        params.is_ca = IsCa::Ca(BasicConstraints::Unconstrained);
634        params.self_signed(&key).unwrap().pem()
635    }
636
637    fn leaf_pem() -> String {
638        let key = KeyPair::generate().unwrap();
639        let mut params = CertificateParams::new(vec!["ldap.example.com".to_string()]).unwrap();
640        params.is_ca = IsCa::NoCa;
641        params.self_signed(&key).unwrap().pem()
642    }
