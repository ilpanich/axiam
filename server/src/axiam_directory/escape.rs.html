1//! RFC 4515 filter-value escaping, and the one function that puts a login
2//! name into a search filter.
3//!
4//! # The rule
5//!
6//! A value that came from outside AXIAM — a login name typed into a form, or a
7//! distinguished name a directory returned — reaches an LDAP filter **only**
8//! through [`escape_filter_value`], and only through [`user_filter_for`] (which
9//! substitutes the escaped login name into the single `{username}` placeholder
10//! of the tenant's template) or [`reverse_member_filter`] (which does the same
11//! for the DNs of a reverse group-membership search), or, for the sync job
12//! (T23.3.5), [`external_id_filter`] and [`changed_since_filter`]. Nothing in this
13//! crate builds a filter with `format!` around raw input, and nothing builds a
14//! distinguished name at all: the DN the user binds as is the one the directory
15//! returned from the search, so RFC 4514 DN escaping is never needed and is not
16//! implemented (a DN that is never constructed cannot be injected into).
17//!
18//! # What is escaped
19//!
20//! RFC 4515 §3 requires the five octets `*`, `(`, `)`, `\` and NUL to be
21//! written as `\XX` inside an assertion value. This function escapes those
22//! **and every octet outside printable ASCII** (`0x20..=0x7E`): control
23//! characters and every byte of a multi-byte UTF-8 sequence. The RFC permits
24//! UTF-8 to appear literally, and `\XX` is equally valid for any octet, so the
25//! stricter form costs nothing in interoperability and buys an output that is
26//! pure printable ASCII — no byte in it can be read as filter syntax by a
27//! server, a proxy or a log viewer, whatever its notion of encoding.
28//!
29//! The server decodes each `\XX` back to the original octet, so `é`
30//! (`c3 a9`) is matched as `é`: escaping changes how the value is written,
31//! never which value is asserted.
32
33use crate::config::USERNAME_PLACEHOLDER;

35/// Longest login name, in bytes, that is ever sent to a directory.
36///
37/// Longer input is refused before any network I/O, as a generic failure. Real
38/// directory login names are short (`sAMAccountName` is at most 20 characters,
39/// a UPN at most a few hundred); the bound exists so a hostile caller cannot
40/// make AXIAM ship a multi-kilobyte filter, three times inflated by escaping,
41/// to a tenant's directory on every attempt.
42pub const LOGIN_NAME_MAX_LEN: usize = 256;
43
44/// Escape `value` for use as an RFC 4515 assertion value.
45///
46/// See the module documentation for exactly which octets are escaped. The
47/// output is always printable ASCII and never contains `*`, `(`, `)` or an
48/// unescaped `\`, so it cannot widen, close or extend the filter it is placed
49/// in.
50#[must_use]
51pub fn escape_filter_value(value: &str) -> String {
52    let mut out = String::with_capacity(value.len() + 8);
53    for &byte in value.as_bytes() {
54        let literal = matches!(byte, 0x20..=0x7E) && !matches!(byte, b'*' | b'(' | b')' | b'\\');
55        if literal {
56            out.push(char::from(byte));
57        } else {
58            out.push('\\');
59            out.push(hex_digit(byte >> 4));
60            out.push(hex_digit(byte & 0x0F));
61        }
62    }
63    out
64}
65
66fn hex_digit(nibble: u8) -> char {
67    char::from(match nibble {
68        0..=9 => b'0' + nibble,
69        _ => b'a' + (nibble - 10),
70    })
71}

73/// Why a user filter could not be built.
74#[derive(Debug, Clone, Copy, PartialEq, Eq)]
75pub enum UserFilterError {
76    /// The login name is empty or longer than [`LOGIN_NAME_MAX_LEN`]. Answered
77    /// as a generic sign-in failure, before any network I/O.
78    UnusableLoginName,
79    /// The stored template does not carry exactly one placeholder. Validation
80    /// refuses such a template at configuration time, so reaching this means
81    /// the row was written by something that skipped validation; it is a
82    /// misconfiguration, never a reason to send an unparameterised filter.
83    TemplateWithoutSinglePlaceholder,
84}
85
86/// Build the user-lookup filter: the tenant's template with its single
87/// `{username}` placeholder replaced by the RFC 4515-escaped login name.
88///
89/// This is the only way a login name enters a filter.
90///
91/// # Errors
92///
93/// [`UserFilterError`] when the login name is unusable or the template does not
94/// carry exactly one placeholder.
95pub fn user_filter_for(template: &str, login_name: &str) -> Result<String, UserFilterError> {
96    if login_name.is_empty() || login_name.len() > LOGIN_NAME_MAX_LEN {
97        return Err(UserFilterError::UnusableLoginName);
98    }
99    if template.matches(USERNAME_PLACEHOLDER).count() != 1 {
100        return Err(UserFilterError::TemplateWithoutSinglePlaceholder);
101    }
102    Ok(template.replacen(USERNAME_PLACEHOLDER, &escape_filter_value(login_name), 1))
103}

106/// the size of a filter built from directory-supplied names.
107pub const REVERSE_MEMBER_BATCH: usize = 16;

109/// Build the reverse group-membership filter (T23.3.4, D-30): the groups, under
110/// the group base, whose member attribute names one of `dns`.
111///
112/// ```text
113/// (&<group_filter>(<attribute>=<escaped dn>))            one DN
114/// (&<group_filter>(|(<attribute>=<dn>)(<attribute>=<dn>)))   several
115/// ```
116///
117/// Without a `group_filter` the `(&...)` wrapper is dropped. **Each DN enters
118/// the filter only through [`escape_filter_value`]**, exactly as a login name
119/// does: a user's DN is a directory-supplied string and may carry `*`, `(`,
120/// `)`, `\` or NUL, none of which can then widen the search. `group_filter` is
121/// the tenant's own, validated, static filter; `attribute` is checked to be a
122/// plain attribute name here as well, because it too is placed in the filter.
123///
124/// `None` when `dns` is empty or `attribute` is not a plain attribute name
125/// (letters, digits and `-`, starting with a letter) — the caller treats it as
126/// a misconfiguration and refuses; an unparameterised filter is never sent.
127#[must_use]
128pub fn reverse_member_filter(
129    group_filter: Option<&str>,
130    attribute: &str,
131    dns: &[&str],
132) -> Option<String> {
133    let plain = attribute.starts_with(|c: char| c.is_ascii_alphabetic())
134        && attribute
135            .chars()
136            .all(|c| c.is_ascii_alphanumeric() || c == '-');
137    if dns.is_empty() || !plain {
138        return None;
139    }
140    let clauses: Vec<String> = dns
141        .iter()
142        .map(|dn| format!("({attribute}={})", escape_filter_value(dn)))
143        .collect();
144    let membership = match clauses.as_slice() {
145        [only] => only.clone(),
146        many => format!("(|{})", many.concat()),
147    };
148    Some(match group_filter {
149        Some(group_filter) => format!("(&{group_filter}{membership})"),
150        None => membership,
151    })
152}

154/// Escape arbitrary octets for use as an RFC 4515 assertion value: **every**
155/// byte is written as `\XX`, so the output is printable ASCII whatever the
156/// bytes were. For binary attributes (`objectGUID`), where the value is not
157/// text and no octet may be left to be read as filter syntax or as part of a
158/// multi-byte character.
159#[must_use]
160pub fn escape_filter_bytes(bytes: &[u8]) -> String {
161    let mut out = String::with_capacity(bytes.len() * 3);
162    for &byte in bytes {
163        out.push('\\');
164        out.push(hex_digit(byte >> 4));
165        out.push(hex_digit(byte & 0x0F));
166    }
167    out
168}

170/// A plain attribute name — letters, digits and `-`, starting with a letter —
171/// the only kind that is ever placed in a filter.
172fn is_plain_attribute(attribute: &str) -> bool {
173    attribute.starts_with(|c: char| c.is_ascii_alphabetic())
174        && attribute
175            .chars()
176            .all(|c| c.is_ascii_alphanumeric() || c == '-')
177}

179/// Longest external identifier that is looked up, in bytes (the client's own
180/// bound on what it accepts from a directory).
181const EXTERNAL_ID_MAX_LEN: usize = 256;
182
183/// Build the filter that finds the one entry whose immutable identifier is
184/// `external_id` (T23.3.5, D-31): `(<attribute>=<value>)`.
185///
186/// * `objectGUID` (Active Directory) is **binary**: the stored identifier is
187///   the canonical text of the GUID, and the directory matches the 16 raw
188///   octets in Microsoft's mixed-endian layout, so the text is parsed and
189///   written back with `Uuid::to_bytes_le` — the inverse of the decoding the
190///   client applies to what the directory returns — and every octet is escaped
191///   by [`escape_filter_bytes`].
192/// * Any other attribute (`entryUUID`, or whatever the tenant mapped) carries
193///   text, which enters the filter through [`escape_filter_value`].
194///
195/// **This, with [`user_filter_for`], [`reverse_member_filter`] and
196/// [`changed_since_filter`], is the only way a value reaches a filter.** `None`
197/// when the attribute is not a plain name, the identifier is empty or overlong,
198/// or (for `objectGUID`) is not a GUID: the caller treats that as "cannot ask",
199/// never as "not found".
200#[must_use]
201pub fn external_id_filter(attribute: &str, external_id: &str) -> Option<String> {
202    if !is_plain_attribute(attribute) || external_id.is_empty() {
203        return None;
204    }
205    if attribute.eq_ignore_ascii_case("objectGUID") {
206        let guid = uuid::Uuid::parse_str(external_id.trim()).ok()?;
207        return Some(format!(
208            "({attribute}={})",
209            escape_filter_bytes(&guid.to_bytes_le())
210        ));
211    }
212    if external_id.len() > EXTERNAL_ID_MAX_LEN {
213        return None;
214    }
215    Some(format!(
216        "({attribute}={})",
217        escape_filter_value(external_id)
218    ))
219}

221/// Whether `value` is an LDAP generalized time as `modifyTimestamp` carries it:
222/// fourteen digits, an optional fraction, and `Z` (`20261003120000Z`,
223/// `20261003120000.0Z`).
224#[must_use]
225pub fn is_generalized_time(value: &str) -> bool {
226    let Some(body) = value.strip_suffix('Z') else {
227        return false;
228    };
229    let (whole, fraction) = match body.split_once(['.', ',']) {
230        Some((whole, fraction)) => (whole, Some(fraction)),
231        None => (body, None),
232    };
233    whole.len() == 14
234        && whole.bytes().all(|b| b.is_ascii_digit())
235        && fraction
236            .is_none_or(|f| !f.is_empty() && f.len() <= 6 && f.bytes().all(|b| b.is_ascii_digit()))
237}

239/// Whether `value` is an update sequence number: one to twenty decimal digits.
240#[must_use]
241pub fn is_usn(value: &str) -> bool {
242    (1..=20).contains(&value.len()) && value.bytes().all(|b| b.is_ascii_digit())
243}
244
245/// Build the incremental-sync filter (T23.3.5, D-31):
246/// `(<attribute>>=<watermark>)`.
247///
248/// The watermark is a value AXIAM stored from the directory's own answer and
249/// still never reaches a filter unchecked: it must be a generalized time or a
250/// USN, and passes through [`escape_filter_value`] as well. `None` for an
251/// attribute that is not a plain name or a watermark that is neither.
252#[must_use]
253pub fn changed_since_filter(attribute: &str, watermark: &str) -> Option<String> {
254    if !is_plain_attribute(attribute) || !(is_generalized_time(watermark) || is_usn(watermark)) {
255        return None;
256    }
257    Some(format!("({attribute}>={})", escape_filter_value(watermark)))
258}

314    /// The structural property, over a wide sample: no output ever contains a
315    /// byte that could be filter syntax, and an escape is always a full `\XX`.
316    #[test]
317    fn no_output_contains_a_syntax_byte() {
318        let mut inputs: Vec<String> = (0u8..=0x7F).map(|b| char::from(b).to_string()).collect();
319        inputs.push("x".repeat(LOGIN_NAME_MAX_LEN * 4));
320        inputs.push("é*)(\\\0ü".repeat(50));
321        for input in inputs {
322            let out = escape_filter_value(&input);
323            assert!(out.bytes().all(|b| (0x20..=0x7E).contains(&b)));
324            assert!(!out.contains(['*', '(', ')']));
325            let bytes = out.as_bytes();
326            let mut i = 0;
327            while i < bytes.len() {
328                if bytes[i] == b'\\' {
329                    assert!(i + 2 < bytes.len(), "a truncated escape");
330                    assert!(bytes[i + 1].is_ascii_hexdigit() && bytes[i + 2].is_ascii_hexdigit());
331                    i += 3;
332                } else {
333                    i += 1;
334                }
335            }
336        }
337    }
