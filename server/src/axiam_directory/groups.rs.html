1//! Which directory groups a user is in, and which AXIAM groups that maps to
2//! (G-3, T23.3.4, D-30).
3//!
4//! This module is the **pure half**: the nested walk over an abstract
5//! [`GroupParents`] source, and the pass of the mapping table over what the
6//! walk found. The LDAP half — the two sources, `memberOf` for Active
7//! Directory and the reverse `member` search for OpenLDAP — is
8//! [`crate::client::DirectoryClient::resolve_groups`]; applying the result to
9//! AXIAM's memberships is [`crate::mapper`].
10//!
11//! # The walk
12//!
13//! Level 0 is the user's own groups (their entry's `memberOf`, or the groups
14//! that list the user's DN as a member). Each further level is the groups that
15//! contain a group found at the previous one, to
16//! [`DirectoryConfig::group_nesting_depth`]
17//! levels: depth `0` follows no nesting, depth `N` follows `N` levels, and
18//! level `N + 1` is never read.
19//!
20//! * **A cycle terminates.** A group is expanded once, whatever it is named
21//!   as: groups are keyed by their [normalised DN](crate::dn::normalize), so a
22//!   group that contains itself, or two that contain each other, or one that
23//!   contains the user, adds nothing the second time round.
24//! * **A hard cap of [`MAX_GROUPS_PER_USER`] distinct groups.** One more is
25//!   [`ResolveError::CapExceeded`], not a truncated answer: a user in more
26//!   groups than that is either a hostile directory or one this feature was not
27//!   sized for, and the answer to both is to refuse the sign-in rather than map
28//!   a prefix of the groups.
29//! * **A DN that cannot be read is skipped** — not traversed, not counted, and
30//!   unmappable anyway, since a mapping DN must parse.
31//!
32//! # Fail closed
33//!
34//! Every error here ends the whole resolution. A partial set is never
35//! returned: the caller removes memberships the directory no longer backs, and
36//! "no longer backs" read off half a walk would be wrong in the dangerous
37//! direction (a lost removal keeps access; a lost addition only withholds it,
38//! but the walk cannot tell which it is looking at).
39//!
40//! [`DirectoryConfig::group_nesting_depth`]: axiam_core::models::directory::DirectoryConfig::group_nesting_depth
41

50/// Most distinct directory groups one user may belong to, nesting included.
51pub const MAX_GROUPS_PER_USER: usize = 1_000;
52
53/// Why a resolution produced no answer.
54#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
55pub enum ResolveError {
56    /// The user is in more than [`MAX_GROUPS_PER_USER`] groups.
57    #[error("the user belongs to more directory groups than the cap allows")]
58    CapExceeded,
59    /// The directory could not be asked, or answered something unusable. The
60    /// reason is fixed text, safe for a log line.
61    #[error("the directory group lookup failed: {0}")]
62    Lookup(&'static str),
63}
64
65/// What the directory says about the groups containing a set of entries.
66pub trait GroupParents {
67    /// The DNs of the groups that **directly** contain any entry of `of` (the
68    /// user's DN, then group DNs), as the directory wrote them. May repeat a
69    /// DN; the walk deduplicates.
70    ///
71    /// # Errors
72    ///
73    /// Any [`ResolveError`]; the walk stops at the first.
74    fn parents_of<'a>(
75        &'a mut self,
76        of: &'a [String],
77    ) -> impl Future<Output = Result<Vec<String>, ResolveError>> + Send + 'a;
78}

80/// The groups a user belongs to, as the walk found them.
81#[derive(Debug, Clone, Default, PartialEq, Eq)]
82pub struct ResolvedGroups {
83    /// Each distinct group's DN as the directory wrote it, in the order found.
84    pub dns: Vec<String>,
85    /// The same groups as [normalised](crate::dn::normalize) comparison keys.
86    pub keys: BTreeSet<String>,
87    /// DNs the directory returned that could not be read as DNs and were
88    /// skipped.
89    pub skipped_unreadable: usize,
90}
91
92impl ResolvedGroups {
93    /// How many distinct groups were found.
94    #[must_use]
95    pub fn len(&self) -> usize {
96        self.dns.len()
97    }
98
99    /// Whether no group was found.
100    #[must_use]
101    pub fn is_empty(&self) -> bool {
102        self.dns.is_empty()
103    }
104}

106/// Walk the nesting above `user_dn` through `source`.
107///
108/// `max_depth` is the tenant's `group_nesting_depth`; `cap` is
109/// [`MAX_GROUPS_PER_USER`] (a parameter so the tests can state the boundary
110/// without a thousand fixtures).
111///
112/// # Errors
113///
114/// [`ResolveError::CapExceeded`] when more than `cap` distinct groups turn up,
115/// and whatever `source` returns.
116pub async fn resolve_nested<S: GroupParents>(
117    source: &mut S,
118    user_dn: &str,
119    max_depth: u8,
120    cap: usize,
121) -> Result<ResolvedGroups, ResolveError> {
122    let mut found = ResolvedGroups::default();
123    let mut seen: BTreeSet<String> = BTreeSet::new();
124    // The user is never a group of themselves, however a loop reaches them.
125    if let Ok(key) = normalize(user_dn) {
126        seen.insert(key);
127    }
128    let mut frontier = vec![user_dn.to_string()];
129    for _level in 0..=max_depth {
130        let parents = source.parents_of(&frontier).await?;
131        let mut next = Vec::new();
132        for dn in parents {
133            let Ok(key) = normalize(&dn) else {
134                found.skipped_unreadable += 1;
135                continue;
136            };
137            if !seen.insert(key.clone()) {
138                continue;
139            }
140            if found.dns.len() >= cap {
141                return Err(ResolveError::CapExceeded);
142            }
143            found.keys.insert(key);
144            found.dns.push(dn.clone());
145            next.push(dn);
146        }
147        if next.is_empty() {
148            break;
149        }
150        frontier = next;
151    }
152    Ok(found)
153}

155/// The AXIAM groups the mapping table puts a member of `resolved` into.
156///
157/// A row matches when its DN, [normalised](crate::dn::normalize), is one of the
158/// resolved groups' keys — and in no other way: not by name, not by prefix, not
159/// by a parent, not by a wildcard. A row whose DN cannot be read matches
160/// nothing (validation refuses such a row at write; this is the read path not
161/// trusting that it did).
162#[must_use]
163pub fn mapped_group_ids(mappings: &[GroupMapping], resolved: &ResolvedGroups) -> BTreeSet<Uuid> {
164    mappings
165        .iter()
166        .filter(|mapping| {
167            normalize(&mapping.directory_group_dn).is_ok_and(|key| resolved.keys.contains(&key))
168        })
169        .map(|mapping| mapping.group_id)
170        .collect()
171}

173/// What [`crate::client::DirectoryClient::resolve_groups`] needs to know about
174/// a tenant's group layout. Resolved from the stored configuration by the
175/// caller, like [`crate::client::DirectoryTarget`].
176#[derive(Debug, Clone, PartialEq, Eq)]
177pub struct GroupLookup {
178    /// How groups are discovered: off the entry (`memberOf`) or by a reverse
179    /// search (`member`).
180    pub strategy: GroupStrategy,
181    /// Where the reverse search runs. Required for
182    /// [`GroupStrategy::ReverseMember`], unused for `MemberOf`.
183    pub base_dn: Option<String>,
184    /// The tenant's static group filter, if it set one.
185    pub filter: Option<String>,
186    /// `memberOf` or `member` (or `uniqueMember`, `isMemberOf`, ...).
187    pub member_attribute: String,
188    /// The tenant's `group_nesting_depth`.
189    pub max_depth: u8,
190}

214                            parents.iter().map(|p| (*p).to_string()).collect(),

224    impl GroupParents for Fixture {
225        async fn parents_of(&mut self, of: &[String]) -> Result<Vec<String>, ResolveError> {
226            {
227                self.calls.push(of.to_vec());
228                if self.fail_on_call == Some(self.calls.len()) {
229                    return Err(ResolveError::Lookup("the fixture failed on purpose"));
230                }
231                Ok(of
232                    .iter()
233                    .flat_map(|dn| {
234                        self.parents
235                            .get(&normalize(dn).unwrap())
236                            .cloned()
237                            .unwrap_or_default()
238                    })
239                    .collect())
240            }
241        }
242    }

272    #[tokio::test]
273    async fn nesting_is_followed_to_depth_n_and_level_n_plus_one_is_not_read() {
274        for depth in 0u8..=3 {
275            let mut source = chain();
276            let found = resolve_nested(&mut source, USER, depth, 1000)
277                .await
278                .unwrap();
279            let want: Vec<String> = (1..=usize::from(depth) + 1)
280                .map(|n| format!("cn=g{n}"))
281                .collect();
282            assert_eq!(names(&found), want, "depth {depth}");
283            assert!(
284                !found
285                    .dns
286                    .iter()
287                    .any(|dn| dn.starts_with(&format!("cn=g{},", depth + 2))),
288                "the group at level {} must not be found at depth {depth}",
289                depth + 1
290            );
291            // One read for the user's groups, one per nesting level, none more.
292            assert_eq!(source.calls.len(), usize::from(depth) + 1, "depth {depth}");
293        }

410    fn resolved(dns: &[&str]) -> ResolvedGroups {
411        ResolvedGroups {
412            dns: dns.iter().map(|d| (*d).to_string()).collect(),
413            keys: dns.iter().map(|d| normalize(d).unwrap()).collect(),
414            skipped_unreadable: 0,
415        }
416    }
