1//! The TLS client configuration every directory connection uses.
2//!
3//! # Mandatory, and verified
4//!
5//! There is no code path in this crate that opens an LDAP connection without
6//! TLS, and none that turns verification off: `ldap3`'s `no_tls_verify` switch
7//! is never set, and the [`rustls::ClientConfig`] built here uses rustls' own
8//! WebPKI verifier over the tenant's anchors. The server name checked is the
9//! host of the configured URL — `ldap3` derives it from the same URL it dials —
10//! so a certificate for another host is refused even when it chains to a
11//! trusted anchor.
12//!
13//! # Whose anchors
14//!
15//! A tenant's `trust_anchors_pem` (validated at configuration time to be CA
16//! certificates) is the **whole** trust store for that tenant's directory: a
17//! private corporate CA — the organisation's own AXIAM CA included — is the
18//! common case, and adding the public roots beside it would let any public CA
19//! vouch for the directory's name. An empty list means the Mozilla bundle in
20//! `webpki-roots`, which is what the rest of the workspace's outbound TLS
21//! (`reqwest`'s `rustls-tls`, `lettre`) already trusts.
22//!
23//! # TLS 1.2 minimum, not 1.3 only
24//!
25//! The protocol floor toward a directory is **TLS 1.2**, with 1.3 preferred
26//! when the server offers it. AXIAM's own listeners pin 1.3, but a directory is
27//! somebody else's server: Active Directory domain controllers on Windows
28//! Server 2019 and earlier do not speak TLS 1.3 on LDAPS at all, and many
29//! OpenLDAP builds in the field are linked against TLS stacks that stop at 1.2.
30//! A 1.3-only client would make the feature unusable against exactly the
31//! directories it exists to federate. TLS 1.2 under rustls is still restricted
32//! to AEAD suites with forward secrecy (no RSA key exchange, no CBC, no RC4),
33//! so the floor is 1.2-done-well rather than anything older.
34//!
35//! # Crypto provider
36//!
37//! The configuration names the `ring` provider explicitly rather than reading
38//! the process default. The workspace links both `ring` and `aws-lc-rs`, so a
39//! process that never installed a default (a test binary, an embedder) would
40//! otherwise panic at the first handshake; `ring` matches the REST listener and
41//! the AMQP client.
42
43use std::sync::Arc;

45use rustls::pki_types::CertificateDer;
46use rustls::pki_types::pem::PemObject;
47use rustls::{ClientConfig, RootCertStore};

49/// Why a TLS client configuration could not be built from a tenant's anchors.
50///
51/// Configuration-time validation refuses every one of these, so reaching one
52/// means the stored row was not validated. The bind path treats it as a
53/// misconfigured directory and fails closed — it never falls back to another
54/// trust store.
55#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
56pub enum TlsSetupError {
57    /// An anchor entry held no PEM certificate.
58    #[error("directory trust anchor #{0} holds no certificate")]
59    NoCertificate(usize),
60    /// An anchor was refused by rustls as a trust anchor.
61    #[error("directory trust anchor #{0} is not usable as a trust anchor")]
62    Unusable(usize),
63    /// The protocol versions could not be applied (a rustls build without
64    /// TLS 1.2 or 1.3 support; not reachable with this workspace's features).
65    #[error("directory TLS configuration could not be built")]
66    Protocol,
67}
68
69/// Build the client configuration for a tenant's directory from its trust
70/// anchors.
71///
72/// # Errors
73///
74/// [`TlsSetupError`] when an anchor cannot be used. An empty list is not an
75/// error: it selects the `webpki-roots` bundle.
76pub fn client_config(trust_anchors_pem: &[String]) -> Result<Arc<ClientConfig>, TlsSetupError> {
77    let roots = root_store(trust_anchors_pem)?;
78    let provider = Arc::new(rustls::crypto::ring::default_provider());
79    let config = ClientConfig::builder_with_provider(provider)
80        .with_protocol_versions(&[&rustls::version::TLS13, &rustls::version::TLS12])
81        .map_err(|_| TlsSetupError::Protocol)?
82        .with_root_certificates(roots)
83        .with_no_client_auth();
84    Ok(Arc::new(config))
85}
86
87/// The trust store for one tenant: its anchors, or the public bundle when it
88/// configured none. Never both.
89fn root_store(trust_anchors_pem: &[String]) -> Result<RootCertStore, TlsSetupError> {
90    let mut roots = RootCertStore::empty();
91    if trust_anchors_pem.is_empty() {
92        roots.extend(webpki_roots::TLS_SERVER_ROOTS.iter().cloned());
93        return Ok(roots);
94    }
95    for (index, pem) in trust_anchors_pem.iter().enumerate() {
96        let mut found = false;
97        for cert in CertificateDer::pem_slice_iter(pem.as_bytes()) {
98            let cert = cert.map_err(|_| TlsSetupError::NoCertificate(index))?;
99            roots
100                .add(cert)
101                .map_err(|_| TlsSetupError::Unusable(index))?;
102            found = true;
103        }
104        if !found {
105            return Err(TlsSetupError::NoCertificate(index));
106        }
107    }
108    Ok(roots)
109}

111#[cfg(test)]
112mod tests {
113    use super::*;
114    use rcgen::{BasicConstraints, CertificateParams, IsCa, KeyPair};
115
116    fn ca_pem() -> String {
117        let key = KeyPair::generate().unwrap();
118        let mut params = CertificateParams::new(Vec::<String>::new()).unwrap();
119        params.is_ca = IsCa::Ca(BasicConstraints::Unconstrained);
120        params.self_signed(&key).unwrap().pem()
121    }
122
123    #[test]
124    fn an_empty_anchor_list_selects_the_public_bundle() {
125        let roots = root_store(&[]).unwrap();
126        assert_eq!(roots.len(), webpki_roots::TLS_SERVER_ROOTS.len());
127        assert!(client_config(&[]).is_ok());
128    }
129
130    /// Configured anchors replace the public bundle rather than joining it: a
131    /// tenant that pinned its corporate CA must not also trust every public CA
132    /// to vouch for its directory's name.
133    #[test]
134    fn configured_anchors_are_the_whole_trust_store() {
135        let roots = root_store(&[ca_pem(), ca_pem()]).unwrap();
136        assert_eq!(roots.len(), 2);
137    }
138
139    #[test]
140    fn an_entry_with_no_certificate_fails_closed() {
141        assert_eq!(
142            root_store(&[ca_pem(), "not pem".into()]).unwrap_err(),
143            TlsSetupError::NoCertificate(1)
144        );
145        assert!(client_config(&["".into()]).is_err());
146    }
147
148    #[test]
149    fn a_corrupt_certificate_fails_closed() {
150        let pem = "-----BEGIN CERTIFICATE-----\nAAAAAAAA\n-----END CERTIFICATE-----\n";
151        assert!(root_store(&[pem.into()]).is_err());
152    }
153
154    /// No client certificate is ever presented to a directory: AXIAM
155    /// authenticates to it with the bind DN over the verified channel, and a
156    /// configuration that could present one would be a credential this crate
157    /// does not manage. (That TLS 1.2 is accepted and verification is enforced
158    /// is pinned end to end in `tests/client_test.rs`, against a live server.)
159    #[test]
160    fn no_client_certificate_is_configured() {
161        let config = client_config(&[]).unwrap();
162        assert!(!config.client_auth_cert_resolver.has_certs());
163        assert!(!config.key_log.will_log("CLIENT_RANDOM"));
164    }
165}
