1//! Where an email provider may connect (#529, P23W3-11, T-473).
2//!
3//! An organization or tenant administrator chooses the provider: an SMTP host
4//! and port, or an HTTP provider's `api_url`. Both used to be dialled as
5//! written — `lettre` resolved the SMTP host itself and `reqwest` the API URL —
6//! so the email configuration was T-300's problem again: a field that makes
7//! AXIAM open a connection to loopback, the cloud metadata service or the pod
8//! network, and send an SMTP greeting or an HTTP `POST` carrying the provider
9//! key there.
10//!
11//! * **SMTP** is held to the connector address guard the directory uses
12//!   ([`axiam_pki::address`], D-32): the host is resolved once; loopback,
13//!   link-local (the metadata service), unspecified, multicast and
14//!   special-purpose addresses and AXIAM's own listeners are always refused; a
15//!   private address only inside the networks the operator lists in
16//!   [`ALLOWED_PRIVATE_NETWORKS_ENV`]. The SMTP connection is opened to the
17//!   vetted address, with the configured host as the TLS name.
18//! * **HTTP providers** go through [`guarded_fetch_no_redirect`] — the API key
19//!   is a credential, so no redirect is ever followed — and are held to its
20//!   rule: HTTPS, globally routable, an exception only for a host the operator
21//!   names in `AXIAM__PKI__SSRF_ALLOWED_HOSTS` (SEC-107).
22//!
23//! The same check runs when a configuration is saved ([`EmailEgress::check`])
24//! and at every send, so a name re-pointed after the save is caught on the next
25//! message. What an administrator is told about a refusal of a host **name** is
26//! one sentence whatever the name resolved to ([`HOST_NOT_PERMITTED`], the
27//! P23W3-04 lesson); the specific rule goes to the operator's log. A connection
28//! that fails after the guard admitted the address is reported as
29//! [`PROVIDER_UNREACHABLE`], so a refused port and a silent one read the same.
30

42/// The environment variable that lists the private networks an SMTP provider
43/// may be in: comma-separated CIDR blocks (`10.20.0.0/16`, `fd12:3456::/48`)
44/// or single addresses. Unset or empty — the default — admits no private
45/// address at all. Separate from the directory's list on purpose: where mail
46/// relays live and where directories live are two decisions.
47pub const ALLOWED_PRIVATE_NETWORKS_ENV: &str = "AXIAM__EMAIL__ALLOWED_PRIVATE_NETWORKS";

49/// The one answer every resolution-dependent refusal of a host **name** gets
50/// (P23W3-04): it says nothing about which names exist or where they point.
51pub const HOST_NOT_PERMITTED: &str = "the email provider host does not resolve to an address \
52     this deployment permits an email connection to; check the name, or ask the deployment's \
53     operator whether its network is allowed";
54
55/// The one answer every connection failure after the guard gets: refused,
56/// reset, timed out and a failed TLS handshake read the same. The operator's
57/// log has the cause.
58pub const PROVIDER_UNREACHABLE: &str =
59    "the connection to the email provider failed; the deployment's log names the cause";

61/// A refused provider address: what the administrator is told and what the
62/// operator's log records.
63#[derive(Debug, Clone, PartialEq, Eq)]
64pub struct EgressRefusal {
65    /// The field refused: `smtp host` or `api_url`.
66    pub field: &'static str,
67    /// The specific rule, for the operator's log (never the administrator's
68    /// answer when the host is a name).
69    pub rule: &'static str,
70    detail: String,
71    about_the_text: bool,
72}
73
74impl EgressRefusal {
75    fn smtp(host: &str, error: &GuardError) -> Self {
76        Self {
77            field: "smtp host",
78            rule: error.rule(),
79            detail: error.to_string(),
80            about_the_text: matches!(error, GuardError::InvalidUrl) || is_ip_literal(host),
81        }
82    }
83
84    fn api_url(host: Option<&str>, error: &SsrfError) -> Self {
85        let (rule, about_the_text) = match error {
86            SsrfError::InvalidUrl => ("ssrf.invalid_url", true),
87            SsrfError::InsecureScheme => ("ssrf.insecure_scheme", true),
88            SsrfError::ResolveFailed => ("ssrf.unresolvable", false),
89            _ => ("ssrf.blocked", false),
90        };
91        Self {
92            field: "api_url",
93            rule,
94            detail: error.to_string(),
95            about_the_text: about_the_text || host.is_some_and(is_ip_literal),
96        }
97    }
98
99    /// What the administrator is told. For a host **name**, every refusal that
100    /// depends on what it resolved to is [`HOST_NOT_PERMITTED`]; for an IP
101    /// literal, and for a value refused for its own text, the specific rule —
102    /// it reveals nothing the administrator did not type.
103    #[must_use]
104    pub fn public_message(&self) -> String {
105        if self.about_the_text {
106            format!("{}: {}", self.field, self.detail)
107        } else {
108            format!("{}: {HOST_NOT_PERMITTED}", self.field)
109        }
110    }
111
112    /// The refusal at send time, as the error the caller sees.
113    #[must_use]
114    pub fn into_error(self) -> AxiamError {
115        AxiamError::EmailConfig(self.public_message())
116    }
117
118    fn log(&self, when: &'static str) {
119        tracing::warn!(
120            target: "axiam::email",
121            field = self.field,
122            rule = self.rule,
123            detail = %self.detail,
124            when,
125            "an email provider address was refused by the outbound address policy"
126        );
127    }
128}

130/// The deployment's outbound rule for email providers. Built once at
131/// composition from deployment configuration; tenants cannot change it.
132/// Cheap to clone.
133#[derive(Clone)]
134pub struct EmailEgress {
135    policy: Arc<AddressPolicy>,
136    resolver: Arc<dyn Resolver>,
137    allow_private_http: bool,
138}
139
140impl Default for EmailEgress {
141    /// The strict rule: no private network, no listener known, the system
142    /// resolver.
143    fn default() -> Self {
144        Self::new(AddressPolicy::new())
145    }
146}
147
148impl std::fmt::Debug for EmailEgress {
149    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
150        f.debug_struct("EmailEgress")
151            .field("policy", &self.policy)
152            .field("allow_private_http", &self.allow_private_http)
153            .finish_non_exhaustive()
154    }
155}

157impl EmailEgress {
158    /// SMTP held to `policy` (the operator's allow-list and AXIAM's listener
159    /// ports), names resolved by the system resolver.
160    #[must_use]
161    pub fn new(policy: AddressPolicy) -> Self {
162        Self {
163            policy: Arc::new(policy),
164            resolver: Arc::new(SystemResolver),
165            allow_private_http: false,
166        }
167    }
168
169    /// Resolve SMTP host names with `resolver` (tests script answers).
170    #[must_use]
171    pub fn with_resolver(mut self, resolver: Arc<dyn Resolver>) -> Self {
172        self.resolver = resolver;
173        self
174    }
175
176    /// **Tests only.** Let an HTTP provider reach a plain-`http` loopback mock
177    /// server — `guarded_fetch`'s `allow_private` seam. Never called by
178    /// production code; a grep for this name finds every exception.
179    #[doc(hidden)]
180    #[must_use]
181    pub fn allowing_private_http_for_tests(mut self) -> Self {
182        self.allow_private_http = true;
183        self
184    }
185
186    /// The SMTP address policy.
187    #[must_use]
188    pub fn policy(&self) -> &AddressPolicy {
189        &self.policy
190    }
191
192    /// Resolve the SMTP `host` once and vet every address.
193    ///
194    /// # Errors
195    ///
196    /// The refusal, when the host is empty, does not resolve, or resolves to
197    /// an address the policy refuses.
198    pub async fn guard_smtp(&self, host: &str, port: u16) -> Result<GuardedTarget, EgressRefusal> {
199        guard_host(host, port, &self.policy, self.resolver.as_ref())
200            .await
201            .map_err(|error| EgressRefusal::smtp(host, &error))
202    }
203
204    /// The save-time check of an HTTP provider's `api_url`: the rule
205    /// [`guarded_fetch_no_redirect`] applies at send — an `https` URL whose
206    /// host resolves to globally routable addresses only (or is on
207    /// `AXIAM__PKI__SSRF_ALLOWED_HOSTS`). Nothing is sent.
208    ///
209    /// # Errors
210    ///
211    /// The refusal.
212    pub async fn check_api_url(&self, api_url: &str) -> Result<(), EgressRefusal> {
213        let refuse = |host: Option<&str>, error: SsrfError| EgressRefusal::api_url(host, &error);
214        let parsed = url::Url::parse(api_url).map_err(|_| refuse(None, SsrfError::InvalidUrl))?;
215        let Some(host) = parsed.host_str().filter(|h| !h.is_empty()) else {
216            return Err(refuse(None, SsrfError::InvalidUrl));
217        };
218        if parsed.scheme() != "https" && !self.allow_private_http {
219            return Err(refuse(Some(host), SsrfError::InsecureScheme));
220        }
221        let port = parsed.port_or_known_default().unwrap_or(443);
222        // The host exactly as the send path hands it to the resolver, so the
223        // save and the send cannot disagree.
224        resolve_and_pick(host, port, self.allow_private_http)
225            .await
226            .map(|_| ())
227            .map_err(|error| refuse(Some(host), error))
228    }
229
230    /// The save-time check of a provider configuration: the SMTP host through
231    /// the address guard, an explicit `api_url` through the HTTP rule. A
232    /// provider's built-in API URL is AXIAM's own constant and is not checked.
233    ///
234    /// # Errors
235    ///
236    /// The refusal; the caller answers [`EgressRefusal::public_message`].
237    pub async fn check(&self, provider: &ProviderConfig) -> Result<(), EgressRefusal> {
238        let result = match provider {
239            ProviderConfig::Smtp(smtp) => self.guard_smtp(&smtp.host, smtp.port).await.map(|_| ()),
240            ProviderConfig::SendGrid(api)
241            | ProviderConfig::Postmark(api)
242            | ProviderConfig::Resend(api)
243            | ProviderConfig::Brevo(api) => match api.api_url.as_deref() {
244                Some(url) => self.check_api_url(url).await,
245                None => Ok(()),
246            },
247        };
248        if let Err(refusal) = &result {
249            refusal.log("save");
250        }
251        result
252    }
253
254    /// One HTTP provider request through [`guarded_fetch_no_redirect`]: `url`
255    /// resolved once, vetted, pinned, and never redirected. A refusal is the
256    /// administrator-safe [`EgressRefusal`] error; a failed connection is
257    /// [`PROVIDER_UNREACHABLE`].
258    pub(crate) async fn post(
259        &self,
260        provider: &'static str,
261        url: &str,
262        build: impl Fn(&reqwest::Client, &str) -> reqwest::RequestBuilder,
263    ) -> Result<reqwest::Response, AxiamError> {
264        match guarded_fetch_no_redirect(url, self.allow_private_http, build).await {
265            Ok(response) => Ok(response),
266            Err(
267                error @ (SsrfError::InvalidUrl
268                | SsrfError::InsecureScheme
269                | SsrfError::ResolveFailed
270                | SsrfError::Blocked),
271            ) => {
272                let host = url::Url::parse(url)
273                    .ok()
274                    .and_then(|u| u.host_str().map(str::to_owned));
275                let refusal = EgressRefusal::api_url(host.as_deref(), &error);
276                refusal.log("send");
277                Err(refusal.into_error())
278            }
279            Err(error) => {
280                tracing::warn!(
281                    target: "axiam::email",
282                    provider,
283                    error = %error,
284                    "the email provider's API could not be reached"
285                );
286                Err(AxiamError::EmailDelivery(format!(
287                    "{provider}: {PROVIDER_UNREACHABLE}"
288                )))
289            }
290        }
291    }
292
293    /// The SMTP send-time guard: the refusal, logged, as the caller's error.
294    pub(crate) async fn smtp_target(
295        &self,
296        host: &str,
297        port: u16,
298    ) -> Result<GuardedTarget, AxiamError> {
299        self.guard_smtp(host, port).await.map_err(|refusal| {
300            refusal.log("send");
301            refusal.into_error()
302        })
303    }
304}

306#[cfg(test)]
307mod tests {
308    use super::*;
309
310    #[test]
311    fn a_names_refusal_is_one_sentence_and_a_literals_names_the_rule() {
312        let by_name = EgressRefusal::smtp("relay.example.com", &GuardError::Loopback);
313        assert_eq!(by_name.rule, "address_guard.loopback");
314        assert_eq!(
315            by_name.public_message(),
316            format!("smtp host: {HOST_NOT_PERMITTED}")
317        );
318        let by_literal = EgressRefusal::smtp("127.0.0.1", &GuardError::Loopback);
319        assert!(by_literal.public_message().contains("loopback"));
320        let unresolved = EgressRefusal::api_url(Some("api.example.com"), &SsrfError::ResolveFailed);
321        assert_eq!(
322            unresolved.public_message(),
323            format!("api_url: {HOST_NOT_PERMITTED}")
324        );
325        let plaintext = EgressRefusal::api_url(Some("api.example.com"), &SsrfError::InsecureScheme);
326        assert!(plaintext.public_message().contains("non-HTTPS"));
327    }
