17/// SMTP email provider using `lettre` with STARTTLS or implicit TLS, held to
18/// the deployment's outbound address policy at every send (#529).
19pub struct SmtpProvider {
20    host: String,
21    port: u16,
22    credentials: Credentials,
23    starttls: bool,
24    /// Built once from the configured host: the name the server's certificate
25    /// must carry, whatever address the connection is pinned to.
26    tls: TlsParameters,
27    egress: EmailEgress,
28}
29
30impl SmtpProvider {
31    /// A provider for `config`. Nothing is resolved or dialled here; an
32    /// unusable host name for the TLS check is refused now.
33    pub fn new(config: &SmtpConfig, egress: EmailEgress) -> Result<Self, AxiamError> {
34        let host = config.host.trim().to_string();
35        let tls_name = host
36            .trim_start_matches('[')
37            .trim_end_matches(']')
38            .to_string();
39        let tls = TlsParameters::new(tls_name)
40            .map_err(|e| AxiamError::EmailConfig(format!("SMTP TLS parameters error: {e}")))?;
41        Ok(Self {
42            host,
43            port: config.port,
44            credentials: Credentials::new(config.username.clone(), config.password.clone()),
45            starttls: config.starttls,
46            tls,
47            egress,
48        })
49    }
50
51    /// The transport for one send: the host resolved once and vetted by the
52    /// address guard, the TCP connection opened to the vetted address — an IP
53    /// literal, so `lettre` resolves nothing — and TLS checked against the
54    /// configured host.
55    ///
56    /// What `relay()` / `starttls_relay()` did, minus their own resolution:
57    /// implicit TLS (SMTPS, typically port 465) wraps the connection from the
58    /// start; STARTTLS (typically port 587) connects in plaintext and upgrades.
59    /// Neither falls back to cleartext.
60    async fn pinned_transport(&self) -> AxiamResult<AsyncSmtpTransport<Tokio1Executor>> {
61        let target = self.egress.smtp_target(&self.host, self.port).await?;
62        let address = target.addresses[0];
63        let tls = if self.starttls {
64            Tls::Required(self.tls.clone())
65        } else {
66            Tls::Wrapper(self.tls.clone())
67        };
68        Ok(
69            AsyncSmtpTransport::<Tokio1Executor>::builder_dangerous(address.ip().to_string())
70                .port(address.port())
71                .credentials(self.credentials.clone())
72                .tls(tls)
73                .build(),
74        )
75    }
76}

78impl EmailProvider for SmtpProvider {
79    fn send(
80        &self,
81        from_name: &str,
82        from_email: &str,
83        reply_to: Option<&str>,
84        message: &EmailMessage,
85    ) -> Pin<Box<dyn Future<Output = AxiamResult<SendResult>> + Send + '_>> {
86        let from_name = from_name.to_string();
87        let from_email = from_email.to_string();
88        let reply_to = reply_to.map(str::to_string);
89        let message = message.clone();
90
91        Box::pin(async move {
92            let from_mailbox: Mailbox = format!("{from_name} <{from_email}>")
93                .parse()
94                .map_err(|e| AxiamError::EmailConfig(format!("invalid from address: {e}")))?;
95
96            let to_mailbox: Mailbox = message
97                .to
98                .parse()
99                .map_err(|e| AxiamError::EmailDelivery(format!("invalid to address: {e}")))?;
100
101            let mut builder = MessageBuilder::new()
102                .from(from_mailbox)
103                .to(to_mailbox)
104                .subject(&message.subject);
105
106            if let Some(ref rt) = reply_to {
107                let rt_mailbox: Mailbox = rt.parse().map_err(|e| {
108                    AxiamError::EmailDelivery(format!("invalid reply-to address: {e}"))
109                })?;
110                builder = builder.reply_to(rt_mailbox);
111            }
112
113            let email = match (&message.html_body, &message.text_body) {
114                (Some(html), Some(text)) => {
115                    use lettre::message::MultiPart;
116                    builder
117                        .multipart(MultiPart::alternative_plain_html(
118                            text.clone(),
119                            html.clone(),
120                        ))
121                        .map_err(|e| {
122                            AxiamError::EmailDelivery(format!(
123                                "failed to build multipart email: {e}"
124                            ))
125                        })?
126                }
127                (Some(html), None) => {
128                    use lettre::message::header::ContentType;
129                    builder
130                        .header(ContentType::TEXT_HTML)
131                        .body(html.clone())
132                        .map_err(|e| {
133                            AxiamError::EmailDelivery(format!("failed to build HTML email: {e}"))
134                        })?
135                }
136                (None, Some(text)) => builder.body(text.clone()).map_err(|e| {
137                    AxiamError::EmailDelivery(format!("failed to build text email: {e}"))
138                })?,
139                (None, None) => return Err(AxiamError::EmailDelivery("email has no body".into())),
140            };
141
142            let transport = self.pinned_transport().await?;
143            let response = transport.send(email).await.map_err(|e| {
144                if e.is_permanent() || e.is_transient() {
145                    // The server's own answer to a command (a rejected sender,
146                    // failed authentication): it reached a vetted SMTP server.
147                    AxiamError::EmailDelivery(format!("SMTP send failed: {e}"))
148                } else {
149                    // Refused, reset, timed out or a failed handshake: one
150                    // answer, so the failure is no port scanner (#529).
151                    tracing::warn!(
152                        target: "axiam::email",
153                        error = %e,
154                        "the SMTP server could not be reached"
155                    );
156                    AxiamError::EmailDelivery(format!("smtp: {PROVIDER_UNREACHABLE}"))
157                }
158            })?;
159
160            Ok(SendResult {
161                message_id: response.message().next().map(str::to_string),
162            })
163        })
164    }
165
166    fn provider_name(&self) -> &'static str {
167        "smtp"
168    }
