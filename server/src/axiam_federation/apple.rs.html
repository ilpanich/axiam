1//! Sign in with Apple's client secret.
2//!
3//! Apple's `client_secret` is not a string an operator pastes once. It is an
4//! ES256-signed JWT with `iss` = Team ID, `sub` = Services ID (the `client_id`),
5//! `aud` = `https://appleid.apple.com`, and `exp` at most 15 777 000 seconds
6//! (six months) after `iat`, signed with a `.p8` key whose 10-character Key ID
7//! goes in the JOSE header `kid`.
8//!
9//! # Why AXIAM mints it rather than storing it
10//!
11//! *A secret that silently expires in six months is an outage nobody diagnoses
12//! quickly.* It lapses on a Tuesday, "Sign in with Apple" starts answering
13//! `invalid_client`, and the person who generated it left. Minting a fresh
14//! five-minute JWT per token exchange removes the failure mode rather than
15//! scheduling a warning about it.
16//!
17//! What the operator stores is the `.p8` private key, in the existing
18//! `client_secret_ciphertext` / `client_secret_nonce` /
19//! `client_secret_key_version` columns — it *is* the secret, so it gets the
20//! AES-256-GCM-at-rest treatment SEC-045 already provides, unchanged. The Team
21//! ID and Key ID are not secret and live in their own plain columns.
22//!
23//! A config with no `apple_team_id`/`apple_key_id` keeps the stored secret and
24//! sends it verbatim, which is the escape hatch for an operator who prefers to
25//! manage the JWT themselves.
26
27use chrono::Utc;
28use jsonwebtoken::{Algorithm, EncodingKey, Header};
29use serde::Serialize;
30
31use crate::error::FederationError;
32
33/// The audience Apple requires on the client secret.
34pub const APPLE_AUDIENCE: &str = "https://appleid.apple.com";
35
36/// Lifetime of a minted client secret.
37///
38/// Apple's ceiling is six months. Five minutes is used instead because the JWT
39/// is consumed by exactly one token exchange, seconds after it is minted, and a
40/// credential's lifetime should match its job rather than its permitted
41/// maximum. Clock skew between AXIAM and Apple is the only reason it is not
42/// shorter still.
43pub const CLIENT_SECRET_TTL_SECS: i64 = 300;
44
45/// Apple's Team and Key identifiers are both exactly ten characters.
46const APPLE_ID_LEN: usize = 10;
47
48#[derive(Debug, Serialize)]
49struct AppleClientSecretClaims<'a> {
50    iss: &'a str,
51    iat: i64,
52    exp: i64,
53    aud: &'a str,
54    sub: &'a str,
55}
56
57/// Validate an Apple Team ID or Key ID.
58///
59/// Ten alphanumerics. Checked at config-write time so an operator
60/// learns about a mistyped identifier while they are looking at the form,
61/// rather than from an `invalid_client` at the first sign-in attempt.
62pub fn validate_apple_identifier(kind: &str, value: &str) -> Result<(), String> {
63    if value.len() != APPLE_ID_LEN || !value.bytes().all(|b| b.is_ascii_alphanumeric()) {
64        return Err(format!(
65            "{kind} must be Apple's 10-character alphanumeric identifier"
66        ));
67    }
68    Ok(())
69}
70
71/// Mint an Apple client secret.
72///
73/// `private_key_pem` is the contents of the `.p8` file Apple issues — a PKCS#8
74/// `PRIVATE KEY` PEM over the P-256 curve.
75pub fn mint_client_secret(
76    team_id: &str,
77    key_id: &str,
78    client_id: &str,
79    private_key_pem: &str,
80) -> Result<String, FederationError> {
81    validate_apple_identifier("apple_team_id", team_id).map_err(FederationError::ConfigInvalid)?;
82    validate_apple_identifier("apple_key_id", key_id).map_err(FederationError::ConfigInvalid)?;
83
84    let now = Utc::now().timestamp();
85    let claims = AppleClientSecretClaims {
86        iss: team_id,
87        iat: now,
88        exp: now + CLIENT_SECRET_TTL_SECS,
89        aud: APPLE_AUDIENCE,
90        sub: client_id,
91    };
92
93    let mut header = Header::new(Algorithm::ES256);
94    header.kid = Some(key_id.to_string());
95
96    // `from_ec_pem` accepts both SEC1 and PKCS#8; Apple issues PKCS#8. A wrong
97    // key type fails here rather than producing a JWT Apple rejects for
98    // reasons the operator cannot see.
99    let key = EncodingKey::from_ec_pem(private_key_pem.as_bytes()).map_err(|e| {
100        FederationError::ConfigInvalid(format!(
101            "the Apple signing key is not a usable EC private key: {e}"
102        ))
103    })?;
104
105    jsonwebtoken::encode(&header, &claims, &key).map_err(|e| {
106        FederationError::ConfigInvalid(format!("failed to sign the Apple client secret: {e}"))
107    })
108}

110#[cfg(test)]
111mod tests {
112    use super::*;
113    use jsonwebtoken::{DecodingKey, Validation, decode, decode_header};
114
115    /// A throwaway P-256 keypair, PKCS#8 PEM for signing and SPKI PEM for
116    /// verification — the same shape Apple's `.p8` has.
117    fn ec_keypair() -> (String, String) {
118        let key = rcgen::KeyPair::generate_for(&rcgen::PKCS_ECDSA_P256_SHA256).unwrap();
119        (key.serialize_pem(), key.public_key_pem())
120    }
121
122    #[test]
123    fn a_minted_secret_carries_exactly_the_claims_apple_requires() {
124        let (private_pem, public_pem) = ec_keypair();
125        let token = mint_client_secret("ABCDE12345", "KEYID67890", "com.example.svc", &private_pem)
126            .expect("minting must succeed");
127
128        // The kid header is how Apple finds the key; without it the JWT is
129        // unverifiable and the failure is a bare invalid_client.
130        let header = decode_header(&token).unwrap();
131        assert_eq!(header.alg, Algorithm::ES256);
132        assert_eq!(header.kid.as_deref(), Some("KEYID67890"));
133
134        let mut validation = Validation::new(Algorithm::ES256);
135        validation.set_audience(&[APPLE_AUDIENCE]);
136        validation.set_issuer(&["ABCDE12345"]);
137        let key = DecodingKey::from_ec_pem(public_pem.as_bytes()).expect("verification key");
138        let data =
139            decode::<serde_json::Value>(&token, &key, &validation).expect("signature must verify");
140
141        assert_eq!(data.claims["iss"], "ABCDE12345");
142        assert_eq!(data.claims["sub"], "com.example.svc");
143        assert_eq!(data.claims["aud"], APPLE_AUDIENCE);
144
145        let iat = data.claims["iat"].as_i64().unwrap();
146        let exp = data.claims["exp"].as_i64().unwrap();
147        assert_eq!(exp - iat, CLIENT_SECRET_TTL_SECS);
148        // The point of minting per exchange: nowhere near Apple's six-month
149        // ceiling, so there is no expiry for an operator to be surprised by.
150        assert!(exp - iat < 15_777_000);
151    }

153    /// The property that makes storing the JWT unnecessary: every call stamps
154    /// the current time, so the secret in flight is always minutes old.
155    ///
156    /// Deliberately not `assert_ne!(a, b)` on two same-second mints:
157    /// `jsonwebtoken`'s `rust_crypto` backend signs ES256 deterministically
158    /// (RFC 6979), so two mints of identical claims are byte-identical. That is
159    /// fine — Apple verifies a signature, not its novelty — but it means token
160    /// inequality would be asserting the signature backend rather than the
161    /// freshness this test is about.
162    #[test]
163    fn each_mint_stamps_the_current_time() {
164        let (private_pem, public_pem) = ec_keypair();
165        let before = Utc::now().timestamp();
166        let token = mint_client_secret("ABCDE12345", "KEYID67890", "svc", &private_pem).unwrap();
167        let after = Utc::now().timestamp();
168
169        let mut validation = Validation::new(Algorithm::ES256);
170        validation.set_audience(&[APPLE_AUDIENCE]);
171        validation.set_issuer(&["ABCDE12345"]);
172        let key = DecodingKey::from_ec_pem(public_pem.as_bytes()).unwrap();
173        let data = decode::<serde_json::Value>(&token, &key, &validation).unwrap();
174
175        let iat = data.claims["iat"].as_i64().unwrap();
176        assert!(
177            (before..=after).contains(&iat),
178            "iat {iat} must be minted now, not read from storage"
179        );
180    }
181
182    #[test]
183    fn a_mistyped_identifier_is_refused_at_the_form_not_at_the_idp() {
184        let (private_pem, _) = ec_keypair();
185        for (team, key) in [
186            ("SHORT", "KEYID67890"),
187            ("ABCDE12345", "SHORT"),
188            ("ABCDE-2345", "KEYID67890"),
189            ("", ""),
190        ] {
191            assert!(
192                mint_client_secret(team, key, "svc", &private_pem).is_err(),
193                "({team}, {key}) must be refused"
194            );
195        }
196    }
197
198    #[test]
199    fn a_key_that_is_not_an_ec_private_key_fails_here() {
200        let err =
201            mint_client_secret("ABCDE12345", "KEYID67890", "svc", "not a pem at all").unwrap_err();
202        assert!(
203            err.to_string().contains("EC private key"),
204            "the message must point at the key, not at Apple: {err}"
205        );
206    }
