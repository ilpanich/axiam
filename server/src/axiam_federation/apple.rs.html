1//! Sign in with Apple's client secret.
2//!
3//! Apple's `client_secret` is not a string an operator pastes once. It is an
4//! ES256-signed JWT with `iss` = Team ID, `sub` = Services ID (the `client_id`),
5//! `aud` = `https://appleid.apple.com`, and `exp` at most 15 777 000 seconds
6//! (six months) after `iat`, signed with a `.p8` key whose 10-character Key ID
7//! goes in the JOSE header `kid`.
8//!
9//! # Why AXIAM mints it rather than storing it
10//!
11//! *A secret that silently expires in six months is an outage nobody diagnoses
12//! quickly.* It lapses on a Tuesday, "Sign in with Apple" starts answering
13//! `invalid_client`, and the person who generated it left. Minting a fresh
14//! five-minute JWT per token exchange removes the failure mode rather than
15//! scheduling a warning about it.
16//!
17//! What the operator stores is the `.p8` private key, in the existing
18//! `client_secret_ciphertext` / `client_secret_nonce` /
19//! `client_secret_key_version` columns — it *is* the secret, so it gets the
20//! AES-256-GCM-at-rest treatment SEC-045 already provides, unchanged. The Team
21//! ID and Key ID are not secret and live in their own plain columns.
22//!
23//! A config with no `apple_team_id`/`apple_key_id` keeps the stored secret and
24//! sends it verbatim, which is the escape hatch for an operator who prefers to
25//! manage the JWT themselves.
26//!
27//! # Why the signature is `ring`'s and not `jsonwebtoken`'s
28//!
29//! This is the one place AXIAM performs an ECDSA *secret-key* operation, and an
30//! unauthenticated caller can make it happen: every OIDC callback for an Apple
31//! config mints a secret before the code exchange. `jsonwebtoken`'s
32//! `rust_crypto` backend signs ES256 with `p256` 0.13, whose scalar arithmetic
33//! (`add_mod`, `sub_mod`, `sbb`, `mul_wide`) is `crypto-bigint` 0.5.5 —
34//! RUSTSEC-2026-0387, timing variability in operations meant to be constant
35//! time, fixed only in 0.7.6, which no `p256` 0.13 release can take. So the JWS
36//! is assembled here and signed by `ring`'s P-256 (already the TLS stack's
37//! provider), which owes nothing to `crypto-bigint`. The advisory's `deny.toml`
38//! entry rests on this: do not route this signature back through
39//! `jsonwebtoken::encode`.
40

52/// The audience Apple requires on the client secret.
53pub const APPLE_AUDIENCE: &str = "https://appleid.apple.com";
54
55/// Lifetime of a minted client secret.
56///
57/// Apple's ceiling is six months. Five minutes is used instead because the JWT
58/// is consumed by exactly one token exchange, seconds after it is minted, and a
59/// credential's lifetime should match its job rather than its permitted
60/// maximum. Clock skew between AXIAM and Apple is the only reason it is not
61/// shorter still.
62pub const CLIENT_SECRET_TTL_SECS: i64 = 300;
63
64/// Apple's Team and Key identifiers are both exactly ten characters.
65const APPLE_ID_LEN: usize = 10;
66
67#[derive(Debug, Serialize)]
68struct AppleClientSecretClaims<'a> {
69    iss: &'a str,
70    iat: i64,
71    exp: i64,
72    aud: &'a str,
73    sub: &'a str,
74}
75
76/// Validate an Apple Team ID or Key ID.
77///
78/// Ten alphanumerics. Checked at config-write time so an operator
79/// learns about a mistyped identifier while they are looking at the form,
80/// rather than from an `invalid_client` at the first sign-in attempt.
81pub fn validate_apple_identifier(kind: &str, value: &str) -> Result<(), String> {
82    if value.len() != APPLE_ID_LEN || !value.bytes().all(|b| b.is_ascii_alphanumeric()) {
83        return Err(format!(
84            "{kind} must be Apple's 10-character alphanumeric identifier"
85        ));
86    }
87    Ok(())
88}

90/// Mint an Apple client secret.
91///
92/// `private_key_pem` is the contents of the `.p8` file Apple issues — a PKCS#8
93/// `PRIVATE KEY` PEM over the P-256 curve.
94pub fn mint_client_secret(
95    team_id: &str,
96    key_id: &str,
97    client_id: &str,
98    private_key_pem: &str,
99) -> Result<String, FederationError> {
100    validate_apple_identifier("apple_team_id", team_id).map_err(FederationError::ConfigInvalid)?;
101    validate_apple_identifier("apple_key_id", key_id).map_err(FederationError::ConfigInvalid)?;
102
103    let now = Utc::now().timestamp();
104    let claims = AppleClientSecretClaims {
105        iss: team_id,
106        iat: now,
107        exp: now + CLIENT_SECRET_TTL_SECS,
108        aud: APPLE_AUDIENCE,
109        sub: client_id,
110    };
111
112    let mut header = Header::new(Algorithm::ES256);
113    header.kid = Some(key_id.to_string());
114
115    let signing_input = format!(
116        "{}.{}",
117        b64_json(&header, "header")?,
118        b64_json(&claims, "claims")?
119    );
120    let signature = sign_es256(signing_input.as_bytes(), private_key_pem)?;
121    Ok(format!(
122        "{signing_input}.{}",
123        URL_SAFE_NO_PAD.encode(signature)
124    ))
125}

127fn b64_json<T: Serialize>(part: &T, what: &str) -> Result<String, FederationError> {
128    let json = serde_json::to_vec(part).map_err(|e| {
129        FederationError::ConfigInvalid(format!(
130            "failed to encode the Apple client secret {what}: {e}"
131        ))
132    })?;
133    Ok(URL_SAFE_NO_PAD.encode(json))
134}

136/// ES256 over `message` with the `.p8` key, as the JWS `r || s` octets.
137///
138/// Only PKCS#8 (`PRIVATE KEY`) is accepted — what Apple issues. A wrong key
139/// type fails here rather than producing a JWT Apple rejects for reasons the
140/// operator cannot see.
141fn sign_es256(message: &[u8], private_key_pem: &str) -> Result<Vec<u8>, FederationError> {
142    let unusable = |why: String| {
143        FederationError::ConfigInvalid(format!(
144            "the Apple signing key is not a usable EC private key: {why}"
145        ))
146    };
147    let pem = ::pem::parse(private_key_pem).map_err(|e| unusable(e.to_string()))?;
148    if pem.tag() != "PRIVATE KEY" {
149        return Err(unusable(format!(
150            "expected a PKCS#8 PRIVATE KEY PEM (Apple's .p8), got {}",
151            pem.tag()
152        )));
153    }
154    let der = Zeroizing::new(pem.into_contents());
155
156    let rng = SystemRandom::new();
157    let key = EcdsaKeyPair::from_pkcs8(&ECDSA_P256_SHA256_FIXED_SIGNING, &der, &rng)
158        .map_err(|e| unusable(e.to_string()))?;
159    let signature = key.sign(&rng, message).map_err(|_| {
160        FederationError::ConfigInvalid("failed to sign the Apple client secret".into())
161    })?;
162    Ok(signature.as_ref().to_vec())
163}

170    /// A throwaway P-256 keypair, PKCS#8 PEM for signing and SPKI PEM for
171    /// verification — the same shape Apple's `.p8` has.
172    fn ec_keypair() -> (String, String) {
173        let key = rcgen::KeyPair::generate_for(&rcgen::PKCS_ECDSA_P256_SHA256).unwrap();
174        (key.serialize_pem(), key.public_key_pem())
175    }
176
177    #[test]
178    fn a_minted_secret_carries_exactly_the_claims_apple_requires() {
179        let (private_pem, public_pem) = ec_keypair();
180        let token = mint_client_secret("ABCDE12345", "KEYID67890", "com.example.svc", &private_pem)
181            .expect("minting must succeed");
182
183        // The kid header is how Apple finds the key; without it the JWT is
184        // unverifiable and the failure is a bare invalid_client.
185        let header = decode_header(&token).unwrap();
186        assert_eq!(header.alg, Algorithm::ES256);
187        assert_eq!(header.kid.as_deref(), Some("KEYID67890"));
188
189        let mut validation = Validation::new(Algorithm::ES256);
190        validation.set_audience(&[APPLE_AUDIENCE]);
191        validation.set_issuer(&["ABCDE12345"]);
192        let key = DecodingKey::from_ec_pem(public_pem.as_bytes()).expect("verification key");
193        let data =
194            decode::<serde_json::Value>(&token, &key, &validation).expect("signature must verify");
195
196        assert_eq!(data.claims["iss"], "ABCDE12345");
197        assert_eq!(data.claims["sub"], "com.example.svc");
198        assert_eq!(data.claims["aud"], APPLE_AUDIENCE);
199
200        let iat = data.claims["iat"].as_i64().unwrap();
201        let exp = data.claims["exp"].as_i64().unwrap();
202        assert_eq!(exp - iat, CLIENT_SECRET_TTL_SECS);
203        // The point of minting per exchange: nowhere near Apple's six-month
204        // ceiling, so there is no expiry for an operator to be surprised by.
205        assert!(exp - iat < 15_777_000);
206    }

208    /// The property that makes storing the JWT unnecessary: every call stamps
209    /// the current time, so the secret in flight is always minutes old.
210    ///
211    /// Deliberately not `assert_ne!(a, b)` on two same-second mints: token
212    /// inequality would be asserting the signature backend (see
213    /// `the_signature_is_rings_not_rust_cryptos`) rather than the freshness
214    /// this test is about.
215    #[test]
216    fn each_mint_stamps_the_current_time() {
217        let (private_pem, public_pem) = ec_keypair();
218        let before = Utc::now().timestamp();
219        let token = mint_client_secret("ABCDE12345", "KEYID67890", "svc", &private_pem).unwrap();
220        let after = Utc::now().timestamp();
221
222        let mut validation = Validation::new(Algorithm::ES256);
223        validation.set_audience(&[APPLE_AUDIENCE]);
224        validation.set_issuer(&["ABCDE12345"]);
225        let key = DecodingKey::from_ec_pem(public_pem.as_bytes()).unwrap();
226        let data = decode::<serde_json::Value>(&token, &key, &validation).unwrap();
227
228        let iat = data.claims["iat"].as_i64().unwrap();
229        assert!(
230            (before..=after).contains(&iat),
231            "iat {iat} must be minted now, not read from storage"
232        );
233    }

261    /// Pins the RUSTSEC-2026-0387 mitigation. `jsonwebtoken`'s `rust_crypto`
262    /// backend (`p256` on `crypto-bigint` 0.5.5) signs ES256 deterministically
263    /// per RFC 6979, so the same message under the same key always yields the
264    /// same bytes; `ring` mixes fresh randomness into every nonce. Two distinct
265    /// signatures that both verify therefore show the secret key is not being
266    /// handled by the affected backend.
267    #[test]
268    fn the_signature_is_rings_not_rust_cryptos() {
269        let (private_pem, public_pem) = ec_keypair();
270        let split = |t: &str| {
271            let (input, sig) = t.rsplit_once('.').unwrap();
272            (input.to_string(), sig.to_string())
273        };
274        // Two mints in the same second carry identical header and claims; retry
275        // on the rare pair that straddles a second boundary.
276        let (a, b) = (0..5)
277            .map(|_| {
278                let mint =
279                    || mint_client_secret("ABCDE12345", "KEYID67890", "svc", &private_pem).unwrap();
280                (mint(), mint())
281            })
282            .find(|(a, b)| split(a).0 == split(b).0)
283            .expect("two mints within one second");
284        assert_ne!(
285            split(&a).1,
286            split(&b).1,
287            "a deterministic (RFC 6979) signer is back on this path"
288        );
289
290        let key = DecodingKey::from_ec_pem(public_pem.as_bytes()).unwrap();
291        let mut validation = Validation::new(Algorithm::ES256);
292        validation.set_audience(&[APPLE_AUDIENCE]);
293        validation.set_issuer(&["ABCDE12345"]);
294        for token in [a, b] {
295            decode::<serde_json::Value>(&token, &key, &validation).expect("must verify");
296        }
297    }
