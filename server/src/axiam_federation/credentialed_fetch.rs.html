1//! The one-hop rule for requests that carry a credential (R1W2-04, T-112).
2//!
3//! The four calls that send AXIAM's client secret, an authorization code, a
4//! PKCE verifier or a user's access token to an IdP — the OIDC and OAuth2
5//! token exchanges, the OAuth2 userinfo call and GitHub's verified-email call —
6//! go through [`crate::ssrf::guarded_fetch_no_redirect`]. That guard returns a
7//! `3xx` to the caller instead of following it, so a redirect never re-sends the
8//! body or the `Authorization` header to the `Location`. What remains is to
9//! refuse the `3xx`: a token or userinfo endpoint has no business redirecting,
10//! and reading a redirect's body as if it were the answer would only produce a
11//! confusing parse error further down.
12//!
13//! Discovery, JWKS and metadata fetches carry no credential and keep the
14//! redirect-following [`crate::ssrf::guarded_fetch`].
15
16/// The wording every refusal starts with.
17pub(crate) const REDIRECT_NOT_FOLLOWED: &str = "redirect not followed";
18
19/// `Some(reason)` when `response` is a `3xx`, `None` otherwise.
20///
21/// `what` names the call (`"token endpoint"`, `"userinfo endpoint"`) and
22/// `endpoint` is the **configured** URL it was sent to. The reason carries the
23/// endpoint's host and path only: not its query (a configured endpoint may carry
24/// one), not the `Location` header (attacker-chosen, and it may carry the very
25/// credential the redirect was meant to collect) and not the response body.
26pub(crate) fn redirect_refusal(
27    what: &str,
28    endpoint: &str,
29    response: &reqwest::Response,
30) -> Option<String> {
31    if !response.status().is_redirection() {
32        return None;
33    }
34    Some(format!(
35        "{REDIRECT_NOT_FOLLOWED}: the {what} {} answered HTTP {}",
36        endpoint_label(endpoint),
37        response.status().as_u16()
38    ))
39}
40
41/// `host[:port]/path` of `endpoint`, without scheme, userinfo, query or
42/// fragment. An unparseable value yields a placeholder rather than the raw
43/// string, which could itself hold a secret.
44fn endpoint_label(endpoint: &str) -> String {
45    match url::Url::parse(endpoint) {
46        Ok(u) => {
47            let host = u.host_str().unwrap_or("?");
48            match u.port() {
49                Some(p) => format!("{host}:{p}{}", u.path()),
50                None => format!("{host}{}", u.path()),
51            }
52        }
53        Err(_) => "<unparseable endpoint>".to_string(),
54    }
55}

57#[cfg(test)]
58mod tests {
59    use super::endpoint_label;
60
61    #[test]
62    fn the_label_is_host_and_path_and_nothing_else() {
63        assert_eq!(
64            endpoint_label("https://user:pw@idp.example.com/oauth/token?api_key=s3cr3t#frag"),
65            "idp.example.com/oauth/token"
66        );
67        assert_eq!(
68            endpoint_label("http://127.0.0.1:8080/token"),
69            "127.0.0.1:8080/token"
70        );
71        assert_eq!(endpoint_label("not a url"), "<unparseable endpoint>");
72    }
