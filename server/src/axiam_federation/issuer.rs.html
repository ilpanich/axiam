1//! Templated OIDC issuers.
2//!
3//! # The problem, verified live
4//!
5//! `https://login.microsoftonline.com/common/v2.0/.well-known/openid-configuration`
6//! publishes
7//!
8//! ```text
9//! issuer = https://login.microsoftonline.com/{tenantid}/v2.0
10//! ```
11//!
12//! with the placeholder literally in the string. `verify_id_token` does
13//! `validation.set_issuer(&[&discovery.issuer])`, so a token whose `iss` is
14//! `…/9188040d-…/v2.0` fails against `…/{tenantid}/v2.0` and every multi-tenant
15//! Entra login is rejected with a claim error.
16//!
17//! # The rule
18//!
19//! When the discovered issuer is templated, the token's `tid` claim is read
20//! from the **unverified** payload — solely to pick which concrete issuer
21//! string to require — checked against the config's `allowed_issuer_tenants`,
22//! and substituted. Signature verification is unchanged: the JWKS still comes
23//! from the discovery document, and the *verified* token's `iss` must then equal
24//! the constructed string.
25//!
26//! Reading a claim before verifying the signature is the part that deserves
27//! scrutiny. It is safe here because the value can only *select* from a closed,
28//! operator-written set: a forged `tid` either names a tenant the operator
29//! listed — and then the signature still has to verify against the provider's
30//! keys, and the verified `iss` still has to match — or it does not, and the
31//! login is refused. It can never widen the accepted set.
32//!
33//! # Why the allow-list is mandatory
34//!
35//! Microsoft signs every tenant's tokens at `common` with the same keys. A
36//! templated issuer with an empty allow-list therefore means *every Microsoft
37//! account on earth may sign in here*. That is occasionally what an operator
38//! wants and never what they want by accident, so the configuration is refused
39//! rather than accepted with that meaning — and the message says how to get
40//! either outcome deliberately.

42use base64::Engine;
43use base64::engine::general_purpose::URL_SAFE_NO_PAD;
44
45use crate::error::FederationError;

47/// The placeholder Entra ID publishes at the `common`/`organizations`
48/// authorities.
49pub const TENANT_PLACEHOLDER: &str = "{tenantid}";
50
51/// Whether a discovered issuer names a tenant it does not know yet.
52pub fn is_templated(issuer: &str) -> bool {
53    issuer.contains(TENANT_PLACEHOLDER)
54}
55
56/// Refuse a templated issuer that accepts everyone.
57///
58/// Called at config write time, which is the only moment an operator is in a
59/// position to do something about it.
60pub fn validate_allow_list(issuer: &str, allowed: &[String]) -> Result<(), FederationError> {
61    if !is_templated(issuer) {
62        return Ok(());
63    }
64    if allowed.iter().any(|t| !t.trim().is_empty()) {
65        return Ok(());
66    }
67    Err(FederationError::ConfigInvalid(format!(
68        "the provider's discovery document publishes a templated issuer \
69         ('{issuer}'), which means any of the provider's tenants can sign in. \
70         Either point metadata_url at a tenant-specific authority, which \
71         publishes a concrete issuer, or list the provider tenant IDs you \
72         accept in allowed_issuer_tenants — there is deliberately no accept-all \
73         value"
74    )))
75}

77/// Resolve the concrete issuer to require for one token.
78///
79/// Returns the discovered issuer unchanged when it is not templated, so the
80/// overwhelmingly common path costs one `contains`.
81pub fn resolve_expected_issuer(
82    discovered_issuer: &str,
83    token: &str,
84    allowed: &[String],
85) -> Result<String, FederationError> {
86    if !is_templated(discovered_issuer) {
87        return Ok(discovered_issuer.to_string());
88    }
89
90    // Belt: the empty allow-list is refused at write time, but a row written
91    // before that check existed — or a provider that started templating its
92    // issuer after the config was saved — must not fall through to "accept
93    // anyone" at sign-in time.
94    validate_allow_list(discovered_issuer, allowed)?;
95
96    let tid = unverified_tenant_id(token)?;
97
98    // A `tid` must be a UUID. Without this, a crafted value could substitute
99    // path segments into the issuer string and make it match some *other*
100    // allowed entry by construction.
101    if uuid::Uuid::parse_str(&tid).is_err() {
102        return Err(FederationError::JwtClaimRejected(
103            "the ID token's `tid` claim is not a tenant identifier".into(),
104        ));
105    }
106
107    // Case-insensitive, because Entra emits lowercase GUIDs and operators paste
108    // whatever the portal showed them. Comparing the *parsed* UUIDs would be
109    // stricter still, but an operator's list can legitimately hold a
110    // non-Microsoft provider's tenant identifier, so the comparison stays
111    // textual and the UUID check above is what bounds the shape.
112    if !allowed.iter().any(|a| a.trim().eq_ignore_ascii_case(&tid)) {
113        return Err(FederationError::IssuerTenantNotAllowed);
114    }
115
116    Ok(discovered_issuer.replace(TENANT_PLACEHOLDER, &tid))
117}

119/// Read `tid` from a JWT payload **without verifying the signature**.
120///
121/// See the module docs for why that is sound here. The value is used only to
122/// select an expected issuer from a closed set; it authorises nothing.
123fn unverified_tenant_id(token: &str) -> Result<String, FederationError> {
124    let payload_b64 = token.split('.').nth(1).ok_or_else(|| {
125        FederationError::JwtClaimRejected("ID token is not a well-formed JWT".into())
126    })?;
127    let bytes = URL_SAFE_NO_PAD.decode(payload_b64).map_err(|_| {
128        FederationError::JwtClaimRejected("ID token payload is not valid base64url".into())
129    })?;
130    let payload: serde_json::Value = serde_json::from_slice(&bytes).map_err(|_| {
131        FederationError::JwtClaimRejected("ID token payload is not valid JSON".into())
132    })?;
133    payload
134        .get("tid")
135        .and_then(|v| v.as_str())
136        .map(str::to_string)
137        .ok_or_else(|| {
138            FederationError::JwtClaimRejected(
139                "the provider publishes a templated issuer but the ID token carries no `tid` claim"
140                    .into(),
141            )
142        })
143}

145#[cfg(test)]
146mod tests {
147    use super::*;

149    const MS_COMMON: &str = "https://login.microsoftonline.com/{tenantid}/v2.0";
150    const TENANT_A: &str = "9188040d-6c67-4c5b-b112-36a304b66dad";
151    const TENANT_B: &str = "72f988bf-86f1-41af-91ab-2d7cd011db47";
152
153    fn token_with_tid(tid: &str) -> String {
154        let payload = serde_json::json!({ "tid": tid, "sub": "s" });
155        format!(
156            "eyJhbGciOiJSUzI1NiJ9.{}.sig",
157            URL_SAFE_NO_PAD.encode(serde_json::to_vec(&payload).unwrap())
158        )
159    }

161    /// Verified live against Microsoft's own document while this was written:
162    /// the placeholder is literal, not a docs convention.
163    #[test]
164    fn microsofts_common_authority_issuer_is_recognised_as_templated() {
165        assert!(is_templated(MS_COMMON));
166        assert!(!is_templated("https://accounts.google.com"));
167        assert!(!is_templated(
168            "https://login.microsoftonline.com/72f988bf-86f1-41af-91ab-2d7cd011db47/v2.0"
169        ));
170    }
171
172    /// The headline rule: templated and unlisted is refused, and the refusal
173    /// explains both ways out rather than just saying no.
174    #[test]
175    fn a_templated_issuer_with_no_allow_list_is_refused() {
176        let err = validate_allow_list(MS_COMMON, &[]).unwrap_err();
177        let msg = err.to_string();
178        assert!(msg.contains("tenant-specific authority"));
179        assert!(msg.contains("allowed_issuer_tenants"));
180        // Whitespace is not a list.
181        assert!(validate_allow_list(MS_COMMON, &["  ".to_string()]).is_err());
182        assert!(validate_allow_list(MS_COMMON, &[TENANT_A.to_string()]).is_ok());
183        // A concrete issuer needs no list at all.
184        assert!(validate_allow_list("https://accounts.google.com", &[]).is_ok());
185    }
186
187    #[test]
188    fn a_concrete_issuer_passes_through_untouched() {
189        let got = resolve_expected_issuer(
190            "https://accounts.google.com",
191            &token_with_tid(TENANT_A),
192            &[],
193        )
194        .unwrap();
195        assert_eq!(got, "https://accounts.google.com");
196    }
197
198    #[test]
199    fn a_listed_tenant_substitutes_into_the_issuer() {
200        let got = resolve_expected_issuer(
201            MS_COMMON,
202            &token_with_tid(TENANT_A),
203            &[TENANT_A.to_string()],
204        )
205        .unwrap();
206        assert_eq!(
207            got,
208            format!("https://login.microsoftonline.com/{TENANT_A}/v2.0")
209        );
210        // Case is not what distinguishes two tenants.
211        let upper = TENANT_A.to_uppercase();
212        assert!(
213            resolve_expected_issuer(MS_COMMON, &token_with_tid(&upper), &[TENANT_A.to_string()])
214                .is_ok()
215        );
216    }

218    /// The property the whole mechanism exists to preserve: a token from a
219    /// tenant the operator did not list is refused, even though its signature
220    /// would verify against exactly the same Microsoft keys.
221    #[test]
222    fn an_unlisted_tenant_is_refused_even_though_the_signature_would_verify() {
223        let err = resolve_expected_issuer(
224            MS_COMMON,
225            &token_with_tid(TENANT_B),
226            &[TENANT_A.to_string()],
227        )
228        .unwrap_err();
229        assert!(matches!(err, FederationError::IssuerTenantNotAllowed));
230    }

232    /// A `tid` that is not a tenant identifier could otherwise inject path
233    /// segments into the issuer string.
234    #[test]
235    fn a_tid_that_is_not_a_uuid_is_refused() {
236        for bad in [
237            "../../evil",
238            "9188040d-6c67-4c5b-b112-36a304b66dad/v2.0",
239            "",
240            "not-a-uuid",
241        ] {
242            let err = resolve_expected_issuer(
243                MS_COMMON,
244                &token_with_tid(bad),
245                &[bad.to_string(), TENANT_A.to_string()],
246            )
247            .unwrap_err();
248            assert!(
249                matches!(err, FederationError::JwtClaimRejected(_)),
250                "{bad} produced {err:?}"
251            );
252        }
253    }

255    #[test]
256    fn a_templated_issuer_with_no_tid_claim_is_refused_with_a_useful_message() {
257        let token = format!(

262        assert!(err.to_string().contains("`tid` claim"));
