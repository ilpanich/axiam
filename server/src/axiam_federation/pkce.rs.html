1//! PKCE (RFC 7636) for the outbound federation flows.
2//!
3//! AXIAM is the *relying party* here, so this is PKCE from the client side: a
4//! high-entropy verifier generated at authorize time, its `S256` challenge sent
5//! to the IdP, and the verifier itself echoed at the token exchange.
6//!
7//! # Where it is mandatory, and why
8//!
9//! On the [`OAuth2`](axiam_core::models::federation::FederationProtocol::OAuth2)
10//! variant it is not optional. That path has no ID token, so it has no `nonce`,
11//! no `aud` and no signature — PKCE is the only thing left that binds the
12//! authorization code to the browser session that started the flow.
13//!
14//! The honest caveat, stated here because it is easy to mistake sending PKCE
15//! for having PKCE: a provider that *ignores* `code_challenge` gives us nothing
16//! for it, and we cannot make a remote server verify something. GitHub has
17//! supported `S256` since July 2025. Where a provider does not, the residual
18//! protections are the single-use 256-bit server-side `state` and the
19//! confidential-client secret at the token endpoint, and that is what an
20//! operator is relying on.
21//!
22//! On the OIDC path PKCE is opt-in (`require_pkce`), because a server-side
23//! `nonce` the client never sees already provides the binding, and because
24//! making it mandatory would break every config written before the column
25//! existed.
26
27use base64::Engine;
28use base64::engine::general_purpose::URL_SAFE_NO_PAD;
29use sha2::{Digest, Sha256};
30
31/// Length in bytes of the random material behind a verifier.
32///
33/// 32 bytes → 43 base64url characters, the RFC 7636 §4.1 minimum, and the
34/// length every major provider is tested against. The RFC permits up to 128
35/// characters; more entropy than 256 bits buys nothing here.
36const VERIFIER_BYTES: usize = 32;
37
38/// A PKCE verifier and the challenge derived from it.
39///
40/// They are produced together and never separately: a challenge without its
41/// verifier cannot complete a flow, and a verifier whose challenge was computed
42/// somewhere else is how the two drift apart.
43#[derive(Debug, Clone)]
44pub struct PkcePair {
45    /// The secret. Stored server-side in `federation_login_state` and echoed at
46    /// the token endpoint. It must never reach the browser.
47    pub verifier: String,
48    /// `BASE64URL(SHA256(verifier))`, sent to the IdP as `code_challenge`.
49    pub challenge: String,
50}
51
52/// The only challenge method AXIAM sends.
53///
54/// `plain` is deliberately absent. RFC 7636 §7.2 permits it and it protects
55/// against nothing an attacker who can read the authorization request cannot
56/// defeat, which is exactly the attacker PKCE exists for. Every provider AXIAM
57/// federates to supports `S256`.
58pub const CHALLENGE_METHOD: &str = "S256";
59
60/// Generate a fresh verifier and its `S256` challenge.
61pub fn generate() -> PkcePair {
62    use rand::Rng;
63    let mut bytes = [0u8; VERIFIER_BYTES];
64    rand::rng().fill_bytes(&mut bytes);
65    let verifier = URL_SAFE_NO_PAD.encode(bytes);
66    let challenge = challenge_for(&verifier);
67    PkcePair {
68        verifier,
69        challenge,
70    }
71}
72
73/// Derive the `S256` challenge for a verifier.
74///
75/// Split out so a test can pin the RFC 7636 Appendix B vector against the same
76/// function the flow uses, rather than against a second copy of it.
77pub fn challenge_for(verifier: &str) -> String {
78    let digest = Sha256::digest(verifier.as_bytes());
79    URL_SAFE_NO_PAD.encode(digest)
80}

82#[cfg(test)]
83mod tests {
84    use super::*;
85
86    /// RFC 7636 Appendix B, verbatim. The one test that proves this is PKCE
87    /// and not merely a hash of a random string.
88    #[test]
89    fn the_rfc_7636_appendix_b_vector_holds() {
90        assert_eq!(
91            challenge_for("dBjftJeZ4CVP-mB92K27uhbUJU1p1r_wW1gFWFOEjXk"),
92            "E9Melhoa2OwvFrEMTJguCHaoeK1t8URWbuGJSstw-cM"
93        );
94    }
95
96    #[test]
97    fn a_verifier_meets_the_rfc_length_floor_and_is_url_safe() {
98        let p = generate();
99        // RFC 7636 §4.1: 43..=128 characters, from the unreserved set.
100        assert!(
101            (43..=128).contains(&p.verifier.len()),
102            "{}",
103            p.verifier.len()
104        );
105        assert!(
106            p.verifier
107                .bytes()
108                .all(|b| b.is_ascii_alphanumeric() || b"-._~".contains(&b)),
109            "verifier must be URL-safe without escaping: {}",
110            p.verifier
111        );
112    }
113
114    #[test]
115    fn each_pair_is_fresh_and_internally_consistent() {
116        let a = generate();
117        let b = generate();
118        assert_ne!(a.verifier, b.verifier, "verifiers must not repeat");
119        assert_ne!(a.challenge, b.challenge);
120        // The property that makes the pair usable at all.
121        assert_eq!(challenge_for(&a.verifier), a.challenge);
122    }
123
124    /// The challenge must not be reversible to the verifier by construction —
125    /// restated as the property that actually matters: they are different
126    /// values, and only the verifier is ever stored as the secret.
127    #[test]
128    fn the_challenge_is_not_the_verifier() {
129        let p = generate();
130        assert_ne!(p.verifier, p.challenge);
131    }
132}
