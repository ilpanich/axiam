132    /// Fetch and parse the IdP SAML metadata from the given URL.
133    ///
134    /// Only HTTPS URLs are accepted to mitigate SSRF risks.
135    pub async fn fetch_idp_metadata(
136        &self,
137        metadata_url: &str,
138    ) -> Result<IdpMetadata, FederationError> {
139        validate_metadata_url(metadata_url)?;
140
141        // SECHRD-02: route the metadata GET through the shared, IP-pinning
142        // SSRF guard (D-01a/b/c). Production always fails closed against
143        // private/loopback/link-local addresses and internal redirect
144        // targets — `allow_private=false`.
145        let response = crate::ssrf::guarded_fetch(metadata_url, false, |c, u| c.get(u))
146            .await
147            .map_err(|e| FederationError::SamlMetadataFailed(e.to_string()))?;
148
149        if !response.status().is_success() {
150            return Err(FederationError::SamlMetadataFailed(format!(
151                "HTTP {} from metadata endpoint",
152                response.status()
153            )));
154        }
155
156        // Enforce a maximum body size to prevent memory exhaustion from
157        // a large metadata document.
158        const MAX_METADATA_SIZE: usize = 512 * 1024; // 512 KiB
159        let bytes = response.bytes().await.map_err(|e| {
160            FederationError::SamlMetadataFailed(format!("Failed to read metadata body: {e}"))
161        })?;
162        if bytes.len() > MAX_METADATA_SIZE {
163            return Err(FederationError::SamlMetadataFailed(format!(
164                "Metadata document too large: {} bytes (max {})",
165                bytes.len(),
166                MAX_METADATA_SIZE
167            )));
168        }
169
170        let text = String::from_utf8(bytes.to_vec()).map_err(|e| {
171            FederationError::SamlMetadataFailed(format!("Invalid UTF-8 in metadata: {e}"))
172        })?;
173
174        let descriptor_type: EntityDescriptorType = text.parse().map_err(|e| {
175            FederationError::SamlMetadataFailed(format!("Failed to parse metadata XML: {e}"))
176        })?;
177
178        // Extract the first EntityDescriptor from the parsed metadata.
179        let ed = descriptor_type.iter().next().ok_or_else(|| {
180            FederationError::SamlMetadataFailed("No EntityDescriptor found in metadata".into())
181        })?;
182
183        let entity_id = ed.entity_id.clone().ok_or_else(|| {
184            FederationError::SamlMetadataFailed("EntityDescriptor missing entityID".into())
185        })?;
186
187        let idp_descriptors = ed.idp_sso_descriptors.as_ref().ok_or_else(|| {
188            FederationError::SamlMetadataFailed("No IDPSSODescriptor in metadata".into())
189        })?;
190
191        let idp = idp_descriptors.first().ok_or_else(|| {
192            FederationError::SamlMetadataFailed("Empty IDPSSODescriptor list in metadata".into())
193        })?;
194
195        // Prefer HTTP-POST binding, fall back to HTTP-Redirect.
196        let sso_endpoint = idp
197            .single_sign_on_services
198            .iter()
199            .find(|ep| ep.binding == HTTP_POST_BINDING)
200            .or_else(|| {
201                idp.single_sign_on_services
202                    .iter()
203                    .find(|ep| ep.binding == HTTP_REDIRECT_BINDING)
204            })
205            .ok_or_else(|| {
206                FederationError::SamlMetadataFailed(
207                    "No HTTP-POST or HTTP-Redirect SSO endpoint".into(),
208                )
209            })?;
210
211        // Validate that the SSO endpoint uses HTTPS to prevent
212        // redirecting users to insecure origins. Fail closed on
213        // parse errors — reject malformed URLs outright.
214        let sso_parsed = url::Url::parse(&sso_endpoint.location).map_err(|e| {
215            FederationError::SamlMetadataFailed(format!("IdP SSO endpoint is not a valid URL: {e}"))
216        })?;
217        if sso_parsed.scheme() != "https" {
218            return Err(FederationError::SamlMetadataFailed(
219                "IdP SSO endpoint must use HTTPS".into(),
220            ));
221        }
222
223        Ok(IdpMetadata {
224            entity_id,
225            sso_url: sso_endpoint.location.clone(),
226            sso_binding: sso_endpoint.binding.clone(),
227        })
228    }

230    /// Build a SAML AuthnRequest for the specified federation config.
231    ///
232    /// Returns the SSO URL, base64-encoded request, binding type, and
233    /// optional relay state for the caller to use in the redirect or
234    /// POST form.
235    pub async fn build_authn_request(
236        &self,
237        tenant_id: Uuid,
238        config_id: Uuid,
239        acs_url: &str,
240        relay_state: Option<String>,
241    ) -> Result<SamlAuthnRequestResult, FederationError> {
242        let config = self
243            .federation_config_repo
244            .get_by_id(tenant_id, config_id)
245            .await
246            .map_err(|e| match e {
247                AxiamError::NotFound { id, .. } => FederationError::ConfigNotFound(id),
248                other => FederationError::Internal(other.to_string()),
249            })?;
250
251        if !config.enabled {
252            return Err(FederationError::ConfigDisabled);
253        }
254
255        if config.protocol != FederationProtocol::Saml {
256            return Err(FederationError::ProtocolMismatch(
257                "expected Saml protocol".into(),
258            ));
259        }
260
261        let metadata_url = config.metadata_url.as_deref().ok_or_else(|| {
262            FederationError::SamlMetadataFailed("No metadata URL configured".into())
263        })?;
264
265        let idp = self.fetch_idp_metadata(metadata_url).await?;
266
267        // Deliberately v4, not `new_id()`: the AuthnRequest ID is echoed back
268        // in `InResponseTo` and gates replay detection, so it must not be
269        // predictable from the request's timing.
270        let authn_request_id = format!("_{}", Uuid::new_v4());
271        let authn_request = AuthnRequest {
272            id: authn_request_id.clone(),
273            version: "2.0".into(),
274            issue_instant: Utc::now(),
275            destination: Some(idp.sso_url.clone()),
276            issuer: Some(Issuer {
277                value: Some(config.client_id.clone()),
278                ..Default::default()
279            }),
280            assertion_consumer_service_url: Some(acs_url.to_string()),
281            protocol_binding: Some(HTTP_POST_BINDING.to_string()),
282            name_id_policy: Some(NameIdPolicy {
283                allow_create: Some(true),
284                format: Some(NAMEID_FORMAT_PERSISTENT.into()),
285                ..Default::default()
286            }),
287            ..Default::default()
288        };
289
290        let xml = authn_request.to_string().map_err(|e| {
291            FederationError::Internal(format!("Failed to serialize AuthnRequest to XML: {e:?}"))
292        })?;
293
294        let (saml_request, url) = if idp.sso_binding == HTTP_REDIRECT_BINDING {
295            // HTTP-Redirect: DEFLATE compress then base64 encode.
296            let deflated = deflate_encode(xml.as_bytes())?;
297            let encoded = STANDARD.encode(&deflated);
298
299            let mut redirect_url = url::Url::parse(&idp.sso_url).map_err(|e| {
300                FederationError::SamlMetadataFailed(format!("Invalid SSO URL: {e}"))
301            })?;
302
303            redirect_url
304                .query_pairs_mut()
305                .append_pair("SAMLRequest", &encoded);
306
307            if let Some(ref rs) = relay_state {
308                redirect_url.query_pairs_mut().append_pair("RelayState", rs);
309            }
310
311            (encoded, redirect_url.to_string())
312        } else {
313            // HTTP-POST: base64 encode the raw XML (no deflate).
314            let encoded = STANDARD.encode(xml.as_bytes());
315            (encoded, idp.sso_url.clone())
316        };
317
318        info!(
319            tenant_id = %tenant_id,
320            config_id = %config_id,
321            binding = %idp.sso_binding,
322            "Built SAML AuthnRequest"
323        );
324
325        Ok(SamlAuthnRequestResult {
326            url,
327            saml_request,
328            binding: idp.sso_binding,
329            relay_state,
330            request_id: authn_request_id,
331        })
332    }

334    /// Handle a SAML Response received at the Assertion Consumer Service.
335    ///
336    /// Decodes, parses, validates the response, extracts the assertion
337    /// claims, and provisions or links the local user.
338    ///
339    /// `expected_request_id` — if `Some`, checked against `Response.InResponseTo`
340    /// (SEC-005/REQ-14 AC-5).  Pass the ID stored in `FederationLoginState.request_id`.
341    ///
342    /// `expected_destination` — if `Some` and non-empty, checked against
343    /// `Response.Destination` (SEC-005/REQ-14 AC-5).
344    ///
345    /// `require_in_response_to` — if `true` and `expected_request_id` is `None`,
346    /// reject a response with no `InResponseTo` at all (unsolicited-response
347    /// defense for callers, such as the authenticated ACS path, that have no
348    /// stored request ID to compare against but still must not accept an
349    /// out-of-band response). Has no effect when `expected_request_id` is
350    /// `Some` — that already enforces presence-and-equality (SECFIX-04/SEC-005).
351    pub async fn handle_saml_response(
352        &self,
353        tenant_id: Uuid,
354        config_id: Uuid,
355        saml_response_b64: &str,
356        _relay_state: Option<&str>,
357        expected_request_id: Option<&str>,
358        expected_destination: Option<&str>,
359        require_in_response_to: bool,
360    ) -> Result<FederationCallbackResult, FederationError> {
361        let config = self
362            .federation_config_repo
363            .get_by_id(tenant_id, config_id)
364            .await
365            .map_err(|e| match e {
366                AxiamError::NotFound { id, .. } => FederationError::ConfigNotFound(id),
367                other => FederationError::Internal(other.to_string()),
368            })?;
369
370        if !config.enabled {
371            return Err(FederationError::ConfigDisabled);
372        }
373
374        if config.protocol != FederationProtocol::Saml {
375            return Err(FederationError::ProtocolMismatch(
376                "expected Saml protocol".into(),
377            ));
378        }
379
380        // Decode base64 SAML response.
381        let decoded = STANDARD.decode(saml_response_b64).map_err(|e| {
382            FederationError::SamlResponseFailed(format!("Base64 decode failed: {e}"))
383        })?;
384
385        let xml = String::from_utf8(decoded).map_err(|e| {
386            FederationError::SamlResponseFailed(format!("Invalid UTF-8 in SAML response: {e}"))
387        })?;
388
389        let response: samael::schema::Response = xml.parse().map_err(|e| {
390            FederationError::SamlResponseFailed(format!("Failed to parse SAML Response XML: {e}"))
391        })?;
392
393        // Step 1 — XML signature verification (D-06/D-07/D-08).
394        // MUST run before any claims are trusted. Fails closed when:
395        //   - idp_signing_cert_pem is None (config incomplete)
396        //   - <ds:Signature> is absent from the document
397        //   - The digest or signature value does not verify
398        self.verify_signature(xml.as_bytes(), &config)?;
399
400        // Step 1a — Protocol binding checks (SEC-005/REQ-14 AC-5).
401        // These run after signature verification so we only check authentic responses.
402
403        // InResponseTo: reject unsolicited responses when a request ID is available.
404        if let Some(expected_id) = expected_request_id
405            && !expected_id.is_empty()
406        {
407            match response.in_response_to.as_deref() {
408                None => {
409                    return Err(FederationError::SamlResponseFailed(
410                        "SAML Response missing InResponseTo (unsolicited response rejected)".into(),
411                    ));
412                }
413                Some(actual_id) if actual_id != expected_id => {
414                    return Err(FederationError::SamlResponseFailed(format!(
415                        "SAML Response InResponseTo mismatch: expected {expected_id}, \
416                         got {actual_id}"
417                    )));
418                }
419                _ => {}
420            }
421        } else if require_in_response_to && response.in_response_to.is_none() {
422            // No stored expected_request_id is available (e.g. the authenticated
423            // ACS path has no FederationLoginState row), but the caller still
424            // requires presence to reject unsolicited responses (SECFIX-04/SEC-005).
425            return Err(FederationError::SamlResponseFailed(
426                "SAML Response missing InResponseTo (unsolicited response rejected)".into(),
427            ));
428        }
429
430        // Destination: reject responses not addressed to this ACS URL.
431        if let Some(expected_dest) = expected_destination
432            && !expected_dest.is_empty()
433        {
434            match response.destination.as_deref() {
435                None => {
436                    return Err(FederationError::SamlResponseFailed(
437                        "SAML Response missing Destination".into(),
438                    ));
439                }
440                Some(actual_dest) if actual_dest != expected_dest => {
441                    return Err(FederationError::SamlResponseFailed(format!(
442                        "SAML Response Destination mismatch: expected {expected_dest}, \
443                         got {actual_dest}"
444                    )));
445                }
446                _ => {}
447            }
448        }
449
450        // Validate status.
451        let status = response.status.as_ref().ok_or_else(|| {
452            FederationError::SamlResponseFailed("SAML Response missing Status element".into())
453        })?;
454
455        let status_value = status.status_code.value.as_deref().unwrap_or("");
456        if status_value != SAML_STATUS_SUCCESS {
457            return Err(FederationError::SamlResponseFailed(format!(
458                "SAML Response status is not Success: {status_value}"
459            )));
460        }
461
462        // Extract assertion.
463        let assertion = response.assertion.as_ref().ok_or_else(|| {
464            FederationError::SamlResponseFailed("SAML Response missing Assertion".into())
465        })?;
466
467        // Step 2 — XSW (XML Signature Wrapping) binding check (SECFIX-04/SEC-005).
468        // `verify_signature` above only proves SOME valid <ds:Signature> exists
469        // somewhere in the document; it never surfaces which element ID it
470        // verified. `response.assertion` is a scalar field, so a wrapped/duplicated
471        // second <Assertion> sibling would otherwise be trusted unchallenged. Bind
472        // the cryptographically verified signature to THIS consumed assertion by
473        // raw-XML introspection: exactly one <Assertion> must exist document-wide,
474        // and at least one <Signature>'s <Reference URI> must resolve to this
475        // assertion's ID. Must run AFTER verify_signature (378) and AFTER the
476        // assertion is read (above), BEFORE any of its claims are trusted.
477        bind_signature_to_assertion(xml.as_bytes(), &assertion.id)?;
478
479        // Validate conditions — REQUIRED (SEC-005/REQ-14 AC-5).
480        // An assertion without a Conditions block has no validity window and no
481        // audience restriction, so it must be rejected to prevent XSW attacks.
482        let conditions = assertion.conditions.as_ref().ok_or_else(|| {
483            FederationError::SamlResponseFailed(
484                "Assertion missing required Conditions element".into(),
485            )
486        })?;
487
488        {
489            let now = Utc::now();
490            if let Some(not_before) = conditions.not_before
491                && now < not_before
492            {
493                return Err(FederationError::SamlResponseFailed(format!(
494                    "Assertion not yet valid (NotBefore: {not_before})"
495                )));
496            }
497            if let Some(not_on_or_after) = conditions.not_on_or_after
498                && now >= not_on_or_after
499            {
500                return Err(FederationError::SamlResponseFailed(format!(
501                    "Assertion expired (NotOnOrAfter: {not_on_or_after})"
502                )));
503            }
504
505            // Validate audience restriction matches our SP entity ID.
506            if let Some(audience_restrictions) = &conditions.audience_restrictions {
507                let sp_entity_id = &config.client_id;
508                let audience_matches = audience_restrictions
509                    .iter()
510                    .any(|ar| ar.audience.iter().any(|aud| aud == sp_entity_id));
511                if !audience_matches {
512                    return Err(FederationError::SamlResponseFailed(format!(
513                        "Audience restriction does not match \
514                             SP entity ID: {sp_entity_id}"
515                    )));
516                }
517            }
518        }
519
520        // Step 3 — Assertion replay protection (D-09).
521        // Record the assertion ID in saml_assertion_replay. Returns
522        // ReplayDetected on UNIQUE violation (same tenant_id + assertion_id).
523        // Use NotOnOrAfter as the row TTL (Conditions is now required above;
524        // fall back to 1 hour from now if NotOnOrAfter is not set).
525        let replay_expires_at = conditions
526            .not_on_or_after
527            .unwrap_or_else(|| Utc::now() + chrono::Duration::hours(1));
528
529        self.replay_repo
530            .insert_assertion(tenant_id, &assertion.id, replay_expires_at)
531            .await
532            .map_err(|e| match e {
533                AxiamError::ReplayDetected => FederationError::AssertionReplay,
534                other => FederationError::Internal(other.to_string()),
535            })?;
536
537        // Extract claims from assertion.
538        let claims = extract_assertion_claims(assertion)?;
539
540        info!(
541            tenant_id = %tenant_id,
542            config_id = %config_id,
543            name_id = %claims.name_id,
544            "SAML ACS: assertion validated successfully"
545        );
546
547        // Apply attribute_map to resolve email and display name.
548        let (email, display_name) = apply_attribute_map(&claims, &config.attribute_map);
549
550        self.provision_or_link_user(
551            tenant_id,
552            config_id,
553            &claims.name_id,
554            email.as_deref(),
555            display_name.as_deref(),
556        )
557        .await
558    }

606    /// Verify the XML signature(s) in a SAML document against the configured
607    /// IdP signing certificate.
608    ///
609    /// Behaviour (D-06/D-07/D-08):
610    /// - If `config.idp_signing_cert_pem` is `None` → `ConfigIncomplete` (fail
611    ///   closed: the config is not finished).
612    /// - If no `<ds:Signature>` is present in the document → `SamlSignatureInvalid`
613    ///   (samael's `verify_signed_xml` returns an error when there is no signature).
614    /// - If the digest or signature value does not match → `SamlSignatureInvalid`.
615    ///
616    /// Must be called BEFORE `validate_conditions` so that a forged/unsigned
617    /// assertion is rejected before any claims are trusted.
618    fn verify_signature(
619        &self,
620        xml_bytes: &[u8],
621        config: &axiam_core::models::federation::FederationConfig,
622    ) -> Result<(), FederationError> {
623        let pem = config
624            .idp_signing_cert_pem
625            .as_deref()
626            .ok_or(FederationError::ConfigIncomplete)?;
627
628        let der = crate::cert::pem_cert_to_der(pem)?;
629        let cert = samael::crypto::CertificateDer::from(der);
630
631        <samael::crypto::XmlSec as samael::crypto::CryptoProvider>::verify_signed_xml(
632            xml_bytes,
633            &cert,
634            Some("ID"),
635        )
636        .map_err(|e| FederationError::SamlSignatureInvalid(e.to_string()))
637    }

690    /// Create a new local user and federation link for a SAML identity.
691    async fn provision_new_user(
692        &self,
693        tenant_id: Uuid,
694        config_id: Uuid,
695        name_id: &str,
696        email: Option<&str>,
697        _display_name: Option<&str>,
698    ) -> Result<FederationCallbackResult, FederationError> {
699        // Derive a deterministic unique username: prefer email (likely
700        // unique per-tenant), otherwise use config_id + name_id. Display
701        // names are neither stable nor unique, so they are not used.
702        let username = email
703            .map(String::from)
704            .unwrap_or_else(|| format!("federated-{config_id}-{name_id}"));
705
706        // Prefer an explicit email attribute; otherwise, if the SAML NameID is
707        // itself email-shaped (the common `emailAddress` NameID format), use it
708        // directly. Only fall back to a synthetic, namespaced address when the
709        // NameID is opaque (e.g. a persistent/transient identifier).
710        let user_email = email
711            .map(String::from)
712            .or_else(|| name_id.contains('@').then(|| name_id.to_string()))
713            .unwrap_or_else(|| format!("{name_id}.{config_id}@federated.local"));
714
715        // Federated users get a random non-usable password since they
716        // authenticate through the external IdP.
717        //
718        // Deliberately v4, not `new_id()`: this value must be unguessable, and
719        // a v7 id spends 48 bits on a timestamp that an attacker can bound from
720        // the user's creation time. See `axiam_core::id` — v7 is for
721        // identifiers, never for secrets.
722        let random_password = Uuid::new_v4().to_string();
723
724        let create_user = CreateUser {
725            tenant_id,
726            username,
727            email: user_email.clone(),
728            password: random_password,
729            metadata: Some(serde_json::json!({
730                "federation_config_id": config_id.to_string(),
731                "external_subject": name_id,
732                "provisioned_by": "saml_federation",
733            })),
734        };
735
736        let user = self.user_repo.create(create_user).await.map_err(|e| {
737            FederationError::ProvisioningFailed(format!("Failed to create user: {e}"))
738        })?;
739
740        let create_link = CreateFederationLink {
741            tenant_id,
742            user_id: user.id,
743            federation_config_id: config_id,
744            external_subject: name_id.to_string(),
745            external_email: Some(user_email),
746        };
747
748        let link = self
749            .federation_link_repo
750            .create(create_link)
751            .await
752            .map_err(|e| {
753                FederationError::ProvisioningFailed(format!(
754                    "Failed to create federation link: {e}"
755                ))
756            })?;
757
758        info!(
759            tenant_id = %tenant_id,
760            user_id = %user.id,
761            name_id = %name_id,
762            "Provisioned new SAML-federated user"
763        );
764
765        Ok(FederationCallbackResult {
766            user,
767            federation_link: link,
768            newly_provisioned: true,
769        })
770    }

777/// Bind the cryptographically verified XML signature to the assertion actually
778/// consumed by `handle_saml_response` (SECFIX-04/SEC-005 — XML Signature
779/// Wrapping defense).
780///
781/// `samael::crypto::verify_signed_xml` (called by `verify_signature` above)
782/// only proves that SOME valid `<ds:Signature>` exists somewhere in the
783/// document — it never surfaces which element ID the verified signature's
784/// `<Reference URI="#...">` pointed to. Meanwhile `samael::schema::Response`
785/// exposes `assertion` as a *scalar* `Option<Assertion>`, so an attacker can
786/// keep the original signed assertion intact somewhere in the tree (so the
787/// lone-signature check still passes) and inject a second, forged, unsigned
788/// `<Assertion>` sibling that the deserializer happens to bind to
789/// `response.assertion`.
790///
791/// This performs an independent raw-XML introspection pass (via `libxml`,
792/// already resolved transitively through samael 0.0.19's `xmlsec` feature)
793/// to close that gap:
794///
795/// 1. Exactly one `<Assertion>` element (namespace-agnostic local name) must
796///    exist anywhere in the document — rejects the wrapped/duplicated payload
797///    shape outright, regardless of signature status.
798/// 2. At least one `<Signature>`'s `<Reference URI="#...">` must resolve to
799///    `claimed_assertion_id` — binds "the element that was cryptographically
800///    verified" to "the element whose claims are about to be trusted".
801///    Rejects on an empty, absent, or non-matching reference.
802///
803/// Never uses regex/string search on the XML (a real, namespace-aware XPath
804/// parser is required — see 23-RESEARCH.md Anti-Patterns).
805fn bind_signature_to_assertion(
806    xml_bytes: &[u8],
807    claimed_assertion_id: &str,
808) -> Result<(), FederationError> {
809    let parser = libxml::parser::Parser::default();
810    let doc = parser.parse_string(xml_bytes).map_err(|e| {
811        FederationError::SamlResponseFailed(format!(
812            "XSW binding check: failed to re-parse SAML response XML: {e}"
813        ))
814    })?;
815
816    let mut context = libxml::xpath::Context::new(&doc).map_err(|()| {
817        FederationError::SamlResponseFailed(
818            "XSW binding check: failed to create XPath context".into(),
819        )
820    })?;
821
822    // 1. Exactly one Assertion element must exist anywhere in the document.
823    let assertions = context
824        .findnodes("//*[local-name()='Assertion']", None)
825        .map_err(|()| {
826            FederationError::SamlResponseFailed(
827                "XSW binding check: XPath evaluation failed (Assertion)".into(),
828            )
829        })?;
830    if assertions.len() != 1 {
831        return Err(FederationError::SamlResponseFailed(format!(
832            "expected exactly 1 Assertion element in SAML Response, found {} \
833             (possible XML Signature Wrapping attack)",
834            assertions.len()
835        )));
836    }
837
838    // 2. Every <Signature>'s Reference URI must resolve to the consumed
839    //    assertion's ID. Reject on empty/absent/non-matching references.
840    let references = context
841        .findnodes(
842            "//*[local-name()='Signature']//*[local-name()='Reference']/@URI",
843            None,
844        )
845        .map_err(|()| {
846            FederationError::SamlResponseFailed(
847                "XSW binding check: XPath evaluation failed (Signature/Reference)".into(),
848            )
849        })?;
850
851    let expected_reference = format!("#{claimed_assertion_id}");
852    let bound = references.iter().any(|node| {
853        let uri = node.get_content();
854        !uri.is_empty() && uri == expected_reference
855    });
856    if !bound {
857        return Err(FederationError::SamlResponseFailed(
858            "no verified Signature references the consumed Assertion \
859             (XML Signature Wrapping rejected)"
860                .into(),
861        ));
862    }
863
864    Ok(())
865}

867/// Extract NameID, session index, and attributes from a SAML assertion.
868fn extract_assertion_claims(
869    assertion: &samael::schema::Assertion,
870) -> Result<SamlAssertionClaims, FederationError> {
871    let name_id = assertion
872        .subject
873        .as_ref()
874        .and_then(|s| s.name_id.as_ref())
875        .map(|n| n.value.clone())
876        .ok_or_else(|| {
877            FederationError::SamlResponseFailed("Assertion missing Subject/NameID".into())
878        })?;
879
880    let session_index = assertion
881        .authn_statements
882        .as_ref()
883        .and_then(|stmts| stmts.first())
884        .and_then(|stmt| stmt.session_index.clone());
885
886    let mut attributes: HashMap<String, Vec<String>> = HashMap::new();
887    if let Some(attr_statements) = &assertion.attribute_statements {
888        for statement in attr_statements {
889            for attr in &statement.attributes {
890                let attr_name = attr
891                    .name
892                    .as_deref()
893                    .or(attr.friendly_name.as_deref())
894                    .unwrap_or("unknown");
895
896                let values: Vec<String> =
897                    attr.values.iter().filter_map(|v| v.value.clone()).collect();
898
899                attributes
900                    .entry(attr_name.to_string())
901                    .or_default()
902                    .extend(values);
903            }
904        }
905    }
906
907    Ok(SamlAssertionClaims {
908        name_id,
909        session_index,
910        attributes,
911    })
912}

914/// Apply the federation config's attribute_map to resolve email and
915/// display name from SAML assertion attributes.
916///
917/// The attribute_map is a JSON object mapping AXIAM field names to SAML
918/// attribute names, e.g.: `{"email": "mail", "name": "displayName"}`.
919fn apply_attribute_map(
920    claims: &SamlAssertionClaims,
921    attribute_map: &serde_json::Value,
922) -> (Option<String>, Option<String>) {
923    let get_mapped = |field: &str| -> Option<String> {
924        let saml_attr_name = attribute_map.get(field)?.as_str()?;
925        claims
926            .attributes
927            .get(saml_attr_name)
928            .and_then(|vals| vals.first())
929            .cloned()
930    };
931
932    let email = get_mapped("email");
933    let display_name = get_mapped("name").or_else(|| get_mapped("displayName"));
934
935    (email, display_name)
936}

948/// DEFLATE-compress the input bytes (raw deflate, no zlib/gzip headers).
