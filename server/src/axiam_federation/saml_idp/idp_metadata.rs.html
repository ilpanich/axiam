1//! The tenant's IdP metadata document (G-2, T23.2.5, D-40).
2//!
3//! What a service provider's administrator pastes into their SP: one
4//! `EntityDescriptor` for the tenant's IdP, with the certificates it will sign
5//! assertions with and the SSO locations. Built from a **fixed template**, with
6//! every value escaped through [`xml::escape`](super::xml::escape) — nothing is
7//! serialized from a data structure, so nothing that is not named here can reach
8//! the document.
9//!
10//! # What it carries
11//!
12//! * `entityID`, the two `SingleSignOnService` locations (HTTP-Redirect and
13//!   HTTP-POST, both at the SSO endpoint) and the two `SingleLogoutService`
14//!   locations (both at the SLO endpoint, T23.2.4) — passed in by the caller from
15//!   [`idp_entity_id`](super::idp_entity_id),
16//!   [`idp_sso_url`](super::idp_sso_url) and
17//!   [`idp_slo_url`](super::idp_slo_url) of the **path** tenant, never from a
18//!   stored row (T-307, T-367);
19//! * one `KeyDescriptor use="signing"` per publishable credential, **`active`
20//!   first and then `next`**, each an `X509Certificate` and nothing else, so an
21//!   SP has the successor before it ever sees an assertion signed with it
22//!   (T-309). A retired credential's key is destroyed and its certificate must
23//!   stop being trusted, so it is never published;
24//! * both `NameIDFormat`s the IdP can issue (persistent and `emailAddress`).
25//!
26//! # What it does not carry
27//!
28//! No encryption key (AXIAM decrypts nothing), no `validUntil`,
29//! `cacheDuration`, `Organization` or `ContactPerson`. The `SingleLogoutService`
30//! elements arrived in the commit that added the `/slo` route (T23.2.4): the
31//! document never advertises a route that is not there.
32//!
33//! # Unsigned, on purpose
34//!
35//! Signing the metadata with the key it publishes anchors nothing — whoever
36//! could swap the document could swap the key — and a signed `EntityDescriptor`
37//! is one more document under the tenant key (T-316). Trust comes from TLS to the
38//! deployment's origin and, for a careful SP administrator, from comparing the
39//! SHA-256 fingerprint contract §29 shows.
40

51/// `Cache-Control` of the document: an hour (D-40). The rotation guidance is to
52/// issue `next`, wait at least this long and the SPs' own refresh interval, and
53/// only then promote.
54pub const IDP_METADATA_CACHE_CONTROL: &str = "public, max-age=3600";

56/// The SAML 2.0 protocol URN `protocolSupportEnumeration` names.
57const PROTOCOL_SAML2: &str = "urn:oasis:names:tc:SAML:2.0:protocol";
58/// HTTP-Redirect binding.
59const BINDING_REDIRECT: &str = "urn:oasis:names:tc:SAML:2.0:bindings:HTTP-Redirect";
60/// HTTP-POST binding.
61const BINDING_POST: &str = "urn:oasis:names:tc:SAML:2.0:bindings:HTTP-POST";
62/// Persistent `NameIDFormat`.
63const NAME_ID_PERSISTENT: &str = "urn:oasis:names:tc:SAML:2.0:nameid-format:persistent";
64/// `emailAddress` `NameIDFormat`.
65const NAME_ID_EMAIL: &str = "urn:oasis:names:tc:SAML:1.1:nameid-format:emailAddress";
66
67/// A rendered metadata document and its validator.
68#[derive(Debug, Clone, PartialEq, Eq)]
69pub struct IdpMetadataDocument {
70    /// The document, UTF-8 XML.
71    pub xml: String,
72    /// A strong `ETag` — the SHA-256 of [`Self::xml`], hex, in double quotes.
73    pub etag: String,
74}
75
76/// The credentials a metadata document publishes, in the order it publishes
77/// them: `active`, then `next`; nothing else. A credential whose certificate
78/// does not decode is not publishable (and is skipped, never half-written).
79///
80/// Returns each with its certificate's DER.
81fn publishable(credentials: &[SamlIdpCredential]) -> Vec<(&SamlIdpCredential, Vec<u8>)> {
82    [
83        SamlIdpCredentialStatus::Active,
84        SamlIdpCredentialStatus::Next,
85    ]
86    .into_iter()
87    .filter_map(|wanted| {
88        credentials
89            .iter()
90            .find(|c| c.status == wanted)
91            .and_then(|c| {
92                crate::cert::pem_cert_to_der(&c.certificate_pem)
93                    .ok()
94                    .map(|d| (c, d))
95            })
96    })
97    .collect()
98}

100/// The tenant's metadata, or `None` when there is nothing to publish — no
101/// `active` or `next` credential with a decodable certificate. The caller
102/// answers that with the D-20 `404`, the same as a tenant that serves no SAML
103/// (a `503` would tell anyone the tenant exists, T-368).
104///
105/// `slo_url` is the tenant's SLO endpoint when this deployment serves one —
106/// always, since T23.2.4 — and `None` for a document that must not advertise a
107/// logout route.
108///
109/// `credentials` is the tenant's keyless list ([`SamlIdpCredentialRepository::list`],
110/// never `get_active_sealed`): at most one credential per status is published.
111///
112/// [`SamlIdpCredentialRepository::list`]: axiam_core::repository::SamlIdpCredentialRepository::list
113#[must_use]
114pub fn build_idp_metadata(
115    entity_id: &str,
116    sso_url: &str,
117    slo_url: Option<&str>,
118    credentials: &[SamlIdpCredential],
119) -> Option<IdpMetadataDocument> {
120    let keys = publishable(credentials);
121    if keys.is_empty() {
122        return None;
123    }
124
125    let mut xml = String::with_capacity(2048 + keys.len() * 2048);
126    xml.push_str("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n");
127    xml.push_str(
128        "<md:EntityDescriptor xmlns:md=\"urn:oasis:names:tc:SAML:2.0:metadata\" \
129         xmlns:ds=\"http://www.w3.org/2000/09/xmldsig#\" entityID=\"",
130    );
131    xml.push_str(&escape(entity_id));
132    xml.push_str("\">\n  <md:IDPSSODescriptor protocolSupportEnumeration=\"");
133    xml.push_str(PROTOCOL_SAML2);
134    xml.push_str("\">\n");
135    for (_, der) in &keys {
136        // One line of standard base64: what every SP's parser reads, and what
137        // the credential's own certificate decodes back from.
138        xml.push_str(
139            "    <md:KeyDescriptor use=\"signing\"><ds:KeyInfo><ds:X509Data>\
140             <ds:X509Certificate>",
141        );
142        xml.push_str(&STANDARD.encode(der));
143        xml.push_str("</ds:X509Certificate></ds:X509Data></ds:KeyInfo></md:KeyDescriptor>\n");
144    }
145    // Schema order (SAML Metadata §2.4.3): the key descriptors, then the logout
146    // services, then the `NameIDFormat`s, then the sign-on services.
147    if let Some(slo_url) = slo_url {
148        for binding in [BINDING_REDIRECT, BINDING_POST] {
149            xml.push_str("    <md:SingleLogoutService Binding=\"");
150            xml.push_str(binding);
151            xml.push_str("\" Location=\"");
152            xml.push_str(&escape(slo_url));
153            xml.push_str("\"/>\n");
154        }
155    }
156    for format in [NAME_ID_PERSISTENT, NAME_ID_EMAIL] {
157        xml.push_str("    <md:NameIDFormat>");
158        xml.push_str(format);
159        xml.push_str("</md:NameIDFormat>\n");
160    }
161    for binding in [BINDING_REDIRECT, BINDING_POST] {
162        xml.push_str("    <md:SingleSignOnService Binding=\"");
163        xml.push_str(binding);
164        xml.push_str("\" Location=\"");
165        xml.push_str(&escape(sso_url));
166        xml.push_str("\"/>\n");
167    }
168    xml.push_str("  </md:IDPSSODescriptor>\n</md:EntityDescriptor>\n");
169
170    let etag = format!("\"{}\"", hex::encode(Sha256::digest(xml.as_bytes())));
171    Some(IdpMetadataDocument { xml, etag })
172}

174/// Whether an `If-None-Match` header value matches `etag` (RFC 9110 §13.1.2,
175/// weak comparison): `*`, or any listed entity tag equal to `etag` once a `W/`
176/// prefix is ignored. The caller answers `304` on `true`.
177#[must_use]
178pub fn if_none_match_matches(header: &str, etag: &str) -> bool {
179    let wanted = etag.trim_start_matches("W/");
180    header
181        .split(',')
182        .map(str::trim)
183        .any(|candidate| candidate == "*" || candidate.trim_start_matches("W/") == wanted)
184}

193    /// A self-signed certificate generated now; no PEM literal in the source.
194    fn cert_pem() -> String {
195        let key = rcgen::KeyPair::generate().expect("key pair");
196        rcgen::CertificateParams::new(vec!["idp.example.test".to_string()])
197            .expect("params")
198            .self_signed(&key)
199            .expect("self-signed")
200            .pem()
201    }

203    fn credential(status: SamlIdpCredentialStatus, pem: String) -> SamlIdpCredential {
204        SamlIdpCredential {
205            id: Uuid::new_v4(),
206            tenant_id: Uuid::nil(),
207            issuer_ca_id: Uuid::nil(),
208            certificate_pem: pem,
209            serial: "01".into(),
210            fingerprint: "00".into(),
211            not_before: Utc::now() - Duration::hours(1),
212            not_after: Utc::now() + Duration::days(30),
213            status,
214            key_custody: CaKeyCustody::Database,
215            created_at: Utc::now(),
216            retired_at: None,
217        }
218    }

229    fn der_b64(pem: &str) -> String {
230        STANDARD.encode(crate::cert::pem_cert_to_der(pem).expect("der"))
231    }

233    #[test]
234    fn active_is_published_before_next_whatever_the_list_order() {
235        let (active, next) = (cert_pem(), cert_pem());
236        let doc = body(&[
237            credential(SamlIdpCredentialStatus::Next, next.clone()),
238            credential(SamlIdpCredentialStatus::Active, active.clone()),
239        ])
240        .expect("publishable");
241        let first = doc.xml.find(&der_b64(&active)).expect("active published");
242        let second = doc.xml.find(&der_b64(&next)).expect("next published");
243        assert!(first < second, "active comes first");
244        assert_eq!(doc.xml.matches("<md:KeyDescriptor").count(), 2);
245    }

323    fn every_value_is_escaped_into_the_template() {
324        let doc = build_idp_metadata(
325            "https://x.test/a\"><evil/>&",
326            "https://x.test/s?a=1&b=\"2\"<",

360    #[test]
361    fn the_etag_is_a_strong_digest_of_the_body_and_changes_with_it() {
362        let one = body(&[credential(SamlIdpCredentialStatus::Active, cert_pem())]).unwrap();
363        let again = build_idp_metadata(
364            "https://iam.example.test/saml/v2/t/metadata",
365            "https://iam.example.test/saml/v2/t/sso",
366            Some("https://iam.example.test/saml/v2/t/slo"),
367            &[credential(SamlIdpCredentialStatus::Active, cert_pem())],
368        )
369        .unwrap();
370        assert!(one.etag.starts_with('"') && one.etag.ends_with('"'));
371        assert!(!one.etag.starts_with("W/"));
372        assert_eq!(one.etag.len(), 64 + 2);
373        assert_ne!(
374            one.etag, again.etag,
375            "a different certificate, a different tag"
376        );
377        assert_eq!(
378            one.etag,
379            format!("\"{}\"", hex::encode(Sha256::digest(one.xml.as_bytes())))
380        );
381    }

383    #[test]
384    fn if_none_match_uses_weak_comparison_over_a_list_and_star() {
385        let tag = "\"abc\"";
386        assert!(if_none_match_matches("\"abc\"", tag));
387        assert!(if_none_match_matches("W/\"abc\"", tag));
388        assert!(if_none_match_matches("\"x\", \"abc\"", tag));
389        assert!(if_none_match_matches("*", tag));
390        assert!(!if_none_match_matches("\"abd\"", tag));
391        assert!(!if_none_match_matches("", tag));
