1//! SAML single logout: receiving an SP's `LogoutRequest` / `LogoutResponse` and
2//! building AXIAM's own (G-2, T23.2.4, D-37, D-38, D-39).
3//!
4//! Pure and synchronous, like the rest of this module: no I/O, no lookups. The
5//! SLO endpoint (`handlers::saml_idp_slo`) calls these in order — decode with
6//! [`super::request`]'s functions, [`parse_logout_message`], find the SP by the
7//! parsed `Issuer`, verify — and every refusal here happens **before** any SP is
8//! looked up, any session is touched or anything is sent.
9//!
10//! # Receiving
11//!
12//! Exactly [`super::request`]'s receiver: encoded value at most 96 KiB, document
13//! at most 64 KiB, the inflate cap, markup declarations and non-UTF-8 encodings
14//! refused on the bytes, libxml without recovery and without network. Then:
15//!
16//! * `Version` 2.0; `ID` a request id ([`super::check_request_id`]); an
17//!   `IssueInstant` at most five minutes old and 60 s ahead; one `Issuer`;
18//! * a `LogoutRequest`'s `NotOnOrAfter`, when present, has not passed;
19//! * a `LogoutRequest` names **exactly one `NameID`**; a `BaseID` or an
20//!   `EncryptedID` is refused, because AXIAM issues neither;
21//! * at most [`MAX_SESSION_INDEXES`] `SessionIndex` elements; none means every
22//!   session in which the SP holds that `NameID`;
23//! * a `LogoutResponse` has an `InResponseTo` and a `Status`.
24//!
25//! The signature is **not** checked here: [`parse_logout_message`] reports where
26//! it sits ([`signature_placement`](super::request::signature_placement), the
27//! D-23 rule, for whichever root the message has), and the endpoint verifies it
28//! with [`super::request::verify_post_signature`] (xmlsec, SHA-1 refused) or
29//! [`RedirectQuery::verify_signature`](super::request::RedirectQuery::verify_signature)
30//! — never `verify_signed_xml`, which checks only the first signature.
31//!
32//! # Sending, narrowly
33//!
34//! AXIAM signs a logout message in two cases only, and the endpoint enforces
35//! both before calling here: a `LogoutRequest` for a session that was ended —
36//! by its holder, or by a **verified** SP request — and a `LogoutResponse`
37//! replying to a verified `LogoutRequest`. Nothing here signs for an unverified
38//! party, so the tenant's key is never a signing oracle (T-373).
39//!
40//! * **HTTP-Redirect**: the signature is the **detached query signature**
41//!   (`rsa-sha256` over `SAMLRequest=…|SAMLResponse=…[&RelayState=…]&SigAlg=…`),
42//!   so **no XML signature exists** to be harvested as a wrapping gadget.
43//! * **HTTP-POST**: enveloped exactly as the assertion's (the root's child, one
44//!   reference to the root `ID`, exclusive c14n, `rsa-sha256`/`sha256`), and
45//!   **re-verified** — shape and xmlsec — before it is returned.
46//!
47//! The destination is the SP's registered `slo_url` and the binding its
48//! registered `slo_binding`, never a location from a message.
49

154/// A received logout message, parsed and not yet trusted.
155#[derive(Debug, Clone, PartialEq, Eq)]
156pub enum ParsedLogoutMessage {
157    /// An SP asks AXIAM to end a session.
158    Request(ParsedLogoutRequest),
159    /// An SP answers a request of AXIAM's.
160    Response(ParsedLogoutResponse),
161}

163impl ParsedLogoutMessage {
164    /// The message's `Issuer`: the entity id to look the SP up by.
165    #[must_use]
166    pub fn issuer(&self) -> &str {
167        match self {
168            Self::Request(r) => &r.issuer,
169            Self::Response(r) => &r.issuer,
170        }
171    }
172
173    /// The message's `Destination`, when it has one.
174    #[must_use]
175    pub fn destination(&self) -> Option<&str> {
176        match self {
177            Self::Request(r) => r.destination.as_deref(),
178            Self::Response(r) => r.destination.as_deref(),
179        }
180    }
181
182    /// Whether the document carries an enveloped signature.
183    #[must_use]
184    pub fn enveloped_signature(&self) -> bool {
185        match self {
186            Self::Request(r) => r.enveloped_signature,
187            Self::Response(r) => r.enveloped_signature,
188        }
189    }
190}

192/// Parse a received logout message and check everything that needs no SP:
193/// shape, `Version`, `ID`, `IssueInstant` against `now`, `Issuer`, the
194/// request's `NotOnOrAfter`, `NameID`, the `SessionIndex` count, a response's
195/// `InResponseTo` and `Status`, and where a signature sits.
196///
197/// # Errors
198///
199/// A [`RequestError`]; nothing about the document is in it.
200pub fn parse_logout_message(
201    document: &str,
202    now: DateTime<Utc>,
203) -> Result<ParsedLogoutMessage, RequestError> {
204    let doc = parse_xml(document)?;
205    let root = doc.get_root_element().ok_or(RequestError::Malformed)?;
206    let is_request = is_element(&root, NS_PROTOCOL, "LogoutRequest");
207    let is_response = is_element(&root, NS_PROTOCOL, "LogoutResponse");
208    if !is_request && !is_response {
209        return Err(RequestError::NotALogoutMessage);
210    }
211    if root.get_attribute("Version").as_deref() != Some("2.0") {
212        return Err(RequestError::NotALogoutMessage);
213    }
214    let id = root.get_attribute("ID").ok_or(RequestError::InvalidId)?;
215    check_request_id(&id).map_err(|_| RequestError::InvalidId)?;
216    let issue_instant = fresh_issue_instant(&root, now)?;
217    let children = root.get_child_elements();
218    let issuer = read_issuer(&children)?;
219    let destination = root.get_attribute("Destination");
220    let enveloped_signature = signature_placement(&doc, &root, &id)?;
221
222    if is_response {
223        let in_response_to = root
224            .get_attribute("InResponseTo")
225            .ok_or(RequestError::InvalidField)?;
226        check_request_id(&in_response_to).map_err(|_| RequestError::InvalidField)?;
227        return Ok(ParsedLogoutMessage::Response(ParsedLogoutResponse {
228            id,
229            issue_instant,
230            destination,
231            issuer,
232            in_response_to,
233            status: read_status(&children)?,
234            enveloped_signature,
235        }));
236    }
237
238    let not_on_or_after = root
239        .get_attribute("NotOnOrAfter")
240        .map(|raw| {
241            DateTime::parse_from_rfc3339(raw.trim())
242                .map(|t| t.with_timezone(&Utc))
243                .map_err(|_| RequestError::InvalidField)
244        })
245        .transpose()?;
246    let skew = Duration::seconds(crate::oidc::CLOCK_SKEW_LEEWAY_SECS as i64);
247    if not_on_or_after.is_some_and(|limit| limit + skew <= now) {
248        return Err(RequestError::Expired);
249    }
250
251    // The principal. AXIAM issues a `NameID` and nothing else, so a `BaseID` or
252    // an `EncryptedID` names nobody it knows: refused outright, and never read
253    // as "no NameID, therefore every session".
254    if children.iter().any(|c| {
255        is_element(c, NS_ASSERTION, "BaseID") || is_element(c, NS_ASSERTION, "EncryptedID")
256    }) {
257        return Err(RequestError::NameIdUnsupported);
258    }
259    let names: Vec<_> = children
260        .iter()
261        .filter(|c| is_element(c, NS_ASSERTION, "NameID"))
262        .collect();
263    let [name_id] = names.as_slice() else {
264        return Err(RequestError::NameIdMissing);
265    };
266    let name_id_format = name_id.get_attribute("Format");
267    let name_id = name_id.get_content().trim().to_owned();
268    if name_id.is_empty() {
269        return Err(RequestError::NameIdMissing);
270    }
271
272    let mut session_indexes: Vec<String> = Vec::new();
273    let mut seen = 0usize;
274    for child in children
275        .iter()
276        .filter(|c| is_element(c, NS_PROTOCOL, "SessionIndex"))
277    {
278        seen += 1;
279        if seen > MAX_SESSION_INDEXES {
280            return Err(RequestError::TooManySessionIndexes);
281        }
282        let index = child.get_content().trim().to_owned();
283        if index.is_empty() || index.len() > super::MAX_SESSION_INDEX_BYTES {
284            return Err(RequestError::InvalidField);
285        }
286        if !session_indexes.contains(&index) {
287            session_indexes.push(index);
288        }
289    }
290
291    Ok(ParsedLogoutMessage::Request(ParsedLogoutRequest {
292        id,
293        issue_instant,
294        destination,
295        issuer,
296        not_on_or_after,
297        name_id,
298        name_id_format,
299        session_indexes,
300        enveloped_signature,
301    }))
302}

304/// The status a `LogoutResponse` reports: one `samlp:Status` with one top-level
305/// `StatusCode`, which may carry one nested code.
306fn read_status(children: &[libxml::tree::Node]) -> Result<LogoutStatus, RequestError> {
307    let statuses: Vec<_> = children
308        .iter()
309        .filter(|c| is_element(c, NS_PROTOCOL, "Status"))
310        .collect();
311    let [status] = statuses.as_slice() else {
312        return Err(RequestError::InvalidField);
313    };
314    let codes: Vec<_> = status
315        .get_child_elements()
316        .into_iter()
317        .filter(|c| is_element(c, NS_PROTOCOL, "StatusCode"))
318        .collect();
319    let [top] = codes.as_slice() else {
320        return Err(RequestError::InvalidField);
321    };
322    let top_value = top
323        .get_attribute("Value")
324        .ok_or(RequestError::InvalidField)?;
325    if top_value != format!("{STATUS_PREFIX}Success") {
326        return Ok(LogoutStatus::Failed);
327    }
328    let nested: Vec<_> = top
329        .get_child_elements()
330        .into_iter()
331        .filter(|c| is_element(c, NS_PROTOCOL, "StatusCode"))
332        .collect();
333    match nested.as_slice() {
334        [] => Ok(LogoutStatus::Success),
335        [second] if second.get_attribute("Value").as_deref() == Some(STATUS_PARTIAL_LOGOUT) => {
336            Ok(LogoutStatus::PartialLogout)
337        }
338        _ => Ok(LogoutStatus::Failed),
339    }
340}

358/// A logout message ready to deliver through the browser.
359#[derive(Debug, Clone, PartialEq, Eq)]
360pub struct OutboundLogout {
361    /// The message's `ID`. For a `LogoutRequest`, 256 random bits: the caller
362    /// stores only its SHA-256 and expects the SP's `InResponseTo` to name it.
363    pub id: String,
364    /// How to deliver it.
365    pub delivery: LogoutDelivery,
366}

368/// How an outbound logout message reaches the SP's registered endpoint.
369#[derive(Clone, PartialEq, Eq)]
370pub enum LogoutDelivery {
371    /// HTTP-Redirect: answer with a redirect to this URL — the SP's registered
372    /// `slo_url` with the message and its detached signature in the query.
373    Redirect {
374        /// The full `Location`.
375        location: String,
376    },
377    /// HTTP-POST: render an auto-submitting form posting `value` as `field` to
378    /// `destination`.
379    Post {
380        /// The SP's registered `slo_url`, also the message's `Destination`.
381        destination: String,
382        /// `SAMLRequest` or `SAMLResponse`.
383        field: &'static str,
384        /// The signed document, base64, one line.
385        value: String,
386        /// The `RelayState` to echo, verbatim.
387        relay_state: Option<String>,
388    },
389}
390
391// The message is a signed logout of a person: never formatted.
392impl std::fmt::Debug for LogoutDelivery {
393    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
394        match self {
395            Self::Redirect { .. } => f.write_str("LogoutDelivery::Redirect([REDACTED])"),
396            Self::Post { field, .. } => write!(f, "LogoutDelivery::Post({field}, [REDACTED])"),
397        }
398    }
399}

401/// A fresh `LogoutRequest` `ID`: `_` and 64 lower-case hex digits of CSPRNG
402/// output — **256 random bits** (D-39), because the SP's `InResponseTo` is the
403/// only thing that advances a logout chain, and the run stores its digest.
404fn new_logout_request_id() -> String {
405    let mut bytes = [0u8; 32];
406    rand::rng().fill_bytes(&mut bytes);
407    format!("_{}", hex::encode(bytes))
408}

410fn url_encode(value: &str) -> String {
411    url::form_urlencoded::byte_serialize(value.as_bytes()).collect()
412}

414/// Raw DEFLATE then base64: an HTTP-Redirect message value before URL encoding.
415fn deflate_base64(document: &[u8]) -> Result<String, SamlIdpError> {
416    let mut encoder =
417        flate2::write::DeflateEncoder::new(Vec::new(), flate2::Compression::default());
418    encoder
419        .write_all(document)
420        .map_err(|_| SamlIdpError::SigningFailed)?;
421    let compressed = encoder.finish().map_err(|_| SamlIdpError::SigningFailed)?;
422    Ok(STANDARD.encode(compressed))
423}

425/// The registered endpoint and binding, or [`SamlIdpError::SloNotRegistered`].
426fn slo_endpoint(sp: &SamlServiceProvider) -> Result<(&str, SamlBinding), SamlIdpError> {
427    match (sp.slo_url.as_deref(), sp.slo_binding) {
428        (Some(url), Some(binding)) => Ok((url, binding)),
429        _ => Err(SamlIdpError::SloNotRegistered),
430    }
431}

433impl SamlIdpIssuer {
434    /// A signed `LogoutRequest` to `sp`, for the session described by `subject`.
435    ///
436    /// **The caller has established that the session ended** — its holder asked,
437    /// or an SP's verified request did — and never calls this for anyone else
438    /// (T-373). Delivered to the SP's registered `slo_url` on its registered
439    /// `slo_binding`: HTTP-Redirect with a detached query signature (no XML
440    /// signature exists), or HTTP-POST with an enveloped one, re-verified here.
441    ///
442    /// The request carries `NotOnOrAfter` five minutes ahead and the `NameID` and
443    /// `SessionIndex` the SP was given, nothing else.
444    ///
445    /// # Errors
446    ///
447    /// A [`SamlIdpError`]: [`SamlIdpError::SloNotRegistered`],
448    /// [`SamlIdpError::TenantMismatch`], [`SamlIdpError::SessionIndexInvalid`],
449    /// the credential's refusals, [`SamlIdpError::SigningFailed`].
450    pub fn logout_request(
451        &self,
452        tenant_id: Uuid,
453        sp: &SamlServiceProvider,
454        subject: &LogoutSubject<'_>,
455        signing_key: &SamlIdpSigningKey,
456        now: DateTime<Utc>,
457    ) -> Result<OutboundLogout, SamlIdpError> {
458        if sp.tenant_id != tenant_id {
459            return Err(SamlIdpError::TenantMismatch);
460        }
461        let (slo_url, binding) = slo_endpoint(sp)?;
462        check_session_index(subject.session_index)?;
463        if subject.name_id.is_empty() || subject.name_id_format.is_empty() {
464            return Err(SamlIdpError::NameIdUnavailable);
465        }
466        let now = now.trunc_subsecs(0);
467        sign::check_credential(&signing_key.credential, tenant_id, now)?;
468
469        let id = new_logout_request_id();
470        let issuer = self.entity_id(tenant_id);
471        let not_on_or_after = now + Duration::seconds(ASSERTION_LIFETIME_SECS);
472        let build = |template: Option<&str>| {
473            let mut out = String::with_capacity(2048);
474            out.push_str(&format!(
475                r#"<samlp:LogoutRequest xmlns:samlp="{NS_PROTOCOL}" xmlns:saml="{NS_ASSERTION}" ID="{id}" Version="2.0" IssueInstant="{}" Destination="{}" NotOnOrAfter="{}">"#,
476                instant(now),
477                escape(slo_url),
478                instant(not_on_or_after),
479            ));
480            out.push_str(&format!("<saml:Issuer>{}</saml:Issuer>", escape(&issuer)));
481            if let Some(template) = template {
482                out.push_str(template);
483            }
484            if subject.name_id_format == NAME_ID_PERSISTENT {
485                out.push_str(&format!(
486                    r#"<saml:NameID Format="{}" NameQualifier="{}" SPNameQualifier="{}">{}</saml:NameID>"#,
487                    escape(subject.name_id_format),
488                    escape(&issuer),
489                    escape(&sp.entity_id),
490                    escape(subject.name_id),
491                ));
492            } else {
493                out.push_str(&format!(
494                    r#"<saml:NameID Format="{}">{}</saml:NameID>"#,
495                    escape(subject.name_id_format),
496                    escape(subject.name_id),
497                ));
498            }
499            out.push_str(&format!(
500                "<samlp:SessionIndex>{}</samlp:SessionIndex></samlp:LogoutRequest>",
501                escape(subject.session_index)
502            ));
503            out
504        };
505        let delivery = deliver(
506            MessageParam::Request,
507            "LogoutRequest",
508            &id,
509            slo_url,
510            binding,
511            None,
512            signing_key,
513            build,
514        )?;
515        Ok(OutboundLogout { id, delivery })
516    }
517
518    /// A signed `LogoutResponse` to `sp`, answering the **verified**
519    /// `LogoutRequest` whose `ID` is `in_response_to`: `Success`, or
520    /// `Success` with a nested `PartialLogout` when `partial`.
521    ///
522    /// The caller has verified that request against the SP's registered
523    /// certificate and never calls this for an unverified one (T-373).
524    /// `relay_state` is the SP's own, at most 80 bytes, echoed to that SP only.
525    ///
526    /// # Errors
527    ///
528    /// As [`Self::logout_request`], and [`SamlIdpError::InvalidRequestId`],
529    /// [`SamlIdpError::RelayStateTooLong`].
530    pub fn logout_response(
531        &self,
532        tenant_id: Uuid,
533        sp: &SamlServiceProvider,
534        in_response_to: &str,
535        relay_state: Option<&str>,
536        partial: bool,
537        signing_key: &SamlIdpSigningKey,
538        now: DateTime<Utc>,
539    ) -> Result<OutboundLogout, SamlIdpError> {
540        if sp.tenant_id != tenant_id {
541            return Err(SamlIdpError::TenantMismatch);
542        }
543        let (slo_url, binding) = slo_endpoint(sp)?;
544        check_request_id(in_response_to)?;
545        if relay_state.is_some_and(|r| r.len() > MAX_RELAY_STATE_BYTES) {
546            return Err(SamlIdpError::RelayStateTooLong);
547        }
548        let now = now.trunc_subsecs(0);
549        sign::check_credential(&signing_key.credential, tenant_id, now)?;
550
551        let id = xml::new_id();
552        let issuer = self.entity_id(tenant_id);
553        let build = |template: Option<&str>| {
554            let mut out = String::with_capacity(1536);
555            out.push_str(&format!(
556                r#"<samlp:LogoutResponse xmlns:samlp="{NS_PROTOCOL}" xmlns:saml="{NS_ASSERTION}" ID="{id}" Version="2.0" IssueInstant="{}" Destination="{}" InResponseTo="{}">"#,
557                instant(now),
558                escape(slo_url),
559                escape(in_response_to),
560            ));
561            out.push_str(&format!("<saml:Issuer>{}</saml:Issuer>", escape(&issuer)));
562            if let Some(template) = template {
563                out.push_str(template);
564            }
565            out.push_str("<samlp:Status>");
566            if partial {
567                out.push_str(&format!(
568                    r#"<samlp:StatusCode Value="{STATUS_PREFIX}Success"><samlp:StatusCode Value="{STATUS_PARTIAL_LOGOUT}"/></samlp:StatusCode>"#
569                ));
570            } else {
571                out.push_str(&format!(
572                    r#"<samlp:StatusCode Value="{STATUS_PREFIX}Success"/>"#
573                ));
574            }
575            out.push_str("</samlp:Status></samlp:LogoutResponse>");
576            out
577        };
578        let delivery = deliver(
579            MessageParam::Response,
580            "LogoutResponse",
581            &id,
582            slo_url,
583            binding,
584            relay_state,
585            signing_key,
586            build,
587        )?;
588        Ok(OutboundLogout { id, delivery })
589    }
590}
591
592/// Sign and package one message for `binding`. `build` renders the document,
593/// with the signature template when given one.
594#[allow(clippy::too_many_arguments)]
595fn deliver(
596    param: MessageParam,
597    root_name: &str,
598    id: &str,
599    slo_url: &str,
600    binding: SamlBinding,
601    relay_state: Option<&str>,
602    signing_key: &SamlIdpSigningKey,
603    build: impl Fn(Option<&str>) -> String,
604) -> Result<LogoutDelivery, SamlIdpError> {
605    let cert_der = sign::certificate_der(&signing_key.credential)?;
606    let key_der = sign::private_key_der(signing_key)?;
607    match binding {
608        SamlBinding::HttpRedirect => {
609            // No `ds:Signature` template: the signature is detached, over the
610            // query octets, so the document carries none (T-373).
611            let document = build(None);
612            let mut query = format!(
613                "{}={}",
614                param.name(),
615                url_encode(&deflate_base64(document.as_bytes())?)
616            );
617            if let Some(relay_state) = relay_state {
618                query.push_str("&RelayState=");
619                query.push_str(&url_encode(relay_state));
620            }
621            query.push_str("&SigAlg=");
622            query.push_str(&url_encode(SIG_ALG_RSA_SHA256));
623            let signature = sign::sign_octets(&query, &key_der)?;
624            query.push_str("&Signature=");
625            query.push_str(&url_encode(&STANDARD.encode(signature)));
626            drop(key_der);
627
628            // What an SP will do with it, done here first, over the octets as
629            // they will be sent.
630            RedirectQuery::parse_logout(&query)
631                .and_then(|parsed| parsed.verify_signature(&cert_der))
632                .map_err(|_| {
633                    tracing::error!("SAML IdP: a Redirect logout failed its own check");
634                    SamlIdpError::SigningFailed
635                })?;
636            let separator = if slo_url.contains('?') { '&' } else { '?' };
637            Ok(LogoutDelivery::Redirect {
638                location: format!("{slo_url}{separator}{query}"),
639            })
640        }
641        SamlBinding::HttpPost => {
642            let template = sign::signature_template(id, &cert_der);
643            let signed = sign::sign(&build(Some(&template)), &key_der)?;
644            drop(key_der);
645            sign::verify_root_signed(&signed, &cert_der, root_name, id)?;
646            Ok(LogoutDelivery::Post {
647                destination: slo_url.to_owned(),
648                field: param.name(),
649                value: STANDARD.encode(signed.as_bytes()),
650                relay_state: relay_state.map(str::to_owned),
651            })
652        }
653    }
654}
