1//! SAML 2.0 identity provider: assertion issuance (G-2, T23.2.2).
2//!
3//! AXIAM as a SAML IdP turns an authenticated AXIAM session into a signed
4//! `<samlp:Response>` for a registered service provider. This module is the
5//! **issuance library** — pure, synchronous, no I/O, no routes. The SSO endpoint
6//! (T23.2.3) resolves the tenant from the request path, parses and checks the
7//! `AuthnRequest`, runs the login hop, loads the user's groups and roles and the
8//! tenant's active signing credential, and then calls
9//! [`SamlIdpIssuer::issue`]. The `SessionIndex` it writes is the **per-SP random
10//! index** the endpoint recorded in `saml_sp_session` just before (D-37, T23.2.4),
11//! never the AXIAM session id: single logout ([`logout`]) maps it back through
12//! that record.
13//!
14//! # What is issued
15//!
16//! | Element | Value |
17//! |---|---|
18//! | `Response/@Destination`, `SubjectConfirmationData/@Recipient` | the ACS URL used, which must be one the SP registered (exact match) |
19//! | `Response/@InResponseTo`, `SubjectConfirmationData/@InResponseTo` | the `AuthnRequest` id when SP-initiated; **absent** when IdP-initiated (SAML Profiles §4.1.4.2) |
20//! | `Issuer` (response and assertion) | [`idp_entity_id`] of the tenant from the request path |
21//! | `NameID` | persistent pairwise ([`pairwise_name_id`], D-22) by default; the user's email when the SP asks for `emailAddress` |
22//! | `SubjectConfirmation/@Method` | bearer |
23//! | `SubjectConfirmationData/@NotOnOrAfter`, `Conditions/@NotOnOrAfter` | now + [`ASSERTION_LIFETIME_SECS`] (5 minutes) |
24//! | `Conditions/@NotBefore` | now − [`crate::oidc::CLOCK_SKEW_LEEWAY_SECS`] (the existing skew allowance, 60 s) |
25//! | `AudienceRestriction/Audience` | the SP's entity id |
26//! | `AuthnStatement/@AuthnInstant` | the session's `authenticated_at` |
27//! | `AuthnStatement/@SessionIndex` | the per-SP random index recorded in `saml_sp_session` (D-37); **not** the AXIAM session id, so SPs that compare notes cannot correlate a person's sessions (T-312) |
28//! | `AuthnContextClassRef` | [`authn_context_class_ref`] of the session's `amr` |
29//! | `AttributeStatement` | the SP's attribute mapping over user fields, group names and role names |
30//!
31//! # Signing
32//!
33//! Enveloped XML-DSig, `rsa-sha256` over a `sha256` digest with exclusive
34//! canonicalization, the reference URI naming the signed element's `ID`, and
35//! the credential's certificate in `KeyInfo/X509Data`. The **assertion is
36//! always signed**; the response is signed as well when the SP's
37//! `sign_responses` is set, and then *after* the assertion, so its signature
38//! covers the signed assertion. Before anything is returned, every signature in
39//! the output is verified against the credential's certificate and the output's
40//! shape is checked: exactly one `Assertion`, exactly the signatures intended,
41//! each referencing the element it sits in. A builder bug that would emit a
42//! wrapped or mis-referenced document fails closed here rather than at an SP.
43//!
44//! The credential must be the tenant's, `active`, and inside its validity
45//! window (`not_before ≤ now < not_after`); anything else is refused, never
46//! signed. The private key is turned into DER inside a zeroizing buffer and is
47//! never formatted, logged or carried in an error.
48//!
49//! # Failure responses are never signed
50//!
51//! [`SamlIdpIssuer::failure`] builds the status-only envelopes the SSO endpoint
52//! needs (`Requester`, `Responder`, `NoPassive`, `AuthnFailed`,
53//! `RequestDenied`, `InvalidNameIDPolicy`), with no status message or detail.
54//! They carry no assertion, and SAML Profiles §4.1.3.5 asks for a signature only
55//! over a response that does. Signing them would hand anyone who can send an
56//! `AuthnRequest` a document signed with the tenant's key, containing a request
57//! id of their choosing and no assertion — exactly the gadget a signature
58//! wrapping attack against a lax SP needs. So the key signs one shape of
59//! document only: a response carrying exactly one assertion.
60//!
61//! # Encryption (D-2)
62//!
63//! Not implemented: `samael` decrypts assertions but has no encryption API, and
64//! no oracle in the tree verifies an `EncryptedAssertion` built with AES-256-GCM.
65//! An SP registered with `encrypt_assertions` is **refused** with `Responder`
66//! ([`SamlIdpError::EncryptionUnsupported`]) — never silently sent plaintext.

139/// The SAML status a refusal is answered with (SAML Core §3.2.2.2).
140///
141/// Each value is a fixed (top-level, second-level) pair, so the SSO endpoint
142/// can only ever say one of these and nothing more: no `StatusMessage`, no
143/// `StatusDetail`.
144#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
145pub enum SamlStatus {
146    /// The request was at fault.
147    Requester,
148    /// AXIAM could not or would not answer it.
149    Responder,
150    /// `IsPassive` was set and a passive answer is impossible.
151    NoPassive,
152    /// The principal could not be authenticated.
153    AuthnFailed,
154    /// The request is refused by policy.
155    RequestDenied,
156    /// The requested or configured `NameID` cannot be formed for this user.
157    InvalidNameIdPolicy,
158}
159
160const STATUS_PREFIX: &str = "urn:oasis:names:tc:SAML:2.0:status:";
161
162impl SamlStatus {
163    /// The top-level `StatusCode` URN.
164    #[must_use]
165    pub const fn top_level(self) -> &'static str {
166        match self {
167            Self::Requester => "urn:oasis:names:tc:SAML:2.0:status:Requester",
168            Self::Responder
169            | Self::NoPassive
170            | Self::AuthnFailed
171            | Self::RequestDenied
172            | Self::InvalidNameIdPolicy => "urn:oasis:names:tc:SAML:2.0:status:Responder",
173        }
174    }
175
176    /// The nested second-level `StatusCode` URN, if there is one.
177    #[must_use]
178    pub const fn second_level(self) -> Option<&'static str> {
179        match self {
180            Self::Requester | Self::Responder => None,
181            Self::NoPassive => Some("urn:oasis:names:tc:SAML:2.0:status:NoPassive"),
182            Self::AuthnFailed => Some("urn:oasis:names:tc:SAML:2.0:status:AuthnFailed"),
183            Self::RequestDenied => Some("urn:oasis:names:tc:SAML:2.0:status:RequestDenied"),
184            Self::InvalidNameIdPolicy => {
185                Some("urn:oasis:names:tc:SAML:2.0:status:InvalidNameIDPolicy")
186            }
187        }
188    }
189}

191/// Why an assertion was not issued.
192///
193/// Every variant is a fixed string with **no value in it** — not the user, the
194/// SP, the ACS URL or anything from the key — so it can be logged as is and
195/// never leaks through [`SamlIdpError::status`], which is all an SP is told.
196#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
197pub enum SamlIdpError {
198    /// The SP row, user, session, credential, a group or a role belongs to a
199    /// tenant other than the one of the request path.
200    #[error("a SAML issuance input belongs to another tenant")]
201    TenantMismatch,
202    /// The session is not the user's, or has expired.
203    #[error("the session does not authenticate this user")]
204    SessionMismatch,
205    /// The account may not act (`axiam_auth::service::account_may_act`).
206    #[error("the account may not sign in")]
207    AccountMayNotAct,
208    /// The SP is registered but disabled.
209    #[error("the service provider is disabled")]
210    SpDisabled,
211    /// The ACS URL is not one the SP registered. The SSO endpoint must not
212    /// deliver anything to it — not even a failure response.
213    #[error("the ACS URL is not registered for this service provider")]
214    AcsNotRegistered,
215    /// The ACS endpoint is registered with the HTTP-Redirect binding, which
216    /// SAML Profiles §4.1.2 forbids for a response.
217    #[error("the ACS endpoint does not accept the HTTP-POST binding")]
218    AcsBindingUnsupported,
219    /// The `InResponseTo` value is empty, too long or not an `NCName`.
220    #[error("the request id is not a valid SAML identifier")]
221    InvalidRequestId,
222    /// `RelayState` is longer than [`MAX_RELAY_STATE_BYTES`].
223    #[error("RelayState exceeds 80 bytes")]
224    RelayStateTooLong,
225    /// An unsolicited response for an SP that has not opted in (D-3).
226    #[error("IdP-initiated sign-on is not enabled for this service provider")]
227    IdpInitiatedNotAllowed,
228    /// The SP's `allowed_groups` does not include any of the user's groups.
229    #[error("the user is not in a group allowed to use this service provider")]
230    GroupNotAllowed,
231    /// The SP asks for an email `NameID` and the user has no email address.
232    #[error("the user has no email address for an emailAddress NameID")]
233    NameIdUnavailable,
234    /// The SP asks for an email `NameID` and nothing vouches for the user's
235    /// address (D-25): it was never verified and the account is not `Active`.
236    #[error("the user's email address is not vouched for")]
237    NameIdUnverified,
238    /// A persistent `NameID` is required and the deployment holds no
239    /// `saml_pairwise_key`.
240    #[error("the SAML pairwise-identifier key is not configured")]
241    PairwiseKeyMissing,
242    /// The SP asks for encrypted assertions, which this build cannot produce.
243    #[error("assertion encryption is not supported")]
244    EncryptionUnsupported,
245    /// The `SessionIndex` is empty, too long or not the opaque token the
246    /// participant record mints (D-37).
247    #[error("the SessionIndex is not a valid opaque token")]
248    SessionIndexInvalid,
249    /// The SP registered no single-logout endpoint to deliver a logout message to.
250    #[error("the service provider registered no single-logout endpoint")]
251    SloNotRegistered,
252    /// The tenant has no active signing credential.
253    #[error("the tenant has no active SAML signing credential")]
254    NoActiveCredential,
255    /// The credential is not `active`.
256    #[error("the SAML signing credential is not active")]
257    CredentialNotActive,
258    /// `now` is outside the credential's `not_before..not_after`.
259    #[error("the SAML signing credential is outside its validity period")]
260    CredentialNotValid,
261    /// The signing step, or the check of its output, failed.
262    #[error("the SAML response could not be signed")]
263    SigningFailed,
264}
265
266impl SamlIdpError {
267    /// The status an SP is told. Nothing else about the refusal leaves AXIAM.
268    #[must_use]
269    pub const fn status(self) -> SamlStatus {
270        match self {
271            Self::AcsNotRegistered
272            | Self::AcsBindingUnsupported
273            | Self::InvalidRequestId
274            | Self::RelayStateTooLong => SamlStatus::Requester,
275            Self::SpDisabled | Self::IdpInitiatedNotAllowed | Self::GroupNotAllowed => {
276                SamlStatus::RequestDenied
277            }
278            Self::AccountMayNotAct | Self::SessionMismatch => SamlStatus::AuthnFailed,
279            Self::NameIdUnavailable | Self::NameIdUnverified => SamlStatus::InvalidNameIdPolicy,
280            Self::TenantMismatch
281            | Self::PairwiseKeyMissing
282            | Self::EncryptionUnsupported
283            | Self::SessionIndexInvalid
284            | Self::SloNotRegistered
285            | Self::NoActiveCredential
286            | Self::CredentialNotActive
287            | Self::CredentialNotValid
288            | Self::SigningFailed => SamlStatus::Responder,
289        }
290    }
291}

297/// Whether a user in `user_groups` may use `sp`: yes when the SP names no
298/// groups, otherwise only when one of the user's groups is listed.
299///
300/// The SSO endpoint calls this once the user is known and before issuing;
301/// [`SamlIdpIssuer::issue`] calls it again, so a caller that forgets cannot
302/// issue past it.
303///
304/// # Errors
305///
306/// [`SamlIdpError::GroupNotAllowed`] (`RequestDenied`).
307pub fn check_allowed_groups(
308    sp: &SamlServiceProvider,
309    user_groups: &[Group],
310) -> Result<(), SamlIdpError> {
311    if sp.allowed_groups.is_empty()
312        || user_groups
313            .iter()
314            .any(|g| g.tenant_id == sp.tenant_id && sp.allowed_groups.contains(&g.id))
315    {
316        Ok(())
317    } else {
318        Err(SamlIdpError::GroupNotAllowed)
319    }
320}

322/// The ACS endpoint a response may be delivered to: the one `acs_url` names,
323/// by exact string equality against the SP's registration, and only with the
324/// HTTP-POST binding.
325///
326/// # Errors
327///
328/// [`SamlIdpError::AcsNotRegistered`] — **deliver nothing** to that URL, not even
329/// a failure response; [`SamlIdpError::AcsBindingUnsupported`].
330pub fn check_acs_url(sp: &SamlServiceProvider, acs_url: &str) -> Result<(), SamlIdpError> {
331    let endpoint = sp
332        .acs_by_url(acs_url)
333        .ok_or(SamlIdpError::AcsNotRegistered)?;
334    if endpoint.binding != SamlBinding::HttpPost {
335        return Err(SamlIdpError::AcsBindingUnsupported);
336    }
337    Ok(())
338}

353/// Check a `RelayState` before the login hop: it is echoed **verbatim** in the
354/// response, so one that cannot be echoed is refused up front.
355///
356/// # Errors
357///
358/// [`SamlIdpError::RelayStateTooLong`] above [`MAX_RELAY_STATE_BYTES`].
359pub fn check_relay_state(relay_state: Option<&str>) -> Result<(), SamlIdpError> {
360    match relay_state {
361        Some(value) if value.len() > MAX_RELAY_STATE_BYTES => Err(SamlIdpError::RelayStateTooLong),
362        _ => Ok(()),
363    }
364}

366/// The longest `SessionIndex` this IdP asserts or accepts back, in bytes. The
367/// ones it mints are 43 (32 bytes, base64url without padding); the bound is
368/// there so a caller cannot put an arbitrary string under the tenant's
369/// signature.
370pub const MAX_SESSION_INDEX_BYTES: usize = 256;
371
372/// Check a `SessionIndex` before it is signed: non-empty, at most
373/// [`MAX_SESSION_INDEX_BYTES`], and only the characters of the opaque token
374/// (ASCII letters and digits, `-` and `_`).
375///
376/// # Errors
377///
378/// [`SamlIdpError::SessionIndexInvalid`] (`Responder`).
379pub fn check_session_index(index: &str) -> Result<(), SamlIdpError> {
380    if !index.is_empty()
381        && index.len() <= MAX_SESSION_INDEX_BYTES
382        && index
383            .bytes()
384            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_'))
385    {
386        Ok(())
387    } else {
388        Err(SamlIdpError::SessionIndexInvalid)
389    }
390}

392/// The `AuthnContextClassRef` an authentication achieved, from its evidence
393/// and nothing else.
394///
395/// Like `axiam_oauth2::acr::acr_for`, this takes the session's `amr` and no
396/// request: a `RequestedAuthnContext` cannot move the answer, only the
397/// SSO endpoint's decision whether to re-authenticate. The table, first match
398/// wins:
399///
400/// | Evidence | Class |
401/// |---|---|
402/// | `mfa`, or `hwk`/`swk` with `user` (a passkey or security key that verified the user) | [`AUTHN_CONTEXT_MFA`] |
403/// | `x509` | [`AUTHN_CONTEXT_X509`] |
404/// | `pwd` | [`AUTHN_CONTEXT_PASSWORD_PROTECTED_TRANSPORT`] |
405/// | anything else: `fed` alone, a presence-only key, no evidence | [`AUTHN_CONTEXT_UNSPECIFIED`] |
406///
407/// The multi-factor row is `acr_for`'s multi-factor rule less `x509`, which
408/// SAML can name precisely. `fed` is unspecified because what an upstream IdP
409/// did is its claim, not AXIAM's evidence (the rule `acr_for` follows too).
410#[must_use]
411pub fn authn_context_class_ref(amr: &[Amr]) -> &'static str {
412    let has = |needle: Amr| amr.contains(&needle);
413    if has(Amr::Mfa) || ((has(Amr::Hwk) || has(Amr::Swk)) && has(Amr::User)) {
414        AUTHN_CONTEXT_MFA
415    } else if has(Amr::X509) {
416        AUTHN_CONTEXT_X509
417    } else if has(Amr::Pwd) {
418        AUTHN_CONTEXT_PASSWORD_PROTECTED_TRANSPORT
419    } else {
420        AUTHN_CONTEXT_UNSPECIFIED
421    }
422}

428/// Everything one assertion is about.
429///
430/// Built by the SSO endpoint from what it verified. `tenant_id` is the tenant
431/// **of the request path**; every other input must belong to it, and the
432/// issuer refuses otherwise (a row from one tenant must never be signed with
433/// another tenant's key, or under another tenant's `Issuer`).
434#[derive(Debug, Clone, Copy)]
435pub struct SsoIssuance<'a> {
436    /// The tenant of the request path.
437    pub tenant_id: Uuid,
438    /// The registered service provider.
439    pub sp: &'a SamlServiceProvider,
440    /// The ACS URL the response is delivered to: the `AuthnRequest`'s, resolved
441    /// against the registration, or the SP's default endpoint.
442    pub acs_url: &'a str,
443    /// The `AuthnRequest` id; `None` for IdP-initiated sign-on.
444    pub in_response_to: Option<&'a str>,
445    /// The `RelayState` to echo, verbatim.
446    pub relay_state: Option<&'a str>,
447    /// The AXIAM session that authenticated the user.
448    pub session: &'a Session,
449    /// The `SessionIndex` to assert: the per-SP random token the endpoint
450    /// recorded in `saml_sp_session` for this (session, SP) **before** calling
451    /// [`SamlIdpIssuer::issue`] (D-37). The session id itself never reaches the
452    /// XML. A caller that has not recorded one has no assertion to sign.
453    pub session_index: &'a str,
454    /// The user.
455    pub user: &'a User,
456    /// The user's groups (for `allowed_groups` and the `groups` attribute).
457    pub groups: &'a [Group],
458    /// The roles the user holds, directly or through a group (for the `roles`
459    /// attribute).
460    pub roles: &'a [Role],
461}

463/// An HTTP-POST binding message (SAML Bindings §3.5): what the SSO endpoint
464/// renders as an auto-submitting form.
465///
466/// **The renderer must HTML-escape all three values** — `acs_url` into the
467/// form's `action`, the other two into hidden `input` values — even though the
468/// response is base64 and the URL was registered: `RelayState` is the SP's
469/// string, echoed verbatim.
470#[derive(Debug, Clone, PartialEq, Eq)]
471pub struct PostBinding {
472    /// The form's `action`: the ACS URL, also the response's `Destination`.
473    pub acs_url: String,
474    /// The `SAMLResponse` field: the response XML, base64 (no line breaks).
475    pub saml_response: String,
476    /// The `RelayState` field, verbatim; omitted from the form when `None`.
477    pub relay_state: Option<String>,
478}

497/// The issuing side of the SAML IdP, one per deployment.
498///
499/// Holds the two deployment-wide inputs: the public base URL every IdP entity id
500/// is derived from, and the pairwise-identifier key. The per-tenant input, the
501/// signing credential, is passed per call, because it is loaded per tenant.
502///
503/// Signing is CPU work (an RSA-4096 private-key operation per signature, and the
504/// verification of each before returning); the SSO endpoint should call
505/// [`Self::issue`] from a blocking task.
506pub struct SamlIdpIssuer {
507    public_base_url: String,
508    pairwise_key: Option<PairwiseKey>,
509}

535    /// [`idp_entity_id`] under this deployment's base URL.

541    /// Issue the signed response for one sign-on.
542    ///
543    /// Every input check of this module runs again here, so a caller that
544    /// skipped one cannot issue past it: tenant agreement, the session, the
545    /// account status, the SP being enabled, the ACS URL and its binding, the
546    /// request id, `RelayState`, the IdP-initiated opt-in, `allowed_groups`,
547    /// the `NameID` policy, encryption, and the credential's tenant, status and
548    /// validity window.
549    ///
550    /// # Errors
551    ///
552    /// A [`SamlIdpError`]; answer the SP with its [`SamlIdpError::status`] (via
553    /// [`Self::failure`]), except for [`SamlIdpError::AcsNotRegistered`] and
554    /// [`SamlIdpError::AcsBindingUnsupported`]: with no registered POST
555    /// endpoint to deliver to, nothing is posted anywhere — render an error
556    /// page.
557    pub fn issue(
558        &self,
559        req: &SsoIssuance<'_>,
560        signing_key: &SamlIdpSigningKey,
561        now: DateTime<Utc>,
562    ) -> Result<IssuedResponse, SamlIdpError> {
563        let now = now.trunc_subsecs(0);
564        self.check(req, now)?;
565        sign::check_credential(&signing_key.credential, req.tenant_id, now)?;
566        let name_id = self.name_id(req)?;
567
568        let issuer = self.entity_id(req.tenant_id);
569        let response_id = xml::new_id();
570        let assertion_id = xml::new_id();
571        let not_on_or_after = now + Duration::seconds(ASSERTION_LIFETIME_SECS);
572        let cert_der = sign::certificate_der(&signing_key.credential)?;
573
574        let assertion = assertion_template(&AssertionParts {
575            id: &assertion_id,
576            issuer: &issuer,
577            now,
578            not_on_or_after,
579            name_id: &name_id,
580            req,
581            cert_der: &cert_der,
582        });
583        let key_der = sign::private_key_der(signing_key)?;
584        let signed_assertion = sign::sign(&assertion, &key_der)?;
585
586        let response = response_envelope(&EnvelopeParts {
587            id: &response_id,
588            issuer: &issuer,
589            now,
590            destination: req.acs_url,
591            in_response_to: req.in_response_to,
592            status: None,
593            signature_template: req
594                .sp
595                .sign_responses
596                .then(|| sign::signature_template(&response_id, &cert_der)),
597            assertion: Some(&signed_assertion),
598        });
599        let response = if req.sp.sign_responses {
600            sign::sign(&response, &key_der)?
601        } else {
602            response
603        };
604        drop(key_der);
605
606        sign::verify_output(
607            &response,
608            &cert_der,
609            &response_id,
610            &assertion_id,
611            req.sp.sign_responses,
612        )?;
613
614        Ok(IssuedResponse {
615            binding: PostBinding {
616                acs_url: req.acs_url.to_owned(),
617                saml_response: STANDARD.encode(response.as_bytes()),
618                relay_state: req.relay_state.map(str::to_owned),
619            },
620            response_id,
621            assertion_id,
622            name_id,
623            session_index: req.session_index.to_owned(),
624            not_on_or_after,
625        })
626    }
627
628    /// A status-only response: no assertion, no signature, no message (see the
629    /// module docs for why it is unsigned).
630    ///
631    /// `acs_url` must be a registered POST endpoint of `sp` — checked here
632    /// again, because a failure response to an unregistered URL would make the
633    /// SSO endpoint an open redirect with a SAML body. A request id or
634    /// `RelayState` that could not be echoed is omitted rather than refused, so
635    /// the SP still learns that its request failed.
636    ///
637    /// # Errors
638    ///
639    /// [`SamlIdpError::AcsNotRegistered`] and
640    /// [`SamlIdpError::AcsBindingUnsupported`]: render an error page instead.
641    pub fn failure(
642        &self,
643        tenant_id: Uuid,
644        sp: &SamlServiceProvider,
645        acs_url: &str,
646        in_response_to: Option<&str>,
647        relay_state: Option<&str>,
648        status: SamlStatus,
649        now: DateTime<Utc>,
650    ) -> Result<PostBinding, SamlIdpError> {
651        if sp.tenant_id != tenant_id {
652            return Err(SamlIdpError::TenantMismatch);
653        }
654        check_acs_url(sp, acs_url)?;
655        let issuer = self.entity_id(tenant_id);
656        let response = response_envelope(&EnvelopeParts {
657            id: &xml::new_id(),
658            issuer: &issuer,
659            now: now.trunc_subsecs(0),
660            destination: acs_url,
661            in_response_to: in_response_to.filter(|id| xml::is_request_id(id)),
662            status: Some(status),
663            signature_template: None,
664            assertion: None,
665        });
666        Ok(PostBinding {
667            acs_url: acs_url.to_owned(),
668            saml_response: STANDARD.encode(response.as_bytes()),
669            relay_state: relay_state
670                .filter(|r| r.len() <= MAX_RELAY_STATE_BYTES)
671                .map(str::to_owned),
672        })
673    }
674
675    /// Every refusal that does not need the key.
676    fn check(&self, req: &SsoIssuance<'_>, now: DateTime<Utc>) -> Result<(), SamlIdpError> {
677        let tenant = req.tenant_id;
678        if req.sp.tenant_id != tenant
679            || req.user.tenant_id != tenant
680            || req.session.tenant_id != tenant
681            || req.groups.iter().any(|g| g.tenant_id != tenant)
682            || req.roles.iter().any(|r| r.tenant_id != tenant)
683        {
684            return Err(SamlIdpError::TenantMismatch);
685        }
686        if req.session.user_id != req.user.id || req.session.expires_at <= now {
687            return Err(SamlIdpError::SessionMismatch);
688        }
689        axiam_auth::service::account_may_act(req.user)
690            .map_err(|_| SamlIdpError::AccountMayNotAct)?;
691        if !req.sp.enabled {
692            return Err(SamlIdpError::SpDisabled);
693        }
694        check_acs_url(req.sp, req.acs_url)?;
695        match req.in_response_to {
696            Some(id) => check_request_id(id)?,
697            None if !req.sp.allow_idp_initiated => {
698                return Err(SamlIdpError::IdpInitiatedNotAllowed);
699            }
700            None => {}
701        }
702        check_relay_state(req.relay_state)?;
703        check_session_index(req.session_index)?;
704        check_allowed_groups(req.sp, req.groups)?;
705        if req.sp.encrypt_assertions {
706            return Err(SamlIdpError::EncryptionUnsupported);
707        }
708        Ok(())
709    }

711    /// The `NameID` the SP is given for this user, and its format, under the SP's
712    /// policy — the value [`Self::issue`] will assert.
713    ///
714    /// The SSO endpoint asks it **before** signing, because the participant
715    /// record (D-37) must hold the `NameID` the SP is about to be given, and
716    /// compares it with [`IssuedResponse::name_id`] afterwards. Same function,
717    /// same inputs: they agree unless the user's address changed between the two
718    /// calls, which the comparison catches.
719    ///
720    /// # Errors
721    ///
722    /// [`SamlIdpError::TenantMismatch`], [`SamlIdpError::PairwiseKeyMissing`],
723    /// [`SamlIdpError::NameIdUnavailable`], [`SamlIdpError::NameIdUnverified`].
724    pub fn name_id_for(
725        &self,
726        tenant_id: Uuid,
727        sp: &SamlServiceProvider,
728        user: &User,
729    ) -> Result<(String, NameIdFormat), SamlIdpError> {
730        if sp.tenant_id != tenant_id || user.tenant_id != tenant_id {
731            return Err(SamlIdpError::TenantMismatch);
732        }
733        let value = match sp.name_id_format {
734            NameIdFormat::Persistent => {
735                let pairwise_key = self
736                    .pairwise_key
737                    .as_ref()
738                    .ok_or(SamlIdpError::PairwiseKeyMissing)?;
739                pairwise_name_id(pairwise_key, tenant_id, &sp.entity_id, user.id)
740            }
741            NameIdFormat::EmailAddress => {
742                let email = user_email(user).ok_or(SamlIdpError::NameIdUnavailable)?;
743                if !email_is_vouched_for(user) {
744                    return Err(SamlIdpError::NameIdUnverified);
745                }
746                email.to_owned()
747            }
748        };
749        Ok((value, sp.name_id_format))
750    }
751
752    /// The `NameID` value under the SP's policy.
753    fn name_id(&self, req: &SsoIssuance<'_>) -> Result<String, SamlIdpError> {
754        self.name_id_for(req.tenant_id, req.sp, req.user)
755            .map(|(value, _)| value)
756    }
757}

759/// The user's email address, or `None` when the row holds none.
760fn user_email(user: &User) -> Option<&str> {
761    let email = user.email.trim();
762    (!email.is_empty()).then_some(email)
763}

765/// Whether something vouches for the user's address (D-25, T-313): it was
766/// verified (`email_verified_at`), or the account is `Active` — which only the
767/// verification flow, an administrator, SCIM or the directory path make it, and
768/// each of those either proved the address or wrote it.
769///
770/// What this refuses is the account T-313 is about: one still
771/// `PendingVerification` with an address nobody checked — a self-registration
772/// inside its grace period, or an account provisioned pending and never
773/// activated. Such an account keeps working at every SP whose `NameID` is the
774/// pairwise identifier; at an email-keyed SP it is answered
775/// `InvalidNameIDPolicy`, never with a weaker identifier. The `email` attribute
776/// is held to the same rule (omitted, not refused), since an SP may key
777/// accounts on it just as well.
778fn email_is_vouched_for(user: &User) -> bool {
779    user.email_verified_at.is_some() || user.status == axiam_core::models::user::UserStatus::Active
780}

782/// The values of one attribute source for this user, in a stable order, empty
783/// values dropped. An attribute with no values is not emitted at all.
784fn attribute_values(source: AttributeSource, req: &SsoIssuance<'_>) -> Vec<String> {
785    let profile = || ProfileClaims::from_metadata(&req.user.metadata);
786    let single = |value: Option<String>| {
787        value
788            .filter(|v| !v.trim().is_empty())
789            .into_iter()
790            .collect::<Vec<_>>()
791    };
792    match source {
793        AttributeSource::Username => single(Some(req.user.username.clone())),
794        AttributeSource::Email => single(
795            user_email(req.user)
796                .filter(|_| email_is_vouched_for(req.user))
797                .map(str::to_owned),
798        ),
799        AttributeSource::DisplayName => single(profile().name),
800        AttributeSource::GivenName => single(profile().given_name),
801        AttributeSource::FamilyName => single(profile().family_name),
802        AttributeSource::Groups => names(req.groups.iter().map(|g| g.name.as_str())),
803        AttributeSource::Roles => names(req.roles.iter().map(|r| r.name.as_str())),
804    }
805}

807/// Sorted, de-duplicated, non-empty names.
808fn names<'n>(iter: impl Iterator<Item = &'n str>) -> Vec<String> {
809    iter.filter(|n| !n.trim().is_empty())
810        .collect::<BTreeSet<_>>()
811        .into_iter()
812        .map(str::to_owned)
813        .collect()
814}

820struct AssertionParts<'a> {
821    id: &'a str,
822    issuer: &'a str,
823    now: DateTime<Utc>,
824    not_on_or_after: DateTime<Utc>,
825    name_id: &'a str,
826    req: &'a SsoIssuance<'a>,
827    cert_der: &'a [u8],
828}
829
830/// The unsigned assertion, with its signature template after `Issuer` (the
831/// position SAML Core §2.3.3's schema gives `ds:Signature`).
832fn assertion_template(p: &AssertionParts<'_>) -> String {
833    use xml::{escape, instant};
834
835    let req = p.req;
836    let sp = req.sp;
837    let not_before = p.now - Duration::seconds(crate::oidc::CLOCK_SKEW_LEEWAY_SECS as i64);
838
839    let mut out = String::with_capacity(4096);
840    out.push_str(&format!(
841        r#"<saml:Assertion xmlns:saml="{NS_ASSERTION}" ID="{}" Version="2.0" IssueInstant="{}">"#,
842        p.id,
843        instant(p.now)
844    ));
845    out.push_str(&format!("<saml:Issuer>{}</saml:Issuer>", escape(p.issuer)));
846    out.push_str(&sign::signature_template(p.id, p.cert_der));
847
848    // --- Subject ---
849    out.push_str("<saml:Subject>");
850    match sp.name_id_format {
851        NameIdFormat::Persistent => out.push_str(&format!(
852            r#"<saml:NameID Format="{}" NameQualifier="{}" SPNameQualifier="{}">{}</saml:NameID>"#,
853            NameIdFormat::Persistent.urn(),
854            escape(p.issuer),
855            escape(&sp.entity_id),
856            escape(p.name_id)
857        )),
858        NameIdFormat::EmailAddress => out.push_str(&format!(
859            r#"<saml:NameID Format="{}">{}</saml:NameID>"#,
860            NameIdFormat::EmailAddress.urn(),
861            escape(p.name_id)
862        )),
863    }
864    out.push_str(&format!(
865        r#"<saml:SubjectConfirmation Method="{CM_BEARER}">"#
866    ));
867    out.push_str(&format!(
868        r#"<saml:SubjectConfirmationData Recipient="{}" NotOnOrAfter="{}""#,
869        escape(req.acs_url),
870        instant(p.not_on_or_after)
871    ));
872    if let Some(request_id) = req.in_response_to {
873        out.push_str(&format!(r#" InResponseTo="{}""#, escape(request_id)));
874    }
875    out.push_str("/></saml:SubjectConfirmation></saml:Subject>");
876
877    // --- Conditions ---
878    out.push_str(&format!(
879        r#"<saml:Conditions NotBefore="{}" NotOnOrAfter="{}"><saml:AudienceRestriction><saml:Audience>{}</saml:Audience></saml:AudienceRestriction></saml:Conditions>"#,
880        instant(not_before),
881        instant(p.not_on_or_after),
882        escape(&sp.entity_id)
883    ));
884
885    // --- AuthnStatement ---
886    out.push_str(&format!(
887        r#"<saml:AuthnStatement AuthnInstant="{}" SessionIndex="{}"><saml:AuthnContext><saml:AuthnContextClassRef>{}</saml:AuthnContextClassRef></saml:AuthnContext></saml:AuthnStatement>"#,
888        instant(req.session.authenticated_at),
889        escape(req.session_index),
890        authn_context_class_ref(&req.session.amr)
891    ));
892
893    // --- AttributeStatement ---
894    let attributes: Vec<String> = sp
895        .attribute_mappings
896        .iter()
897        .filter_map(|mapping| {
898            let values = attribute_values(mapping.source, req);
899            if values.is_empty() {
900                return None;
901            }
902            let mut attr = format!(r#"<saml:Attribute Name="{}""#, escape(&mapping.saml_name));
903            if let Some(format) = &mapping.name_format {
904                attr.push_str(&format!(r#" NameFormat="{}""#, escape(format)));
905            }
906            attr.push('>');
907            for value in values {
908                attr.push_str(&format!(
909                    "<saml:AttributeValue>{}</saml:AttributeValue>",
910                    escape(&value)
911                ));
912            }
913            attr.push_str("</saml:Attribute>");
914            Some(attr)
915        })
916        .collect();
917    if !attributes.is_empty() {
918        out.push_str("<saml:AttributeStatement>");
919        for attr in attributes {
920            out.push_str(&attr);
921        }
922        out.push_str("</saml:AttributeStatement>");
923    }
924
925    out.push_str("</saml:Assertion>");
926    out
927}

929struct EnvelopeParts<'a> {
930    id: &'a str,
931    issuer: &'a str,
932    now: DateTime<Utc>,
933    destination: &'a str,
934    in_response_to: Option<&'a str>,
935    /// `None` is `Success`.
936    status: Option<SamlStatus>,
937    signature_template: Option<String>,
938    assertion: Option<&'a str>,
939}
940
941/// The `samlp:Response` around an (already signed) assertion, or around a
942/// status alone.
943fn response_envelope(p: &EnvelopeParts<'_>) -> String {
944    use xml::{escape, instant};
945
946    let mut out = String::with_capacity(8192);
947    out.push_str(&format!(
948        r#"<samlp:Response xmlns:samlp="{NS_PROTOCOL}" xmlns:saml="{NS_ASSERTION}" ID="{}" Version="2.0" IssueInstant="{}" Destination="{}""#,
949        p.id,
950        instant(p.now),
951        escape(p.destination)
952    ));
953    if let Some(request_id) = p.in_response_to {
954        out.push_str(&format!(r#" InResponseTo="{}""#, escape(request_id)));
955    }
956    out.push('>');
957    out.push_str(&format!("<saml:Issuer>{}</saml:Issuer>", escape(p.issuer)));
958    if let Some(template) = &p.signature_template {
959        out.push_str(template);
960    }
961    out.push_str("<samlp:Status>");
962    match p.status {
963        None => out.push_str(&format!(
964            r#"<samlp:StatusCode Value="{STATUS_PREFIX}Success"/>"#
965        )),
966        Some(status) => match status.second_level() {
967            None => out.push_str(&format!(
968                r#"<samlp:StatusCode Value="{}"/>"#,
969                status.top_level()
970            )),
971            Some(second) => out.push_str(&format!(
972                r#"<samlp:StatusCode Value="{}"><samlp:StatusCode Value="{second}"/></samlp:StatusCode>"#,
973                status.top_level()
974            )),
975        },
976    }
977    out.push_str("</samlp:Status>");
978    if let Some(assertion) = p.assertion {
979        out.push_str(assertion);
980    }
981    out.push_str("</samlp:Response>");
982    out
983}

1018    /// An RSA key of `bits` and a self-signed certificate for it, valid from an
1019    /// hour ago for `days`.
1020    ///
1021    /// # Panics
1022    ///
1023    /// When OpenSSL fails, which a test should surface.
1024    #[must_use]
1025    pub fn rsa_material(bits: u32, common_name: &str, days: u32) -> Material {
1026        use openssl::{asn1, bn, hash, nid, pkey, rsa, x509};
1027        let pair = pkey::PKey::from_rsa(rsa::Rsa::generate(bits).expect("rsa")).expect("pkey");
1028        let mut name = x509::X509NameBuilder::new().expect("name");
1029        name.append_entry_by_nid(nid::Nid::COMMONNAME, common_name)
1030            .expect("cn");
1031        let name = name.build();
1032        let mut builder = x509::X509Builder::new().expect("builder");
1033        builder.set_version(2).expect("version");
1034        let mut serial = bn::BigNum::new().expect("bn");
1035        serial
1036            .rand(127, bn::MsbOption::MAYBE_ZERO, false)
1037            .expect("serial");
1038        builder
1039            .set_serial_number(&serial.to_asn1_integer().expect("serial"))
1040            .expect("serial");
1041        builder.set_subject_name(&name).expect("subject");
1042        builder.set_issuer_name(&name).expect("issuer");
1043        builder.set_pubkey(&pair).expect("pubkey");
1044        let not_before =
1045            asn1::Asn1Time::from_unix(chrono::Utc::now().timestamp() - 3600).expect("not before");
1046        builder.set_not_before(&not_before).expect("not before");
1047        builder
1048            .set_not_after(&asn1::Asn1Time::days_from_now(days).expect("not after"))
1049            .expect("not after");
1050        builder
1051            .sign(&pair, hash::MessageDigest::sha256())
1052            .expect("sign");
1053        let cert = builder.build();
1054        Material {
1055            pkcs8_der: pair.private_key_to_pkcs8().expect("pkcs8 der"),
1056            pkcs8_pem: String::from_utf8(pair.private_key_to_pem_pkcs8().expect("pkcs8 pem"))
1057                .expect("utf8"),
1058            cert_der: cert.to_der().expect("der"),
1059            cert_pem: String::from_utf8(cert.to_pem().expect("pem")).expect("utf8"),
1060        }
1061    }

1083    /// library this crate's own SP side uses to consume one (`fetch_idp_metadata`)
1084    /// — and read out what an SP would.
1085    ///

1088    /// The parser's text, for a document it cannot read as one

1090    pub fn parse_idp_metadata(xml: &str) -> Result<IdpMetadataSummary, String> {
1091        let parsed: samael::metadata::EntityDescriptorType =
1092            xml.parse().map_err(|e| format!("{e}"))?;

1095            .next()

1100            .and_then(|d| d.first())
1101            .ok_or_else(|| "no IDPSSODescriptor".to_string())?;
1102        Ok(IdpMetadataSummary {
1103            entity_id: entity.entity_id.clone().ok_or("no entityID")?,
1104            keys: idp
1105                .key_descriptors
1106                .iter()
1107                .map(|k| {
1108                    (
1109                        k.key_use.clone(),
1110                        k.key_info
1111                            .x509_data
1112                            .as_ref()
1113                            .and_then(|d| d.certificates.first().cloned())
1114                            .unwrap_or_default(),
1115                    )
1116                })
1117                .collect(),
1118            single_sign_on: idp
1119                .single_sign_on_services
1120                .iter()
1121                .map(|e| (e.binding.clone(), e.location.clone()))
1122                .collect(),
1123            name_id_formats: idp.name_id_formats.clone(),
1124            single_logout: idp
1125                .single_logout_services
1126                .iter()
1127                .map(|e| (e.binding.clone(), e.location.clone()))
1128                .collect(),
1129            has_validity_or_signature: entity.valid_until.is_some()
1130                || entity.cache_duration.is_some()
1131                || entity.signature.is_some()
1132                || idp.valid_until.is_some()
1133                || idp.cache_duration.is_some()
1134                || idp.signature.is_some(),
1135        })
1136    }

1156    /// RSA-SHA256 over `octets`, base64: an HTTP-Redirect `Signature`.
1157    ///
1158    /// # Panics
1159    ///
1160    /// When OpenSSL fails.
1161    #[must_use]
1162    pub fn sign_octets(octets: &str, pkcs8_der: &[u8]) -> String {
1163        let pair = openssl::pkey::PKey::private_key_from_pkcs8(pkcs8_der).expect("key");
1164        let mut signer = openssl::sign::Signer::new(openssl::hash::MessageDigest::sha256(), &pair)
1165            .expect("signer");
1166        signer.update(octets.as_bytes()).expect("update");
1167        STANDARD.encode(signer.sign_to_vec().expect("sign"))
1168    }
