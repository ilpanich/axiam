1//! The persistent, pairwise `NameID` (D-22).
2//!
3//! A persistent `NameID` is what a service provider keys its account on, for as
4//! long as the account lives. Pairwise means each SP sees a different value for
5//! the same person, so two SPs comparing notes cannot tell they share a user.
6//! The value is
7//!
8//! ```text
9//! hex( HMAC-SHA256( saml_pairwise_key,
10//!        "axiam/saml-pairwise/v1" ‖ 0x00
11//!        ‖ tenant_id (16 bytes) ‖ user_id (16 bytes)
12//!        ‖ len(sp_entity_id) as u32 big-endian ‖ sp_entity_id ) )
13//! ```
14//!
15//! — 64 lower-case hex characters, within SAML Core §8.3.7's 256-character
16//! bound. What each property rests on:
17//!
18//! * **Stable** per (tenant, SP entity id, user): a pure function of the three
19//!   and a key that does not rotate. Deleting and re-registering an SP under the
20//!   same entity id gives its users their old identifiers back.
21//! * **Unlinkable** across SPs and tenants: different inputs to a PRF under a
22//!   key no SP holds.
23//! * **Not reversible** to the user id: HMAC is one-way; recovering the user
24//!   would need the key and a search over the tenant's users.
25//! * **Independent of the signing credential** (T23.2.1, D-21): the key is a
26//!   separate deployment secret, so rotating, retiring or re-issuing the
27//!   tenant's SAML signing certificate changes no identifier.
28//!
29//! The input encoding is injective — fixed-width ids, a length-prefixed entity
30//! id and a versioned label — so no two (tenant, user, entity id) triples can
31//! collide by concatenation, and a later derivation under the same key with a
32//! different label cannot produce these values.
33
34use hmac::{Hmac, KeyInit, Mac};
35use sha2::Sha256;
36use uuid::Uuid;
37use zeroize::Zeroizing;
38
39type HmacSha256 = Hmac<Sha256>;
40
41/// Domain-separation label, versioned so that a second scheme can coexist.
42const LABEL: &[u8] = b"axiam/saml-pairwise/v1";
43
44/// The deployment's pairwise-identifier key (`saml_pairwise_key`, D-22).
45///
46/// Held in a zeroizing buffer; `Debug` prints no key material.
47pub struct PairwiseKey(Zeroizing<[u8; 32]>);
48
49impl PairwiseKey {
50    /// Wrap the 256-bit key the secret provider returned for
51    /// [`axiam_core::secrets::SAML_PAIRWISE_KEY`].
52    #[must_use]
53    pub fn new(bytes: [u8; 32]) -> Self {
54        Self(Zeroizing::new(bytes))
55    }
56}
57
58impl std::fmt::Debug for PairwiseKey {
59    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
60        f.write_str("PairwiseKey([REDACTED])")
61    }
62}
63
64/// The persistent pairwise `NameID` of `user_id` at the SP `sp_entity_id` of
65/// tenant `tenant_id`. See the module docs for the construction.
66#[must_use]
67pub fn pairwise_name_id(
68    pairwise_key: &PairwiseKey,
69    tenant_id: Uuid,
70    sp_entity_id: &str,
71    user_id: Uuid,
72) -> String {
73    let mut mac = <HmacSha256 as KeyInit>::new_from_slice(pairwise_key.0.as_slice())
74        .expect("HMAC accepts a 32-byte key");
75    mac.update(LABEL);
76    mac.update(&[0]);
77    mac.update(tenant_id.as_bytes());
78    mac.update(user_id.as_bytes());
79    // An entity id is bounded at 1 KiB by the registry validator, so the
80    // length always fits; saturating keeps the function total regardless.
81    let len = u32::try_from(sp_entity_id.len()).unwrap_or(u32::MAX);
82    mac.update(&len.to_be_bytes());
83    mac.update(sp_entity_id.as_bytes());
84    hex::encode(mac.finalize().into_bytes())
85}
