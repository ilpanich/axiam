1//! Receiving an `AuthnRequest` (T23.2.3): decoding, parsing and signature
2//! verification for the SSO endpoint, on both bindings.
3//!
4//! Pure and synchronous, like the rest of this module: no I/O, no lookups. The
5//! SSO endpoint calls these in order — decode, parse, find the SP by the parsed
6//! `Issuer`, verify — and every refusal here happens **before** any service
7//! provider is looked up, any row is written or any person is sent to a
8//! sign-in page.
9//!
10//! # What is refused before XML parsing
11//!
12//! * **Size.** The encoded value is bounded ([`MAX_ENCODED_REQUEST_BYTES`]),
13//!   and so is the document it decodes to ([`MAX_REQUEST_XML_BYTES`]). On the
14//!   HTTP-Redirect binding the document is raw DEFLATE, and inflating stops one
15//!   byte past the bound: a decompression bomb costs at most that much memory,
16//!   however small its compressed form.
17//! * **Any markup declaration.** An `AuthnRequest` has no use for a DTD, so a
18//!   document containing `<!` other than a comment or a CDATA section —
19//!   `<!DOCTYPE`, `<!ENTITY`, `<!ELEMENT`, `<!ATTLIST` — is refused on its
20//!   bytes, before libxml sees it. That is the whole defence against external
21//!   entities and entity expansion ("billion laughs"): there is no entity to
22//!   expand. libxml is then run without recovery and without network access, so
23//!   a malformed document is refused rather than repaired into something the
24//!   signature check and the field reader might read differently.
25//!
26//! # Signatures
27//!
28//! * **HTTP-POST** — an enveloped `ds:Signature`, and only one place may hold it:
29//!   the child of the `AuthnRequest` root, with exactly one reference, naming the
30//!   root's `ID`. Any other `ds:Signature` anywhere refuses the document (the
31//!   placement rule D-23 gave the SP side). The one admitted signature is
32//!   verified by xmlsec against the SP's registered certificate, on its own
33//!   node, with SHA-1 algorithms refused.
34//! * **HTTP-Redirect** — the `Signature` query parameter over the octets SAML
35//!   Bindings §3.4.4.1 names, **exactly as received**:
36//!   `SAMLRequest=…[&RelayState=…]&SigAlg=…`, each value still percent-encoded
37//!   the way the SP encoded it. Re-encoding the decoded values (what
38//!   `samael`'s `UrlVerifier` does) would verify a string the SP never signed.
39//!   Each of the four parameters may appear once; a repeated one is refused,
40//!   since two readers could pick different copies. RSA with SHA-256, -384 or
41//!   -512 only.
42//!
43//! An enveloped signature inside a Redirect-binding document is not a signature
44//! of that binding (§3.4.4.1 says it must be removed); it is refused rather than
45//! ignored, so no document carries a signature nobody checked.
46//!
47//! # Reused by single logout (T23.2.4)
48//!
49//! [`super::logout`] receives `LogoutRequest`s and `LogoutResponse`s with the same
50//! decoding, the same refusals before libxml, the same placement rule —
51//! [`signature_placement`] is a function over **any root element** — and the same
52//! two signature verifiers. [`RedirectQuery`] carries either `SAMLRequest` or
53//! `SAMLResponse` ([`MessageParam`]) and signs over whichever it was, exactly as
54//! received. `verify_signed_xml`, which checks only the first signature, is never
55//! called (D-23, D-38).

105/// Why an `AuthnRequest` was refused before any lookup.
106///
107/// Fixed strings with no value in them, so a refusal can be logged as is. None
108/// of these is delivered to an SP: the request is not yet known to come from
109/// one, so the endpoint answers with an error page that posts nowhere.
110#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
111pub enum RequestError {
112    /// The message is larger than this endpoint reads.
113    #[error("the SAML request is too large")]
114    TooLarge,
115    /// No `SAMLRequest`, or it is not base64.
116    #[error("the SAML request is missing or not base64")]
117    Encoding,
118    /// The Redirect binding's DEFLATE stream is not valid.
119    #[error("the SAML request does not inflate")]
120    Inflate,
121    /// The document contains a markup declaration (a DTD, an entity).
122    #[error("the SAML request declares a DTD or an entity")]
123    Dtd,
124    /// The document is not well-formed UTF-8 XML.
125    #[error("the SAML request is not well-formed XML")]
126    Malformed,
127    /// The root is not a SAML 2.0 `samlp:AuthnRequest`.
128    #[error("the message is not a SAML 2.0 AuthnRequest")]
129    NotAnAuthnRequest,
130    /// `ID` is missing or not a request id this IdP will echo.
131    #[error("the AuthnRequest ID is missing or invalid")]
132    InvalidId,
133    /// An attribute or child element is missing, repeated or malformed.
134    #[error("the AuthnRequest is malformed")]
135    InvalidField,
136    /// `IssueInstant` is outside the accepted window.
137    #[error("the AuthnRequest IssueInstant is outside the accepted window")]
138    Stale,
139    /// The request names a `Subject`, which this IdP does not honour.
140    #[error("an AuthnRequest naming a Subject is not supported")]
141    SubjectUnsupported,
142    /// A query parameter of the Redirect binding appears more than once.
143    #[error("a SAML query parameter is repeated")]
144    DuplicateParameter,
145    /// A signature sits somewhere a signature may not.
146    #[error("a signature is placed where none may be")]
147    SignaturePlacement,
148    /// The SP requires signed requests and this one is not signed.
149    #[error("the AuthnRequest is not signed")]
150    SignatureMissing,
151    /// The signature algorithm is not one this IdP accepts.
152    #[error("the signature algorithm is not accepted")]
153    SignatureAlgorithm,
154    /// The root is not a SAML 2.0 `samlp:LogoutRequest` or `samlp:LogoutResponse`.
155    #[error("the message is not a SAML 2.0 LogoutRequest or LogoutResponse")]
156    NotALogoutMessage,
157    /// A `LogoutRequest` that names no `NameID`, or more than one.
158    #[error("the LogoutRequest does not name exactly one NameID")]
159    NameIdMissing,
160    /// A `LogoutRequest` naming its principal by `BaseID` or `EncryptedID`,
161    /// neither of which this IdP issues.
162    #[error("a BaseID or EncryptedID is not supported")]
163    NameIdUnsupported,
164    /// More `SessionIndex` elements than one request may carry.
165    #[error("the LogoutRequest carries too many SessionIndex elements")]
166    TooManySessionIndexes,
167    /// The request's `NotOnOrAfter` has passed.
168    #[error("the LogoutRequest has expired")]
169    Expired,
170    /// The message carries no `Destination`.
171    #[error("the message has no Destination")]
172    DestinationMissing,
173    /// The signature does not verify against the SP's certificate.
174    #[error("the AuthnRequest signature does not verify")]
175    SignatureInvalid,
176}

205/// Decode an HTTP-Redirect `SAMLRequest` value (already URL-decoded): base64,
206/// then raw DEFLATE inflated to at most [`MAX_REQUEST_XML_BYTES`].
207///
208/// # Errors
209///
210/// [`RequestError::TooLarge`], [`RequestError::Encoding`],
211/// [`RequestError::Inflate`], [`RequestError::Malformed`] (not UTF-8).
212pub fn decode_redirect(value: &str) -> Result<String, RequestError> {
213    let compressed = decode_base64(value, false)?;
214    let mut inflated = Vec::with_capacity(4096);
215    flate2::read::DeflateDecoder::new(compressed.as_slice())
216        .take(MAX_REQUEST_XML_BYTES as u64 + 1)
217        .read_to_end(&mut inflated)
218        .map_err(|_| RequestError::Inflate)?;
219    if inflated.len() > MAX_REQUEST_XML_BYTES {
220        return Err(RequestError::TooLarge);
221    }
222    String::from_utf8(inflated).map_err(|_| RequestError::Malformed)
223}

225/// Decode an HTTP-POST `SAMLRequest` value: base64 of the document, line breaks
226/// tolerated (some SPs wrap at 76 columns).
227///
228/// # Errors
229///
230/// [`RequestError::TooLarge`], [`RequestError::Encoding`],
231/// [`RequestError::Malformed`] (not UTF-8).
232pub fn decode_post(value: &str) -> Result<String, RequestError> {
233    let bytes = decode_base64(value, true)?;
234    if bytes.len() > MAX_REQUEST_XML_BYTES {
235        return Err(RequestError::TooLarge);
236    }
237    String::from_utf8(bytes).map_err(|_| RequestError::Malformed)
238}

240fn decode_base64(value: &str, allow_line_breaks: bool) -> Result<Vec<u8>, RequestError> {
241    if value.len() > MAX_ENCODED_REQUEST_BYTES {
242        return Err(RequestError::TooLarge);
243    }
244    let compact: String = if allow_line_breaks {
245        value
246            .chars()
247            .filter(|c| !matches!(c, '\r' | '\n' | ' ' | '\t'))
248            .collect()
249    } else {
250        value.to_owned()
251    };
252    if compact.is_empty() {
253        return Err(RequestError::Encoding);
254    }
255    STANDARD
256        .decode(compact.as_bytes())
257        .map_err(|_| RequestError::Encoding)
258}

260/// Refuse a document that contains any markup declaration.
261///
262/// `<!--` (a comment) and `<![CDATA[` are the only two constructs beginning
263/// `<!` that may appear in a document without a DTD; anything else beginning
264/// `<!` — `<!DOCTYPE`, `<!ENTITY`, `<!ELEMENT`, `<!ATTLIST`, `<!NOTATION`, in
265/// any case — is a declaration, and an `AuthnRequest` needs none.
266///
267/// # Errors
268///
269/// [`RequestError::Dtd`].
270pub fn refuse_markup_declarations(document: &str) -> Result<(), RequestError> {
271    let mut rest = document;
272    while let Some(at) = rest.find("<!") {
273        let tail = &rest[at..];
274        if !(tail.starts_with("<!--") || tail.starts_with("<![CDATA[")) {
275            return Err(RequestError::Dtd);
276        }
277        rest = &rest[at + 2..];
278    }
279    Ok(())
280}
281
282/// Refuse a document that is not plainly UTF-8: one containing a NUL (which is
283/// how UTF-16 and UTF-32 spell ASCII, and would hide a `<!DOCTYPE` from
284/// [`refuse_markup_declarations`]), or one whose XML declaration names another
285/// encoding. The parse here forces UTF-8, and xmlsec's honours the declaration;
286/// refusing every other declaration keeps the two reading the same characters.
287///
288/// # Errors
289///
290/// [`RequestError::Malformed`].
291pub fn refuse_other_encodings(document: &str) -> Result<(), RequestError> {
292    if document.contains('\0') {
293        return Err(RequestError::Malformed);
294    }
295    let trimmed = document.trim_start();
296    if let Some(rest) = trimmed.strip_prefix("<?xml") {
297        let declaration = rest.split("?>").next().unwrap_or_default();
298        if let Some(at) = declaration.find("encoding") {
299            let value = declaration[at + "encoding".len()..]
300                .trim_start()
301                .trim_start_matches('=')
302                .trim_start()
303                .trim_start_matches(['"', '\''])
304                .split(['"', '\''])
305                .next()
306                .unwrap_or_default();
307            if !value.eq_ignore_ascii_case("utf-8") {
308                return Err(RequestError::Malformed);
309            }
310        }
311    }
312    Ok(())
313}

315/// Parse an `AuthnRequest` and check what needs no SP: shape, `Version`, `ID`,
316/// `IssueInstant` against `now`, `Issuer`, and where a signature sits.
317///
318/// # Errors
319///
320/// A [`RequestError`]; nothing about the document is in it.
321pub fn parse_authn_request(
322    document: &str,
323    now: DateTime<Utc>,
324) -> Result<ParsedAuthnRequest, RequestError> {
325    let doc = parse_xml(document)?;
326    let root = doc.get_root_element().ok_or(RequestError::Malformed)?;
327    if !is_element(&root, NS_PROTOCOL, "AuthnRequest") {
328        return Err(RequestError::NotAnAuthnRequest);
329    }
330    if root.get_attribute("Version").as_deref() != Some("2.0") {
331        return Err(RequestError::NotAnAuthnRequest);
332    }
333
334    let id = root.get_attribute("ID").ok_or(RequestError::InvalidId)?;
335    super::check_request_id(&id).map_err(|_| RequestError::InvalidId)?;
336
337    let issue_instant = fresh_issue_instant(&root, now)?;
338
339    let children = root.get_child_elements();
340    let issuer = read_issuer(&children)?;
341    if children
342        .iter()
343        .any(|c| is_element(c, NS_ASSERTION, "Subject"))
344    {
345        return Err(RequestError::SubjectUnsupported);
346    }
347    let policies: Vec<_> = children
348        .iter()
349        .filter(|c| is_element(c, NS_PROTOCOL, "NameIDPolicy"))
350        .collect();
351    let name_id_format = match policies.as_slice() {
352        [] => None,
353        [policy] => policy.get_attribute("Format"),
354        _ => return Err(RequestError::InvalidField),
355    };
356
357    let enveloped_signature = signature_placement(&doc, &root, &id)?;
358
359    Ok(ParsedAuthnRequest {
360        id,
361        issue_instant,
362        destination: root.get_attribute("Destination"),
363        issuer,
364        force_authn: xs_boolean(root.get_attribute("ForceAuthn"))?,
365        is_passive: xs_boolean(root.get_attribute("IsPassive"))?,
366        acs_url: root.get_attribute("AssertionConsumerServiceURL"),
367        acs_index: root
368            .get_attribute("AssertionConsumerServiceIndex")
369            .map(|raw| raw.trim().parse::<u16>())
370            .transpose()
371            .map_err(|_| RequestError::InvalidField)?,
372        protocol_binding: root.get_attribute("ProtocolBinding"),
373        name_id_format,
374        enveloped_signature,
375    })
376}

378/// Parse a document the way every message this IdP receives is parsed: size
379/// bound, no markup declaration, no other encoding — all on the bytes, before
380/// libxml — then libxml without recovery and without network access.
381pub(super) fn parse_xml(document: &str) -> Result<libxml::tree::Document, RequestError> {
382    if document.len() > MAX_REQUEST_XML_BYTES {
383        return Err(RequestError::TooLarge);
384    }
385    refuse_markup_declarations(document)?;
386    refuse_other_encodings(document)?;
387
388    let options = libxml::parser::ParserOptions {
389        recover: false,
390        no_net: true,
391        no_def_dtd: true,
392        ignore_enc: true,
393        encoding: Some("UTF-8"),
394        ..Default::default()
395    };
396    libxml::parser::Parser::default()
397        .parse_string_with_options(document.as_bytes(), options)
398        .map_err(|_| RequestError::Malformed)
399}

401/// `IssueInstant`, required, and inside the accepted window: at most
402/// [`REQUEST_MAX_AGE_SECS`] old and no further ahead than the clock-skew
403/// allowance.
404pub(super) fn fresh_issue_instant(
405    root: &libxml::tree::Node,
406    now: DateTime<Utc>,
407) -> Result<DateTime<Utc>, RequestError> {
408    let issue_instant = root
409        .get_attribute("IssueInstant")
410        .and_then(|raw| DateTime::parse_from_rfc3339(raw.trim()).ok())
411        .map(|t| t.with_timezone(&Utc))
412        .ok_or(RequestError::InvalidField)?;
413    let skew = Duration::seconds(crate::oidc::CLOCK_SKEW_LEEWAY_SECS as i64);
414    if issue_instant > now + skew
415        || issue_instant < now - Duration::seconds(REQUEST_MAX_AGE_SECS) - skew
416    {
417        return Err(RequestError::Stale);
418    }
419    Ok(issue_instant)
420}

422/// The one `saml:Issuer` among `children`: its text, the SP's entity id. Exactly
423/// one, non-empty, and when it carries a `Format` the entity one.
424pub(super) fn read_issuer(children: &[libxml::tree::Node]) -> Result<String, RequestError> {
425    let issuers: Vec<_> = children
426        .iter()
427        .filter(|c| is_element(c, NS_ASSERTION, "Issuer"))
428        .collect();
429    let [issuer] = issuers.as_slice() else {
430        return Err(RequestError::InvalidField);
431    };
432    if issuer
433        .get_attribute("Format")
434        .is_some_and(|f| f != NAME_ID_FORMAT_ENTITY)
435    {
436        return Err(RequestError::InvalidField);
437    }
438    let issuer = issuer.get_content().trim().to_owned();
439    if issuer.is_empty() {
440        return Err(RequestError::InvalidField);
441    }
442    Ok(issuer)
443}

445/// `xs:boolean`, absent meaning `false`.
446fn xs_boolean(raw: Option<String>) -> Result<bool, RequestError> {
447    match raw.as_deref().map(str::trim) {
448        None | Some("false" | "0") => Ok(false),
449        Some("true" | "1") => Ok(true),
450        Some(_) => Err(RequestError::InvalidField),
451    }
452}

454/// Where the document's signatures sit: `Ok(false)` for none, `Ok(true)` for
455/// exactly one enveloped child of the root referencing the root's `ID`, and
456/// [`RequestError::SignaturePlacement`] for anything else.
457///
458/// **A function over any root element** (D-38): `root` is whatever the caller
459/// established the document's root to be — an `AuthnRequest` here, a
460/// `LogoutRequest` or `LogoutResponse` in [`super::logout`] — and `root_id` its
461/// `ID`. The rule never looks at the root's name, so it cannot differ between
462/// message kinds.
463pub(super) fn signature_placement(
464    doc: &libxml::tree::Document,
465    root: &libxml::tree::Node,
466    root_id: &str,
467) -> Result<bool, RequestError> {
468    let mut context = libxml::xpath::Context::new(doc).map_err(|()| RequestError::Malformed)?;
469    let all = context
470        .findnodes(
471            &format!("//*[local-name()='Signature' and namespace-uri()='{NS_DSIG}']"),
472            None,
473        )
474        .map_err(|()| RequestError::Malformed)?;
475    if all.is_empty() {
476        return Ok(false);
477    }
478    let at_root: Vec<_> = root
479        .get_child_elements()
480        .into_iter()
481        .filter(|c| is_element(c, NS_DSIG, "Signature"))
482        .collect();
483    let ([signature], 1) = (at_root.as_slice(), all.len()) else {
484        return Err(RequestError::SignaturePlacement);
485    };
486    let references: Vec<_> = signature
487        .get_child_elements()
488        .into_iter()
489        .filter(|c| is_element(c, NS_DSIG, "SignedInfo"))
490        .flat_map(|info| info.get_child_elements())
491        .filter(|c| is_element(c, NS_DSIG, "Reference"))
492        .collect();
493    let expected = format!("#{root_id}");
494    match references.as_slice() {
495        [reference] if reference.get_attribute("URI").as_deref() == Some(expected.as_str()) => {
496            Ok(true)
497        }
498        _ => Err(RequestError::SignaturePlacement),
499    }
500}

502pub(super) fn is_element(node: &libxml::tree::Node, namespace: &str, name: &str) -> bool {
503    node.get_name() == name
504        && node
505            .get_namespace()
506            .is_some_and(|ns| ns.get_href() == namespace)

509/// Verify a POST-binding request's enveloped signature against the SP's
510/// registered certificate (DER). Call only after [`parse_authn_request`]
511/// answered `enveloped_signature: true`, which is what established that the one
512/// signature in the document is the root's own.
513///
514/// # Errors
515///
516/// [`RequestError::SignatureInvalid`] — a wrong key, a changed byte, a SHA-1
517/// algorithm, or anything xmlsec refuses.
518pub fn verify_post_signature(document: &str, sp_cert_der: &[u8]) -> Result<(), RequestError> {
519    let cert = CertificateDer::from(sp_cert_der.to_vec());
520    <XmlSec as CryptoProvider>::reduce_xml_to_signed_with_allowed_algorithms(
521        document,
522        &[cert],
523        ReduceMode::PreDigest,
524        Some(&ALLOWED_XML_SIGNATURE_ALGORITHMS),
525    )
526    .map(|_| ())
527    .map_err(|_| RequestError::SignatureInvalid)
528}

567impl<'q> RedirectQuery<'q> {
568    /// Split a raw query string for the SSO endpoint, which takes a
569    /// `SAMLRequest`. Parameters other than the four are ignored (a
570    /// `SAMLResponse` among them); each of the four may appear at most once, and
571    /// `SAMLRequest` must.
572    ///
573    /// # Errors
574    ///
575    /// [`RequestError::DuplicateParameter`], [`RequestError::Encoding`] (no
576    /// `SAMLRequest`), [`RequestError::TooLarge`].
577    pub fn parse(query: &'q str) -> Result<Self, RequestError> {
578        Self::split(query, false)
579    }
580
581    /// Split a raw query string for the SLO endpoint, which takes **either**
582    /// `SAMLRequest` or `SAMLResponse`, never both: a query naming both is
583    /// ambiguous (two readers could pick different ones) and is refused. Each of
584    /// the parameters may appear at most once.
585    ///
586    /// # Errors
587    ///
588    /// [`RequestError::DuplicateParameter`] (a repeated parameter, or both
589    /// messages), [`RequestError::Encoding`] (neither), [`RequestError::TooLarge`].
590    pub fn parse_logout(query: &'q str) -> Result<Self, RequestError> {
591        Self::split(query, true)
592    }
593
594    fn split(query: &'q str, either: bool) -> Result<Self, RequestError> {
595        if query.len() > MAX_ENCODED_REQUEST_BYTES * 3 {
596            return Err(RequestError::TooLarge);
597        }
598        let (mut request, mut response, mut relay_state, mut sig_alg, mut signature) =
599            (None, None, None, None, None);
600        for pair in query.split('&') {
601            let (name, value) = pair.split_once('=').unwrap_or((pair, ""));
602            let slot = match name {
603                "SAMLRequest" => &mut request,
604                "SAMLResponse" if either => &mut response,
605                "RelayState" => &mut relay_state,
606                "SigAlg" => &mut sig_alg,
607                "Signature" => &mut signature,
608                _ => continue,
609            };
610            if slot.replace(value).is_some() {
611                return Err(RequestError::DuplicateParameter);
612            }
613        }
614        let request = request.filter(|v: &&str| !v.is_empty());
615        let response = response.filter(|v: &&str| !v.is_empty());
616        let (param, message) = match (request, response) {
617            (Some(message), None) => (MessageParam::Request, message),
618            (None, Some(message)) => (MessageParam::Response, message),
619            (Some(_), Some(_)) => return Err(RequestError::DuplicateParameter),
620            (None, None) => return Err(RequestError::Encoding),
621        };
622        Ok(Self {
623            param,
624            message,
625            relay_state,
626            sig_alg,
627            signature,
628        })
629    }
630
631    /// Whether the query carries a signature (or a half of one).
632    #[must_use]
633    pub fn is_signed(&self) -> bool {
634        self.sig_alg.is_some() || self.signature.is_some()
635    }
636
637    /// The message parameter, URL-decoded. A base64 value contains no space, so a
638    /// `+` an SP left unencoded (and form decoding turned into a space) is put
639    /// back.
640    #[must_use]
641    pub fn message(&self) -> String {
642        url_decode(self.message).replace(' ', "+")
643    }
644
645    /// `RelayState`, URL-decoded, as it will be echoed.
646    #[must_use]
647    pub fn relay_state(&self) -> Option<String> {
648        self.relay_state.map(url_decode)
649    }
650
651    /// Verify the query signature against the SP's certificate (DER), over the
652    /// octets SAML Bindings §3.4.4.1 names, exactly as received:
653    /// `SAMLRequest=…` (or `SAMLResponse=…`)`[&RelayState=…]&SigAlg=…`.
654    ///
655    /// # Errors
656    ///
657    /// [`RequestError::SignatureMissing`] (no `SigAlg` or no `Signature`),
658    /// [`RequestError::SignatureAlgorithm`], [`RequestError::SignatureInvalid`].
659    pub fn verify_signature(&self, sp_cert_der: &[u8]) -> Result<(), RequestError> {
660        let (Some(sig_alg_raw), Some(signature_raw)) = (self.sig_alg, self.signature) else {
661            return Err(RequestError::SignatureMissing);
662        };
663        let digest = match url_decode(sig_alg_raw).as_str() {
664            "http://www.w3.org/2001/04/xmldsig-more#rsa-sha256" => {
665                openssl::hash::MessageDigest::sha256()
666            }
667            "http://www.w3.org/2001/04/xmldsig-more#rsa-sha384" => {
668                openssl::hash::MessageDigest::sha384()
669            }
670            "http://www.w3.org/2001/04/xmldsig-more#rsa-sha512" => {
671                openssl::hash::MessageDigest::sha512()
672            }
673            _ => return Err(RequestError::SignatureAlgorithm),
674        };
675        let signature = STANDARD
676            .decode(url_decode(signature_raw).replace(' ', "+").as_bytes())
677            .map_err(|_| RequestError::SignatureInvalid)?;
678
679        let mut signed = format!("{}={}", self.param.name(), self.message);
680        if let Some(relay_state) = self.relay_state {
681            signed.push_str("&RelayState=");
682            signed.push_str(relay_state);
683        }
684        signed.push_str("&SigAlg=");
685        signed.push_str(sig_alg_raw);
686
687        let public_key = openssl::x509::X509::from_der(sp_cert_der)
688            .and_then(|cert| cert.public_key())
689            .map_err(|_| RequestError::SignatureInvalid)?;
690        if public_key.id() != openssl::pkey::Id::RSA {
691            return Err(RequestError::SignatureAlgorithm);
692        }
693        let mut verifier = openssl::sign::Verifier::new(digest, &public_key)
694            .map_err(|_| RequestError::SignatureInvalid)?;
695        verifier
696            .update(signed.as_bytes())
697            .map_err(|_| RequestError::SignatureInvalid)?;
698        match verifier.verify(&signature) {
699            Ok(true) => Ok(()),
700            _ => Err(RequestError::SignatureInvalid),
701        }
702    }
703}
704
705/// `application/x-www-form-urlencoded` decoding of one value.
706fn url_decode(raw: &str) -> String {
707    url::form_urlencoded::parse(format!("v={raw}").as_bytes())
708        .next()
709        .map(|(_, v)| v.into_owned())
710        .unwrap_or_default()
711}

713/// Whether `requested` — an `AuthnRequest`'s `NameIDPolicy@Format` — can be
714/// honoured by an SP configured for `configured`: absent, `unspecified`, or
715/// the configured format's URN.
716#[must_use]
717pub fn name_id_format_compatible(
718    requested: Option<&str>,
719    configured: axiam_core::models::saml_sp::NameIdFormat,
720) -> bool {
721    match requested {
722        None => true,
723        Some(format) => format == NAME_ID_FORMAT_UNSPECIFIED || format == configured.urn(),
724    }
