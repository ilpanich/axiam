1//! Enveloped XML-DSig with the tenant's credential, and the check of what was
2//! signed.
3//!
4//! Signing goes through `samael`'s `xmlsec` provider, the library the SP side
5//! verifies with: it signs the **first** `ds:Signature` template in document
6//! order (a pre-order walk from the root), registering every `ID` attribute so
7//! the template's `#id` reference resolves. The issuer relies on that order and
8//! nothing else: an assertion is signed on its own, as the document root, so its
9//! template is the only one; a response's template sits after its `Issuer`,
10//! before the embedded (already signed) assertion, so it is reached first.
11

16use samael::crypto::{CertificateDer, CryptoProvider, ReduceMode, XmlSec};
17use uuid::Uuid;
18use zeroize::Zeroizing;

22/// XML-DSig namespace.
23const NS_DSIG: &str = "http://www.w3.org/2000/09/xmldsig#";
24/// Exclusive XML canonicalization 1.0, without comments.
25const C14N_EXCLUSIVE: &str = "http://www.w3.org/2001/10/xml-exc-c14n#";
26/// The enveloped-signature transform.
27const TRANSFORM_ENVELOPED: &str = "http://www.w3.org/2000/09/xmldsig#enveloped-signature";
28/// RSA PKCS#1 v1.5 over SHA-256 (D-21).
29const SIG_RSA_SHA256: &str = "http://www.w3.org/2001/04/xmldsig-more#rsa-sha256";
30/// SHA-256 digest.
31const DIGEST_SHA256: &str = "http://www.w3.org/2001/04/xmlenc#sha256";

33/// Refuse a credential that is not the tenant's, not `active`, or not valid at
34/// `now`.
35///
36/// `SamlIdpCredentialService::get_active_signing_key` returns the active row
37/// whatever its dates (T23.2.1); this is where an expired or not-yet-valid one
38/// stops. The interval is the certificate's, `not_before ≤ now < not_after`.
39pub(super) fn check_credential(
40    credential: &SamlIdpCredential,
41    tenant_id: Uuid,
42    now: DateTime<Utc>,
43) -> Result<(), SamlIdpError> {
44    if credential.tenant_id != tenant_id {
45        return Err(SamlIdpError::TenantMismatch);
46    }
47    if credential.status != SamlIdpCredentialStatus::Active {
48        return Err(SamlIdpError::CredentialNotActive);
49    }
50    if now < credential.not_before || now >= credential.not_after {
51        return Err(SamlIdpError::CredentialNotValid);
52    }
53    Ok(())
54}

56/// The credential's certificate as DER, for `KeyInfo` and for checking the
57/// output.
58pub(super) fn certificate_der(credential: &SamlIdpCredential) -> Result<Vec<u8>, SamlIdpError> {
59    crate::cert::pem_cert_to_der(&credential.certificate_pem)
60        .map_err(|_| SamlIdpError::SigningFailed)
61}

63/// The private key as DER, decoded straight into a zeroizing buffer.
64///
65/// Decoded here rather than by the `pem` crate, whose parser keeps
66/// intermediate copies of the base64 text it does not wipe. Accepts exactly one
67/// `PRIVATE KEY` (PKCS#8, what T23.2.1 stores) or `RSA PRIVATE KEY` (PKCS#1)
68/// block. A failure says nothing about the input.
69pub(super) fn private_key_der(
70    signing_key: &SamlIdpSigningKey,
71) -> Result<Zeroizing<Vec<u8>>, SamlIdpError> {
72    let pem: &str = signing_key.private_key_pem.as_str();
73    let (begin, end) = [
74        ("-----BEGIN PRIVATE KEY-----", "-----END PRIVATE KEY-----"),
75        (
76            "-----BEGIN RSA PRIVATE KEY-----",
77            "-----END RSA PRIVATE KEY-----",
78        ),
79    ]
80    .into_iter()
81    .find(|(begin, _)| pem.trim_start().starts_with(begin))
82    .ok_or(SamlIdpError::SigningFailed)?;
83    let body_start = pem.find(begin).ok_or(SamlIdpError::SigningFailed)? + begin.len();
84    let body_len = pem[body_start..]
85        .find(end)
86        .ok_or(SamlIdpError::SigningFailed)?;
87    if !pem[body_start + body_len + end.len()..].trim().is_empty() {
88        return Err(SamlIdpError::SigningFailed);
89    }
90
91    let mut base64_body = Zeroizing::new(String::with_capacity(body_len));
92    base64_body.extend(
93        pem[body_start..body_start + body_len]
94            .chars()
95            .filter(|c| !c.is_ascii_whitespace()),
96    );
97    let mut der = Zeroizing::new(vec![0u8; base64_body.len().div_ceil(4) * 3]);
98    let written = STANDARD
99        .decode_slice(base64_body.as_bytes(), der.as_mut_slice())
100        .map_err(|_| SamlIdpError::SigningFailed)?;
101    der.truncate(written);
102    if der.is_empty() {
103        return Err(SamlIdpError::SigningFailed);
104    }
105    Ok(der)
106}

108/// The `ds:Signature` template for the element whose `ID` is `id`, with the
109/// signing certificate in `KeyInfo/X509Data`.
110///
111/// No whitespace between elements: canonicalization keeps text nodes, and a
112/// template with none has nothing for a re-indenting proxy to change.
113pub(super) fn signature_template(id: &str, cert_der: &[u8]) -> String {
114    format!(
115        concat!(
116            r#"<ds:Signature xmlns:ds="{ns}">"#,
117            r#"<ds:SignedInfo>"#,
118            r#"<ds:CanonicalizationMethod Algorithm="{c14n}"/>"#,
119            r#"<ds:SignatureMethod Algorithm="{sig}"/>"#,
120            r##"<ds:Reference URI="#{id}">"##,
121            r#"<ds:Transforms>"#,
122            r#"<ds:Transform Algorithm="{enveloped}"/>"#,
123            r#"<ds:Transform Algorithm="{c14n}"/>"#,
124            r#"</ds:Transforms>"#,
125            r#"<ds:DigestMethod Algorithm="{digest}"/>"#,
126            r#"<ds:DigestValue></ds:DigestValue>"#,
127            r#"</ds:Reference>"#,
128            r#"</ds:SignedInfo>"#,
129            r#"<ds:SignatureValue></ds:SignatureValue>"#,
130            r#"<ds:KeyInfo><ds:X509Data><ds:X509Certificate>{cert}</ds:X509Certificate></ds:X509Data></ds:KeyInfo>"#,
131            r#"</ds:Signature>"#
132        ),
133        ns = NS_DSIG,
134        c14n = C14N_EXCLUSIVE,
135        sig = SIG_RSA_SHA256,
136        id = xml::escape(id),
137        enveloped = TRANSFORM_ENVELOPED,
138        digest = DIGEST_SHA256,
139        cert = STANDARD.encode(cert_der),
140    )
141}

143/// Sign the first signature template in `document`.
144pub(super) fn sign(document: &str, key_der: &[u8]) -> Result<String, SamlIdpError> {
145    match <XmlSec as CryptoProvider>::sign_xml(document.as_bytes(), key_der) {
146        Ok(signed) => Ok(xml::strip_declaration(&signed).to_owned()),
147        Err(error) => {
148            // The error is xmlsec's or libxml's fixed text; it never carries
149            // the key or the document.
150            tracing::error!(%error, "SAML IdP: signing failed");
151            Err(SamlIdpError::SigningFailed)
152        }
153    }
154}

156/// Check the finished response before it leaves: every signature verifies
157/// against the credential's certificate, and the document has exactly the
158/// shape the issuer meant to sign.
159///
160/// * exactly one `saml:Assertion`, the root's child, with `ID = assertion_id`;
161/// * one `ds:Signature` in the assertion, referencing `#assertion_id`;
162/// * when `response_signed`, one more, a child of the root, referencing
163///   `#response_id`; otherwise none there;
164/// * no other `ds:Signature` anywhere, and no duplicate `ID`.
165///
166/// Signatures are checked the way the SP side checks them since D-23: every
167/// `ds:Signature` in the response as delivered is verified on its own node by
168/// xmlsec (`reduce_xml_to_signed`), not just the first one.
169pub(super) fn verify_output(
170    response: &str,
171    cert_der: &[u8],
172    response_id: &str,
173    assertion_id: &str,
174    response_signed: bool,
175) -> Result<(), SamlIdpError> {
176    let fail = |reason: &'static str| {
177        tracing::error!(reason, "SAML IdP: the signed response failed its own check");
178        SamlIdpError::SigningFailed
179    };
180
181    let parser = libxml::parser::Parser::default();
182    let doc = parser
183        .parse_string(response.as_bytes())
184        .map_err(|_| fail("unparseable output"))?;
185    let root = doc.get_root_element().ok_or_else(|| fail("no root"))?;
186    if !is_element(&root, super::NS_PROTOCOL, "Response")
187        || root.get_attribute("ID").as_deref() != Some(response_id)
188    {
189        return Err(fail("root is not the response"));
190    }
191
192    let mut context = libxml::xpath::Context::new(&doc).map_err(|()| fail("no XPath context"))?;
193    let count = |context: &mut libxml::xpath::Context, path: &str| {
194        context
195            .findnodes(path, None)
196            .map(|nodes| nodes.len())
197            .map_err(|()| fail("XPath failed"))
198    };
199    if count(&mut context, "//*[local-name()='Assertion']")? != 1 {
200        return Err(fail("not exactly one Assertion"));
201    }
202    let signatures = count(
203        &mut context,
204        &format!("//*[local-name()='Signature' and namespace-uri()='{NS_DSIG}']"),
205    )?;
206    if signatures != if response_signed { 2 } else { 1 } {
207        return Err(fail("unexpected number of signatures"));
208    }
209    let ids = context
210        .findnodes("//@ID", None)
211        .map_err(|()| fail("XPath failed"))?
212        .iter()
213        .map(libxml::tree::Node::get_content)
214        .collect::<Vec<_>>();
215    if ids.len() != 2 || ids[0] == ids[1] {
216        return Err(fail("unexpected IDs"));
217    }
218
219    let children = root.get_child_elements();
220    let assertions: Vec<_> = children
221        .iter()
222        .filter(|c| is_element(c, super::NS_ASSERTION, "Assertion"))
223        .collect();
224    let [assertion] = assertions.as_slice() else {
225        return Err(fail("the Assertion is not the root's child"));
226    };
227    if assertion.get_attribute("ID").as_deref() != Some(assertion_id)
228        || !signature_child_references(assertion, assertion_id)
229    {
230        return Err(fail("the assertion's signature does not reference it"));
231    }
232    if signature_child_references(&root, response_id) != response_signed {
233        return Err(fail("the response signature is not as intended"));
234    }
235
236    let cert = CertificateDer::from(cert_der.to_vec());
237    <XmlSec as CryptoProvider>::reduce_xml_to_signed(response, &[cert], ReduceMode::PreDigest)
238        .map(|_| ())
239        .map_err(|_| fail("a signature does not verify"))
240}

242/// RSA PKCS#1 v1.5 with SHA-256 over `octets`: the HTTP-Redirect binding's
243/// **detached** query signature (SAML Bindings §3.4.4.1). Nothing XML is signed,
244/// so the signature is not a `ds:Signature` anyone could lift into another
245/// document (T-373).
246pub(super) fn sign_octets(octets: &str, key_der: &[u8]) -> Result<Vec<u8>, SamlIdpError> {
247    let fail = |reason: &'static str| {
248        tracing::error!(reason, "SAML IdP: signing a query failed");
249        SamlIdpError::SigningFailed
250    };
251    let key = openssl::pkey::PKey::private_key_from_der(key_der).map_err(|_| fail("key"))?;
252    let mut signer = openssl::sign::Signer::new(openssl::hash::MessageDigest::sha256(), &key)
253        .map_err(|_| fail("signer"))?;
254    signer
255        .update(octets.as_bytes())
256        .map_err(|_| fail("update"))?;
257    signer.sign_to_vec().map_err(|_| fail("sign"))
258}

260/// Check a logout message signed with an enveloped signature before it leaves:
261/// the root is the expected `samlp` element with `ID = root_id`; the document
262/// holds **exactly one** `ds:Signature`, the root's own child, whose one
263/// reference is `#root_id`; and xmlsec verifies it against the credential's
264/// certificate with the SHA-2 family only. A builder bug that would emit a
265/// wrapped or mis-referenced message fails closed here, not at an SP.
266pub(super) fn verify_root_signed(
267    document: &str,
268    cert_der: &[u8],
269    root_name: &str,
270    root_id: &str,
271) -> Result<(), SamlIdpError> {
272    let fail = |reason: &'static str| {
273        tracing::error!(reason, "SAML IdP: a logout message failed its own check");
274        SamlIdpError::SigningFailed
275    };
276    let doc = libxml::parser::Parser::default()
277        .parse_string(document.as_bytes())
278        .map_err(|_| fail("unparseable output"))?;
279    let root = doc.get_root_element().ok_or_else(|| fail("no root"))?;
280    if !is_element(&root, super::NS_PROTOCOL, root_name)
281        || root.get_attribute("ID").as_deref() != Some(root_id)
282    {
283        return Err(fail("root is not the message"));
284    }
285    let mut context = libxml::xpath::Context::new(&doc).map_err(|()| fail("no XPath context"))?;
286    let signatures = context
287        .findnodes(
288            &format!("//*[local-name()='Signature' and namespace-uri()='{NS_DSIG}']"),
289            None,
290        )
291        .map_err(|()| fail("XPath failed"))?;
292    if signatures.len() != 1 || !signature_child_references(&root, root_id) {
293        return Err(fail("the signature is not the root's own"));
294    }
295    let ids = context
296        .findnodes("//@ID", None)
297        .map_err(|()| fail("XPath failed"))?;
298    if ids.len() != 1 {
299        return Err(fail("unexpected IDs"));
300    }
301    super::request::verify_post_signature(document, cert_der)
302        .map_err(|_| fail("the signature does not verify"))
303}

305fn is_element(node: &libxml::tree::Node, namespace: &str, name: &str) -> bool {
306    node.get_name() == name
307        && node
308            .get_namespace()
309            .is_some_and(|ns| ns.get_href() == namespace)
310}

312/// Whether `parent` has exactly one `ds:Signature` child, whose one reference
313/// is `#id`.
314fn signature_child_references(parent: &libxml::tree::Node, id: &str) -> bool {
315    let signatures: Vec<_> = parent
316        .get_child_elements()
317        .into_iter()
318        .filter(|c| is_element(c, NS_DSIG, "Signature"))
319        .collect();
320    let [signature] = signatures.as_slice() else {
321        return false;
322    };
323    let references: Vec<_> = signature
324        .get_child_elements()
325        .into_iter()
326        .filter(|c| is_element(c, NS_DSIG, "SignedInfo"))
327        .flat_map(|info| info.get_child_elements())
328        .filter(|c| is_element(c, NS_DSIG, "Reference"))
329        .collect();
330    let expected = format!("#{id}");
331    matches!(references.as_slice(), [reference] if reference.get_attribute("URI").as_deref() == Some(expected.as_str()))
332}
