1//! Importing a service provider's metadata — a **parse to a draft, never a
2//! write** (G-2, T23.2.5, D-41).
3//!
4//! An SP's metadata document is attacker-influenced XML (an upload, or whatever
5//! a URL serves) and it is usually unsigned. So this module only *suggests*: it
6//! turns a document into a [`SpMetadataDraft`] — a
7//! [`SamlServiceProviderInput`] the administrator reads and then submits through
8//! the ordinary create or update, where the validator and the write-time
9//! refusals apply as to a manual entry. Nothing here touches the registry.
10//!
11//! # How the XML is refused
12//!
13//! As the receiver refuses an `AuthnRequest` (`request.rs`), in the same order:
14//!
15//! 1. the size ([`MAX_SP_METADATA_BYTES`], 512 KiB) and UTF-8;
16//! 2. [`refuse_markup_declarations`] — any `<!` other than a comment or CDATA
17//!    section — and [`refuse_other_encodings`] on the bytes, before any parser
18//!    sees them: with no DTD there is no entity, external or internal, to read
19//!    or to expand;
20//! 3. libxml, without recovery, network or DTD defaults, which bounds the nesting
21//!    depth and decides that the root really is `md:EntityDescriptor` (an
22//!    `EntitiesDescriptor` aggregate is refused here, before `samael`'s
23//!    recursive deserializer can be fed a deeply nested one);
24//! 4. `samael`'s metadata types, which open no network, and exactly one
25//!    `SPSSODescriptor` supporting SAML 2.0.
26//!
27//! Every refusal is one of three generic [`MetadataError`] categories. The
28//! message never carries the document, a parser's text, a status line or an
29//! address (T-356's lesson), so the route cannot be used to read an internal
30//! response back.
31//!
32//! # What is taken, and what is trusted
33//!
34//! Taken, as a draft: `entityID`; the HTTP-POST `AssertionConsumerService`
35//! endpoints with their `index` and `isDefault`; one `SingleLogoutService`
36//! (HTTP-Redirect preferred over HTTP-POST — no signature to harvest, D-38);
37//! the first signing `KeyDescriptor` as the request-signing certificate and an
38//! `use="encryption"` one as the encryption certificate (**`encrypt_assertions`
39//! is never set**, D-2); `AuthnRequestsSigned`; the first of persistent and
40//! `emailAddress` among the `NameIDFormat`s; a display name.
41//!
42//! **Trusted from an unsigned document: nothing.** A `ds:Signature` in it is not
43//! evaluated — there is no anchor, and evaluating it against a certificate the
44//! same document carries would be circular — and its presence is reported as a
45//! warning. `validUntil` and `cacheDuration` are ignored (D-54: a `cacheDuration` on the
46//! `SPSSODescriptor` is dropped from the parsed tree before `samael` reads it, and an
47//! `AssertionConsumerService` with no `index` is given the lowest unused one — each
48//! with a warning). The SHA-256
49//! fingerprints of the certificates are returned for the administrator to compare
50//! out of band. AXIAM never re-reads an SP's metadata on its own.
51

66/// The largest metadata document accepted, in bytes — the cap the SP side's
67/// `fetch_idp_metadata` already uses.
68pub const MAX_SP_METADATA_BYTES: usize = 512 * 1024;
69
70/// The metadata namespace.
71const NS_METADATA: &str = "urn:oasis:names:tc:SAML:2.0:metadata";
72/// XML-DSig namespace, for noticing a document signature.
73const NS_DSIG: &str = "http://www.w3.org/2000/09/xmldsig#";
74/// The SAML 2.0 protocol URN an `SPSSODescriptor` must support.
75const PROTOCOL_SAML2: &str = "urn:oasis:names:tc:SAML:2.0:protocol";

77/// Why metadata could not be imported: one of three generic categories, each
78/// with a fixed message. Nothing about the document, the response or the
79/// resolved address is in any of them.
80#[derive(Debug, Clone, Copy, PartialEq, Eq)]
81pub enum MetadataError {
82    /// The URL was refused before or while fetching: not `https`, carries
83    /// credentials, does not resolve, resolves to an address the SSRF guard
84    /// refuses, or redirects somewhere it refuses.
85    UrlRefused,
86    /// The fetch itself failed: a transport error, or a response that was not a
87    /// success.
88    FetchFailed,
89    /// The document is not usable SP metadata: too large, not UTF-8, containing
90    /// a DTD or entity declaration, not well-formed, not exactly one
91    /// `EntityDescriptor` with one SAML 2.0 `SPSSODescriptor`.
92    NotSpMetadata,
93}
94
95impl MetadataError {
96    /// The message every caller gives — and the audit row records the category
97    /// by [`Self::category`].
98    #[must_use]
99    pub const fn message(self) -> &'static str {
100        match self {
101            Self::UrlRefused => "metadata_url refused",
102            Self::FetchFailed => "metadata fetch failed",
103            Self::NotSpMetadata => "not SAML service-provider metadata",
104        }
105    }
106
107    /// The category as a stable machine-readable word, for the audit row.
108    #[must_use]
109    pub const fn category(self) -> &'static str {
110        match self {
111            Self::UrlRefused => "url_refused",
112            Self::FetchFailed => "fetch_failed",
113            Self::NotSpMetadata => "not_sp_metadata",
114        }
115    }
116}
117
118impl std::fmt::Display for MetadataError {
119    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
120        f.write_str(self.message())
121    }
122}

126/// What importing produced: a suggestion and what the administrator needs to
127/// judge it. Not a registration.
128#[derive(Debug, Clone, PartialEq, Eq)]
129pub struct SpMetadataDraft {
130    /// The registration the document suggests, as a create or update body.
131    /// `encrypt_assertions` is `false`.
132    pub service_provider: SamlServiceProviderInput,
133    /// Lower-case hex SHA-256 of the signing certificate's DER the draft
134    /// carries, or `None`.
135    pub signing_certificate_fingerprint: Option<String>,
136    /// Lower-case hex SHA-256 of the encryption certificate's DER the draft
137    /// carries, or `None`.
138    pub encryption_certificate_fingerprint: Option<String>,
139    /// What the administrator should know before submitting it. Human text; a
140    /// caller must not parse it.
141    pub warnings: Vec<String>,
142}

148/// Fetch the metadata at `url`: **one** request through
149/// [`axiam_pki::ssrf::guarded_fetch`] — the name resolved once and the
150/// connection pinned to the vetted address, private, loopback, link-local and
151/// cloud-metadata addresses refused, every redirect hop re-validated, a
152/// transport timeout and cap — then held to [`MAX_SP_METADATA_BYTES`].
153///
154/// `allow_private` is the guard's test seam and the production caller passes
155/// **`false`**: it relaxes the address rule and the scheme rule for the first
156/// hop only (never for a redirect), so a test can reach a loopback mock.
157/// With it `false`, only `https` is accepted, and a URL with a user name or
158/// password is refused (no credentials are ever sent).
159///
160/// No refresh, no retry, no cookies: a changed SP is re-imported by a person.
161///
162/// # Errors
163///
164/// [`MetadataError::UrlRefused`] for everything decided by the URL or the
165/// address it resolves to — **one** category, so the answer cannot be used to map
166/// internal DNS; [`MetadataError::FetchFailed`] for a transport failure or a
167/// non-success status; [`MetadataError::NotSpMetadata`] for a response over the
168/// cap.
169pub async fn fetch_sp_metadata(url: &str, allow_private: bool) -> Result<Vec<u8>, MetadataError> {
170    use axiam_pki::ssrf::{SsrfError, guarded_fetch, read_capped_body};
171
172    let parsed = url::Url::parse(url).map_err(|_| MetadataError::UrlRefused)?;
173    if parsed.host_str().is_none()
174        || !parsed.username().is_empty()
175        || parsed.password().is_some()
176        || (!allow_private && parsed.scheme() != "https")
177    {
178        return Err(MetadataError::UrlRefused);
179    }
180
181    let response = guarded_fetch(url, allow_private, |client, target| client.get(target))
182        .await
183        .map_err(|error| match error {
184            SsrfError::InvalidUrl
185            | SsrfError::ResolveFailed
186            | SsrfError::Blocked
187            | SsrfError::InsecureScheme
188            | SsrfError::TooManyRedirects => MetadataError::UrlRefused,
189            SsrfError::ResponseTooLarge(_) => MetadataError::NotSpMetadata,
190            SsrfError::ClientBuildFailed | SsrfError::RequestFailed(_) => {
191                MetadataError::FetchFailed
192            }
193        })?;
194    if !response.status().is_success() {
195        return Err(MetadataError::FetchFailed);
196    }
197    read_capped_body(response, MAX_SP_METADATA_BYTES)
198        .await
199        .map_err(|error| match error {
200            SsrfError::ResponseTooLarge(_) => MetadataError::NotSpMetadata,
201            _ => MetadataError::FetchFailed,
202        })
203}

209/// Parse `document` into a draft.
210///
211/// # Errors
212///
213/// [`MetadataError::NotSpMetadata`] for every way the document is refused. Never
214/// anything else, and never any text of the document.
215pub fn parse_sp_metadata(document: &[u8]) -> Result<SpMetadataDraft, MetadataError> {
216    let refused = MetadataError::NotSpMetadata;
217    if document.len() > MAX_SP_METADATA_BYTES {
218        return Err(refused);
219    }
220    let text = std::str::from_utf8(document).map_err(|_| refused)?;
221    refuse_markup_declarations(text).map_err(|_| refused)?;
222    refuse_other_encodings(text).map_err(|_| refused)?;
223
224    // libxml first: well-formedness, a bounded depth, and the root's identity.
225    let options = libxml::parser::ParserOptions {
226        recover: false,
227        no_net: true,
228        no_def_dtd: true,
229        ignore_enc: true,
230        encoding: Some("UTF-8"),
231        ..Default::default()
232    };
233    let doc = libxml::parser::Parser::default()
234        .parse_string_with_options(document, options)
235        .map_err(|_| refused)?;
236    let root = doc.get_root_element().ok_or(refused)?;
237    if root.get_name() != "EntityDescriptor"
238        || !root
239            .get_namespace()
240            .is_some_and(|ns| ns.get_href() == NS_METADATA)
241    {
242        // Includes an `EntitiesDescriptor` aggregate.
243        return Err(refused);
244    }
245
246    // D-54: two things `samael`'s types cannot read, normalised on the parsed
247    // tree (never by rewriting bytes). A document needing neither goes to
248    // `samael` exactly as it arrived.
249    let (normalised, normalisation_warnings) = normalise_tree(&doc, &root);
250    let parsed_text = normalised.as_deref().unwrap_or(text);
251
252    let descriptor: EntityDescriptorType = parsed_text.parse().map_err(|_| refused)?;
253    let EntityDescriptorType::EntityDescriptor(entity) = descriptor else {
254        return Err(refused);
255    };
256    let mut draft = draft_from(&entity, text)?;
257    draft.warnings.splice(0..0, normalisation_warnings);
258    Ok(draft)
259}

261/// What `samael` cannot type, fixed on the libxml tree (D-54): the `cacheDuration`
262/// of the `SPSSODescriptor` (an ISO-8601 duration where `samael` wants an
263/// integer — D-41 never uses it, and the draft has no cache field) is dropped,
264/// and an `AssertionConsumerService` with no `index` (`samael` requires one) gets
265/// the lowest unused non-negative index in document order. An `index` that is
266/// present is left alone: a malformed one is still refused by `samael`.
267///
268/// Returns the re-serialised document when something changed, else `None`, and a
269/// warning for each change that names it, so the administrator sees what was
270/// altered before saving.
271fn normalise_tree(
272    doc: &libxml::tree::Document,
273    root: &libxml::tree::Node,
274) -> (Option<String>, Vec<String>) {
275    let in_metadata = |node: &libxml::tree::Node, name: &str| {
276        node.get_name() == name
277            && node
278                .get_namespace()
279                .is_some_and(|ns| ns.get_href() == NS_METADATA)
280    };
281    let mut warnings = Vec::new();
282    let mut changed = false;
283    for mut descriptor in root
284        .get_child_elements()
285        .into_iter()
286        .filter(|n| in_metadata(n, "SPSSODescriptor"))
287    {
288        if descriptor.has_attribute("cacheDuration")
289            && descriptor.remove_attribute("cacheDuration").is_ok()
290        {
291            changed = true;
292            warnings.push(
293                "the SPSSODescriptor's cacheDuration was ignored: AXIAM never re-reads an SP's \
294                 metadata on its own"
295                    .to_string(),
296            );
297        }
298        let acs: Vec<libxml::tree::Node> = descriptor
299            .get_child_elements()
300            .into_iter()
301            .filter(|n| in_metadata(n, "AssertionConsumerService"))
302            .collect();
303        let mut used: std::collections::BTreeSet<u64> = acs
304            .iter()
305            .filter_map(|n| n.get_attribute("index")?.trim().parse::<u64>().ok())
306            .collect();
307        for mut endpoint in acs {
308            if endpoint.has_attribute("index") {
309                continue;
310            }
311            let index = (0u64..).find(|i| !used.contains(i)).unwrap_or(0);
312            if endpoint.set_attribute("index", &index.to_string()).is_ok() {
313                used.insert(index);
314                changed = true;
315                warnings.push(format!(
316                    "the AssertionConsumerService at {} has no index: it was given index \
317                     {index}",
318                    endpoint
319                        .get_attribute("Location")
320                        .map_or_else(|| "(no Location)".to_string(), |l| acs_location_label(&l)),
321                ));
322            }
323        }
324    }
325    (changed.then(|| doc.to_string()), warnings)
326}

328/// A `Location` as it goes into a warning: trimmed, free of control characters,
329/// cut to 200 characters. It is the SP's own text, shown to the administrator who
330/// imported that SP's document.
331fn acs_location_label(location: &str) -> String {
332    location
333        .trim()
334        .chars()
335        .filter(|c| !c.is_control())
336        .take(200)
337        .collect()
338}

340/// Map one `EntityDescriptor` to a draft.
341fn draft_from(entity: &EntityDescriptor, text: &str) -> Result<SpMetadataDraft, MetadataError> {
342    let refused = MetadataError::NotSpMetadata;
343    let entity_id = entity
344        .entity_id
345        .clone()
346        .filter(|id| !id.is_empty())
347        .ok_or(refused)?;
348    let sp = match entity.sp_sso_descriptors.as_deref() {
349        Some([only]) => only,
350        _ => return Err(refused),
351    };
352    if !sp
353        .protocol_support_enumeration
354        .as_deref()
355        .is_some_and(|list| list.split_whitespace().any(|p| p == PROTOCOL_SAML2))
356    {
357        return Err(refused);
358    }
359
360    let mut warnings = Vec::new();
361
362    // --- the signature of the document itself: noticed, never evaluated ---
363    if entity.signature.is_some() || text.contains(NS_DSIG) && text.contains("SignatureValue") {
364        warnings.push(
365            "metadata signature not verified: the document's own signature is not evaluated, \
366             so nothing in this draft is trusted because the document carried it"
367                .to_string(),
368        );
369    }
370
371    // --- AssertionConsumerService: HTTP-POST only ---
372    let mut dropped_bindings = 0usize;
373    let mut acs_urls: Vec<AcsEndpoint> = Vec::new();
374    let mut dropped_indexes = 0usize;
375    for endpoint in &sp.assertion_consumer_services {
376        if endpoint.binding != HTTP_POST_BINDING {
377            dropped_bindings += 1;
378            continue;
379        }
380        let Ok(index) = u16::try_from(endpoint.index) else {
381            dropped_indexes += 1;
382            continue;
383        };
384        if acs_urls.iter().any(|e| e.index == index) {
385            dropped_indexes += 1;
386            continue;
387        }
388        if acs_urls.len() == MAX_ACS_ENDPOINTS {
389            dropped_indexes += 1;
390            continue;
391        }
392        acs_urls.push(AcsEndpoint {
393            url: endpoint.location.clone(),
394            binding: SamlBinding::HttpPost,
395            index,
396            is_default: endpoint.is_default == Some(true),
397        });
398    }
399    if dropped_bindings > 0 {
400        warnings.push(format!(
401            "{dropped_bindings} AssertionConsumerService endpoint(s) with a binding other than \
402             HTTP-POST were not imported: AXIAM answers by HTTP-POST only"
403        ));
404    }
405    if dropped_indexes > 0 {
406        warnings.push(format!(
407            "{dropped_indexes} AssertionConsumerService endpoint(s) were not imported: an \
408             index that is out of range or repeated, or more than {MAX_ACS_ENDPOINTS} endpoints"
409        ));
410    }
411    if acs_urls.is_empty() {
412        warnings.push(
413            "the metadata names no HTTP-POST AssertionConsumerService: add one before \
414             registering"
415                .to_string(),
416        );
417    }
418    if let Some(first_default) = acs_urls.iter().position(|e| e.is_default) {
419        let mut extra = 0usize;
420        for (at, endpoint) in acs_urls.iter_mut().enumerate() {
421            if endpoint.is_default && at != first_default {
422                endpoint.is_default = false;
423                extra += 1;
424            }
425        }
426        if extra > 0 {
427            warnings.push(
428                "more than one AssertionConsumerService is marked default: only the first \
429                 stays the default"
430                    .to_string(),
431            );
432        }
433    }
434
435    // --- SingleLogoutService: one, HTTP-Redirect preferred ---
436    let slo_services = sp.single_logout_services.as_deref().unwrap_or_default();
437    let slo = slo_services
438        .iter()
439        .find(|e| e.binding == HTTP_REDIRECT_BINDING)
440        .or_else(|| slo_services.iter().find(|e| e.binding == HTTP_POST_BINDING));
441    let (slo_url, slo_binding) = match slo {
442        Some(endpoint) => (
443            Some(endpoint.location.clone()),
444            Some(if endpoint.binding == HTTP_REDIRECT_BINDING {
445                SamlBinding::HttpRedirect
446            } else {
447                SamlBinding::HttpPost
448            }),
449        ),
450        None => (None, None),
451    };
452    if !slo_services.is_empty() && slo_url.is_none() {
453        warnings.push(
454            "the metadata's SingleLogoutService uses a binding AXIAM does not support (only \
455             HTTP-Redirect and HTTP-POST): no logout endpoint was imported"
456                .to_string(),
457        );
458    } else if slo_services.len() > 1 {
459        warnings.push(
460            "more than one SingleLogoutService: one was imported, HTTP-Redirect before HTTP-POST"
461                .to_string(),
462        );
463    }
464
465    // --- keys ---
466    let keys = sp.key_descriptors.as_deref().unwrap_or_default();
467    let signing_keys: Vec<&KeyDescriptor> = keys
468        .iter()
469        .filter(|k| k.key_use.as_deref().is_none_or(|u| u == "signing"))
470        .collect();
471    let encryption_keys: Vec<&KeyDescriptor> = keys
472        .iter()
473        .filter(|k| k.key_use.as_deref() == Some("encryption"))
474        .collect();
475    if signing_keys.len() > 1 {
476        warnings.push("more than one signing certificate: only the first was imported".to_string());
477    }
478    let signing = signing_keys
479        .first()
480        .and_then(|k| certificate_of(k, "signing", &mut warnings));
481    let encryption = encryption_keys
482        .first()
483        .and_then(|k| certificate_of(k, "encryption", &mut warnings));
484    if encryption.is_some() {
485        warnings.push(
486            "an encryption certificate was imported, but assertion encryption is not supported \
487             yet: encrypt_assertions stays false"
488                .to_string(),
489        );
490    }
491
492    // --- NameID format and display name ---
493    let name_id_format = sp
494        .name_id_formats
495        .as_deref()
496        .unwrap_or_default()
497        .iter()
498        .find_map(|f| {
499            [NameIdFormat::Persistent, NameIdFormat::EmailAddress]
500                .into_iter()
501                .find(|known| known.urn() == f)
502        });
503    if name_id_format.is_none() && !sp.name_id_formats.as_deref().unwrap_or_default().is_empty() {
504        warnings.push(
505            "the metadata lists no NameIDFormat AXIAM issues (persistent, emailAddress): \
506             persistent was chosen"
507                .to_string(),
508        );
509    }
510    let display_name = display_name_of(sp.organization.as_ref(), &entity_id);
511
512    let service_provider = SamlServiceProviderInput {
513        enabled: true,
514        display_name,
515        entity_id,
516        acs_urls,
517        slo_url,
518        slo_binding,
519        name_id_format: name_id_format.unwrap_or_default(),
520        sign_responses: true,
521        // D-2: never set from a document.
522        encrypt_assertions: false,
523        sp_signing_cert_pem: signing.as_ref().map(|c| c.pem.clone()),
524        sp_encryption_cert_pem: encryption.as_ref().map(|c| c.pem.clone()),
525        want_authn_requests_signed: sp.authn_requests_signed == Some(true),
526        allow_idp_initiated: false,
527        attribute_mappings: Vec::new(),
528        allowed_groups: Vec::new(),
529    };
530
531    // What a write of this draft would be refused for, said now: the same rules,
532    // run on the draft, so the administrator is not told one thing here and
533    // another on submit. (The rules that need the datastore are the write's.)
534    for violation in crate::saml_sp::saml_sp_violations(&service_provider) {
535        warnings.push(format!(
536            "a write of this draft would be refused: {violation}"
537        ));
538    }
539    for refusal in crate::saml_sp::saml_sp_write_refusals(&service_provider) {
540        warnings.push(format!("a write of this draft would be refused: {refusal}"));
541    }
542
543    Ok(SpMetadataDraft {
544        signing_certificate_fingerprint: signing.map(|c| c.fingerprint),
545        encryption_certificate_fingerprint: encryption.map(|c| c.fingerprint),
546        service_provider,
547        warnings,
548    })
549}

551/// A certificate lifted out of a `KeyDescriptor`.
552struct ImportedCertificate {
553    pem: String,
554    fingerprint: String,
555}
556
557/// The first X.509 certificate of a `KeyDescriptor`, as PEM and fingerprint, or
558/// `None` with a warning when it carries none that parses. An expired
559/// certificate is imported and warned about: SAML trusts a registered key as a
560/// key, not by its dates.
561fn certificate_of(
562    key: &KeyDescriptor,
563    kind: &str,
564    warnings: &mut Vec<String>,
565) -> Option<ImportedCertificate> {
566    let encoded = key
567        .key_info
568        .x509_data
569        .as_ref()
570        .and_then(|data| data.certificates.first());
571    let Some(encoded) = encoded else {
572        warnings.push(format!(
573            "the {kind} KeyDescriptor carries no X509Certificate: it was not imported"
574        ));
575        return None;
576    };
577    let compact: String = encoded.chars().filter(|c| !c.is_whitespace()).collect();
578    let der = STANDARD.decode(compact.as_bytes()).ok()?;
579    let Ok((_, parsed)) = X509Certificate::from_der(&der) else {
580        warnings.push(format!(
581            "the {kind} certificate is not a parseable X.509 certificate: it was not imported"
582        ));
583        return None;
584    };
585    if parsed.validity().not_after.timestamp() < chrono::Utc::now().timestamp() {
586        warnings.push(format!(
587            "the {kind} certificate has expired; it is imported because SAML trusts a \
588             registered key as a key, but the SP will need to rotate it"
589        ));
590    }
591    // LF line endings, as every other certificate PEM in the registry.
592    let pem = ::pem::encode_config(
593        &::pem::Pem::new("CERTIFICATE", der.clone()),
594        ::pem::EncodeConfig::new().set_line_ending(::pem::LineEnding::LF),
595    );
596    Some(ImportedCertificate {

602/// A display name that satisfies the registry's own rule: the first
603/// `OrganizationDisplayName`, else the entity id's host, else the entity id —
604/// trimmed, free of control characters, at most
605/// [`MAX_DISPLAY_NAME_BYTES`] bytes.
606fn display_name_of(
607    organization: Option<&samael::metadata::Organization>,
608    entity_id: &str,
609) -> String {
610    let from_document = organization
611        .and_then(|o| o.organization_display_names.as_ref())
612        .and_then(|names| names.first())
613        .map(|n| n.value.as_str());
614    let host = url::Url::parse(entity_id)
615        .ok()
616        .and_then(|u| u.host_str().map(str::to_owned));
617    let candidate = from_document
618        .map(str::to_owned)
619        .or(host)
620        .unwrap_or_else(|| entity_id.to_owned());
621    let mut cleaned: String = candidate.chars().filter(|c| !c.is_control()).collect();
622    cleaned = cleaned.trim().to_owned();
623    while cleaned.len() > MAX_DISPLAY_NAME_BYTES {
624        cleaned.pop();
625    }
626    let cleaned = cleaned.trim().to_owned();
627    if cleaned.is_empty() {
628        "service provider".to_owned()
629    } else {
630        cleaned
631    }
632}
