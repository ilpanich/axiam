1//! The SAML identity provider's identifiers: its entity id and endpoint URLs
2//! (G-2, T23.2.2 / T23.2.5).
3//!
4//! **Not behind the `saml` feature.** They are string functions of the
5//! deployment's public base URL and a tenant id, and contract §29's `get_idp`
6//! answers them in every build — including one that serves no SAML — so the
7//! administrator sees the metadata URL and entity id an SP will be given before
8//! the IdP is switched on. They were in `saml_idp`, which is gated; they moved
9//! here so the two can never disagree (D-40).
10//!
11//! The base is the deployment's public root, `AuthConfig::root_issuer()` — the
12//! value the OIDC issuer and the T21.6 per-tenant issuers are built on — with or
13//! without a trailing slash. The tenant is the tenant **of the request path**,
14//! never one read from a stored row.
15
16use uuid::Uuid;
17
18/// The tenant's IdP entity id: `{public_base_url}/saml/v2/{tenant_id}/metadata`.
19///
20/// **The one definition.** The `Issuer` of every response and assertion, the
21/// metadata document's `entityID` (T23.2.5) and anything else that names the
22/// IdP must come from here, so they cannot disagree.
23///
24/// The entity id is the metadata URL itself, the common convention that lets an
25/// SP administrator paste one URL and fetch the metadata from it.
26#[must_use]
27pub fn idp_entity_id(public_base_url: &str, tenant_id: Uuid) -> String {
28    idp_endpoint(public_base_url, tenant_id, "metadata")
29}
30
31/// The tenant's SSO endpoint, `{public_base_url}/saml/v2/{tenant_id}/sso`, for
32/// the metadata `SingleSignOnService` locations (T23.2.5).
33#[must_use]
34pub fn idp_sso_url(public_base_url: &str, tenant_id: Uuid) -> String {
35    idp_endpoint(public_base_url, tenant_id, "sso")
36}
37
38/// The tenant's SLO endpoint, `{public_base_url}/saml/v2/{tenant_id}/slo`, for
39/// the metadata `SingleLogoutService` locations (T23.2.4) and the `Destination`
40/// every logout message sent to it must carry.
41#[must_use]
42pub fn idp_slo_url(public_base_url: &str, tenant_id: Uuid) -> String {
43    idp_endpoint(public_base_url, tenant_id, "slo")
44}
45
46fn idp_endpoint(public_base_url: &str, tenant_id: Uuid, leaf: &str) -> String {
47    format!(
48        "{}/saml/v2/{tenant_id}/{leaf}",
49        public_base_url.trim_end_matches('/')
50    )
51}
52
53#[cfg(test)]
54mod tests {
55    use super::*;
56
57    #[test]
58    fn the_three_urls_are_one_function_of_the_base_and_the_tenant() {
59        let tenant = Uuid::nil();
60        for base in ["https://iam.example.com", "https://iam.example.com/"] {
61            assert_eq!(
62                idp_entity_id(base, tenant),
63                format!("https://iam.example.com/saml/v2/{tenant}/metadata")
64            );
65            assert_eq!(
66                idp_sso_url(base, tenant),
67                format!("https://iam.example.com/saml/v2/{tenant}/sso")
68            );
69            assert_eq!(
70                idp_slo_url(base, tenant),
71                format!("https://iam.example.com/saml/v2/{tenant}/slo")
72            );
73        }
74    }
75}
