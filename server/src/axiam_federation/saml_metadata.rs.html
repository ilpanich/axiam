1//! SAML SP: the IdP's metadata document — its signature and its cache (#530,
2//! P23W3-07).
3//!
4//! The SP reads three things from an IdP's metadata: its entity ID, its SSO
5//! URL and the binding of that URL. The assertion signing certificate is
6//! **not** among them (it is pinned on the federation configuration), so forged
7//! metadata cannot forge an assertion. It can still choose the HTTPS URL a
8//! person is redirected to with the `AuthnRequest` — a phishing redirect — and
9//! before this module every SP-initiated sign-in fetched the document again, so
10//! every sign-in also depended on the metadata host being up.
11//!
12//! # Signature
13//!
14//! When the configuration carries `idp_metadata_signing_cert_pem`, the
15//! document must be signed, and only one place may hold the signature: the
16//! enveloped child of the `md:EntityDescriptor` root, with one reference,
17//! naming the root's `ID` — the IdP receiver's placement rule
18//! ([`signature_placement`], D-23), applied to a different root. Any other
19//! `ds:Signature` refuses the document, and so does one naming a transform or
20//! canonicalization other than `enveloped-signature` and the c14n family
21//! without comments (R101W2-01). The one admitted signature is verified
22//! by xmlsec on its own node against the configured certificate, with the SHA-2
23//! algorithms only ([`ALLOWED_XML_SIGNATURE_ALGORITHMS`]; a configuration's
24//! `allow_sha1_signatures` applies to responses, not to metadata), and what is
25//! then read is xmlsec's **pre-digest** output — the bytes the digest covered —
26//! not the document as fetched, so nothing outside the signature is read.
27//!
28//! Every document, signed or not, is first refused on its bytes if it carries a
29//! markup declaration or an encoding other than UTF-8, as a SAML response is
30//! (#531): metadata has no use for a DTD.
31//!
32//! A fetched document is verified and parsed on the blocking pool, under the
33//! deployment's crypto semaphore ([`parse_idp_metadata_off_worker`],
34//! R101W2-01), never on an async worker.
35//!
36//! # Cache
37//!
38//! [`SamlMetadataCache`] holds the parsed document per process, keyed by the
39//! federation configuration's id and checked against the metadata URL and the
40//! configuration's `updated_at`, so any edit of the configuration — a new URL,
41//! a new certificate — misses. An entry lives for the document's
42//! `cacheDuration` ([`DEFAULT_TTL`] when it states none), held between
43//! [`MIN_TTL`] and [`MAX_TTL`], and never past its `validUntil`. A document
44//! whose `validUntil` has passed is refused.
45//!
46//! A refetch whose SSO URL names another host than the entry it replaces is
47//! reported ([`SsoHostChange`]) for the caller to audit: the metadata host can
48//! move users, so a move should be visible.

69/// SAML 2.0 metadata namespace.
70const NS_METADATA: &str = "urn:oasis:names:tc:SAML:2.0:metadata";

72/// The largest metadata document read: 512 KiB.
73pub const MAX_METADATA_SIZE: usize = 512 * 1024;

75/// How long a document that states no `cacheDuration` is kept: one hour, the
76/// OIDC discovery cache's TTL.
77pub const DEFAULT_TTL: Duration = Duration::hours(1);
78
79/// The shortest time an entry is kept, whatever `cacheDuration` says — so a
80/// document stating `PT0S` does not turn the cache off — unless its
81/// `validUntil` comes first.
82pub const MIN_TTL: Duration = Duration::minutes(5);
83
84/// The longest time an entry is kept, whatever `cacheDuration` says: a moved
85/// or re-keyed IdP is picked up within a day without an edit.
86pub const MAX_TTL: Duration = Duration::hours(24);

101/// Fetch a metadata document (no parsing, no cache).
102///
103/// # Errors
104///
105/// [`FederationError::InvalidMetadataUrl`] for a URL that is not HTTPS;
106/// [`FederationError::SamlMetadataFailed`] for a refused or failed fetch, a
107/// non-2xx answer, a body over [`MAX_METADATA_SIZE`] or one that is not UTF-8.
108pub async fn fetch_metadata_document(metadata_url: String) -> Result<String, FederationError> {
109    crate::validate_metadata_url(&metadata_url)?;
110
111    // SECHRD-02: route the metadata GET through the shared, IP-pinning SSRF
112    // guard (D-01a/b/c). Production always fails closed against
113    // private/loopback/link-local addresses and internal redirect targets —
114    // `allow_private=false`.
115    let response = crate::ssrf::guarded_fetch(&metadata_url, false, |c, u| c.get(u))
116        .await
117        .map_err(|e| FederationError::SamlMetadataFailed(e.to_string()))?;
118
119    if !response.status().is_success() {
120        return Err(FederationError::SamlMetadataFailed(format!(
121            "HTTP {} from metadata endpoint",
122            response.status()
123        )));
124    }
125
126    let bytes = response.bytes().await.map_err(|e| {
127        FederationError::SamlMetadataFailed(format!("Failed to read metadata body: {e}"))
128    })?;
129    if bytes.len() > MAX_METADATA_SIZE {
130        return Err(FederationError::SamlMetadataFailed(format!(
131            "Metadata document too large: {} bytes (max {MAX_METADATA_SIZE})",
132            bytes.len(),
133        )));
134    }
135
136    String::from_utf8(bytes.to_vec())
137        .map_err(|e| FederationError::SamlMetadataFailed(format!("Invalid UTF-8 in metadata: {e}")))
138}

140/// A metadata document, verified when a certificate was given, and read.
141#[derive(Debug, Clone)]
142pub struct ParsedIdpMetadata {
143    /// What the SP uses.
144    pub metadata: IdpMetadata,
145    /// `validUntil`, when the document states one (still in the future).
146    pub valid_until: Option<DateTime<Utc>>,
147    /// `cacheDuration`, when the document states one AXIAM can read.
148    pub cache_duration: Option<Duration>,
149}
150
151impl ParsedIdpMetadata {
152    /// When a cache entry made from this document at `now` expires:
153    /// `cacheDuration` (or [`DEFAULT_TTL`]) held to [`MIN_TTL`]..=[`MAX_TTL`],
154    /// and never later than `validUntil`.
155    #[must_use]
156    pub fn expires_at(&self, now: DateTime<Utc>) -> DateTime<Utc> {
157        let ttl = self
158            .cache_duration
159            .unwrap_or(DEFAULT_TTL)
160            .clamp(MIN_TTL, MAX_TTL);
161        let by_ttl = now + ttl;
162        self.valid_until.map_or(by_ttl, |until| by_ttl.min(until))
163    }
164}

167    FederationError::SamlMetadataFailed(reason.into())

170/// Read an IdP metadata document; with `signing_cert_pem`, only after its
171/// signature verified (see the module documentation for the rule).
172///
173/// # Errors
174///
175/// [`FederationError::SamlMetadataFailed`]: a markup declaration or another
176/// encoding; with a certificate, an unsigned document, a signature anywhere but
177/// on the `EntityDescriptor` root, or one that does not verify; a document
178/// whose `validUntil` is not after `now`; no `EntityDescriptor`, `entityID`,
179/// `IDPSSODescriptor` or HTTPS SSO endpoint on either binding.
180/// [`FederationError::InvalidIdpCert`] for a certificate that is not PEM.
181pub fn parse_idp_metadata(
182    text: &str,
183    signing_cert_pem: Option<&str>,
184    now: DateTime<Utc>,
185) -> Result<ParsedIdpMetadata, FederationError> {
186    // #531's rule for SAML responses, on the bytes and before any parser.
187    // R101W2-02: from here on `text` is the document without its one leading
188    // byte-order mark (Entra ID's metadata has one), so the signature check
189    // and samael read the same characters.
190    let text =
191        refuse_other_encodings(text).map_err(|_| metadata_failed("metadata is not plain UTF-8"))?;
192    refuse_markup_declarations(text)
193        .map_err(|_| metadata_failed("metadata declares a DTD or an entity"))?;
194
195    let verified;
196    let readable = match signing_cert_pem {
197        Some(pem) => {
198            verified = verify_metadata_signature(text, pem)?;
199            verified.as_str()
200        }
201        None => text,
202    };
203
204    let descriptor_type: EntityDescriptorType = readable
205        .parse()
206        .map_err(|e| metadata_failed(format!("Failed to parse metadata XML: {e}")))?;
207
208    // A signed document is one `EntityDescriptor` (the root the signature is
209    // on); an unsigned one may be an aggregate, whose first entity is read.
210    let (outer_valid_until, outer_cache_duration) = match &descriptor_type {
211        EntityDescriptorType::EntitiesDescriptor(all) => {
212            (all.valid_until, all.cache_duration.clone())
213        }
214        EntityDescriptorType::EntityDescriptor(_) => (None, None),
215    };
216    let ed = descriptor_type
217        .iter()
218        .next()
219        .ok_or_else(|| metadata_failed("No EntityDescriptor found in metadata"))?;
220
221    let valid_until = match (ed.valid_until, outer_valid_until) {
222        (Some(a), Some(b)) => Some(a.min(b)),
223        (a, b) => a.or(b),
224    };
225    if valid_until.is_some_and(|until| until <= now) {
226        return Err(metadata_failed("metadata validUntil has passed"));
227    }
228    let cache_duration = ed
229        .cache_duration
230        .as_deref()
231        .or(outer_cache_duration.as_deref())
232        .and_then(parse_xs_duration);
233
234    let entity_id = ed
235        .entity_id
236        .clone()
237        .ok_or_else(|| metadata_failed("EntityDescriptor missing entityID"))?;
238    let idp = ed
239        .idp_sso_descriptors
240        .as_ref()
241        .and_then(|d| d.first())
242        .ok_or_else(|| metadata_failed("No IDPSSODescriptor in metadata"))?;
243
244    // Prefer HTTP-POST binding, fall back to HTTP-Redirect.
245    let sso_endpoint = idp
246        .single_sign_on_services
247        .iter()
248        .find(|ep| ep.binding == HTTP_POST_BINDING)
249        .or_else(|| {
250            idp.single_sign_on_services
251                .iter()
252                .find(|ep| ep.binding == HTTP_REDIRECT_BINDING)
253        })
254        .ok_or_else(|| metadata_failed("No HTTP-POST or HTTP-Redirect SSO endpoint"))?;
255
256    // The SSO endpoint must be HTTPS, so a person is never sent to an insecure
257    // origin. Fail closed on a URL that does not parse.
258    let sso_parsed = url::Url::parse(&sso_endpoint.location)
259        .map_err(|e| metadata_failed(format!("IdP SSO endpoint is not a valid URL: {e}")))?;
260    if sso_parsed.scheme() != "https" {
261        return Err(metadata_failed("IdP SSO endpoint must use HTTPS"));
262    }
263
264    Ok(ParsedIdpMetadata {
265        metadata: IdpMetadata {
266            entity_id,
267            sso_url: sso_endpoint.location.clone(),
268            sso_binding: sso_endpoint.binding.clone(),
269        },
270        valid_until,
271        cache_duration,
272    })
273}

275/// [`parse_idp_metadata`] on the blocking pool, under `gate` when one is given
276/// (R101W2-01; the wait for a permit is bounded, #654): the signature check canonicalizes and hashes up to
277/// [`MAX_METADATA_SIZE`] bytes, which must not hold an async worker. A
278/// sign-in's cache miss and the uncached administrative fetch both go through
279/// here.
280///
281/// # Errors
282///
283/// What [`parse_idp_metadata`] returns, or [`FederationError::SamlMetadataFailed`]
284/// when the blocking task does not complete.
285pub async fn parse_idp_metadata_off_worker(
286    text: String,
287    signing_cert_pem: Option<String>,
288    now: DateTime<Utc>,
289    gate: Option<CryptoGate>,
290) -> Result<ParsedIdpMetadata, FederationError> {
291    let permit = match gate {
292        // #654: the wait is bounded; a busy gate is the refusal this path
293        // already has for a verification that cannot run.
294        Some(gate) => Some(gate.acquire("saml_metadata").await.map_err(|refusal| {
295            metadata_failed(match refusal {
296                GateRefusal::Busy => "metadata verification is busy; retry shortly",
297                GateRefusal::Closed => "metadata verification is unavailable",
298            })
299        })?),
300        None => None,
301    };
302    tokio::task::spawn_blocking(move || {
303        let _permit = permit;
304        parse_idp_metadata(&text, signing_cert_pem.as_deref(), now)
305    })
306    .await
307    .map_err(|_| metadata_failed("metadata verification did not complete"))?
308}

310/// Verify the metadata's one enveloped signature and return the bytes it
311/// covers (xmlsec's pre-digest output: the `EntityDescriptor`, canonicalized,
312/// without its signature).
313fn verify_metadata_signature(text: &str, pem: &str) -> Result<String, FederationError> {
314    let der = crate::cert::pem_cert_to_der(pem)?;
315
316    let doc = libxml::parser::Parser::default()
317        .parse_string_with_options(text.as_bytes(), receiver_parser_options())
318        .map_err(|_| metadata_failed("metadata is not XML"))?;
319    let root = doc
320        .get_root_element()
321        .ok_or_else(|| metadata_failed("metadata has no root element"))?;
322    let root_is_entity = root.get_name() == "EntityDescriptor"
323        && root
324            .get_namespace()
325            .is_some_and(|ns| ns.get_href() == NS_METADATA);
326    if !root_is_entity {
327        return Err(metadata_failed(
328            "signed metadata must be one md:EntityDescriptor",
329        ));
330    }
331    let root_id = root
332        .get_attribute("ID")
333        .filter(|id| !id.is_empty())
334        .ok_or_else(|| metadata_failed("metadata is not signed (its root has no ID)"))?;
335    // The reference names the root by `ID`; a second element carrying the same
336    // value would leave xmlsec and this check free to mean different nodes.
337    let mut context = libxml::xpath::Context::new(&doc)
338        .map_err(|()| metadata_failed("metadata: XPath context failed"))?;
339    let bearers = context
340        .findnodes("//*[@ID]", None)
341        .map_err(|()| metadata_failed("metadata: XPath evaluation failed"))?;
342    if bearers
343        .iter()
344        .filter(|n| n.get_attribute("ID").as_deref() == Some(root_id.as_str()))
345        .count()
346        != 1
347    {
348        return Err(metadata_failed("metadata repeats its root ID"));
349    }
350
351    // The placement rule, and with it the transform allow-list (R101W2-01):
352    // before xmlsec, which evaluates the transforms before the signature value.
353    match signature_placement(&doc, &root, &root_id) {
354        Ok(true) => {}
355        Ok(false) => return Err(metadata_failed("metadata is not signed")),
356        Err(_) => {
357            return Err(metadata_failed(
358                "a metadata signature may only be the EntityDescriptor root's own, \
359                 with enveloped-signature and canonicalization transforms only",
360            ));
361        }
362    }
363
364    <XmlSec as CryptoProvider>::reduce_xml_to_signed_with_allowed_algorithms(
365        text,
366        &[CertificateDer::from(der)],
367        ReduceMode::PreDigest,
368        Some(&ALLOWED_XML_SIGNATURE_ALGORITHMS),
369    )
370    .map_err(|_| metadata_failed("metadata signature does not verify"))
371}

373/// An `xs:duration` (`PnYnMnDTnHnMnS`, seconds may be fractional) as a
374/// [`Duration`]. A year counts 365 days and a month 30: the result is only
375/// ever compared with [`MAX_TTL`]. `None` for a negative or malformed value,
376/// which then reads as "no `cacheDuration`".
377#[must_use]
378pub fn parse_xs_duration(raw: &str) -> Option<Duration> {
379    let rest = raw.trim().strip_prefix('P')?;
380    let (date, time) = match rest.split_once('T') {
381        Some((d, t)) if !t.is_empty() => (d, Some(t)),
382        Some(_) => return None,
383        None => (rest, None),
384    };
385    if date.is_empty() && time.is_none() {
386        return None;
387    }
388    let mut millis: i64 = 0;
389    let mut add = |digits: &str, unit_ms: i64| -> Option<()> {
390        let value = if let Some((whole, frac)) = digits.split_once('.') {
391            if unit_ms != 1_000 || whole.is_empty() {
392                return None;
393            }
394            let whole: i64 = whole.parse().ok()?;
395            let frac_ms: i64 = format!("{frac:0<3}")[..3].parse().ok()?;
396            whole.checked_mul(1_000)?.checked_add(frac_ms)?
397        } else {
398            digits.parse::<i64>().ok()?.checked_mul(unit_ms)?
399        };
400        millis = millis.checked_add(value)?;
401        Some(())
402    };
403    let mut walk = |part: &str, units: &[(char, i64)]| -> Option<()> {
404        let mut digits = String::new();
405        let mut next_unit = 0;
406        for c in part.chars() {
407            if c.is_ascii_digit() || c == '.' {
408                digits.push(c);
409                continue;
410            }
411            let at = units[next_unit..].iter().position(|(u, _)| *u == c)? + next_unit;
412            if digits.is_empty() {
413                return None;
414            }
415            add(&digits, units[at].1)?;
416            digits.clear();
417            next_unit = at + 1;
418        }
419        digits.is_empty().then_some(())
420    };
421    const DAY: i64 = 86_400_000;
422    walk(date, &[('Y', 365 * DAY), ('M', 30 * DAY), ('D', DAY)])?;
423    if let Some(time) = time {
424        walk(time, &[('H', 3_600_000), ('M', 60_000), ('S', 1_000)])?;
425    }
426    Some(Duration::milliseconds(millis))
427}

448#[derive(Debug, Clone)]
449struct CacheEntry {
450    metadata_url: String,
451    config_version: DateTime<Utc>,
452    metadata: IdpMetadata,
453    expires_at: DateTime<Utc>,
454}

456/// Per-process cache of parsed IdP metadata, keyed by federation
457/// configuration. Cheap to clone; clones share the map.
458#[derive(Clone, Default)]
459pub struct SamlMetadataCache {
460    entries: Arc<Mutex<HashMap<Uuid, CacheEntry>>>,
461    /// R101W2-01: the gate a miss verifies and parses under, on the blocking
462    /// pool ([`parse_idp_metadata_off_worker`]). `None`: the blocking pool
463    /// only.
464    gate: Option<CryptoGate>,
465}
466
467impl SamlMetadataCache {
468    /// An empty cache.
469    #[must_use]
470    pub fn new() -> Self {
471        Self::default()
472    }
473
474    /// Verify and parse a fetched document under `gate` — the deployment's
475    /// crypto semaphore and the wait for it. Keeps the entries (clones still
476    /// share them).
477    #[must_use]
478    pub fn with_crypto_gate(mut self, gate: CryptoGate) -> Self {
479        self.gate = Some(gate);
480        self
481    }
482
483    fn entries(&self) -> std::sync::MutexGuard<'_, HashMap<Uuid, CacheEntry>> {
484        // A panic while the lock was held leaves a map of whole entries; keep
485        // using it rather than failing every sign-in.
486        self.entries
487            .lock()
488            .unwrap_or_else(std::sync::PoisonError::into_inner)
489    }
490
491    /// The configuration's IdP metadata: the cached entry when it was made
492    /// from this configuration (same id, metadata URL and `updated_at`) and has
493    /// not expired at `now`, otherwise `fetch`ed, verified, parsed and stored.
494    ///
495    /// The lock is never held across `fetch`: two misses at once both fetch,
496    /// and the later store wins.
497    ///
498    /// # Errors
499    ///
500    /// [`FederationError::SamlMetadataFailed`] when the configuration has no
501    /// metadata URL; otherwise whatever `fetch` or [`parse_idp_metadata`]
502    /// returns. A failure stores nothing and leaves an earlier entry in place.
503    pub async fn resolve<F, Fut>(
504        &self,
505        config: &FederationConfig,
506        now: DateTime<Utc>,
507        fetch: F,
508    ) -> Result<ResolvedIdpMetadata, FederationError>
509    where
510        F: FnOnce(String) -> Fut,
511        Fut: Future<Output = Result<String, FederationError>>,
512    {
513        let metadata_url = config
514            .metadata_url
515            .as_deref()
516            .ok_or_else(|| metadata_failed("No metadata URL configured"))?;
517
518        if let Some(entry) = self.entries().get(&config.id)
519            && entry.metadata_url == metadata_url
520            && entry.config_version == config.updated_at
521            && now < entry.expires_at
522        {
523            return Ok(ResolvedIdpMetadata {
524                metadata: entry.metadata.clone(),
525                sso_host_change: None,
526            });
527        }
528
529        let text = fetch(metadata_url.to_owned()).await?;
530        let parsed = parse_idp_metadata_off_worker(
531            text,
532            config.idp_metadata_signing_cert_pem.clone(),
533            now,
534            self.gate.clone(),
535        )
536        .await?;
537        let entry = CacheEntry {
538            metadata_url: metadata_url.to_owned(),
539            config_version: config.updated_at,
540            metadata: parsed.metadata.clone(),
541            expires_at: parsed.expires_at(now),
542        };
543        let previous = self.entries().insert(config.id, entry);
544
545        let new_host = sso_host(&parsed.metadata.sso_url);
546        let sso_host_change = previous
547            .map(|p| sso_host(&p.metadata.sso_url))
548            .filter(|old_host| *old_host != new_host)
549            .map(|old_host| SsoHostChange { old_host, new_host });
550        Ok(ResolvedIdpMetadata {
551            metadata: parsed.metadata,
552            sso_host_change,
553        })
554    }
555}

557/// The host of an SSO URL `parse_idp_metadata` accepted (so it parses).
558fn sso_host(sso_url: &str) -> String {
559    url::Url::parse(sso_url)
560        .ok()
561        .and_then(|u| u.host_str().map(str::to_ascii_lowercase))
562        .unwrap_or_default()
563}

583    /// A fetcher that serves `doc` and counts its calls.
584    fn serving(
585        doc: String,
586        calls: &AtomicUsize,
587    ) -> impl FnOnce(String) -> std::future::Ready<Result<String, FederationError>> + '_ {
588        move |_url| {
589            calls.fetch_add(1, Ordering::SeqCst);
590            std::future::ready(Ok(doc))
591        }
592    }

766    /// #530: a second sign-in within the document's freshness is served from
767    /// the cache; it is fetched again once the entry expires, and at once when
768    /// the configuration is edited (its `updated_at` moves).
769    #[tokio::test]
770    async fn p23w3_07_metadata_is_cached_until_it_expires_or_the_configuration_changes() {
771        let cache = SamlMetadataCache::new();
772        let mut config = config();
773        let doc = document("https://sso.example.test/sso", r#"cacheDuration="PT1H""#);
774        let calls = AtomicUsize::new(0);
775        let now = Utc::now();
776
777        let first = cache
778            .resolve(&config, now, serving(doc.clone(), &calls))
779            .await
780            .unwrap();
781        assert_eq!(first.metadata.sso_url, "https://sso.example.test/sso");
782        assert!(first.sso_host_change.is_none());
783        assert_eq!(calls.load(Ordering::SeqCst), 1);
784
785        // A hit: no second fetch within the hour.
786        let hit = cache
787            .resolve(
788                &config,
789                now + Duration::minutes(59),
790                serving(doc.clone(), &calls),
791            )
792            .await
793            .unwrap();
794        assert_eq!(hit.metadata.sso_url, first.metadata.sso_url);
795        assert_eq!(calls.load(Ordering::SeqCst), 1, "served from the cache");
796
797        // Expired: fetched again.
798        cache
799            .resolve(
800                &config,
801                now + Duration::hours(1),
802                serving(doc.clone(), &calls),
803            )
804            .await
805            .unwrap();
806        assert_eq!(calls.load(Ordering::SeqCst), 2, "refetched after expiry");
807
808        // An edited configuration misses at once.
809        config.updated_at += Duration::seconds(1);
810        cache
811            .resolve(
812                &config,
813                now + Duration::hours(1),
814                serving(doc.clone(), &calls),
815            )
816            .await
817            .unwrap();
818        assert_eq!(calls.load(Ordering::SeqCst), 3, "an edit invalidates");
819
820        // So does a new metadata URL.
821        config.metadata_url = Some("https://idp.example.test/other".into());
822        cache
823            .resolve(&config, now + Duration::hours(1), serving(doc, &calls))
824            .await
825            .unwrap();
826        assert_eq!(calls.load(Ordering::SeqCst), 4);
827    }
