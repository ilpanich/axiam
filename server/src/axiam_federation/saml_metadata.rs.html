1//! SAML SP: the IdP's metadata document — its signature and its cache (#530,
2//! P23W3-07).
3//!
4//! The SP reads three things from an IdP's metadata: its entity ID, its SSO
5//! URL and the binding of that URL. The assertion signing certificate is
6//! **not** among them (it is pinned on the federation configuration), so forged
7//! metadata cannot forge an assertion. It can still choose the HTTPS URL a
8//! person is redirected to with the `AuthnRequest` — a phishing redirect — and
9//! before this module every SP-initiated sign-in fetched the document again, so
10//! every sign-in also depended on the metadata host being up.
11//!
12//! # Signature
13//!
14//! When the configuration carries `idp_metadata_signing_cert_pem`, the
15//! document must be signed, and only one place may hold the signature: the
16//! enveloped child of the `md:EntityDescriptor` root, with one reference,
17//! naming the root's `ID` — the IdP receiver's placement rule
18//! ([`signature_placement`], D-23), applied to a different root. Any other
19//! `ds:Signature` refuses the document. The one admitted signature is verified
20//! by xmlsec on its own node against the configured certificate, with the SHA-2
21//! algorithms only ([`ALLOWED_XML_SIGNATURE_ALGORITHMS`]; a configuration's
22//! `allow_sha1_signatures` applies to responses, not to metadata), and what is
23//! then read is xmlsec's **pre-digest** output — the bytes the digest covered —
24//! not the document as fetched, so nothing outside the signature is read.
25//!
26//! Every document, signed or not, is first refused on its bytes if it carries a
27//! markup declaration or an encoding other than UTF-8, as a SAML response is
28//! (#531): metadata has no use for a DTD.
29//!
30//! # Cache
31//!
32//! [`SamlMetadataCache`] holds the parsed document per process, keyed by the
33//! federation configuration's id and checked against the metadata URL and the
34//! configuration's `updated_at`, so any edit of the configuration — a new URL,
35//! a new certificate — misses. An entry lives for the document's
36//! `cacheDuration` ([`DEFAULT_TTL`] when it states none), held between
37//! [`MIN_TTL`] and [`MAX_TTL`], and never past its `validUntil`. A document
38//! whose `validUntil` has passed is refused.
39//!
40//! A refetch whose SSO URL names another host than the entry it replaces is
41//! reported ([`SsoHostChange`]) for the caller to audit: the metadata host can
42//! move users, so a move should be visible.
43

62/// SAML 2.0 metadata namespace.
63const NS_METADATA: &str = "urn:oasis:names:tc:SAML:2.0:metadata";

65/// The largest metadata document read: 512 KiB.
66pub const MAX_METADATA_SIZE: usize = 512 * 1024;
67
68/// How long a document that states no `cacheDuration` is kept: one hour, the
69/// OIDC discovery cache's TTL.
70pub const DEFAULT_TTL: Duration = Duration::hours(1);
71
72/// The shortest time an entry is kept, whatever `cacheDuration` says — so a
73/// document stating `PT0S` does not turn the cache off — unless its
74/// `validUntil` comes first.
75pub const MIN_TTL: Duration = Duration::minutes(5);
76
77/// The longest time an entry is kept, whatever `cacheDuration` says: a moved
78/// or re-keyed IdP is picked up within a day without an edit.
79pub const MAX_TTL: Duration = Duration::hours(24);

94/// Fetch a metadata document (no parsing, no cache).
95///
96/// # Errors
97///
98/// [`FederationError::InvalidMetadataUrl`] for a URL that is not HTTPS;
99/// [`FederationError::SamlMetadataFailed`] for a refused or failed fetch, a
100/// non-2xx answer, a body over [`MAX_METADATA_SIZE`] or one that is not UTF-8.
101pub async fn fetch_metadata_document(metadata_url: String) -> Result<String, FederationError> {
102    crate::validate_metadata_url(&metadata_url)?;
103
104    // SECHRD-02: route the metadata GET through the shared, IP-pinning SSRF
105    // guard (D-01a/b/c). Production always fails closed against
106    // private/loopback/link-local addresses and internal redirect targets —
107    // `allow_private=false`.
108    let response = crate::ssrf::guarded_fetch(&metadata_url, false, |c, u| c.get(u))
109        .await
110        .map_err(|e| FederationError::SamlMetadataFailed(e.to_string()))?;
111
112    if !response.status().is_success() {
113        return Err(FederationError::SamlMetadataFailed(format!(
114            "HTTP {} from metadata endpoint",
115            response.status()
116        )));
117    }
118
119    let bytes = response.bytes().await.map_err(|e| {
120        FederationError::SamlMetadataFailed(format!("Failed to read metadata body: {e}"))
121    })?;
122    if bytes.len() > MAX_METADATA_SIZE {
123        return Err(FederationError::SamlMetadataFailed(format!(
124            "Metadata document too large: {} bytes (max {MAX_METADATA_SIZE})",
125            bytes.len(),
126        )));
127    }
128
129    String::from_utf8(bytes.to_vec())
130        .map_err(|e| FederationError::SamlMetadataFailed(format!("Invalid UTF-8 in metadata: {e}")))
131}

133/// A metadata document, verified when a certificate was given, and read.
134#[derive(Debug, Clone)]
135pub struct ParsedIdpMetadata {
136    /// What the SP uses.
137    pub metadata: IdpMetadata,
138    /// `validUntil`, when the document states one (still in the future).
139    pub valid_until: Option<DateTime<Utc>>,
140    /// `cacheDuration`, when the document states one AXIAM can read.
141    pub cache_duration: Option<Duration>,
142}
143
144impl ParsedIdpMetadata {
145    /// When a cache entry made from this document at `now` expires:
146    /// `cacheDuration` (or [`DEFAULT_TTL`]) held to [`MIN_TTL`]..=[`MAX_TTL`],
147    /// and never later than `validUntil`.
148    #[must_use]
149    pub fn expires_at(&self, now: DateTime<Utc>) -> DateTime<Utc> {
150        let ttl = self
151            .cache_duration
152            .unwrap_or(DEFAULT_TTL)
153            .clamp(MIN_TTL, MAX_TTL);
154        let by_ttl = now + ttl;
155        self.valid_until.map_or(by_ttl, |until| by_ttl.min(until))
156    }
157}

163/// Read an IdP metadata document; with `signing_cert_pem`, only after its
164/// signature verified (see the module documentation for the rule).
165///
166/// # Errors
167///
168/// [`FederationError::SamlMetadataFailed`]: a markup declaration or another
169/// encoding; with a certificate, an unsigned document, a signature anywhere but
170/// on the `EntityDescriptor` root, or one that does not verify; a document
171/// whose `validUntil` is not after `now`; no `EntityDescriptor`, `entityID`,
172/// `IDPSSODescriptor` or HTTPS SSO endpoint on either binding.
173/// [`FederationError::InvalidIdpCert`] for a certificate that is not PEM.
174pub fn parse_idp_metadata(
175    text: &str,
176    signing_cert_pem: Option<&str>,
177    now: DateTime<Utc>,
178) -> Result<ParsedIdpMetadata, FederationError> {
179    // #531's rule for SAML responses, on the bytes and before any parser.
180    refuse_other_encodings(text).map_err(|_| metadata_failed("metadata is not plain UTF-8"))?;
181    refuse_markup_declarations(text)
182        .map_err(|_| metadata_failed("metadata declares a DTD or an entity"))?;
183
184    let verified;
185    let readable = match signing_cert_pem {
186        Some(pem) => {
187            verified = verify_metadata_signature(text, pem)?;
188            verified.as_str()
189        }
190        None => text,
191    };
192
193    let descriptor_type: EntityDescriptorType = readable
194        .parse()
195        .map_err(|e| metadata_failed(format!("Failed to parse metadata XML: {e}")))?;
196
197    // A signed document is one `EntityDescriptor` (the root the signature is
198    // on); an unsigned one may be an aggregate, whose first entity is read.
199    let (outer_valid_until, outer_cache_duration) = match &descriptor_type {
200        EntityDescriptorType::EntitiesDescriptor(all) => {
201            (all.valid_until, all.cache_duration.clone())
202        }
203        EntityDescriptorType::EntityDescriptor(_) => (None, None),
204    };
205    let ed = descriptor_type
206        .iter()
207        .next()
208        .ok_or_else(|| metadata_failed("No EntityDescriptor found in metadata"))?;
209
210    let valid_until = match (ed.valid_until, outer_valid_until) {
211        (Some(a), Some(b)) => Some(a.min(b)),
212        (a, b) => a.or(b),
213    };
214    if valid_until.is_some_and(|until| until <= now) {
215        return Err(metadata_failed("metadata validUntil has passed"));
216    }
217    let cache_duration = ed
218        .cache_duration
219        .as_deref()
220        .or(outer_cache_duration.as_deref())
221        .and_then(parse_xs_duration);
222
223    let entity_id = ed
224        .entity_id
225        .clone()
226        .ok_or_else(|| metadata_failed("EntityDescriptor missing entityID"))?;
227    let idp = ed
228        .idp_sso_descriptors
229        .as_ref()
230        .and_then(|d| d.first())
231        .ok_or_else(|| metadata_failed("No IDPSSODescriptor in metadata"))?;
232
233    // Prefer HTTP-POST binding, fall back to HTTP-Redirect.
234    let sso_endpoint = idp
235        .single_sign_on_services
236        .iter()
237        .find(|ep| ep.binding == HTTP_POST_BINDING)
238        .or_else(|| {
239            idp.single_sign_on_services
240                .iter()
241                .find(|ep| ep.binding == HTTP_REDIRECT_BINDING)
242        })
243        .ok_or_else(|| metadata_failed("No HTTP-POST or HTTP-Redirect SSO endpoint"))?;
244
245    // The SSO endpoint must be HTTPS, so a person is never sent to an insecure
246    // origin. Fail closed on a URL that does not parse.
247    let sso_parsed = url::Url::parse(&sso_endpoint.location)
248        .map_err(|e| metadata_failed(format!("IdP SSO endpoint is not a valid URL: {e}")))?;
249    if sso_parsed.scheme() != "https" {
250        return Err(metadata_failed("IdP SSO endpoint must use HTTPS"));
251    }
252
253    Ok(ParsedIdpMetadata {
254        metadata: IdpMetadata {
255            entity_id,
256            sso_url: sso_endpoint.location.clone(),
257            sso_binding: sso_endpoint.binding.clone(),
258        },
259        valid_until,
260        cache_duration,
261    })
262}

264/// Verify the metadata's one enveloped signature and return the bytes it
265/// covers (xmlsec's pre-digest output: the `EntityDescriptor`, canonicalized,
266/// without its signature).
267fn verify_metadata_signature(text: &str, pem: &str) -> Result<String, FederationError> {
268    let der = crate::cert::pem_cert_to_der(pem)?;
269
270    let doc = libxml::parser::Parser::default()
271        .parse_string(text.as_bytes())
272        .map_err(|_| metadata_failed("metadata is not XML"))?;
273    let root = doc
274        .get_root_element()
275        .ok_or_else(|| metadata_failed("metadata has no root element"))?;
276    let root_is_entity = root.get_name() == "EntityDescriptor"
277        && root
278            .get_namespace()
279            .is_some_and(|ns| ns.get_href() == NS_METADATA);
280    if !root_is_entity {
281        return Err(metadata_failed(
282            "signed metadata must be one md:EntityDescriptor",
283        ));
284    }
285    let root_id = root
286        .get_attribute("ID")
287        .filter(|id| !id.is_empty())
288        .ok_or_else(|| metadata_failed("metadata is not signed (its root has no ID)"))?;
289    // The reference names the root by `ID`; a second element carrying the same
290    // value would leave xmlsec and this check free to mean different nodes.
291    let mut context = libxml::xpath::Context::new(&doc)
292        .map_err(|()| metadata_failed("metadata: XPath context failed"))?;
293    let bearers = context
294        .findnodes("//*[@ID]", None)
295        .map_err(|()| metadata_failed("metadata: XPath evaluation failed"))?;
296    if bearers
297        .iter()
298        .filter(|n| n.get_attribute("ID").as_deref() == Some(root_id.as_str()))
299        .count()
300        != 1
301    {
302        return Err(metadata_failed("metadata repeats its root ID"));
303    }
304
305    match signature_placement(&doc, &root, &root_id) {
306        Ok(true) => {}
307        Ok(false) => return Err(metadata_failed("metadata is not signed")),
308        Err(_) => {
309            return Err(metadata_failed(
310                "a metadata signature may only be the EntityDescriptor root's own",
311            ));
312        }
313    }
314
315    <XmlSec as CryptoProvider>::reduce_xml_to_signed_with_allowed_algorithms(
316        text,
317        &[CertificateDer::from(der)],
318        ReduceMode::PreDigest,
319        Some(&ALLOWED_XML_SIGNATURE_ALGORITHMS),
320    )
321    .map_err(|_| metadata_failed("metadata signature does not verify"))
322}

324/// An `xs:duration` (`PnYnMnDTnHnMnS`, seconds may be fractional) as a
325/// [`Duration`]. A year counts 365 days and a month 30: the result is only
326/// ever compared with [`MAX_TTL`]. `None` for a negative or malformed value,
327/// which then reads as "no `cacheDuration`".
328#[must_use]
329pub fn parse_xs_duration(raw: &str) -> Option<Duration> {
330    let rest = raw.trim().strip_prefix('P')?;
331    let (date, time) = match rest.split_once('T') {
332        Some((d, t)) if !t.is_empty() => (d, Some(t)),
333        Some(_) => return None,
334        None => (rest, None),
335    };
336    if date.is_empty() && time.is_none() {
337        return None;
338    }
339    let mut millis: i64 = 0;
340    let mut add = |digits: &str, unit_ms: i64| -> Option<()> {
341        let value = if let Some((whole, frac)) = digits.split_once('.') {
342            if unit_ms != 1_000 || whole.is_empty() {
343                return None;
344            }
345            let whole: i64 = whole.parse().ok()?;
346            let frac_ms: i64 = format!("{frac:0<3}")[..3].parse().ok()?;
347            whole.checked_mul(1_000)?.checked_add(frac_ms)?
348        } else {
349            digits.parse::<i64>().ok()?.checked_mul(unit_ms)?
350        };
351        millis = millis.checked_add(value)?;
352        Some(())
353    };
354    let mut walk = |part: &str, units: &[(char, i64)]| -> Option<()> {
355        let mut digits = String::new();
356        let mut next_unit = 0;
357        for c in part.chars() {
358            if c.is_ascii_digit() || c == '.' {
359                digits.push(c);
360                continue;
361            }
362            let at = units[next_unit..].iter().position(|(u, _)| *u == c)? + next_unit;
363            if digits.is_empty() {
364                return None;
365            }
366            add(&digits, units[at].1)?;
367            digits.clear();
368            next_unit = at + 1;
369        }
370        digits.is_empty().then_some(())
371    };
372    const DAY: i64 = 86_400_000;
373    walk(date, &[('Y', 365 * DAY), ('M', 30 * DAY), ('D', DAY)])?;
374    if let Some(time) = time {
375        walk(time, &[('H', 3_600_000), ('M', 60_000), ('S', 1_000)])?;
376    }
377    Some(Duration::milliseconds(millis))
378}

389/// Metadata as the SP uses it, and whether this resolution moved the SSO host.
390#[derive(Debug, Clone)]
391pub struct ResolvedIdpMetadata {
392    /// The parsed document (from the cache or just fetched).
393    pub metadata: IdpMetadata,
394    /// `Some` only on a fetch whose SSO URL names another host than the entry
395    /// it replaced in this process.
396    pub sso_host_change: Option<SsoHostChange>,
397}

399#[derive(Debug, Clone)]
400struct CacheEntry {
401    metadata_url: String,
402    config_version: DateTime<Utc>,
403    metadata: IdpMetadata,
404    expires_at: DateTime<Utc>,
405}

407/// Per-process cache of parsed IdP metadata, keyed by federation
408/// configuration. Cheap to clone; clones share the map.
409#[derive(Clone, Default)]
410pub struct SamlMetadataCache(Arc<Mutex<HashMap<Uuid, CacheEntry>>>);
411
412impl SamlMetadataCache {
413    /// An empty cache.
414    #[must_use]
415    pub fn new() -> Self {
416        Self::default()
417    }
418
419    fn entries(&self) -> std::sync::MutexGuard<'_, HashMap<Uuid, CacheEntry>> {
420        // A panic while the lock was held leaves a map of whole entries; keep
421        // using it rather than failing every sign-in.
422        self.0
423            .lock()
424            .unwrap_or_else(std::sync::PoisonError::into_inner)
425    }
426
427    /// The configuration's IdP metadata: the cached entry when it was made
428    /// from this configuration (same id, metadata URL and `updated_at`) and has
429    /// not expired at `now`, otherwise `fetch`ed, verified, parsed and stored.
430    ///
431    /// The lock is never held across `fetch`: two misses at once both fetch,
432    /// and the later store wins.
433    ///
434    /// # Errors
435    ///
436    /// [`FederationError::SamlMetadataFailed`] when the configuration has no
437    /// metadata URL; otherwise whatever `fetch` or [`parse_idp_metadata`]
438    /// returns. A failure stores nothing and leaves an earlier entry in place.
439    pub async fn resolve<F, Fut>(
440        &self,
441        config: &FederationConfig,
442        now: DateTime<Utc>,
443        fetch: F,
444    ) -> Result<ResolvedIdpMetadata, FederationError>
445    where
446        F: FnOnce(String) -> Fut,
447        Fut: Future<Output = Result<String, FederationError>>,
448    {
449        let metadata_url = config
450            .metadata_url
451            .as_deref()
452            .ok_or_else(|| metadata_failed("No metadata URL configured"))?;
453
454        if let Some(entry) = self.entries().get(&config.id)
455            && entry.metadata_url == metadata_url
456            && entry.config_version == config.updated_at
457            && now < entry.expires_at
458        {
459            return Ok(ResolvedIdpMetadata {
460                metadata: entry.metadata.clone(),
461                sso_host_change: None,
462            });
463        }
464
465        let text = fetch(metadata_url.to_owned()).await?;
466        let parsed =
467            parse_idp_metadata(&text, config.idp_metadata_signing_cert_pem.as_deref(), now)?;
468        let entry = CacheEntry {
469            metadata_url: metadata_url.to_owned(),
470            config_version: config.updated_at,
471            metadata: parsed.metadata.clone(),
472            expires_at: parsed.expires_at(now),
473        };
474        let previous = self.entries().insert(config.id, entry);
475
476        let new_host = sso_host(&parsed.metadata.sso_url);
477        let sso_host_change = previous
478            .map(|p| sso_host(&p.metadata.sso_url))
479            .filter(|old_host| *old_host != new_host)
480            .map(|old_host| SsoHostChange { old_host, new_host });
481        Ok(ResolvedIdpMetadata {
482            metadata: parsed.metadata,
483            sso_host_change,
484        })
485    }
486}

488/// The host of an SSO URL `parse_idp_metadata` accepted (so it parses).
489fn sso_host(sso_url: &str) -> String {
490    url::Url::parse(sso_url)
491        .ok()
492        .and_then(|u| u.host_str().map(str::to_ascii_lowercase))
493        .unwrap_or_default()
494}

620    /// #530: a second sign-in within the document's freshness is served from
621    /// the cache; it is fetched again once the entry expires, and at once when
622    /// the configuration is edited (its `updated_at` moves).
623    #[tokio::test]
624    async fn p23w3_07_metadata_is_cached_until_it_expires_or_the_configuration_changes() {
625        let cache = SamlMetadataCache::new();
626        let mut config = config();
627        let doc = document("https://sso.example.test/sso", r#"cacheDuration="PT1H""#);
628        let calls = AtomicUsize::new(0);
629        let now = Utc::now();
630
631        let first = cache
632            .resolve(&config, now, serving(doc.clone(), &calls))
633            .await
634            .unwrap();
635        assert_eq!(first.metadata.sso_url, "https://sso.example.test/sso");
636        assert!(first.sso_host_change.is_none());
637        assert_eq!(calls.load(Ordering::SeqCst), 1);
638
639        // A hit: no second fetch within the hour.
640        let hit = cache
641            .resolve(
642                &config,
643                now + Duration::minutes(59),
644                serving(doc.clone(), &calls),
645            )
646            .await
647            .unwrap();
648        assert_eq!(hit.metadata.sso_url, first.metadata.sso_url);
649        assert_eq!(calls.load(Ordering::SeqCst), 1, "served from the cache");
650
651        // Expired: fetched again.
652        cache
653            .resolve(
654                &config,
655                now + Duration::hours(1),
656                serving(doc.clone(), &calls),
657            )
658            .await
659            .unwrap();
660        assert_eq!(calls.load(Ordering::SeqCst), 2, "refetched after expiry");
661
662        // An edited configuration misses at once.
663        config.updated_at += Duration::seconds(1);
664        cache
665            .resolve(
666                &config,
667                now + Duration::hours(1),
668                serving(doc.clone(), &calls),
669            )
670            .await
671            .unwrap();
672        assert_eq!(calls.load(Ordering::SeqCst), 3, "an edit invalidates");
673
674        // So does a new metadata URL.
675        config.metadata_url = Some("https://idp.example.test/other".into());
676        cache
677            .resolve(&config, now + Duration::hours(1), serving(doc, &calls))
678            .await
679            .unwrap();
680        assert_eq!(calls.load(Ordering::SeqCst), 4);
681    }
