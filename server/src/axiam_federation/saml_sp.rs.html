1//! Write-time rules for the SAML service-provider registry (G-2, T23.2.1).
2//!
3//! One pure function, [`validate_saml_service_provider`], decides whether a
4//! [`SamlServiceProviderInput`] may be stored. It lives here and not in
5//! `axiam-core` because it needs URL and X.509 parsing, which layer 0 does not
6//! carry; and it is **not** behind the `saml` feature, because it parses no
7//! XML and the registry is plain data (the SAML protocol code that uses it is
8//! what the feature gates).
9//!
10//! # The ACS allow-list is a redirect-URI registration
11//!
12//! An `AssertionConsumerServiceURL` is where AXIAM sends a signed assertion, so
13//! it is a redirect target in the OAuth2 sense: the same attacks (an assertion
14//! delivered to an attacker's endpoint) and therefore the same registration
15//! rules. [`check_redirect_uri_registration`] is that rule, and it is **the one
16//! copy**: the admin API's `validate_redirect_uris`
17//! (`axiam-api-rest`, layer 6) calls it rather than carrying its own, so the two
18//! can never disagree about what a registrable redirect is. On top of it the ACS
19//! list refuses any `*`, because the allow-list is exact strings and a wildcard
20//! would be a glob by another name (the URL parser admits `*` in a host).
21//!
22//! The same URL rule is applied to `slo_url`.
23

34/// Structural rules for a redirect URI, shared by the admin OAuth2 client
35/// registration API, `POST /oauth2/register` and the SAML ACS allow-list.
36///
37/// Returns the refusal text on failure. The strings are the ones the admin API
38/// has always answered with; callers wrap them in their own error type.
39///
40/// * absolute, with a host;
41/// * `https`, except `http` for the loopback hosts `localhost`, `127.0.0.1` and
42///   `[::1]` (`Url::host_str` spells an IPv6 literal with its brackets);
43/// * no fragment (RFC 6749 §3.1.2).
44///
45/// # Errors
46///
47/// A human-readable message naming the offending URI.
48pub fn check_redirect_uri_registration(uri: &str) -> Result<(), String> {
49    let parsed: url::Url = uri
50        .parse()
51        .map_err(|_| format!("invalid redirect_uri: {uri}"))?;
52    // Redirect URIs must be absolute with an authority (host)
53    let host = parsed
54        .host_str()
55        .ok_or_else(|| format!("redirect_uri must be an absolute URL with a host: {uri}"))?;
56    // Allow http for localhost/loopback only, require HTTPS otherwise.
57    //
58    // `[::1]`, with the brackets, is what `Url::host_str` returns for an IPv6
59    // literal. The rest of the stack always spelled it with brackets: the
60    // redirect matcher's loopback arm (`axiam_oauth2::redirect_uri`), the DCR
61    // host allow-list and the CIMD document validator all do. RFC 8252 §7.3
62    // lists the IPv6 loopback beside `127.0.0.1`.
63    let is_localhost = host == "localhost" || host == "127.0.0.1" || host == "[::1]";
64    if parsed.scheme() != "https" && !(parsed.scheme() == "http" && is_localhost) {
65        return Err(format!(
66            "redirect_uri must use https (http is only allowed for localhost/127.0.0.1/[::1]): {uri}"
67        ));
68    }
69    // RFC 6749 §3.1.2: redirect URIs must not include a fragment
70    if parsed.fragment().is_some() {
71        return Err(format!("redirect_uri must not contain a fragment: {uri}"));
72    }
73    Ok(())
74}

76/// Why a certificate PEM was refused. Never carries the PEM itself.
77#[derive(Debug, Clone, Copy, PartialEq, Eq)]
78enum CertDefect {
79    TooLarge,
80    NotPem,
81    NotCertificateBlock,
82    MoreThanOneBlock,
83    Unparseable,
84}
85
86impl CertDefect {
87    fn describe(self) -> &'static str {
88        match self {
89            Self::TooLarge => "is larger than the permitted size",
90            Self::NotPem => "is not PEM",
91            Self::NotCertificateBlock => {
92                "is not a CERTIFICATE block (a private key must never be stored here)"
93            }
94            Self::MoreThanOneBlock => "holds more than one PEM block; exactly one is required",
95            Self::Unparseable => "is not a parseable X.509 certificate",
96        }
97    }
98}

100/// Exactly one `CERTIFICATE` PEM block that parses as X.509.
101///
102/// A private key, a bundle, trailing blocks and DER-as-text are all refused:
103/// the registry holds public material only.
104fn check_certificate_pem(pem: &str) -> Result<(), CertDefect> {
105    if pem.len() > MAX_SP_CERT_PEM_BYTES {
106        return Err(CertDefect::TooLarge);
107    }
108    let (rest, block) =
109        x509_parser::pem::parse_x509_pem(pem.as_bytes()).map_err(|_| CertDefect::NotPem)?;
110    if block.label != "CERTIFICATE" {
111        return Err(CertDefect::NotCertificateBlock);
112    }
113    if !rest.iter().all(u8::is_ascii_whitespace) {
114        return Err(CertDefect::MoreThanOneBlock);
115    }
116    X509Certificate::from_der(&block.contents).map_err(|_| CertDefect::Unparseable)?;
117    Ok(())
118}

120/// A name that is non-empty, within bounds, free of control characters and not
121/// padded with whitespace.
122fn check_plain_text(label: &str, value: &str, max_bytes: usize, out: &mut Vec<String>) {
123    if value.trim().is_empty() {
124        out.push(format!("{label} must not be empty"));
125    } else if value.len() > max_bytes {
126        out.push(format!("{label} must be at most {max_bytes} bytes"));
127    } else if value != value.trim() {
128        out.push(format!("{label} must not start or end with whitespace"));
129    } else if value.chars().any(char::is_control) {
130        out.push(format!("{label} must not contain control characters"));
131    }
132}

134/// Every reason `input` may not be stored, in a stable order. Empty means it
135/// may.
136#[must_use]
137pub fn saml_sp_violations(input: &SamlServiceProviderInput) -> Vec<String> {
138    let mut v = Vec::new();
139
140    check_plain_text(
141        "display_name",
142        &input.display_name,
143        MAX_DISPLAY_NAME_BYTES,
144        &mut v,
145    );
146    check_plain_text("entity_id", &input.entity_id, MAX_ENTITY_ID_BYTES, &mut v);
147
148    // --- ACS allow-list ---
149    if input.acs_urls.is_empty() {
150        v.push("acs_urls must contain at least one endpoint".into());
151    }
152    if input.acs_urls.len() > MAX_ACS_ENDPOINTS {
153        v.push(format!(
154            "acs_urls may hold at most {MAX_ACS_ENDPOINTS} endpoints"
155        ));
156    }
157    let mut seen_urls = HashSet::new();
158    let mut seen_indexes = HashSet::new();
159    for (position, endpoint) in input.acs_urls.iter().enumerate() {
160        if let Err(msg) = check_redirect_uri_registration(&endpoint.url) {
161            v.push(format!("acs_urls[{position}]: {msg}"));
162        } else if endpoint.url.contains('*') {
163            v.push(format!(
164                "acs_urls[{position}]: wildcards are not allowed; an ACS URL is matched exactly"
165            ));
166        }
167        if !seen_urls.insert(endpoint.url.as_str()) {
168            v.push(format!(
169                "acs_urls[{position}]: duplicate ACS URL (already registered above)"
170            ));
171        }
172        if !seen_indexes.insert(endpoint.index) {
173            v.push(format!(
174                "acs_urls[{position}]: duplicate index {} (an AssertionConsumerServiceIndex must \
175                 name one endpoint)",
176                endpoint.index
177            ));
178        }
179    }
180    if input.acs_urls.iter().filter(|e| e.is_default).count() > 1 {
181        v.push("acs_urls may mark at most one endpoint as the default".into());
182    }
183
184    // --- single logout ---
185    match (&input.slo_url, &input.slo_binding) {
186        (Some(url), Some(_)) => {
187            if let Err(msg) = check_redirect_uri_registration(url) {
188                v.push(format!("slo_url: {msg}"));
189            } else if url.contains('*') {
190                v.push("slo_url: wildcards are not allowed".into());
191            }
192        }
193        (None, None) => {}
194        (Some(_), None) => v.push("slo_binding is required when slo_url is set".into()),
195        (None, Some(_)) => v.push("slo_binding must not be set without slo_url".into()),
196    }
197
198    // --- certificates ---
199    for (label, cert) in [
200        ("sp_signing_cert_pem", &input.sp_signing_cert_pem),
201        ("sp_encryption_cert_pem", &input.sp_encryption_cert_pem),
202    ] {
203        if let Some(pem) = cert
204            && let Err(defect) = check_certificate_pem(pem)
205        {
206            v.push(format!("{label} {}", defect.describe()));
207        }
208    }
209    if input.encrypt_assertions && input.sp_encryption_cert_pem.is_none() {
210        v.push("encrypt_assertions requires sp_encryption_cert_pem".into());
211    }
212    if input.want_authn_requests_signed && input.sp_signing_cert_pem.is_none() {
213        v.push("want_authn_requests_signed requires sp_signing_cert_pem".into());
214    }
215
216    // --- attribute mappings ---
217    if input.attribute_mappings.len() > MAX_ATTRIBUTE_MAPPINGS {
218        v.push(format!(
219            "attribute_mappings may hold at most {MAX_ATTRIBUTE_MAPPINGS} entries"
220        ));
221    }
222    let mut seen_names = HashSet::new();
223    for (position, mapping) in input.attribute_mappings.iter().enumerate() {
224        let label = format!("attribute_mappings[{position}].saml_name");
225        check_plain_text(&label, &mapping.saml_name, MAX_ATTRIBUTE_NAME_BYTES, &mut v);
226        if !seen_names.insert(mapping.saml_name.as_str()) {
227            v.push(format!(
228                "attribute_mappings[{position}]: duplicate attribute name (names are \
229                 case-sensitive and must be unique)"
230            ));
231        }
232        if let Some(format) = &mapping.name_format
233            && !ATTRIBUTE_NAME_FORMATS.contains(&format.as_str())
234        {
235            v.push(format!(
236                "attribute_mappings[{position}].name_format is not a SAML attribute name format"
237            ));
238        }
239    }
240
241    // --- group restriction ---
242    if input.allowed_groups.len() > MAX_ALLOWED_GROUPS {
243        v.push(format!(
244            "allowed_groups may name at most {MAX_ALLOWED_GROUPS} groups"
245        ));
246    }
247
248    v
249}

251/// Whether `input` may be stored.
252///
253/// # Errors
254///
255/// [`AxiamError::Validation`] carrying every violation, joined by `; `.
256pub fn validate_saml_service_provider(input: &SamlServiceProviderInput) -> AxiamResult<()> {
257    let violations = saml_sp_violations(input);
258    if violations.is_empty() {
259        Ok(())
260    } else {
261        Err(AxiamError::Validation {
262            message: violations.join("; "),
263        })
264    }
265}

271/// The smallest RSA modulus an SP's request-signing certificate may carry, in
272/// bits.
273pub const MIN_SP_RSA_BITS: usize = 2048;
274
275/// NIST curves an SP's ECDSA request-signing certificate may use, by OID: P-256,
276/// P-384, P-521 (the three xmlsec verifies with `ecdsa-sha256/384/512`).
277const SP_EC_CURVE_OIDS: [&str; 3] = ["1.2.840.10045.3.1.7", "1.3.132.0.34", "1.3.132.0.35"];
278
279/// Why the SSO endpoint could not use `pem` as an SP's request-signing
280/// certificate, or `None` when it can. The text names the rule and never the
281/// certificate.
282///
283/// *Could not use* means one of: its decoder, [`crate::cert::pem_cert_to_der`] —
284/// the very function the SSO endpoint calls, so the two cannot disagree — refuses
285/// it; it is not X.509; or its public key is one no verifier AXIAM runs can use:
286/// RSA under [`MIN_SP_RSA_BITS`] bits, or anything but RSA or ECDSA on P-256,
287/// P-384 or P-521. (An ECDSA certificate verifies HTTP-POST requests only; the
288/// HTTP-Redirect binding is RSA-only, which contract §29 says.) A certificate's
289/// validity dates are **not** judged: SAML trusts a registered key as a key.
290#[must_use]
291pub fn sp_signing_certificate_refusal(pem: &str) -> Option<String> {
292    const FIELD: &str = "sp_signing_cert_pem";
293    let Ok(der) = crate::cert::pem_cert_to_der(pem) else {
294        return Some(format!(
295            "{FIELD} is not a certificate the SSO endpoint can decode"
296        ));
297    };
298    let Ok((_, cert)) = X509Certificate::from_der(&der) else {
299        return Some(format!("{FIELD} is not a parseable X.509 certificate"));
300    };
301    let spki = cert.public_key();
302    match spki.parsed() {
303        Ok(x509_parser::public_key::PublicKey::RSA(rsa)) => {
304            // Bits of the modulus, without the DER sign byte or leading zeros.
305            let significant: &[u8] = rsa
306                .modulus
307                .iter()
308                .position(|b| *b != 0)
309                .map_or(&[][..], |at| &rsa.modulus[at..]);
310            let bits = significant.first().map_or(0, |top| {
311                significant.len() * 8 - top.leading_zeros() as usize
312            });
313            (bits < MIN_SP_RSA_BITS)
314                .then(|| format!("{FIELD}: an RSA key must be at least {MIN_SP_RSA_BITS} bits"))
315        }
316        Ok(x509_parser::public_key::PublicKey::EC(_)) => {
317            let curve = spki
318                .algorithm
319                .parameters
320                .as_ref()
321                .and_then(|p| p.as_oid().ok())
322                .map(|oid| oid.to_id_string());
323            curve
324                .is_none_or(|oid| !SP_EC_CURVE_OIDS.contains(&oid.as_str()))
325                .then(|| format!("{FIELD}: an ECDSA key must be on P-256, P-384 or P-521"))
326        }
327        _ => Some(format!(
328            "{FIELD}: the key must be RSA (at least {MIN_SP_RSA_BITS} bits) or ECDSA on \
329             P-256, P-384 or P-521"
330        )),
331    }
332}

334/// The write-time refusals of D-42 that need no database, each naming its rule
335/// — beyond [`saml_sp_violations`], which the caller runs first:
336///
337/// * `encrypt_assertions: true` — assertion encryption is not implemented (D-2);
338///   an SP asking for it would be refused at every sign-on, so it is refused
339///   here, where the administrator can read why;
340/// * an `sp_signing_cert_pem` the SSO endpoint cannot use
341///   ([`sp_signing_certificate_refusal`]).
342///
343/// The other two — an `allowed_groups` entry outside the tenant and a changed
344/// `entity_id` — need the datastore and the stored row, and are the route's.
345#[must_use]
346pub fn saml_sp_write_refusals(input: &SamlServiceProviderInput) -> Vec<String> {
347    let mut refusals = Vec::new();
348    if input.encrypt_assertions {
349        refusals.push(
350            "encrypt_assertions: assertion encryption is not supported yet; leave it false"
351                .to_string(),
352        );
353    }
354    if let Some(pem) = input.sp_signing_cert_pem.as_deref() {
355        refusals.extend(sp_signing_certificate_refusal(pem));
356    }
357    refusals
358}
359
360/// [`validate_saml_service_provider`], then [`saml_sp_write_refusals`]: every
361/// rule a registry write must satisfy that needs neither the datastore nor the
362/// stored row.
363///
364/// # Errors
365///
366/// [`AxiamError::Validation`] carrying the violations of the first of the two
367/// that has any, joined by `; `. The message never echoes a certificate.
368pub fn validate_saml_service_provider_write(input: &SamlServiceProviderInput) -> AxiamResult<()> {
369    validate_saml_service_provider(input)?;
370    let refusals = saml_sp_write_refusals(input);
371    if refusals.is_empty() {
372        Ok(())
373    } else {
374        Err(AxiamError::Validation {
375            message: refusals.join("; "),
376        })
377    }
378}

815    fn ec_pem(algorithm: &'static rcgen::SignatureAlgorithm) -> String {
816        let key = rcgen::KeyPair::generate_for(algorithm).expect("key pair");
817        rcgen::CertificateParams::new(vec!["sp.example.test".to_string()])
818            .expect("params")
819            .self_signed(&key)
820            .expect("self-signed")
821            .pem()
822    }
823
824    /// A self-signed certificate over an EC key on `curve`, by OpenSSL (rcgen
825    /// here has no P-521).
826    fn openssl_ec_pem(curve: openssl::nid::Nid) -> String {
827        use openssl::{asn1, ec, hash, nid, pkey, x509};
828        let group = ec::EcGroup::from_curve_name(curve).expect("curve");
829        let pair =
830            pkey::PKey::from_ec_key(ec::EcKey::generate(&group).expect("ec key")).expect("pkey");
831        let mut name = x509::X509NameBuilder::new().expect("name");
832        name.append_entry_by_nid(nid::Nid::COMMONNAME, "sp")
833            .expect("cn");
834        let name = name.build();
835        let mut builder = x509::X509Builder::new().expect("builder");
836        builder.set_version(2).expect("version");
837        builder.set_subject_name(&name).expect("subject");
838        builder.set_issuer_name(&name).expect("issuer");
839        builder.set_pubkey(&pair).expect("pubkey");
840        builder
841            .set_not_before(&asn1::Asn1Time::days_from_now(0).expect("nb"))
842            .expect("nb");
843        builder
844            .set_not_after(&asn1::Asn1Time::days_from_now(30).expect("na"))
845            .expect("na");
846        builder
847            .sign(&pair, hash::MessageDigest::sha256())
848            .expect("sign");
849        String::from_utf8(builder.build().to_pem().expect("pem")).expect("utf8")
850    }
