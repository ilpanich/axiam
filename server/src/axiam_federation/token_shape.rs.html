1//! What kind of JWT a provider-issued token is, judged by its shape.
2//!
3//! One definition, read by two checks that point in opposite directions:
4//!
5//! * X4 ([`crate::token_exchange`]) accepts an **access token** as an RFC 8693
6//!   subject token and refuses anything carrying an ID-token marker or a
7//!   refresh/logout type.
8//! * Native sign-in ([`crate::native_login`]) accepts **only an ID token** and
9//!   refuses an access-token-shaped JWT.
10//!
11//! Were the marker list written twice, a marker added to one side and not the
12//! other would open exactly the hole each check closes: an ID token buying an
13//! API credential through X4, or an access token buying a sign-in here. So the
14//! lists live here and both modules read them.
15
16/// Claims that are legal only in an OIDC ID token: `nonce` (OIDC Core
17/// §3.1.3.7) and the hashes of §3.3.2.11 / §3.2.2.9.
18///
19/// The order is the order X4 reports them in, and is part of its behaviour.
20pub(crate) const ID_TOKEN_ONLY_CLAIMS: [&str; 4] = ["nonce", "at_hash", "c_hash", "s_hash"];
21
22/// `typ` values (JOSE header or claim, after [`normalize_typ`]) that name an ID
23/// token.
24pub(crate) const ID_TOKEN_TYPES: [&str; 2] = ["id_token", "id+jwt"];
25
26/// `typ` values that name a token which is neither an access token nor an ID
27/// token: a refresh token, an offline token, a logout token. Neither path
28/// accepts one.
29pub(crate) const NEITHER_ACCESS_NOR_ID_TYPES: [&str; 4] =
30    ["refresh", "refresh_token", "offline", "logout+jwt"];
31
32/// `typ` values that name an access token (RFC 9068 §2.1). Native sign-in
33/// refuses them; X4 has no need to name them, because it refuses by marker
34/// rather than requiring one.
35pub(crate) const ACCESS_TOKEN_TYPES: [&str; 1] = ["at+jwt"];
36
37/// A `typ` as compared: trimmed, ASCII-lowercased, an `application/` media-type
38/// prefix removed (RFC 7515 §4.1.9 allows it to be omitted).
39pub(crate) fn normalize_typ(typ: &str) -> String {
40    let lowered = typ.trim().to_ascii_lowercase();
41    match lowered.strip_prefix("application/") {
42        Some(rest) => rest.to_owned(),
43        None => lowered,
44    }
45}
46
47/// Whether X4 refuses a token whose (normalized) `typ` is this: an ID token
48/// type, or one that is neither an access nor an ID token.
49pub(crate) fn refused_as_subject_token(normalized: &str) -> bool {
50    ID_TOKEN_TYPES.contains(&normalized) || NEITHER_ACCESS_NOR_ID_TYPES.contains(&normalized)
51}
52
53/// Whether native sign-in refuses a token whose (normalized) `typ` is this: an
54/// access token type, or one that is neither an access nor an ID token.
55pub(crate) fn refused_as_native_id_token(normalized: &str) -> bool {
56    ACCESS_TOKEN_TYPES.contains(&normalized) || NEITHER_ACCESS_NOR_ID_TYPES.contains(&normalized)
57}

59#[cfg(test)]
60mod tests {
61    use super::*;
62
63    /// The two refusals partition the named types: no `typ` is acceptable to
64    /// both paths, and an ID-token type is refused by exactly one of them.
65    #[test]
66    fn no_named_type_is_accepted_by_both_paths() {
67        for t in ID_TOKEN_TYPES
68            .iter()
69            .chain(&NEITHER_ACCESS_NOR_ID_TYPES)
70            .chain(&ACCESS_TOKEN_TYPES)
71        {
72            assert!(
73                refused_as_subject_token(t) || refused_as_native_id_token(t),
74                "{t} is refused by at least one path"
75            );
76        }
77        for t in ID_TOKEN_TYPES {
78            assert!(refused_as_subject_token(t) && !refused_as_native_id_token(t));
79        }
80        for t in ACCESS_TOKEN_TYPES {
81            assert!(!refused_as_subject_token(t) && refused_as_native_id_token(t));
82        }
83    }
84
85    #[test]
86    fn a_typ_is_compared_without_case_whitespace_or_media_type_prefix() {
87        assert_eq!(normalize_typ(" Application/ID+JWT "), "id+jwt");
88        assert_eq!(normalize_typ("at+jwt"), "at+jwt");
89        assert_eq!(normalize_typ("JWT"), "jwt");
90    }
91}
