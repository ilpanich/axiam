1//! OAuth2 Device Authorization Grant (RFC 8628) — B2.
2//!
3//! The grant for input-constrained devices: a TV, a CLI, a headless IoT
4//! sensor. The device shows a short code, the user types it on a phone or
5//! laptop, and the device polls until the user has approved.
6//!
7//! # The two codes, and why they are shaped so differently
8//!
9//! | | `device_code` | `user_code` |
10//! |---|---|---|
11//! | who handles it | the device, over TLS | a **human**, by reading and typing |
12//! | entropy | 256 bits | ~34 bits |
13//! | stored as | SHA-256 hash | normalised plaintext |
14//! | lifetime | the grant's | the grant's |
15//!
16//! The device code is an ordinary bearer secret and is treated like one: CSPRNG,
17//! hashed at rest, never logged.
18//!
19//! The user code cannot be. It has to be read off a screen and typed on a
20//! phone, so it is short, and short means guessable — which makes **rate
21//! limiting the verification endpoint a security control, not a courtesy**.
22//! See [`brute_force_attempts_required`] for the arithmetic that sizes it.
23
24use rand::RngExt;
25use sha2::{Digest, Sha256};
26
27/// Alphabet for the user code (RFC 8628 §6.1 recommends exactly this shape).
28///
29/// 20 uppercase consonants. What is excluded, and why:
30///
31/// - **every digit.** This is what removes the `O`/`0` and `I`/`1`/`L`
32///   confusions at the root: with no digits in the alphabet, a user squinting
33///   at a television cannot mistake a letter for one.
34/// - **every vowel** (`A E I O U`) **and `Y`.** Partly the `O` confusion
35///   again, but mainly so the generated code cannot spell a word — nobody
36///   should have to read an obscenity off their screen to sign in.
37///
38/// Excluding characters costs entropy, and that is the right trade: a user who
39/// misreads a character retries, and a retry is an *attempt* against the rate
40/// limit — so ambiguity does not merely annoy, it consumes the budget that
41/// protects the code (see [`brute_force_attempts_required`]).
42pub const USER_CODE_CHARSET: &[u8] = b"BCDFGHJKLMNPQRSTVWXZ";
43
44/// Length of the generated user code, excluding the display hyphen.
45pub const USER_CODE_LEN: usize = 8;
46
47/// Default lifetime of a device grant (RFC 8628 `expires_in`).
48///
49/// Long enough for a user to find their phone, short enough that a code left
50/// on a screen in a public place is not a standing invitation.
51pub const DEFAULT_EXPIRES_IN_SECS: u64 = 600;
52
53/// Default minimum seconds between token polls (RFC 8628 `interval`).
54pub const DEFAULT_INTERVAL_SECS: u64 = 5;
55
56/// Generate a `user_code` in its display form, e.g. `BCDF-GHJK`.
57///
58/// The hyphen is presentation only — [`normalize_user_code`] strips it, so a
59/// user may type it either way.
60pub fn generate_user_code() -> String {
61    let mut rng = rand::rng();
62    let raw: String = (0..USER_CODE_LEN)
63        .map(|_| {
64            let idx = rng.random_range(0..USER_CODE_CHARSET.len());
65            USER_CODE_CHARSET[idx] as char
66        })
67        .collect();
68    format!("{}-{}", &raw[..4], &raw[4..])
69}
70
71/// Normalise a user-typed code for lookup.
72///
73/// Uppercases, and drops everything that is not in the charset — hyphens,
74/// spaces, and the stray characters a phone keyboard inserts. Without this a
75/// user typing `bcdf ghjk` would simply not be found, and would reasonably
76/// conclude the code was wrong rather than that they had typed it in a form
77/// the server did not accept.
78pub fn normalize_user_code(input: &str) -> String {
79    input
80        .chars()
81        .flat_map(|c| c.to_uppercase())
82        .filter(|c| USER_CODE_CHARSET.contains(&(*c as u8)))
83        .collect()
84}

86/// Generate a `device_code`: 256 bits of CSPRNG, base64url, unpadded.
87pub fn generate_device_code() -> String {
88    use base64::Engine;
89    use base64::engine::general_purpose::URL_SAFE_NO_PAD;
90    let mut rng = rand::rng();
91    let bytes: [u8; 32] = rng.random();
92    URL_SAFE_NO_PAD.encode(bytes)
93}
94
95/// SHA-256 of a device code, hex-encoded — what is stored.
96///
97/// SHA-256 rather than a KDF for the same reason refresh tokens use it: the
98/// value is a 256-bit CSPRNG string, so there is no offline-guessing threat
99/// for a KDF to defend against, and the poll path runs every few seconds per
100/// device by design.
101pub fn hash_device_code(raw: &str) -> String {
102    let mut hasher = Sha256::new();
103    hasher.update(raw.as_bytes());
104    hex::encode(hasher.finalize())
105}

107/// How many guesses an attacker needs, on average, to hit a live user code —
108/// given `rate_per_minute` verification attempts and a `lifetime_secs` window.
109///
110/// # The number this exists to keep honest
111///
112/// The user code is ~34 bits (`20^8 ≈ 2.56 × 10^10`), which sounds ample and
113/// is not the relevant figure. What matters is how many guesses an attacker
114/// can *make* while a code is live, against how many codes are live — a
115/// deployment with many concurrent pending grants shrinks the search space
116/// proportionally.
117///
118/// The OWASP bar this implements: `charset^len / (rate × lifetime)` must
119/// exceed `10^6`. At the shipped 10-minute lifetime, that caps the
120/// verification endpoint at roughly `2.56 × 10^10 / (10^6 × 600) ≈ 42` attempts
121/// per second — orders of magnitude above any human, and the reason the
122/// verification endpoint gets its own rate-limit bucket rather than sharing
123/// the generic one.
124pub fn brute_force_attempts_required(rate_per_minute: u64, lifetime_secs: u64) -> f64 {
125    let space = (USER_CODE_CHARSET.len() as f64).powi(USER_CODE_LEN as i32);
126    let attempts_in_window = (rate_per_minute as f64) * (lifetime_secs as f64) / 60.0;
127    if attempts_in_window <= 0.0 {
128        return f64::INFINITY;
129    }
130    space / attempts_in_window
131}

133#[cfg(test)]
134mod tests {
135    use super::*;
136    use std::collections::HashSet;

138    #[test]
139    fn user_codes_use_only_the_unambiguous_charset() {
140        for _ in 0..200 {
141            let code = generate_user_code();
142            for c in code.chars().filter(|c| *c != '-') {
143                assert!(
144                    USER_CODE_CHARSET.contains(&(c as u8)),
145                    "generated {c:?}, which is not in the unambiguous charset"
146                );
147            }
148        }
149    }
150
151    /// Every excluded character is excluded for a reason a user would hit.
152    #[test]
153    fn ambiguous_characters_are_absent() {
154        // No digits at all — this is what kills the O/0 and I/1/L confusions
155        // at the root rather than case by case.
156        for c in '0'..='9' {
157            assert!(
158                !USER_CODE_CHARSET.contains(&(c as u8)),
159                "{c:?}: the alphabet must contain no digits, so no letter can \
160                 be mistaken for one"
161            );
162        }
163        // No vowels or Y — so a generated code cannot spell a word.
164        for c in ['A', 'E', 'I', 'O', 'U', 'Y'] {
165            assert!(
166                !USER_CODE_CHARSET.contains(&(c as u8)),
167                "{c:?} must be absent so a code cannot spell a word"
168            );
169        }
170    }
171
172    /// The alphabet size is what the brute-force arithmetic is computed
173    /// against, so shrinking it silently would silently weaken that bound.
174    #[test]
175    fn alphabet_size_is_what_the_brute_force_math_assumes() {
176        assert_eq!(USER_CODE_CHARSET.len(), 20);
177        assert_eq!(
178            USER_CODE_CHARSET.len(),
179            USER_CODE_CHARSET
180                .iter()
181                .collect::<std::collections::HashSet<_>>()
182                .len(),
183            "duplicate characters would inflate the assumed key space"
184        );
185    }
186
187    #[test]
188    fn user_codes_are_displayed_hyphenated() {
189        let code = generate_user_code();
190        assert_eq!(code.len(), USER_CODE_LEN + 1);
191        assert_eq!(code.chars().nth(4), Some('-'));
192    }

194    /// A user typing the code back in any reasonable form must be found.
195    #[test]
196    fn normalisation_accepts_what_a_human_actually_types() {
197        let canonical = normalize_user_code("BCDF-GHJK");
198        assert_eq!(canonical, "BCDFGHJK");
199
200        for variant in [
201            "bcdf-ghjk",
202            "BCDFGHJK",
203            "bcdf ghjk",
204            " BCDF-GHJK ",
205            "b c d f - g h j k",
206            "BCDF—GHJK", // em dash, which some phone keyboards substitute
207        ] {
208            assert_eq!(
209                normalize_user_code(variant),
210                canonical,
211                "{variant:?} must normalise to the same code — otherwise the \
212                 user retries, and a retry spends the rate-limit budget that \
213                 protects the code"
214            );
215        }
216    }
217
218    /// Normalisation must not invent characters that are not in the alphabet.
219    #[test]
220    fn normalisation_drops_out_of_charset_characters() {
221        assert_eq!(normalize_user_code("BC0DF1GH"), "BCDFGH");
222        assert_eq!(normalize_user_code("!!!!"), "");
223    }

225    #[test]
226    fn device_codes_are_high_entropy_and_unique() {
227        let codes: HashSet<String> = (0..500).map(|_| generate_device_code()).collect();
228        assert_eq!(codes.len(), 500, "device codes must not collide");
229        // 32 bytes base64url-unpadded.
230        assert_eq!(generate_device_code().len(), 43);
231    }

233    #[test]
234    fn device_code_hashing_is_stable_and_not_reversible_by_length() {
235        let raw = generate_device_code();
236        assert_eq!(hash_device_code(&raw), hash_device_code(&raw));
237        assert_ne!(hash_device_code(&raw), raw);
238        assert_eq!(hash_device_code(&raw).len(), 64);
239    }

241    /// The shipped posture must clear the OWASP bar with room to spare.
242    ///
243    /// This is the test that stops someone raising the verification rate limit
244    /// "because users complain" without noticing what it costs.
245    #[test]
246    fn shipped_rate_limit_clears_the_owasp_brute_force_bar() {
247        // The verification endpoint's own bucket, sized like the other human
248        // endpoints rather than like the machine ones.
249        let shipped_rate_per_min = 10;
250        let required = brute_force_attempts_required(shipped_rate_per_min, DEFAULT_EXPIRES_IN_SECS);
251
252        assert!(
253            required > 1e6,
254            "OWASP bar: charset^len / (rate x lifetime) must exceed 10^6, got {required:e}"
255        );
256    }
257
258    /// ...and the bar is genuinely a bar: a wide-open verification endpoint
259    /// fails it, which is what makes the assertion above meaningful.
260    #[test]
261    fn an_unthrottled_verification_endpoint_fails_the_bar() {
262        // 100k attempts/min is what an unthrottled endpoint on this hardware
263        // would comfortably serve.
264        let required = brute_force_attempts_required(100_000, DEFAULT_EXPIRES_IN_SECS);
265        assert!(
266            required < 1e6,
267            "the bar must actually reject an unthrottled endpoint, got {required:e}"
268        );
269    }
270
271    /// A longer lifetime widens the attack window linearly — worth pinning, so
272    /// that raising `expires_in` is a decision rather than a tweak.
273    #[test]
274    fn a_longer_lifetime_lowers_the_margin() {
275        let ten_min = brute_force_attempts_required(10, 600);
276        let one_hour = brute_force_attempts_required(10, 3_600);
277        assert!(one_hour < ten_min);
278        assert!((ten_min / one_hour - 6.0).abs() < 0.01);
279    }
