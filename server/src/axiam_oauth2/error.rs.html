5/// OAuth2-specific errors following RFC 6749 error codes.
6#[derive(Debug, Error)]
7pub enum OAuth2Error {
8    #[error("invalid_request: {0}")]
9    InvalidRequest(String),
10    #[error("unauthorized_client: {0}")]
11    UnauthorizedClient(String),
12    #[error("access_denied: {0}")]
13    AccessDenied(String),
14    #[error("unsupported_response_type: only 'code' response type is supported")]
15    UnsupportedResponseType,
16    #[error("invalid_scope: {0}")]
17    InvalidScope(String),
18    #[error("invalid_grant: {0}")]
19    InvalidGrant(String),
20    #[error("invalid_client: {0}")]
21    InvalidClient(String),
22    /// RFC 9449 §5 — a DPoP proof was required and was missing or unusable.
23    ///
24    /// # Why this is not `invalid_client`
25    ///
26    /// It was, and it was wrong in both directions. A `private_key_jwt` client
27    /// that presented a perfectly valid assertion and no proof was told
28    /// `invalid_client: invalid client credentials`, which names the one thing
29    /// that was not the problem — so the client cannot tell what to fix, and an
30    /// operator reading logs sees an authentication failure that did not
31    /// happen. RFC 9449 §5 gives the code for exactly this case, and §7.1 gives
32    /// the same code for a proof that fails verification at a resource.
33    ///
34    /// The OIDF FAPI 2.0 DPoP lane checks it by name:
35    /// `CheckTokenEndpointReturnedInvalidRequestGrantOrDPopProofError` accepts
36    /// `invalid_request`, `invalid_grant` or `invalid_dpop_proof` and rejects
37    /// anything else, `invalid_client` included.
38    ///
39    /// Carries HTTP **400**, not 401: 401 is for a failure to authenticate the
40    /// client (RFC 6749 §5.2), and this is a client that authenticated.
41    #[error("invalid_dpop_proof: {0}")]
42    InvalidDpopProof(String),
43    #[error("invalid_request: {0}")]
44    InvalidRedirectUri(String),
45    /// B5 — the client is registered `require_par` and sent its parameters
46    /// through the browser anyway.
47    ///
48    /// A distinct variant rather than a plain `InvalidRequest` because of how
49    /// the authorize handler decides whether to redirect: `InvalidRequest`
50    /// means "we got past client and redirect_uri validation", and is
51    /// therefore safe to report by redirecting. This refusal happens
52    /// *before* the redirect_uri is validated — deliberately, since the whole
53    /// point is that this client's parameters must not travel through the
54    /// browser — so redirecting would bounce the user agent to an unvalidated
55    /// URI supplied by the very channel the setting forbids. On the wire it
56    /// is still `invalid_request` per RFC 9126 §5; the variant exists to keep
57    /// the response non-redirecting.
58    #[error("invalid_request: {0}")]
59    ParRequired(String),
60    #[error("unsupported_grant_type: grant type is not supported")]
61    UnsupportedGrantType,
62    #[error("server_error: {0}")]
63    ServerError(String),
64
65    // --- RFC 8628 device-flow polling errors (B2) --------------------------
66    //
67    // These are NOT failures in the usual sense: `authorization_pending` is
68    // the *expected* answer to almost every poll, and `slow_down` is
69    // corrective rather than terminal. They are modelled as errors because
70    // that is what RFC 8628 §3.5 puts on the wire — a 400 with an `error`
71    // field — and because keeping them in one type is what stops the token
72    // endpoint growing a second, parallel response path.
73    /// The user has not yet approved or denied. The device keeps polling.
74    #[error("authorization_pending: the user has not yet completed authorization")]
75    AuthorizationPending,
76    /// The device is polling faster than the interval it was given. The
77    /// interval has been raised; the device must honour the new one.
78    #[error("slow_down: polling faster than the permitted interval")]
79    SlowDown,
80    /// The device code expired before the user acted.
81    #[error("expired_token: the device code has expired; restart the flow")]
82    ExpiredToken,
83
84    // --- RFC 8693 token exchange (B3) --------------------------------------
85    /// RFC 8693 §2.2.2 — the requested `audience`/`resource` is not one the
86    /// exchanging client may address. Its own error code rather than
87    /// `invalid_request` because a caller can act on it: the target is
88    /// well-formed, it is simply not theirs to address.
89    #[error("invalid_target: {0}")]
90    InvalidTarget(String),
91
92    // --- OIDC Core §3.1.2.6 request objects (X7 G12) -----------------------
93    //
94    // Their own variants rather than `InvalidRequest` because the codes are
95    // the whole point: OIDC Core defines `request_not_supported` and
96    // `request_uri_not_supported` precisely so a relying party can tell "I
97    // sent a malformed request" from "this server does not do request
98    // objects", and a conformance suite matches on the distinction. AXIAM
99    // implements neither form — see `authorize::RequestObject` for why.
100    /// OIDC Core §3.1.2.6 — `login_required`: the authorization server needs
101    /// the end user to authenticate and cannot ask again (W3, plan §4.0).
102    ///
103    /// The **terminal state of the login hop's loop guard**. A `browser_sso`
104    /// client's anonymous authorization request is redirected to the login page
105    /// once; if the request that redirect produces comes back still carrying no
106    /// principal, this is the answer, and it is deliberately not another
107    /// redirect. See `crate::login_hop` for the argument that this bounds the
108    /// chain at two authorization requests.
109    ///
110    /// Answered **directly**, not by redirecting to the relying party. W3
111    /// honours no authentication-request parameter, so it builds no
112    /// parameter-driven redirect error; the loop guard is a server-side safety
113    /// valve for a deployment that is misconfigured or a browser that is
114    /// refusing cookies, and neither is something the relying party can act on
115    /// by being told in a query string.
116    #[error("login_required: {0}")]
117    LoginRequired(String),
118
119    /// OIDC Core §3.1.2.6 — `consent_required`: the end user has not consented
120    /// to something this request needs, and `prompt=none` forbids asking.
121    ///
122    /// **Unreachable in W4, deliberately and stated rather than hidden.** The
123    /// only thing that could require consent is a consent-gated scope, and
124    /// there are none until W7 defines `address` and `phone` (plan §4.8). The
125    /// variant exists because the four OIDC interaction errors are one
126    /// vocabulary and splitting it across two waves is how an error code comes
127    /// to be spelled twice; the honour lane raises it nowhere, and
128    /// `crate::honour` has no branch that could.
129    #[error("consent_required: {0}")]
130    ConsentRequired(String),
131
132    /// OIDC Core §3.1.2.6 — `interaction_required`: some interaction other
133    /// than authentication or consent is needed, and `prompt=none` forbids it.
134    ///
135    /// **Also unreachable in W4.** Every `prompt=none` refusal the honour lane
136    /// can produce has a more specific name — `login_required` for a missing,
137    /// stale or mismatched authentication, `account_selection_required` when a
138    /// hint mismatch is the cause, `unmet_authentication_requirements` for an
139    /// essential `acr` — and OIDC Core asks for the most specific code that
140    /// applies. It is declared with its siblings for the same reason
141    /// [`Self::ConsentRequired`] is.
142    #[error("interaction_required: {0}")]
143    InteractionRequired(String),
144
145    /// OIDC Core §3.1.2.6 — `account_selection_required`: the end user needs
146    /// to choose a session, and this request could not choose for them.
147    ///
148    /// Raised on the return leg of a login hop when `prompt=select_account`
149    /// was asked and the `id_token_hint` still names somebody other than
150    /// whoever signed in (plan §4.2). Without `select_account` the same state
151    /// is `login_required`: the relying party did not ask about accounts, so
152    /// naming accounts in the answer would tell it something it did not ask.
153    #[error("account_selection_required: {0}")]
154    AccountSelectionRequired(String),
155
156    /// OpenID Connect Core Error Code `unmet_authentication_requirements` 1.0
157    /// — an **essential** authentication context class the end user did not
158    /// reach.
159    ///
160    /// Distinct from `login_required` because the two ask the relying party
161    /// for different things: `login_required` says "send them back and they
162    /// can sign in", and this says "they signed in, and this deployment cannot
163    /// give you the assurance level you require of them" — commonly a user
164    /// with no second factor enrolled against a request for
165    /// [`crate::acr::Acr::MultiFactor`]. Answering `login_required` there
166    /// invites a loop the relying party drives.
167    ///
168    /// Never accompanied by a token, at any point. That is the whole
169    /// difference between an essential and a voluntary `acr` request.
170    #[error("unmet_authentication_requirements: {0}")]
171    UnmetAuthenticationRequirements(String),
172
173    /// OIDC Core §3.1.2.6 — `invalid_request_uri`, raised on the **return leg
174    /// of a login hop** when the pushed request the browser left with is gone
175    /// (W3, plan §4.0 and F10).
176    ///
177    /// A PAR `request_uri` lives 60 seconds (RFC 9126 §2.2, and `par.rs`'s own
178    /// constant). A user who takes longer than that to type a password comes
179    /// back to an authorization endpoint that has nothing to consume — and
180    /// telling them `invalid_request: request_uri is unknown, expired, or used`
181    /// describes a client bug that did not happen. This code says the recoverable
182    /// thing instead, which is what OIDC Core defines it for: the relying party
183    /// pushes again and restarts.
184    ///
185    /// Raised **only** on a request carrying `axiam_login_hop`; an ordinary
186    /// authorization request with a dead `request_uri` gets the same
187    /// `invalid_request` it has always got, because changing that would change
188    /// behaviour for a client registered today.
189    #[error("invalid_request_uri: {0}")]
190    InvalidRequestUri(String),
191
192    /// OIDC Core §3.1.2.6 — a `request` parameter (a request object by value).
193    #[error(
194        "request_not_supported: this server does not accept request objects; send the \
195             authorization parameters directly, or push them to /oauth2/par"
196    )]
197    RequestNotSupported,
198    /// OIDC Core §3.1.2.6 — a `request_uri` that is not a PAR handle.
199    #[error(
200        "request_uri_not_supported: request_uri accepts only a \
201             urn:ietf:params:oauth:request_uri: value obtained from /oauth2/par (RFC 9126); \
202             this server does not fetch request objects by reference"
203    )]
204    RequestUriNotSupported,
205}

208    /// RFC 6749 error code string.
209    pub fn error_code(&self) -> &'static str {
210        match self {
211            Self::InvalidRequest(_) => "invalid_request",
212            Self::UnauthorizedClient(_) => "unauthorized_client",
213            Self::AccessDenied(_) => "access_denied",
214            Self::UnsupportedResponseType => "unsupported_response_type",
215            Self::InvalidScope(_) => "invalid_scope",
216            Self::InvalidGrant(_) => "invalid_grant",
217            Self::InvalidClient(_) => "invalid_client",
218            Self::InvalidDpopProof(_) => "invalid_dpop_proof",
219            Self::InvalidRedirectUri(_) => "invalid_request",
220            Self::ParRequired(_) => "invalid_request",
221            Self::UnsupportedGrantType => "unsupported_grant_type",
222            Self::ServerError(_) => "server_error",
223            Self::AuthorizationPending => "authorization_pending",
224            Self::SlowDown => "slow_down",
225            Self::ExpiredToken => "expired_token",
226            Self::InvalidTarget(_) => "invalid_target",
227            Self::LoginRequired(_) => "login_required",
228            Self::ConsentRequired(_) => "consent_required",
229            Self::InteractionRequired(_) => "interaction_required",
230            Self::AccountSelectionRequired(_) => "account_selection_required",
231            Self::UnmetAuthenticationRequirements(_) => "unmet_authentication_requirements",
232            Self::InvalidRequestUri(_) => "invalid_request_uri",
233            Self::RequestNotSupported => "request_not_supported",
234            Self::RequestUriNotSupported => "request_uri_not_supported",
235        }
236    }

238    /// Human-readable error description for the `error_description` field.
239    ///
240    /// Strips the RFC error-code prefix from the Display output so that
241    /// `error_description` contains only the message (the code goes in
242    /// the separate `error` field per RFC 6749 §5.2).
243    pub fn error_description(&self) -> String {
244        let full = self.to_string();
245        // Display format is "error_code: message"; extract the message part.
246        let msg = match full.split_once(": ") {
247            Some((_, msg)) => msg,
248            None => &full,
249        };
250        nqschar(msg)
251    }
252}
253
254/// Render text as RFC 6749 §5.2 `error_description`.
255///
256/// The grammar is not advisory and it is narrower than "text":
257///
258/// ```text
259/// error_description = 1*NQSCHAR
260/// NQSCHAR           = %x20-21 / %x23-5B / %x5D-7E
261/// ```
262///
263/// Printable US-ASCII, minus `"` (%x22) and `\` (%x5C) — the two characters
264/// that would need escaping inside the JSON string the field is delivered in.
265/// Everything above %x7E is excluded, which rules out every character this
266/// codebase reaches for when it writes prose: `§`, `—`, curly quotes, `…`.
267///
268/// # Why here and not at the call sites
269///
270/// Because there are hundreds of call sites and one of this function. A
271/// message is written by whoever is fixing the bug it describes, in the house
272/// style of the file around it, and that style cites specifications as `§`.
273/// Asking every future author to remember a character-set rule from RFC 6749
274/// is asking for the defect back; `error_description()` is the one place every
275/// one of those strings passes through on its way to a client.
276///
277/// It is a real defect and not pedantry. The OIDF module
278/// `fapi2-security-profile-final-ensure-holder-of-key-required` fails on
279/// exactly this, and it failed on a description that had just been *corrected*
280/// to say `RFC 9449 §5` — the section reference is what broke it.
281///
282/// # Transliterated, not stripped
283///
284/// The characters that actually occur are given ASCII spellings rather than
285/// dropped, because the description exists to be read by an integrator: "RFC
286/// 9449 5" and "RFC 9449 section 5" are not equally useful, and a message with
287/// its punctuation silently deleted reads like a corrupted string. Anything
288/// not spelled out here is replaced by a single `?`, which is visible — a
289/// dropped character is not, and an invisible truncation is how a message
290/// comes to mean something it did not say.
291pub fn nqschar(msg: &str) -> String {
292    let mut out = String::with_capacity(msg.len());
293    for c in msg.chars() {
294        match c {
295            // The permitted set, verbatim.
296            '\x20'..='\x21' | '\x23'..='\x5b' | '\x5d'..='\x7e' => out.push(c),
297            // Excluded by NQSCHAR even though they are ASCII.
298            '"' => out.push('\''),
299            '\\' => out.push('/'),
300            // Whitespace a multi-line Rust string literal picks up.
301            '\n' | '\r' | '\t' => out.push(' '),
302            // The house style's punctuation, spelled out.
303            '§' => out.push_str("section "),
304            '—' | '–' => out.push('-'),
305            '‘' | '’' => out.push('\''),
306            '“' | '”' => out.push('\''),
307            '…' => out.push_str("..."),
308            '\u{a0}' => out.push(' '),
309            _ => out.push('?'),
310        }
311    }
312    out
313}

319    /// The character set is the whole of the rule, so it is asserted as the
320    /// whole of the rule rather than by example.
321    #[test]
322    fn nqschar_admits_exactly_the_rfc6749_set() {
323        for c in '\u{0}'..='\u{ff}' {
324            let permitted = matches!(c, '\x20'..='\x21' | '\x23'..='\x5b' | '\x5d'..='\x7e');
325            let rendered = nqschar(&c.to_string());
326            if permitted {
327                assert_eq!(rendered, c.to_string(), "{c:?} is NQSCHAR and must survive");
328            } else {
329                assert!(
330                    rendered.chars().all(|r| matches!(
331                        r,
332                        '\x20'..='\x21' | '\x23'..='\x5b' | '\x5d'..='\x7e'
333                    )),
334                    "{c:?} rendered as {rendered:?}, which is not NQSCHAR"
335                );
336            }
337        }
338    }

344    fn a_section_sign_becomes_a_readable_word_not_a_hole() {

358    /// Every variant that carries a message, rendered — because the rule binds
359    /// the field and not one variant of it.

361    fn every_error_description_is_nqschar() {
362        let prose = "an em dash — a section §7 — curly ‘quotes’ and an ellipsis…";
363        for e in [
364            OAuth2Error::InvalidRequest(prose.into()),

368            OAuth2Error::InvalidGrant(prose.into()),
369            OAuth2Error::InvalidClient(prose.into()),
370            OAuth2Error::InvalidDpopProof(prose.into()),
371            OAuth2Error::ServerError(prose.into()),

373            let d = e.error_description();
374            assert!(
375                d.chars()
376                    .all(|c| matches!(c, '\x20'..='\x21' | '\x23'..='\x5b' | '\x5d'..='\x7e')),
377                "{d:?} escaped the NQSCHAR rule"

396    /// The OIDC interaction vocabulary, pinned as strings.
397    ///
398    /// These five codes are what a relying party matches on and what the
399    /// conformance suite's
400    /// `CheckErrorFromAuthorizationEndpointIsOneThatRequiredAUserInterface`
401    /// accepts; a typo in one of them is invisible to every type in the
402    /// codebase and visible to every client. Two of the five are unreachable
403    /// in W4 — see their doc comments — and are pinned anyway so the wave that
404    /// reaches them inherits the spelling rather than choosing it again.
405    #[test]
406    fn the_oidc_interaction_error_codes_are_spelled_as_the_specification_spells_them() {
407        for (error, code) in [
408            (OAuth2Error::LoginRequired(String::new()), "login_required"),
409            (
410                OAuth2Error::ConsentRequired(String::new()),
411                "consent_required",
412            ),
413            (
414                OAuth2Error::InteractionRequired(String::new()),
415                "interaction_required",
416            ),
417            (
418                OAuth2Error::AccountSelectionRequired(String::new()),
419                "account_selection_required",
420            ),
421            (
422                OAuth2Error::UnmetAuthenticationRequirements(String::new()),
423                "unmet_authentication_requirements",
424            ),
425        ] {
426            assert_eq!(error.error_code(), code);
427        }
428    }
