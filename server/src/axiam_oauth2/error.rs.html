1//! OAuth2 error types per RFC 6749.
2
3use thiserror::Error;
4
5/// OAuth2-specific errors following RFC 6749 error codes.
6#[derive(Debug, Error)]
7pub enum OAuth2Error {
8    #[error("invalid_request: {0}")]
9    InvalidRequest(String),
10    #[error("unauthorized_client: {0}")]
11    UnauthorizedClient(String),
12    #[error("access_denied: {0}")]
13    AccessDenied(String),
14    #[error("unsupported_response_type: only 'code' response type is supported")]
15    UnsupportedResponseType,
16    #[error("invalid_scope: {0}")]
17    InvalidScope(String),
18    #[error("invalid_grant: {0}")]
19    InvalidGrant(String),
20    #[error("invalid_client: {0}")]
21    InvalidClient(String),
22    #[error("invalid_request: {0}")]
23    InvalidRedirectUri(String),
24    /// B5 — the client is registered `require_par` and sent its parameters
25    /// through the browser anyway.
26    ///
27    /// A distinct variant rather than a plain `InvalidRequest` because of how
28    /// the authorize handler decides whether to redirect: `InvalidRequest`
29    /// means "we got past client and redirect_uri validation", and is
30    /// therefore safe to report by redirecting. This refusal happens
31    /// *before* the redirect_uri is validated — deliberately, since the whole
32    /// point is that this client's parameters must not travel through the
33    /// browser — so redirecting would bounce the user agent to an unvalidated
34    /// URI supplied by the very channel the setting forbids. On the wire it
35    /// is still `invalid_request` per RFC 9126 §5; the variant exists to keep
36    /// the response non-redirecting.
37    #[error("invalid_request: {0}")]
38    ParRequired(String),
39    #[error("unsupported_grant_type: grant type is not supported")]
40    UnsupportedGrantType,
41    #[error("server_error: {0}")]
42    ServerError(String),
43
44    // --- RFC 8628 device-flow polling errors (B2) --------------------------
45    //
46    // These are NOT failures in the usual sense: `authorization_pending` is
47    // the *expected* answer to almost every poll, and `slow_down` is
48    // corrective rather than terminal. They are modelled as errors because
49    // that is what RFC 8628 §3.5 puts on the wire — a 400 with an `error`
50    // field — and because keeping them in one type is what stops the token
51    // endpoint growing a second, parallel response path.
52    /// The user has not yet approved or denied. The device keeps polling.
53    #[error("authorization_pending: the user has not yet completed authorization")]
54    AuthorizationPending,
55    /// The device is polling faster than the interval it was given. The
56    /// interval has been raised; the device must honour the new one.
57    #[error("slow_down: polling faster than the permitted interval")]
58    SlowDown,
59    /// The device code expired before the user acted.
60    #[error("expired_token: the device code has expired; restart the flow")]
61    ExpiredToken,
62
63    // --- RFC 8693 token exchange (B3) --------------------------------------
64    /// RFC 8693 §2.2.2 — the requested `audience`/`resource` is not one the
65    /// exchanging client may address. Its own error code rather than
66    /// `invalid_request` because a caller can act on it: the target is
67    /// well-formed, it is simply not theirs to address.
68    #[error("invalid_target: {0}")]
69    InvalidTarget(String),
70}
71
72impl OAuth2Error {
73    /// RFC 6749 error code string.
74    pub fn error_code(&self) -> &'static str {
75        match self {
76            Self::InvalidRequest(_) => "invalid_request",
77            Self::UnauthorizedClient(_) => "unauthorized_client",
78            Self::AccessDenied(_) => "access_denied",
79            Self::UnsupportedResponseType => "unsupported_response_type",
80            Self::InvalidScope(_) => "invalid_scope",
81            Self::InvalidGrant(_) => "invalid_grant",
82            Self::InvalidClient(_) => "invalid_client",
83            Self::InvalidRedirectUri(_) => "invalid_request",
84            Self::ParRequired(_) => "invalid_request",
85            Self::UnsupportedGrantType => "unsupported_grant_type",
86            Self::ServerError(_) => "server_error",
87            Self::AuthorizationPending => "authorization_pending",
88            Self::SlowDown => "slow_down",
89            Self::ExpiredToken => "expired_token",
90            Self::InvalidTarget(_) => "invalid_target",
91        }
92    }
93
94    /// Human-readable error description for the `error_description` field.
95    ///
96    /// Strips the RFC error-code prefix from the Display output so that
97    /// `error_description` contains only the message (the code goes in
98    /// the separate `error` field per RFC 6749 §5.2).
99    pub fn error_description(&self) -> String {
100        let full = self.to_string();
101        // Display format is "error_code: message"; extract the message part.
102        match full.split_once(": ") {
103            Some((_, msg)) => msg.to_string(),
104            None => full,
105        }
106    }
107}
