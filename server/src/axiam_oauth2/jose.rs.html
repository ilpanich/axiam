1//! The JOSE rules both asymmetric proof-of-possession mechanisms share (X5.1).
2//!
3//! `private_key_jwt` (RFC 7523 §2.2) and DPoP (RFC 9449) are different
4//! protocols solving different problems, but they make the *same* three
5//! decisions about a JWT signed by a key AXIAM did not mint:
6//!
7//! 1. which algorithms are permitted at all,
8//! 2. **which key decides the algorithm** — and it is never the header, and
9//! 3. how a public key reduces to the `jkt` thumbprint that names it.
10//!
11//! Those decisions live here once. Two copies would be two chances to fix one
12//! and not the other, and the thing being decided is which signatures the
13//! server accepts.
14//!
15//! # `alg` comes from the key, not from the assertion
16//!
17//! This is the load-bearing rule of the whole module. An attacker who can
18//! choose the algorithm a verifier uses does not need to break the signature:
19//! the classic failures — `alg: none`, and RSA-public-key-as-HMAC-secret —
20//! are both "the token told the verifier how to check the token". So
21//! [`algorithm_for_key`] derives the algorithm from the **key material**, and
22//! [`verify_permitted_header`] then requires the header to agree with what the
23//! key already decided. A header naming anything else is refused rather than
24//! honoured.
25//!
26//! `none` is unreachable twice over: [`jsonwebtoken::Algorithm`] has no such
27//! variant, so a proof carrying it fails to parse before any of this runs, and
28//! [`PERMITTED_ALGORITHMS`] would not contain it if it did.
29//!
30//! # Why only three algorithms
31//!
32//! FAPI 2.0 §5.3.1.1 permits `PS256`, `ES256` and `EdDSA`. AXIAM implements
33//! exactly those and refuses the rest — including `RS256`, which is the one an
34//! operator is most likely to be surprised by, and which the profile excludes
35//! deliberately (PKCS#1 v1.5 padding). A P-384 or P-521 key is likewise
36//! refused: it is a perfectly good key, but it is not on the profile's list,
37//! and quietly accepting `ES384` here would mean AXIAM's answer to "which
38//! algorithms do you accept?" differed from the profile's.
39

45/// The signature algorithms AXIAM accepts on a client assertion or a DPoP
46/// proof (FAPI 2.0 §5.3.1.1).
47pub const PERMITTED_ALGORITHMS: [Algorithm; 3] =
48    [Algorithm::PS256, Algorithm::ES256, Algorithm::EdDSA];
49
50/// Why a key or a header could not be used.
51#[derive(Debug, Clone, PartialEq, Eq)]
52pub enum JoseKeyError {
53    /// The key set contained no key usable under the permitted algorithms.
54    NoUsableKey,
55    /// A `kid` was named and no registered key carries it.
56    UnknownKid { kid: String },
57    /// The key's own type or curve is outside the profile.
58    UnsupportedKeyType { detail: String },
59    /// The header named an algorithm the selected key does not use.
60    AlgorithmMismatch { header: Algorithm, key: Algorithm },
61    /// The key's own `alg` member disagrees with its key material, or names an
62    /// algorithm outside the profile.
63    DeclaredAlgorithmMismatch { declared: String, key: Algorithm },
64    /// The header named an algorithm outside the profile.
65    AlgorithmNotPermitted { header: Algorithm },
66    /// The key could not be turned into a verifier.
67    Unusable { detail: String },
68}
69
70impl std::fmt::Display for JoseKeyError {
71    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
72        match self {
73            Self::NoUsableKey => write!(
74                f,
75                "no registered key uses an algorithm this profile permits (PS256, ES256, EdDSA)"
76            ),
77            Self::UnknownKid { kid } => {
78                write!(f, "no registered key carries the key id {kid:?}")
79            }
80            Self::UnsupportedKeyType { detail } => write!(f, "unsupported key: {detail}"),
81            Self::AlgorithmMismatch { header, key } => write!(
82                f,
83                "the JWS header names {header:?} but the registered key is a {key:?} key; the \
84                 algorithm is decided by the key, never by the header"
85            ),
86            Self::DeclaredAlgorithmMismatch { declared, key } => write!(
87                f,
88                "the registered key declares alg {declared:?} but its key material is a {key:?} \
89                 key; this profile permits PS256, ES256 and EdDSA, and a key that declares \
90                 anything else is refused rather than reinterpreted"
91            ),
92            Self::AlgorithmNotPermitted { header } => write!(
93                f,
94                "{header:?} is not permitted by this profile (PS256, ES256, EdDSA)"
95            ),
96            Self::Unusable { detail } => write!(f, "key cannot be used for verification: {detail}"),
97        }
98    }
99}

103/// Whether `alg` is one of the three the profile permits.
104pub fn is_permitted(alg: Algorithm) -> bool {
105    PERMITTED_ALGORITHMS.contains(&alg)
106}
107
108/// The one algorithm this key may be used with.
109///
110/// Derived from the key material — `kty` and, for the curve families, `crv` —
111/// so that the answer does not depend on anything the presenter controls. A
112/// `alg` member on the JWK itself is honoured only as a *consistency check*: if
113/// the key declares one, it must equal what the material implies. A JWK saying
114/// `{"kty":"RSA","alg":"ES256"}` is not a request to verify an RSA signature
115/// with ECDSA; it is a malformed key, and it is refused.
116pub fn algorithm_for_key(jwk: &Jwk) -> Result<Algorithm, JoseKeyError> {
117    let intrinsic = match &jwk.algorithm {
118        AlgorithmParameters::RSA(_) => Algorithm::PS256,
119        AlgorithmParameters::EllipticCurve(ec) => match ec.curve {
120            EllipticCurve::P256 => Algorithm::ES256,
121            ref other => {
122                return Err(JoseKeyError::UnsupportedKeyType {
123                    detail: format!(
124                        "EC curve {other:?} is outside this profile, which permits ES256 (P-256) \
125                         only"
126                    ),
127                });
128            }
129        },
130        AlgorithmParameters::OctetKeyPair(okp) => match okp.curve {
131            EllipticCurve::Ed25519 => Algorithm::EdDSA,
132            ref other => {
133                return Err(JoseKeyError::UnsupportedKeyType {
134                    detail: format!(
135                        "OKP curve {other:?} is outside this profile, which permits EdDSA over \
136                         Ed25519 only"
137                    ),
138                });
139            }
140        },
141        // A symmetric key is not a proof of possession of anything the server
142        // does not already hold, and accepting one here is how "verify this
143        // RSA public key as an HMAC secret" becomes a signature forgery.
144        AlgorithmParameters::OctetKey(_) => {
145            return Err(JoseKeyError::UnsupportedKeyType {
146                detail: "symmetric (oct) keys cannot authenticate a client asymmetrically".into(),
147            });
148        }
149        // `Other`, and any variant a future jsonwebtoken adds. Both are
150        // "a key type this profile does not list", and the catch-all is
151        // deliberate: a new key type must arrive as a refusal until somebody
152        // decides it belongs, not as a silently accepted credential.
153        _ => {
154            return Err(JoseKeyError::UnsupportedKeyType {
155                detail: "unrecognised key type".into(),
156            });
157        }
158    };
159
160    // A declared `alg` on the key is honoured only as a consistency check.
161    // `KeyAlgorithm` covers algorithms `Algorithm` does not (RSA-OAEP and
162    // friends), so the comparison is made on the declared value's own terms:
163    // anything that is not the intrinsic algorithm's name is a disagreement,
164    // whether it is a *different signature* algorithm or not a signature
165    // algorithm at all.
166    if let Some(declared) = jwk.common.key_algorithm
167        && !declares(declared, intrinsic)
168    {
169        return Err(JoseKeyError::DeclaredAlgorithmMismatch {
170            declared: format!("{declared:?}"),
171            key: intrinsic,
172        });
173    }
174
175    Ok(intrinsic)
176}

178/// Whether a JWK's declared `alg` member names `expected`.
179///
180/// Compared by name rather than by converting `KeyAlgorithm` into `Algorithm`:
181/// the conversion is private to `jsonwebtoken`, and more importantly it is
182/// partial — `RSA-OAEP` has no `Algorithm` — so a key declaring an encryption
183/// algorithm would otherwise fall through the check it is most important not to
184/// fall through.
185fn declares(declared: jsonwebtoken::jwk::KeyAlgorithm, expected: Algorithm) -> bool {
186    use jsonwebtoken::jwk::KeyAlgorithm;
187    matches!(
188        (declared, expected),
189        (KeyAlgorithm::PS256, Algorithm::PS256)
190            | (KeyAlgorithm::ES256, Algorithm::ES256)
191            | (KeyAlgorithm::EdDSA, Algorithm::EdDSA)
192    )
193}

195/// Check a JWS header against the algorithm the key already decided.
196///
197/// Returns the algorithm to verify with — always the key's, never the
198/// header's. The header is only ever an assertion to be *contradicted*.
199pub fn verify_permitted_header(
200    header_alg: Algorithm,
201    key_alg: Algorithm,
202) -> Result<Algorithm, JoseKeyError> {
203    if !is_permitted(key_alg) {
204        return Err(JoseKeyError::AlgorithmNotPermitted { header: key_alg });
205    }
206    if !is_permitted(header_alg) {
207        return Err(JoseKeyError::AlgorithmNotPermitted { header: header_alg });
208    }
209    if header_alg != key_alg {
210        return Err(JoseKeyError::AlgorithmMismatch {
211            header: header_alg,
212            key: key_alg,
213        });
214    }
215    Ok(key_alg)
216}

218/// Build a verifier from a public JWK.
219pub fn decoding_key_for(jwk: &Jwk) -> Result<DecodingKey, JoseKeyError> {
220    DecodingKey::from_jwk(jwk).map_err(|e| JoseKeyError::Unusable {
221        detail: e.to_string(),
222    })
223}

225/// RFC 7638 JWK thumbprint, SHA-256, base64url-unpadded — the `jkt` of
226/// RFC 9449 §6.1 and the `cnf.jkt` of §6.
227///
228/// Delegated to `jsonwebtoken`'s implementation rather than hand-rolled: the
229/// canonicalisation (required members only, lexicographic order, no
230/// whitespace) is the entire security property, and a second implementation of
231/// it would be a second chance to canonicalise differently from every other
232/// participant — which shows up as a `jkt` that never matches, or worse, one
233/// that matches a key it should not.
234pub fn jwk_thumbprint(jwk: &Jwk) -> Result<String, JoseKeyError> {
235    jwk.thumbprint(ThumbprintHash::SHA256)
236        .map_err(|e| JoseKeyError::Unusable {
237            detail: format!("cannot compute a JWK thumbprint: {e}"),
238        })
239}
240
241/// RFC 9449 §4.2 `ath`: base64url-unpadded SHA-256 of the access token's
242/// ASCII form.
243///
244/// Unpadded for the same reason `x5t#S256` is: the value travels in a JWT
245/// claim, and RFC 7515 §2 defines JOSE base64url as omitting the padding. A
246/// padded value would not compare equal at a conforming peer.
247pub fn access_token_hash(access_token: &str) -> String {
248    base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(Sha256::digest(access_token.as_bytes()))
249}

251/// Select the key a JWS was signed with from a registered key set.
252///
253/// `kid` is honoured when the header carries one, because that is what a `kid`
254/// is for and because a client rotating keys publishes both. When the header
255/// carries none, every key in the set is a candidate — which is correct and
256/// not a weakening: each candidate is a key the *client registered*, so
257/// verifying against all of them proves possession of one of them, which is
258/// exactly the claim being made.
259///
260/// Keys the profile cannot use are skipped rather than failing the whole set. A
261/// client that publishes one Ed25519 key and one P-521 key has published one
262/// key AXIAM can use, and refusing the set would make an unrelated key an
263/// outage.
264pub fn candidate_keys<'a>(
265    keys: &'a [Jwk],
266    kid: Option<&str>,
267) -> Result<Vec<(&'a Jwk, Algorithm)>, JoseKeyError> {
268    let scoped: Vec<&Jwk> = match kid {
269        Some(kid) => {
270            let matched: Vec<&Jwk> = keys
271                .iter()
272                .filter(|k| k.common.key_id.as_deref() == Some(kid))
273                .collect();
274            if matched.is_empty() {
275                return Err(JoseKeyError::UnknownKid {
276                    kid: kid.to_owned(),
277                });
278            }
279            matched
280        }
281        None => keys.iter().collect(),
282    };
283
284    let usable: Vec<(&Jwk, Algorithm)> = scoped
285        .into_iter()
286        .filter_map(|k| algorithm_for_key(k).ok().map(|alg| (k, alg)))
287        .collect();
288
289    if usable.is_empty() {
290        return Err(JoseKeyError::NoUsableKey);
291    }
292    Ok(usable)
293}
