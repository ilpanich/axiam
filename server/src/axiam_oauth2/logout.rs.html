1//! OIDC RP-Initiated Logout 1.0 and Back-Channel Logout 1.0 — B5.
2//!
3//! # Why the two exist together
4//!
5//! A federation deployment is AXIAM plus N relying parties. Each half closes a
6//! hole the other leaves:
7//!
8//! - Without RP-initiated logout, a user who logs out of an RP stays logged in
9//!   at AXIAM, so the next "Login with AXIAM" silently re-authenticates them.
10//!   From the user's point of view they did not log out.
11//! - Without back-channel logout, a user who logs out **of AXIAM** stays
12//!   logged in at every RP, indefinitely, because nothing tells the RPs. This
13//!   is the one that matters: an admin revoking a compromised account leaves N
14//!   live sessions behind.
15//!
16//! # Session precision
17//!
18//! Both halves operate on a **session**, not a user. A user with a phone and a
19//! laptop who logs out on the laptop expects the phone to stay signed in, and
20//! RP-Initiated Logout 1.0 §2 agrees. That is why the ID token carries `sid`
21//! and why the logout token always names it.
22

31/// The `events` member every logout token must carry (Back-Channel Logout 1.0
32/// §2.4).
33pub const BACKCHANNEL_LOGOUT_EVENT: &str = "http://schemas.openid.net/event/backchannel-logout";
34
35/// How long a logout token stays valid.
36///
37/// A logout token is delivered immediately or not at all, so its lifetime only
38/// has to cover one HTTP request plus clock skew. A long-lived one is a
39/// replayable session-termination command.
40pub const LOGOUT_TOKEN_LIFETIME_SECS: i64 = 120;
41
42/// Claims of an OIDC back-channel logout token.
43#[derive(Debug, Clone, Serialize, Deserialize)]
44pub struct LogoutTokenClaims {
45    pub iss: String,
46    /// The receiving client. One token per client, because `aud` names one.
47    pub aud: String,
48    pub iat: i64,
49    pub exp: i64,
50    /// Replay identifier the RP is required to track.
51    pub jti: String,
52    /// The session that ended. AXIAM always sends this — see the module docs.
53    pub sid: String,
54    pub sub: String,
55    /// `{ "http://schemas.openid.net/event/backchannel-logout": {} }`.
56    ///
57    /// This member is what distinguishes a logout token from an ID token, and
58    /// the receiving RP is required to check it.
59    pub events: Map<String, Value>,
60}
61
62/// Build the `events` claim.
63fn logout_events() -> Map<String, Value> {
64    let mut events = Map::new();
65    events.insert(
66        BACKCHANNEL_LOGOUT_EVENT.to_string(),
67        Value::Object(Map::new()),
68    );
69    events
70}

72/// Issue a signed back-channel logout token for one client.
73///
74/// Deliberately **not** given a `nonce` parameter. Back-Channel Logout 1.0 §2.4
75/// forbids the claim, and its presence is the documented way an attacker
76/// replays an ID token as a logout token. Making it un-passable is stronger
77/// than remembering not to pass it.
78pub fn issue_logout_token(
79    issuer: &str,
80    client_id: &str,
81    session_id: Uuid,
82    user_id: Uuid,
83    config: &AuthConfig,
84) -> Result<String, AuthError> {
85    let now = Utc::now().timestamp();
86    let claims = LogoutTokenClaims {
87        iss: issuer.to_string(),
88        aud: client_id.to_string(),
89        iat: now,
90        exp: now + LOGOUT_TOKEN_LIFETIME_SECS,
91        jti: Uuid::new_v4().to_string(),
92        sid: session_id.to_string(),
93        sub: user_id.to_string(),
94        events: logout_events(),
95    };
96
97    let owned;
98    let key: &EncodingKey = if let Some(ref cached) = config.jwt_encoding_key {
99        cached.as_ref()
100    } else {
101        owned = EncodingKey::from_ed_pem(config.jwt_private_key_pem.as_bytes())
102            .map_err(|e| AuthError::Crypto(format!("bad private key: {e}")))?;
103        &owned
104    };
105    jsonwebtoken::encode(&Header::new(Algorithm::EdDSA), &claims, key)
106        .map_err(|e| AuthError::Crypto(format!("logout token encode: {e}")))
107}

109/// The parts of an `id_token_hint` the end-session endpoint acts on.
110#[derive(Debug, Clone, PartialEq, Eq)]
111pub struct IdTokenHint {
112    /// The session to end. `None` when the token predates `sid` (B5).
113    pub session_id: Option<Uuid>,
114    /// The client the token was issued to — the allow-list to check the
115    /// `post_logout_redirect_uri` against.
116    pub client_id: String,
117    pub subject_id: Option<Uuid>,
118}
119
120/// Decode an `id_token_hint`.
121///
122/// Two validation choices, both deliberate:
123///
124/// - **Expiry is not checked.** An ID token presented here has routinely
125///   expired already: the user is logging out, which commonly happens long
126///   after a 15-minute token was minted. Rejecting expired hints would make
127///   the hint useless exactly when it is most needed.
128/// - **The signature IS checked.** That is what makes the hint an
129///   authenticated statement of which session and client this is about rather
130///   than a suggestion. Without it, anyone could name any session.
131///
132/// `aud` is read out rather than validated against — there is no expected
133/// value to compare to, since the client's identity is what we are learning.
134///
135/// Anything unverifiable returns `None` and is treated by the caller as an
136/// absent hint, per RP-Initiated Logout 1.0 §2 — not as an error.
137pub fn decode_id_token_hint(token: &str, public_key_pem: &str) -> Option<IdTokenHint> {
138    use jsonwebtoken::{DecodingKey, Validation};
139
140    let key = DecodingKey::from_ed_pem(public_key_pem.as_bytes()).ok()?;
141    let mut validation = Validation::new(Algorithm::EdDSA);
142    validation.validate_exp = false;
143    validation.validate_aud = false;
144
145    let claims = jsonwebtoken::decode::<axiam_auth::token::IdTokenClaims>(token, &key, &validation)
146        .ok()?
147        .claims;
148    Some(IdTokenHint {
149        session_id: claims.sid.as_deref().and_then(|s| Uuid::parse_str(s).ok()),
150        client_id: claims.aud,
151        subject_id: Uuid::parse_str(&claims.sub).ok(),
152    })
153}

155/// What the end-session endpoint decided to do.
156#[derive(Debug, Clone, PartialEq, Eq)]
157pub enum LogoutOutcome {
158    /// Session ended; send the browser to this URI (with `state` appended when
159    /// the RP supplied one).
160    Redirect { uri: String, state: Option<String> },
161    /// Session ended; render AXIAM's own logged-out page.
162    ///
163    /// This is **not** an error path. It is what happens when the RP supplied
164    /// no `post_logout_redirect_uri`, or supplied one that is not on its
165    /// allow-list. Refusing to log the user out because their RP sent a bad
166    /// parameter would be the wrong failure: they asked to log out, and they
167    /// are logged out.
168    Rendered,
169}
170
171/// Resolve where to send the browser after a logout.
172///
173/// The allow-list check is **exact string equality** — no prefix matching, no
174/// wildcards, no scheme or host normalisation — the same discipline
175/// `redirect_uris` already uses. `post_logout_redirect_uri` is reachable
176/// without authentication by design, so a loose match here is an open redirect
177/// on an unauthenticated endpoint.
178pub fn resolve_post_logout_redirect(
179    requested: Option<&str>,
180    allow_list: &[String],
181    state: Option<&str>,
182) -> LogoutOutcome {
183    match requested {
184        Some(uri) if allow_list.iter().any(|a| a == uri) => LogoutOutcome::Redirect {
185            uri: uri.to_string(),
186            state: state.map(str::to_owned),
187        },
188        // Includes both "no URI requested" and "URI not on the allow-list".
189        // `state` is echoed only on a redirect that actually happens — there
190        // is nowhere to echo it to otherwise, and it is never interpreted.
191        _ => LogoutOutcome::Rendered,
192    }
193}

226    fn decode(token: &str, config: &AuthConfig, aud: &str) -> LogoutTokenClaims {
227        let key = DecodingKey::from_ed_pem(config.jwt_public_key_pem.as_bytes()).unwrap();
228        let mut v = Validation::new(Algorithm::EdDSA);
229        v.set_audience(&[aud]);
230        v.set_issuer(&["https://id.test.example"]);
231        jsonwebtoken::decode::<LogoutTokenClaims>(token, &key, &v)
232            .unwrap()
233            .claims
234    }

236    #[test]
237    fn logout_token_carries_the_exact_events_member() {
238        let cfg = test_config();
239        let sid = Uuid::new_v4();
240        let sub = Uuid::new_v4();
241        let t = issue_logout_token("https://id.test.example", "rp-1", sid, sub, &cfg).unwrap();
242        let c = decode(&t, &cfg, "rp-1");

244        // The RP is required to check this; an empty or differently-keyed
245        // events member makes the token indistinguishable from an ID token.
246        assert_eq!(c.events.len(), 1);
247        let member = c.events.get(BACKCHANNEL_LOGOUT_EVENT).expect("events key");
248        assert_eq!(member, &Value::Object(Map::new()));
249    }

253        // Back-Channel Logout 1.0 §2.4 forbids `nonce`; its presence is how an

386    #[test]
387    fn an_expired_id_token_hint_is_still_decoded() {
388        // The common case: the user logs out well after their 15-minute token
389        // expired. Rejecting it would make the hint useless when it matters.
390        let cfg = test_config();
391        let sid = Uuid::new_v4();
392        let sub = Uuid::new_v4();
393        let mut expired_cfg = cfg.clone();
394        expired_cfg.access_token_lifetime_secs = 1;
395        let token = axiam_auth::token::issue_id_token(
396            sub,
397            Uuid::new_v4(),
398            Uuid::new_v4(),
399            "rp-1",
400            None,
401            None,
402            None,
403            &[],
404            &expired_cfg,
405            Some(sid),
406        )
407        .unwrap();
408        std::thread::sleep(std::time::Duration::from_millis(1100));
409        let hint = decode_id_token_hint(&token, &cfg.jwt_public_key_pem).expect("decodes");
