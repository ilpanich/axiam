1//! OIDC RP-Initiated Logout 1.0 and Back-Channel Logout 1.0 — B5.
2//!
3//! # Why the two exist together
4//!
5//! A federation deployment is AXIAM plus N relying parties. Each half closes a
6//! hole the other leaves:
7//!
8//! - Without RP-initiated logout, a user who logs out of an RP stays logged in
9//!   at AXIAM, so the next "Login with AXIAM" silently re-authenticates them.
10//!   From the user's point of view they did not log out.
11//! - Without back-channel logout, a user who logs out **of AXIAM** stays
12//!   logged in at every RP, indefinitely, because nothing tells the RPs. This
13//!   is the one that matters: an admin revoking a compromised account leaves N
14//!   live sessions behind.
15//!
16//! # Session precision
17//!
18//! Both halves operate on a **session**, not a user. A user with a phone and a
19//! laptop who logs out on the laptop expects the phone to stay signed in, and
20//! RP-Initiated Logout 1.0 §2 agrees. That is why the ID token carries `sid`
21//! and why the logout token always names it.
22

31/// The `events` member every logout token must carry (Back-Channel Logout 1.0
32/// §2.4).
33pub const BACKCHANNEL_LOGOUT_EVENT: &str = "http://schemas.openid.net/event/backchannel-logout";
34
35/// How long a logout token stays valid.
36///
37/// A logout token is delivered immediately or not at all, so its lifetime only
38/// has to cover one HTTP request plus clock skew. A long-lived one is a
39/// replayable session-termination command.
40pub const LOGOUT_TOKEN_LIFETIME_SECS: i64 = 120;
41
42/// Claims of an OIDC back-channel logout token.
43#[derive(Debug, Clone, Serialize, Deserialize)]
44pub struct LogoutTokenClaims {
45    pub iss: String,
46    /// The receiving client. One token per client, because `aud` names one.
47    pub aud: String,
48    pub iat: i64,
49    pub exp: i64,
50    /// Replay identifier the RP is required to track.
51    pub jti: String,
52    /// The session that ended. AXIAM always sends this — see the module docs.
53    pub sid: String,
54    pub sub: String,
55    /// `{ "http://schemas.openid.net/event/backchannel-logout": {} }`.
56    ///
57    /// This member is what distinguishes a logout token from an ID token, and
58    /// the receiving RP is required to check it.
59    pub events: Map<String, Value>,
60}
61
62/// Build the `events` claim.
63fn logout_events() -> Map<String, Value> {
64    let mut events = Map::new();
65    events.insert(
66        BACKCHANNEL_LOGOUT_EVENT.to_string(),
67        Value::Object(Map::new()),
68    );
69    events
70}

72/// Issue a signed back-channel logout token for one client.
73///
74/// Deliberately **not** given a `nonce` parameter. Back-Channel Logout 1.0 §2.4
75/// forbids the claim, and its presence is the documented way an attacker
76/// replays an ID token as a logout token. Making it un-passable is stronger
77/// than remembering not to pass it.
78pub fn issue_logout_token(
79    issuer: &str,
80    client_id: &str,
81    session_id: Uuid,
82    user_id: Uuid,
83    config: &AuthConfig,
84) -> Result<String, AuthError> {
85    let now = Utc::now().timestamp();
86    let claims = LogoutTokenClaims {
87        iss: issuer.to_string(),
88        aud: client_id.to_string(),
89        iat: now,
90        exp: now + LOGOUT_TOKEN_LIFETIME_SECS,
91        jti: Uuid::new_v4().to_string(),
92        sid: session_id.to_string(),
93        sub: user_id.to_string(),
94        events: logout_events(),
95    };
96
97    // SA-03 — explicitly typed (`logout+jwt`), as Back-Channel Logout 1.0 §2.4
98    // recommends, so no verifier of another AXIAM token kind can mistake it
99    // for its own. #880 (contract 1.69) — and naming its JWKS `kid`, because a
100    // relying party selects the verification key by it.
101    axiam_auth::token::sign_logout_token(&claims, config)
102}

104/// The parts of an `id_token_hint` the end-session endpoint acts on.
105#[derive(Debug, Clone, PartialEq, Eq)]
106pub struct IdTokenHint {
107    /// The session to end. `None` when the token predates `sid` (B5).
108    pub session_id: Option<Uuid>,
109    /// The client the token was issued to — the allow-list to check the
110    /// `post_logout_redirect_uri` against.
111    pub client_id: String,
112    pub subject_id: Option<Uuid>,
113}
114
115/// Decode an `id_token_hint`.
116///
117/// Two validation choices, both deliberate:
118///
119/// - **Expiry is not checked.** An ID token presented here has routinely
120///   expired already: the user is logging out, which commonly happens long
121///   after a 15-minute token was minted. Rejecting expired hints would make
122///   the hint useless exactly when it is most needed.
123/// - **The signature IS checked.** That is what makes the hint an
124///   authenticated statement of which session and client this is about rather
125///   than a suggestion. Without it, anyone could name any session.
126///
127/// The header `typ` must be an ID token's (`JWT`, or absent), and the claims
128/// must carry no member an ID token never has (`jti`, `purpose`, `scope`,
129/// `client_id`): an access token — including one minted before SA-03 with
130/// `typ: JWT` — a purpose token or a logout token is not a hint. See
131/// [`axiam_auth::token::decode_id_token_hint_claims`].
132///
133/// `aud` is read out rather than validated against — there is no expected
134/// value to compare to, since the client's identity is what we are learning.
135///
136/// Anything unverifiable returns `None` and is treated by the caller as an
137/// absent hint, per RP-Initiated Logout 1.0 §2 — not as an error.
138pub fn decode_id_token_hint(token: &str, public_key_pem: &str) -> Option<IdTokenHint> {
139    use jsonwebtoken::{DecodingKey, Validation};
140
141    let key = DecodingKey::from_ed_pem(public_key_pem.as_bytes()).ok()?;
142    let mut validation = Validation::new(Algorithm::EdDSA);
143    validation.validate_exp = false;
144    validation.validate_aud = false;
145
146    // Q-8 / S-10, SR2-2: only an ID token is a hint. An access token's claims
147    // fit `IdTokenClaims` (and carry `sid`), so the header and the claim set
148    // must refuse it.
149    let claims = axiam_auth::token::decode_id_token_hint_claims(token, &key, &validation)?;
150    Some(IdTokenHint {
151        session_id: claims.sid.as_deref().and_then(|s| Uuid::parse_str(s).ok()),
152        client_id: claims.aud,
153        subject_id: Uuid::parse_str(&claims.sub).ok(),
154    })
155}

157/// What the end-session endpoint decided to do.
158#[derive(Debug, Clone, PartialEq, Eq)]
159pub enum LogoutOutcome {
160    /// Session ended; send the browser to this URI (with `state` appended when
161    /// the RP supplied one).
162    Redirect { uri: String, state: Option<String> },
163    /// Session ended; render AXIAM's own logged-out page.
164    ///
165    /// This is **not** an error path. It is what happens when the RP supplied
166    /// no `post_logout_redirect_uri`, or supplied one that is not on its
167    /// allow-list. Refusing to log the user out because their RP sent a bad
168    /// parameter would be the wrong failure: they asked to log out, and they
169    /// are logged out.
170    Rendered,
171}
172
173/// Resolve where to send the browser after a logout.
174///
175/// The allow-list check is **exact string equality** — no prefix matching, no
176/// wildcards, no scheme or host normalisation — the same discipline
177/// `redirect_uris` already uses. `post_logout_redirect_uri` is reachable
178/// without authentication by design, so a loose match here is an open redirect
179/// on an unauthenticated endpoint.
180pub fn resolve_post_logout_redirect(
181    requested: Option<&str>,
182    allow_list: &[String],
183    state: Option<&str>,
184) -> LogoutOutcome {
185    match requested {
186        Some(uri) if allow_list.iter().any(|a| a == uri) => LogoutOutcome::Redirect {
187            uri: uri.to_string(),
188            state: state.map(str::to_owned),
189        },
190        // Includes both "no URI requested" and "URI not on the allow-list".
191        // `state` is echoed only on a redirect that actually happens — there
192        // is nowhere to echo it to otherwise, and it is never interpreted.
193        _ => LogoutOutcome::Rendered,
194    }
195}

311    /// #880: verify the way a relying party does — read the JWKS as
312    /// `/oauth2/jwks` serves it, during a rotation (two keys), select the key
313    /// **by `kid`** and nothing else, then verify. Before contract 1.69 the
314    /// token named no key, so a strictly-by-`kid` verifier found none.
315    #[test]
316    fn a_relying_party_selects_the_logout_key_by_kid() {
317        let cfg = test_config();
318        let sid = Uuid::new_v4();
319        let t = issue_logout_token("https://id.test.example", "rp-1", sid, Uuid::new_v4(), &cfg)
320            .unwrap();
321
322        // What the endpoint serves, plus the next key of a rotation.
323        let mut served =
324            serde_json::to_value(crate::oidc::build_jwks(&cfg.jwt_public_key_pem).unwrap())
325                .unwrap();
326        let next = crate::oidc::build_jwks(OTHER_PUBLIC_KEY_PEM).unwrap();
327        served["keys"]
328            .as_array_mut()
329            .unwrap()
330            .insert(0, serde_json::to_value(&next.keys[0]).unwrap());
331        let jwks: jsonwebtoken::jwk::JwkSet = serde_json::from_value(served).unwrap();
332        assert_eq!(jwks.keys.len(), 2);
333
334        let header = jsonwebtoken::decode_header(&t).unwrap();
335        let kid = header.kid.expect("a logout token must carry a kid");
336        let jwk = jwks.find(&kid).expect("the kid must name a published key");
337        let key = DecodingKey::from_jwk(jwk).unwrap();
338        let mut v = Validation::new(Algorithm::EdDSA);
339        v.set_audience(&["rp-1"]);
340        v.set_issuer(&["https://id.test.example"]);
341        let claims = jsonwebtoken::decode::<LogoutTokenClaims>(&t, &key, &v)
342            .expect("the key the kid names verifies the token")
343            .claims;
344        assert_eq!(claims.sid, sid.to_string());
345
346        // The other key of the rotation does not.
347        let other = jwks
348            .keys
349            .iter()
350            .find(|k| k.common.key_id.as_deref() != Some(kid.as_str()))
351            .unwrap();
352        let other = DecodingKey::from_jwk(other).unwrap();
353        assert!(jsonwebtoken::decode::<LogoutTokenClaims>(&t, &other, &v).is_err());
354    }

356    /// #880 security argument, the claim half: with a `kid`, a logout token is
357    /// a token a strictly-by-`kid` verifier finds a key for, so what must keep
358    /// it from being an access token is not the key lookup. The header refuses
359    /// it (`typ`), and so — independently — does its claim set, which lacks
360    /// the `tenant_id` and `org_id` every access token carries. Re-signed as
361    /// `at+jwt`, addressed to `axiam:user` and issued by the root issuer (so
362    /// neither `typ`, `aud` nor `iss` can be what refuses it), it is still
363    /// refused by both decoders, strict and relaxed.
364    #[test]
365    fn a_logout_token_is_refused_as_an_access_token_by_its_claims_alone() {
366        let cfg = test_config();
367        let mut relaxed = cfg.clone();
368        relaxed.require_access_token_typ = false;
369        let t = issue_logout_token(
370            cfg.root_issuer(),
371            axiam_auth::token::AUD_USER,
372            Uuid::new_v4(),
373            Uuid::new_v4(),
374            &cfg,
375        )
376        .unwrap();
377        for typ in [Some("at+jwt"), Some("JWT"), None] {
378            let reshaped = resign_with_typ(&t, typ, &cfg);
379            for c in [&cfg, &relaxed] {
380                assert!(
381                    axiam_auth::token::decode_access_token(&reshaped, c).is_err(),
382                    "typ {typ:?}, relaxed {}",
383                    !c.require_access_token_typ
384                );
385                assert!(
386                    axiam_auth::token::decode_access_token_any_audience(&reshaped, c).is_err(),
387                    "typ {typ:?}, relaxed {}",
388                    !c.require_access_token_typ
389                );
390            }
391        }
392        // And as minted, kid and all: refused in both modes.
393        for c in [&cfg, &relaxed] {
394            assert!(axiam_auth::token::decode_access_token(&t, c).is_err());
395            assert!(axiam_auth::token::decode_access_token_any_audience(&t, c).is_err());
396        }
397    }

553        assert_eq!(
554            jsonwebtoken::decode_header(&access).unwrap().typ.as_deref(),
555            Some(axiam_auth::token::TYP_ACCESS_TOKEN)
556        );
557        let hint = decode_id_token_hint(&access, &cfg.jwt_public_key_pem);
558        assert_eq!(
559            hint, None,
560            "an at+jwt access token was accepted as an id_token_hint"
561        );
562
563        // Every other AXIAM kind is refused too; an untyped token (every ID
564        // token before SA-03 was `JWT`) and a real ID token are not.
565        let now = chrono::Utc::now().timestamp();
566        let claims = serde_json::json!({
567            "iss": cfg.root_issuer(), "sub": Uuid::new_v4().to_string(),
568            "aud": "rp-1", "iat": now, "exp": now + 300,
569        });
570        let key = EncodingKey::from_ed_pem(cfg.jwt_private_key_pem.as_bytes()).unwrap();
571        for (typ, accepted) in [
572            (Some(axiam_auth::token::TYP_WEBAUTHN_STATE), false),
573            (Some(axiam_auth::token::TYP_MFA_CHALLENGE), false),
574            (Some(axiam_auth::token::TYP_MFA_SETUP), false),
575            (Some(axiam_auth::token::TYP_LOGOUT_TOKEN), false),
576            (Some("JWT"), true),
577            (None, true),
578        ] {
579            let mut header = Header::new(Algorithm::EdDSA);
580            header.typ = typ.map(str::to_owned);
581            let token = jsonwebtoken::encode(&header, &claims, &key).unwrap();
582            assert_eq!(
583                decode_id_token_hint(&token, &cfg.jwt_public_key_pem).is_some(),
584                accepted,
585                "typ {typ:?}"
586            );
587        }
588        let id = axiam_auth::token::issue_id_token(
589            Uuid::new_v4(),
590            "rp-1",
591            None,
592            None,
593            &[],
594            &cfg,
595            Some(sid),
596            &axiam_auth::token::IdTokenEvidence::NONE,
597        )
598        .unwrap();
599        assert!(decode_id_token_hint(&id, &cfg.jwt_public_key_pem).is_some());
600    }

602    /// Re-sign `token`'s payload, unchanged, under `typ`: what an access token
603    /// minted before SA-03 (`Header::new(EdDSA)`, so `typ: "JWT"`) looks like.
604    fn resign_with_typ(token: &str, typ: Option<&str>, cfg: &AuthConfig) -> String {
605        use base64::Engine;
606        let payload = token.split('.').nth(1).unwrap();
607        let claims: Value = serde_json::from_slice(
608            &base64::engine::general_purpose::URL_SAFE_NO_PAD
609                .decode(payload)
610                .unwrap(),
611        )
612        .unwrap();
613        let key = EncodingKey::from_ed_pem(cfg.jwt_private_key_pem.as_bytes()).unwrap();
614        let mut header = Header::new(Algorithm::EdDSA);
615        header.typ = typ.map(str::to_owned);
616        jsonwebtoken::encode(&header, &claims, &key).unwrap()
617    }

619    /// SR2-2: an access token minted before SA-03 carries `typ: "JWT"` (or
620    /// none) and claims that fit `IdTokenClaims`, `sid` included. The header
621    /// cannot refuse it, and the decoder skips `exp`, so without a claims rule
622    /// every such token would name a session to the end-session endpoint and
623    /// a subject to `/oauth2/authorize` forever. `jti` (and `purpose`,
624    /// `scope`, `client_id`) is what no ID token has ever carried.
625    #[test]
626    fn a_legacy_untyped_access_token_is_not_an_id_token_hint() {
627        let cfg = test_config();
628        let sid = Uuid::new_v4();
629        let access = axiam_auth::token::AccessTokenSpec::user(
630            Uuid::new_v4(),
631            Uuid::new_v4(),
632            Uuid::new_v4(),
633            sid.to_string(),
634        )
635        .aud("rp-1")
636        .session(Some(sid))
637        .issue(&cfg)
638        .unwrap();
639        for typ in [Some("JWT"), None] {
640            let legacy = resign_with_typ(&access, typ, &cfg);
641            assert_eq!(
642                decode_id_token_hint(&legacy, &cfg.jwt_public_key_pem),
643                None,
644                "a legacy (typ {typ:?}) access token was accepted as an id_token_hint"
645            );
646        }
647
648        // Each access-token-only member alone refuses an otherwise ID-shaped,
649        // untyped token.
650        let now = chrono::Utc::now().timestamp();
651        let key = EncodingKey::from_ed_pem(cfg.jwt_private_key_pem.as_bytes()).unwrap();
652        for (member, value) in [
653            ("jti", serde_json::json!(Uuid::new_v4().to_string())),
654            ("purpose", serde_json::json!("webauthn_authentication")),
655            ("scope", serde_json::json!("openid")),
656            ("client_id", serde_json::json!("rp-1")),
657        ] {
658            let mut claims = serde_json::json!({
659                "iss": cfg.root_issuer(), "sub": Uuid::new_v4().to_string(),
660                "aud": "rp-1", "iat": now, "exp": now + 300, "sid": sid.to_string(),
661            });
662            claims[member] = value;
663            let token =
664                jsonwebtoken::encode(&Header::new(Algorithm::EdDSA), &claims, &key).unwrap();
665            assert_eq!(
666                decode_id_token_hint(&token, &cfg.jwt_public_key_pem),
667                None,
668                "an untyped token carrying `{member}` was accepted as an id_token_hint"
669            );
670        }
671
672        // A real ID token as minted today, every optional claim populated,
673        // still is a hint.
674        let sub = Uuid::new_v4();
675        let id = axiam_auth::token::issue_id_token(
676            sub,
677            "rp-1",
678            Some("n-0S6_WzA2Mj"),
679            Some("alice"),
680            &["openid".to_owned(), "profile".to_owned()],
681            &cfg,
682            Some(sid),
683            &axiam_auth::token::IdTokenEvidence {
684                auth_time: Some(now - 60),
685                acr: Some("urn:axiam:acr:mfa".into()),
686                amr: vec!["pwd".into(), "otp".into()],
687            },
688        )
689        .unwrap();
690        let hint = decode_id_token_hint(&id, &cfg.jwt_public_key_pem).expect("a real ID token");
691        assert_eq!(hint.session_id, Some(sid));
692        assert_eq!(hint.client_id, "rp-1");
693        assert_eq!(hint.subject_id, Some(sub));
694    }

699        // expired. Rejecting it would make the hint useless when it matters.
