12/// OpenID Connect Discovery 1.0 metadata document.
13#[derive(Debug, Serialize, utoipa::ToSchema)]
14pub struct OidcDiscoveryDocument {
15    pub issuer: String,
16    pub authorization_endpoint: String,
17    pub token_endpoint: String,
18    pub userinfo_endpoint: String,
19    pub jwks_uri: String,
20    pub revocation_endpoint: String,
21    pub introspection_endpoint: String,
22    /// RFC 8628 §4 — B2. Advertised unconditionally because the grant is
23    /// always mounted; a device that reads discovery is exactly the client
24    /// that cannot be told the URL out of band.
25    pub device_authorization_endpoint: String,
26    /// RFC 9126 §5 — B5. Advertised unconditionally because the endpoint is
27    /// always mounted.
28    pub pushed_authorization_request_endpoint: String,
29    /// The **server-wide** default, which is `false`: PAR is available to
30    /// every client but demanded of none. Per-client enforcement is
31    /// `require_par` on the registration and is deliberately not discoverable
32    /// — RFC 9126 §5 scopes this metadata to the server, and publishing a
33    /// per-client answer here would leak one client's posture to every other
34    /// reader of the document.
35    pub require_pushed_authorization_requests: bool,
36    /// OIDC RP-Initiated Logout 1.0 §3 — B5.
37    pub end_session_endpoint: String,
38    /// Back-Channel Logout 1.0 §3.
39    pub backchannel_logout_supported: bool,
40    /// The claim that AXIAM puts `sid` in its logout tokens — which it does,
41    /// unconditionally. An RP reads this to know it can match a logout token
42    /// to one session rather than having to end every session for the subject.
43    pub backchannel_logout_session_supported: bool,
44    pub response_types_supported: Vec<String>,
45    pub subject_types_supported: Vec<String>,
46    pub id_token_signing_alg_values_supported: Vec<String>,
47    pub scopes_supported: Vec<String>,
48    pub token_endpoint_auth_methods_supported: Vec<String>,
49    pub claims_supported: Vec<String>,
50    pub grant_types_supported: Vec<String>,
51    /// RFC 9207 §3 — X5.1. AXIAM emits `iss` on **every** authorization
52    /// response, success and error alike, for every client, so this is
53    /// unconditionally true. See `handlers::oauth2::append_issuer` for why it
54    /// is not made conditional.
55    pub authorization_response_iss_parameter_supported: bool,
56    /// RFC 8705 §3.3 — X5.1. Certificate-bound access tokens are available;
57    /// whether a given client *gets* them is that client's
58    /// `tls_client_certificate_bound_access_tokens` registration, which is
59    /// deliberately not discoverable for the same reason `require_par` is not:
60    /// this document is scoped to the server, and a per-client answer here
61    /// would leak one client's posture to every reader.
62    pub tls_client_certificate_bound_access_tokens: bool,
63    /// RFC 9449 §5.1 — X5.1 second half. The JWS algorithms AXIAM accepts on a
64    /// DPoP proof.
65    ///
66    /// Its **presence** is what tells a client DPoP is supported at all; RFC
67    /// 9449 defines no separate boolean. The list is the profile's three, and
68    /// notably excludes `RS256` — the omission a client library defaulting to
69    /// RSA will hit first, and the reason advertising the list matters rather
70    /// than merely advertising support.
71    pub dpop_signing_alg_values_supported: Vec<String>,
72}

74/// Build a fully-populated OIDC discovery document for the given issuer URL.
75pub fn build_discovery_document(issuer: &str) -> OidcDiscoveryDocument {
76    let issuer = issuer.trim_end_matches('/');
77    OidcDiscoveryDocument {
78        issuer: issuer.to_string(),
79        authorization_endpoint: format!("{issuer}/oauth2/authorize"),
80        token_endpoint: format!("{issuer}/oauth2/token"),
81        userinfo_endpoint: format!("{issuer}/oauth2/userinfo"),
82        jwks_uri: format!("{issuer}/oauth2/jwks"),
83        revocation_endpoint: format!("{issuer}/oauth2/revoke"),
84        introspection_endpoint: format!("{issuer}/oauth2/introspect"),
85        device_authorization_endpoint: format!("{issuer}/oauth2/device_authorization"),
86        pushed_authorization_request_endpoint: format!("{issuer}/oauth2/par"),
87        require_pushed_authorization_requests: false,
88        end_session_endpoint: format!("{issuer}/oauth2/end_session"),
89        backchannel_logout_supported: true,
90        backchannel_logout_session_supported: true,
91        response_types_supported: vec!["code".into()],
92        subject_types_supported: vec!["public".into()],
93        id_token_signing_alg_values_supported: vec!["EdDSA".into()],
94        scopes_supported: vec!["openid".into(), "profile".into(), "email".into()],
95        token_endpoint_auth_methods_supported: vec![
96            "client_secret_post".into(),
97            // X5.1 / RFC 8705 §2. Advertised unconditionally: whether a mTLS
98            // handshake is actually available is a deployment's listener
99            // configuration (the p3 profile), and a client that cannot reach
100            // an mTLS listener discovers that at connect time rather than by
101            // reading a metadata field that would have to lie one way or the
102            // other on a multi-listener deployment.
103            "tls_client_auth".into(),
104            "self_signed_tls_client_auth".into(),
105            // X5.1 second half / RFC 7523 §2.2. Unlike the two mTLS methods
106            // this one needs nothing from the deployment's listeners at all,
107            // so advertising it carries no caveat: every AXIAM deployment can
108            // serve it.
109            "private_key_jwt".into(),
110        ],
111        claims_supported: vec![
112            "sub".into(),
113            "iss".into(),
114            "aud".into(),
115            "exp".into(),
116            "iat".into(),
117            "nonce".into(),
118            "email".into(),
119            "preferred_username".into(),
120            "tenant_id".into(),
121            "org_id".into(),
122        ],
123        grant_types_supported: vec![
124            "authorization_code".into(),
125            "client_credentials".into(),
126            "refresh_token".into(),
127            // B2: the URN is the grant's identifier on the wire; a device
128            // matches on this exact string, so it is spelled out rather than
129            // referenced, and the constant it must equal
130            // (`device_service::DEVICE_CODE_GRANT_TYPE`) is asserted against
131            // it in this module's tests.
132            "urn:ietf:params:oauth:grant-type:device_code".into(),
133            // B3 / RFC 8693. Advertised so a mesh service can discover that
134            // narrowing a token is possible here rather than being told out
135            // of band; whether a given client MAY exchange is still its own
136            // registration's business.
137            "urn:ietf:params:oauth:grant-type:token-exchange".into(),
138        ],
139        authorization_response_iss_parameter_supported: true,
140        tls_client_certificate_bound_access_tokens: true,
141        dpop_signing_alg_values_supported: vec!["PS256".into(), "ES256".into(), "EdDSA".into()],
142    }
143}

149/// JSON Web Key per RFC 7517.

151pub struct Jwk {

157    pub use_: String,
158    pub alg: String,
159}

161/// JSON Web Key Set document.
162#[derive(Debug, Serialize, utoipa::ToSchema)]
163pub struct JwksDocument {
164    pub keys: Vec<Jwk>,
165}
166
167/// Build a JWKS document from an Ed25519 public key in PEM format.
168///
169/// The PEM must contain a SubjectPublicKeyInfo structure (44 bytes
170/// when DER-decoded: 12-byte OID header + 32-byte raw Ed25519 key).
171/// The `kid` is derived deterministically as the first 16 hex
172/// characters (64 bits) of the SHA-256 hash of the raw public key bytes.
173pub fn build_jwks(public_key_pem: &str) -> Result<JwksDocument, String> {
174    // Strip PEM headers, trim whitespace (handles CRLF / trailing
175    // spaces from env vars or Windows-formatted PEMs), and decode.
176    let b64: String = public_key_pem
177        .lines()
178        .map(str::trim)
179        .filter(|l| !l.starts_with("-----"))
180        .collect();
181    let der = STANDARD
182        .decode(&b64)
183        .map_err(|e| format!("PEM decode: {e}"))?;
184
185    // Ed25519 SubjectPublicKeyInfo is exactly 44 bytes:
186    // 12-byte ASN.1/OID header + 32-byte raw public key.
187    if der.len() != 44 {
188        return Err(format!("expected 44-byte Ed25519 SPKI, got {}", der.len()));
189    }
190    let raw_key = &der[12..44];
191
192    // Base64url-encode the raw key bytes for the JWK `x` parameter.
193    let x = URL_SAFE_NO_PAD.encode(raw_key);
194
195    // Deterministic kid: first 16 hex chars of SHA-256(raw_key).
196    let kid = {
197        let mut h = Sha256::new();
198        h.update(raw_key);
199        hex::encode(h.finalize())[..16].to_string()
200    };
201
202    Ok(JwksDocument {
203        keys: vec![Jwk {
204            kty: "OKP".into(),
205            crv: "Ed25519".into(),
206            x,
207            kid,
208            use_: "sig".into(),
209            alg: "EdDSA".into(),
210        }],
211    })
212}

234#[cfg(test)]
235mod tests {
236    use super::*;
237
238    #[test]
239    fn discovery_document_has_required_fields() {
240        let doc = build_discovery_document("https://auth.example.com");
241        assert_eq!(doc.issuer, "https://auth.example.com");
242        assert_eq!(
243            doc.authorization_endpoint,
244            "https://auth.example.com/oauth2/authorize"
245        );
246        assert!(doc.response_types_supported.contains(&"code".into()));
247        assert!(doc.scopes_supported.contains(&"openid".into()));
248    }
249
250    #[test]
251    fn jwks_parses_ed25519_pem() {
252        let pem = "\
253-----BEGIN PUBLIC KEY-----
254MCowBQYDK2VwAyEAcweT2rPwpUxadO56wIhW1XBoMF63aWOE2UMAVsRudhs=
255-----END PUBLIC KEY-----";
256        let jwks = build_jwks(pem).unwrap();
257        assert_eq!(jwks.keys.len(), 1);
258        assert_eq!(jwks.keys[0].kty, "OKP");
259        assert_eq!(jwks.keys[0].crv, "Ed25519");
260        assert_eq!(jwks.keys[0].alg, "EdDSA");
261        assert_eq!(jwks.keys[0].use_, "sig");
262        assert!(!jwks.keys[0].x.is_empty());
263        assert!(!jwks.keys[0].kid.is_empty());
264    }
265
266    #[test]
267    fn jwk_kid_is_deterministic() {
268        let pem = "\
269-----BEGIN PUBLIC KEY-----
270MCowBQYDK2VwAyEAcweT2rPwpUxadO56wIhW1XBoMF63aWOE2UMAVsRudhs=
271-----END PUBLIC KEY-----";
272        let jwks1 = build_jwks(pem).unwrap();
273        let jwks2 = build_jwks(pem).unwrap();
274        assert_eq!(jwks1.keys[0].kid, jwks2.keys[0].kid);
275    }
