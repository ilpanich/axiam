12/// RFC 8705 §5 `mtls_endpoint_aliases` — the endpoints re-based on the host
13/// that performs the mutual-TLS handshake.
14///
15/// Present only when the deployment configured a separate mTLS host
16/// (`AXIAM__AUTH__OAUTH2_MTLS_BASE_URL`); a client doing mTLS MUST prefer an
17/// alias over the top-level endpoint of the same name.
18//
19// The rest of the rationale is deliberately NOT a doc comment: utoipa lifts a
20// doc block verbatim into the OpenAPI `description`, and eleven SDK repos
21// vendor `sdks/openapi.json` byte-for-byte. A design argument belongs in the
22// source, where it is read by whoever changes this; a two-kilobyte markdown
23// table belongs in neither a generated client nor eleven re-vendor rounds.
24// Same reasoning as the F3 note on `handlers::oauth2::jwks`.
25//
26// # What is aliased, and what deliberately is not
27//
28// An endpoint belongs here exactly when reaching it over mTLS is meaningful —
29// either the authorization server authenticates the client there (§2), or the
30// caller presents a certificate-bound token there (§3.2):
31//
32// | Endpoint | Why it is aliased |
33// |---|---|
34// | `token_endpoint` | client authentication, and where a bound token is minted |
35// | `revocation_endpoint` | client authentication (RFC 7009 §2.1) |
36// | `introspection_endpoint` | client authentication (RFC 7662 §2.1) |
37// | `device_authorization_endpoint` | client authentication (RFC 8628 §3.1) |
38// | `pushed_authorization_request_endpoint` | client authentication (RFC 9126 §2) |
39// | `backchannel_authentication_endpoint` | client authentication (CIBA Core §7.1) — a `tls_client_auth` CIBA client, the FAPI-CIBA shape, has no other way to present its certificate (D-61; contract §21.3.1 vector A amended to seven) |
40// | `userinfo_endpoint` | a `cnf`-bound access token is refused unless presented on the connection it is bound to |
41//
42// Three endpoints AXIAM publishes are absent, and their absence is the design
43// rather than an omission:
44//
45// - `authorization_endpoint` and `end_session_endpoint` are front-channel: the
46//   visitor is a browser, the authorization server authenticates the *user*
47//   there and never the client, and prompting a browser for a client
48//   certificate produces a native chooser dialog most users cannot answer.
49// - `jwks_uri` is unauthenticated public key material. Putting it behind a
50//   handshake gains nothing and costs every client that only wanted to verify a
51//   signature.
52//
53// `issuer` is absent for a different reason again: it is an identifier, not an
54// endpoint. OIDC Core §2 requires it to match the `iss` of every token exactly,
55// so a per-transport `issuer` would break token validation on the transport
56// that got the alias.
57#[derive(Debug, Serialize, utoipa::ToSchema)]
58pub struct MtlsEndpointAliases {
59    /// RFC 8705 §2 and §3 — client authentication and the mint of a
60    /// certificate-bound token.
61    pub token_endpoint: String,
62    /// OIDC Core §5.3, reached with an access token that may carry `cnf`.
63    pub userinfo_endpoint: String,
64    /// RFC 7009 §2.1 — authenticates the client.
65    pub revocation_endpoint: String,
66    /// RFC 7662 §2.1 — authenticates the caller.
67    pub introspection_endpoint: String,
68    /// RFC 8628 §3.1 — authenticates the client.
69    pub device_authorization_endpoint: String,
70    /// RFC 9126 §2 — authenticates the client.
71    pub pushed_authorization_request_endpoint: String,
72    /// CIBA Core §7.1 — authenticates the client.
73    pub backchannel_authentication_endpoint: String,
74}

270/// Build the endpoint paths for a base URL.
271///
272/// The single place the endpoint *paths* are written down, so a top-level
273/// endpoint and its RFC 8705 §5 alias cannot drift apart. They did in the first
274/// draft of this change — a `/oauth2/device_authorization` alias against a
275/// `/oauth2/device_authorization` top-level is easy to keep in step by hand
276/// right up until one of them is renamed.
277macro_rules! endpoint {
278    ($base:expr, $path:literal) => {
279        format!("{}{}", $base, $path)
280    };
281}

283/// Append `?tenant_id=<uuid>` to an endpoint URL, when this document describes
284/// one tenant.
285///
286/// # Why an endpoint URL carries a query string at all
287///
288/// Because AXIAM's OAuth2 endpoints need one, and OIDC Discovery exists so that
289/// a relying party needs nothing else. Every endpoint that authenticates a
290/// *client* — token, PAR, introspection, revocation, device authorization,
291/// end-session — takes a **required** `tenant_id`, and `/oauth2/authorize`
292/// needs one for a request with no principal, which is every browser arriving
293/// from a relying party. Until this function existed the document advertised
294/// all of them bare, so a client that followed the document exactly got
295/// `400 Query deserialize error: missing field tenant_id` at the token
296/// endpoint. That is what the first OpenID Foundation conformance run hit, and
297/// it is equally what any third-party client would hit.
298///
299/// RFC 6749 §3.1 and §3.2 both allow these endpoints to include a query
300/// component and require the client to **retain** it when adding parameters of
301/// its own, so this is the mechanism the specification provides rather than a
302/// workaround for the lack of one.
303///
304/// `userinfo_endpoint` and `jwks_uri` are deliberately NOT passed through here:
305/// UserInfo resolves the tenant from the bearer token that authenticates the
306/// call, and a JWKS is a deployment-wide document. Adding a parameter neither
307/// reads would be noise a client has to carry.
308fn tenant_scoped(url: String, tenant_id: Option<Uuid>) -> String {
309    match tenant_id {
310        // Every caller passes a URL this module just built from a validated
311        // issuer, so none of them already carries a query — `?` is correct and
312        // `&` would be wrong. Asserted by
313        // `every_tenant_scoped_endpoint_has_exactly_one_query_string`.
314        Some(id) => format!("{url}?tenant_id={id}"),
315        None => url,
316    }
317}

319/// Build the RFC 8705 §5 aliases for a base URL, or `None` when there is no
320/// separate mTLS host.
321///
322/// `Err` when `mtls_base_url` is `Some` but not an absolute URL. That is a
323/// deliberate refusal rather than a silent `None`: the whole point of the
324/// member is to stop an mTLS client using the conventional endpoints, so
325/// dropping it on a typo would produce exactly the failure the operator
326/// configured it to prevent — and would do so invisibly, since a client cannot
327/// tell an absent alias from one the server meant to send.
328fn build_mtls_aliases(
329    mtls_base_url: Option<&str>,
330    tenant_id: Option<Uuid>,
331) -> Result<Option<MtlsEndpointAliases>, String> {
332    let Some(base) = mtls_base_url else {
333        return Ok(None);
334    };
335    let base = base.trim_end_matches('/');
336
337    // `Url::parse` accepts only absolute URLs, which is the property wanted
338    // here: a relative alias is not resolvable against anything a client holds.
339    let parsed = url::Url::parse(base).map_err(|e| format!("not a valid absolute URL: {e}"))?;
340    if !matches!(parsed.scheme(), "https" | "http") {
341        // `http` is tolerated for the same reason the rest of the test and
342        // compose topology tolerates it, and for no other; mutual TLS over
343        // cleartext is a contradiction, so a deployment reaching this with
344        // `http` has a configuration error the operator needs to see rather
345        // than a document to serve.
346        return Err(format!(
347            "scheme must be https (or http for local development), got {:?}",
348            parsed.scheme()
349        ));
350    }
351    // A base URL is concatenated with an endpoint path, so a query or fragment
352    // on it lands in the middle of the result:
353    // `https://mtls.example.com?x=1` + `/oauth2/token` is a URL that parses and
354    // points nowhere. Refusing is the same fail-closed call as the scheme
355    // check — a malformed alias is worse than no alias, because a client will
356    // dutifully try it.
357    if parsed.query().is_some() || parsed.fragment().is_some() {
358        return Err(
359            "must be a bare base URL: a query string or fragment cannot carry an endpoint path"
360                .to_owned(),
361        );
362    }
363
364    // An alias must be usable on arrival, and every one of these but UserInfo
365    // authenticates a client and therefore needs the tenant. An alias that
366    // omitted it would be strictly worse than no alias: RFC 8705 §5 tells an
367    // mTLS client it MUST use these, so it would follow a URL that cannot work
368    // and would have no conventional endpoint to fall back to.
369    Ok(Some(MtlsEndpointAliases {
370        token_endpoint: tenant_scoped(endpoint!(base, "/oauth2/token"), tenant_id),
371        userinfo_endpoint: endpoint!(base, "/oauth2/userinfo"),
372        revocation_endpoint: tenant_scoped(endpoint!(base, "/oauth2/revoke"), tenant_id),
373        introspection_endpoint: tenant_scoped(endpoint!(base, "/oauth2/introspect"), tenant_id),
374        device_authorization_endpoint: tenant_scoped(
375            endpoint!(base, "/oauth2/device_authorization"),
376            tenant_id,
377        ),
378        pushed_authorization_request_endpoint: tenant_scoped(
379            endpoint!(base, "/oauth2/par"),
380            tenant_id,
381        ),
382        backchannel_authentication_endpoint: tenant_scoped(
383            endpoint!(base, "/oauth2/bc-authorize"),
384            tenant_id,
385        ),
386    }))
387}

407/// [`build_discovery_document`], told whether the tenant this document
408/// describes has the X7 G8 sensitive scopes enabled (W7, plan §6).
409///
410/// # The plan assumed a tenant-scoped document; there is not one
411///
412/// Plan §6 heads its table "tenant-scoped discovery document" and gates two
413/// rows on the tenant switch. `GET /.well-known/openid-configuration` takes no
414/// tenant: it is registered at the host root, reads only `AuthConfig`, and
415/// describes the deployment. So the gate had to be built rather than used, and
416/// `false` — omit the two scopes — is what a caller that names no tenant gets.
417///
418/// That default is the honest answer rather than a cautious one. Discovery
419/// says what a relying party may ask for, and with no tenant named the server
420/// genuinely does not know: one tenant may have the capability on and the next
421/// may not. Advertising it unconditionally would tell every relying party in
422/// every tenant that `address` is available, and most of them would be refused
423/// `invalid_scope` on the first request.
424///
425/// Note the contrast with `auth_time`/`acr`/`amr`, added to `claims_supported`
426/// by W1 *unconditionally* with the argument that "discovery describes the
427/// server's capabilities, not any one client's grant". That argument holds
428/// there and not here, and the difference is who decides: those three are a
429/// per-**client** registration, and this is a per-**tenant** switch. A document
430/// that cannot name the client is still truthful about what the server can do
431/// for some client; a document that cannot name the tenant cannot say whether
432/// the capability exists at all in the deployment the caller is talking to.
433/// [`build_discovery_document_for`]'s per-tenant capability inputs.
434///
435/// A struct rather than a growing tail of `bool` parameters: there are two
436/// today and T5 adds a third, and three adjacent booleans at a call site is
437/// three chances to transpose two of them.
438#[derive(Debug, Clone, Copy, Default)]
439pub struct TenantCapabilities {
440    /// X7 G8 (W7) — whether `address` and `phone` are available to this
441    /// tenant.
442    pub sensitive_scopes_enabled: bool,
443    /// T21.4 — whether `POST /oauth2/register` will do anything for this
444    /// tenant. `false` omits `registration_endpoint` entirely.
445    pub dynamic_registration_enabled: bool,
446    /// T21.5 — whether a URL-shaped `client_id` is resolved for this tenant.
447    /// `false` omits `client_id_metadata_document_supported` entirely.
448    pub cimd_enabled: bool,
449}

451pub fn build_discovery_document_for(
452    issuer: &str,
453    mtls_base_url: Option<&str>,
454    capabilities: TenantCapabilities,
455    tenant_id: Option<Uuid>,
456) -> Result<OidcDiscoveryDocument, String> {
457    let TenantCapabilities {
458        sensitive_scopes_enabled,
459        dynamic_registration_enabled,
460        cimd_enabled,
461    } = capabilities;
462    let issuer = issuer.trim_end_matches('/');
463    let mtls_endpoint_aliases = build_mtls_aliases(mtls_base_url, tenant_id)?;
464    // The order is the operator's recommendation, and the three endpoints that
465    // authenticate a client with it publish it (or, for introspection, all of
466    // it but `none`) below.
467    let token_endpoint_auth_methods: Vec<String> = vec![
468        "client_secret_post".into(),
469        // W8 / RFC 6749 §2.3.1. Advertised unconditionally, like the two
470        // mTLS methods and for a narrower version of the same reason:
471        // whether a *particular* client may use it is decided by its
472        // registration, and this document describes the deployment's
473        // capabilities rather than any one client's. A `fapi2` client
474        // reading this list still cannot register for the method —
475        // `validate_registration` refuses it — which is the intended
476        // shape: the server can speak Basic, and the FAPI profile will
477        // not let a client that must not, do so.
478        //
479        // Listed second rather than first: the order is the operator's
480        // recommendation, and `client_secret_post` remains it (the header
481        // channel is the one intermediaries log). SDKs are forbidden from
482        // sending Basic at all — `sdks/CONTRACT.md` §5 rule 3.
483        "client_secret_basic".into(),

681/// JSON Web Key per RFC 7517.
682#[derive(Debug, Serialize, utoipa::ToSchema)]
683pub struct Jwk {
684    pub kty: String,
685    pub crv: String,
686    pub x: String,
687    pub kid: String,
688    #[serde(rename = "use")]
689    pub use_: String,
690    pub alg: String,
691}
692
693/// JSON Web Key Set document.
694#[derive(Debug, Serialize, utoipa::ToSchema)]
695pub struct JwksDocument {
696    pub keys: Vec<Jwk>,
697}
698
699/// Build a JWKS document from an Ed25519 public key in PEM format.
700///
701/// The PEM must contain a SubjectPublicKeyInfo structure (44 bytes
702/// when DER-decoded: 12-byte OID header + 32-byte raw Ed25519 key).
703/// The `kid` is derived deterministically as the first 16 hex
704/// characters (64 bits) of the SHA-256 hash of the raw public key bytes.
705pub fn build_jwks(public_key_pem: &str) -> Result<JwksDocument, String> {
706    // Strip PEM headers, trim whitespace (handles CRLF / trailing
707    // spaces from env vars or Windows-formatted PEMs), and decode.
708    let b64: String = public_key_pem
709        .lines()
710        .map(str::trim)
711        .filter(|l| !l.starts_with("-----"))
712        .collect();
713    let der = STANDARD
714        .decode(&b64)
715        .map_err(|e| format!("PEM decode: {e}"))?;
716
717    // Ed25519 SubjectPublicKeyInfo is exactly 44 bytes:
718    // 12-byte ASN.1/OID header + 32-byte raw public key.
719    if der.len() != 44 {
720        return Err(format!("expected 44-byte Ed25519 SPKI, got {}", der.len()));
721    }
722    let raw_key = &der[12..44];
723
724    // Base64url-encode the raw key bytes for the JWK `x` parameter.
725    let x = URL_SAFE_NO_PAD.encode(raw_key);
726
727    // Deterministic kid: first 16 hex chars of SHA-256(raw_key).
728    //
729    // X7 §1.3 — derived by `axiam_auth::token::ed25519_jwk_kid`, which is also
730    // what stamps the `kid` into every signed header. One definition, because
731    // a JWKS advertising one `kid` while the tokens name another is worse than
732    // no `kid` at all: the relying party looks up a key that is not there and
733    // rejects a good signature. The PEM has already been validated above, so
734    // the `None` arm is unreachable in practice and is answered with the same
735    // message the length check would have given.
736    let kid = axiam_auth::token::ed25519_jwk_kid(public_key_pem)
737        .ok_or_else(|| "expected a 44-byte Ed25519 SPKI".to_owned())?;
738
739    Ok(JwksDocument {
740        keys: vec![Jwk {
741            kty: "OKP".into(),
742            crv: "Ed25519".into(),
743            x,
744            kid,
745            use_: "sig".into(),
746            alg: "EdDSA".into(),
747        }],
748    })
749}
