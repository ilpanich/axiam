12/// RFC 8705 §5 `mtls_endpoint_aliases` — the endpoints re-based on the host
13/// that performs the mutual-TLS handshake.
14///
15/// Present only when the deployment configured a separate mTLS host
16/// (`AXIAM__AUTH__OAUTH2_MTLS_BASE_URL`); a client doing mTLS MUST prefer an
17/// alias over the top-level endpoint of the same name.
18//
19// The rest of the rationale is deliberately NOT a doc comment: utoipa lifts a
20// doc block verbatim into the OpenAPI `description`, and eleven SDK repos
21// vendor `sdks/openapi.json` byte-for-byte. A design argument belongs in the
22// source, where it is read by whoever changes this; a two-kilobyte markdown
23// table belongs in neither a generated client nor eleven re-vendor rounds.
24// Same reasoning as the F3 note on `handlers::oauth2::jwks`.
25//
26// # What is aliased, and what deliberately is not
27//
28// An endpoint belongs here exactly when reaching it over mTLS is meaningful —
29// either the authorization server authenticates the client there (§2), or the
30// caller presents a certificate-bound token there (§3.2):
31//
32// | Endpoint | Why it is aliased |
33// |---|---|
34// | `token_endpoint` | client authentication, and where a bound token is minted |
35// | `revocation_endpoint` | client authentication (RFC 7009 §2.1) |
36// | `introspection_endpoint` | client authentication (RFC 7662 §2.1) |
37// | `device_authorization_endpoint` | client authentication (RFC 8628 §3.1) |
38// | `pushed_authorization_request_endpoint` | client authentication (RFC 9126 §2) |
39// | `userinfo_endpoint` | a `cnf`-bound access token is refused unless presented on the connection it is bound to |
40//
41// Three endpoints AXIAM publishes are absent, and their absence is the design
42// rather than an omission:
43//
44// - `authorization_endpoint` and `end_session_endpoint` are front-channel: the
45//   visitor is a browser, the authorization server authenticates the *user*
46//   there and never the client, and prompting a browser for a client
47//   certificate produces a native chooser dialog most users cannot answer.
48// - `jwks_uri` is unauthenticated public key material. Putting it behind a
49//   handshake gains nothing and costs every client that only wanted to verify a
50//   signature.
51//
52// `issuer` is absent for a different reason again: it is an identifier, not an
53// endpoint. OIDC Core §2 requires it to match the `iss` of every token exactly,
54// so a per-transport `issuer` would break token validation on the transport
55// that got the alias.
56#[derive(Debug, Serialize, utoipa::ToSchema)]
57pub struct MtlsEndpointAliases {
58    /// RFC 8705 §2 and §3 — client authentication and the mint of a
59    /// certificate-bound token.
60    pub token_endpoint: String,
61    /// OIDC Core §5.3, reached with an access token that may carry `cnf`.
62    pub userinfo_endpoint: String,
63    /// RFC 7009 §2.1 — authenticates the client.
64    pub revocation_endpoint: String,
65    /// RFC 7662 §2.1 — authenticates the caller.
66    pub introspection_endpoint: String,
67    /// RFC 8628 §3.1 — authenticates the client.
68    pub device_authorization_endpoint: String,
69    /// RFC 9126 §2 — authenticates the client.
70    pub pushed_authorization_request_endpoint: String,
71}

87#[derive(Debug, Serialize, utoipa::ToSchema)]
88pub struct OidcDiscoveryDocument {
89    pub issuer: String,
90    pub authorization_endpoint: String,
91    pub token_endpoint: String,
92    pub userinfo_endpoint: String,
93    pub jwks_uri: String,
94    pub revocation_endpoint: String,
95    pub introspection_endpoint: String,

112    /// Back-Channel Logout 1.0 §3.
113    pub backchannel_logout_supported: bool,

119    pub subject_types_supported: Vec<String>,

128    /// is not made conditional.

130    /// RFC 8705 §3.3 — X5.1. Certificate-bound access tokens are available;
131    /// whether a given client *gets* them is that client's

140    ///
141    /// `request_uri_parameter_supported` is deliberately **absent** rather
142    /// than `false`: its default is `true`, and that is the truthful answer —

182    /// RFC 8414 §2 — the JWS algorithms AXIAM accepts on a `private_key_jwt`

185    /// Derived from [`crate::jose::permitted_algorithm_names`] rather than

195    /// RFC 8705 §5 — the mTLS-specific endpoint URLs, when this deployment

198    //
199    // The `skip_serializing_if` is load-bearing, not cosmetic: a *present*

201    // `"mtls_endpoint_aliases": null` on the single-listener deployment would

203    #[serde(skip_serializing_if = "Option::is_none")]
204    pub mtls_endpoint_aliases: Option<MtlsEndpointAliases>,
205}

207/// Build the endpoint paths for a base URL.
208///
209/// The single place the endpoint *paths* are written down, so a top-level
210/// endpoint and its RFC 8705 §5 alias cannot drift apart. They did in the first
211/// draft of this change — a `/oauth2/device_authorization` alias against a
212/// `/oauth2/device_authorization` top-level is easy to keep in step by hand
213/// right up until one of them is renamed.
214macro_rules! endpoint {
215    ($base:expr, $path:literal) => {
216        format!("{}{}", $base, $path)
217    };
218}

220/// Append `?tenant_id=<uuid>` to an endpoint URL, when this document describes
221/// one tenant.
222///
223/// # Why an endpoint URL carries a query string at all
224///
225/// Because AXIAM's OAuth2 endpoints need one, and OIDC Discovery exists so that
226/// a relying party needs nothing else. Every endpoint that authenticates a
227/// *client* — token, PAR, introspection, revocation, device authorization,
228/// end-session — takes a **required** `tenant_id`, and `/oauth2/authorize`
229/// needs one for a request with no principal, which is every browser arriving
230/// from a relying party. Until this function existed the document advertised
231/// all of them bare, so a client that followed the document exactly got
232/// `400 Query deserialize error: missing field tenant_id` at the token
233/// endpoint. That is what the first OpenID Foundation conformance run hit, and
234/// it is equally what any third-party client would hit.
235///
236/// RFC 6749 §3.1 and §3.2 both allow these endpoints to include a query
237/// component and require the client to **retain** it when adding parameters of
238/// its own, so this is the mechanism the specification provides rather than a
239/// workaround for the lack of one.
240///
241/// `userinfo_endpoint` and `jwks_uri` are deliberately NOT passed through here:
242/// UserInfo resolves the tenant from the bearer token that authenticates the
243/// call, and a JWKS is a deployment-wide document. Adding a parameter neither
244/// reads would be noise a client has to carry.
245fn tenant_scoped(url: String, tenant_id: Option<Uuid>) -> String {
246    match tenant_id {
247        // Every caller passes a URL this module just built from a validated
248        // issuer, so none of them already carries a query — `?` is correct and
249        // `&` would be wrong. Asserted by
250        // `every_tenant_scoped_endpoint_has_exactly_one_query_string`.
251        Some(id) => format!("{url}?tenant_id={id}"),
252        None => url,
253    }
254}

256/// Build the RFC 8705 §5 aliases for a base URL, or `None` when there is no
257/// separate mTLS host.
258///
259/// `Err` when `mtls_base_url` is `Some` but not an absolute URL. That is a
260/// deliberate refusal rather than a silent `None`: the whole point of the
261/// member is to stop an mTLS client using the conventional endpoints, so
262/// dropping it on a typo would produce exactly the failure the operator
263/// configured it to prevent — and would do so invisibly, since a client cannot
264/// tell an absent alias from one the server meant to send.
265fn build_mtls_aliases(
266    mtls_base_url: Option<&str>,
267    tenant_id: Option<Uuid>,
268) -> Result<Option<MtlsEndpointAliases>, String> {
269    let Some(base) = mtls_base_url else {
270        return Ok(None);
271    };
272    let base = base.trim_end_matches('/');
273
274    // `Url::parse` accepts only absolute URLs, which is the property wanted
275    // here: a relative alias is not resolvable against anything a client holds.
276    let parsed = url::Url::parse(base).map_err(|e| format!("not a valid absolute URL: {e}"))?;
277    if !matches!(parsed.scheme(), "https" | "http") {
278        // `http` is tolerated for the same reason the rest of the test and
279        // compose topology tolerates it, and for no other; mutual TLS over
280        // cleartext is a contradiction, so a deployment reaching this with
281        // `http` has a configuration error the operator needs to see rather
282        // than a document to serve.
283        return Err(format!(
284            "scheme must be https (or http for local development), got {:?}",
285            parsed.scheme()
286        ));
287    }
288    // A base URL is concatenated with an endpoint path, so a query or fragment
289    // on it lands in the middle of the result:
290    // `https://mtls.example.com?x=1` + `/oauth2/token` is a URL that parses and
291    // points nowhere. Refusing is the same fail-closed call as the scheme
292    // check — a malformed alias is worse than no alias, because a client will
293    // dutifully try it.
294    if parsed.query().is_some() || parsed.fragment().is_some() {
295        return Err(
296            "must be a bare base URL: a query string or fragment cannot carry an endpoint path"
297                .to_owned(),
298        );
299    }
300
301    // An alias must be usable on arrival, and every one of these but UserInfo
302    // authenticates a client and therefore needs the tenant. An alias that
303    // omitted it would be strictly worse than no alias: RFC 8705 §5 tells an
304    // mTLS client it MUST use these, so it would follow a URL that cannot work
305    // and would have no conventional endpoint to fall back to.
306    Ok(Some(MtlsEndpointAliases {
307        token_endpoint: tenant_scoped(endpoint!(base, "/oauth2/token"), tenant_id),
308        userinfo_endpoint: endpoint!(base, "/oauth2/userinfo"),
309        revocation_endpoint: tenant_scoped(endpoint!(base, "/oauth2/revoke"), tenant_id),
310        introspection_endpoint: tenant_scoped(endpoint!(base, "/oauth2/introspect"), tenant_id),
311        device_authorization_endpoint: tenant_scoped(
312            endpoint!(base, "/oauth2/device_authorization"),
313            tenant_id,
314        ),
315        pushed_authorization_request_endpoint: tenant_scoped(
316            endpoint!(base, "/oauth2/par"),
317            tenant_id,
318        ),
319    }))
320}

322/// Build a fully-populated OIDC discovery document for the given issuer URL.
323///
324/// `mtls_base_url` is the deployment's separate mutual-TLS host, when it has
325/// one; `None` omits RFC 8705 §5 `mtls_endpoint_aliases` from the document.
326///
327/// # Errors
328///
329/// Only when `mtls_base_url` is `Some` and not a usable absolute URL. Every
330/// other field is derived from `issuer`, whose validation is the caller's —
331/// the discovery handler parses it before calling and answers `500` itself, and
332/// that check predates this one.
333pub fn build_discovery_document(
334    issuer: &str,
335    mtls_base_url: Option<&str>,
336) -> Result<OidcDiscoveryDocument, String> {
337    build_discovery_document_for(issuer, mtls_base_url, false, None)
338}

340/// [`build_discovery_document`], told whether the tenant this document
341/// describes has the X7 G8 sensitive scopes enabled (W7, plan §6).
342///
343/// # The plan assumed a tenant-scoped document; there is not one
344///
345/// Plan §6 heads its table "tenant-scoped discovery document" and gates two
346/// rows on the tenant switch. `GET /.well-known/openid-configuration` takes no
347/// tenant: it is registered at the host root, reads only `AuthConfig`, and
348/// describes the deployment. So the gate had to be built rather than used, and
349/// `false` — omit the two scopes — is what a caller that names no tenant gets.
350///
351/// That default is the honest answer rather than a cautious one. Discovery
352/// says what a relying party may ask for, and with no tenant named the server
353/// genuinely does not know: one tenant may have the capability on and the next
354/// may not. Advertising it unconditionally would tell every relying party in
355/// every tenant that `address` is available, and most of them would be refused
356/// `invalid_scope` on the first request.
357///
358/// Note the contrast with `auth_time`/`acr`/`amr`, added to `claims_supported`
359/// by W1 *unconditionally* with the argument that "discovery describes the
360/// server's capabilities, not any one client's grant". That argument holds
361/// there and not here, and the difference is who decides: those three are a
362/// per-**client** registration, and this is a per-**tenant** switch. A document
363/// that cannot name the client is still truthful about what the server can do
364/// for some client; a document that cannot name the tenant cannot say whether
365/// the capability exists at all in the deployment the caller is talking to.
366pub fn build_discovery_document_for(
367    issuer: &str,
368    mtls_base_url: Option<&str>,
369    sensitive_scopes_enabled: bool,
370    tenant_id: Option<Uuid>,
371) -> Result<OidcDiscoveryDocument, String> {
372    let issuer = issuer.trim_end_matches('/');
373    let mtls_endpoint_aliases = build_mtls_aliases(mtls_base_url, tenant_id)?;
374    let mut doc = OidcDiscoveryDocument {
375        issuer: issuer.to_string(),
376        // `tenant_scoped` on everything that authenticates a client, and its
377        // doc comment is where the argument lives. UserInfo and the JWKS are
378        // bare on purpose: one resolves the tenant from its bearer token, the
379        // other is deployment-wide.
380        authorization_endpoint: tenant_scoped(format!("{issuer}/oauth2/authorize"), tenant_id),
381        token_endpoint: tenant_scoped(endpoint!(issuer, "/oauth2/token"), tenant_id),
382        userinfo_endpoint: endpoint!(issuer, "/oauth2/userinfo"),
383        jwks_uri: format!("{issuer}/oauth2/jwks"),
384        revocation_endpoint: tenant_scoped(endpoint!(issuer, "/oauth2/revoke"), tenant_id),
385        introspection_endpoint: tenant_scoped(endpoint!(issuer, "/oauth2/introspect"), tenant_id),
386        device_authorization_endpoint: tenant_scoped(
387            endpoint!(issuer, "/oauth2/device_authorization"),
388            tenant_id,
389        ),
390        pushed_authorization_request_endpoint: tenant_scoped(
391            endpoint!(issuer, "/oauth2/par"),
392            tenant_id,
393        ),
394        require_pushed_authorization_requests: false,
395        end_session_endpoint: tenant_scoped(format!("{issuer}/oauth2/end_session"), tenant_id),
396        backchannel_logout_supported: true,
397        backchannel_logout_session_supported: true,
398        response_types_supported: vec!["code".into()],
399        subject_types_supported: vec!["public".into()],
400        id_token_signing_alg_values_supported: vec!["EdDSA".into()],
401        scopes_supported: {
402            let mut scopes = vec!["openid".into(), "profile".into(), "email".into()];
403            if sensitive_scopes_enabled {
404                scopes.extend(crate::sensitive::SENSITIVE_SCOPES.map(String::from));
405            }
406            scopes
407        },
408        token_endpoint_auth_methods_supported: vec![
409            "client_secret_post".into(),
410            // W8 / RFC 6749 §2.3.1. Advertised unconditionally, like the two
411            // mTLS methods and for a narrower version of the same reason:
412            // whether a *particular* client may use it is decided by its
413            // registration, and this document describes the deployment's
414            // capabilities rather than any one client's. A `fapi2` client
415            // reading this list still cannot register for the method —
416            // `validate_registration` refuses it — which is the intended
417            // shape: the server can speak Basic, and the FAPI profile will
418            // not let a client that must not, do so.
419            //
420            // Listed second rather than first: the order is the operator's
421            // recommendation, and `client_secret_post` remains it (the header
422            // channel is the one intermediaries log). SDKs are forbidden from
423            // sending Basic at all — `sdks/CONTRACT.md` §5 rule 3.
424            "client_secret_basic".into(),
425            // X5.1 / RFC 8705 §2. Advertised unconditionally: whether a mTLS
426            // handshake is actually available is a deployment's listener
427            // configuration (the p3 profile), and a client that cannot reach
428            // an mTLS listener discovers that at connect time rather than by
429            // reading a metadata field that would have to lie one way or the
430            // other on a multi-listener deployment.
431            "tls_client_auth".into(),
432            "self_signed_tls_client_auth".into(),
433            // X5.1 second half / RFC 7523 §2.2. Unlike the two mTLS methods
434            // this one needs nothing from the deployment's listeners at all,
435            // so advertising it carries no caveat: every AXIAM deployment can
436            // serve it.
437            "private_key_jwt".into(),
438        ],
439        claims_supported: vec![
440            "sub".into(),
441            "iss".into(),
442            "aud".into(),
443            "exp".into(),
444            "iat".into(),
445            "nonce".into(),
446            "email".into(),
447            // OIDC Core §5.1 — advertised because UserInfo releases it
448            // whenever it releases `email`, and a relying party reading
449            // `claims_supported` to decide whether it can trust an address
450            // needs to know the verification status is available.
451            "email_verified".into(),
452            "preferred_username".into(),
453            // The `profile` scope's claims, all of which AXIAM can now hold
454            // and release — see `axiam_core::models::user::ProfileClaims` for
455            // where each is stored. Advertised because a relying party reads
456            // `claims_supported` to decide what it can ask for; a claim listed
457            // here is one AXIAM *can* assert, not one every subject has, and
458            // §5.3.2 governs the difference at release time.
459            "name".into(),
460            "given_name".into(),
461            "family_name".into(),
462            "middle_name".into(),
463            "nickname".into(),
464            "profile".into(),
465            "picture".into(),
466            "website".into(),
467            "gender".into(),
468            "birthdate".into(),
469            "zoneinfo".into(),
470            "locale".into(),
471            "updated_at".into(),
472            "tenant_id".into(),
473            "org_id".into(),
474            // X7 — the three authentication-evidence claims. Advertised as
475            // claims AXIAM *can* assert; which clients receive them is their
476            // own `authn_request_params` registration, and in this wave the
477            // answer is none of them. Discovery describes the server's
478            // capabilities, not any one client's grant — the same distinction
479            // `require_pushed_authorization_requests` above draws.
480            "auth_time".into(),
481            "acr".into(),
482            "amr".into(),
483        ],
484        // X7 G8 (W7). Appended after the closing bracket above rather than
485        // inside it because these three are conditional and the ten are not —
486        // and a `Vec` built by one expression with an `if` in the middle of it
487        // is a `Vec` whose unconditional members are hard to read off.
488        grant_types_supported: vec![
489            "authorization_code".into(),
490            "client_credentials".into(),
491            "refresh_token".into(),
492            // B2: the URN is the grant's identifier on the wire; a device
493            // matches on this exact string, so it is spelled out rather than
494            // referenced, and the constant it must equal
495            // (`device_service::DEVICE_CODE_GRANT_TYPE`) is asserted against
496            // it in this module's tests.
497            "urn:ietf:params:oauth:grant-type:device_code".into(),
498            // B3 / RFC 8693. Advertised so a mesh service can discover that
499            // narrowing a token is possible here rather than being told out
500            // of band; whether a given client MAY exchange is still its own
501            // registration's business.
502            "urn:ietf:params:oauth:grant-type:token-exchange".into(),
503        ],
504        authorization_response_iss_parameter_supported: true,
505        request_parameter_supported: false,
506        // OIDC Core §5.5. `true` since AXIAM honours the `userinfo` member of

512        acr_values_supported: vec![ACR_SINGLE_FACTOR.into(), ACR_MULTI_FACTOR.into()],

530/// JSON Web Key per RFC 7517.
531#[derive(Debug, Serialize, utoipa::ToSchema)]
532pub struct Jwk {
533    pub kty: String,
534    pub crv: String,
535    pub x: String,
536    pub kid: String,
537    #[serde(rename = "use")]
538    pub use_: String,
539    pub alg: String,
540}
541
542/// JSON Web Key Set document.
543#[derive(Debug, Serialize, utoipa::ToSchema)]
544pub struct JwksDocument {
545    pub keys: Vec<Jwk>,
546}
547
548/// Build a JWKS document from an Ed25519 public key in PEM format.
549///
550/// The PEM must contain a SubjectPublicKeyInfo structure (44 bytes
551/// when DER-decoded: 12-byte OID header + 32-byte raw Ed25519 key).
552/// The `kid` is derived deterministically as the first 16 hex
553/// characters (64 bits) of the SHA-256 hash of the raw public key bytes.
554pub fn build_jwks(public_key_pem: &str) -> Result<JwksDocument, String> {
555    // Strip PEM headers, trim whitespace (handles CRLF / trailing
556    // spaces from env vars or Windows-formatted PEMs), and decode.
557    let b64: String = public_key_pem
558        .lines()
559        .map(str::trim)
560        .filter(|l| !l.starts_with("-----"))
561        .collect();
562    let der = STANDARD
563        .decode(&b64)
564        .map_err(|e| format!("PEM decode: {e}"))?;
565
566    // Ed25519 SubjectPublicKeyInfo is exactly 44 bytes:
567    // 12-byte ASN.1/OID header + 32-byte raw public key.
568    if der.len() != 44 {
569        return Err(format!("expected 44-byte Ed25519 SPKI, got {}", der.len()));
570    }
571    let raw_key = &der[12..44];
572
573    // Base64url-encode the raw key bytes for the JWK `x` parameter.
574    let x = URL_SAFE_NO_PAD.encode(raw_key);
575
576    // Deterministic kid: first 16 hex chars of SHA-256(raw_key).
577    //
578    // X7 §1.3 — derived by `axiam_auth::token::ed25519_jwk_kid`, which is also
579    // what stamps the `kid` into every signed header. One definition, because
580    // a JWKS advertising one `kid` while the tokens name another is worse than
581    // no `kid` at all: the relying party looks up a key that is not there and
582    // rejects a good signature. The PEM has already been validated above, so
583    // the `None` arm is unreachable in practice and is answered with the same
584    // message the length check would have given.
585    let kid = axiam_auth::token::ed25519_jwk_kid(public_key_pem)
586        .ok_or_else(|| "expected a 44-byte Ed25519 SPKI".to_owned())?;
587
588    Ok(JwksDocument {
589        keys: vec![Jwk {
590            kty: "OKP".into(),
591            crv: "Ed25519".into(),
592            x,
593            kid,
594            use_: "sig".into(),
595            alg: "EdDSA".into(),
596        }],
597    })
598}

604/// OIDC UserInfo response.
605///
606/// # The three W7 claims are absent unless four gates all said yes
607///
608/// `phone_number`, `phone_number_verified` and `address` (X7 G8) are the only
609/// members of this type that are not derivable from the token alone. They are
610/// present only when the tenant switch is on, the access token carries the
611/// scope, the relying party the token names holds a live consent record, and
612/// that relying party is not on the `fapi2` profile — see
613/// `crate::sensitive` for why each of the four exists and who closes it.
614///
615/// `Debug` is manual so that neither value can reach a log line through the
616/// most natural diagnostic anybody writes, the same rule
617/// `axiam_core::models::user::User` follows.
618#[derive(Serialize, utoipa::ToSchema)]
619pub struct UserInfoResponse {
620    pub sub: String,
621    #[serde(skip_serializing_if = "Option::is_none")]
622    pub email: Option<String>,
623    /// Whether [`Self::email`] has been verified (OIDC Core §5.1).
624    ///
625    /// Emitted only alongside the address, for [`Self::phone_number_verified`]'s
626    /// reason: it is a statement *about* a value, and making it about one the
627    /// relying party was not given asserts nothing it can act on.
628    ///
629    /// `email` scope promises both members, and the OIDF suite says so —
630    /// `VerifyScopesReturnedInUserInfoClaims` listed `email_verified` as
631    /// missing while `email` itself was released. AXIAM has always held the
632    /// fact (`User::email_verified_at`); it simply had nowhere to put it.
633    ///
634    /// `false` rather than omitted when the address is present and
635    /// unverified. AXIAM does run an email verification ceremony, so unlike
636    /// the telephone case this is a claim with a real ceremony behind it, and
637    /// "we have not verified this" is the answer a relying party needs in
638    /// order to decide whether to trust the address.
639    #[serde(skip_serializing_if = "Option::is_none")]
640    pub email_verified: Option<bool>,
641    #[serde(skip_serializing_if = "Option::is_none")]
642    pub preferred_username: Option<String>,
643    /// OIDC Core §5.1 `name`, released under the `profile` scope.
644    ///
645    /// This and the eleven that follow are
646    /// `axiam_core::models::user::ProfileClaims`, which documents where each
647    /// one is stored. Every one is omitted rather than `null` when AXIAM holds
648    /// nothing: §5.3.2 says a claim the OP cannot assert is simply absent, and
649    /// a `null` would be AXIAM asserting the subject has no name.
650    #[serde(skip_serializing_if = "Option::is_none")]
651    pub name: Option<String>,
652    /// OIDC Core §5.1 `given_name` — SCIM `name.givenName`.
653    #[serde(skip_serializing_if = "Option::is_none")]
654    pub given_name: Option<String>,
655    /// OIDC Core §5.1 `family_name` — SCIM `name.familyName`.
656    #[serde(skip_serializing_if = "Option::is_none")]
657    pub family_name: Option<String>,
658    /// OIDC Core §5.1 `middle_name` — SCIM `name.middleName`.
659    #[serde(skip_serializing_if = "Option::is_none")]
660    pub middle_name: Option<String>,
661    /// OIDC Core §5.1 `nickname` — SCIM `nickName`.
662    #[serde(skip_serializing_if = "Option::is_none")]
663    pub nickname: Option<String>,
664    /// OIDC Core §5.1 `profile` — SCIM `profileUrl`.
665    #[serde(skip_serializing_if = "Option::is_none")]
666    pub profile: Option<String>,
667    /// OIDC Core §5.1 `picture` — SCIM `photos`.
668    #[serde(skip_serializing_if = "Option::is_none")]
669    pub picture: Option<String>,
670    /// OIDC Core §5.1 `website` — `metadata.oidc.website`.
671    #[serde(skip_serializing_if = "Option::is_none")]
672    pub website: Option<String>,
673    /// OIDC Core §5.1 `gender` — `metadata.oidc.gender`.
674    #[serde(skip_serializing_if = "Option::is_none")]
675    pub gender: Option<String>,
676    /// OIDC Core §5.1 `birthdate` — `metadata.oidc.birthdate`.
677    #[serde(skip_serializing_if = "Option::is_none")]
678    pub birthdate: Option<String>,
679    /// OIDC Core §5.1 `zoneinfo` — SCIM `timezone`.
680    #[serde(skip_serializing_if = "Option::is_none")]
681    pub zoneinfo: Option<String>,
682    /// OIDC Core §5.1 `locale` — SCIM `locale`.
683    #[serde(skip_serializing_if = "Option::is_none")]
684    pub locale: Option<String>,
685    /// OIDC Core §5.1 `updated_at` — seconds since the epoch.
686    ///
687    /// From `User::updated_at`, a column, and deliberately not from metadata:
688    /// it says when AXIAM last changed the row, which is not something a
689    /// provisioning client should be able to assert about AXIAM.
690    ///
691    /// A NumericDate, per §5.1's own table — an RFC 3339 string here is the
692    /// obvious and wrong thing, and `EnsureUserInfoUpdatedAtValid` is the
693    /// module that says so.
694    #[serde(skip_serializing_if = "Option::is_none")]
695    pub updated_at: Option<i64>,
696    /// OIDC Core §5.1, released under the `phone` scope (X7 G8).
697    #[serde(skip_serializing_if = "Option::is_none")]
698    pub phone_number: Option<String>,
699    /// Whether [`Self::phone_number`] has been verified out of band.
700    ///
701    /// Emitted **only alongside** the number, per OIDC Core §5.1: a
702    /// `phone_number_verified` with no `phone_number` asserts something about
703    /// a value the relying party was not given. `false` rather than omitted
704    /// when the number is present and unverified — that is a statement AXIAM

1255    /// X7 §1.3 — the JWKS `kid` and the `kid` stamped into every signed header
1256    /// come from **one** derivation. A JWKS publishing one `kid` while the
1257    /// tokens name another is worse than no `kid` at all: the relying party
1258    /// looks up a key that is not there and rejects a good signature.
1259    #[test]
1260    fn the_published_kid_is_the_one_the_signer_stamps() {

1265        let published = build_jwks(pem).unwrap().keys[0].kid.clone();
1266        let stamped =
1267            axiam_auth::token::ed25519_jwk_kid(pem).expect("a valid Ed25519 SPKI yields a kid");
1268        assert_eq!(published, stamped);
1269        // 64 bits of SHA-256, hex-encoded.
1270        assert_eq!(published.len(), 16);
1271        assert!(published.chars().all(|c| c.is_ascii_hexdigit()));
