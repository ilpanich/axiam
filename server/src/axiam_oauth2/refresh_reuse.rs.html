1//! G-C (#818) — what a **refused** refresh-token replay does besides refusing.
2//!
3//! T-254 made every presentation of an already-rotated refresh token visible:
4//! marked on the session, audited as `oauth2.refresh_token_replayed`. It did
5//! not revoke anything. This module is where the token endpoint learns, per
6//! tenant, whether it now should: the tenant's effective
7//! `oauth2.refresh_reuse_response` (the organization baseline merged with the
8//! tenant override), `revoke_session` unless somebody chose `refuse`.
9//!
10//! # Why a port
11//!
12//! For the reason [`crate::scope_ceiling::ExternalClientScopePolicy`] is one:
13//! the settings live in a repository `TokenService` has no type parameter for,
14//! and the composition root builds one value over the settings repository.
15//!
16//! # Read only on the refusal path
17//!
18//! The policy is asked only after a replay has been recognised and refused, so
19//! a grant that succeeds — every honest request — pays nothing for it.
20//!
21//! # A read that fails, or no port at all, means the default
22//!
23//! Unlike the scope ceiling, a failed read here does not refuse anything: the
24//! refusal has already been decided. It falls back to
25//! [`RefreshReuseResponse::default`], `revoke_session`, which is both the
26//! system default (D-N2) and the stricter of the two values — a datastore
27//! fault can make the endpoint revoke where a tenant asked it only to refuse,
28//! never the reverse. A service built without a port ([`default_policy`])
29//! behaves the same way.
30
31use std::future::Future;
32use std::pin::Pin;
33use std::sync::Arc;
34
35use axiam_core::error::AxiamResult;
36use axiam_core::models::settings::RefreshReuseResponse;
37use uuid::Uuid;
38
39/// Where a tenant's effective `oauth2.refresh_reuse_response` is read from.
40pub trait RefreshReusePolicy: Send + Sync {
41    /// The tenant's **effective** response to a refused refresh-token replay.
42    ///
43    /// An `Err` is a failed read; the caller falls back to the default.
44    fn refresh_reuse_response<'a>(
45        &'a self,
46        tenant_id: Uuid,
47    ) -> Pin<Box<dyn Future<Output = AxiamResult<RefreshReuseResponse>> + Send + 'a>>;
48}
49
50/// The shared form every service holds.
51pub type SharedRefreshReusePolicy = Arc<dyn RefreshReusePolicy>;
52
53/// A policy that answers the same response for every tenant.
54///
55/// What a service holds until its composition root wires the settings-backed
56/// one, and what a test uses to pin either behaviour.
57#[derive(Debug, Clone, Copy)]
58pub struct FixedRefreshReusePolicy(pub RefreshReuseResponse);
59
60impl RefreshReusePolicy for FixedRefreshReusePolicy {
61    fn refresh_reuse_response<'a>(
62        &'a self,
63        _tenant_id: Uuid,
64    ) -> Pin<Box<dyn Future<Output = AxiamResult<RefreshReuseResponse>> + Send + 'a>> {
65        let response = self.0;
66        Box::pin(async move { Ok(response) })
67    }
68}
69
70/// The policy a service holds when nothing else is wired: the system default,
71/// `revoke_session`, for every tenant.
72pub fn default_policy() -> SharedRefreshReusePolicy {
73    fixed(RefreshReuseResponse::default())
74}
75
76/// A [`SharedRefreshReusePolicy`] answering `response` for every tenant.
77pub fn fixed(response: RefreshReuseResponse) -> SharedRefreshReusePolicy {
78    Arc::new(FixedRefreshReusePolicy(response))
79}
80
81/// The response to apply for `tenant_id`, never failing: a failed read is
82/// logged and answered with the default (see the module documentation).
83pub async fn resolve(policy: &dyn RefreshReusePolicy, tenant_id: Uuid) -> RefreshReuseResponse {
84    match policy.refresh_reuse_response(tenant_id).await {
85        Ok(response) => response,
86        Err(e) => {
87            let fallback = RefreshReuseResponse::default();
88            tracing::error!(
89                error = %e,
90                %tenant_id,
91                fallback = fallback.as_str(),
92                "oauth2: could not read the tenant's refresh_reuse_response; applying the default"
93            );
94            fallback
95        }
96    }
97}

99#[cfg(test)]
100mod tests {
101    use super::*;
102    use axiam_core::error::AxiamError;
103
104    struct Failing;
105
106    impl RefreshReusePolicy for Failing {
107        fn refresh_reuse_response<'a>(
108            &'a self,
109            _tenant_id: Uuid,
110        ) -> Pin<Box<dyn Future<Output = AxiamResult<RefreshReuseResponse>> + Send + 'a>> {
111            Box::pin(async { Err(AxiamError::Database("settings unreadable".into())) })
112        }
113    }
114
115    #[tokio::test]
116    async fn an_unwired_service_revokes_by_default() {
117        assert_eq!(
118            resolve(default_policy().as_ref(), Uuid::new_v4()).await,
119            RefreshReuseResponse::RevokeSession
120        );
121    }
122
123    #[tokio::test]
124    async fn a_fixed_policy_answers_what_it_was_given() {
125        assert_eq!(
126            resolve(fixed(RefreshReuseResponse::Refuse).as_ref(), Uuid::new_v4()).await,
127            RefreshReuseResponse::Refuse
128        );
129    }
130
131    #[tokio::test]
132    async fn a_failed_read_falls_back_to_the_stricter_default() {
133        assert_eq!(
134            resolve(&Failing, Uuid::new_v4()).await,
135            RefreshReuseResponse::RevokeSession,
136            "a datastore fault may make the endpoint revoke, never stop it revoking"
137        );
138    }
