1//! #613 (T-289) — the tenant's `dcr_allowed_scopes` as a **ceiling** on what an
2//! externally registered client may obtain, applied where scopes are read.
3//!
4//! # The gap this closes
5//!
6//! `dcr_allowed_scopes` bounds what a self-registered client (`managed_by:
7//! dcr`) or a client ID metadata document (`managed_by: cimd`) may *register*.
8//! It was checked at registration, at an RFC 7592 `PUT` and at a CIMD fetch —
9//! and nowhere else. A tenant that withdrew a scope from the list therefore
10//! narrowed nothing already registered: the client row kept the scope, every
11//! grant that reads `client.scopes` kept offering it, and the refresh grant's
12//! `stored.scopes ∩ client.scopes` (#520) intersected against a row that still
13//! held it. The withdrawal took effect only when the client itself chose to
14//! re-register.
15//!
16//! # Enforcement at the read points, not a rewrite of rows
17//!
18//! The settings handler does not narrow client rows when the list changes.
19//! Rewriting rows would be a second writer on `oauth2_client` racing RFC 7592
20//! `PUT`s and CIMD re-materialisations, and it would make the *row* say
21//! something the client never registered. Instead every point that decides
22//! what a token may carry reads the client's scopes through
23//! [`apply_to_client`], which intersects them with the tenant's **effective**
24//! list (the organization baseline merged with the tenant override, exactly
25//! what registration validates against). A withdrawn scope is gone at the
26//! client's next request, whatever the row says; a scope put back is back at
27//! the next request too, because nothing was deleted.
28//!
29//! The read points, each of which calls into this module:
30//!
31//! | Path | Where | What is narrowed |
32//! |---|---|---|
33//! | `/oauth2/authorize` (query, PAR `request_uri`) | `AuthorizeService::authorize` step 5 | the scopes a request may name |
34//! | `authorization_code` | `TokenService::handle_authorization_code` | the code's scopes, before it is spent |
35//! | `client_credentials` | `TokenService::handle_client_credentials` | requested / default scopes |
36//! | `refresh_token` | `TokenService::handle_refresh_token` | `stored.scopes ∩ client.scopes` |
37//! | CIBA token (`urn:openid:params:grant-type:ciba`) | `TokenService::exchange_ciba` | the approved request's scopes |
38//! | `/oauth2/bc-authorize` | the REST handler, through `TokenService::apply_scope_ceiling` | the scopes a request may name |
39//! | token exchange (RFC 8693) | the REST handler, through `TokenService::apply_scope_ceiling` | the client side of the narrowing |
40//! | `/oauth2/device_authorization` | `DeviceAuthorizationService::authorize` | the scopes the device grant records |
41//!
42//! # Who pays for it
43//!
44//! Only an external client. [`applies_to`] is answered from the client row the
45//! path already holds, so a client an administrator created — every client in
46//! most deployments — costs no read and sees byte-identical behaviour. An
47//! external client costs one effective-settings read per request on the paths
48//! above, which is the read `materialise_if_cimd` already makes for every
49//! URL-shaped `client_id`. Nothing is cached: a cache would be the staleness
50//! this module exists to remove.
51//!
52//! # A read that fails refuses
53//!
54//! For an external client a failed policy read is a `server_error`, never a
55//! fall-back to the row's scopes: the row is exactly the value this module
56//! exists to distrust. An unwired policy ([`unwired`]) is the same refusal, so
57//! a composition root that forgets the port fails loudly on the first DCR
58//! request rather than silently restoring the gap.
59//!
60//! # What it does not do
61//!
62//! An access token already minted keeps its `scope` until it expires (15
63//! minutes by default): it is a stateless JWT and nothing re-reads it. The
64//! refresh token behind it is narrowed at its next use. A device code pending
65//! when the list changes mints with the scopes recorded at
66//! `/oauth2/device_authorization` (its poll never reads the client), and the
67//! refresh token it yields is narrowed at its first use.

79/// Where the tenant's effective `dcr_allowed_scopes` is read from.
80///
81/// A port rather than a repository type parameter, for the reason
82/// [`crate::mtls::IssuedCertificateLookup`] is one: the services that need it
83/// keep their type parameters, and the composition root builds one value that
84/// every service shares.
85pub trait ExternalClientScopePolicy: Send + Sync {
86    /// The tenant's **effective** `dcr_allowed_scopes` — the organization
87    /// baseline merged with the tenant override, as registration reads it.
88    ///
89    /// An `Err` is a failed read, never "no ceiling": the caller refuses.
90    fn allowed_scopes<'a>(
91        &'a self,
92        tenant_id: Uuid,
93    ) -> Pin<Box<dyn Future<Output = AxiamResult<Vec<String>>> + Send + 'a>>;
94}
95
96/// The shared form every service holds.
97pub type SharedScopePolicy = Arc<dyn ExternalClientScopePolicy>;
98
99/// The policy a service holds until its composition root wires a real one.
100///
101/// Refuses every read, so an external client is refused with `server_error`
102/// and an administrator's client — which never reads it — is unaffected.
103#[derive(Debug, Clone, Copy, Default)]
104pub struct UnwiredScopePolicy;
105
106impl ExternalClientScopePolicy for UnwiredScopePolicy {
107    fn allowed_scopes<'a>(
108        &'a self,
109        _tenant_id: Uuid,
110    ) -> Pin<Box<dyn Future<Output = AxiamResult<Vec<String>>> + Send + 'a>> {
111        Box::pin(async {
112            Err(AxiamError::Internal(
113                "no external-client scope policy is wired into this service".into(),
114            ))
115        })
116    }
117}
118
119/// A [`SharedScopePolicy`] that refuses — see [`UnwiredScopePolicy`].
120pub fn unwired() -> SharedScopePolicy {
121    Arc::new(UnwiredScopePolicy)
122}
123
124/// Whether the ceiling binds a client of this provenance.
125///
126/// Exhaustive on purpose: a new `ManagedBy` variant must decide here whether
127/// the tenant's list bounds it.
128pub const fn applies_to(managed_by: ManagedBy) -> bool {
129    match managed_by {
130        ManagedBy::Admin => false,
131        ManagedBy::Dcr | ManagedBy::Cimd => true,
132    }
133}
134
135/// Keep, in their order, only the scopes the ceiling names.
136pub fn narrow(scopes: &mut Vec<String>, ceiling: &[String]) {
137    scopes.retain(|s| ceiling.iter().any(|c| c == s));
138}
139
140/// The ceiling that binds `client`: `None` for an administrator's client (no
141/// read is made), the tenant's effective list for an external one.
142///
143/// # Errors
144///
145/// `server_error` when the list cannot be read — fail closed, see the module
146/// documentation.
147pub async fn ceiling_for(
148    policy: &dyn ExternalClientScopePolicy,
149    client: &OAuth2Client,
150) -> Result<Option<Vec<String>>, OAuth2Error> {
151    if !applies_to(client.managed_by) {
152        return Ok(None);
153    }
154    match policy.allowed_scopes(client.tenant_id).await {
155        Ok(list) => Ok(Some(list)),
156        Err(e) => {
157            tracing::error!(
158                error = %e,
159                tenant_id = %client.tenant_id,
160                client_id = %client.client_id,
161                managed_by = %client.managed_by,
162                "could not read the tenant's dcr_allowed_scopes for an externally registered \
163                 client; refusing rather than trusting the client row"
164            );
165            Err(OAuth2Error::ServerError(
166                "the tenant's scope policy could not be read".into(),
167            ))
168        }
169    }
170}

172/// Narrow `client.scopes` to the tenant's current ceiling, in place, and
173/// return the ceiling so a caller holding scopes granted *earlier* (a code, a
174/// refresh token, an approved CIBA request) can narrow those with the same
175/// list.
176///
177/// A no-op returning `Ok(None)` for an administrator's client.
178///
179/// # Errors
180///
181/// As [`ceiling_for`].
182pub async fn apply_to_client(
183    policy: &dyn ExternalClientScopePolicy,
184    client: &mut OAuth2Client,
185) -> Result<Option<Vec<String>>, OAuth2Error> {
186    let ceiling = ceiling_for(policy, client).await?;
187    if let Some(ceiling) = &ceiling {
188        let before = client.scopes.len();
189        narrow(&mut client.scopes, ceiling);
190        if client.scopes.len() != before {
191            tracing::debug!(
192                tenant_id = %client.tenant_id,
193                client_id = %client.client_id,
194                kept = %client.scopes.join(" "),
195                "an externally registered client holds scopes the tenant no longer offers; \
196                 they are withheld"
197            );
198        }
199    }
200    Ok(ceiling)
201}

207    fn s(v: &[&str]) -> Vec<String> {
208        v.iter().map(|x| (*x).to_owned()).collect()
209    }

211    #[test]
212    fn only_external_provenance_is_bound() {
213        assert!(!applies_to(ManagedBy::Admin));
214        assert!(applies_to(ManagedBy::Dcr));
215        assert!(applies_to(ManagedBy::Cimd));
216    }
217
218    #[test]
219    fn narrowing_keeps_order_and_drops_only_what_the_ceiling_lacks() {
220        let mut scopes = s(&["profile", "openid", "email"]);
221        narrow(&mut scopes, &s(&["openid", "profile"]));
222        assert_eq!(scopes, s(&["profile", "openid"]));
223
224        // An empty ceiling is "offer nothing", exactly as registration reads
225        // it (`dcr::validate` refuses every scope under an empty list) — not
226        // "no ceiling". Withdrawing the last scope must not restore the rest.
227        let mut scopes = s(&["openid"]);
228        narrow(&mut scopes, &[]);
229        assert!(scopes.is_empty());
230
231        // A scope the ceiling gained is never added.
232        let mut scopes = s(&["openid"]);
233        narrow(&mut scopes, &s(&["openid", "profile"]));
234        assert_eq!(scopes, s(&["openid"]));
235    }

237    #[tokio::test]
238    async fn an_unwired_policy_refuses_an_external_client_and_ignores_an_admin_one() {
239        let policy = UnwiredScopePolicy;
240        let mut client = OAuth2Client {
241            scopes: s(&["openid"]),
242            managed_by: ManagedBy::Admin,
243            ..test_client()
244        };
245        assert_eq!(apply_to_client(&policy, &mut client).await.unwrap(), None);
246        assert_eq!(client.scopes, s(&["openid"]));
247
248        client.managed_by = ManagedBy::Dcr;
249        let err = apply_to_client(&policy, &mut client).await.unwrap_err();
250        assert_eq!(err.error_code(), "server_error");
251        assert_eq!(client.scopes, s(&["openid"]), "a refusal narrows nothing");
252    }
