1//! X7 G8 — the GDPR-sensitive OIDC scopes `address` and `phone` (plan §4.8).
2//!
3//! # Four gates, and none of them is enough on its own
4//!
5//! A postal address or a telephone number reaches a relying party only when
6//! **all four** of these hold, and each is closed by a different party:
7//!
8//! 1. The **organization** turned `sensitive_scopes_enabled` on
9//!    (`axiam_core::models::settings::OidcPolicy`). Off by default; a tenant
10//!    may turn it off again but never on.
11//! 2. The **operator** registered the scope on the client. Nothing could
12//!    register it before this wave, so no client that exists today has it, and
13//!    `crate::authorize`'s step 5 refuses an unregistered scope as it always
14//!    has (invariant I1).
15//! 3. The **end user** consented, per client and per scope set, and the record
16//!    is still there. Withdrawal takes effect on the next UserInfo call — not
17//!    on the next token.
18//! 4. The client is **not** on the `fapi2` profile, checked at registration
19//!    (`crate::fapi::validate_registration`), again at the authorization
20//!    endpoint (`crate::fapi::enforce_authorization_request`), and a third
21//!    time at UserInfo, where no authorization request is in hand.
22//!
23//! # What this module is
24//!
25//! The pure half: which scopes are sensitive, what a consent record for them is
26//! called, and what an authorization request has earned given the answers the
27//! caller resolved. Every database read lives in the caller — the REST handler
28//! owns the settings and consent repositories — for the same reason
29//! `crate::authorize` takes `session_evidence` rather than a session
30//! repository: this crate decides, it does not fetch.
31
32use axiam_core::repository::OIDC_SCOPE_RELEASE_CONSENT_PREFIX;

34pub use crate::fapi::SENSITIVE_SCOPES;

36/// The sensitive scopes present in a requested set, in the canonical order
37/// [`SENSITIVE_SCOPES`] declares.
38///
39/// Canonical rather than request order because the result names a consent
40/// record, and `address phone` and `phone address` are the same consent. A
41/// relying party that reorders its `scope` parameter between two requests must
42/// not be asked to collect consent twice.
43///
44/// Duplicates in the request collapse: `scope=openid phone phone` asks for one
45/// telephone number.
46pub fn requested(scopes: &[String]) -> Vec<&'static str> {
47    SENSITIVE_SCOPES
48        .into_iter()
49        .filter(|sensitive| scopes.iter().any(|s| s == sensitive))
50        .collect()
51}
52
53/// The `consent_type` of the record covering releases to one relying party.
54///
55/// Per client, because consenting to give a postal address to one relying
56/// party says nothing about any other — that is the whole content of "informed"
57/// in Art. 4(11). The `client_id` is used verbatim: it is a registered
58/// identifier this deployment issued, not request input.
59pub fn consent_type(client_id: &str) -> String {
60    format!("{OIDC_SCOPE_RELEASE_CONSENT_PREFIX}{client_id}")
61}
62
63/// The `version` of that record: the consented scopes, canonically ordered and
64/// space-joined.
65///
66/// Putting the scope set in the version is what makes a client that later adds
67/// a scope re-prompt rather than inherit. `phone` consented to yesterday does
68/// not answer for `address phone` today, and the comparison that decides is
69/// string equality on a canonical form rather than a subset test somebody
70/// could get backwards.
71pub fn consent_version(scopes: &[&str]) -> String {
72    let mut canonical: Vec<&str> = SENSITIVE_SCOPES
73        .into_iter()
74        .filter(|s| scopes.contains(s))
75        .collect();
76    canonical.dedup();
77    canonical.join(" ")
78}

80/// What the caller resolved about this request's sensitive scopes.
81///
82/// Resolved by the REST handler, which owns the settings and consent
83/// repositories, and consumed by [`decide`]. The default is the state of every
84/// authorization request in every deployment today.
85#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
86pub enum Requested {
87    /// No sensitive scope was asked for. Nothing in this module applies.
88    #[default]
89    None,
90    /// Asked for, and the tenant's effective settings say the scopes are off.
91    Disabled,
92    /// Asked for, allowed, and a consent record covers exactly this set.
93    Consented,
94    /// Asked for, allowed, and no consent record covers this set.
95    ConsentMissing,
96}
97
98/// What an authorization request carrying sensitive scopes has earned.
99#[derive(Debug, Clone, PartialEq, Eq)]
100pub enum Decision {
101    /// Carry on. Either nothing sensitive was asked for, or it was consented.
102    Proceed,
103    /// Send the end user to the consent screen.
104    AskForConsent,
105    /// Answer the relying party with a terminal error.
106    Refuse(Refusal),
107}
108
109/// Why a request carrying sensitive scopes was refused.
110///
111/// A typed reason rather than a built `OAuth2Error` so that this module stays
112/// free of the error type's redirect semantics, and so a test can assert
113/// *which* refusal happened rather than matching prose.
114#[derive(Debug, Clone, Copy, PartialEq, Eq)]
115pub enum Refusal {
116    /// The tenant's effective settings have the scopes off. Answered
117    /// `invalid_scope`, which is what an unregistrable scope has always
118    /// earned — the switch being off and the scope never having been
119    /// registrable are the same fact from the relying party's side.
120    Disabled,
121    /// `prompt=none` forbade the interaction that consent needs
122    /// (OIDC Core §3.1.2.6).
123    ConsentRequired,
124    /// The end user has already been through the consent screen for this
125    /// request and consent still is not recorded, so they declined.
126    Declined,
127}
128
129/// Decide what a request carrying sensitive scopes earns.
130///
131/// The three inputs beyond the resolved state are all facts about *this*
132/// request, and the order they are consulted in is the argument:
133///
134/// * `switch_is_off` is answered first because it is the operator's decision
135///   and it outranks anything the end user or the relying party wants.
136/// * `consent_leg` — the request has already been through the **consent
137///   screen** once — is answered before `prompt_none` because it is the
138///   stronger statement: the user was asked, in person, and did not grant.
139///   Redirecting again is the non-terminating case, so the chain is bounded at
140///   one consent hop exactly as `crate::honour`'s is bounded at one login hop.
141///
142///   It must be the *consent* marker and not the login one
143///   (`crate::login_hop::CONSENT_HOP_MARKER`). A request carrying
144///   `prompt=consent` and `address` goes to the sign-in page first and comes
145///   back carrying the login marker with nobody having been asked about
146///   consent; read as a decline, that is `access_denied` for somebody who was
147///   never shown the question.
148/// * `prompt_none` last, because it only applies to a first attempt.
149///
150/// `prompt_none` must be passed as `true` **only** when the relying party sent
151/// it *and* the client is on the honour lane. A client registered `ignore` has
152/// its `prompt` dropped everywhere else, and this is not the place to start
153/// reading it — that would be invariant 4 broken by a wave that promised not
154/// to.
155pub fn decide(
156    requested: Requested,
157    switch_is_off: bool,
158    consent_leg: bool,
159    prompt_none: bool,
160) -> Decision {
161    match requested {
162        Requested::None => Decision::Proceed,
163        Requested::Disabled => Decision::Refuse(Refusal::Disabled),
164        Requested::Consented => {
165            // A resolved `Consented` cannot coexist with the switch being off:
166            // the caller only looks for a consent record once the switch has
167            // said yes. Asserted rather than assumed, because the two facts
168            // arrive from two different reads and a caller that transposed
169            // them would otherwise release data on a tenant that forbade it.
170            if switch_is_off {
171                Decision::Refuse(Refusal::Disabled)
172            } else {
173                Decision::Proceed
174            }
175        }
176        Requested::ConsentMissing => {
177            if switch_is_off {
178                Decision::Refuse(Refusal::Disabled)
179            } else if consent_leg {
180                Decision::Refuse(Refusal::Declined)
181            } else if prompt_none {
182                Decision::Refuse(Refusal::ConsentRequired)
183            } else {
184                Decision::AskForConsent
185            }
186        }
187    }
188}

190#[cfg(test)]
191mod tests {
192    use super::*;
193
194    fn scopes(list: &[&str]) -> Vec<String> {
195        list.iter().map(|s| (*s).to_owned()).collect()
196    }
197
198    #[test]
199    fn only_the_two_scopes_are_sensitive() {
200        assert_eq!(SENSITIVE_SCOPES, ["address", "phone"]);
201        assert!(requested(&scopes(&["openid", "profile", "email"])).is_empty());
202    }
203
204    /// The result is in the canonical order, whatever order the relying party
205    /// sent — otherwise the same consent would have two names.
206    #[test]
207    fn the_requested_set_is_canonically_ordered() {
208        assert_eq!(
209            requested(&scopes(&["phone", "openid", "address"])),
210            vec!["address", "phone"]
211        );
212        assert_eq!(
213            consent_version(&requested(&scopes(&["phone", "address"]))),
214            consent_version(&requested(&scopes(&["address", "phone"]))),
215        );
216    }
217
218    /// A repeated scope is one scope. A relying party sending `phone phone`
219    /// must not be asked to consent to something with a different name.
220    #[test]
221    fn duplicates_collapse() {
222        assert_eq!(requested(&scopes(&["phone", "phone"])), vec!["phone"]);
223        assert_eq!(consent_version(&["phone", "phone"]), "phone");
224    }
225
226    /// Adding a scope changes the version, which is what makes a client that
227    /// widens its request re-prompt instead of inheriting.
228    #[test]
229    fn widening_the_scope_set_changes_the_consent_version() {
230        assert_ne!(
231            consent_version(&["phone"]),
232            consent_version(&["address", "phone"])
233        );
234        assert_eq!(consent_version(&["address", "phone"]), "address phone");
235    }
236
237    /// Consent is per relying party.
238    #[test]
239    fn the_consent_type_names_the_client() {
240        assert_eq!(consent_type("shop"), "oidc_scope_release:shop");
241        assert_ne!(consent_type("shop"), consent_type("shop2"));
242    }
243
244    /// The ordinary request — every request in every deployment today.
245    #[test]
246    fn a_request_asking_for_nothing_sensitive_proceeds() {
247        assert_eq!(
248            decide(Requested::None, false, false, false),
249            Decision::Proceed
250        );
251        // Including under `prompt=none`, and on a return leg: neither means
252        // anything when nothing sensitive was asked for.
253        assert_eq!(decide(Requested::None, true, true, true), Decision::Proceed);
254    }
255
256    /// T8.1 — the switch outranks everything, consent included.
257    #[test]
258    fn the_switch_outranks_a_recorded_consent() {
259        assert_eq!(
260            decide(Requested::Disabled, true, false, false),
261            Decision::Refuse(Refusal::Disabled)
262        );
263        assert_eq!(
264            decide(Requested::Consented, true, false, false),
265            Decision::Refuse(Refusal::Disabled),
266            "a consent record must not survive the operator turning the capability off"
267        );
268    }
269
270    /// T8.2 — a first authorization asks; `prompt=none` cannot.
271    #[test]
272    fn a_first_authorization_asks_and_prompt_none_is_refused() {
273        assert_eq!(
274            decide(Requested::ConsentMissing, false, false, false),
275            Decision::AskForConsent
276        );
277        assert_eq!(
278            decide(Requested::ConsentMissing, false, false, true),
279            Decision::Refuse(Refusal::ConsentRequired)
280        );
281    }
282
283    /// The chain is bounded at one hop: a request that has already been to the
284    /// consent screen and come back without consent is answered, not sent
285    /// again.
286    #[test]
287    fn a_return_leg_without_consent_is_a_decline_not_a_second_redirect() {
288        assert_eq!(
289            decide(Requested::ConsentMissing, false, true, false),
290            Decision::Refuse(Refusal::Declined)
291        );
292        // And the decline outranks `prompt=none`: the stronger statement is
293        // that the user was asked and said no.
294        assert_eq!(
295            decide(Requested::ConsentMissing, false, true, true),
296            Decision::Refuse(Refusal::Declined)
297        );
298    }
299
300    /// Consent recorded, switch on: the request proceeds and nothing has been
301    /// released yet — release is UserInfo's decision, taken again there.
302    #[test]
303    fn a_consented_request_proceeds() {
304        assert_eq!(
305            decide(Requested::Consented, false, false, false),
306            Decision::Proceed
307        );
308    }
