1//! Shared Signals Framework delivery: the push deliverer (RFC 8935) and the
2//! outbox that routes a produced event (G-5, T23.5.3, D-36, D-48, D-49, D-51).
3//!
4//! Two pieces, one on each side of the shared outbound dispatcher:
5//!
6//! * [`SsfOutboxService`] implements the core [`SsfOutbox`] port **producers**
7//!   call: an enabled push stream's event becomes one
8//!   [`OutboundKind::SsfPush`] message on the dispatcher; an enabled poll
9//!   stream's, and any paused stream's, goes to the stream's bounded buffer; a
10//!   disabled stream's is refused ([`SsfOutboxError::Disabled`]).
11//! * [`SsfPushDeliverer`] implements the core [`OutboundDeliverer`] port the
12//!   dispatcher's consumer calls: **one attempt** to push one event.
13//!
14//! The retry schedule, the attempt counter, the dead-letter queue and the
15//! audit rows (`ssf_push.delivery_succeeded`, `.delivery_attempt`,
16//! `.delivery_failed`) are the dispatcher's. A deliverer classifies; it does
17//! not decide.
18//!
19//! # What one attempt does
20//!
21//! The queued message holds the **unsigned** event (D-48), so the attempt
22//! starts by reading the stream *as it is now* and signs only against that
23//! (D-51):
24//!
25//! | The stream, at the attempt | Outcome |
26//! |---|---|
27//! | any stream, while the D-55 gate holds (tenant issuer paths off, more than one tenant) | dead-letter — nothing is signed, sent or held |
28//! | gone | dead-letter |
29//! | `disabled` | dead-letter — nothing is signed or sent |
30//! | `paused`, or now a poll stream | the event goes to the buffer; the message is acknowledged |
31//! | `enabled`, event no longer carried | dead-letter |
32//! | `enabled` | sign with [`sign_set`], open the credential, read the stream again — a different version is a retry (F4 W4 P23W4-01) — then push |
33//!
34//! The one exception is a **stream-updated** announcement (SSF §8.1.5), whose
35//! whole point is to follow a status change: it is not refused for the status
36//! it announces, and [`sign_set`] signs it only while that is still the
37//! stream's status.
38//!
39//! # The push (D-49)
40//!
41//! `POST` of the compact SET, `Content-Type: application/secevent+jwt`,
42//! `Accept: application/json`, the stored `Authorization` header (opened by
43//! `decrypt_authorization_header`, held in `Zeroizing` and marked sensitive on
44//! the request), ten seconds, **no redirect followed**, a response body read
45//! to at most 64 KiB and never logged.
46//!
47//! **The only way out of the process is `guarded_fetch_no_redirect` with
48//! `allow_private = false`** (`axiam_pki::ssrf`, D-53 (9)): the name is resolved
49//! fresh, every address must be globally routable, the validated address is
50//! pinned into the connection, and `https` is required. A name an administrator
51//! registered that resolves to an internal address is refused here, at delivery,
52//! which is what the write-time policy cannot catch (T-392). The fetch makes
53//! **one hop and returns a `3xx` as a response**: a redirect is never followed,
54//! so neither the SET nor the credential can reach a host the administrator
55//! never named — by construction, not by inspecting a later hop.
56//!
57//! | The receiver answers | Outcome |
58//! |---|---|
59//! | any `2xx` (RFC 8935: `202`) | delivered |
60//! | `400` with an RFC 8935 `err` | dead-letter, the code in the audit row |
61//! | `401`, `403` | dead-letter: the credential is wrong until someone fixes it |
62//! | `404`, `408`, `429`, `5xx`, a timeout, no connection | retry |
63//! | a `3xx` | retry; it is never followed |
64//! | any other `4xx` (a `400` without a known code, `410`, `422` …) | dead-letter, reason `HTTP <status>`: it will not change on retry (D-53 (8)) |
65//! | anything else | retry |
66//!
67//! A reason string reaches the audit log, so it is one of a fixed vocabulary:
68//! never a header, a URL, a response body or a transport error's text (which
69//! would carry the endpoint's URL).

120/// The [`SsfOutbox`] a deployment runs: routes by the stream's status and
121/// method (D-48, D-51).
122///
123/// | Status | Method | |
124/// |---|---|---|
125/// | `enabled` | push | one [`OutboundKind::SsfPush`] message, unsigned event as payload |
126/// | `enabled` | poll | the stream's buffer |
127/// | `paused` | either | the stream's buffer (held) |
128/// | `disabled` | either | [`SsfOutboxError::Disabled`]; nothing is kept |
129///
130/// A stream-updated announcement of a push stream is the exception to the last
131/// two rows: it is enqueued whatever the status it announces, since telling the
132/// receiver of a pause or a disable is the event's job and [`sign_set`] will
133/// sign it only while it is true.
134pub struct SsfOutboxService<B> {
135    buffer: B,
136    publisher: Arc<dyn OutboundPublisher>,
137}
138
139impl<B: SsfEventBufferRepository + 'static> SsfOutboxService<B> {
140    /// An outbox over `buffer` and the dispatcher's publisher.
141    pub fn new(buffer: B, publisher: Arc<dyn OutboundPublisher>) -> Self {
142        Self { buffer, publisher }
143    }
144
145    fn message(
146        stream: &SsfStream,
147        event: &SsfPendingEvent,
148    ) -> Result<OutboundMessage, SsfOutboxError> {
149        let delivery_id = delivery_id_of(event)
150            .ok_or_else(|| SsfOutboxError::Failed("the event has no valid jti".into()))?;
151        let payload = serde_json::to_value(event)
152            .map_err(|_| SsfOutboxError::Failed("the event could not be encoded".into()))?;
153        Ok(OutboundMessage {
154            kind: OutboundKind::SsfPush,
155            tenant_id: stream.tenant_id,
156            target_id: stream.id,
157            delivery_id,
158            event_type: event.event_uri.clone(),
159            payload,
160            attempt: 0,
161        })
162    }
163
164    async fn enqueue(
165        &self,
166        stream: &SsfStream,
167        event: &SsfPendingEvent,
168    ) -> Result<(), SsfOutboxError> {
169        let message = Self::message(stream, event)?;
170        self.publisher.enqueue(&message).await.map_err(|error| {
171            tracing::warn!(
172                target: "axiam::ssf",
173                tenant_id = %stream.tenant_id,
174                stream_id = %stream.id,
175                %error,
176                "an SSF push could not be enqueued"
177            );
178            SsfOutboxError::Failed("the event could not be enqueued".into())
179        })
180    }
181
182    async fn hold(
183        &self,
184        stream: &SsfStream,
185        event: &SsfPendingEvent,
186    ) -> Result<(), SsfOutboxError> {
187        self.buffer
188            .push(stream.tenant_id, stream.id, event, Utc::now())
189            .await
190            .map_err(|error| {
191                tracing::warn!(
192                    target: "axiam::ssf",
193                    tenant_id = %stream.tenant_id,
194                    stream_id = %stream.id,
195                    %error,
196                    "an SSF event could not be buffered"
197                );
198                SsfOutboxError::Failed("the event could not be buffered".into())
199            })
200    }
201}

203impl<B: SsfEventBufferRepository + 'static> SsfOutbox for SsfOutboxService<B> {
204    fn submit<'a>(
205        &'a self,
206        stream: &'a SsfStream,
207        event: &'a SsfPendingEvent,
208    ) -> SsfFuture<'a, Result<(), SsfOutboxError>> {
209        Box::pin(async move {
210            let announcement = event.event_uri == STREAM_UPDATED_EVENT_URI;
211            match (stream.status, stream.delivery_method) {
212                // The announcement of the change itself: it is what tells the
213                // receiver, so it goes out whatever the new status is.
214                (_, SsfDeliveryMethod::Push) if announcement => self.enqueue(stream, event).await,
215                (SsfStreamStatus::Disabled, _) => Err(SsfOutboxError::Disabled),
216                (SsfStreamStatus::Paused, _) => self.hold(stream, event).await,
217                (SsfStreamStatus::Enabled, SsfDeliveryMethod::Push) => {
218                    self.enqueue(stream, event).await
219                }
220                (SsfStreamStatus::Enabled, SsfDeliveryMethod::Poll) => {
221                    self.hold(stream, event).await
222                }
223            }
224        })
225    }
226
227    fn resume<'a>(&'a self, stream: &'a SsfStream) -> SsfFuture<'a, Result<usize, SsfOutboxError>> {
228        Box::pin(async move {
229            if stream.delivery_method != SsfDeliveryMethod::Push
230                || stream.status != SsfStreamStatus::Enabled
231            {
232                return Ok(0);
233            }
234            let mut released = 0usize;
235            for _ in 0..RESUME_MAX_BATCHES {
236                let batch = self
237                    .buffer
238                    .list_oldest(stream.tenant_id, stream.id, RESUME_BATCH, Utc::now())
239                    .await
240                    .map_err(|_| {
241                        SsfOutboxError::Failed("the held events could not be read".into())
242                    })?;
243                if batch.is_empty() {
244                    break;
245                }
246                // Oldest first, and a row is removed only once its message is
247                // durably queued: a failure part-way leaves the rest held, and
248                // a crash between the two repeats one event, which carries one
249                // `jti` the receiver de-duplicates.
250                let mut done = Vec::with_capacity(batch.len());
251                let mut failure = None;
252                for event in &batch {
253                    match self.enqueue(stream, event).await {
254                        Ok(()) => done.push(event.jti.clone()),
255                        Err(error) => {
256                            failure = Some(error);
257                            break;
258                        }
259                    }
260                }
261                released += done.len();
262                self.buffer
263                    .delete_by_jti(stream.tenant_id, stream.id, &done)
264                    .await
265                    .map_err(|_| {
266                        SsfOutboxError::Failed("the released events could not be removed".into())
267                    })?;
268                if let Some(error) = failure {
269                    return Err(error);
270                }
271            }
272            Ok(released)
273        })
274    }
275}

281/// The [`OutboundDeliverer`] of [`OutboundKind::SsfPush`]: one RFC 8935 push of
282/// one event. See the module documentation for the whole contract.
283pub struct SsfPushDeliverer<S, B> {
284    streams: S,
285    buffer: B,
286    auth_config: AuthConfig,
287    /// D-55: while the deployment's tenants would share one issuer, nothing is
288    /// signed, sent or held; a queued message is dead-lettered.
289    gate: Arc<SsfIssuerGate>,
290    /// The guarded fetch's `allow_private`. **Always `false`** except for the
291    /// integration tests' loopback receiver, which turn it on through
292    /// [`Self::admitting_private_networks_for_tests`] and nothing else.
293    allow_private: bool,
294}
295
296impl<S, B> SsfPushDeliverer<S, B>
297where
298    S: SsfStreamRepository + 'static,
299    B: SsfEventBufferRepository + 'static,
300{
301    /// The production deliverer: every push goes through `guarded_fetch_no_redirect` with
302    /// `allow_private = false`.
303    pub fn new(streams: S, buffer: B, auth_config: AuthConfig, gate: Arc<SsfIssuerGate>) -> Self {
304        Self {
305            streams,
306            buffer,
307            auth_config,
308            gate,
309            allow_private: false,
310        }
311    }
312
313    /// **Test seam, never used by the composition root.** Lets the first hop
314    /// reach a loopback receiver over plain `http`, the way
315    /// `JwksCache::new_allow_private_networks` and the CIMD policy do for their
316    /// tests. A deliverer built by [`Self::new`] cannot be switched afterwards
317    /// by anything but this call, and production never makes it.
318    #[doc(hidden)]
319    #[must_use]
320    pub fn admitting_private_networks_for_tests(mut self) -> Self {
321        self.allow_private = true;
322        self
323    }

325    async fn attempt(&self, msg: &OutboundMessage) -> Result<DeliveryOutcome, OutboundError> {
326        if msg.kind != OutboundKind::SsfPush {
327            return Ok(dead("the message is not an SSF push"));
328        }
329        let Ok(pending) = serde_json::from_value::<SsfPendingEvent>(msg.payload.clone()) else {
330            return Ok(dead("the queued event is malformed"));
331        };
332        if delivery_id_of(&pending) != Some(msg.delivery_id) {
333            return Ok(dead("the queued event does not match its delivery id"));
334        }
335
336        // The stream as it is now (D-48, D-51).
337        let stream = match self.streams.get(msg.tenant_id, msg.target_id).await {
338            Ok(stream) => stream,
339            Err(AxiamError::NotFound { .. }) => return Ok(dead("the stream no longer exists")),
340            Err(_) => {
341                return Err(OutboundError::Delivery(
342                    "the stream could not be read".into(),
343                ));
344            }
345        };
346
347        // D-55, before anything is held or signed: while the deployment's
348        // tenants share one issuer SSF is off for every tenant, so a queued
349        // message is dead-lettered — not held, not retried.
350        let issuer = match self.gate.check().await {
351            Ok(issuer) => issuer,
352            Err(_) => {
353                return Err(OutboundError::Delivery(
354                    "the deployment's tenants could not be counted".into(),
355                ));
356            }
357        };
358        if issuer.holds() {
359            return Ok(dead(SHARED_ISSUER_DEAD_LETTER));
360        }
361        let announcement = pending.event_uri == STREAM_UPDATED_EVENT_URI;
362        if !announcement {
363            match stream.status {
364                SsfStreamStatus::Disabled => return Ok(dead("the stream is disabled")),
365                SsfStreamStatus::Paused => return self.hold(&stream, &pending).await,
366                SsfStreamStatus::Enabled => {}
367            }
368        }
369        if stream.delivery_method != SsfDeliveryMethod::Push {
370            // The stream became a poll stream while the message was queued.
371            if stream.status == SsfStreamStatus::Disabled {
372                return Ok(dead("the stream is disabled"));
373            }
374            return self.hold(&stream, &pending).await;
375        }
376
377        // Signed against the stream as it is *now*, or not at all.
378        let set = match sign_set(&self.auth_config, issuer, &stream, &pending) {
379            Ok(set) => zeroize::Zeroizing::new(set),
380            Err(SsfError::SharedIssuer) => return Ok(dead(SHARED_ISSUER_DEAD_LETTER)),
381            Err(SsfError::Signing(_)) => {
382                return Err(OutboundError::Delivery(
383                    "the SET could not be signed".into(),
384                ));
385            }
386            Err(_) => return Ok(dead("the stream no longer carries this event")),
387        };
388        let Some(endpoint) = stream.endpoint_url.clone() else {
389            return Ok(dead("the push stream has no endpoint"));
390        };
391        let authorization = match self
392            .streams
393            .decrypt_authorization_header(msg.tenant_id, msg.target_id)
394            .await
395        {
396            Ok(value) => value,
397            Err(AxiamError::NotFound { .. }) => return Ok(dead("the stream no longer exists")),
398            Err(_) => {
399                return Err(OutboundError::Delivery(
400                    "the stream's push credential could not be opened".into(),
401                ));
402            }
403        };
404        // F4 W4 P23W4-01 (D-49, T-406): the stream and its credential are two
405        // reads. A credential opened after the endpoint moved is the *new*
406        // endpoint's, and must not go to the one read above, so the stream is
407        // read once more and the push goes out only if it is the same version
408        // that was signed against and whose endpoint is about to be used.
409        match self.streams.get(msg.tenant_id, msg.target_id).await {
410            Ok(again) if again.updated_at == stream.updated_at => {}
411            Ok(_) => return Ok(retry("the stream changed during the attempt")),
412            Err(AxiamError::NotFound { .. }) => return Ok(dead("the stream no longer exists")),
413            Err(_) => {
414                return Err(OutboundError::Delivery(
415                    "the stream could not be read".into(),
416                ));
417            }
418        }
419        let authorization = match authorization {
420            None => None,
421            Some(value) => match HeaderValue::from_str(&value) {
422                Ok(mut header) => {
423                    header.set_sensitive(true);
424                    Some(header)
425                }
426                Err(_) => return Ok(dead("the stored push credential is not a valid header")),
427            },
428        };
429
430        self.push(&endpoint, &set, authorization).await
431    }
432
433    async fn hold(
434        &self,
435        stream: &SsfStream,
436        pending: &SsfPendingEvent,
437    ) -> Result<DeliveryOutcome, OutboundError> {
438        self.buffer
439            .push(stream.tenant_id, stream.id, pending, Utc::now())
440            .await
441            .map_err(|_| OutboundError::Delivery("the event could not be buffered".into()))?;
442        Ok(DeliveryOutcome::Delivered {
443            response_status: None,
444        })
445    }
446
447    /// The one HTTP exchange, through the shared guard and nothing else.
448    async fn push(
449        &self,
450        endpoint: &str,
451        set: &str,
452        authorization: Option<HeaderValue>,
453    ) -> Result<DeliveryOutcome, OutboundError> {
454        // One guarded hop that returns a redirect instead of following it
455        // (D-53 (9)): neither the SET nor the credential can reach a host the
456        // administrator did not name, by construction.
457        let response =
458            match guarded_fetch_no_redirect(endpoint, self.allow_private, |client, target| {
459                let mut request = client
460                    .post(target)
461                    .header(CONTENT_TYPE, SET_CONTENT_TYPE)
462                    .header(ACCEPT, "application/json")
463                    .body(set.to_owned());
464                if let Some(value) = &authorization {
465                    request = request.header(AUTHORIZATION, value.clone());
466                }
467                request
468            })
469            .await
470            {
471                Ok(response) => response,
472                Err(error) => return Ok(classify_transport(&error)),
473            };
474
475        let status = response.status().as_u16();
476        Ok(match status {
477            200..=299 => DeliveryOutcome::Delivered {
478                response_status: Some(status),
479            },
480            // A 3xx is never followed, and is a retry: the administrator may
481            // fix the endpoint.
482            300..=399 => retry("the receiver answered with a redirect, which is not followed"),
483            400..=499 => match status {
484                400 => match read_capped_body(response, MAX_RESPONSE_BODY_BYTES)
485                    .await
486                    .ok()
487                    .as_deref()
488                    .and_then(rfc_8935_error)
489                {
490                    Some(code) => dead(&format!("the receiver rejected the SET: {code}")),
491                    // A 400 without a known code will not change on retry either.
492                    None => dead("HTTP 400"),
493                },
494                // The one 4xx that says the credential is wrong until someone
495                // fixes it.
496                401 | 403 => dead(&format!(
497                    "the receiver refused the push credential (HTTP {status})"
498                )),
499                // The 4xx that can pass: a receiver not ready, slow or limiting.
500                404 | 408 | 429 => retry(&format!("the receiver answered HTTP {status}")),
501                // Any other 4xx will not change on retry (D-53 (8)).
502                _ => dead(&format!("HTTP {status}")),
503            },
504            500..=599 => retry(&format!("the receiver answered HTTP {status}")),
505            // Anything else (1xx, an unassigned code) retries.
506            _ => retry(&format!(
507                "the receiver answered an unexpected HTTP {status}"
508            )),
509        })
510    }
511}

542/// The audit-safe reading of a guard error: a fixed phrase, never the error's
543/// own text (a transport error names the URL it was fetching).
544fn classify_transport(error: &SsrfError) -> DeliveryOutcome {
545    match error {
546        SsrfError::Blocked => {
547            retry("the push endpoint resolves to an address AXIAM does not connect to")
548        }
549        SsrfError::ResolveFailed => retry("the push endpoint's host did not resolve"),
550        SsrfError::InsecureScheme => dead("the push endpoint is not https"),
551        SsrfError::InvalidUrl => dead("the push endpoint is not a valid URL"),
552        SsrfError::TooManyRedirects => retry("the receiver redirected too often"),
553        SsrfError::ResponseTooLarge(_) => retry("the receiver's response was too large"),
554        SsrfError::ClientBuildFailed => retry("the HTTP client could not be built"),
555        SsrfError::RequestFailed(_) => {
556            retry("the receiver could not be reached or did not answer in time")
557        }
558    }
559}

561/// The RFC 8935 §2.4 error code in a `400` body, if it names one.
562fn rfc_8935_error(body: &[u8]) -> Option<&'static str> {
563    let value: serde_json::Value = serde_json::from_slice(body).ok()?;
564    let code = value.get("err")?.as_str()?;
565    RFC_8935_ERROR_CODES
566        .into_iter()
567        .find(|known| *known == code)
568}

596            SsrfError::RequestFailed(format!("error sending request for url ({url})")),
597            SsrfError::Blocked,
598            SsrfError::ResolveFailed,
599            SsrfError::TooManyRedirects,
600            SsrfError::ResponseTooLarge(1),
601            SsrfError::ClientBuildFailed,
602            SsrfError::InsecureScheme,
603            SsrfError::InvalidUrl,
604        ] {
605            let reason = match classify_transport(&error) {
606                DeliveryOutcome::Retry { reason } | DeliveryOutcome::DeadLetter { reason } => {
607                    reason
608                }
609                other => panic!("a transport failure is never {other:?}"),
610            };
611            assert!(!reason.contains("rp.example.test") && !reason.contains("token"));
612        }
613    }
