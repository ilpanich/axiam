32/// The single `error_description` every token-endpoint client-authentication
33/// failure returns, whatever actually went wrong (SEC-086).
34///
35/// The `invalid_client` error *code* was already uniform across "no such
36/// client" and "wrong secret". The *description* was not: an unknown client
37/// said `"client not found"`, which let an unauthenticated caller probe
38/// whether a given `client_id` — including an `sa_…` service-account id —
39/// exists. The masquerade the QUAL-03/D-11 comments describe was applied to
40/// the code and not to the message; this constant applies it to both.
41///
42/// The `NotFound`-versus-other-error distinction those comments protect is
43/// unaffected: it stays internal, mapping to `invalid_client` versus
44/// `server_error`, so a DB outage still never reads as bad credentials. Only
45/// the caller-visible wording is collapsed.
46///
47/// Deliberately **not** used at the authorization endpoint
48/// (`authorize.rs`): there `client_id` is public by construction, RFC 6749
49/// §4.1.2.1 requires informing the resource owner directly rather than
50/// redirecting, and no credential is presented — so "invalid client
51/// credentials" would be actively misleading there.
52///
53/// W8 made it `pub`: the REST layer now refuses a malformed
54/// `Authorization: Basic` header before `TokenService` is ever reached, and
55/// that refusal must be worded identically to the ones raised here. A second
56/// copy of the string in `axiam-api-rest` would be a second thing to keep
57/// equal, and SEC-086's whole property is that these answers are
58/// indistinguishable.
59pub const CLIENT_AUTH_FAILED: &str = "invalid client credentials";

61/// How long a rotated refresh token stays usable after its successor is issued.
62///
63/// FAPI 2.0 Security Profile §5.3.2.1-9 requires the previous token to be
64/// accepted for a period after rotation; the OIDF module that checks it
65/// (`fapi2-security-profile-final-refresh-token`) sleeps thirty seconds and
66/// then replays the old token expecting a 200.
67///
68/// Sixty seconds, not thirty. The tested number is the suite's, not the
69/// profile's, and a grace period exactly as long as the test that measures it
70/// passes conformance by arriving first — a client on a slow link retrying the
71/// request whose response it lost has no such guarantee. Sixty is the smallest
72/// value that is comfortably longer than the observation and still far shorter
73/// than the thirty-day life of the token being retired.
74///
75/// Not configurable: a deployment that shortened it below the profile's floor
76/// would silently stop conforming, and one that lengthened it would widen the
77/// replay window for every tenant at once. If a deployment ever needs to differ
78/// this belongs in the client's own registration, where the FAPI profile switch
79/// already lives, rather than in a global.
80pub const REFRESH_ROTATION_GRACE_SECS: i64 = 60;

98    /// `grant_type=urn:ietf:params:oauth:grant-type:device_code`

100    /// to [`crate::device_service::DeviceAuthorizationService`] rather than

105    //
106    // Carried here rather than in a second form type because actix
107    // deserializes one body once, and the token endpoint cannot know which

109    // `exchange_request()` below projects them into the shape the exchange
110    // service takes, so the grant's own module never sees these Options.

112    pub subject_token_type: Option<String>,
113    pub requested_token_type: Option<String>,
114    pub actor_token: Option<String>,
115    pub actor_token_type: Option<String>,
116    pub audience: Option<String>,
117    pub resource: Option<String>,
118
119    // --- X2 / UMA 2.0 ticket grant --------------------------------------
120    /// The permission ticket minted by `/uma2/perm`, for
121    /// `grant_type=urn:ietf:params:oauth:grant-type:uma-ticket`.
122    pub ticket: Option<String>,
123    /// The requesting party's identity, as an AXIAM access token.
124    ///
125    /// UMA 2.0 §3.3.1 makes this optional because a client may instead present
126    /// its own RPT for incremental authorization, which v1 does not implement.
127    /// With claims-gathering also deferred, this is the only channel through

139    /// RFC 8693 parameters are: actix deserializes one body once, and the

144    /// Must be `urn:ietf:params:oauth:client-assertion-type:jwt-bearer`.

148impl TokenRequest {
149    /// Project the RFC 8693 parameters, or `None` when the two mandatory ones
150    /// are absent.
151    ///
152    /// Returning `None` rather than a half-populated request means the caller
153    /// answers `invalid_request` naming what is missing, instead of the
154    /// exchange service reporting a confusing downstream failure about a
155    /// token it was handed as an empty string.
156    pub fn exchange_request(&self) -> Option<crate::token_exchange::TokenExchangeRequest> {
157        let subject_token = self.subject_token.clone().filter(|t| !t.is_empty())?;
158        let subject_token_type = self.subject_token_type.clone().filter(|t| !t.is_empty())?;
159        Some(crate::token_exchange::TokenExchangeRequest {
160            subject_token,
161            subject_token_type,
162            requested_token_type: self.requested_token_type.clone().filter(|t| !t.is_empty()),
163            actor_token: self.actor_token.clone().filter(|t| !t.is_empty()),
164            actor_token_type: self.actor_token_type.clone().filter(|t| !t.is_empty()),
165            scope: self.scope.clone(),
166            audience: self.audience.clone().filter(|t| !t.is_empty()),
167            resource: self.resource.clone().filter(|t| !t.is_empty()),
168        })
169    }

260/// What the token endpoint knows about the **connection** a request arrived
261/// on, as opposed to its body (X5.1).
262///
263/// Exists so `TokenService` can authenticate a client by certificate without
264/// depending on actix, rustls, or the REST layer's extractor types. The REST
265/// handler builds one from `HttpRequest::conn_data::<VerifiedClientCert>()`;
266/// a test builds one from a certificate it generated.
267///
268/// [`Default`] is "no certificate", which is what a plaintext or
269/// server-only-TLS connection legitimately has. Every path that *needs* a
270/// certificate checks for it explicitly; nothing infers authentication from
271/// this type's absence.
272#[derive(Debug, Clone, Default)]
273pub struct TokenRequestContext {
274    /// The client certificate rustls verified during the TLS handshake, if
275    /// this connection carried one.
276    ///
277    /// **Only** ever populated from a verified peer chain. See
278    /// `crate::mtls`'s module docs for why the `X-Client-Certificate` proxy
279    /// header is deliberately not a source here.
280    pub client_certificate: Option<PresentedCertificate>,
281    /// A DPoP proof that has **already been verified** for this request, and
282    /// whose `jti` has already been recorded (X5.1, RFC 9449).
283    ///
284    /// The same discipline as [`Self::client_certificate`] and for the same
285    /// reason: this field carries a *conclusion*, never an input. The REST
286    /// layer verifies the `DPoP` header against the actual method and URI —
287    /// which only it knows — and hands the result here. A caller that
288    /// populated this from an unverified header would have turned DPoP into a
289    /// self-signed permission slip, which is why the type is
290    /// `VerifiedDpopProof` and not `String`.
291    pub dpop_proof: Option<crate::dpop::VerifiedDpopProof>,
292    /// The `client_assertion` form parameter, verbatim (RFC 7521 §4.2).
293    ///
294    /// Unlike the two fields above this is a request *body* parameter, not a
295    /// property of the connection, and it is carried here rather than passed
296    /// down from the form because `authenticate_client_credential` is reached
297    /// from four grants that share no request type. It is deliberately the raw
298    /// string: nothing has verified it at this point, and the type says so.
299    pub client_assertion: Option<String>,
300    /// The `client_assertion_type` form parameter.
301    pub client_assertion_type: Option<String>,
302    /// W8 — the credentials decoded from an `Authorization: Basic` header
303    /// (RFC 6749 §2.3.1), if this request carried a well-formed one.
304    ///
305    /// The same discipline as [`Self::client_certificate`] and
306    /// [`Self::dpop_proof`]: the REST layer reads the header, hands it to
307    /// [`crate::client_secret_basic::parse_authorization_header`], and puts
308    /// the *decoded* result here. A malformed header never reaches this field
309    /// — it is answered at the edge with `invalid_client` and a
310    /// `WWW-Authenticate: Basic` challenge — so `None` unambiguously means
311    /// "no Basic credentials were presented" rather than "some were, badly".
312    ///
313    /// Presence here authenticates nothing on its own. Only a client whose
314    /// *registration* names `client_secret_basic` is authenticated from it
315    /// (SEC-093); for every other client this field is inert, and the
316    /// `client_secret_post` case logs a `warn` saying so.
317    pub basic_credentials: Option<crate::client_secret_basic::BasicCredentials>,
318    /// The `sub` of [`Self::client_assertion`], for RFC 7521 §4.2's fallback.
319    ///
320    /// Derived by [`Self::with_assertion`] and by nothing else, so that it
321    /// cannot be set independently of the assertion it is supposed to describe.
322    /// It answers "which client row should be loaded" and never "which client
323    /// is authenticated" — see
324    /// [`crate::private_key_jwt::unverified_client_id`].
325    pub assertion_client_id: Option<String>,
326}

328/// Which client id a token request is for, given a body parameter that may be
329/// absent and an `Authorization: Basic` header that may name one (W8, RFC 6749
330/// §2.3.1).
331///
332/// # Why the header may supply it at all
333///
334/// RFC 6749 §2.3.1 makes the body's `client_id` **optional** for a client
335/// authenticating with the header, and the OpenID Foundation's Basic OP suite
336/// takes it up: 37 of its 38 modules send the id only in the header. AXIAM's
337/// three grants each required `req.client_id`, so without this the new method
338/// would have been unusable by exactly the clients it was added for. The plan
339/// (§4.7) did not mention it; the tree did. See the W8 amendment in
340/// `claude_dev/basic-op-gap-plan.md` §4.7.
341///
342/// # Why disagreement is `invalid_request` and not a preference
343///
344/// A request naming `a` in the header and `b` in the body has made two claims
345/// of identity, and any rule for picking a winner is a rule an attacker can
346/// aim: whichever half a proxy, a log or an audit record reads, the other is
347/// the one that authenticated. Refusing is the only answer that cannot be
348/// exploited by arranging for the two readers to disagree.
349///
350/// It is refused **before** the client lookup, and that placement is load
351/// bearing: the question "do these two strings differ" is answered from the
352/// request alone, so it creates no client-existence oracle and SEC-086's
353/// ordering property is untouched.
354fn resolve_client_id<'a>(
355    from_body: Option<&'a str>,
356    ctx: &'a TokenRequestContext,
357) -> Result<&'a str, OAuth2Error> {
358    match (from_body, ctx.basic_client_id()) {
359        (Some(body), Some(header)) if body != header => Err(OAuth2Error::InvalidRequest(
360            "client_id in the Authorization header and in the request body disagree".into(),
361        )),
362        (Some(body), _) => Ok(body),
363        (None, Some(header)) => Ok(header),
364        // RFC 7521 §4.2: "The `client_id` parameter is optional when using an
365        // assertion for client authentication." OIDC Core §9 then makes the
366        // assertion's `sub` the client identifier, so a `private_key_jwt`
367        // client that sends only `client_assertion` — which is what the OIDF
368        // FAPI 2.0 plan's DPoP lane does on every request — is identifying
369        // itself perfectly well and was being answered "client_id is required".
370        //
371        // The value routes the lookup and authenticates nothing:
372        // `authenticate_client_credential` still verifies the assertion against
373        // the loaded client's registered key and still requires `iss` and `sub`
374        // to equal that client's id, so naming somebody else here only picks
375        // the key the forgery is checked against.
376        (None, None) => ctx
377            .assertion_client_id()
378            .ok_or_else(|| OAuth2Error::InvalidRequest("client_id is required".into())),
379    }
380}

402        self.basic_credentials.as_ref().map(|c| c.client_id())

405    /// Whether this request carries **no** client credential of any kind.
406    ///
407    /// The guard three grants run before they look a client up, in one place.
408    ///
409    /// # Why it has to be one place
410    ///
411    /// It has been widened three times. X5.1 added the client certificate,
412    /// because an mTLS client sends no secret at all; W8 added the
413    /// `Authorization: Basic` header; and the third widening — the assertion —
414    /// was missed at all three sites at once, so a `private_key_jwt` client
415    /// presenting a perfectly good `client_assertion` and nothing else was
416    /// refused `invalid_client: client authentication is required` before
417    /// anything looked at it. `private_key_jwt` verification had been
418    /// implemented and wired; this guard sat in front of it. It cost every
419    /// module of the OIDF FAPI 2.0 DPoP lane.
420    ///
421    /// # What it still does not do
422    ///
423    /// It asks only what the REQUEST carries, never what the client
424    /// registered, so it stays decidable before the lookup and SEC-086's
425    /// property holds: a caller with no credential gets the same answer whether
426    /// or not the client id exists. Presence of a credential here authenticates
427    /// nothing — `authenticate_client_credential` decides that, against the
428    /// registration.
429    pub fn carries_no_client_credential(&self, body_secret: Option<&str>) -> bool {
430        body_secret.is_none()
431            && self.client_certificate.is_none()
432            && self.basic_credentials.is_none()
433            && self.client_assertion.is_none()
434    }

472    /// Same as [`Self::with_assertion_from`], for the endpoints whose request
473    /// body is not a [`TokenRequest`] — revoke, introspect and PAR (SEC-093).
474    ///
475    /// One implementation, so the "empty string is not a credential" filter
476    /// cannot be present at one endpoint and missing at another.
477    #[must_use]
478    pub fn with_assertion(
479        mut self,
480        client_assertion: Option<&str>,
481        client_assertion_type: Option<&str>,
482    ) -> Self {
483        self.client_assertion = client_assertion
484            .filter(|a| !a.is_empty())
485            .map(str::to_owned);
486        self.client_assertion_type = client_assertion_type
487            .filter(|t| !t.is_empty())
488            .map(str::to_owned);
489        // Derived here, once, so that every endpoint that attaches an assertion
490        // gets the RFC 7521 §4.2 fallback without four call sites remembering
491        // to — the same argument this builder exists for.
492        self.assertion_client_id = self
493            .client_assertion
494            .as_deref()
495            .and_then(crate::private_key_jwt::unverified_client_id);
496        self
497    }

534#[derive(Clone)]
535pub struct TokenService<OC, AC, TR, RT, UR, SA, SR> {
536    client_repo: OC,
537    /// X5.1 — resolves a `private_key_jwt` client's keys, verifies its
538    /// assertion, and records the `jti`.
539    ///
540    /// `None` means no client may authenticate with `private_key_jwt` in this
541    /// deployment — a refusal for clients registered for the method, never a
542    /// fallback to a weaker credential. See
543    /// [`crate::private_key_jwt::ClientAssertionVerifier`].
544    assertion_verifier: Option<std::sync::Arc<dyn crate::private_key_jwt::ClientAssertionVerifier>>,
545    /// Service-account repository (client-credentials for `sa_…` clients).
546    service_account_repo: SA,
547    code_repo: AC,
548    tenant_repo: TR,
549    refresh_token_repo: RT,
550    user_repo: UR,
551    /// W4 — the sessions behind refresh grants (plan §4.3).
552    ///
553    /// A repository rather than a snapshot on the refresh-token row because
554    /// the authentication event lives on the session and refresh rotation
555    /// **copies** it there (`axiam_auth::service`, X7.2): reading it is
556    /// therefore reading the same value the authorization code recorded, and
557    /// storing a second copy per refresh token would be a second thing that
558    /// can drift from it.
559    ///
560    /// Read on one path only — re-issuing an ID token for a client on the
561    /// honour lane — and a failed read yields no evidence rather than an
562    /// error, so no grant that works today can start failing because of it.
563    session_repo: SR,
564    auth_config: AuthConfig,
565    refresh_token_lifetime_secs: i64,
566    /// X1 — the `token.pre_issue` interceptor chain.
567    ///
568    /// A field rather than a seventh type parameter, and never an `Option`;
569    /// see `axiam_auth::service::AuthService::reactor_gate` for why. A
570    /// deployment without reactors holds the no-op gate, so every grant runs
571    /// the same code.
572    reactor_gate: SharedReactorGate,
573}

612    /// The evidence a refreshed ID token carries (W4, plan §4.3).
613    ///
614    /// Read from the session the original grant was minted in. Three things
615    /// are worth saying about what comes back:
616    ///
617    /// - **`auth_time` equals the original's**, which is what OIDC Core §12.2
618    ///   requires and what `OIDCCRefreshToken` compares. It holds because
619    ///   refresh rotation copies `authenticated_at` onto the session row it
620    ///   writes rather than stamping the clock (X7.2).
621    /// - **`acr` is the class the authentication achieved**, not the one the
622    ///   original authorization request selected. A refresh carries no
623    ///   authorization request, so there is no relying-party preference to
624    ///   express: where the original may have reported the weaker of two
625    ///   satisfied classes because the RP listed it first
626    ///   ([`crate::acr::report_acr`]), a refreshed token reports what the
627    ///   session proves. Both statements are true of the same authentication,
628    ///   and the one available here is the only one that can be derived
629    ///   without inventing a request.
630    /// - **A session that cannot be read yields no evidence at all**, never an
631    ///   error and never a guess. A refresh that works today must not begin to
632    ///   fail because a session row was reaped.
633    async fn session_evidence_for_refresh(
634        &self,
635        tenant_id: Uuid,
636        session_id: Option<Uuid>,
637    ) -> IdTokenEvidence {
638        let Some(session_id) = session_id else {
639            return IdTokenEvidence::NONE;
640        };
641        match self.session_repo.get_by_id(tenant_id, session_id).await {
642            Ok(session) => IdTokenEvidence {
643                auth_time: Some(session.authenticated_at.timestamp()),
644                acr: Some(crate::acr::acr_for(&session.amr).as_str().to_owned()),
645                amr: axiam_core::models::session::Amr::encode_list(&session.amr),
646            },
647            Err(e) => {
648                tracing::debug!(
649                    error = %e,
650                    %session_id,
651                    "no session row behind this refresh grant; the re-issued ID token \
652                     carries no authentication evidence"
653                );
654                IdTokenEvidence::NONE
655            }
656        }
657    }

670    /// Run `token.pre_issue` and return the `ext` claims to mint with.
671    ///
672    /// Every grant that mints an access token goes through here, which is the
673    /// point: three call sites (authorization code, refresh, client
674    /// credentials) share one hook, so a fourth grant added later cannot
675    /// quietly skip it by forgetting to copy the block.
676    ///
677    /// A veto surfaces as `invalid_grant` — the plan's pinned mapping, and the
678    /// right one: RFC 6749 §5.2 `invalid_grant` is "the grant is not valid for
679    /// this request", which is exactly what a reactor is saying. The reactor's
680    /// reason is **not** echoed to the client (it is third-party text on an
681    /// endpoint reachable with a client credential); the gate has already
682    /// written it to the audit trail with the reactor's id.
683    async fn pre_issue_ext_claims(
684        &self,
685        tenant_id: Uuid,
686        subject: &str,
687        subject_kind: &'static str,
688        client_id: &str,
689        scopes: &[String],
690    ) -> Result<Option<std::collections::BTreeMap<String, String>>, OAuth2Error> {
691        match self
692            .reactor_gate
693            .intercept(
694                tenant_id,
695                reactor_events::TOKEN_PRE_ISSUE,
696                serde_json::json!({
697                    // No token, no credential, no signing key (§22.3) — the
698                    // reactor is told what is about to be minted, not handed
699                    // the means to mint it.
700                    "sub": subject,
701                    "sub_kind": subject_kind,
702                    "client_id": client_id,
703                    "tenant_id": tenant_id,
704                    "scopes": scopes,
705                }),
706            )
707            .await
708        {
709            ReactorOutcome::Allow => Ok(None),
710            ReactorOutcome::Mutate { patch } => {
711                Ok(axiam_auth::token::ext_claims_from_patch(&patch))
712            }
713            ReactorOutcome::Deny { reason } => {
714                tracing::info!(
715                    target: "axiam::reactor",
716                    tenant_id = %tenant_id,
717                    client_id,
718                    "token issuance vetoed by a reactor on token.pre_issue"
719                );
720                debug_assert!(!reason.is_empty());
721                Err(OAuth2Error::InvalidGrant(
722                    "token issuance was refused by policy".into(),
723                ))
724            }
725            // `require_mfa` is a `login.post_auth` answer. The reply validator
726            // refuses it on any other event, so this arm is unreachable through
727            // the wire path — and if it were reachable, the token endpoint has
728            // no step-up to perform, so proceeding "as if allowed" would be
729            // silently discarding a security demand.
730            ReactorOutcome::RequireMfa => {
731                tracing::error!(
732                    target: "axiam::reactor",
733                    tenant_id = %tenant_id,
734                    "a reactor demanded step-up on token.pre_issue, which has no \
735                     step-up to perform — refusing the grant"
736                );
737                Err(OAuth2Error::InvalidGrant(
738                    "token issuance was refused by policy".into(),
739                ))
740            }
741        }
742    }

760    /// Verify a presented client secret against the stored hash — the single
761    /// point at which every grant authenticates a confidential client.
762    ///
763    /// The comparison is constant-time and keyed (HMAC-SHA256 under the server
764    /// pepper, `axiam_auth::client_secret`). A pre-OBS-1 row still stored as a
765    /// bare SHA-256 digest verifies against the legacy scheme and is then
766    /// **transparently upgraded**: the replacement hash is written back with a
767    /// compare-and-swap so a concurrent rotation is never clobbered.
768    ///
769    /// Two properties this function exists to guarantee:
770    ///
771    /// - the upgrade is driven by [`ClientSecretVerdict::MatchNeedsUpgrade`],
772    ///   which the hasher only ever returns on a **successful** verification —
773    ///   a wrong secret can never cause a write; and
774    /// - a failed *write* never becomes a failed *authentication*. The client
775    ///   proved possession of the secret; losing the migration is a warning,
776    ///   and the next request simply retries it.
777    ///
778    /// Cost on the hot path: one HMAC-SHA256 over the secret into a stack
779    /// buffer, plus an `OnceLock` acquire load. No heap allocation and no lock
780    /// — strictly less than the `String` the previous `hash_client_secret`
781    /// allocated per request. The DB write happens at most once per legacy
782    /// client, on its first authentication after the upgrade.
783    async fn verify_client_secret(
784        &self,
785        tenant_id: Uuid,
786        client: &OAuth2Client,
787        presented: &str,
788    ) -> Result<(), OAuth2Error> {
789        let hasher =
790            client_secret::global().map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
791
792        match hasher.verify(presented, &client.client_secret_hash) {
793            ClientSecretVerdict::Match => Ok(()),
794            ClientSecretVerdict::MatchNeedsUpgrade { upgraded_hash } => {
795                match self
796                    .client_repo
797                    .upgrade_client_secret_hash(
798                        tenant_id,
799                        &client.client_id,
800                        &client.client_secret_hash,
801                        &upgraded_hash,
802                    )
803                    .await
804                {
805                    Ok(true) => tracing::info!(
806                        client_id = %client.client_id,
807                        "upgraded legacy client-secret hash to the peppered scheme (OBS-1)"
808                    ),
809                    Ok(false) => tracing::debug!(
810                        client_id = %client.client_id,
811                        "legacy client-secret hash upgrade skipped — a concurrent write won \
812                         the compare-and-swap"
813                    ),
814                    Err(e) => tracing::warn!(
815                        client_id = %client.client_id,
816                        error = %e,
817                        "failed to persist the legacy client-secret hash upgrade (OBS-1); \
818                         authentication succeeded, the upgrade will be retried"
819                    ),
820                }
821                Ok(())
822            }
823            ClientSecretVerdict::Mismatch => {
824                Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()))
825            }
826        }
827    }

829    /// Authenticate a registered client by whichever credential its
830    /// registration says it uses (X5.1).
831    ///
832    /// The **registration** decides, not the request. A client registered for
833    /// `tls_client_auth` is authenticated by its certificate even if it also
834    /// posted a `client_secret`, and a client registered for
835    /// `client_secret_post` is authenticated by its secret even over an mTLS
836    /// connection. Letting the request choose would turn the two methods into
837    /// an OR: an attacker holding either credential could authenticate, which
838    /// is strictly weaker than holding the one the operator registered.
839    ///
840    /// This preserves every ordering property SEC-086 established for the
841    /// secret path — see [`Self::verify_client_secret`] and the callers, where
842    /// the *presence* of a credential is still checked before the client
843    /// lookup so that "no such client" and "client lacks this grant" stay
844    /// indistinguishable.
845    async fn authenticate_client_credential(
846        &self,
847        tenant_id: Uuid,
848        client: &OAuth2Client,
849        presented_secret: Option<&str>,
850        ctx: &TokenRequestContext,
851    ) -> Result<(), OAuth2Error> {
852        if client.token_endpoint_auth_method.is_mtls() {
853            return crate::mtls::authenticate_mtls_client(client, ctx.client_certificate.as_ref());
854        }
855
856        if client.token_endpoint_auth_method.is_private_key_jwt() {
857            return self
858                .authenticate_client_assertion(tenant_id, client, ctx)
859                .await;
860        }
861
862        if client.token_endpoint_auth_method == ClientAuthMethod::ClientSecretBasic {
863            return self
864                .authenticate_client_secret_basic(tenant_id, client, presented_secret, ctx)
865                .await;
866        }
867
868        // `client_secret_post`. W8: a Basic header on this client is **not** a
869        // second way in. It is ignored — the registration decided which
870        // channel carries the credential — and said out loud, because the
871        // silent version of this is indistinguishable from the bug where the
872        // header quietly wins.
873        if ctx.basic_credentials.is_some() {
874            tracing::warn!(
875                client_id = %client.client_id,
876                "an Authorization header using the Basic scheme was presented by a client \
877                 registered for client_secret_post; it is ignored and the request is \
878                 authenticated by the form-body secret (SEC-093: the registration \
879                 decides). If this client meant to use HTTP Basic, register it for \
880                 client_secret_basic"
881            );
882        }
883
884        let secret = presented_secret
885            .ok_or_else(|| OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()))?;
886        self.verify_client_secret(tenant_id, client, secret).await
887    }

889    /// `client_secret_basic` client authentication (W8, RFC 6749 §2.3.1).
890    ///
891    /// The credential is the same peppered-hash comparison
892    /// [`Self::verify_client_secret`] performs for `client_secret_post` — one
893    /// secret, two spellings — so this branch adds no cryptography and no new
894    /// rate-limit bucket. What it adds is the RFC's two negative rules.
895    ///
896    /// # The body secret is refused, *after* the header is verified
897    ///
898    /// RFC 6749 §2.3 forbids a client from using more than one
899    /// authentication method in one request, so a `client_secret` in the body
900    /// of a `client_secret_basic` client is `invalid_request`. The plan
901    /// (§4.7) specified that refusal; it did not specify where in the order it
902    /// goes, and the obvious placement — first — reopens SEC-086. `invalid_request`
903    /// would then be reachable *only* for a client that exists and is
904    /// registered for Basic, making both facts decidable by an unauthenticated
905    /// caller who sends a junk secret in each channel. Verifying the header
906    /// credential first means the malformed-request answer is only ever given
907    /// to a caller who has already proven possession of the secret, and
908    /// everybody else gets the same uniform `invalid_client`. See the W8
909    /// amendment in `claude_dev/basic-op-gap-plan.md` §4.7.
910    ///
911    /// # The header's client id must be the one being authenticated
912    ///
913    /// A header naming `a` and a body naming `b` is two claims of identity in
914    /// one request. It is refused *before* the lookup — see
915    /// [`resolve_client_id`], where it is decidable from the request alone —
916    /// so this method only ever sees an agreeing pair, and the equality check
917    /// here is the defence-in-depth restatement of that.
918    async fn authenticate_client_secret_basic(
919        &self,
920        tenant_id: Uuid,
921        client: &OAuth2Client,
922        presented_secret: Option<&str>,
923        ctx: &TokenRequestContext,
924    ) -> Result<(), OAuth2Error> {
925        let Some(credentials) = ctx.basic_credentials.as_ref() else {
926            // Registered for Basic, presented no Basic header. The body secret
927            // it may have sent instead is deliberately not consulted: that
928            // would be the OR over two credentials SEC-093 exists to prevent.
929            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
930        };
931
932        if credentials.client_id() != client.client_id {
933            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
934        }
935
936        self.verify_client_secret(tenant_id, client, credentials.client_secret())
937            .await?;
938
939        if presented_secret.is_some() {
940            tracing::warn!(
941                client_id = %client.client_id,
942                "a client_secret_basic client presented a client_secret in the request body as \
943                 well as in the Authorization header; RFC 6749 §2.3 permits exactly one \
944                 authentication method per request, so the request is refused"
945            );
946            return Err(OAuth2Error::InvalidRequest(
947                "a client may use only one authentication method per request (RFC 6749 §2.3): \
948                 remove client_secret from the request body"
949                    .into(),
950            ));
951        }
952
953        Ok(())
954    }

956    /// `private_key_jwt` client authentication (X5.1, RFC 7523 §2.2).
957    ///
958    /// The assertion comes from the request body, which is why it is read out
959    /// of [`TokenRequestContext`]'s sibling — the form — rather than the
960    /// context: unlike a certificate or a DPoP proof, it is not a property of
961    /// the connection. Everything else about the shape matches the mTLS branch
962    /// exactly: the **registration** decides the method, the credential is
963    /// checked against what the client registered, and every failure answers
964    /// the same `invalid_client`.
965    async fn authenticate_client_assertion(
966        &self,
967        tenant_id: Uuid,
968        client: &OAuth2Client,
969        ctx: &TokenRequestContext,
970    ) -> Result<(), OAuth2Error> {
971        let Some(verifier) = self.assertion_verifier.as_ref() else {
972            tracing::error!(
973                client_id = %client.client_id,
974                "a client is registered for private_key_jwt but this deployment has no assertion \
975                 verifier configured; refusing rather than falling back to another credential"
976            );
977            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
978        };
979
980        let Some(assertion) = ctx.client_assertion.as_deref() else {
981            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
982        };
983        crate::private_key_jwt::check_assertion_type(ctx.client_assertion_type.as_deref())
984            .map_err(OAuth2Error::from)?;
985
986        verifier.verify(tenant_id, client, assertion).await
987    }

989    /// Decide the `cnf` confirmation claim for a token about to be issued
990    /// (X5.1, RFC 8705 §3).
991    ///
992    /// Returns `Ok(None)` for every client that has not asked for binding —
993    /// which is every client that existed before X5.1 — so this is a branch
994    /// predictor's dream and costs nothing on the ordinary path.
995    ///
996    /// When a client *has* asked for binding and no certificate is present,
997    /// this **refuses** rather than issuing an unbound token. Silently
998    /// downgrading would be the worst possible behaviour: the operator
999    /// configured binding, the resource servers were told to expect it, and
1000    /// the one request that arrives without a certificate would be the one
1001    /// that gets a bearer token.
1002    /// SEC-096: `pub` because the token-exchange and uma-ticket grants are
1003    /// dispatched from `axiam-api-rest` rather than through
1004    /// [`Self::exchange`], and they must reach the same decision. A second
1005    /// copy of this rule in the handler is how one of the two ends up
1006    /// silently issuing an unbound token.
1007    pub fn certificate_binding_for(
1008        &self,
1009        client: &OAuth2Client,
1010        ctx: &TokenRequestContext,
1011    ) -> Result<Option<axiam_auth::token::CnfClaim>, OAuth2Error> {
1012        let mut cnf: Option<axiam_auth::token::CnfClaim> = None;
1013
1014        if crate::fapi::wants_certificate_binding(client) {
1015            let Some(thumbprint) = ctx.certificate_thumbprint() else {
1016                tracing::warn!(
1017                    client_id = %client.client_id,
1018                    "client requires certificate-bound access tokens but presented no client \
1019                     certificate; refusing rather than issuing an unbound token"
1020                );
1021                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1022            };
1023            cnf = Some(axiam_auth::token::CnfClaim::from_certificate_thumbprint(
1024                thumbprint,
1025            ));
1026        }
1027
1028        if crate::fapi::wants_dpop_binding(client) {
1029            let Some(jkt) = ctx.dpop_thumbprint() else {
1030                tracing::warn!(
1031                    client_id = %client.client_id,
1032                    "client requires DPoP-bound access tokens but presented no verified DPoP \
1033                     proof; refusing rather than issuing an unbound token"
1034                );
1035                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1036            };
1037            // A client that asked for both gets both, in one `cnf`. The
1038            // resource server then has to satisfy both — see
1039            // `axiam_auth::token::verify_token_binding`, where "both present"
1040            // is a conjunction.
1041            cnf = Some(match cnf {
1042                Some(existing) => existing.with_dpop_thumbprint(jkt),
1043                None => axiam_auth::token::CnfClaim::from_dpop_thumbprint(jkt),
1044            });
1045        }
1046
1047        Ok(cnf)
1048    }

1050    /// Client-credentials for a **service account** (`sa_…` client id).
1051    ///
1052    /// Service accounts are AXIAM's machine principal, and until now the only
1053    /// way one could authenticate in a running server was mTLS
1054    /// (`POST /api/v1/auth/device`). `create`/`rotate_secret` handed the
1055    /// operator a `client_secret` that **no flow accepted** — this is the path
1056    /// that makes it usable.
1057    ///
1058    /// Three deliberate differences from the `oauth2_client` branch:
1059    ///
1060    /// 1. **The secret is verified BEFORE the status check.** Rejecting a
1061    ///    disabled account without verifying would let an unauthenticated
1062    ///    caller distinguish "exists but disabled" from "does not exist" by
1063    ///    timing. Verifying first means status is only ever consulted for a
1064    ///    caller who already proved possession of the secret — and even then
1065    ///    the error returned is the same generic `invalid_client`.
1066    /// 2. **No scope may be requested.** A service account registers no scopes,
1067    ///    and the `oauth2_client` branch's rule is that a requested scope must
1068    ///    be a subset of the registered ones. With an empty registered set the
1069    ///    only subset is the empty one, so any requested scope is
1070    ///    `invalid_scope`. Authorization for a service account comes from the
1071    ///    roles assigned to its subject, exactly as on the mTLS path.
1072    /// 3. **The token's `sub` is the service-account id**, not the client id,
1073    ///    so the RBAC engine resolves the same principal whether the account
1074    ///    authenticated by certificate or by secret.
1075    async fn handle_client_credentials_service_account(
1076        &self,
1077        tenant_id: Uuid,
1078        client_id: &str,
1079        client_secret: &str,
1080        requested_scope: Option<&str>,
1081        started: std::time::Instant,
1082    ) -> Result<TokenResponse, OAuth2Error> {
1083        let t_client_lookup = std::time::Instant::now();
1084        let sa = self
1085            .service_account_repo
1086            .get_by_client_id(tenant_id, client_id)
1087            .await
1088            .map_err(|e| match e {
1089                // Same taxonomy as the oauth2_client branch: only a genuinely
1090                // unknown client is `invalid_client`; a DB outage must never
1091                // masquerade as bad credentials (error-oracle, QUAL-03/D-11).
1092                // The description is CLIENT_AUTH_FAILED here too, so an
1093                // unauthenticated caller cannot learn whether an `sa_…`
1094                // client id exists (SEC-086).
1095                AxiamError::NotFound { .. } => {
1096                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
1097                }
1098                other => OAuth2Error::ServerError(other.to_string()),
1099            })?;
1100        let client_lookup_us = t_client_lookup.elapsed().as_micros() as u64;
1101
1102        // Defence in depth (§17.2 residual 6). The lookup above is already
1103        // tenant-scoped in SQL, so this can only fire if that predicate is
1104        // ever dropped or a repository implementation ignores `tenant_id` —
1105        // exactly the regression the assertion exists to catch. Without it the
1106        // cross-tenant property of this grant lives entirely in a `WHERE`
1107        // clause two crates away, and no test in this crate can prove it
1108        // because the mocks do not filter by tenant.
1109        //
1110        // Deliberately checked *before* the secret is verified: a row from the
1111        // wrong tenant must not have its secret compared at all, and the
1112        // presented client id is the caller's own input, so refusing early
1113        // leaks nothing it did not already know.
1114        if sa.tenant_id != tenant_id {
1115            tracing::error!(
1116                requested_tenant = %tenant_id,
1117                account_tenant = %sa.tenant_id,
1118                "service-account lookup returned a row from another tenant — refusing; \
1119                 this indicates a dropped tenant predicate in the repository layer"
1120            );
1121            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1122        }
1123
1124        let t_secret_verify = std::time::Instant::now();
1125        let secret_result = self
1126            .verify_service_account_secret(tenant_id, &sa, client_secret)
1127            .await;
1128        let secret_verify_us = t_secret_verify.elapsed().as_micros() as u64;
1129        secret_result?;
1130
1131        // Only now — see point 1 in the doc comment.
1132        if sa.status != UserStatus::Active {
1133            tracing::info!(
1134                client_id = %sa.client_id,
1135                status = ?sa.status,
1136                "service-account client-credentials refused: account is not active"
1137            );
1138            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1139        }
1140
1141        if let Some(scope) = requested_scope.filter(|s| !s.trim().is_empty()) {
1142            return Err(OAuth2Error::InvalidScope(format!(
1143                "unregistered scopes: {} — a service account registers no scopes; \
1144                 its authorization comes from the roles assigned to it",
1145                scope.split_whitespace().collect::<Vec<_>>().join(", ")
1146            )));
1147        }
1148
1149        let t_tenant_lookup = std::time::Instant::now();
1150        let tenant = self
1151            .tenant_repo
1152            .get_by_id(tenant_id)
1153            .await
1154            .map_err(|e| match e {
1155                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
1156                other => OAuth2Error::ServerError(other.to_string()),
1157            })?;
1158        let tenant_lookup_us = t_tenant_lookup.elapsed().as_micros() as u64;
1159
1160        let t_token_mint = std::time::Instant::now();
1161        // X1 — the fourth issuance path, hooked like the other three.
1162        let ext = self
1163            .pre_issue_ext_claims(
1164                tenant_id,
1165                &sa.id.to_string(),
1166                "service_account",
1167                &sa.client_id,
1168                &[],
1169            )
1170            .await?;
1171        let access_token = issue_service_account_client_credentials_token_enriched(
1172            sa.id,
1173            tenant_id,
1174            tenant.organization_id,
1175            &[],
1176            &self.auth_config,
1177            ext,
1178        )
1179        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1180        let token_mint_us = t_token_mint.elapsed().as_micros() as u64;
1181
1182        let span = tracing::Span::current();
1183        span.record("client_lookup_us", client_lookup_us);
1184        span.record("secret_verify_us", secret_verify_us);
1185        span.record("tenant_lookup_us", tenant_lookup_us);
1186        span.record("token_mint_us", token_mint_us);
1187        span.record("handler_total_us", started.elapsed().as_micros() as u64);
1188
1189        Ok(TokenResponse {
1190            access_token,
1191            token_type: "Bearer".to_string(),
1192            expires_in: self.auth_config.access_token_lifetime_secs,
1193            refresh_token: None,
1194            scope: None,
1195            id_token: None,
1196        })
1197    }

1199    /// [`Self::verify_client_secret`] for a service account.
1200    ///
1201    /// Same contract, against the service-account table: the upgrade is driven
1202    /// only by `MatchNeedsUpgrade` (so a wrong secret can never cause a write),
1203    /// and a failed write is a warning rather than a failed authentication.
1204    ///
1205    /// This is also what makes the §15.2 Obs 4 lazy migration *reachable*: with
1206    /// no verification path, `upgrade_client_secret_hash` could never fire for a
1207    /// service account, so legacy rows could not drain. They can now.
1208    async fn verify_service_account_secret(
1209        &self,
1210        tenant_id: Uuid,
1211        sa: &ServiceAccount,
1212        presented: &str,
1213    ) -> Result<(), OAuth2Error> {
1214        let hasher =
1215            client_secret::global().map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1216
1217        match hasher.verify(presented, &sa.client_secret_hash) {
1218            ClientSecretVerdict::Match => Ok(()),
1219            ClientSecretVerdict::MatchNeedsUpgrade { upgraded_hash } => {
1220                match self
1221                    .service_account_repo
1222                    .upgrade_client_secret_hash(
1223                        tenant_id,
1224                        &sa.client_id,
1225                        &sa.client_secret_hash,
1226                        &upgraded_hash,
1227                    )
1228                    .await
1229                {
1230                    Ok(true) => tracing::info!(
1231                        client_id = %sa.client_id,
1232                        "upgraded legacy service-account secret hash to the peppered scheme"
1233                    ),
1234                    Ok(false) => tracing::debug!(
1235                        client_id = %sa.client_id,
1236                        "legacy service-account secret-hash upgrade skipped — a concurrent \
1237                         write won the compare-and-swap"
1238                    ),
1239                    Err(e) => tracing::warn!(
1240                        client_id = %sa.client_id,
1241                        error = %e,
1242                        "failed to persist the legacy service-account secret-hash upgrade; \
1243                         authentication succeeded, the upgrade will be retried"
1244                    ),
1245                }
1246                Ok(())
1247            }
1248            ClientSecretVerdict::Mismatch => {
1249                Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()))
1250            }
1251        }
1252    }

1258    /// [`TokenRequestContext::default()`] on a connection that has none; every

1274    /// Exchange an authorization code for tokens (RFC 6749 section 4.1.3).
1275    /// Revoke what a replayed authorization code minted (RFC 6749 §10.5).
1276    ///
1277    /// # What gets revoked, and why it is the session
1278    ///
1279    /// The spec says "all tokens previously issued based on that authorization
1280    /// code". An AXIAM access token is a stateless EdDSA JWT — there is
1281    /// nothing to delete — but every resource request already checks that the
1282    /// session named by the token's `sid` is still live
1283    /// (`is_session_active`, on the hot path and cached). Invalidating that
1284    /// session therefore stops the minted access token at the next request,
1285    /// with no new store and no new per-request read.
1286    ///
1287    /// # The two costs, stated plainly
1288    ///
1289    /// The session is the **browser** session the authorization happened in,
1290    /// so revoking it signs the user out of everything that session reaches,
1291    /// AXIAM's own admin UI included. And a legitimate client that retries
1292    /// after losing a `200` response replays a code exactly as an attacker
1293    /// does — the server cannot tell them apart, which is precisely why
1294    /// §10.5's answer is to revoke rather than to guess.
1295    ///
1296    /// # Not an oracle
1297    ///
1298    /// Both branches return the same `invalid_grant` from the caller, and a
1299    /// hash naming no row revokes nothing. A replay does cost one extra read
1300    /// and one write, so the two are distinguishable by timing — to an
1301    /// attacker who already holds a real authorization code, which is the
1302    /// position this whole path exists to answer.
1303    async fn revoke_after_code_replay(
1304        &self,
1305        tenant_id: Uuid,
1306        code_hash: &str,
1307        client_id: &str,
1308        redirect_uri: &str,
1309    ) {
1310        let session_id = match self
1311            .code_repo
1312            .replayed_session(tenant_id, code_hash, client_id, redirect_uri)
1313            .await
1314        {
1315            Ok(Some(id)) => id,
1316            // Nothing to revoke: an unknown code, or one issued with no
1317            // session behind it. Not a replay worth reporting.
1318            Ok(None) => return,
1319            Err(e) => {
1320                tracing::warn!(
1321                    error = %e,
1322                    "could not check whether a refused authorization code was a replay; \
1323                     the redemption is still denied"
1324                );
1325                return;
1326            }
1327        };
1328
1329        match self.session_repo.invalidate(tenant_id, session_id).await {
1330            Ok(()) => tracing::warn!(
1331                client_id = %client_id,
1332                session_id = %session_id,
1333                "authorization code replay detected; revoked the session it was issued from \
1334                 (RFC 6749 §10.5)"
1335            ),
1336            Err(e) => tracing::warn!(
1337                error = %e,
1338                client_id = %client_id,
1339                session_id = %session_id,
1340                "authorization code replay detected but the session could not be revoked; \
1341                 the redemption is still denied"
1342            ),
1343        }
1344    }

1346    async fn handle_authorization_code(
1347        &self,
1348        tenant_id: Uuid,
1349        req: TokenRequest,
1350        ctx: &TokenRequestContext,
1351    ) -> Result<TokenResponse, OAuth2Error> {
1352        let code = req
1353            .code
1354            .as_deref()
1355            .ok_or_else(|| OAuth2Error::InvalidRequest("code is required".into()))?;
1356        let redirect_uri = req
1357            .redirect_uri
1358            .as_deref()
1359            .ok_or_else(|| OAuth2Error::InvalidRequest("redirect_uri is required".into()))?;
1360        let client_id = resolve_client_id(req.client_id.as_deref(), ctx)?;
1361
1362        // Require client_secret — all clients are confidential (no
1363        // public-client distinction exists yet).
1364        //
1365        // SEC-086 (second pass): this check MUST precede the client lookup.
1366        // When it ran after, a caller posting no `client_secret` at all got
1367        // three distinguishable answers — `invalid_client`/generic for an
1368        // unknown client, `unauthorized_client` for a client lacking the
1369        // grant, and `invalid_client`/"client_secret is required" for a
1370        // client that had it — so client existence stayed decidable on this
1371        // grant even after the description was unified everywhere else.
1372        // `client_credentials` and `refresh_token` already ordered it this
1373        // way; only this grant did not.
1374        //
1375        // X5.1 widens "presented a client_secret" to "presented *a*
1376        // credential", because an mTLS client sends no secret at all. W8
1377        // widens it once more, for the client that sent its secret in the
1378        // `Authorization` header instead of the body. The widened check is
1379        // still decidable **before** the lookup — it asks only what this
1380        // request carries, never what this client registered — so SEC-086's
1381        // property is preserved exactly: a caller with no credential gets one
1382        // answer regardless of whether the client id exists.
1383        let client_secret = req.client_secret.as_deref();
1384        if ctx.carries_no_client_credential(client_secret) {
1385            return Err(OAuth2Error::InvalidClient(
1386                "client authentication is required".into(),
1387            ));
1388        }
1389
1390        // Authenticate client
1391        let client = self
1392            .client_repo
1393            .get_by_client_id(tenant_id, client_id)
1394            .await
1395            .map_err(|e| match e {
1396                // QUAL-03/D-11: only a genuinely-unknown client maps to
1397                // invalid_client. Any other error (e.g. a DB outage) must
1398                // surface as a distinct server error, never masquerade as
1399                // bad client credentials (error-oracle). The distinction is
1400                // internal only — the caller-visible description is
1401                // CLIENT_AUTH_FAILED either way, so it cannot be used to
1402                // probe which client ids exist (SEC-086).
1403                AxiamError::NotFound { .. } => {
1404                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
1405                }
1406                other => OAuth2Error::ServerError(other.to_string()),
1407            })?;
1408
1409        // Secret verification must also precede the grant-type check, and for
1410        // the same reason: `unauthorized_client` is only reachable by a caller
1411        // who has *already proven possession of the secret*. Ordered the other
1412        // way, supplying any dummy secret still separated "client exists but
1413        // lacks this grant" from "no such client" — a second oracle on the
1414        // same grant, which moving the presence check alone does not close.
1415        // Both safe grants verify first; this now matches them exactly.
1416        //
1417        // X5.1: the profile gate runs *after* authentication, for the same
1418        // reason. `enforce_token_request` reports a fact about the client's
1419        // registration (its profile, its auth method), and reporting that to a
1420        // caller who has not proven possession of a credential would be a new
1421        // oracle on a path that just spent three comment blocks closing one.
1422        self.authenticate_client_credential(tenant_id, &client, client_secret, ctx)
1423            .await?;
1424        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
1425
1426        // Verify client is authorized for authorization_code grant
1427        if !client.grant_types.iter().any(|s| s == "authorization_code") {
1428            return Err(OAuth2Error::UnauthorizedClient(
1429                "client not authorized for authorization_code grant".into(),
1430            ));
1431        }
1432
1433        // Look up the authorization code without consuming it so we
1434        // can verify PKCE *before* marking it as used.  This prevents
1435        // an attacker from burning a valid code by intentionally
1436        // failing PKCE verification.
1437        let code_hash = crate::authorize::hash_code(code);
1438        let auth_code = match self
1439            .code_repo
1440            .get_by_hash(tenant_id, &code_hash, client_id, redirect_uri)
1441            .await
1442        {
1443            Ok(row) => row,
1444            Err(_) => {
1445                // RFC 6749 §10.5's revocation belongs HERE, not only on the
1446                // `consume` failure below — and putting it only there made it
1447                // unreachable for the case it was written for.
1448                //
1449                // `get_by_hash` requires `used = false`. A replayed code is
1450                // therefore refused by this lookup and never reaches `consume`,
1451                // so the revocation guarded by `consume` fired for a race and
1452                // for nothing else. The observable consequence was that an
1453                // access token minted from a replayed code kept working:
1454                // `oidcc-codereuse-30seconds` and
1455                // `fapi2-security-profile-final-attempt-reuse-authorization-code-after-one-second`
1456                // both call the resource endpoint afterwards expecting 4xx and
1457                // both got 200, and the server logged nothing at all — the
1458                // revocation was running, finding no row, and returning quietly.
1459                //
1460                // Safe to call on every lookup failure, including the two that
1461                // are not replays. `replayed_session` matches `used = true`
1462                // together with the same client and redirect_uri pair, so an
1463                // unknown or merely expired code selects nothing and revokes
1464                // nothing; the cost is one read on a request that has already
1465                // failed.
1466                self.revoke_after_code_replay(tenant_id, &code_hash, client_id, redirect_uri)
1467                    .await;
1468                return Err(OAuth2Error::InvalidGrant(
1469                    "authorization code is invalid, expired, or already used".into(),
1470                ));
1471            }
1472        };
1473
1474        // Verify PKCE before consuming the code
1475        if let Some(ref challenge) = auth_code.code_challenge {
1476            let verifier = req.code_verifier.as_deref().ok_or_else(|| {
1477                OAuth2Error::InvalidGrant("code_verifier required for PKCE".into())
1478            })?;
1479            if !pkce::verify_pkce(verifier, challenge) {
1480                return Err(OAuth2Error::InvalidGrant("PKCE verification failed".into()));
1481            }
1482        }
1483
1484        // RFC 9449 §10, and beside PKCE for the same reason PKCE is here.
1485        //
1486        //   When a token request is received, the authorization server computes
1487        //   the JWK Thumbprint of the proof-of-possession public key in the DPoP
1488        //   proof and verifies that it matches the dpop_jkt parameter value in
1489        //   the authorization request.  If they do not match, it MUST reject
1490        //   the request.
1491        //
1492        // `auth_code.dpop_jkt` is that parameter value as it stood when the
1493        // client committed to it — whether it arrived as the `dpop_jkt`
1494        // parameter or as the thumbprint of a DPoP proof presented at the PAR
1495        // endpoint, which §10.1 says binds identically ("behave as if the
1496        // contained public key's thumbprint was provided using dpop_jkt").
1497        //
1498        // Three cases, and only the first is new behaviour:
1499        //
1500        //   - bound, and the proof's key differs (or there is no proof at all)
1501        //     — refuse. A code pinned to a key is worthless to whoever holds
1502        //     the code without the key, which is the entire point of §10.
1503        //   - bound, and the keys match — proceed.
1504        //   - not bound — proceed, and let `certificate_binding_for` apply
1505        //     whatever the *registration* requires. A client that never used
1506        //     the parameter sees exactly the answer it saw before §10 existed.
1507        //
1508        // `invalid_grant` rather than `invalid_dpop_proof`: the proof is
1509        // perfectly valid — `dpop_from_request` already verified its
1510        // signature, `htm`, `htu` and freshness — and what fails is the
1511        // *grant*, which is bound to a key this caller has not demonstrated.
1512        // Saying `invalid_dpop_proof` would send an honest client debugging
1513        // its proof generation over a code it should simply push again.
1514        if let Some(ref bound_jkt) = auth_code.dpop_jkt {
1515            let presented = ctx.dpop_thumbprint();
1516            if presented != Some(bound_jkt.as_str()) {
1517                tracing::debug!(
1518                    client_id = %client_id,
1519                    presented = presented.unwrap_or("<none>"),
1520                    "authorization code is bound to a DPoP key the token request did not prove"
1521                );
1522                return Err(OAuth2Error::InvalidGrant(
1523                    "this authorization code is bound to a DPoP key (RFC 9449 §10); the token \
1524                     request must carry a DPoP proof for that same key"
1525                        .into(),
1526                ));
1527            }
1528        }
1529
1530        // Now atomically consume (mark as used) the code.
1531        if self
1532            .code_repo
1533            .consume(tenant_id, &code_hash, client_id, redirect_uri)
1534            .await
1535            .is_err()
1536        {
1537            // RFC 6749 §10.5: a code used more than once must be denied — the
1538            // `Err` above already does that — and the server "SHOULD revoke
1539            // (when possible) all tokens previously issued based on that
1540            // authorization code".
1541            //
1542            // Best-effort and never fatal: the denial is the guarantee, the
1543            // revocation is the clean-up, and a caller must not learn from a
1544            // 500 that its replay found something.
1545            self.revoke_after_code_replay(tenant_id, &code_hash, client_id, redirect_uri)
1546                .await;
1547            return Err(OAuth2Error::InvalidGrant(
1548                "authorization code is invalid, expired, or already used".into(),
1549            ));
1550        }
1551
1552        // Resolve org_id from tenant
1553        let tenant = self
1554            .tenant_repo
1555            .get_by_id(tenant_id)
1556            .await
1557            .map_err(|e| match e {
1558                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
1559                other => OAuth2Error::ServerError(other.to_string()),
1560            })?;
1561
1562        // Issue access token (include scopes from the authorization code).
1563        // OAuth2 auth-code flow has no persistent session row — use random jti.
1564        //
1565        // X5.1: `cnf` is `None` for every client that has not asked for
1566        // certificate binding, which makes this byte-identical to what the
1567        // pre-X5 call produced.
1568        let cnf = self.certificate_binding_for(&client, ctx)?;
1569        // Read before `cnf` is moved into the claims. RFC 9449 §5: a DPoP-bound
1570        // token is announced as such, and everything else stays `Bearer`.
1571        let token_type = crate::dpop::token_type_for(cnf.as_ref());
1572        // X1 — the code has already been consumed at this point, so a veto
1573        // costs the caller the code. That is the correct order: the code is
1574        // single-use and PKCE-verified, and leaving it spendable while an
1575        // external process decides would hand an attacker a retry window.
1576        let ext = self
1577            .pre_issue_ext_claims(
1578                tenant_id,
1579                &auth_code.user_id.to_string(),
1580                "user",
1581                client_id,
1582                &auth_code.scopes,
1583            )
1584            .await?;
1585        // W7 — the token names the relying party it was granted to
1586        // (RFC 9068 §2.2). UserInfo needs it to know *whose* consent record to
1587        // look for before releasing `phone_number` or `address`, and to
1588        // recognise a `fapi2`-issued token at a point where no authorization
1589        // request is in hand. See `AccessTokenClaims::client_id`.
1590        //
1591        // Built from `AccessTokenSpec` rather than through
1592        // `issue_access_token_for_client`, which every other issuance site
1593        // still uses. That wrapper already carries eleven parameters, and OIDC
1594        // Core §5.5 would have made twelve — the builder exists so that a
1595        // capability only this path needs does not become an argument every
1596        // caller has to pass `None` for.
1597        let access_token = axiam_auth::token::AccessTokenSpec::user(
1598            auth_code.user_id,
1599            tenant_id,
1600            tenant.organization_id,
1601            uuid::Uuid::new_v4().to_string(),
1602        )
1603        .aud(axiam_auth::token::AUD_USER)
1604        .scopes(&auth_code.scopes)
1605        .cnf(cnf)
1606        .ext(ext)
1607        .client_id(Some(client_id))
1608        // The session this authorization was performed in. `jti` stays a
1609        // unique per-token id; the session travels in `sid`, which is what
1610        // the session-revocation check reads. Without it this token is
1611        // refused at every session-validated endpoint, UserInfo included.
1612        .session(auth_code.session_id)
1613        // OIDC Core §5.5 — the claims this authorization asked for by name,
1614        // resolved at the authorization endpoint and carried on the code.
1615        // Empty for every grant that sent no `claims` parameter, and an empty
1616        // list leaves the token byte-identical to what it was before §5.5.
1617        .requested_userinfo_claims(&auth_code.requested_userinfo_claims)
1618        .issue(&self.auth_config)
1619        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1620
1621        // Only issue a refresh token when the client is authorized
1622        // for the refresh_token grant type.
1623        let refresh_token = if client.grant_types.iter().any(|s| s == "refresh_token") {
1624            let raw_refresh = generate_refresh_token();
1625            let refresh_hash = hash_refresh_token(&raw_refresh);
1626            let refresh_expires =
1627                Utc::now() + chrono::Duration::seconds(self.refresh_token_lifetime_secs);
1628
1629            self.refresh_token_repo
1630                .create(CreateRefreshToken {
1631                    tenant_id,
1632                    token_hash: refresh_hash,
1633                    client_id: client_id.to_string(),
1634                    user_id: Some(auth_code.user_id),
1635                    scopes: auth_code.scopes.clone(),
1636                    session_id: auth_code.session_id,
1637                    expires_at: refresh_expires,
1638                })
1639                .await
1640                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1641            Some(raw_refresh)
1642        } else {
1643            None
1644        };
1645
1646        // Issue an ID token when the `openid` scope was requested.
1647        let id_token = if auth_code.scopes.iter().any(|s| s == "openid") {
1648            let user = self
1649                .user_repo
1650                .get_by_id(tenant_id, auth_code.user_id)
1651                .await
1652                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1653            // X7.2 — the evidence the code was issued with, released only to a
1654            // client on the lane that asked for it. `emits_session_evidence`
1655            // answers `false` for every client in this wave, so this is
1656            // `NONE` and the claim set is what it has always been.
1657            let evidence = if crate::fapi::emits_session_evidence(&client) {
1658                session_evidence_of(&auth_code)
1659            } else {
1660                IdTokenEvidence::NONE
1661            };
1662            Some(
1663                issue_id_token(
1664                    auth_code.user_id,
1665                    client_id,
1666                    auth_code.nonce.as_deref(),
1667                    Some(&user.username),
1668                    &auth_code.scopes,
1669                    &self.auth_config,
1670                    auth_code.session_id,
1671                    &evidence,
1672                )
1673                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?,
1674            )
1675        } else {
1676            None
1677        };
1678
1679        let scope = if auth_code.scopes.is_empty() {
1680            None
1681        } else {
1682            Some(auth_code.scopes.join(" "))
1683        };
1684
1685        Ok(TokenResponse {
1686            access_token,
1687            token_type,
1688            expires_in: self.auth_config.access_token_lifetime_secs,
1689            refresh_token,
1690            scope,
1691            id_token,
1692        })
1693    }

1695    /// Client credentials grant (RFC 6749 section 4.4).
1696    ///
1697    /// Machine-to-machine flow — no user context, no refresh token.
1698    ///
1699    /// # Per-stage timing (I5)
1700    ///
1701    /// This handler is the subject of the run-4 TLS plateau investigation, so it
1702    /// carries a stage breakdown on its span: `client_lookup_us`,
1703    /// `secret_verify_us`, `tenant_lookup_us`, `token_mint_us` and
1704    /// `handler_total_us`, recorded on the `oauth2.client_credentials` span and
1705    /// re-emitted as one DEBUG event on `target: "axiam::perf"`.
1706    ///
1707    /// **Cost, and why it is left permanently on:** five `Instant::now()` calls
1708    /// (~20 ns each on a `clock_gettime` vDSO read) and five
1709    /// `Span::record` calls that are no-ops when no subscriber is interested —
1710    /// together well under 0.1 % of the ~370 µs this handler takes even in the
1711    /// *fast* (plaintext, 2 727 req/s) configuration. No flag gates the
1712    /// measurement; only the *reporting* is gated, by the ordinary tracing
1713    /// level (`RUST_LOG=axiam_oauth2=debug`, or any subscriber that samples the
1714    /// span's fields).
1715    ///
1716    /// There is deliberately **no "token persist" stage**: the
1717    /// `client_credentials` grant issues no refresh token and writes nothing —
1718    /// the only DB work is the two reads below. That, plus the fact that the
1719    /// client secret is verified with a single keyed HMAC-SHA256 rather than
1720    /// Argon2id (see `axiam_auth::client_secret`), is why this endpoint
1721    /// measures 2 727 req/s where password login measures 69. The OBS-1
1722    /// remediation deliberately kept this MAC-bound rather than KDF-bound:
1723    /// client secrets are 256-bit CSPRNG values *and* the digest is now keyed,
1724    /// so the offline-guessing threat a KDF defends against does not apply.
1725    #[tracing::instrument(
1726        name = "oauth2.client_credentials",
1727        skip(self, req),
1728        fields(
1729            client_lookup_us = tracing::field::Empty,
1730            secret_verify_us = tracing::field::Empty,
1731            tenant_lookup_us = tracing::field::Empty,
1732            token_mint_us = tracing::field::Empty,
1733            handler_total_us = tracing::field::Empty,
1734        )
1735    )]
1736    async fn handle_client_credentials(
1737        &self,
1738        tenant_id: Uuid,
1739        req: TokenRequest,
1740        ctx: &TokenRequestContext,
1741    ) -> Result<TokenResponse, OAuth2Error> {
1742        let started = std::time::Instant::now();
1743        let client_id = resolve_client_id(req.client_id.as_deref(), ctx)?;
1744        // X5.1: as on the authorization-code grant, "presented a client_secret"
1745        // widens to "presented a credential", so an mTLS client that sends no
1746        // secret is not turned away before its certificate is ever looked at,
1747        // and (W8) neither is a client whose secret arrived in the
1748        // `Authorization` header. Still decidable from the request alone, so
1749        // no client-existence oracle is created.
1750        let client_secret = req.client_secret.as_deref();
1751        if ctx.carries_no_client_credential(client_secret) {
1752            return Err(OAuth2Error::InvalidClient(
1753                "client authentication is required".into(),
1754            ));
1755        }
1756
1757        // Service accounts are the other machine principal in AXIAM and live in
1758        // their own table. Both client-id families are **server-generated** with
1759        // disjoint prefixes (`oa_` for `oauth2_client`, `sa_` for
1760        // `service_account`), so the prefix selects the table unambiguously and
1761        // keeps this path at one lookup.
1762        //
1763        // The prefix is NOT a security decision: it only picks which table to
1764        // look in. The presented secret must still verify against the row that
1765        // is actually found, so guessing a prefix only routes a caller to a
1766        // table where their `client_id` does not exist — an `invalid_client`,
1767        // exactly as before.
1768        if client_id.starts_with(SERVICE_ACCOUNT_CLIENT_ID_PREFIX) {
1769            // A service account has no `token_endpoint_auth_method` — it is a
1770            // different table with a different lifecycle, and its mTLS story is
1771            // the device-auth path (`POST /api/v1/auth/device`), not RFC 8705.
1772            // So this branch still requires a secret. The error is the generic
1773            // one rather than "client_secret is required", because by this
1774            // point the caller has named an `sa_…` id and a specific message
1775            // would confirm that the id belongs to the service-account family.
1776            let Some(client_secret) = client_secret else {
1777                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1778            };
1779            return self
1780                .handle_client_credentials_service_account(
1781                    tenant_id,
1782                    client_id,
1783                    client_secret,
1784                    req.scope.as_deref(),
1785                    started,
1786                )
1787                .await;
1788        }
1789
1790        // Stage 1 — client lookup (DB round-trip #1).
1791        let t_client_lookup = std::time::Instant::now();
1792        let client = self
1793            .client_repo
1794            .get_by_client_id(tenant_id, client_id)
1795            .await
1796            .map_err(|e| match e {
1797                // QUAL-03/D-11: only a genuinely-unknown client maps to
1798                // invalid_client. Any other error (e.g. a DB outage) must
1799                // surface as a distinct server error, never masquerade as
1800                // bad client credentials (error-oracle). The distinction is
1801                // internal only — the caller-visible description is
1802                // CLIENT_AUTH_FAILED either way, so it cannot be used to
1803                // probe which client ids exist (SEC-086).
1804                AxiamError::NotFound { .. } => {
1805                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
1806                }
1807                other => OAuth2Error::ServerError(other.to_string()),
1808            })?;
1809        let client_lookup_us = t_client_lookup.elapsed().as_micros() as u64;
1810
1811        // Stage 2 — client-secret verification. Keyed HMAC-SHA256 into a stack
1812        // buffer + constant-time compare (`axiam_auth::client_secret`); no heap
1813        // allocation, no lock, and this does NOT touch the Argon2id
1814        // `crypto_semaphore` that bounds password verification. A pre-OBS-1
1815        // row additionally costs one SHA-256 and, once, one DB write to
1816        // migrate; the timing recorded here therefore includes that one-off
1817        // upgrade for a legacy client's first authentication.
1818        //
1819        // X5.1: for an mTLS client this stage is a string comparison against
1820        // the certificate rustls already verified during the handshake — no
1821        // cryptography at all, since the signature check happened once per
1822        // connection rather than once per request. That is the mechanical
1823        // reason certificate-bound tokens are cheap here, and what
1824        // `bench-quick` measures.
1825        let t_secret_verify = std::time::Instant::now();
1826        let secret_result = self
1827            .authenticate_client_credential(tenant_id, &client, client_secret, ctx)
1828            .await;
1829        let secret_verify_us = t_secret_verify.elapsed().as_micros() as u64;
1830        secret_result?;
1831        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
1832
1833        // Verify grant type is allowed
1834        if !client.grant_types.iter().any(|s| s == "client_credentials") {
1835            return Err(OAuth2Error::UnauthorizedClient(
1836                "client not authorized for client_credentials grant".into(),
1837            ));
1838        }
1839
1840        // Resolve scopes: use requested scope if provided, else client's.
1841        // Validate requested scopes are a subset of the client's allowed
1842        // scopes to prevent minting tokens with arbitrary privileges.
1843        let scopes = match req.scope.as_deref() {
1844            Some(s) => {
1845                let requested: Vec<String> = s.split_whitespace().map(String::from).collect();
1846                let invalid: Vec<&str> = requested
1847                    .iter()
1848                    .filter(|sc| !client.scopes.contains(sc))
1849                    .map(String::as_str)
1850                    .collect();
1851                if !invalid.is_empty() {
1852                    return Err(OAuth2Error::InvalidScope(format!(
1853                        "unregistered scopes: {}",
1854                        invalid.join(", ")
1855                    )));
1856                }
1857                requested
1858            }
1859            None => client.scopes.clone(),
1860        };
1861
1862        // Stage 3 — tenant lookup (DB round-trip #2), for the `org_id` claim.
1863        let t_tenant_lookup = std::time::Instant::now();
1864        let tenant = self
1865            .tenant_repo
1866            .get_by_id(tenant_id)
1867            .await
1868            .map_err(|e| match e {
1869                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
1870                other => OAuth2Error::ServerError(other.to_string()),
1871            })?;
1872        let tenant_lookup_us = t_tenant_lookup.elapsed().as_micros() as u64;
1873
1874        // Stage 4 — token mint + EdDSA (Ed25519) signature. No refresh token is
1875        // issued for client_credentials, so there is no persist stage.
1876        let t_token_mint = std::time::Instant::now();
1877        let cnf = self.certificate_binding_for(&client, ctx)?;
1878        // Read before `cnf` is moved into the claims. RFC 9449 §5: a DPoP-bound
1879        // token is announced as such, and everything else stays `Bearer`.
1880        let token_type = crate::dpop::token_type_for(cnf.as_ref());
1881        // X1. Deliberately measured INSIDE `token_mint_us` rather than as its
1882        // own stage: this is the number the R2.4 bench cell compares against an
1883        // unhooked run, and splitting it out would let the hooked and unhooked
1884        // figures be read from different clocks.
1885        let ext = self
1886            .pre_issue_ext_claims(tenant_id, client_id, "oauth2_client", client_id, &scopes)
1887            .await?;
1888        let access_token = issue_client_credentials_token_enriched(
1889            client_id,
1890            tenant_id,
1891            tenant.organization_id,
1892            &scopes,
1893            &self.auth_config,
1894            cnf,
1895            ext,
1896        )
1897        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1898        let token_mint_us = t_token_mint.elapsed().as_micros() as u64;
1899
1900        let scope = if scopes.is_empty() {
1901            None
1902        } else {
1903            Some(scopes.join(" "))
1904        };
1905
1906        let handler_total_us = started.elapsed().as_micros() as u64;
1907        let span = tracing::Span::current();
1908        span.record("client_lookup_us", client_lookup_us);
1909        span.record("secret_verify_us", secret_verify_us);
1910        span.record("tenant_lookup_us", tenant_lookup_us);
1911        span.record("token_mint_us", token_mint_us);
1912        span.record("handler_total_us", handler_total_us);
1913        tracing::debug!(
1914            target: "axiam::perf",
1915            stage = "oauth2.client_credentials",
1916            client_lookup_us,
1917            secret_verify_us,
1918            tenant_lookup_us,
1919            token_mint_us,
1920            handler_total_us,
1921            access_token_len = access_token.len(),
1922            "client_credentials stage timings (I5)"
1923        );
1924
1925        Ok(TokenResponse {
1926            access_token,
1927            token_type,
1928            expires_in: self.auth_config.access_token_lifetime_secs,
1929            refresh_token: None,
1930            scope,
1931            id_token: None,
1932        })
1933    }

1935    /// Refresh token grant (RFC 6749 section 6).
1936    ///
1937    /// Single-use rotation: the old refresh token is revoked and a new one
1938    /// is issued alongside a fresh access token.
1939    async fn handle_refresh_token(
1940        &self,
1941        tenant_id: Uuid,
1942        req: TokenRequest,
1943        ctx: &TokenRequestContext,
1944    ) -> Result<TokenResponse, OAuth2Error> {
1945        let raw_token = req
1946            .refresh_token
1947            .as_deref()
1948            .ok_or_else(|| OAuth2Error::InvalidRequest("refresh_token is required".into()))?;
1949        let client_id = resolve_client_id(req.client_id.as_deref(), ctx)?;
1950
1951        // Authenticate client BEFORE looking up the refresh token to
1952        // avoid a token-validity oracle (different error for valid vs
1953        // invalid tokens when client auth fails).
1954        //
1955        // X5.1: widened from "a client_secret" to "a credential" so an mTLS
1956        // client is not turned away before its certificate is consulted; W8
1957        // widens it again for a secret carried in the `Authorization` header.
1958        // Still answered from the request alone, so the oracle this ordering
1959        // exists to close stays closed.
1960        let client_secret_val = req.client_secret.as_deref();
1961        if ctx.carries_no_client_credential(client_secret_val) {
1962            return Err(OAuth2Error::InvalidClient(
1963                "client authentication is required".into(),
1964            ));
1965        }
1966
1967        let client = self
1968            .client_repo
1969            .get_by_client_id(tenant_id, client_id)
1970            .await
1971            .map_err(|e| match e {
1972                // QUAL-03/D-11: only a genuinely-unknown client maps to
1973                // invalid_client. Any other error (e.g. a DB outage) must
1974                // surface as a distinct server error, never masquerade as
1975                // bad client credentials (error-oracle). The distinction is
1976                // internal only — the caller-visible description is
1977                // CLIENT_AUTH_FAILED either way, so it cannot be used to
1978                // probe which client ids exist (SEC-086).
1979                AxiamError::NotFound { .. } => {
1980                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
1981                }
1982                other => OAuth2Error::ServerError(other.to_string()),
1983            })?;
1984
1985        self.authenticate_client_credential(tenant_id, &client, client_secret_val, ctx)
1986            .await?;
1987        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
1988
1989        // Verify client is authorized for refresh_token grant
1990        if !client.grant_types.iter().any(|s| s == "refresh_token") {
1991            return Err(OAuth2Error::UnauthorizedClient(
1992                "client not authorized for refresh_token grant".into(),
1993            ));
1994        }
1995
1996        // Look up the refresh token by hash (after client auth)
1997        let token_hash = hash_refresh_token(raw_token);
1998        let stored = self
1999            .refresh_token_repo
2000            .get_by_token_hash(tenant_id, &token_hash)
2001            .await
2002            .map_err(|_| {
2003                OAuth2Error::InvalidGrant("refresh token is invalid, expired, or revoked".into())
2004            })?;
2005
2006        // Verify client ownership
2007        if stored.client_id != client_id {
2008            return Err(OAuth2Error::InvalidGrant(
2009                "refresh token was not issued to this client".into(),
2010            ));
2011        }
2012
2013        // Resolve org_id from tenant
2014        let tenant = self
2015            .tenant_repo
2016            .get_by_id(tenant_id)
2017            .await
2018            .map_err(|e| match e {
2019                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
2020                other => OAuth2Error::ServerError(other.to_string()),
2021            })?;
2022
2023        // Issue new access token.
2024        // OAuth2 refresh flow has no persistent session row — use random jti.
2025        //
2026        // X5.1: the binding is re-derived from the certificate presented on
2027        // THIS connection, never copied from the token being refreshed. A
2028        // refresh is a fresh proof of possession — carrying the old `cnf`
2029        // forward would let a client that has lost its certificate keep
2030        // renewing tokens bound to it, and would survive a certificate
2031        // rotation the operator performed precisely to revoke access.
2032        let cnf = self.certificate_binding_for(&client, ctx)?;
2033        // Read before `cnf` is moved into the claims. RFC 9449 §5: a DPoP-bound
2034        // token is announced as such, and everything else stays `Bearer`.
2035        let token_type = crate::dpop::token_type_for(cnf.as_ref());
2036        // X1 — a refresh mints a new access token, so it is an issuance and it
2037        // is hooked. Enrichment is re-derived here rather than copied from the
2038        // token being refreshed, for the same reason `cnf` is: a claim a
2039        // reactor added fifteen minutes ago is not evidence that it would add
2040        // it now, and carrying it forward would let a revoked enrichment
2041        // survive every rotation.
2042        let ext = self
2043            .pre_issue_ext_claims(
2044                tenant_id,
2045                &stored
2046                    .user_id
2047                    .map_or_else(|| client_id.to_string(), |u| u.to_string()),
2048                if stored.user_id.is_some() {
2049                    "user"
2050                } else {
2051                    "oauth2_client"
2052                },
2053                client_id,
2054                &stored.scopes,
2055            )
2056            .await?;
2057        let access_token = if let Some(user_id) = stored.user_id {
2058            // W7 — carried across the rotation, because the grant it names is
2059            // the same grant. A refreshed token that lost the claim would lose
2060            // access to consented claims at UserInfo fifteen minutes after the
2061            // consent was given, which the end user would experience as the
2062            // consent not having worked.
2063            issue_access_token_for_client(
2064                user_id,
2065                tenant_id,
2066                tenant.organization_id,
2067                &stored.scopes,
2068                &self.auth_config,
2069                uuid::Uuid::new_v4().to_string(),
2070                axiam_auth::token::AUD_USER,
2071                cnf,
2072                ext,
2073                Some(client_id),
2074                // Rotation carries the session forward. A refreshed token that
2075                // dropped `sid` would stop working at UserInfo the moment it
2076                // replaced the one that did, which the end user would
2077                // experience as the session silently ending mid-flow.
2078                stored.session_id,
2079            )
2080            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?
2081        } else {
2082            // Client-credentials-originated refresh (shouldn't normally
2083            // happen, but handle gracefully)
2084            issue_client_credentials_token_enriched(
2085                client_id,
2086                tenant_id,
2087                tenant.organization_id,
2088                &stored.scopes,
2089                &self.auth_config,
2090                cnf,
2091                ext,
2092            )
2093            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?
2094        };
2095
2096        // Refresh token rotation: create new token BEFORE revoking
2097        // the old one. If `create` fails the old token remains valid
2098        // and the client can retry; if `revoke` fails after `create`
2099        // we have a brief window with two valid tokens (less harmful
2100        // than zero tokens forcing a full re-auth).
2101        let new_raw_refresh = generate_refresh_token();
2102        let new_refresh_hash = hash_refresh_token(&new_raw_refresh);
2103        let refresh_expires =
2104            Utc::now() + chrono::Duration::seconds(self.refresh_token_lifetime_secs);
2105
2106        self.refresh_token_repo
2107            .create(CreateRefreshToken {
2108                tenant_id,
2109                token_hash: new_refresh_hash.clone(),
2110                client_id: client_id.to_string(),
2111                user_id: stored.user_id,
2112                scopes: stored.scopes.clone(),
2113                // Rotation preserves the session: the new token descends from
2114                // the same login, so it must carry the same `sid`.
2115                session_id: stored.session_id,
2116                expires_at: refresh_expires,
2117            })
2118            .await
2119            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
2120
2121        // Now retire the old refresh token.
2122        //
2123        // FAPI 2.0 Security Profile §5.3.2.1-9: an authorization server that
2124        // rotates refresh tokens shall keep accepting the previous one for a
2125        // short period after issuing its successor. So this SUPERSEDES rather
2126        // than revokes — the old token stays usable for
2127        // `REFRESH_ROTATION_GRACE_SECS` and then expires on the read path's
2128        // existing expiry check.
2129        //
2130        // What that buys is the only recovery a client has from a rotation
2131        // whose response it never received: under immediate revocation it holds
2132        // a token the server has destroyed, has not been given the replacement,
2133        // and can do nothing but start a whole new authorization. What it costs
2134        // is a window in which a leaked refresh token is still usable — which
2135        // is why the profile that requires it is also the profile that
2136        // sender-constrains every token, so replaying one inside the window
2137        // needs the client's private key as well.
2138        //
2139        // The failure handling below is unchanged, including its NotFound case:
2140        // `supersede` reports NotFound on exactly the same condition `revoke`
2141        // did — no live row matched — so a genuinely concurrent second use is
2142        // still caught and still answered "already consumed".
2143        if let Err(revoke_err) = self
2144            .refresh_token_repo
2145            .supersede(
2146                tenant_id,
2147                &token_hash,
2148                Utc::now() + chrono::Duration::seconds(REFRESH_ROTATION_GRACE_SECS),
2149            )
2150            .await
2151        {
2152            // Best-effort: delete the newly-created token so it
2153            // doesn't linger as an orphan.
2154            if let Err(cleanup_err) = self
2155                .refresh_token_repo
2156                .revoke(tenant_id, &new_refresh_hash)
2157                .await
2158            {
2159                tracing::warn!(
2160                    error = %cleanup_err,
2161                    "token: failed to revoke orphaned refresh token; token may linger"
2162                );
2163            }
2164
2165            return if matches!(revoke_err, AxiamError::NotFound { .. }) {
2166                Err(OAuth2Error::InvalidGrant(
2167                    "refresh token already consumed".into(),
2168                ))
2169            } else {
2170                Err(OAuth2Error::ServerError(
2171                    "failed to revoke old refresh token".into(),
2172                ))
2173            };
2174        }
2175
2176        // Re-issue an ID token when the original grant included `openid`.
2177        let id_token = if stored.scopes.iter().any(|s| s == "openid") {
2178            if let Some(uid) = stored.user_id {
2179                let user = self
2180                    .user_repo
2181                    .get_by_id(tenant_id, uid)
2182                    .await
2183                    .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
2184                // W4 — the same gate as the code-exchange path above. For a
2185                // client on the honour lane the evidence is resolved from the
2186                // session behind `stored.session_id`, which refresh rotation
2187                // preserves precisely so that OIDC Core §12.2 holds: the
2188                // `auth_time` of a refreshed ID token equals the original's,
2189                // and a freshly-stamped one would be a lie that grows younger
2190                // with every refresh.
2191                let evidence = if crate::fapi::emits_session_evidence(&client) {
2192                    self.session_evidence_for_refresh(tenant_id, stored.session_id)
2193                        .await
2194                } else {
2195                    IdTokenEvidence::NONE
2196                };
2197                Some(
2198                    issue_id_token(
2199                        uid,
2200                        client_id,
2201                        None,
2202                        Some(&user.username),
2203                        &stored.scopes,
2204                        &self.auth_config,
2205                        // The same session the original login created: an RP
2206                        // that only ever sees refreshed ID tokens must still
2207                        // be able to match a logout token to its session.
2208                        stored.session_id,
2209                        &evidence,
2210                    )
2211                    .map_err(|e| OAuth2Error::ServerError(e.to_string()))?,
2212                )
2213            } else {
2214                None
2215            }
2216        } else {
2217            None
2218        };
2219
2220        let scope = if stored.scopes.is_empty() {
2221            None
2222        } else {
2223            Some(stored.scopes.join(" "))
2224        };
2225
2226        Ok(TokenResponse {
2227            access_token,
2228            token_type,
2229            expires_in: self.auth_config.access_token_lifetime_secs,
2230            refresh_token: Some(new_raw_refresh),
2231            scope,
2232            id_token,
2233        })
2234    }

2236    /// Revoke a token per RFC 7009.
2237    ///
2238    /// Always returns `Ok(())` — invalid tokens are silently ignored.
2239    pub async fn revoke_token(
2240        &self,
2241        tenant_id: Uuid,
2242        req: RevokeRequest,
2243        ctx: &TokenRequestContext,
2244    ) -> Result<(), OAuth2Error> {
2245        // Authenticate the client making the revocation request, by whichever
2246        // credential its REGISTRATION names (SEC-093) — not by whichever one
2247        // the request happened to carry.
2248        let ctx = ctx.clone().with_assertion(
2249            req.client_assertion.as_deref(),
2250            req.client_assertion_type.as_deref(),
2251        );
2252        self.authenticate_client(
2253            tenant_id,
2254            &req.client_id,
2255            req.client_secret.as_deref(),
2256            &ctx,
2257        )
2258        .await?;
2259
2260        // Try revoking as a refresh token (hash-based lookup).
2261        // For access tokens (short-lived JWTs), revocation is a no-op —
2262        // they expire naturally within minutes.
2263        let token_hash = hash_refresh_token(&req.token);
2264
2265        // Look up the token first to verify client ownership.
2266        // Only revoke if found and belongs to the requesting client;
2267        // otherwise treat as unknown per RFC 7009.
2268        if let Ok(stored) = self
2269            .refresh_token_repo
2270            .get_by_token_hash(tenant_id, &token_hash)
2271            .await
2272            && stored.client_id == req.client_id
2273        {
2274            self.refresh_token_repo
2275                .revoke(tenant_id, &token_hash)
2276                .await
2277                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
2278        }
2279
2280        Ok(())
2281    }

2283    /// Introspect a token per RFC 7662.
2284    pub async fn introspect_token(
2285        &self,
2286        tenant_id: Uuid,
2287        req: IntrospectRequest,
2288        ctx: &TokenRequestContext,
2289    ) -> Result<IntrospectionResponse, OAuth2Error> {
2290        // Authenticate the client making the introspection request, by
2291        // whichever credential its REGISTRATION names (SEC-093).
2292        let ctx = ctx.clone().with_assertion(
2293            req.client_assertion.as_deref(),
2294            req.client_assertion_type.as_deref(),
2295        );
2296        self.authenticate_client(
2297            tenant_id,
2298            &req.client_id,
2299            req.client_secret.as_deref(),
2300            &ctx,
2301        )
2302        .await?;
2303
2304        // First try: decode as JWT access token
2305        if let Ok(validated) = validate_access_token(&req.token, &self.auth_config) {
2306            let claims = &validated.0;
2307
2308            // Verify the token belongs to this tenant to prevent
2309            // cross-tenant introspection / metadata leaks.
2310            let token_tenant = claims.tenant_id.parse::<Uuid>().unwrap_or(Uuid::nil());
2311            if token_tenant != tenant_id {
2312                return Ok(IntrospectionResponse {
2313                    active: false,
2314                    ..Default::default()
2315                });
2316            }
2317
2318            return Ok(IntrospectionResponse {
2319                active: true,
2320                scope: claims.scope.clone(),
2321                // Access tokens don't carry an explicit client_id
2322                // claim; omit it to avoid confusing sub (user ID)
2323                // with client_id.
2324                client_id: None,
2325                sub: Some(claims.sub.clone()),
2326                exp: Some(claims.exp),
2327                iat: Some(claims.iat),
2328                token_type: Some("Bearer".into()),
2329                permissions: claims.permissions.clone(),
2330                // RFC 8705 §3.3 — echoed from the token, like `permissions`.
2331                // A resource server that introspects rather than validating
2332                // locally learns from this field, and only from this field,
2333                // that the token in its hand is not a bearer token.
2334                cnf: claims.cnf.clone(),
2335            });
2336        }
2337
2338        // Second try: look up as refresh token
2339        let token_hash = hash_refresh_token(&req.token);
2340        if let Ok(stored) = self
2341            .refresh_token_repo
2342            .get_by_token_hash(tenant_id, &token_hash)
2343            .await
2344        {
2345            // Only introspect tokens belonging to the requesting
2346            // client — prevent cross-client information leaks.
2347            if stored.client_id != req.client_id {
2348                return Ok(IntrospectionResponse {
2349                    active: false,
2350                    ..Default::default()
2351                });
2352            }
2353
2354            let scope = if stored.scopes.is_empty() {
2355                None
2356            } else {
2357                Some(stored.scopes.join(" "))
2358            };
2359            return Ok(IntrospectionResponse {
2360                active: true,
2361                scope,
2362                client_id: Some(stored.client_id),
2363                sub: stored.user_id.map(|u| u.to_string()),
2364                exp: Some(stored.expires_at.timestamp()),
2365                iat: Some(stored.created_at.timestamp()),
2366                token_type: Some("refresh_token".into()),
2367                // A refresh token is never an RPT: the uma-ticket grant issues
2368                // no refresh token, precisely so that an RPT cannot be renewed
2369                // past the ticket that authorised it.
2370                permissions: None,
2371                // A refresh token is not sender-constrained in AXIAM: it is an
2372                // opaque server-stored value, single-use with rotation, and it
2373                // is presented at the token endpoint where the client
2374                // re-authenticates. Its constraint is client authentication,
2375                // not `cnf`, and the certificate binding of the ACCESS token it
2376                // mints is re-derived from that fresh authentication (see
2377                // `handle_refresh_token`).
2378                cnf: None,
2379            });
2380        }
2381
2382        // Token is not recognised — return inactive per RFC 7662
2383        Ok(IntrospectionResponse {
2384            active: false,
2385            ..Default::default()
2386        })
2387    }

2389    /// Verify client credentials and return the authenticated registration.
2390    ///
2391    /// Shared by every client-authenticating endpoint that is not one of the
2392    /// three grants dispatched inside [`Self::exchange`]: revoke, introspect,
2393    /// PAR, the token-exchange grant and the uma-ticket grant. The first two
2394    /// discard the returned row; the others need it, because the client's
2395    /// registered scopes bound what an exchange may grant and its registered
2396    /// URIs bound which audiences it may address.
2397    ///
2398    /// # SEC-093 — the registration decides here too
2399    ///
2400    /// This used to call [`Self::verify_client_secret`] directly, so those
2401    /// five endpoints authenticated by shared secret **whatever
2402    /// `token_endpoint_auth_method` the client registered**. Because
2403    /// `SurrealOAuth2ClientRepository::create` mints a secret for every client
2404    /// unconditionally, a FAPI client registered for `tls_client_auth` — whose
2405    /// registration validation explicitly *forbids* shared-secret
2406    /// authentication — nonetheless held a live password-equivalent credential
2407    /// that worked at PAR, revoke, introspect, token-exchange and uma-ticket.
2408    /// The operator's model was "this client authenticates with a
2409    /// certificate"; the reality was an OR over two credentials, one of which
2410    /// they were told to discard at creation time.
2411    ///
2412    /// It now delegates to [`Self::authenticate_client_credential`], the same
2413    /// method the authorization_code / client_credentials / refresh_token
2414    /// grants use — deliberately *reused* rather than reimplemented, because a
2415    /// second copy of this rule is how the two drifted apart in the first
2416    /// place. `client_secret` is `Option` because a client registered for a
2417    /// strong method has none to present.
2418    pub async fn authenticate_client(
2419        &self,
2420        tenant_id: Uuid,
2421        client_id: &str,
2422        client_secret: Option<&str>,
2423        ctx: &TokenRequestContext,
2424    ) -> Result<OAuth2Client, OAuth2Error> {
2425        let client = self
2426            .client_repo
2427            .get_by_client_id(tenant_id, client_id)
2428            .await
2429            .map_err(|e| match e {
2430                // QUAL-03/D-11: only a genuinely-unknown client maps to
2431                // invalid_client. Any other error (e.g. a DB outage) must
2432                // surface as a distinct server error, never masquerade as
2433                // bad client credentials (error-oracle). The distinction is
2434                // internal only — the caller-visible description is
2435                // CLIENT_AUTH_FAILED either way, so it cannot be used to
2436                // probe which client ids exist (SEC-086).
2437                AxiamError::NotFound { .. } => {
2438                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
2439                }
2440                other => OAuth2Error::ServerError(other.to_string()),
2441            })?;
2442
2443        // Defence in depth. The lookup above is tenant-scoped in SQL, so this
2444        // can only fire if that predicate is dropped or a repository
2445        // implementation ignores `tenant_id` — which is exactly the
2446        // regression worth catching, since otherwise the cross-tenant
2447        // property lives entirely in a WHERE clause two crates away.
2448        if client.tenant_id != tenant_id {
2449            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
2450        }
2451
2452        self.authenticate_client_credential(tenant_id, &client, client_secret, ctx)
2453            .await?;
2454
2455        Ok(client)
2456    }
2457}
