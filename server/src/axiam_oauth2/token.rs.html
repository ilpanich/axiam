32/// The single `error_description` every token-endpoint client-authentication
33/// failure returns, whatever actually went wrong (SEC-086).
34///
35/// The `invalid_client` error *code* was already uniform across "no such
36/// client" and "wrong secret". The *description* was not: an unknown client
37/// said `"client not found"`, which let an unauthenticated caller probe
38/// whether a given `client_id` — including an `sa_…` service-account id —
39/// exists. The masquerade the QUAL-03/D-11 comments describe was applied to
40/// the code and not to the message; this constant applies it to both.
41///
42/// The `NotFound`-versus-other-error distinction those comments protect is
43/// unaffected: it stays internal, mapping to `invalid_client` versus
44/// `server_error`, so a DB outage still never reads as bad credentials. Only
45/// the caller-visible wording is collapsed.
46///
47/// Deliberately **not** used at the authorization endpoint
48/// (`authorize.rs`): there `client_id` is public by construction, RFC 6749
49/// §4.1.2.1 requires informing the resource owner directly rather than
50/// redirecting, and no credential is presented — so "invalid client
51/// credentials" would be actively misleading there.
52const CLIENT_AUTH_FAILED: &str = "invalid client credentials";

58/// Token request parameters (from form body).
59#[derive(Debug, Deserialize, utoipa::ToSchema)]
60pub struct TokenRequest {
61    pub grant_type: String,
62    pub code: Option<String>,
63    pub redirect_uri: Option<String>,
64    pub client_id: Option<String>,
65    pub client_secret: Option<String>,
66    pub code_verifier: Option<String>,
67    pub refresh_token: Option<String>,
68    pub scope: Option<String>,
69    /// B2 — the secret a device polls with, for
70    /// `grant_type=urn:ietf:params:oauth:grant-type:device_code`
71    /// (RFC 8628 §3.4). Present only on that grant; the REST layer routes it
72    /// to [`crate::device_service::DeviceAuthorizationService`] rather than
73    /// to [`TokenService::exchange`], which never sees it.
74    pub device_code: Option<String>,
75
76    // --- B3 / RFC 8693 token exchange -----------------------------------
77    //
78    // Carried here rather than in a second form type because actix
79    // deserializes one body once, and the token endpoint cannot know which
80    // grant it is serving until it has read `grant_type` out of that body.
81    // `exchange_request()` below projects them into the shape the exchange
82    // service takes, so the grant's own module never sees these Options.
83    pub subject_token: Option<String>,
84    pub subject_token_type: Option<String>,
85    pub requested_token_type: Option<String>,
86    pub actor_token: Option<String>,
87    pub actor_token_type: Option<String>,
88    pub audience: Option<String>,
89    pub resource: Option<String>,
90
91    // --- X2 / UMA 2.0 ticket grant --------------------------------------
92    /// The permission ticket minted by `/uma2/perm`, for
93    /// `grant_type=urn:ietf:params:oauth:grant-type:uma-ticket`.
94    pub ticket: Option<String>,
95    /// The requesting party's identity, as an AXIAM access token.
96    ///
97    /// UMA 2.0 §3.3.1 makes this optional because a client may instead present
98    /// its own RPT for incremental authorization, which v1 does not implement.
99    /// With claims-gathering also deferred, this is the only channel through
100    /// which a requesting party can be named — so the grant requires it and
101    /// says so, rather than resolving to some default subject.
102    pub claim_token: Option<String>,
103    /// Declared format of `claim_token`. Only the AXIAM access-token URN is
104    /// accepted in v1; anything else is refused rather than guessed at.
105    pub claim_token_format: Option<String>,
106
107    // --- X5.1 / RFC 7521 §4.2 client assertion ---------------------------
108    /// The signed JWT a `private_key_jwt` client authenticates with.
109    ///
110    /// Carried here rather than in a second form type for the same reason the
111    /// RFC 8693 parameters are: actix deserializes one body once, and the
112    /// token endpoint cannot know which client-authentication method applies
113    /// until it has read `client_id` out of that body and loaded the
114    /// registration.
115    pub client_assertion: Option<String>,
116    /// Must be `urn:ietf:params:oauth:client-assertion-type:jwt-bearer`.
117    pub client_assertion_type: Option<String>,
118}

120impl TokenRequest {
121    /// Project the RFC 8693 parameters, or `None` when the two mandatory ones
122    /// are absent.
123    ///
124    /// Returning `None` rather than a half-populated request means the caller
125    /// answers `invalid_request` naming what is missing, instead of the
126    /// exchange service reporting a confusing downstream failure about a
127    /// token it was handed as an empty string.
128    pub fn exchange_request(&self) -> Option<crate::token_exchange::TokenExchangeRequest> {
129        let subject_token = self.subject_token.clone().filter(|t| !t.is_empty())?;
130        let subject_token_type = self.subject_token_type.clone().filter(|t| !t.is_empty())?;
131        Some(crate::token_exchange::TokenExchangeRequest {
132            subject_token,
133            subject_token_type,
134            requested_token_type: self.requested_token_type.clone().filter(|t| !t.is_empty()),
135            actor_token: self.actor_token.clone().filter(|t| !t.is_empty()),
136            actor_token_type: self.actor_token_type.clone().filter(|t| !t.is_empty()),
137            scope: self.scope.clone(),
138            audience: self.audience.clone().filter(|t| !t.is_empty()),
139            resource: self.resource.clone().filter(|t| !t.is_empty()),
140        })
141    }

207    ///
208    /// Keycloak's authorization services put the same array under the same key
209    /// in an introspection response, so a resource server migrating from
210    /// Keycloak reads an AXIAM RPT without a translation layer.
211    ///
212    /// This is **echoed from the token, not re-evaluated**. Introspection

215    /// that needs a live answer must ask the authorization endpoints — which is
216    /// why RPT lifetime is bounded rather than long.
217    #[serde(skip_serializing_if = "Option::is_none")]

219    /// RFC 8705 §3.3 `cnf` — present only when the introspected token is
220    /// certificate-bound (X5.1).
221    ///
222    /// RFC 8705 requires the authorization server to expose the confirmation
223    /// through introspection, because a resource server doing token
224    /// introspection rather than local JWT validation has no other way to
225    /// learn that the token it is holding is not a bearer token. Omitting it
226    /// would leave every introspecting resource server treating bound tokens
227    /// as unbound — the exact failure the binding exists to prevent.
228    #[serde(skip_serializing_if = "Option::is_none")]
229    pub cnf: Option<axiam_auth::token::CnfClaim>,

232/// What the token endpoint knows about the **connection** a request arrived
233/// on, as opposed to its body (X5.1).
234///
235/// Exists so `TokenService` can authenticate a client by certificate without
236/// depending on actix, rustls, or the REST layer's extractor types. The REST
237/// handler builds one from `HttpRequest::conn_data::<VerifiedClientCert>()`;
238/// a test builds one from a certificate it generated.
239///
240/// [`Default`] is "no certificate", which is what a plaintext or
241/// server-only-TLS connection legitimately has. Every path that *needs* a
242/// certificate checks for it explicitly; nothing infers authentication from
243/// this type's absence.
244#[derive(Debug, Clone, Default)]
245pub struct TokenRequestContext {
246    /// The client certificate rustls verified during the TLS handshake, if
247    /// this connection carried one.
248    ///
249    /// **Only** ever populated from a verified peer chain. See
250    /// `crate::mtls`'s module docs for why the `X-Client-Certificate` proxy
251    /// header is deliberately not a source here.
252    pub client_certificate: Option<PresentedCertificate>,
253    /// A DPoP proof that has **already been verified** for this request, and
254    /// whose `jti` has already been recorded (X5.1, RFC 9449).
255    ///
256    /// The same discipline as [`Self::client_certificate`] and for the same
257    /// reason: this field carries a *conclusion*, never an input. The REST
258    /// layer verifies the `DPoP` header against the actual method and URI —
259    /// which only it knows — and hands the result here. A caller that
260    /// populated this from an unverified header would have turned DPoP into a
261    /// self-signed permission slip, which is why the type is
262    /// `VerifiedDpopProof` and not `String`.
263    pub dpop_proof: Option<crate::dpop::VerifiedDpopProof>,
264    /// The `client_assertion` form parameter, verbatim (RFC 7521 §4.2).
265    ///
266    /// Unlike the two fields above this is a request *body* parameter, not a
267    /// property of the connection, and it is carried here rather than passed
268    /// down from the form because `authenticate_client_credential` is reached
269    /// from four grants that share no request type. It is deliberately the raw
270    /// string: nothing has verified it at this point, and the type says so.
271    pub client_assertion: Option<String>,
272    /// The `client_assertion_type` form parameter.
273    pub client_assertion_type: Option<String>,
274}

410    /// Run `token.pre_issue` and return the `ext` claims to mint with.
411    ///
412    /// Every grant that mints an access token goes through here, which is the
413    /// point: three call sites (authorization code, refresh, client
414    /// credentials) share one hook, so a fourth grant added later cannot
415    /// quietly skip it by forgetting to copy the block.
416    ///
417    /// A veto surfaces as `invalid_grant` — the plan's pinned mapping, and the
418    /// right one: RFC 6749 §5.2 `invalid_grant` is "the grant is not valid for
419    /// this request", which is exactly what a reactor is saying. The reactor's
420    /// reason is **not** echoed to the client (it is third-party text on an
421    /// endpoint reachable with a client credential); the gate has already
422    /// written it to the audit trail with the reactor's id.
423    async fn pre_issue_ext_claims(
424        &self,
425        tenant_id: Uuid,
426        subject: &str,
427        subject_kind: &'static str,
428        client_id: &str,
429        scopes: &[String],
430    ) -> Result<Option<std::collections::BTreeMap<String, String>>, OAuth2Error> {
431        match self
432            .reactor_gate
433            .intercept(
434                tenant_id,
435                reactor_events::TOKEN_PRE_ISSUE,
436                serde_json::json!({
437                    // No token, no credential, no signing key (§22.3) — the
438                    // reactor is told what is about to be minted, not handed
439                    // the means to mint it.
440                    "sub": subject,
441                    "sub_kind": subject_kind,
442                    "client_id": client_id,
443                    "tenant_id": tenant_id,
444                    "scopes": scopes,
445                }),
446            )
447            .await
448        {
449            ReactorOutcome::Allow => Ok(None),
450            ReactorOutcome::Mutate { patch } => {
451                Ok(axiam_auth::token::ext_claims_from_patch(&patch))
452            }
453            ReactorOutcome::Deny { reason } => {
454                tracing::info!(
455                    target: "axiam::reactor",
456                    tenant_id = %tenant_id,
457                    client_id,
458                    "token issuance vetoed by a reactor on token.pre_issue"
459                );
460                debug_assert!(!reason.is_empty());
461                Err(OAuth2Error::InvalidGrant(
462                    "token issuance was refused by policy".into(),
463                ))
464            }
465            // `require_mfa` is a `login.post_auth` answer. The reply validator
466            // refuses it on any other event, so this arm is unreachable through
467            // the wire path — and if it were reachable, the token endpoint has
468            // no step-up to perform, so proceeding "as if allowed" would be
469            // silently discarding a security demand.
470            ReactorOutcome::RequireMfa => {
471                tracing::error!(
472                    target: "axiam::reactor",
473                    tenant_id = %tenant_id,
474                    "a reactor demanded step-up on token.pre_issue, which has no \
475                     step-up to perform — refusing the grant"
476                );
477                Err(OAuth2Error::InvalidGrant(
478                    "token issuance was refused by policy".into(),
479                ))
480            }
481        }
482    }

500    /// Verify a presented client secret against the stored hash — the single
501    /// point at which every grant authenticates a confidential client.
502    ///
503    /// The comparison is constant-time and keyed (HMAC-SHA256 under the server
504    /// pepper, `axiam_auth::client_secret`). A pre-OBS-1 row still stored as a
505    /// bare SHA-256 digest verifies against the legacy scheme and is then
506    /// **transparently upgraded**: the replacement hash is written back with a
507    /// compare-and-swap so a concurrent rotation is never clobbered.
508    ///
509    /// Two properties this function exists to guarantee:
510    ///
511    /// - the upgrade is driven by [`ClientSecretVerdict::MatchNeedsUpgrade`],
512    ///   which the hasher only ever returns on a **successful** verification —
513    ///   a wrong secret can never cause a write; and
514    /// - a failed *write* never becomes a failed *authentication*. The client
515    ///   proved possession of the secret; losing the migration is a warning,
516    ///   and the next request simply retries it.
517    ///
518    /// Cost on the hot path: one HMAC-SHA256 over the secret into a stack
519    /// buffer, plus an `OnceLock` acquire load. No heap allocation and no lock
520    /// — strictly less than the `String` the previous `hash_client_secret`
521    /// allocated per request. The DB write happens at most once per legacy
522    /// client, on its first authentication after the upgrade.
523    async fn verify_client_secret(
524        &self,
525        tenant_id: Uuid,
526        client: &OAuth2Client,
527        presented: &str,
528    ) -> Result<(), OAuth2Error> {
529        let hasher =
530            client_secret::global().map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
531
532        match hasher.verify(presented, &client.client_secret_hash) {
533            ClientSecretVerdict::Match => Ok(()),
534            ClientSecretVerdict::MatchNeedsUpgrade { upgraded_hash } => {
535                match self
536                    .client_repo
537                    .upgrade_client_secret_hash(
538                        tenant_id,
539                        &client.client_id,
540                        &client.client_secret_hash,
541                        &upgraded_hash,
542                    )
543                    .await
544                {
545                    Ok(true) => tracing::info!(
546                        client_id = %client.client_id,
547                        "upgraded legacy client-secret hash to the peppered scheme (OBS-1)"
548                    ),
549                    Ok(false) => tracing::debug!(
550                        client_id = %client.client_id,
551                        "legacy client-secret hash upgrade skipped — a concurrent write won \
552                         the compare-and-swap"
553                    ),
554                    Err(e) => tracing::warn!(
555                        client_id = %client.client_id,
556                        error = %e,
557                        "failed to persist the legacy client-secret hash upgrade (OBS-1); \
558                         authentication succeeded, the upgrade will be retried"
559                    ),
560                }
561                Ok(())
562            }
563            ClientSecretVerdict::Mismatch => {
564                Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()))
565            }
566        }
567    }

569    /// Authenticate a registered client by whichever credential its
570    /// registration says it uses (X5.1).
571    ///
572    /// The **registration** decides, not the request. A client registered for
573    /// `tls_client_auth` is authenticated by its certificate even if it also
574    /// posted a `client_secret`, and a client registered for
575    /// `client_secret_post` is authenticated by its secret even over an mTLS
576    /// connection. Letting the request choose would turn the two methods into
577    /// an OR: an attacker holding either credential could authenticate, which
578    /// is strictly weaker than holding the one the operator registered.
579    ///
580    /// This preserves every ordering property SEC-086 established for the
581    /// secret path — see [`Self::verify_client_secret`] and the callers, where
582    /// the *presence* of a credential is still checked before the client
583    /// lookup so that "no such client" and "client lacks this grant" stay
584    /// indistinguishable.
585    async fn authenticate_client_credential(
586        &self,
587        tenant_id: Uuid,
588        client: &OAuth2Client,
589        presented_secret: Option<&str>,
590        ctx: &TokenRequestContext,
591    ) -> Result<(), OAuth2Error> {
592        if client.token_endpoint_auth_method.is_mtls() {
593            return crate::mtls::authenticate_mtls_client(client, ctx.client_certificate.as_ref());
594        }
595
596        if client.token_endpoint_auth_method.is_private_key_jwt() {
597            return self
598                .authenticate_client_assertion(tenant_id, client, ctx)
599                .await;
600        }
601
602        let secret = presented_secret
603            .ok_or_else(|| OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()))?;
604        self.verify_client_secret(tenant_id, client, secret).await
605    }

607    /// `private_key_jwt` client authentication (X5.1, RFC 7523 §2.2).
608    ///
609    /// The assertion comes from the request body, which is why it is read out
610    /// of [`TokenRequestContext`]'s sibling — the form — rather than the
611    /// context: unlike a certificate or a DPoP proof, it is not a property of
612    /// the connection. Everything else about the shape matches the mTLS branch
613    /// exactly: the **registration** decides the method, the credential is
614    /// checked against what the client registered, and every failure answers
615    /// the same `invalid_client`.
616    async fn authenticate_client_assertion(
617        &self,
618        tenant_id: Uuid,
619        client: &OAuth2Client,
620        ctx: &TokenRequestContext,
621    ) -> Result<(), OAuth2Error> {
622        let Some(verifier) = self.assertion_verifier.as_ref() else {
623            tracing::error!(
624                client_id = %client.client_id,
625                "a client is registered for private_key_jwt but this deployment has no assertion \
626                 verifier configured; refusing rather than falling back to another credential"
627            );
628            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
629        };
630
631        let Some(assertion) = ctx.client_assertion.as_deref() else {
632            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
633        };
634        crate::private_key_jwt::check_assertion_type(ctx.client_assertion_type.as_deref())
635            .map_err(OAuth2Error::from)?;
636
637        verifier.verify(tenant_id, client, assertion).await
638    }

640    /// Decide the `cnf` confirmation claim for a token about to be issued
641    /// (X5.1, RFC 8705 §3).
642    ///
643    /// Returns `Ok(None)` for every client that has not asked for binding —
644    /// which is every client that existed before X5.1 — so this is a branch
645    /// predictor's dream and costs nothing on the ordinary path.
646    ///
647    /// When a client *has* asked for binding and no certificate is present,
648    /// this **refuses** rather than issuing an unbound token. Silently
649    /// downgrading would be the worst possible behaviour: the operator
650    /// configured binding, the resource servers were told to expect it, and
651    /// the one request that arrives without a certificate would be the one
652    /// that gets a bearer token.
653    /// SEC-096: `pub` because the token-exchange and uma-ticket grants are
654    /// dispatched from `axiam-api-rest` rather than through
655    /// [`Self::exchange`], and they must reach the same decision. A second
656    /// copy of this rule in the handler is how one of the two ends up
657    /// silently issuing an unbound token.
658    pub fn certificate_binding_for(
659        &self,
660        client: &OAuth2Client,
661        ctx: &TokenRequestContext,
662    ) -> Result<Option<axiam_auth::token::CnfClaim>, OAuth2Error> {
663        let mut cnf: Option<axiam_auth::token::CnfClaim> = None;
664
665        if crate::fapi::wants_certificate_binding(client) {
666            let Some(thumbprint) = ctx.certificate_thumbprint() else {
667                tracing::warn!(
668                    client_id = %client.client_id,
669                    "client requires certificate-bound access tokens but presented no client \
670                     certificate; refusing rather than issuing an unbound token"
671                );
672                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
673            };
674            cnf = Some(axiam_auth::token::CnfClaim::from_certificate_thumbprint(
675                thumbprint,
676            ));
677        }
678
679        if crate::fapi::wants_dpop_binding(client) {
680            let Some(jkt) = ctx.dpop_thumbprint() else {
681                tracing::warn!(
682                    client_id = %client.client_id,
683                    "client requires DPoP-bound access tokens but presented no verified DPoP \
684                     proof; refusing rather than issuing an unbound token"
685                );
686                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
687            };
688            // A client that asked for both gets both, in one `cnf`. The
689            // resource server then has to satisfy both — see
690            // `axiam_auth::token::verify_token_binding`, where "both present"
691            // is a conjunction.
692            cnf = Some(match cnf {
693                Some(existing) => existing.with_dpop_thumbprint(jkt),
694                None => axiam_auth::token::CnfClaim::from_dpop_thumbprint(jkt),
695            });
696        }
697
698        Ok(cnf)
699    }

701    /// Client-credentials for a **service account** (`sa_…` client id).
702    ///
703    /// Service accounts are AXIAM's machine principal, and until now the only
704    /// way one could authenticate in a running server was mTLS
705    /// (`POST /api/v1/auth/device`). `create`/`rotate_secret` handed the
706    /// operator a `client_secret` that **no flow accepted** — this is the path
707    /// that makes it usable.
708    ///
709    /// Three deliberate differences from the `oauth2_client` branch:
710    ///
711    /// 1. **The secret is verified BEFORE the status check.** Rejecting a
712    ///    disabled account without verifying would let an unauthenticated
713    ///    caller distinguish "exists but disabled" from "does not exist" by
714    ///    timing. Verifying first means status is only ever consulted for a
715    ///    caller who already proved possession of the secret — and even then
716    ///    the error returned is the same generic `invalid_client`.
717    /// 2. **No scope may be requested.** A service account registers no scopes,
718    ///    and the `oauth2_client` branch's rule is that a requested scope must
719    ///    be a subset of the registered ones. With an empty registered set the
720    ///    only subset is the empty one, so any requested scope is
721    ///    `invalid_scope`. Authorization for a service account comes from the
722    ///    roles assigned to its subject, exactly as on the mTLS path.
723    /// 3. **The token's `sub` is the service-account id**, not the client id,
724    ///    so the RBAC engine resolves the same principal whether the account
725    ///    authenticated by certificate or by secret.
726    async fn handle_client_credentials_service_account(
727        &self,
728        tenant_id: Uuid,
729        client_id: &str,
730        client_secret: &str,
731        requested_scope: Option<&str>,
732        started: std::time::Instant,
733    ) -> Result<TokenResponse, OAuth2Error> {
734        let t_client_lookup = std::time::Instant::now();
735        let sa = self
736            .service_account_repo
737            .get_by_client_id(tenant_id, client_id)
738            .await
739            .map_err(|e| match e {
740                // Same taxonomy as the oauth2_client branch: only a genuinely
741                // unknown client is `invalid_client`; a DB outage must never
742                // masquerade as bad credentials (error-oracle, QUAL-03/D-11).
743                // The description is CLIENT_AUTH_FAILED here too, so an
744                // unauthenticated caller cannot learn whether an `sa_…`
745                // client id exists (SEC-086).
746                AxiamError::NotFound { .. } => {
747                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
748                }
749                other => OAuth2Error::ServerError(other.to_string()),
750            })?;
751        let client_lookup_us = t_client_lookup.elapsed().as_micros() as u64;
752
753        // Defence in depth (§17.2 residual 6). The lookup above is already
754        // tenant-scoped in SQL, so this can only fire if that predicate is
755        // ever dropped or a repository implementation ignores `tenant_id` —
756        // exactly the regression the assertion exists to catch. Without it the
757        // cross-tenant property of this grant lives entirely in a `WHERE`
758        // clause two crates away, and no test in this crate can prove it
759        // because the mocks do not filter by tenant.
760        //
761        // Deliberately checked *before* the secret is verified: a row from the
762        // wrong tenant must not have its secret compared at all, and the
763        // presented client id is the caller's own input, so refusing early
764        // leaks nothing it did not already know.
765        if sa.tenant_id != tenant_id {
766            tracing::error!(
767                requested_tenant = %tenant_id,
768                account_tenant = %sa.tenant_id,
769                "service-account lookup returned a row from another tenant — refusing; \
770                 this indicates a dropped tenant predicate in the repository layer"
771            );
772            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
773        }
774
775        let t_secret_verify = std::time::Instant::now();
776        let secret_result = self
777            .verify_service_account_secret(tenant_id, &sa, client_secret)
778            .await;
779        let secret_verify_us = t_secret_verify.elapsed().as_micros() as u64;
780        secret_result?;
781
782        // Only now — see point 1 in the doc comment.
783        if sa.status != UserStatus::Active {
784            tracing::info!(
785                client_id = %sa.client_id,
786                status = ?sa.status,
787                "service-account client-credentials refused: account is not active"
788            );
789            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
790        }
791
792        if let Some(scope) = requested_scope.filter(|s| !s.trim().is_empty()) {
793            return Err(OAuth2Error::InvalidScope(format!(
794                "unregistered scopes: {} — a service account registers no scopes; \
795                 its authorization comes from the roles assigned to it",
796                scope.split_whitespace().collect::<Vec<_>>().join(", ")
797            )));
798        }
799
800        let t_tenant_lookup = std::time::Instant::now();
801        let tenant = self
802            .tenant_repo
803            .get_by_id(tenant_id)
804            .await
805            .map_err(|e| match e {
806                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
807                other => OAuth2Error::ServerError(other.to_string()),
808            })?;
809        let tenant_lookup_us = t_tenant_lookup.elapsed().as_micros() as u64;
810
811        let t_token_mint = std::time::Instant::now();
812        // X1 — the fourth issuance path, hooked like the other three.
813        let ext = self
814            .pre_issue_ext_claims(
815                tenant_id,
816                &sa.id.to_string(),
817                "service_account",
818                &sa.client_id,
819                &[],
820            )
821            .await?;
822        let access_token = issue_service_account_client_credentials_token_enriched(
823            sa.id,
824            tenant_id,
825            tenant.organization_id,
826            &[],
827            &self.auth_config,
828            ext,
829        )
830        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
831        let token_mint_us = t_token_mint.elapsed().as_micros() as u64;
832
833        let span = tracing::Span::current();
834        span.record("client_lookup_us", client_lookup_us);
835        span.record("secret_verify_us", secret_verify_us);
836        span.record("tenant_lookup_us", tenant_lookup_us);
837        span.record("token_mint_us", token_mint_us);
838        span.record("handler_total_us", started.elapsed().as_micros() as u64);
839
840        Ok(TokenResponse {
841            access_token,
842            token_type: "Bearer".to_string(),
843            expires_in: self.auth_config.access_token_lifetime_secs,
844            refresh_token: None,
845            scope: None,
846            id_token: None,
847        })
848    }

850    /// [`Self::verify_client_secret`] for a service account.
851    ///
852    /// Same contract, against the service-account table: the upgrade is driven
853    /// only by `MatchNeedsUpgrade` (so a wrong secret can never cause a write),
854    /// and a failed write is a warning rather than a failed authentication.
855    ///
856    /// This is also what makes the §15.2 Obs 4 lazy migration *reachable*: with
857    /// no verification path, `upgrade_client_secret_hash` could never fire for a
858    /// service account, so legacy rows could not drain. They can now.
859    async fn verify_service_account_secret(
860        &self,
861        tenant_id: Uuid,
862        sa: &ServiceAccount,
863        presented: &str,
864    ) -> Result<(), OAuth2Error> {
865        let hasher =
866            client_secret::global().map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
867
868        match hasher.verify(presented, &sa.client_secret_hash) {
869            ClientSecretVerdict::Match => Ok(()),
870            ClientSecretVerdict::MatchNeedsUpgrade { upgraded_hash } => {
871                match self
872                    .service_account_repo
873                    .upgrade_client_secret_hash(
874                        tenant_id,
875                        &sa.client_id,
876                        &sa.client_secret_hash,
877                        &upgraded_hash,
878                    )
879                    .await
880                {
881                    Ok(true) => tracing::info!(
882                        client_id = %sa.client_id,
883                        "upgraded legacy service-account secret hash to the peppered scheme"
884                    ),
885                    Ok(false) => tracing::debug!(
886                        client_id = %sa.client_id,
887                        "legacy service-account secret-hash upgrade skipped — a concurrent \
888                         write won the compare-and-swap"
889                    ),
890                    Err(e) => tracing::warn!(
891                        client_id = %sa.client_id,
892                        error = %e,
893                        "failed to persist the legacy service-account secret-hash upgrade; \
894                         authentication succeeded, the upgrade will be retried"
895                    ),
896                }
897                Ok(())
898            }
899            ClientSecretVerdict::Mismatch => {
900                Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()))
901            }
902        }
903    }

907    /// `ctx` carries what the transport knows (X5.1) — today, the client
908    /// certificate rustls verified during the handshake. Pass
909    /// [`TokenRequestContext::default()`] on a connection that has none; every
910    /// path that requires a certificate checks for it rather than assuming.

912        &self,

920            "refresh_token" => self.handle_refresh_token(tenant_id, req, ctx).await,
921            _ => Err(OAuth2Error::UnsupportedGrantType),
922        }
923    }

925    /// Exchange an authorization code for tokens (RFC 6749 section 4.1.3).
926    async fn handle_authorization_code(
927        &self,
928        tenant_id: Uuid,
929        req: TokenRequest,
930        ctx: &TokenRequestContext,
931    ) -> Result<TokenResponse, OAuth2Error> {
932        let code = req
933            .code
934            .as_deref()
935            .ok_or_else(|| OAuth2Error::InvalidRequest("code is required".into()))?;
936        let redirect_uri = req
937            .redirect_uri
938            .as_deref()
939            .ok_or_else(|| OAuth2Error::InvalidRequest("redirect_uri is required".into()))?;
940        let client_id = req
941            .client_id
942            .as_deref()
943            .ok_or_else(|| OAuth2Error::InvalidRequest("client_id is required".into()))?;
944
945        // Require client_secret — all clients are confidential (no
946        // public-client distinction exists yet).
947        //
948        // SEC-086 (second pass): this check MUST precede the client lookup.
949        // When it ran after, a caller posting no `client_secret` at all got
950        // three distinguishable answers — `invalid_client`/generic for an
951        // unknown client, `unauthorized_client` for a client lacking the
952        // grant, and `invalid_client`/"client_secret is required" for a
953        // client that had it — so client existence stayed decidable on this
954        // grant even after the description was unified everywhere else.
955        // `client_credentials` and `refresh_token` already ordered it this
956        // way; only this grant did not.
957        //
958        // X5.1 widens "presented a client_secret" to "presented *a*
959        // credential", because an mTLS client sends no secret at all. The
960        // widened check is still decidable **before** the lookup — it asks
961        // only what this request carries, never what this client registered —
962        // so SEC-086's property is preserved exactly: a caller with no
963        // credential gets one answer regardless of whether the client id
964        // exists.
965        let client_secret = req.client_secret.as_deref();
966        if client_secret.is_none() && ctx.client_certificate.is_none() {
967            return Err(OAuth2Error::InvalidClient(
968                "client authentication is required".into(),
969            ));
970        }
971
972        // Authenticate client
973        let client = self
974            .client_repo
975            .get_by_client_id(tenant_id, client_id)
976            .await
977            .map_err(|e| match e {
978                // QUAL-03/D-11: only a genuinely-unknown client maps to
979                // invalid_client. Any other error (e.g. a DB outage) must
980                // surface as a distinct server error, never masquerade as
981                // bad client credentials (error-oracle). The distinction is
982                // internal only — the caller-visible description is
983                // CLIENT_AUTH_FAILED either way, so it cannot be used to
984                // probe which client ids exist (SEC-086).
985                AxiamError::NotFound { .. } => {
986                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
987                }
988                other => OAuth2Error::ServerError(other.to_string()),
989            })?;
990
991        // Secret verification must also precede the grant-type check, and for
992        // the same reason: `unauthorized_client` is only reachable by a caller
993        // who has *already proven possession of the secret*. Ordered the other
994        // way, supplying any dummy secret still separated "client exists but
995        // lacks this grant" from "no such client" — a second oracle on the
996        // same grant, which moving the presence check alone does not close.
997        // Both safe grants verify first; this now matches them exactly.
998        //
999        // X5.1: the profile gate runs *after* authentication, for the same
1000        // reason. `enforce_token_request` reports a fact about the client's
1001        // registration (its profile, its auth method), and reporting that to a
1002        // caller who has not proven possession of a credential would be a new
1003        // oracle on a path that just spent three comment blocks closing one.
1004        self.authenticate_client_credential(tenant_id, &client, client_secret, ctx)
1005            .await?;
1006        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
1007
1008        // Verify client is authorized for authorization_code grant
1009        if !client.grant_types.iter().any(|s| s == "authorization_code") {
1010            return Err(OAuth2Error::UnauthorizedClient(
1011                "client not authorized for authorization_code grant".into(),
1012            ));
1013        }
1014
1015        // Look up the authorization code without consuming it so we
1016        // can verify PKCE *before* marking it as used.  This prevents
1017        // an attacker from burning a valid code by intentionally
1018        // failing PKCE verification.
1019        let code_hash = crate::authorize::hash_code(code);
1020        let auth_code = self
1021            .code_repo
1022            .get_by_hash(tenant_id, &code_hash, client_id, redirect_uri)
1023            .await
1024            .map_err(|_| {
1025                OAuth2Error::InvalidGrant(
1026                    "authorization code is invalid, expired, or already used".into(),
1027                )
1028            })?;
1029
1030        // Verify PKCE before consuming the code
1031        if let Some(ref challenge) = auth_code.code_challenge {
1032            let verifier = req.code_verifier.as_deref().ok_or_else(|| {
1033                OAuth2Error::InvalidGrant("code_verifier required for PKCE".into())
1034            })?;
1035            if !pkce::verify_pkce(verifier, challenge) {
1036                return Err(OAuth2Error::InvalidGrant("PKCE verification failed".into()));
1037            }
1038        }
1039
1040        // Now atomically consume (mark as used) the code.
1041        self.code_repo
1042            .consume(tenant_id, &code_hash, client_id, redirect_uri)
1043            .await
1044            .map_err(|_| {
1045                OAuth2Error::InvalidGrant(
1046                    "authorization code is invalid, expired, or already used".into(),
1047                )
1048            })?;
1049
1050        // Resolve org_id from tenant
1051        let tenant = self
1052            .tenant_repo
1053            .get_by_id(tenant_id)
1054            .await
1055            .map_err(|e| match e {
1056                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
1057                other => OAuth2Error::ServerError(other.to_string()),
1058            })?;
1059
1060        // Issue access token (include scopes from the authorization code).
1061        // OAuth2 auth-code flow has no persistent session row — use random jti.
1062        //
1063        // X5.1: `cnf` is `None` for every client that has not asked for
1064        // certificate binding, which makes this byte-identical to what the
1065        // pre-X5 call produced.
1066        let cnf = self.certificate_binding_for(&client, ctx)?;
1067        // Read before `cnf` is moved into the claims. RFC 9449 §5: a DPoP-bound
1068        // token is announced as such, and everything else stays `Bearer`.
1069        let token_type = crate::dpop::token_type_for(cnf.as_ref());
1070        // X1 — the code has already been consumed at this point, so a veto
1071        // costs the caller the code. That is the correct order: the code is
1072        // single-use and PKCE-verified, and leaving it spendable while an
1073        // external process decides would hand an attacker a retry window.
1074        let ext = self
1075            .pre_issue_ext_claims(
1076                tenant_id,
1077                &auth_code.user_id.to_string(),
1078                "user",
1079                client_id,
1080                &auth_code.scopes,
1081            )
1082            .await?;
1083        let access_token = issue_access_token_enriched(
1084            auth_code.user_id,
1085            tenant_id,
1086            tenant.organization_id,
1087            &auth_code.scopes,
1088            &self.auth_config,
1089            uuid::Uuid::new_v4().to_string(),
1090            axiam_auth::token::AUD_USER,
1091            cnf,
1092            ext,
1093        )
1094        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1095
1096        // Only issue a refresh token when the client is authorized
1097        // for the refresh_token grant type.
1098        let refresh_token = if client.grant_types.iter().any(|s| s == "refresh_token") {
1099            let raw_refresh = generate_refresh_token();
1100            let refresh_hash = hash_refresh_token(&raw_refresh);
1101            let refresh_expires =
1102                Utc::now() + chrono::Duration::seconds(self.refresh_token_lifetime_secs);
1103
1104            self.refresh_token_repo
1105                .create(CreateRefreshToken {
1106                    tenant_id,
1107                    token_hash: refresh_hash,
1108                    client_id: client_id.to_string(),
1109                    user_id: Some(auth_code.user_id),
1110                    scopes: auth_code.scopes.clone(),
1111                    session_id: auth_code.session_id,
1112                    expires_at: refresh_expires,
1113                })
1114                .await
1115                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1116            Some(raw_refresh)
1117        } else {
1118            None
1119        };
1120
1121        // Issue an ID token when the `openid` scope was requested.
1122        let id_token = if auth_code.scopes.iter().any(|s| s == "openid") {
1123            let user = self
1124                .user_repo
1125                .get_by_id(tenant_id, auth_code.user_id)
1126                .await
1127                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1128            Some(
1129                issue_id_token(
1130                    auth_code.user_id,
1131                    tenant_id,
1132                    tenant.organization_id,
1133                    client_id,
1134                    auth_code.nonce.as_deref(),
1135                    Some(&user.email),
1136                    Some(&user.username),
1137                    &auth_code.scopes,
1138                    &self.auth_config,
1139                    auth_code.session_id,
1140                )
1141                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?,
1142            )
1143        } else {
1144            None
1145        };
1146
1147        let scope = if auth_code.scopes.is_empty() {
1148            None
1149        } else {
1150            Some(auth_code.scopes.join(" "))
1151        };
1152
1153        Ok(TokenResponse {
1154            access_token,
1155            token_type,
1156            expires_in: self.auth_config.access_token_lifetime_secs,
1157            refresh_token,
1158            scope,
1159            id_token,
1160        })
1161    }

1163    /// Client credentials grant (RFC 6749 section 4.4).
1164    ///
1165    /// Machine-to-machine flow — no user context, no refresh token.
1166    ///
1167    /// # Per-stage timing (I5)
1168    ///
1169    /// This handler is the subject of the run-4 TLS plateau investigation, so it
1170    /// carries a stage breakdown on its span: `client_lookup_us`,
1171    /// `secret_verify_us`, `tenant_lookup_us`, `token_mint_us` and
1172    /// `handler_total_us`, recorded on the `oauth2.client_credentials` span and
1173    /// re-emitted as one DEBUG event on `target: "axiam::perf"`.
1174    ///
1175    /// **Cost, and why it is left permanently on:** five `Instant::now()` calls
1176    /// (~20 ns each on a `clock_gettime` vDSO read) and five
1177    /// `Span::record` calls that are no-ops when no subscriber is interested —
1178    /// together well under 0.1 % of the ~370 µs this handler takes even in the
1179    /// *fast* (plaintext, 2 727 req/s) configuration. No flag gates the
1180    /// measurement; only the *reporting* is gated, by the ordinary tracing
1181    /// level (`RUST_LOG=axiam_oauth2=debug`, or any subscriber that samples the
1182    /// span's fields).
1183    ///
1184    /// There is deliberately **no "token persist" stage**: the
1185    /// `client_credentials` grant issues no refresh token and writes nothing —
1186    /// the only DB work is the two reads below. That, plus the fact that the
1187    /// client secret is verified with a single keyed HMAC-SHA256 rather than
1188    /// Argon2id (see `axiam_auth::client_secret`), is why this endpoint
1189    /// measures 2 727 req/s where password login measures 69. The OBS-1
1190    /// remediation deliberately kept this MAC-bound rather than KDF-bound:
1191    /// client secrets are 256-bit CSPRNG values *and* the digest is now keyed,
1192    /// so the offline-guessing threat a KDF defends against does not apply.
1193    #[tracing::instrument(
1194        name = "oauth2.client_credentials",
1195        skip(self, req),
1196        fields(
1197            client_lookup_us = tracing::field::Empty,
1198            secret_verify_us = tracing::field::Empty,
1199            tenant_lookup_us = tracing::field::Empty,
1200            token_mint_us = tracing::field::Empty,
1201            handler_total_us = tracing::field::Empty,
1202        )
1203    )]
1204    async fn handle_client_credentials(
1205        &self,
1206        tenant_id: Uuid,
1207        req: TokenRequest,
1208        ctx: &TokenRequestContext,
1209    ) -> Result<TokenResponse, OAuth2Error> {
1210        let started = std::time::Instant::now();
1211        let client_id = req
1212            .client_id
1213            .as_deref()
1214            .ok_or_else(|| OAuth2Error::InvalidRequest("client_id is required".into()))?;
1215        // X5.1: as on the authorization-code grant, "presented a client_secret"
1216        // widens to "presented a credential", so an mTLS client that sends no
1217        // secret is not turned away before its certificate is ever looked at.
1218        // Still decidable from the request alone, so no client-existence
1219        // oracle is created.
1220        let client_secret = req.client_secret.as_deref();
1221        if client_secret.is_none() && ctx.client_certificate.is_none() {
1222            return Err(OAuth2Error::InvalidClient(
1223                "client authentication is required".into(),
1224            ));
1225        }
1226
1227        // Service accounts are the other machine principal in AXIAM and live in
1228        // their own table. Both client-id families are **server-generated** with
1229        // disjoint prefixes (`oa_` for `oauth2_client`, `sa_` for
1230        // `service_account`), so the prefix selects the table unambiguously and
1231        // keeps this path at one lookup.
1232        //
1233        // The prefix is NOT a security decision: it only picks which table to
1234        // look in. The presented secret must still verify against the row that
1235        // is actually found, so guessing a prefix only routes a caller to a
1236        // table where their `client_id` does not exist — an `invalid_client`,
1237        // exactly as before.
1238        if client_id.starts_with(SERVICE_ACCOUNT_CLIENT_ID_PREFIX) {
1239            // A service account has no `token_endpoint_auth_method` — it is a
1240            // different table with a different lifecycle, and its mTLS story is
1241            // the device-auth path (`POST /api/v1/auth/device`), not RFC 8705.
1242            // So this branch still requires a secret. The error is the generic
1243            // one rather than "client_secret is required", because by this
1244            // point the caller has named an `sa_…` id and a specific message
1245            // would confirm that the id belongs to the service-account family.
1246            let Some(client_secret) = client_secret else {
1247                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1248            };
1249            return self
1250                .handle_client_credentials_service_account(
1251                    tenant_id,
1252                    client_id,
1253                    client_secret,
1254                    req.scope.as_deref(),
1255                    started,
1256                )
1257                .await;
1258        }
1259
1260        // Stage 1 — client lookup (DB round-trip #1).
1261        let t_client_lookup = std::time::Instant::now();
1262        let client = self
1263            .client_repo
1264            .get_by_client_id(tenant_id, client_id)
1265            .await
1266            .map_err(|e| match e {
1267                // QUAL-03/D-11: only a genuinely-unknown client maps to
1268                // invalid_client. Any other error (e.g. a DB outage) must
1269                // surface as a distinct server error, never masquerade as
1270                // bad client credentials (error-oracle). The distinction is
1271                // internal only — the caller-visible description is
1272                // CLIENT_AUTH_FAILED either way, so it cannot be used to
1273                // probe which client ids exist (SEC-086).
1274                AxiamError::NotFound { .. } => {
1275                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
1276                }
1277                other => OAuth2Error::ServerError(other.to_string()),
1278            })?;
1279        let client_lookup_us = t_client_lookup.elapsed().as_micros() as u64;
1280
1281        // Stage 2 — client-secret verification. Keyed HMAC-SHA256 into a stack
1282        // buffer + constant-time compare (`axiam_auth::client_secret`); no heap
1283        // allocation, no lock, and this does NOT touch the Argon2id
1284        // `crypto_semaphore` that bounds password verification. A pre-OBS-1
1285        // row additionally costs one SHA-256 and, once, one DB write to
1286        // migrate; the timing recorded here therefore includes that one-off
1287        // upgrade for a legacy client's first authentication.
1288        //
1289        // X5.1: for an mTLS client this stage is a string comparison against
1290        // the certificate rustls already verified during the handshake — no
1291        // cryptography at all, since the signature check happened once per
1292        // connection rather than once per request. That is the mechanical
1293        // reason certificate-bound tokens are cheap here, and what
1294        // `bench-quick` measures.
1295        let t_secret_verify = std::time::Instant::now();
1296        let secret_result = self
1297            .authenticate_client_credential(tenant_id, &client, client_secret, ctx)
1298            .await;
1299        let secret_verify_us = t_secret_verify.elapsed().as_micros() as u64;
1300        secret_result?;
1301        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
1302
1303        // Verify grant type is allowed
1304        if !client.grant_types.iter().any(|s| s == "client_credentials") {
1305            return Err(OAuth2Error::UnauthorizedClient(
1306                "client not authorized for client_credentials grant".into(),
1307            ));
1308        }
1309
1310        // Resolve scopes: use requested scope if provided, else client's.
1311        // Validate requested scopes are a subset of the client's allowed
1312        // scopes to prevent minting tokens with arbitrary privileges.
1313        let scopes = match req.scope.as_deref() {
1314            Some(s) => {
1315                let requested: Vec<String> = s.split_whitespace().map(String::from).collect();
1316                let invalid: Vec<&str> = requested
1317                    .iter()
1318                    .filter(|sc| !client.scopes.contains(sc))
1319                    .map(String::as_str)
1320                    .collect();
1321                if !invalid.is_empty() {
1322                    return Err(OAuth2Error::InvalidScope(format!(
1323                        "unregistered scopes: {}",
1324                        invalid.join(", ")
1325                    )));
1326                }
1327                requested
1328            }
1329            None => client.scopes.clone(),
1330        };
1331
1332        // Stage 3 — tenant lookup (DB round-trip #2), for the `org_id` claim.
1333        let t_tenant_lookup = std::time::Instant::now();
1334        let tenant = self
1335            .tenant_repo
1336            .get_by_id(tenant_id)
1337            .await
1338            .map_err(|e| match e {
1339                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
1340                other => OAuth2Error::ServerError(other.to_string()),
1341            })?;
1342        let tenant_lookup_us = t_tenant_lookup.elapsed().as_micros() as u64;
1343
1344        // Stage 4 — token mint + EdDSA (Ed25519) signature. No refresh token is
1345        // issued for client_credentials, so there is no persist stage.
1346        let t_token_mint = std::time::Instant::now();
1347        let cnf = self.certificate_binding_for(&client, ctx)?;
1348        // Read before `cnf` is moved into the claims. RFC 9449 §5: a DPoP-bound
1349        // token is announced as such, and everything else stays `Bearer`.
1350        let token_type = crate::dpop::token_type_for(cnf.as_ref());
1351        // X1. Deliberately measured INSIDE `token_mint_us` rather than as its
1352        // own stage: this is the number the R2.4 bench cell compares against an
1353        // unhooked run, and splitting it out would let the hooked and unhooked
1354        // figures be read from different clocks.
1355        let ext = self
1356            .pre_issue_ext_claims(tenant_id, client_id, "oauth2_client", client_id, &scopes)
1357            .await?;
1358        let access_token = issue_client_credentials_token_enriched(
1359            client_id,
1360            tenant_id,
1361            tenant.organization_id,
1362            &scopes,
1363            &self.auth_config,
1364            cnf,
1365            ext,
1366        )
1367        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1368        let token_mint_us = t_token_mint.elapsed().as_micros() as u64;
1369
1370        let scope = if scopes.is_empty() {
1371            None
1372        } else {
1373            Some(scopes.join(" "))
1374        };
1375
1376        let handler_total_us = started.elapsed().as_micros() as u64;
1377        let span = tracing::Span::current();
1378        span.record("client_lookup_us", client_lookup_us);
1379        span.record("secret_verify_us", secret_verify_us);
1380        span.record("tenant_lookup_us", tenant_lookup_us);
1381        span.record("token_mint_us", token_mint_us);
1382        span.record("handler_total_us", handler_total_us);
1383        tracing::debug!(
1384            target: "axiam::perf",
1385            stage = "oauth2.client_credentials",
1386            client_lookup_us,
1387            secret_verify_us,
1388            tenant_lookup_us,
1389            token_mint_us,
1390            handler_total_us,
1391            access_token_len = access_token.len(),
1392            "client_credentials stage timings (I5)"
1393        );
1394
1395        Ok(TokenResponse {
1396            access_token,
1397            token_type,
1398            expires_in: self.auth_config.access_token_lifetime_secs,
1399            refresh_token: None,
1400            scope,
1401            id_token: None,
1402        })
1403    }

1405    /// Refresh token grant (RFC 6749 section 6).
1406    ///
1407    /// Single-use rotation: the old refresh token is revoked and a new one
1408    /// is issued alongside a fresh access token.
1409    async fn handle_refresh_token(
1410        &self,
1411        tenant_id: Uuid,
1412        req: TokenRequest,
1413        ctx: &TokenRequestContext,
1414    ) -> Result<TokenResponse, OAuth2Error> {
1415        let raw_token = req
1416            .refresh_token
1417            .as_deref()
1418            .ok_or_else(|| OAuth2Error::InvalidRequest("refresh_token is required".into()))?;
1419        let client_id = req
1420            .client_id
1421            .as_deref()
1422            .ok_or_else(|| OAuth2Error::InvalidRequest("client_id is required".into()))?;
1423
1424        // Authenticate client BEFORE looking up the refresh token to
1425        // avoid a token-validity oracle (different error for valid vs
1426        // invalid tokens when client auth fails).
1427        //
1428        // X5.1: widened from "a client_secret" to "a credential" so an mTLS
1429        // client is not turned away before its certificate is consulted. Still
1430        // answered from the request alone, so the oracle this ordering exists
1431        // to close stays closed.
1432        let client_secret_val = req.client_secret.as_deref();
1433        if client_secret_val.is_none() && ctx.client_certificate.is_none() {
1434            return Err(OAuth2Error::InvalidClient(
1435                "client authentication is required".into(),
1436            ));
1437        }
1438
1439        let client = self
1440            .client_repo
1441            .get_by_client_id(tenant_id, client_id)
1442            .await
1443            .map_err(|e| match e {
1444                // QUAL-03/D-11: only a genuinely-unknown client maps to
1445                // invalid_client. Any other error (e.g. a DB outage) must
1446                // surface as a distinct server error, never masquerade as
1447                // bad client credentials (error-oracle). The distinction is
1448                // internal only — the caller-visible description is
1449                // CLIENT_AUTH_FAILED either way, so it cannot be used to
1450                // probe which client ids exist (SEC-086).
1451                AxiamError::NotFound { .. } => {
1452                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
1453                }
1454                other => OAuth2Error::ServerError(other.to_string()),
1455            })?;
1456
1457        self.authenticate_client_credential(tenant_id, &client, client_secret_val, ctx)
1458            .await?;
1459        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
1460
1461        // Verify client is authorized for refresh_token grant
1462        if !client.grant_types.iter().any(|s| s == "refresh_token") {
1463            return Err(OAuth2Error::UnauthorizedClient(
1464                "client not authorized for refresh_token grant".into(),
1465            ));
1466        }
1467
1468        // Look up the refresh token by hash (after client auth)
1469        let token_hash = hash_refresh_token(raw_token);
1470        let stored = self
1471            .refresh_token_repo
1472            .get_by_token_hash(tenant_id, &token_hash)
1473            .await
1474            .map_err(|_| {
1475                OAuth2Error::InvalidGrant("refresh token is invalid, expired, or revoked".into())
1476            })?;
1477
1478        // Verify client ownership
1479        if stored.client_id != client_id {
1480            return Err(OAuth2Error::InvalidGrant(
1481                "refresh token was not issued to this client".into(),
1482            ));
1483        }
1484
1485        // Resolve org_id from tenant
1486        let tenant = self
1487            .tenant_repo
1488            .get_by_id(tenant_id)
1489            .await
1490            .map_err(|e| match e {
1491                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
1492                other => OAuth2Error::ServerError(other.to_string()),
1493            })?;
1494
1495        // Issue new access token.
1496        // OAuth2 refresh flow has no persistent session row — use random jti.
1497        //
1498        // X5.1: the binding is re-derived from the certificate presented on
1499        // THIS connection, never copied from the token being refreshed. A
1500        // refresh is a fresh proof of possession — carrying the old `cnf`
1501        // forward would let a client that has lost its certificate keep
1502        // renewing tokens bound to it, and would survive a certificate
1503        // rotation the operator performed precisely to revoke access.
1504        let cnf = self.certificate_binding_for(&client, ctx)?;
1505        // Read before `cnf` is moved into the claims. RFC 9449 §5: a DPoP-bound
1506        // token is announced as such, and everything else stays `Bearer`.
1507        let token_type = crate::dpop::token_type_for(cnf.as_ref());
1508        // X1 — a refresh mints a new access token, so it is an issuance and it
1509        // is hooked. Enrichment is re-derived here rather than copied from the
1510        // token being refreshed, for the same reason `cnf` is: a claim a
1511        // reactor added fifteen minutes ago is not evidence that it would add
1512        // it now, and carrying it forward would let a revoked enrichment
1513        // survive every rotation.
1514        let ext = self
1515            .pre_issue_ext_claims(
1516                tenant_id,
1517                &stored
1518                    .user_id
1519                    .map_or_else(|| client_id.to_string(), |u| u.to_string()),
1520                if stored.user_id.is_some() {
1521                    "user"
1522                } else {
1523                    "oauth2_client"
1524                },
1525                client_id,
1526                &stored.scopes,
1527            )
1528            .await?;
1529        let access_token = if let Some(user_id) = stored.user_id {
1530            issue_access_token_enriched(
1531                user_id,
1532                tenant_id,
1533                tenant.organization_id,
1534                &stored.scopes,
1535                &self.auth_config,
1536                uuid::Uuid::new_v4().to_string(),
1537                axiam_auth::token::AUD_USER,
1538                cnf,
1539                ext,
1540            )
1541            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?
1542        } else {
1543            // Client-credentials-originated refresh (shouldn't normally
1544            // happen, but handle gracefully)
1545            issue_client_credentials_token_enriched(
1546                client_id,
1547                tenant_id,
1548                tenant.organization_id,
1549                &stored.scopes,
1550                &self.auth_config,
1551                cnf,
1552                ext,
1553            )
1554            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?
1555        };
1556
1557        // Refresh token rotation: create new token BEFORE revoking
1558        // the old one. If `create` fails the old token remains valid
1559        // and the client can retry; if `revoke` fails after `create`
1560        // we have a brief window with two valid tokens (less harmful
1561        // than zero tokens forcing a full re-auth).
1562        let new_raw_refresh = generate_refresh_token();
1563        let new_refresh_hash = hash_refresh_token(&new_raw_refresh);
1564        let refresh_expires =
1565            Utc::now() + chrono::Duration::seconds(self.refresh_token_lifetime_secs);
1566
1567        self.refresh_token_repo
1568            .create(CreateRefreshToken {
1569                tenant_id,
1570                token_hash: new_refresh_hash.clone(),
1571                client_id: client_id.to_string(),
1572                user_id: stored.user_id,
1573                scopes: stored.scopes.clone(),
1574                // Rotation preserves the session: the new token descends from
1575                // the same login, so it must carry the same `sid`.
1576                session_id: stored.session_id,
1577                expires_at: refresh_expires,
1578            })
1579            .await
1580            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1581
1582        // Now revoke the old refresh token (single-use rotation).
1583        // If revoke fails, best-effort cleanup of the new token to
1584        // avoid orphaned entries; then surface the appropriate error.
1585        if let Err(revoke_err) = self.refresh_token_repo.revoke(tenant_id, &token_hash).await {
1586            // Best-effort: delete the newly-created token so it
1587            // doesn't linger as an orphan.
1588            if let Err(cleanup_err) = self
1589                .refresh_token_repo
1590                .revoke(tenant_id, &new_refresh_hash)
1591                .await
1592            {
1593                tracing::warn!(
1594                    error = %cleanup_err,
1595                    "token: failed to revoke orphaned refresh token; token may linger"
1596                );
1597            }
1598
1599            return if matches!(revoke_err, AxiamError::NotFound { .. }) {
1600                Err(OAuth2Error::InvalidGrant(
1601                    "refresh token already consumed".into(),
1602                ))
1603            } else {
1604                Err(OAuth2Error::ServerError(
1605                    "failed to revoke old refresh token".into(),
1606                ))
1607            };
1608        }
1609
1610        // Re-issue an ID token when the original grant included `openid`.
1611        let id_token = if stored.scopes.iter().any(|s| s == "openid") {
1612            if let Some(uid) = stored.user_id {
1613                let user = self
1614                    .user_repo
1615                    .get_by_id(tenant_id, uid)
1616                    .await
1617                    .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1618                Some(
1619                    issue_id_token(
1620                        uid,
1621                        tenant_id,
1622                        tenant.organization_id,
1623                        client_id,
1624                        None, // no nonce for refresh
1625                        Some(&user.email),
1626                        Some(&user.username),
1627                        &stored.scopes,
1628                        &self.auth_config,
1629                        // The same session the original login created: an RP
1630                        // that only ever sees refreshed ID tokens must still
1631                        // be able to match a logout token to its session.
1632                        stored.session_id,
1633                    )
1634                    .map_err(|e| OAuth2Error::ServerError(e.to_string()))?,
1635                )
1636            } else {
1637                None
1638            }
1639        } else {
1640            None
1641        };
1642
1643        let scope = if stored.scopes.is_empty() {
1644            None
1645        } else {
1646            Some(stored.scopes.join(" "))
1647        };
1648
1649        Ok(TokenResponse {
1650            access_token,
1651            token_type,
1652            expires_in: self.auth_config.access_token_lifetime_secs,
1653            refresh_token: Some(new_raw_refresh),
1654            scope,
1655            id_token,
1656        })
1657    }

1659    /// Revoke a token per RFC 7009.
1660    ///
1661    /// Always returns `Ok(())` — invalid tokens are silently ignored.
1662    pub async fn revoke_token(
1663        &self,
1664        tenant_id: Uuid,
1665        req: RevokeRequest,
1666        ctx: &TokenRequestContext,
1667    ) -> Result<(), OAuth2Error> {
1668        // Authenticate the client making the revocation request, by whichever
1669        // credential its REGISTRATION names (SEC-093) — not by whichever one
1670        // the request happened to carry.
1671        let ctx = ctx.clone().with_assertion(
1672            req.client_assertion.as_deref(),
1673            req.client_assertion_type.as_deref(),
1674        );
1675        self.authenticate_client(
1676            tenant_id,
1677            &req.client_id,
1678            req.client_secret.as_deref(),
1679            &ctx,
1680        )
1681        .await?;
1682
1683        // Try revoking as a refresh token (hash-based lookup).
1684        // For access tokens (short-lived JWTs), revocation is a no-op —
1685        // they expire naturally within minutes.
1686        let token_hash = hash_refresh_token(&req.token);
1687
1688        // Look up the token first to verify client ownership.
1689        // Only revoke if found and belongs to the requesting client;
1690        // otherwise treat as unknown per RFC 7009.
1691        if let Ok(stored) = self
1692            .refresh_token_repo
1693            .get_by_token_hash(tenant_id, &token_hash)
1694            .await
1695            && stored.client_id == req.client_id
1696        {
1697            self.refresh_token_repo
1698                .revoke(tenant_id, &token_hash)
1699                .await
1700                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1701        }
1702
1703        Ok(())
1704    }

1706    /// Introspect a token per RFC 7662.
1707    pub async fn introspect_token(
1708        &self,
1709        tenant_id: Uuid,
1710        req: IntrospectRequest,
1711        ctx: &TokenRequestContext,
1712    ) -> Result<IntrospectionResponse, OAuth2Error> {
1713        // Authenticate the client making the introspection request, by
1714        // whichever credential its REGISTRATION names (SEC-093).
1715        let ctx = ctx.clone().with_assertion(
1716            req.client_assertion.as_deref(),
1717            req.client_assertion_type.as_deref(),
1718        );
1719        self.authenticate_client(
1720            tenant_id,
1721            &req.client_id,
1722            req.client_secret.as_deref(),
1723            &ctx,
1724        )
1725        .await?;
1726
1727        // First try: decode as JWT access token
1728        if let Ok(validated) = validate_access_token(&req.token, &self.auth_config) {
1729            let claims = &validated.0;
1730
1731            // Verify the token belongs to this tenant to prevent
1732            // cross-tenant introspection / metadata leaks.
1733            let token_tenant = claims.tenant_id.parse::<Uuid>().unwrap_or(Uuid::nil());
1734            if token_tenant != tenant_id {
1735                return Ok(IntrospectionResponse {
1736                    active: false,
1737                    ..Default::default()
1738                });
1739            }
1740
1741            return Ok(IntrospectionResponse {
1742                active: true,
1743                scope: claims.scope.clone(),
1744                // Access tokens don't carry an explicit client_id
1745                // claim; omit it to avoid confusing sub (user ID)
1746                // with client_id.
1747                client_id: None,
1748                sub: Some(claims.sub.clone()),
1749                exp: Some(claims.exp),
1750                iat: Some(claims.iat),
1751                token_type: Some("Bearer".into()),
1752                permissions: claims.permissions.clone(),
1753                // RFC 8705 §3.3 — echoed from the token, like `permissions`.
1754                // A resource server that introspects rather than validating
1755                // locally learns from this field, and only from this field,
1756                // that the token in its hand is not a bearer token.
1757                cnf: claims.cnf.clone(),
1758            });
1759        }
1760
1761        // Second try: look up as refresh token
1762        let token_hash = hash_refresh_token(&req.token);
1763        if let Ok(stored) = self
1764            .refresh_token_repo
1765            .get_by_token_hash(tenant_id, &token_hash)
1766            .await
1767        {
1768            // Only introspect tokens belonging to the requesting
1769            // client — prevent cross-client information leaks.
1770            if stored.client_id != req.client_id {
1771                return Ok(IntrospectionResponse {
1772                    active: false,
1773                    ..Default::default()
1774                });
1775            }
1776
1777            let scope = if stored.scopes.is_empty() {
1778                None
1779            } else {
1780                Some(stored.scopes.join(" "))
1781            };
1782            return Ok(IntrospectionResponse {
1783                active: true,
1784                scope,
1785                client_id: Some(stored.client_id),
1786                sub: stored.user_id.map(|u| u.to_string()),
1787                exp: Some(stored.expires_at.timestamp()),
1788                iat: Some(stored.created_at.timestamp()),
1789                token_type: Some("refresh_token".into()),
1790                // A refresh token is never an RPT: the uma-ticket grant issues
1791                // no refresh token, precisely so that an RPT cannot be renewed
1792                // past the ticket that authorised it.
1793                permissions: None,
1794                // A refresh token is not sender-constrained in AXIAM: it is an
1795                // opaque server-stored value, single-use with rotation, and it
1796                // is presented at the token endpoint where the client
1797                // re-authenticates. Its constraint is client authentication,
1798                // not `cnf`, and the certificate binding of the ACCESS token it
1799                // mints is re-derived from that fresh authentication (see
1800                // `handle_refresh_token`).
1801                cnf: None,
1802            });
1803        }
1804
1805        // Token is not recognised — return inactive per RFC 7662
1806        Ok(IntrospectionResponse {
1807            active: false,
1808            ..Default::default()
1809        })
1810    }

1812    /// Verify client credentials and return the authenticated registration.
1813    ///
1814    /// Shared by every client-authenticating endpoint that is not one of the
1815    /// three grants dispatched inside [`Self::exchange`]: revoke, introspect,
1816    /// PAR, the token-exchange grant and the uma-ticket grant. The first two
1817    /// discard the returned row; the others need it, because the client's
1818    /// registered scopes bound what an exchange may grant and its registered
1819    /// URIs bound which audiences it may address.
1820    ///
1821    /// # SEC-093 — the registration decides here too
1822    ///
1823    /// This used to call [`Self::verify_client_secret`] directly, so those
1824    /// five endpoints authenticated by shared secret **whatever
1825    /// `token_endpoint_auth_method` the client registered**. Because
1826    /// `SurrealOAuth2ClientRepository::create` mints a secret for every client
1827    /// unconditionally, a FAPI client registered for `tls_client_auth` — whose
1828    /// registration validation explicitly *forbids* shared-secret
1829    /// authentication — nonetheless held a live password-equivalent credential
1830    /// that worked at PAR, revoke, introspect, token-exchange and uma-ticket.
1831    /// The operator's model was "this client authenticates with a
1832    /// certificate"; the reality was an OR over two credentials, one of which
1833    /// they were told to discard at creation time.
1834    ///
1835    /// It now delegates to [`Self::authenticate_client_credential`], the same
1836    /// method the authorization_code / client_credentials / refresh_token
1837    /// grants use — deliberately *reused* rather than reimplemented, because a
1838    /// second copy of this rule is how the two drifted apart in the first
1839    /// place. `client_secret` is `Option` because a client registered for a
1840    /// strong method has none to present.
1841    pub async fn authenticate_client(
1842        &self,
1843        tenant_id: Uuid,
1844        client_id: &str,
1845        client_secret: Option<&str>,
1846        ctx: &TokenRequestContext,
1847    ) -> Result<OAuth2Client, OAuth2Error> {
1848        let client = self
1849            .client_repo
1850            .get_by_client_id(tenant_id, client_id)
1851            .await
1852            .map_err(|e| match e {
1853                // QUAL-03/D-11: only a genuinely-unknown client maps to
1854                // invalid_client. Any other error (e.g. a DB outage) must
1855                // surface as a distinct server error, never masquerade as
1856                // bad client credentials (error-oracle). The distinction is
1857                // internal only — the caller-visible description is
1858                // CLIENT_AUTH_FAILED either way, so it cannot be used to
1859                // probe which client ids exist (SEC-086).
1860                AxiamError::NotFound { .. } => {
1861                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
1862                }
1863                other => OAuth2Error::ServerError(other.to_string()),
1864            })?;
1865
1866        // Defence in depth. The lookup above is tenant-scoped in SQL, so this
1867        // can only fire if that predicate is dropped or a repository
1868        // implementation ignores `tenant_id` — which is exactly the
1869        // regression worth catching, since otherwise the cross-tenant
1870        // property lives entirely in a WHERE clause two crates away.
1871        if client.tenant_id != tenant_id {
1872            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1873        }
1874
1875        self.authenticate_client_credential(tenant_id, &client, client_secret, ctx)
1876            .await?;
1877
1878        Ok(client)
1879    }
