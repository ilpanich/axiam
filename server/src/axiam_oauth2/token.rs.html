33/// The single `error_description` every token-endpoint client-authentication
34/// failure returns, whatever actually went wrong (SEC-086).
35///
36/// The `invalid_client` error *code* was already uniform across "no such
37/// client" and "wrong secret". The *description* was not: an unknown client
38/// said `"client not found"`, which let an unauthenticated caller probe
39/// whether a given `client_id` — including an `sa_…` service-account id —
40/// exists. The masquerade the QUAL-03/D-11 comments describe was applied to
41/// the code and not to the message; this constant applies it to both.
42///
43/// The `NotFound`-versus-other-error distinction those comments protect is
44/// unaffected: it stays internal, mapping to `invalid_client` versus
45/// `server_error`, so a DB outage still never reads as bad credentials. Only
46/// the caller-visible wording is collapsed.
47///
48/// Deliberately **not** used at the authorization endpoint
49/// (`authorize.rs`): there `client_id` is public by construction, RFC 6749
50/// §4.1.2.1 requires informing the resource owner directly rather than
51/// redirecting, and no credential is presented — so "invalid client
52/// credentials" would be actively misleading there.
53///
54/// W8 made it `pub`: the REST layer now refuses a malformed
55/// `Authorization: Basic` header before `TokenService` is ever reached, and
56/// that refusal must be worded identically to the ones raised here. A second
57/// copy of the string in `axiam-api-rest` would be a second thing to keep
58/// equal, and SEC-086's whole property is that these answers are
59/// indistinguishable.
60pub const CLIENT_AUTH_FAILED: &str = "invalid client credentials";

73/// The audit action a refresh-token replay is recorded under (T-254).
74///
75/// Its own name in the `oauth2.*` family rather than a metadata key on
76/// `oauth2.token_endpoint`, for the same reason `oauth2.client_auth_failed`

82/// retry are the same event — a rotated refresh token was presented again —

92/// Token request parameters (from form body).
93#[derive(Debug, Deserialize, utoipa::ToSchema)]
94pub struct TokenRequest {
95    pub grant_type: String,
96    pub code: Option<String>,
97    pub redirect_uri: Option<String>,
98    pub client_id: Option<String>,
99    pub client_secret: Option<String>,
100    pub code_verifier: Option<String>,
101    pub refresh_token: Option<String>,
102    pub scope: Option<String>,
103    /// B2 — the secret a device polls with, for
104    /// `grant_type=urn:ietf:params:oauth:grant-type:device_code`
105    /// (RFC 8628 §3.4). Present only on that grant; the REST layer routes it
106    /// to [`crate::device_service::DeviceAuthorizationService`] rather than
107    /// to [`TokenService::exchange`], which never sees it.
108    pub device_code: Option<String>,
109
110    // --- B3 / RFC 8693 token exchange -----------------------------------
111    //
112    // Carried here rather than in a second form type because actix
113    // deserializes one body once, and the token endpoint cannot know which
114    // grant it is serving until it has read `grant_type` out of that body.
115    // `exchange_request()` below projects them into the shape the exchange
116    // service takes, so the grant's own module never sees these Options.
117    pub subject_token: Option<String>,
118    pub subject_token_type: Option<String>,
119    pub requested_token_type: Option<String>,
120    pub actor_token: Option<String>,
121    pub actor_token_type: Option<String>,
122    pub audience: Option<String>,
123    pub resource: Option<String>,
124
125    // --- X2 / UMA 2.0 ticket grant --------------------------------------
126    /// The permission ticket minted by `/uma2/perm`, for
127    /// `grant_type=urn:ietf:params:oauth:grant-type:uma-ticket`.
128    pub ticket: Option<String>,
129    /// The requesting party's identity, as an AXIAM access token.
130    ///
131    /// UMA 2.0 §3.3.1 makes this optional because a client may instead present
132    /// its own RPT for incremental authorization, which v1 does not implement.
133    /// With claims-gathering also deferred, this is the only channel through
134    /// which a requesting party can be named — so the grant requires it and
135    /// says so, rather than resolving to some default subject.
136    pub claim_token: Option<String>,
137    /// Declared format of `claim_token`. Only the AXIAM access-token URN is
138    /// accepted in v1; anything else is refused rather than guessed at.
139    pub claim_token_format: Option<String>,
140
141    // --- X5.1 / RFC 7521 §4.2 client assertion ---------------------------
142    /// The signed JWT a `private_key_jwt` client authenticates with.
143    ///
144    /// Carried here rather than in a second form type for the same reason the
145    /// RFC 8693 parameters are: actix deserializes one body once, and the
146    /// token endpoint cannot know which client-authentication method applies
147    /// until it has read `client_id` out of that body and loaded the
148    /// registration.
149    pub client_assertion: Option<String>,
150    /// Must be `urn:ietf:params:oauth:client-assertion-type:jwt-bearer`.
151    pub client_assertion_type: Option<String>,

154impl TokenRequest {
155    /// Project the RFC 8693 parameters, or `None` when the two mandatory ones
156    /// are absent.
157    ///
158    /// Returning `None` rather than a half-populated request means the caller
159    /// answers `invalid_request` naming what is missing, instead of the
160    /// exchange service reporting a confusing downstream failure about a
161    /// token it was handed as an empty string.
162    pub fn exchange_request(&self) -> Option<crate::token_exchange::TokenExchangeRequest> {
163        let subject_token = self.subject_token.clone().filter(|t| !t.is_empty())?;
164        let subject_token_type = self.subject_token_type.clone().filter(|t| !t.is_empty())?;
165        Some(crate::token_exchange::TokenExchangeRequest {
166            subject_token,
167            subject_token_type,
168            requested_token_type: self.requested_token_type.clone().filter(|t| !t.is_empty()),
169            actor_token: self.actor_token.clone().filter(|t| !t.is_empty()),
170            actor_token_type: self.actor_token_type.clone().filter(|t| !t.is_empty()),
171            scope: self.scope.clone(),
172            audience: self.audience.clone().filter(|t| !t.is_empty()),
173            resource: self.resource.clone().filter(|t| !t.is_empty()),
174        })
175    }

264    /// RFC 7662 §2.2 `aud` — who the token is for (T21.3).

271    /// ask.
272    ///
273    /// `None` on a refresh token (an opaque server-stored value with no

278    pub aud: Option<String>,

281/// What the token endpoint knows about the **connection** a request arrived
282/// on, as opposed to its body (X5.1).
283///
284/// Exists so `TokenService` can authenticate a client by certificate without
285/// depending on actix, rustls, or the REST layer's extractor types. The REST
286/// handler builds one from `HttpRequest::conn_data::<VerifiedClientCert>()`;
287/// a test builds one from a certificate it generated.
288///
289/// [`Default`] is "no certificate", which is what a plaintext or
290/// server-only-TLS connection legitimately has. Every path that *needs* a
291/// certificate checks for it explicitly; nothing infers authentication from
292/// this type's absence.
293#[derive(Debug, Clone, Default)]
294pub struct TokenRequestContext {
295    /// The client certificate rustls verified during the TLS handshake, if
296    /// this connection carried one.
297    ///
298    /// **Only** ever populated from a verified peer chain. See
299    /// `crate::mtls`'s module docs for why the `X-Client-Certificate` proxy
300    /// header is deliberately not a source here.
301    pub client_certificate: Option<PresentedCertificate>,
302    /// A DPoP proof that has **already been verified** for this request, and
303    /// whose `jti` has already been recorded (X5.1, RFC 9449).
304    ///
305    /// The same discipline as [`Self::client_certificate`] and for the same
306    /// reason: this field carries a *conclusion*, never an input. The REST
307    /// layer verifies the `DPoP` header against the actual method and URI —
308    /// which only it knows — and hands the result here. A caller that
309    /// populated this from an unverified header would have turned DPoP into a
310    /// self-signed permission slip, which is why the type is
311    /// `VerifiedDpopProof` and not `String`.
312    pub dpop_proof: Option<crate::dpop::VerifiedDpopProof>,
313    /// The `client_assertion` form parameter, verbatim (RFC 7521 §4.2).
314    ///
315    /// Unlike the two fields above this is a request *body* parameter, not a
316    /// property of the connection, and it is carried here rather than passed
317    /// down from the form because `authenticate_client_credential` is reached
318    /// from four grants that share no request type. It is deliberately the raw
319    /// string: nothing has verified it at this point, and the type says so.
320    pub client_assertion: Option<String>,
321    /// The `client_assertion_type` form parameter.
322    pub client_assertion_type: Option<String>,
323    /// W8 — the credentials decoded from an `Authorization: Basic` header
324    /// (RFC 6749 §2.3.1), if this request carried a well-formed one.
325    ///
326    /// The same discipline as [`Self::client_certificate`] and
327    /// [`Self::dpop_proof`]: the REST layer reads the header, hands it to
328    /// [`crate::client_secret_basic::parse_authorization_header`], and puts
329    /// the *decoded* result here. A malformed header never reaches this field
330    /// — it is answered at the edge with `invalid_client` and a
331    /// `WWW-Authenticate: Basic` challenge — so `None` unambiguously means
332    /// "no Basic credentials were presented" rather than "some were, badly".
333    ///
334    /// Presence here authenticates nothing on its own. Only a client whose
335    /// *registration* names `client_secret_basic` is authenticated from it
336    /// (SEC-093); for every other client this field is inert, and the
337    /// `client_secret_post` case logs a `warn` saying so.
338    pub basic_credentials: Option<crate::client_secret_basic::BasicCredentials>,
339    /// The `sub` of [`Self::client_assertion`], for RFC 7521 §4.2's fallback.
340    ///
341    /// Derived by [`Self::with_assertion`] and by nothing else, so that it
342    /// cannot be set independently of the assertion it is supposed to describe.
343    /// It answers "which client row should be loaded" and never "which client
344    /// is authenticated" — see
345    /// [`crate::private_key_jwt::unverified_client_id`].
346    pub assertion_client_id: Option<String>,
347    /// T21.6 — the issuer identifier this request arrived under, when it
348    /// arrived on a per-tenant path (`{root}/t/{tenant_id}`).
349    ///
350    /// `None` on every request to the deployment-wide `/oauth2/token`, which
351    /// is every request on a deployment that has not set
352    /// `AXIAM__AUTH__TENANT_ISSUER_PATHS` — so `Default` reproduces today's
353    /// behaviour exactly, and so does every test that builds a context
354    /// without naming this field.
355    ///
356    /// It belongs on the *connection* context rather than on `TokenRequest`
357    /// for the same reason the certificate does: it is a property of where the
358    /// request arrived, never of what its body said. A client cannot put it in
359    /// a form field, which is what keeps `iss` from being caller-chosen.
360    pub issuer: Option<String>,
361}

363/// Which client id a token request is for, given a body parameter that may be
364/// absent and an `Authorization: Basic` header that may name one (W8, RFC 6749
365/// §2.3.1).
366///
367/// # Why the header may supply it at all
368///
369/// RFC 6749 §2.3.1 makes the body's `client_id` **optional** for a client
370/// authenticating with the header, and the OpenID Foundation's Basic OP suite
371/// takes it up: 37 of its 38 modules send the id only in the header. AXIAM's
372/// three grants each required `req.client_id`, so without this the new method
373/// would have been unusable by exactly the clients it was added for. The plan
374/// (§4.7) did not mention it; the tree did. See the W8 amendment in
375/// `claude_dev/basic-op-gap-plan.md` §4.7.
376///
377/// # Why disagreement is `invalid_request` and not a preference
378///
379/// A request naming `a` in the header and `b` in the body has made two claims
380/// of identity, and any rule for picking a winner is a rule an attacker can
381/// aim: whichever half a proxy, a log or an audit record reads, the other is
382/// the one that authenticated. Refusing is the only answer that cannot be
383/// exploited by arranging for the two readers to disagree.
384///
385/// It is refused **before** the client lookup, and that placement is load
386/// bearing: the question "do these two strings differ" is answered from the
387/// request alone, so it creates no client-existence oracle and SEC-086's
388/// ordering property is untouched.
389fn resolve_client_id<'a>(
390    from_body: Option<&'a str>,
391    ctx: &'a TokenRequestContext,
392) -> Result<&'a str, OAuth2Error> {
393    match (from_body, ctx.basic_client_id()) {
394        (Some(body), Some(header)) if body != header => Err(OAuth2Error::InvalidRequest(
395            "client_id in the Authorization header and in the request body disagree".into(),
396        )),
397        (Some(body), _) => Ok(body),
398        (None, Some(header)) => Ok(header),
399        // RFC 7521 §4.2: "The `client_id` parameter is optional when using an
400        // assertion for client authentication." OIDC Core §9 then makes the
401        // assertion's `sub` the client identifier, so a `private_key_jwt`
402        // client that sends only `client_assertion` — which is what the OIDF
403        // FAPI 2.0 plan's DPoP lane does on every request — is identifying
404        // itself perfectly well and was being answered "client_id is required".
405        //
406        // The value routes the lookup and authenticates nothing:
407        // `authenticate_client_credential` still verifies the assertion against
408        // the loaded client's registered key and still requires `iss` and `sub`
409        // to equal that client's id, so naming somebody else here only picks
410        // the key the forgery is checked against.
411        (None, None) => ctx
412            .assertion_client_id()
413            .ok_or_else(|| OAuth2Error::InvalidRequest("client_id is required".into())),
414    }
415}

440    /// Whether this request carries **no** client credential of any kind.
441    ///
442    /// The guard three grants run before they look a client up, in one place.
443    ///
444    /// # Why it has to be one place
445    ///
446    /// It has been widened three times. X5.1 added the client certificate,
447    /// because an mTLS client sends no secret at all; W8 added the
448    /// `Authorization: Basic` header; and the third widening — the assertion —
449    /// was missed at all three sites at once, so a `private_key_jwt` client
450    /// presenting a perfectly good `client_assertion` and nothing else was
451    /// refused `invalid_client: client authentication is required` before
452    /// anything looked at it. `private_key_jwt` verification had been
453    /// implemented and wired; this guard sat in front of it. It cost every
454    /// module of the OIDF FAPI 2.0 DPoP lane.
455    ///
456    /// # What it still does not do
457    ///
458    /// It asks only what the REQUEST carries, never what the client
459    /// registered, so it stays decidable before the lookup and SEC-086's
460    /// property holds: a caller with no credential gets the same answer whether
461    /// or not the client id exists. Presence of a credential here authenticates
462    /// nothing — `authenticate_client_credential` decides that, against the
463    /// registration.
464    pub fn carries_no_client_credential(&self, body_secret: Option<&str>) -> bool {
465        body_secret.is_none()
466            && self.client_certificate.is_none()
467            && self.basic_credentials.is_none()
468            && self.client_assertion.is_none()
469    }

685    /// The evidence a refreshed ID token carries (W4, plan §4.3).
686    ///
687    /// Read from the session the original grant was minted in. Three things
688    /// are worth saying about what comes back:
689    ///
690    /// - **`auth_time` equals the original's**, which is what OIDC Core §12.2
691    ///   requires and what `OIDCCRefreshToken` compares. It holds because
692    ///   refresh rotation copies `authenticated_at` onto the session row it
693    ///   writes rather than stamping the clock (X7.2).
694    /// - **`acr` is the class the authentication achieved**, not the one the
695    ///   original authorization request selected. A refresh carries no
696    ///   authorization request, so there is no relying-party preference to
697    ///   express: where the original may have reported the weaker of two
698    ///   satisfied classes because the RP listed it first
699    ///   ([`crate::acr::report_acr`]), a refreshed token reports what the
700    ///   session proves. Both statements are true of the same authentication,
701    ///   and the one available here is the only one that can be derived
702    ///   without inventing a request.
703    /// - **A session that cannot be read yields no evidence at all**, never an
704    ///   error and never a guess. A refresh that works today must not begin to
705    ///   fail because a session row was reaped.
706    async fn session_evidence_for_refresh(
707        &self,
708        tenant_id: Uuid,
709        session_id: Option<Uuid>,
710    ) -> IdTokenEvidence {

712            return IdTokenEvidence::NONE;
713        };

715            Ok(session) => IdTokenEvidence {
716                auth_time: Some(session.authenticated_at.timestamp()),
717                acr: Some(crate::acr::acr_for(&session.amr).as_str().to_owned()),
718                amr: axiam_core::models::session::Amr::encode_list(&session.amr),
719            },
720            Err(e) => {
721                tracing::debug!(
722                    error = %e,
723                    %session_id,
724                    "no session row behind this refresh grant; the re-issued ID token \
725                     carries no authentication evidence"
726                );
727                IdTokenEvidence::NONE
728            }
729        }
730    }

761    /// Make a refresh-token replay visible (T-254).
762    ///
763    /// Two records, because they answer different questions and are read by
764    /// different people:
765    ///
766    /// - **A marker on the session** the token names, which is what an
767    ///   administrator looking at an account sees. Counted per outcome so that
768    ///   "a FAPI client retried a lost rotation" and "a rotated token was
769    ///   presented where no window would accept it" are not one number.
770    /// - **An `oauth2.refresh_token_replayed` audit row**, which is what an
771    ///   operator queries across a tenant, and what survives the session being
772    ///   reaped.
773    ///
774    /// Both are fire-and-forget. A refresh that works today must not begin to
775    /// fail because a marker could not be written, and a refusal must not
776    /// become a `500` for the same reason (T-15-04) — so every error here is
777    /// logged and swallowed.
778    ///
779    /// **The token value is never recorded, in any form** — not the raw token,
780    /// not its hash. The row names the client, the session and the outcome,
781    /// which is everything an investigation needs and nothing that would make
782    /// the audit log a place to steal a credential from.
783    async fn record_refresh_replay(
784        &self,
785        tenant_id: Uuid,
786        client: &OAuth2Client,
787        replayed: &axiam_core::models::oauth2_client::RefreshToken,
788        accepted_under_grace: bool,
789    ) {
790        // The `warn` rather than `info`: on `standard` nothing legitimate
791        // produces one, and on `fapi2` it is at least worth knowing how often
792        // the grace is being leaned on.
793        tracing::warn!(
794            client_id = %client.client_id,
795            profile = client.profile.as_str(),
796            session_id = ?replayed.session_id,
797            accepted_under_grace,
798            "oauth2: a refresh token was presented after it had already been rotated (T-254)"
799        );
800
801        if let Some(session_id) = replayed.session_id
802            && let Err(e) = self
803                .session_repo
804                .mark_refresh_replay(tenant_id, session_id, accepted_under_grace)
805                .await
806        {
807            tracing::error!(
808                error = %e,
809                %session_id,
810                "oauth2: failed to mark a refresh-token replay on its session"
811            );
812        }
813
814        let outcome = if accepted_under_grace {
815            // The request succeeded, and saying otherwise would make every
816            // conformant FAPI retry look like a failure in the audit log.
817            AuditOutcome::Success
818        } else {
819            AuditOutcome::Failure
820        };
821        if let Err(e) = self
822            .audit_repo
823            .append(CreateAuditLogEntry {
824                tenant_id,
825                // The client authenticated, but it is not a user; the same
826                // reasoning `oauth2.client_auth_failed` applies, and the
827                // identity goes to the metadata where it is evidence.
828                actor_id: replayed.user_id.unwrap_or_else(Uuid::nil),
829                actor_type: if replayed.user_id.is_some() {
830                    ActorType::User
831                } else {
832                    ActorType::System
833                },
834                action: REFRESH_REPLAY_AUDIT_ACTION.into(),
835                resource_id: replayed.session_id,
836                outcome,
837                ip_address: None,
838                metadata: Some(serde_json::json!({
839                    "client_id": client.client_id,
840                    "client_profile": client.profile.as_str(),
841                    "session_id": replayed.session_id.map(|id| id.to_string()),
842                    // The discriminator, spelled out rather than left to be
843                    // inferred from `outcome`: an operator filtering the audit
844                    // log must be able to separate the retry the FAPI window
845                    // exists for from a refusal without reading the code.
846                    "disposition": if accepted_under_grace {
847                        "accepted_under_fapi_grace"
848                    } else {
849                        "refused"
850                    },
851                    "rotated_at": replayed.rotated_at.map(|t| t.to_rfc3339()),
852                })),
853            })
854            .await
855        {
856            tracing::error!(
857                error = %e,
858                %tenant_id,
859                "oauth2: failed to write the {REFRESH_REPLAY_AUDIT_ACTION} audit log"
860            );
861        }

875    /// Run `token.pre_issue` and return the `ext` claims to mint with.
876    ///
877    /// Every grant that mints an access token goes through here, which is the
878    /// point: three call sites (authorization code, refresh, client
879    /// credentials) share one hook, so a fourth grant added later cannot
880    /// quietly skip it by forgetting to copy the block.
881    ///
882    /// A veto surfaces as `invalid_grant` — the plan's pinned mapping, and the
883    /// right one: RFC 6749 §5.2 `invalid_grant` is "the grant is not valid for
884    /// this request", which is exactly what a reactor is saying. The reactor's
885    /// reason is **not** echoed to the client (it is third-party text on an
886    /// endpoint reachable with a client credential); the gate has already
887    /// written it to the audit trail with the reactor's id.
888    async fn pre_issue_ext_claims(
889        &self,
890        tenant_id: Uuid,
891        subject: &str,
892        subject_kind: &'static str,
893        client_id: &str,
894        scopes: &[String],
895    ) -> Result<Option<std::collections::BTreeMap<String, String>>, OAuth2Error> {
896        match self
897            .reactor_gate
898            .intercept(
899                tenant_id,
900                reactor_events::TOKEN_PRE_ISSUE,
901                serde_json::json!({
902                    // No token, no credential, no signing key (§22.3) — the
903                    // reactor is told what is about to be minted, not handed
904                    // the means to mint it.
905                    "sub": subject,
906                    "sub_kind": subject_kind,
907                    "client_id": client_id,
908                    "tenant_id": tenant_id,
909                    "scopes": scopes,
910                }),
911            )
912            .await
913        {
914            ReactorOutcome::Allow => Ok(None),
915            ReactorOutcome::Mutate { patch } => {
916                Ok(axiam_auth::token::ext_claims_from_patch(&patch))

918            ReactorOutcome::Deny { reason } => {
919                tracing::info!(
920                    target: "axiam::reactor",
921                    tenant_id = %tenant_id,
922                    client_id,

924                );

927                    "token issuance was refused by policy".into(),
928                ))
929            }
930            // `require_mfa` is a `login.post_auth` answer. The reply validator
931            // refuses it on any other event, so this arm is unreachable through
932            // the wire path — and if it were reachable, the token endpoint has
933            // no step-up to perform, so proceeding "as if allowed" would be
934            // silently discarding a security demand.
935            ReactorOutcome::RequireMfa => {
936                tracing::error!(
937                    target: "axiam::reactor",
938                    tenant_id = %tenant_id,
939                    "a reactor demanded step-up on token.pre_issue, which has no \
940                     step-up to perform — refusing the grant"
941                );
942                Err(OAuth2Error::InvalidGrant(
943                    "token issuance was refused by policy".into(),
944                ))
945            }
946        }

965    /// Verify a presented client secret against the stored hash — the single
966    /// point at which every grant authenticates a confidential client.
967    ///
968    /// The comparison is constant-time and keyed (HMAC-SHA256 under the server
969    /// pepper, `axiam_auth::client_secret`). A pre-OBS-1 row still stored as a
970    /// bare SHA-256 digest verifies against the legacy scheme and is then
971    /// **transparently upgraded**: the replacement hash is written back with a
972    /// compare-and-swap so a concurrent rotation is never clobbered.
973    ///
974    /// Two properties this function exists to guarantee:
975    ///
976    /// - the upgrade is driven by [`ClientSecretVerdict::MatchNeedsUpgrade`],
977    ///   which the hasher only ever returns on a **successful** verification —
978    ///   a wrong secret can never cause a write; and
979    /// - a failed *write* never becomes a failed *authentication*. The client
980    ///   proved possession of the secret; losing the migration is a warning,
981    ///   and the next request simply retries it.
982    ///
983    /// Cost on the hot path: one HMAC-SHA256 over the secret into a stack
984    /// buffer, plus an `OnceLock` acquire load. No heap allocation and no lock
985    /// — strictly less than the `String` the previous `hash_client_secret`
986    /// allocated per request. The DB write happens at most once per legacy
987    /// client, on its first authentication after the upgrade.
988    async fn verify_client_secret(
989        &self,
990        tenant_id: Uuid,
991        client: &OAuth2Client,
992        presented: &str,
993    ) -> Result<(), OAuth2Error> {
994        let hasher =
995            client_secret::global().map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
996
997        match hasher.verify(presented, &client.client_secret_hash) {
998            ClientSecretVerdict::Match => Ok(()),
999            ClientSecretVerdict::MatchNeedsUpgrade { upgraded_hash } => {
1000                match self
1001                    .client_repo
1002                    .upgrade_client_secret_hash(
1003                        tenant_id,
1004                        &client.client_id,
1005                        &client.client_secret_hash,
1006                        &upgraded_hash,
1007                    )
1008                    .await
1009                {
1010                    Ok(true) => tracing::info!(
1011                        client_id = %client.client_id,
1012                        "upgraded legacy client-secret hash to the peppered scheme (OBS-1)"
1013                    ),
1014                    Ok(false) => tracing::debug!(
1015                        client_id = %client.client_id,
1016                        "legacy client-secret hash upgrade skipped — a concurrent write won \
1017                         the compare-and-swap"
1018                    ),
1019                    Err(e) => tracing::warn!(
1020                        client_id = %client.client_id,
1021                        error = %e,
1022                        "failed to persist the legacy client-secret hash upgrade (OBS-1); \
1023                         authentication succeeded, the upgrade will be retried"
1024                    ),
1025                }
1026                Ok(())
1027            }
1028            ClientSecretVerdict::Mismatch => {
1029                Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()))
1030            }
1031        }
1032    }

1034    /// Authenticate a registered client by whichever credential its
1035    /// registration says it uses (X5.1).
1036    ///
1037    /// The **registration** decides, not the request. A client registered for
1038    /// `tls_client_auth` is authenticated by its certificate even if it also
1039    /// posted a `client_secret`, and a client registered for
1040    /// `client_secret_post` is authenticated by its secret even over an mTLS
1041    /// connection. Letting the request choose would turn the two methods into
1042    /// an OR: an attacker holding either credential could authenticate, which
1043    /// is strictly weaker than holding the one the operator registered.
1044    ///
1045    /// This preserves every ordering property SEC-086 established for the
1046    /// secret path — see [`Self::verify_client_secret`] and the callers, where
1047    /// the *presence* of a credential is still checked before the client
1048    /// lookup so that "no such client" and "client lacks this grant" stay
1049    /// indistinguishable.
1050    async fn authenticate_client_credential(
1051        &self,
1052        tenant_id: Uuid,
1053        client: &OAuth2Client,
1054        presented_secret: Option<&str>,
1055        ctx: &TokenRequestContext,
1056    ) -> Result<(), OAuth2Error> {
1057        // T21.2 — the public-client arm, first, because it is the one arm
1058        // that may succeed with no credential at all and the reader should
1059        // not have to establish that the other five cannot.
1060        //
1061        // Two rules, and the second is the one that keeps this from being a
1062        // hole. A client registered for `none` authenticates with nothing, so
1063        // nothing is what it must present: a secret, a Basic header, a client
1064        // assertion or a client certificate arriving on a request from a
1065        // public client is refused, not ignored. RFC 6749 §2.3 permits exactly
1066        // one authentication method per request and this registration names
1067        // none of them, so a credential here is either a misconfigured client
1068        // — pointing at the wrong registration, or at a `client_id` it does
1069        // not own — or somebody probing with one. Accepting it would make the
1070        // registration an OR over "nothing" and "whatever was sent", which is
1071        // strictly weaker than either, and it is the shape SEC-093 exists to
1072        // prevent.
1073        if client.token_endpoint_auth_method.is_public() {
1074            if !ctx.carries_no_client_credential(presented_secret) {
1075                tracing::warn!(
1076                    client_id = %client.client_id,
1077                    "a credential was presented for a client registered as public \
1078                     (token_endpoint_auth_method: none); the request is refused rather than \
1079                     authenticated, because a public registration names no credential to check \
1080                     it against (T21.2)"
1081                );
1082                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1083            }
1084            return Ok(());
1085        }
1086
1087        if client.token_endpoint_auth_method.is_mtls() {
1088            return crate::mtls::authenticate_mtls_client(client, ctx.client_certificate.as_ref());
1089        }
1090
1091        if client.token_endpoint_auth_method.is_private_key_jwt() {
1092            return self
1093                .authenticate_client_assertion(tenant_id, client, ctx)
1094                .await;
1095        }
1096
1097        if client.token_endpoint_auth_method == ClientAuthMethod::ClientSecretBasic {
1098            return self
1099                .authenticate_client_secret_basic(tenant_id, client, presented_secret, ctx)
1100                .await;
1101        }
1102
1103        // `client_secret_post`. W8: a Basic header on this client is **not** a
1104        // second way in. It is ignored — the registration decided which
1105        // channel carries the credential — and said out loud, because the
1106        // silent version of this is indistinguishable from the bug where the
1107        // header quietly wins.
1108        if ctx.basic_credentials.is_some() {
1109            tracing::warn!(
1110                client_id = %client.client_id,
1111                "an Authorization header using the Basic scheme was presented by a client \
1112                 registered for client_secret_post; it is ignored and the request is \
1113                 authenticated by the form-body secret (SEC-093: the registration \
1114                 decides). If this client meant to use HTTP Basic, register it for \
1115                 client_secret_basic"
1116            );
1117        }
1118
1119        let secret = presented_secret
1120            .ok_or_else(|| OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()))?;
1121        self.verify_client_secret(tenant_id, client, secret).await
1122    }

1124    /// `client_secret_basic` client authentication (W8, RFC 6749 §2.3.1).
1125    ///
1126    /// The credential is the same peppered-hash comparison
1127    /// [`Self::verify_client_secret`] performs for `client_secret_post` — one
1128    /// secret, two spellings — so this branch adds no cryptography and no new
1129    /// rate-limit bucket. What it adds is the RFC's two negative rules.
1130    ///
1131    /// # The body secret is refused, *after* the header is verified
1132    ///
1133    /// RFC 6749 §2.3 forbids a client from using more than one
1134    /// authentication method in one request, so a `client_secret` in the body
1135    /// of a `client_secret_basic` client is `invalid_request`. The plan
1136    /// (§4.7) specified that refusal; it did not specify where in the order it
1137    /// goes, and the obvious placement — first — reopens SEC-086. `invalid_request`
1138    /// would then be reachable *only* for a client that exists and is
1139    /// registered for Basic, making both facts decidable by an unauthenticated
1140    /// caller who sends a junk secret in each channel. Verifying the header
1141    /// credential first means the malformed-request answer is only ever given
1142    /// to a caller who has already proven possession of the secret, and
1143    /// everybody else gets the same uniform `invalid_client`. See the W8
1144    /// amendment in `claude_dev/basic-op-gap-plan.md` §4.7.
1145    ///
1146    /// # The header's client id must be the one being authenticated
1147    ///
1148    /// A header naming `a` and a body naming `b` is two claims of identity in
1149    /// one request. It is refused *before* the lookup — see
1150    /// [`resolve_client_id`], where it is decidable from the request alone —
1151    /// so this method only ever sees an agreeing pair, and the equality check
1152    /// here is the defence-in-depth restatement of that.
1153    async fn authenticate_client_secret_basic(
1154        &self,
1155        tenant_id: Uuid,
1156        client: &OAuth2Client,
1157        presented_secret: Option<&str>,
1158        ctx: &TokenRequestContext,
1159    ) -> Result<(), OAuth2Error> {
1160        let Some(credentials) = ctx.basic_credentials.as_ref() else {
1161            // Registered for Basic, presented no Basic header. The body secret
1162            // it may have sent instead is deliberately not consulted: that
1163            // would be the OR over two credentials SEC-093 exists to prevent.
1164            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1165        };
1166
1167        if credentials.client_id() != client.client_id {
1168            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1169        }
1170
1171        self.verify_client_secret(tenant_id, client, credentials.client_secret())
1172            .await?;
1173
1174        if presented_secret.is_some() {
1175            tracing::warn!(
1176                client_id = %client.client_id,
1177                "a client_secret_basic client presented a client_secret in the request body as \
1178                 well as in the Authorization header; RFC 6749 §2.3 permits exactly one \
1179                 authentication method per request, so the request is refused"
1180            );
1181            return Err(OAuth2Error::InvalidRequest(
1182                "a client may use only one authentication method per request (RFC 6749 §2.3): \
1183                 remove client_secret from the request body"
1184                    .into(),
1185            ));
1186        }
1187
1188        Ok(())
1189    }

1224    /// Decide the `cnf` confirmation claim for a token about to be issued
1225    /// (X5.1, RFC 8705 §3).
1226    ///
1227    /// Returns `Ok(None)` for every client that has not asked for binding —
1228    /// which is every client that existed before X5.1 — so this is a branch
1229    /// predictor's dream and costs nothing on the ordinary path.
1230    ///
1231    /// When a client *has* asked for binding and no certificate is present,
1232    /// this **refuses** rather than issuing an unbound token. Silently
1233    /// downgrading would be the worst possible behaviour: the operator
1234    /// configured binding, the resource servers were told to expect it, and
1235    /// the one request that arrives without a certificate would be the one
1236    /// that gets a bearer token.
1237    /// SEC-096: `pub` because the token-exchange and uma-ticket grants are
1238    /// dispatched from `axiam-api-rest` rather than through
1239    /// [`Self::exchange`], and they must reach the same decision. A second
1240    /// copy of this rule in the handler is how one of the two ends up
1241    /// silently issuing an unbound token.
1242    pub fn certificate_binding_for(
1243        &self,
1244        client: &OAuth2Client,
1245        ctx: &TokenRequestContext,
1246    ) -> Result<Option<axiam_auth::token::CnfClaim>, OAuth2Error> {
1247        let mut cnf: Option<axiam_auth::token::CnfClaim> = None;
1248
1249        if crate::fapi::wants_certificate_binding(client) {
1250            let Some(thumbprint) = ctx.certificate_thumbprint() else {
1251                tracing::warn!(
1252                    client_id = %client.client_id,
1253                    "client requires certificate-bound access tokens but presented no client \
1254                     certificate; refusing rather than issuing an unbound token"
1255                );
1256                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1257            };
1258            cnf = Some(axiam_auth::token::CnfClaim::from_certificate_thumbprint(
1259                thumbprint,
1260            ));
1261        }
1262
1263        if crate::fapi::wants_dpop_binding(client) {
1264            let Some(jkt) = ctx.dpop_thumbprint() else {
1265                tracing::warn!(
1266                    client_id = %client.client_id,
1267                    "client requires DPoP-bound access tokens but presented no verified DPoP \
1268                     proof; refusing rather than issuing an unbound token"
1269                );
1270                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1271            };
1272            // A client that asked for both gets both, in one `cnf`. The
1273            // resource server then has to satisfy both — see
1274            // `axiam_auth::token::verify_token_binding`, where "both present"
1275            // is a conjunction.
1276            cnf = Some(match cnf {
1277                Some(existing) => existing.with_dpop_thumbprint(jkt),
1278                None => axiam_auth::token::CnfClaim::from_dpop_thumbprint(jkt),
1279            });
1280        }
1281
1282        Ok(cnf)
1283    }

1285    /// Client-credentials for a **service account** (`sa_…` client id).
1286    ///
1287    /// Service accounts are AXIAM's machine principal, and until now the only
1288    /// way one could authenticate in a running server was mTLS
1289    /// (`POST /api/v1/auth/device`). `create`/`rotate_secret` handed the
1290    /// operator a `client_secret` that **no flow accepted** — this is the path
1291    /// that makes it usable.
1292    ///
1293    /// Three deliberate differences from the `oauth2_client` branch:
1294    ///
1295    /// 1. **The secret is verified BEFORE the status check.** Rejecting a
1296    ///    disabled account without verifying would let an unauthenticated
1297    ///    caller distinguish "exists but disabled" from "does not exist" by
1298    ///    timing. Verifying first means status is only ever consulted for a
1299    ///    caller who already proved possession of the secret — and even then
1300    ///    the error returned is the same generic `invalid_client`.
1301    /// 2. **No scope may be requested.** A service account registers no scopes,
1302    ///    and the `oauth2_client` branch's rule is that a requested scope must
1303    ///    be a subset of the registered ones. With an empty registered set the
1304    ///    only subset is the empty one, so any requested scope is
1305    ///    `invalid_scope`. Authorization for a service account comes from the
1306    ///    roles assigned to its subject, exactly as on the mTLS path.
1307    /// 3. **The token's `sub` is the service-account id**, not the client id,
1308    ///    so the RBAC engine resolves the same principal whether the account
1309    ///    authenticated by certificate or by secret.
1310    async fn handle_client_credentials_service_account(
1311        &self,
1312        tenant_id: Uuid,
1313        client_id: &str,
1314        client_secret: &str,
1315        requested_scope: Option<&str>,
1316        started: std::time::Instant,
1317        // T21.6 — the tenant issuer, when this request arrived on a tenant
1318        // path. Passed rather than read from a context because this branch
1319        // takes no `TokenRequestContext`: RFC 6749 §4.4 client credentials for
1320        // an `sa_…` id authenticate from the body alone.
1321        issuer: Option<&str>,
1322    ) -> Result<TokenResponse, OAuth2Error> {
1323        let t_client_lookup = std::time::Instant::now();
1324        let sa = self
1325            .service_account_repo
1326            .get_by_client_id(tenant_id, client_id)
1327            .await
1328            .map_err(|e| match e {
1329                // Same taxonomy as the oauth2_client branch: only a genuinely
1330                // unknown client is `invalid_client`; a DB outage must never
1331                // masquerade as bad credentials (error-oracle, QUAL-03/D-11).
1332                // The description is CLIENT_AUTH_FAILED here too, so an
1333                // unauthenticated caller cannot learn whether an `sa_…`
1334                // client id exists (SEC-086).
1335                AxiamError::NotFound { .. } => {
1336                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
1337                }
1338                other => OAuth2Error::ServerError(other.to_string()),
1339            })?;
1340        let client_lookup_us = t_client_lookup.elapsed().as_micros() as u64;
1341
1342        // Defence in depth (§17.2 residual 6). The lookup above is already
1343        // tenant-scoped in SQL, so this can only fire if that predicate is
1344        // ever dropped or a repository implementation ignores `tenant_id` —
1345        // exactly the regression the assertion exists to catch. Without it the
1346        // cross-tenant property of this grant lives entirely in a `WHERE`
1347        // clause two crates away, and no test in this crate can prove it
1348        // because the mocks do not filter by tenant.
1349        //
1350        // Deliberately checked *before* the secret is verified: a row from the
1351        // wrong tenant must not have its secret compared at all, and the
1352        // presented client id is the caller's own input, so refusing early
1353        // leaks nothing it did not already know.
1354        if sa.tenant_id != tenant_id {
1355            tracing::error!(
1356                requested_tenant = %tenant_id,
1357                account_tenant = %sa.tenant_id,
1358                "service-account lookup returned a row from another tenant — refusing; \
1359                 this indicates a dropped tenant predicate in the repository layer"
1360            );
1361            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1362        }
1363
1364        let t_secret_verify = std::time::Instant::now();
1365        let secret_result = self
1366            .verify_service_account_secret(tenant_id, &sa, client_secret)
1367            .await;
1368        let secret_verify_us = t_secret_verify.elapsed().as_micros() as u64;
1369        secret_result?;
1370
1371        // Only now — see point 1 in the doc comment.
1372        if sa.status != UserStatus::Active {
1373            tracing::info!(
1374                client_id = %sa.client_id,
1375                status = ?sa.status,
1376                "service-account client-credentials refused: account is not active"
1377            );
1378            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
1379        }
1380
1381        if let Some(scope) = requested_scope.filter(|s| !s.trim().is_empty()) {
1382            return Err(OAuth2Error::InvalidScope(format!(
1383                "unregistered scopes: {} — a service account registers no scopes; \
1384                 its authorization comes from the roles assigned to it",
1385                scope.split_whitespace().collect::<Vec<_>>().join(", ")
1386            )));
1387        }
1388
1389        let t_tenant_lookup = std::time::Instant::now();
1390        let tenant = self
1391            .tenant_repo
1392            .get_by_id(tenant_id)
1393            .await
1394            .map_err(|e| match e {
1395                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
1396                other => OAuth2Error::ServerError(other.to_string()),
1397            })?;
1398        let tenant_lookup_us = t_tenant_lookup.elapsed().as_micros() as u64;
1399
1400        let t_token_mint = std::time::Instant::now();
1401        // X1 — the fourth issuance path, hooked like the other three.
1402        let ext = self
1403            .pre_issue_ext_claims(
1404                tenant_id,
1405                &sa.id.to_string(),
1406                "service_account",
1407                &sa.client_id,
1408                &[],
1409            )
1410            .await?;
1411        let access_token = issue_service_account_client_credentials_token_enriched(
1412            sa.id,
1413            tenant_id,
1414            tenant.organization_id,
1415            &[],
1416            &self.minting_config(issuer),
1417            ext,
1418        )
1419        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1420        let token_mint_us = t_token_mint.elapsed().as_micros() as u64;
1421
1422        let span = tracing::Span::current();
1423        span.record("client_lookup_us", client_lookup_us);
1424        span.record("secret_verify_us", secret_verify_us);
1425        span.record("tenant_lookup_us", tenant_lookup_us);
1426        span.record("token_mint_us", token_mint_us);
1427        span.record("handler_total_us", started.elapsed().as_micros() as u64);
1428
1429        Ok(TokenResponse {
1430            access_token,
1431            token_type: "Bearer".to_string(),

1434            scope: None,
1435            id_token: None,
1436        })

1444    ///
1445    /// This is also what makes the §15.2 Obs 4 lazy migration *reachable*: with
1446    /// no verification path, `upgrade_client_secret_hash` could never fire for a

1449        &self,

1514    /// Exchange an authorization code for tokens (RFC 6749 section 4.1.3).
1515    /// Revoke what a replayed authorization code minted (RFC 6749 §10.5).
1516    ///
1517    /// # What gets revoked, and why it is the session
1518    ///
1519    /// The spec says "all tokens previously issued based on that authorization
1520    /// code". An AXIAM access token is a stateless EdDSA JWT — there is
1521    /// nothing to delete — but every resource request already checks that the
1522    /// session named by the token's `sid` is still live
1523    /// (`is_session_active`, on the hot path and cached). Invalidating that
1524    /// session therefore stops the minted access token at the next request,
1525    /// with no new store and no new per-request read.
1526    ///
1527    /// # The two costs, stated plainly
1528    ///
1529    /// The session is the **browser** session the authorization happened in,
1530    /// so revoking it signs the user out of everything that session reaches,
1531    /// AXIAM's own admin UI included. And a legitimate client that retries
1532    /// after losing a `200` response replays a code exactly as an attacker
1533    /// does — the server cannot tell them apart, which is precisely why
1534    /// §10.5's answer is to revoke rather than to guess.
1535    ///
1536    /// # Not an oracle
1537    ///
1538    /// Both branches return the same `invalid_grant` from the caller, and a
1539    /// hash naming no row revokes nothing. A replay does cost one extra read
1540    /// and one write, so the two are distinguishable by timing — to an
1541    /// attacker who already holds a real authorization code, which is the
1542    /// position this whole path exists to answer.
1543    async fn revoke_after_code_replay(
1544        &self,
1545        tenant_id: Uuid,
1546        code_hash: &str,
1547        client_id: &str,
1548        redirect_uri: &str,
1549    ) {
1550        let session_id = match self
1551            .code_repo
1552            .replayed_session(tenant_id, code_hash, client_id, redirect_uri)
1553            .await
1554        {
1555            Ok(Some(id)) => id,
1556            // Nothing to revoke: an unknown code, or one issued with no
1557            // session behind it. Not a replay worth reporting.
1558            Ok(None) => return,
1559            Err(e) => {
1560                tracing::warn!(
1561                    error = %e,
1562                    "could not check whether a refused authorization code was a replay; \
1563                     the redemption is still denied"
1564                );
1565                return;
1566            }
1567        };
1568
1569        match self.session_repo.invalidate(tenant_id, session_id).await {
1570            Ok(()) => tracing::warn!(
1571                client_id = %client_id,
1572                session_id = %session_id,
1573                "authorization code replay detected; revoked the session it was issued from \
1574                 (RFC 6749 §10.5)"
1575            ),
1576            Err(e) => tracing::warn!(
1577                error = %e,
1578                client_id = %client_id,
1579                session_id = %session_id,
1580                "authorization code replay detected but the session could not be revoked; \
1581                 the redemption is still denied"
1582            ),
1583        }
1584    }

1586    async fn handle_authorization_code(
1587        &self,
1588        tenant_id: Uuid,
1589        req: TokenRequest,
1590        ctx: &TokenRequestContext,
1591    ) -> Result<TokenResponse, OAuth2Error> {
1592        let code = req
1593            .code
1594            .as_deref()
1595            .ok_or_else(|| OAuth2Error::InvalidRequest("code is required".into()))?;
1596        let redirect_uri = req
1597            .redirect_uri
1598            .as_deref()
1599            .ok_or_else(|| OAuth2Error::InvalidRequest("redirect_uri is required".into()))?;
1600        let client_id = resolve_client_id(req.client_id.as_deref(), ctx)?;
1601
1602        // Require client_secret — all clients are confidential (no
1603        // public-client distinction exists yet).
1604        //
1605        // SEC-086 (second pass): this check MUST precede the client lookup.
1606        // When it ran after, a caller posting no `client_secret` at all got
1607        // three distinguishable answers — `invalid_client`/generic for an
1608        // unknown client, `unauthorized_client` for a client lacking the
1609        // grant, and `invalid_client`/"client_secret is required" for a
1610        // client that had it — so client existence stayed decidable on this
1611        // grant even after the description was unified everywhere else.
1612        // `client_credentials` and `refresh_token` already ordered it this
1613        // way; only this grant did not.
1614        //
1615        // X5.1 widens "presented a client_secret" to "presented *a*
1616        // credential", because an mTLS client sends no secret at all. W8
1617        // widens it once more, for the client that sent its secret in the
1618        // `Authorization` header instead of the body. The widened check is
1619        // still decidable **before** the lookup — it asks only what this
1620        // request carries, never what this client registered — so SEC-086's
1621        // property is preserved exactly: a caller with no credential gets one
1622        // answer regardless of whether the client id exists.
1623        //
1624        // T21.2 moves the *answer* after the lookup without moving the
1625        // *question*. A public client presents no credential by construction,
1626        // so the refusal can no longer be decided from the request alone — but
1627        // it must still be indistinguishable for everybody else. It is: both
1628        // the unknown-client branch below and the confidential-client branch
1629        // after it answer `invalid_client: client authentication is required`,
1630        // the same error, in the same order, that this early return produced
1631        // for both cases before. What changed is that a client registered for
1632        // `none` now reaches its grant instead, which is the whole feature and
1633        // is exactly as decidable as the existence of a public client already
1634        // is at `/oauth2/authorize`.
1635        let client_secret = req.client_secret.as_deref();
1636        let carries_no_credential = ctx.carries_no_client_credential(client_secret);
1637        let no_credential_error =
1638            || OAuth2Error::InvalidClient("client authentication is required".into());
1639
1640        // Authenticate client
1641        let client = self
1642            .client_repo
1643            .get_by_client_id(tenant_id, client_id)
1644            .await
1645            .map_err(|e| match e {
1646                // T21.2: an unknown client that presented no credential keeps
1647                // the answer the pre-lookup check gave it, byte for byte.
1648                AxiamError::NotFound { .. } if carries_no_credential => no_credential_error(),
1649                // QUAL-03/D-11: only a genuinely-unknown client maps to
1650                // invalid_client. Any other error (e.g. a DB outage) must
1651                // surface as a distinct server error, never masquerade as
1652                // bad client credentials (error-oracle). The distinction is
1653                // internal only — the caller-visible description is
1654                // CLIENT_AUTH_FAILED either way, so it cannot be used to
1655                // probe which client ids exist (SEC-086).
1656                AxiamError::NotFound { .. } => {
1657                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
1658                }
1659                other => OAuth2Error::ServerError(other.to_string()),
1660            })?;
1661
1662        // T21.2: and a *known* client that presented no credential is refused
1663        // here unless its registration says it has none to present.
1664        if carries_no_credential && !client.token_endpoint_auth_method.is_public() {
1665            return Err(no_credential_error());
1666        }
1667
1668        // Secret verification must also precede the grant-type check, and for
1669        // the same reason: `unauthorized_client` is only reachable by a caller
1670        // who has *already proven possession of the secret*. Ordered the other
1671        // way, supplying any dummy secret still separated "client exists but
1672        // lacks this grant" from "no such client" — a second oracle on the
1673        // same grant, which moving the presence check alone does not close.
1674        // Both safe grants verify first; this now matches them exactly.
1675        //
1676        // X5.1: the profile gate runs *after* authentication, for the same
1677        // reason. `enforce_token_request` reports a fact about the client's
1678        // registration (its profile, its auth method), and reporting that to a
1679        // caller who has not proven possession of a credential would be a new
1680        // oracle on a path that just spent three comment blocks closing one.
1681        self.authenticate_client_credential(tenant_id, &client, client_secret, ctx)
1682            .await?;
1683        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
1684
1685        // Verify client is authorized for authorization_code grant
1686        if !client.grant_types.iter().any(|s| s == "authorization_code") {
1687            return Err(OAuth2Error::UnauthorizedClient(
1688                "client not authorized for authorization_code grant".into(),
1689            ));
1690        }
1691
1692        // Look up the authorization code without consuming it so we
1693        // can verify PKCE *before* marking it as used.  This prevents
1694        // an attacker from burning a valid code by intentionally
1695        // failing PKCE verification.
1696        let code_hash = crate::authorize::hash_code(code);
1697        let auth_code = match self
1698            .code_repo
1699            .get_by_hash(tenant_id, &code_hash, client_id, redirect_uri)
1700            .await
1701        {
1702            Ok(row) => row,
1703            Err(_) => {
1704                // RFC 6749 §10.5's revocation belongs HERE, not only on the
1705                // `consume` failure below — and putting it only there made it
1706                // unreachable for the case it was written for.
1707                //
1708                // `get_by_hash` requires `used = false`. A replayed code is
1709                // therefore refused by this lookup and never reaches `consume`,
1710                // so the revocation guarded by `consume` fired for a race and
1711                // for nothing else. The observable consequence was that an
1712                // access token minted from a replayed code kept working:
1713                // `oidcc-codereuse-30seconds` and
1714                // `fapi2-security-profile-final-attempt-reuse-authorization-code-after-one-second`
1715                // both call the resource endpoint afterwards expecting 4xx and
1716                // both got 200, and the server logged nothing at all — the
1717                // revocation was running, finding no row, and returning quietly.
1718                //
1719                // Safe to call on every lookup failure, including the two that
1720                // are not replays. `replayed_session` matches `used = true`
1721                // together with the same client and redirect_uri pair, so an
1722                // unknown or merely expired code selects nothing and revokes
1723                // nothing; the cost is one read on a request that has already
1724                // failed.
1725                self.revoke_after_code_replay(tenant_id, &code_hash, client_id, redirect_uri)
1726                    .await;
1727                return Err(OAuth2Error::InvalidGrant(
1728                    "authorization code is invalid, expired, or already used".into(),
1729                ));
1730            }
1731        };
1732
1733        // T21.2 — PKCE is required of a public client HERE as well as at
1734        // `/oauth2/authorize`.
1735        //
1736        // The authorization endpoint refuses a public client's request that
1737        // carries no `code_challenge` (step 2b there), so a code with no
1738        // stored challenge should not exist for such a client. "Should not
1739        // exist" is the part worth checking: a code predating the client's
1740        // registration change, a row written by another path, or a bug in that
1741        // gate all produce one, and redeeming it would complete the flow with
1742        // the code as the only credential — which is precisely the attack PKCE
1743        // replaces the client secret with, for a client that has no secret.
1744        //
1745        // `invalid_grant` rather than `invalid_request`: what is wrong is the
1746        // grant being presented, not the shape of this request. The client
1747        // cannot fix it by adding a parameter; it has to start a new
1748        // authorization, which is what this error tells it to do.
1749        if client.token_endpoint_auth_method.is_public() && auth_code.code_challenge.is_none() {
1750            tracing::warn!(
1751                client_id = %client.client_id,
1752                "a public client presented an authorization code that carries no PKCE \
1753                 challenge; refused (T21.2, OAuth 2.0 Security BCP §2.1.1)"
1754            );
1755            return Err(OAuth2Error::InvalidGrant(
1756                "this authorization code was issued without PKCE and cannot be redeemed by a \
1757                 public client (RFC 7636)"
1758                    .into(),
1759            ));
1760        }
1761
1762        // Verify PKCE before consuming the code
1763        if let Some(ref challenge) = auth_code.code_challenge {
1764            let verifier = req.code_verifier.as_deref().ok_or_else(|| {
1765                OAuth2Error::InvalidGrant("code_verifier required for PKCE".into())
1766            })?;
1767            if !pkce::verify_pkce(verifier, challenge) {
1768                return Err(OAuth2Error::InvalidGrant("PKCE verification failed".into()));
1769            }
1770        }
1771
1772        // RFC 9449 §10, and beside PKCE for the same reason PKCE is here.
1773        //
1774        //   When a token request is received, the authorization server computes
1775        //   the JWK Thumbprint of the proof-of-possession public key in the DPoP
1776        //   proof and verifies that it matches the dpop_jkt parameter value in
1777        //   the authorization request.  If they do not match, it MUST reject
1778        //   the request.
1779        //
1780        // `auth_code.dpop_jkt` is that parameter value as it stood when the
1781        // client committed to it — whether it arrived as the `dpop_jkt`
1782        // parameter or as the thumbprint of a DPoP proof presented at the PAR
1783        // endpoint, which §10.1 says binds identically ("behave as if the
1784        // contained public key's thumbprint was provided using dpop_jkt").
1785        //
1786        // Three cases, and only the first is new behaviour:
1787        //
1788        //   - bound, and the proof's key differs (or there is no proof at all)
1789        //     — refuse. A code pinned to a key is worthless to whoever holds
1790        //     the code without the key, which is the entire point of §10.
1791        //   - bound, and the keys match — proceed.
1792        //   - not bound — proceed, and let `certificate_binding_for` apply
1793        //     whatever the *registration* requires. A client that never used
1794        //     the parameter sees exactly the answer it saw before §10 existed.
1795        //
1796        // `invalid_grant` rather than `invalid_dpop_proof`: the proof is
1797        // perfectly valid — `dpop_from_request` already verified its
1798        // signature, `htm`, `htu` and freshness — and what fails is the
1799        // *grant*, which is bound to a key this caller has not demonstrated.
1800        // Saying `invalid_dpop_proof` would send an honest client debugging
1801        // its proof generation over a code it should simply push again.
1802        if let Some(ref bound_jkt) = auth_code.dpop_jkt {
1803            let presented = ctx.dpop_thumbprint();
1804            if presented != Some(bound_jkt.as_str()) {
1805                tracing::debug!(
1806                    client_id = %client_id,
1807                    presented = presented.unwrap_or("<none>"),
1808                    "authorization code is bound to a DPoP key the token request did not prove"
1809                );
1810                return Err(OAuth2Error::InvalidGrant(
1811                    "this authorization code is bound to a DPoP key (RFC 9449 §10); the token \
1812                     request must carry a DPoP proof for that same key"
1813                        .into(),
1814                ));
1815            }
1816        }
1817
1818        // Now atomically consume (mark as used) the code.
1819        if self
1820            .code_repo
1821            .consume(tenant_id, &code_hash, client_id, redirect_uri)
1822            .await
1823            .is_err()
1824        {
1825            // RFC 6749 §10.5: a code used more than once must be denied — the
1826            // `Err` above already does that — and the server "SHOULD revoke
1827            // (when possible) all tokens previously issued based on that
1828            // authorization code".
1829            //
1830            // Best-effort and never fatal: the denial is the guarantee, the
1831            // revocation is the clean-up, and a caller must not learn from a
1832            // 500 that its replay found something.
1833            self.revoke_after_code_replay(tenant_id, &code_hash, client_id, redirect_uri)
1834                .await;
1835            return Err(OAuth2Error::InvalidGrant(
1836                "authorization code is invalid, expired, or already used".into(),
1837            ));
1838        }
1839
1840        // Resolve org_id from tenant
1841        let tenant = self
1842            .tenant_repo
1843            .get_by_id(tenant_id)
1844            .await
1845            .map_err(|e| match e {
1846                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
1847                other => OAuth2Error::ServerError(other.to_string()),
1848            })?;
1849
1850        // Issue access token (include scopes from the authorization code).
1851        // OAuth2 auth-code flow has no persistent session row — use random jti.
1852        //
1853        // X5.1: `cnf` is `None` for every client that has not asked for
1854        // certificate binding, which makes this byte-identical to what the
1855        // pre-X5 call produced.
1856        let cnf = self.certificate_binding_for(&client, ctx)?;
1857        // Read before `cnf` is moved into the claims. RFC 9449 §5: a DPoP-bound
1858        // token is announced as such, and everything else stays `Bearer`.
1859        let token_type = crate::dpop::token_type_for(cnf.as_ref());
1860        // X1 — the code has already been consumed at this point, so a veto
1861        // costs the caller the code. That is the correct order: the code is
1862        // single-use and PKCE-verified, and leaving it spendable while an
1863        // external process decides would hand an attacker a retry window.
1864        let ext = self
1865            .pre_issue_ext_claims(
1866                tenant_id,
1867                &auth_code.user_id.to_string(),
1868                "user",
1869                client_id,
1870                &auth_code.scopes,
1871            )
1872            .await?;
1873        // W7 — the token names the relying party it was granted to
1874        // (RFC 9068 §2.2). UserInfo needs it to know *whose* consent record to
1875        // look for before releasing `phone_number` or `address`, and to
1876        // recognise a `fapi2`-issued token at a point where no authorization
1877        // request is in hand. See `AccessTokenClaims::client_id`.
1878        //
1879        // Built from `AccessTokenSpec` rather than through
1880        // `issue_access_token_for_client`, which every other issuance site
1881        // still uses. That wrapper already carries eleven parameters, and OIDC
1882        // Core §5.5 would have made twelve — the builder exists so that a
1883        // capability only this path needs does not become an argument every
1884        // caller has to pass `None` for.
1885        // T21.3 / RFC 8707 §2 — the audience is the resource this
1886        // authorization was issued for, which the code carries. The token
1887        // request may repeat it or omit it; naming a different one is
1888        // `invalid_target`, because a grant's audience is decided when the
1889        // grant is made and not when the code is spent
1890        // (`crate::resource::resolve_bound`).
1891        //
1892        // The client's `allowed_resources` is deliberately NOT re-consulted
1893        // here: it was consulted at the authorization endpoint, and asking
1894        // again would let a registration edited in the fifteen seconds since
1895        // change what an outstanding code means.
1896        //
1897        // `None` for every code issued without a `resource`, and
1898        // `unwrap_or(AUD_USER)` below then mints exactly the token this grant
1899        // has always minted (I2).
1900        let resource =
1901            crate::resource::resolve_bound(auth_code.resource.as_deref(), req.resource.as_deref())?;
1902        let audience = resource
1903            .as_deref()
1904            .unwrap_or(axiam_auth::token::AUD_USER)
1905            .to_owned();
1906        let access_token = axiam_auth::token::AccessTokenSpec::user(
1907            auth_code.user_id,
1908            tenant_id,
1909            tenant.organization_id,
1910            uuid::Uuid::new_v4().to_string(),
1911        )
1912        .aud(&audience)
1913        .scopes(&auth_code.scopes)
1914        .cnf(cnf)
1915        .ext(ext)
1916        .client_id(Some(client_id))
1917        // The session this authorization was performed in. `jti` stays a
1918        // unique per-token id; the session travels in `sid`, which is what
1919        // the session-revocation check reads. Without it this token is
1920        // refused at every session-validated endpoint, UserInfo included.
1921        .session(auth_code.session_id)
1922        // OIDC Core §5.5 — the claims this authorization asked for by name,
1923        // resolved at the authorization endpoint and carried on the code.
1924        // Empty for every grant that sent no `claims` parameter, and an empty
1925        // list leaves the token byte-identical to what it was before §5.5.
1926        .requested_userinfo_claims(&auth_code.requested_userinfo_claims)
1927        .issue(&self.minting_config(ctx.issuer.as_deref()))
1928        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1929
1930        // Only issue a refresh token when the client is authorized
1931        // for the refresh_token grant type.
1932        let refresh_token = if client.grant_types.iter().any(|s| s == "refresh_token") {
1933            let raw_refresh = generate_refresh_token();
1934            let refresh_hash = hash_refresh_token(&raw_refresh);
1935            let refresh_expires =
1936                Utc::now() + chrono::Duration::seconds(self.refresh_token_lifetime_secs);
1937
1938            self.refresh_token_repo
1939                .create(CreateRefreshToken {
1940                    tenant_id,
1941                    token_hash: refresh_hash,
1942                    client_id: client_id.to_string(),
1943                    user_id: Some(auth_code.user_id),
1944                    scopes: auth_code.scopes.clone(),
1945                    session_id: auth_code.session_id,
1946                    // T-241 — the claims this authorization asked for by name
1947                    // travel with the grant, not with the one access token
1948                    // that happened to be minted first. Without this the
1949                    // client's second token, fifteen minutes later, releases
1950                    // nothing it consented to and it must re-authorize.
1951                    requested_userinfo_claims: auth_code.requested_userinfo_claims.clone(),
1952                    // T21.3 / RFC 8707 — the target travels with the grant,
1953                    // so a refresh re-mints the SAME audience. Without this a
1954                    // resource-bound token would silently become an
1955                    // `axiam:user` token fifteen minutes later, which is a
1956                    // widening performed by the server.
1957                    resource: resource.clone(),
1958                    expires_at: refresh_expires,
1959                })
1960                .await
1961                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1962            Some(raw_refresh)
1963        } else {
1964            None
1965        };
1966
1967        // Issue an ID token when the `openid` scope was requested.
1968        let id_token = if auth_code.scopes.iter().any(|s| s == "openid") {
1969            let user = self
1970                .user_repo
1971                .get_by_id(tenant_id, auth_code.user_id)
1972                .await
1973                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
1974            // X7.2 — the evidence the code was issued with, released only to a
1975            // client on the lane that asked for it. `emits_session_evidence`
1976            // answers `false` for every client in this wave, so this is
1977            // `NONE` and the claim set is what it has always been.
1978            let evidence = if crate::fapi::emits_session_evidence(&client) {
1979                session_evidence_of(&auth_code)
1980            } else {
1981                IdTokenEvidence::NONE
1982            };
1983            Some(
1984                issue_id_token(
1985                    auth_code.user_id,
1986                    client_id,
1987                    auth_code.nonce.as_deref(),
1988                    Some(&user.username),
1989                    &auth_code.scopes,
1990                    &self.minting_config(ctx.issuer.as_deref()),
1991                    auth_code.session_id,
1992                    &evidence,
1993                )
1994                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?,
1995            )
1996        } else {
1997            None
1998        };
1999
2000        let scope = if auth_code.scopes.is_empty() {
2001            None
2002        } else {
2003            Some(auth_code.scopes.join(" "))
2004        };
2005
2006        Ok(TokenResponse {
2007            access_token,
2008            token_type,
2009            expires_in: self.auth_config.access_token_lifetime_secs,
2010            refresh_token,
2011            scope,
2012            id_token,
2013        })
2014    }
2015
2016    /// Client credentials grant (RFC 6749 section 4.4).
2017    ///
2018    /// Machine-to-machine flow — no user context, no refresh token.
2019    ///
2020    /// # Per-stage timing (I5)
2021    ///
2022    /// This handler is the subject of the run-4 TLS plateau investigation, so it
2023    /// carries a stage breakdown on its span: `client_lookup_us`,
2024    /// `secret_verify_us`, `tenant_lookup_us`, `token_mint_us` and
2025    /// `handler_total_us`, recorded on the `oauth2.client_credentials` span and
2026    /// re-emitted as one DEBUG event on `target: "axiam::perf"`.
2027    ///
2028    /// **Cost, and why it is left permanently on:** five `Instant::now()` calls
2029    /// (~20 ns each on a `clock_gettime` vDSO read) and five
2030    /// `Span::record` calls that are no-ops when no subscriber is interested —
2031    /// together well under 0.1 % of the ~370 µs this handler takes even in the
2032    /// *fast* (plaintext, 2 727 req/s) configuration. No flag gates the
2033    /// measurement; only the *reporting* is gated, by the ordinary tracing
2034    /// level (`RUST_LOG=axiam_oauth2=debug`, or any subscriber that samples the
2035    /// span's fields).
2036    ///
2037    /// There is deliberately **no "token persist" stage**: the
2038    /// `client_credentials` grant issues no refresh token and writes nothing —
2039    /// the only DB work is the two reads below. That, plus the fact that the
2040    /// client secret is verified with a single keyed HMAC-SHA256 rather than
2041    /// Argon2id (see `axiam_auth::client_secret`), is why this endpoint
2042    /// measures 2 727 req/s where password login measures 69. The OBS-1
2043    /// remediation deliberately kept this MAC-bound rather than KDF-bound:
2044    /// client secrets are 256-bit CSPRNG values *and* the digest is now keyed,
2045    /// so the offline-guessing threat a KDF defends against does not apply.
2046    #[tracing::instrument(
2047        name = "oauth2.client_credentials",
2048        skip(self, req),
2049        fields(
2050            client_lookup_us = tracing::field::Empty,
2051            secret_verify_us = tracing::field::Empty,
2052            tenant_lookup_us = tracing::field::Empty,
2053            token_mint_us = tracing::field::Empty,
2054            handler_total_us = tracing::field::Empty,
2055        )
2056    )]
2057    async fn handle_client_credentials(
2058        &self,
2059        tenant_id: Uuid,
2060        req: TokenRequest,
2061        ctx: &TokenRequestContext,
2062    ) -> Result<TokenResponse, OAuth2Error> {
2063        let started = std::time::Instant::now();
2064        let client_id = resolve_client_id(req.client_id.as_deref(), ctx)?;
2065        // X5.1: as on the authorization-code grant, "presented a client_secret"
2066        // widens to "presented a credential", so an mTLS client that sends no
2067        // secret is not turned away before its certificate is ever looked at,
2068        // and (W8) neither is a client whose secret arrived in the
2069        // `Authorization` header. Still decidable from the request alone, so
2070        // no client-existence oracle is created.
2071        let client_secret = req.client_secret.as_deref();
2072        if ctx.carries_no_client_credential(client_secret) {
2073            return Err(OAuth2Error::InvalidClient(
2074                "client authentication is required".into(),
2075            ));
2076        }
2077
2078        // Service accounts are the other machine principal in AXIAM and live in
2079        // their own table. Both client-id families are **server-generated** with
2080        // disjoint prefixes (`oa_` for `oauth2_client`, `sa_` for
2081        // `service_account`), so the prefix selects the table unambiguously and
2082        // keeps this path at one lookup.
2083        //
2084        // The prefix is NOT a security decision: it only picks which table to
2085        // look in. The presented secret must still verify against the row that
2086        // is actually found, so guessing a prefix only routes a caller to a
2087        // table where their `client_id` does not exist — an `invalid_client`,
2088        // exactly as before.
2089        if client_id.starts_with(SERVICE_ACCOUNT_CLIENT_ID_PREFIX) {
2090            // A service account has no `token_endpoint_auth_method` — it is a
2091            // different table with a different lifecycle, and its mTLS story is
2092            // the device-auth path (`POST /api/v1/auth/device`), not RFC 8705.
2093            // So this branch still requires a secret. The error is the generic
2094            // one rather than "client_secret is required", because by this
2095            // point the caller has named an `sa_…` id and a specific message
2096            // would confirm that the id belongs to the service-account family.
2097            let Some(client_secret) = client_secret else {
2098                return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
2099            };
2100            return self
2101                .handle_client_credentials_service_account(
2102                    tenant_id,
2103                    client_id,
2104                    client_secret,
2105                    req.scope.as_deref(),
2106                    started,
2107                    ctx.issuer.as_deref(),
2108                )
2109                .await;
2110        }
2111
2112        // Stage 1 — client lookup (DB round-trip #1).
2113        let t_client_lookup = std::time::Instant::now();
2114        let client = self
2115            .client_repo
2116            .get_by_client_id(tenant_id, client_id)
2117            .await
2118            .map_err(|e| match e {
2119                // QUAL-03/D-11: only a genuinely-unknown client maps to
2120                // invalid_client. Any other error (e.g. a DB outage) must
2121                // surface as a distinct server error, never masquerade as
2122                // bad client credentials (error-oracle). The distinction is
2123                // internal only — the caller-visible description is
2124                // CLIENT_AUTH_FAILED either way, so it cannot be used to
2125                // probe which client ids exist (SEC-086).
2126                AxiamError::NotFound { .. } => {
2127                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
2128                }
2129                other => OAuth2Error::ServerError(other.to_string()),
2130            })?;
2131        let client_lookup_us = t_client_lookup.elapsed().as_micros() as u64;
2132
2133        // T21.2 — a public client has no business here. The admin API refuses
2134        // to register `none` together with `client_credentials`, so this is
2135        // the second gate rather than the first: a row that acquired the
2136        // combination some other way must not be able to mint a machine token
2137        // on the strength of its `client_id` alone, which is all a public
2138        // client has. The grant-type check further down would also refuse it,
2139        // and this says why.
2140        if client.token_endpoint_auth_method.is_public() {
2141            return Err(OAuth2Error::UnauthorizedClient(
2142                "a public client (token_endpoint_auth_method: none) cannot use the \
2143                 client_credentials grant: it has no credential to present"
2144                    .into(),
2145            ));
2146        }
2147
2148        // Stage 2 — client-secret verification. Keyed HMAC-SHA256 into a stack
2149        // buffer + constant-time compare (`axiam_auth::client_secret`); no heap
2150        // allocation, no lock, and this does NOT touch the Argon2id
2151        // `crypto_semaphore` that bounds password verification. A pre-OBS-1
2152        // row additionally costs one SHA-256 and, once, one DB write to
2153        // migrate; the timing recorded here therefore includes that one-off
2154        // upgrade for a legacy client's first authentication.
2155        //
2156        // X5.1: for an mTLS client this stage is a string comparison against
2157        // the certificate rustls already verified during the handshake — no
2158        // cryptography at all, since the signature check happened once per
2159        // connection rather than once per request. That is the mechanical
2160        // reason certificate-bound tokens are cheap here, and what
2161        // `bench-quick` measures.
2162        let t_secret_verify = std::time::Instant::now();
2163        let secret_result = self
2164            .authenticate_client_credential(tenant_id, &client, client_secret, ctx)
2165            .await;
2166        let secret_verify_us = t_secret_verify.elapsed().as_micros() as u64;
2167        secret_result?;
2168        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
2169
2170        // Verify grant type is allowed
2171        if !client.grant_types.iter().any(|s| s == "client_credentials") {
2172            return Err(OAuth2Error::UnauthorizedClient(
2173                "client not authorized for client_credentials grant".into(),
2174            ));
2175        }
2176
2177        // Resolve scopes: use requested scope if provided, else client's.
2178        // Validate requested scopes are a subset of the client's allowed
2179        // scopes to prevent minting tokens with arbitrary privileges.
2180        let scopes = match req.scope.as_deref() {
2181            Some(s) => {
2182                let requested: Vec<String> = s.split_whitespace().map(String::from).collect();
2183                let invalid: Vec<&str> = requested
2184                    .iter()
2185                    .filter(|sc| !client.scopes.contains(sc))
2186                    .map(String::as_str)
2187                    .collect();
2188                if !invalid.is_empty() {
2189                    return Err(OAuth2Error::InvalidScope(format!(
2190                        "unregistered scopes: {}",
2191                        invalid.join(", ")
2192                    )));
2193                }
2194                requested
2195            }
2196            None => client.scopes.clone(),
2197        };
2198
2199        // Stage 3 — tenant lookup (DB round-trip #2), for the `org_id` claim.
2200        let t_tenant_lookup = std::time::Instant::now();
2201        let tenant = self
2202            .tenant_repo
2203            .get_by_id(tenant_id)
2204            .await
2205            .map_err(|e| match e {
2206                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
2207                other => OAuth2Error::ServerError(other.to_string()),
2208            })?;
2209        let tenant_lookup_us = t_tenant_lookup.elapsed().as_micros() as u64;
2210
2211        // Stage 4 — token mint + EdDSA (Ed25519) signature. No refresh token is
2212        // issued for client_credentials, so there is no persist stage.
2213        let t_token_mint = std::time::Instant::now();
2214        let cnf = self.certificate_binding_for(&client, ctx)?;
2215        // Read before `cnf` is moved into the claims. RFC 9449 §5: a DPoP-bound
2216        // token is announced as such, and everything else stays `Bearer`.
2217        let token_type = crate::dpop::token_type_for(cnf.as_ref());
2218        // X1. Deliberately measured INSIDE `token_mint_us` rather than as its
2219        // own stage: this is the number the R2.4 bench cell compares against an
2220        // unhooked run, and splitting it out would let the hooked and unhooked
2221        // figures be read from different clocks.
2222        let ext = self
2223            .pre_issue_ext_claims(tenant_id, client_id, "oauth2_client", client_id, &scopes)
2224            .await?;
2225        // T21.3 / RFC 8707 §2 — the machine grant is the one grant with no
2226        // earlier credential to inherit a target from, so this is where the
2227        // client's allow-list is consulted (`resolve_requested`) rather than a
2228        // stored binding (`resolve_bound`). There is no end user to widen
2229        // anything on behalf of: the client is asking for a token for itself,
2230        // and what bounds it is its own registration.
2231        //
2232        // Absent — which is every client-credentials request in every
2233        // deployment today — mints `axiam:m2m`, byte for byte (I2).
2234        let resource =
2235            crate::resource::resolve_requested(&client.allowed_resources, req.resource.as_deref())?;
2236        let access_token = issue_client_credentials_token_for_resource(
2237            client_id,
2238            tenant_id,
2239            tenant.organization_id,
2240            &scopes,
2241            &self.minting_config(ctx.issuer.as_deref()),
2242            cnf,
2243            ext,
2244            resource.as_deref(),
2245        )
2246        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
2247        let token_mint_us = t_token_mint.elapsed().as_micros() as u64;
2248
2249        let scope = if scopes.is_empty() {
2250            None
2251        } else {
2252            Some(scopes.join(" "))
2253        };
2254
2255        let handler_total_us = started.elapsed().as_micros() as u64;
2256        let span = tracing::Span::current();
2257        span.record("client_lookup_us", client_lookup_us);
2258        span.record("secret_verify_us", secret_verify_us);
2259        span.record("tenant_lookup_us", tenant_lookup_us);
2260        span.record("token_mint_us", token_mint_us);
2261        span.record("handler_total_us", handler_total_us);
2262        tracing::debug!(
2263            target: "axiam::perf",
2264            stage = "oauth2.client_credentials",
2265            client_lookup_us,
2266            secret_verify_us,
2267            tenant_lookup_us,
2268            token_mint_us,
2269            handler_total_us,
2270            access_token_len = access_token.len(),
2271            "client_credentials stage timings (I5)"
2272        );
2273
2274        Ok(TokenResponse {
2275            access_token,
2276            token_type,
2277            expires_in: self.auth_config.access_token_lifetime_secs,
2278            refresh_token: None,
2279            scope,
2280            id_token: None,
2281        })
2282    }

2284    /// Refresh token grant (RFC 6749 section 6).
2285    ///
2286    /// Single-use rotation: the old refresh token is revoked and a new one
2287    /// is issued alongside a fresh access token.
2288    async fn handle_refresh_token(
2289        &self,
2290        tenant_id: Uuid,
2291        req: TokenRequest,
2292        ctx: &TokenRequestContext,
2293    ) -> Result<TokenResponse, OAuth2Error> {
2294        let raw_token = req
2295            .refresh_token
2296            .as_deref()
2297            .ok_or_else(|| OAuth2Error::InvalidRequest("refresh_token is required".into()))?;
2298        let client_id = resolve_client_id(req.client_id.as_deref(), ctx)?;
2299
2300        // Authenticate client BEFORE looking up the refresh token to
2301        // avoid a token-validity oracle (different error for valid vs
2302        // invalid tokens when client auth fails).
2303        //
2304        // X5.1: widened from "a client_secret" to "a credential" so an mTLS
2305        // client is not turned away before its certificate is consulted; W8
2306        // widens it again for a secret carried in the `Authorization` header.
2307        // Still answered from the request alone, so the oracle this ordering
2308        // exists to close stays closed.
2309        //
2310        // T21.2 defers the answer past the lookup for the same reason, and
2311        // with the same care, as the authorization-code grant above: a public
2312        // client must be able to refresh the tokens it was legitimately
2313        // issued — that is how a desktop MCP client stays signed in — and
2314        // every other caller keeps the answer it got before.
2315        let client_secret_val = req.client_secret.as_deref();
2316        let carries_no_credential = ctx.carries_no_client_credential(client_secret_val);
2317        let no_credential_error =
2318            || OAuth2Error::InvalidClient("client authentication is required".into());
2319
2320        let client = self
2321            .client_repo
2322            .get_by_client_id(tenant_id, client_id)
2323            .await
2324            .map_err(|e| match e {
2325                // T21.2, as above: unchanged for a caller with no credential.
2326                AxiamError::NotFound { .. } if carries_no_credential => no_credential_error(),
2327                // QUAL-03/D-11: only a genuinely-unknown client maps to
2328                // invalid_client. Any other error (e.g. a DB outage) must
2329                // surface as a distinct server error, never masquerade as
2330                // bad client credentials (error-oracle). The distinction is
2331                // internal only — the caller-visible description is
2332                // CLIENT_AUTH_FAILED either way, so it cannot be used to
2333                // probe which client ids exist (SEC-086).
2334                AxiamError::NotFound { .. } => {
2335                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
2336                }
2337                other => OAuth2Error::ServerError(other.to_string()),
2338            })?;
2339
2340        if carries_no_credential && !client.token_endpoint_auth_method.is_public() {
2341            return Err(no_credential_error());
2342        }
2343
2344        self.authenticate_client_credential(tenant_id, &client, client_secret_val, ctx)
2345            .await?;
2346        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
2347
2348        // Verify client is authorized for refresh_token grant
2349        if !client.grant_types.iter().any(|s| s == "refresh_token") {
2350            return Err(OAuth2Error::UnauthorizedClient(
2351                "client not authorized for refresh_token grant".into(),
2352            ));
2353        }
2354
2355        // Look up the refresh token by hash (after client auth)
2356        let token_hash = hash_refresh_token(raw_token);
2357        let stored = match self
2358            .refresh_token_repo
2359            .get_by_token_hash(tenant_id, &token_hash)
2360            .await
2361        {
2362            Ok(stored) => stored,
2363            Err(_) => {
2364                // T-254 — before answering, find out *why* there was no live
2365                // row. A token that had already been rotated is a replay, and
2366                // a replay is marked and audited whatever the window; a token
2367                // that was revoked at logout or simply ran out is an ordinary
2368                // stale credential and stays the one-line refusal it has
2369                // always been.
2370                //
2371                // The second read runs only on this branch, so the grant that
2372                // succeeds pays nothing for it.
2373                if let Some(rotated) = self.rotated_predecessor(tenant_id, &token_hash).await {
2374                    self.record_refresh_replay(tenant_id, &client, &rotated, false)
2375                        .await;
2376                    return Err(OAuth2Error::InvalidGrant(
2377                        "refresh token already consumed".into(),
2378                    ));
2379                }
2380                return Err(OAuth2Error::InvalidGrant(
2381                    "refresh token is invalid, expired, or revoked".into(),
2382                ));
2383            }
2384        };
2385
2386        // Verify client ownership
2387        if stored.client_id != client_id {
2388            return Err(OAuth2Error::InvalidGrant(
2389                "refresh token was not issued to this client".into(),
2390            ));
2391        }
2392
2393        // T-254 — a live row that already carries `rotated_at` is a
2394        // presentation of a token whose successor has been issued. Only the
2395        // FAPI grace lane can produce one (`supersede` leaves the row
2396        // redeemable; `revoke_rotated` does not), and only a `fapi2`
2397        // registration may be answered from one.
2398        //
2399        // The second half of that sentence is why the profile is asked again
2400        // here rather than inferred from the row's existence: a registration
2401        // moved off `fapi2` between the rotation and the replay leaves a
2402        // superseded row behind it, and the client's registration *now* is
2403        // what decides — never the request, and never a window it used to
2404        // have.
2405        if stored.rotated_at.is_some() {
2406            let accepted = crate::fapi::refresh_rotation_grace_secs(&client).is_some();
2407            self.record_refresh_replay(tenant_id, &client, &stored, accepted)
2408                .await;
2409            if !accepted {
2410                return Err(OAuth2Error::InvalidGrant(
2411                    "refresh token already consumed".into(),
2412                ));
2413            }
2414        }
2415
2416        // Resolve org_id from tenant
2417        let tenant = self
2418            .tenant_repo
2419            .get_by_id(tenant_id)
2420            .await
2421            .map_err(|e| match e {
2422                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
2423                other => OAuth2Error::ServerError(other.to_string()),
2424            })?;
2425
2426        // Issue new access token.
2427        // OAuth2 refresh flow has no persistent session row — use random jti.
2428        //
2429        // X5.1: the binding is re-derived from the certificate presented on
2430        // THIS connection, never copied from the token being refreshed. A
2431        // refresh is a fresh proof of possession — carrying the old `cnf`
2432        // forward would let a client that has lost its certificate keep
2433        // renewing tokens bound to it, and would survive a certificate
2434        // rotation the operator performed precisely to revoke access.
2435        let cnf = self.certificate_binding_for(&client, ctx)?;
2436        // Read before `cnf` is moved into the claims. RFC 9449 §5: a DPoP-bound
2437        // token is announced as such, and everything else stays `Bearer`.
2438        let token_type = crate::dpop::token_type_for(cnf.as_ref());
2439        // X1 — a refresh mints a new access token, so it is an issuance and it
2440        // is hooked. Enrichment is re-derived here rather than copied from the
2441        // token being refreshed, for the same reason `cnf` is: a claim a
2442        // reactor added fifteen minutes ago is not evidence that it would add
2443        // it now, and carrying it forward would let a revoked enrichment
2444        // survive every rotation.
2445        let ext = self
2446            .pre_issue_ext_claims(
2447                tenant_id,
2448                &stored
2449                    .user_id
2450                    .map_or_else(|| client_id.to_string(), |u| u.to_string()),
2451                if stored.user_id.is_some() {
2452                    "user"
2453                } else {
2454                    "oauth2_client"
2455                },
2456                client_id,
2457                &stored.scopes,
2458            )
2459            .await?;
2460        // T21.3 / RFC 8707 §2 — **the rule that stops a token being widened by
2461        // refreshing it.** The refresh token carries the target the grant was
2462        // made for; this request may repeat it or omit it, and anything else —
2463        // a different resource, or a resource on a grant that has none — is
2464        // `invalid_target`.
2465        //
2466        // `resolve_bound` and not `resolve_requested`, deliberately: the
2467        // client's allow-list is not consulted here at all. A client whose
2468        // registration gained a resource yesterday must not be able to
2469        // re-address an outstanding refresh token at it today, because nobody
2470        // asked the end user about that resource when the grant was made.
2471        let resource =
2472            crate::resource::resolve_bound(stored.resource.as_deref(), req.resource.as_deref())?;
2473        let access_token = if let Some(user_id) = stored.user_id {
2474            // W7 — carried across the rotation, because the grant it names is
2475            // the same grant. A refreshed token that lost the claim would lose
2476            // access to consented claims at UserInfo fifteen minutes after the
2477            // consent was given, which the end user would experience as the
2478            // consent not having worked.
2479            issue_access_token_for_client(
2480                user_id,
2481                tenant_id,
2482                tenant.organization_id,
2483                &stored.scopes,
2484                &self.minting_config(ctx.issuer.as_deref()),
2485                uuid::Uuid::new_v4().to_string(),
2486                // RFC 8707 — the SAME audience the grant was issued for.
2487                // `None` for every grant that named no resource, which mints
2488                // `axiam:user` exactly as this path always did (I2).
2489                resource.as_deref().unwrap_or(axiam_auth::token::AUD_USER),
2490                cnf,
2491                ext,
2492                Some(client_id),
2493                // Rotation carries the session forward. A refreshed token that
2494                // dropped `sid` would stop working at UserInfo the moment it
2495                // replaced the one that did, which the end user would
2496                // experience as the session silently ending mid-flow.
2497                stored.session_id,
2498                // And the claims request, for exactly the same reason (T-241).
2499                // Empty for every grant that sent no `claims` parameter, and
2500                // an empty list leaves this token byte-identical to what it
2501                // was before §5.5.
2502                &stored.requested_userinfo_claims,
2503            )
2504            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?
2505        } else {
2506            // Client-credentials-originated refresh (shouldn't normally
2507            // happen, but handle gracefully)
2508            issue_client_credentials_token_for_resource(
2509                client_id,
2510                tenant_id,
2511                tenant.organization_id,
2512                &stored.scopes,
2513                &self.minting_config(ctx.issuer.as_deref()),
2514                cnf,
2515                ext,
2516                // RFC 8707 — carried across the rotation on this branch too.
2517                resource.as_deref(),
2518            )
2519            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?
2520        };
2521
2522        // Refresh token rotation: create new token BEFORE revoking
2523        // the old one. If `create` fails the old token remains valid
2524        // and the client can retry; if `revoke` fails after `create`
2525        // we have a brief window with two valid tokens (less harmful
2526        // than zero tokens forcing a full re-auth).
2527        let new_raw_refresh = generate_refresh_token();
2528        let new_refresh_hash = hash_refresh_token(&new_raw_refresh);
2529        let refresh_expires =
2530            Utc::now() + chrono::Duration::seconds(self.refresh_token_lifetime_secs);
2531
2532        self.refresh_token_repo
2533            .create(CreateRefreshToken {
2534                tenant_id,
2535                token_hash: new_refresh_hash.clone(),
2536                client_id: client_id.to_string(),
2537                user_id: stored.user_id,
2538                scopes: stored.scopes.clone(),
2539                // Rotation preserves the session: the new token descends from
2540                // the same login, so it must carry the same `sid`.
2541                session_id: stored.session_id,
2542                // And the claims request, for the same reason and by the same
2543                // rule (T-241): the successor names the grant its predecessor
2544                // named. Copied, never widened — the list has already been
2545                // through `claims_request::RELEASABLE` at the authorization
2546                // endpoint, and UserInfo re-asks every gate on every call.
2547                requested_userinfo_claims: stored.requested_userinfo_claims.clone(),
2548                // T21.3 / RFC 8707 — and the target, by exactly the same rule.
2549                // The successor is bound to the resource its predecessor was
2550                // bound to, so an unbounded chain of refreshes can never
2551                // reach an audience the original grant did not name.
2552                resource: resource.clone(),
2553                expires_at: refresh_expires,
2554            })
2555            .await
2556            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
2557
2558        // Now retire the old refresh token — on one of two lanes, chosen by
2559        // the client's registration and by nothing else (T-254).
2560        //
2561        // **`fapi2`: supersede.** FAPI 2.0 Security Profile §5.3.2.1-9 says an
2562        // authorization server that rotates refresh tokens shall keep
2563        // accepting the previous one for a short period after issuing its
2564        // successor. That is the only recovery a client has from a rotation
2565        // whose response it never received: under immediate revocation it
2566        // holds a token the server has destroyed, has not been given the
2567        // replacement, and can do nothing but start a whole new authorization.
2568        // The old token therefore stays usable for the grace period and then
2569        // expires on the read path's existing expiry check.
2570        //
2571        // **Everything else: revoke.** The window costs a replay window, and
2572        // it is only affordable where every token is sender-constrained —
2573        // replaying a `fapi2` refresh token inside the window needs the
2574        // client's private key as well. On `standard` the token *is* the
2575        // credential, so the predecessor is revoked at rotation and a second
2576        // presentation is refused, exactly as it was before 065f37c.
2577        //
2578        // Both lanes stamp `rotated_at`, so a later presentation of either is
2579        // recognisable as a replay rather than as an ordinary stale
2580        // credential.
2581        //
2582        // The failure handling below is unchanged, including its NotFound
2583        // case: both methods report NotFound on exactly the condition `revoke`
2584        // did — no live row matched — so a genuinely concurrent second use is
2585        // still caught and still answered "already consumed".
2586        let retirement = match crate::fapi::refresh_rotation_grace_secs(&client) {
2587            Some(grace_secs) => {
2588                self.refresh_token_repo
2589                    .supersede(
2590                        tenant_id,
2591                        &token_hash,
2592                        Utc::now() + chrono::Duration::seconds(grace_secs),
2593                    )
2594                    .await
2595            }
2596            None => {
2597                self.refresh_token_repo
2598                    .revoke_rotated(tenant_id, &token_hash)
2599                    .await
2600            }
2601        };
2602        if let Err(revoke_err) = retirement {
2603            // Best-effort: delete the newly-created token so it
2604            // doesn't linger as an orphan.
2605            if let Err(cleanup_err) = self
2606                .refresh_token_repo
2607                .revoke(tenant_id, &new_refresh_hash)
2608                .await
2609            {
2610                tracing::warn!(
2611                    error = %cleanup_err,
2612                    "token: failed to revoke orphaned refresh token; token may linger"
2613                );
2614            }
2615
2616            return if matches!(revoke_err, AxiamError::NotFound { .. }) {
2617                Err(OAuth2Error::InvalidGrant(
2618                    "refresh token already consumed".into(),
2619                ))
2620            } else {
2621                Err(OAuth2Error::ServerError(
2622                    "failed to revoke old refresh token".into(),
2623                ))
2624            };
2625        }
2626
2627        // Re-issue an ID token when the original grant included `openid`.
2628        let id_token = if stored.scopes.iter().any(|s| s == "openid") {
2629            if let Some(uid) = stored.user_id {
2630                let user = self
2631                    .user_repo
2632                    .get_by_id(tenant_id, uid)
2633                    .await
2634                    .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
2635                // W4 — the same gate as the code-exchange path above. For a
2636                // client on the honour lane the evidence is resolved from the
2637                // session behind `stored.session_id`, which refresh rotation
2638                // preserves precisely so that OIDC Core §12.2 holds: the
2639                // `auth_time` of a refreshed ID token equals the original's,
2640                // and a freshly-stamped one would be a lie that grows younger
2641                // with every refresh.
2642                let evidence = if crate::fapi::emits_session_evidence(&client) {
2643                    self.session_evidence_for_refresh(tenant_id, stored.session_id)
2644                        .await
2645                } else {
2646                    IdTokenEvidence::NONE
2647                };
2648                Some(
2649                    issue_id_token(
2650                        uid,
2651                        client_id,
2652                        None,
2653                        Some(&user.username),
2654                        &stored.scopes,
2655                        &self.minting_config(ctx.issuer.as_deref()),
2656                        // The same session the original login created: an RP
2657                        // that only ever sees refreshed ID tokens must still
2658                        // be able to match a logout token to its session.
2659                        stored.session_id,
2660                        &evidence,
2661                    )
2662                    .map_err(|e| OAuth2Error::ServerError(e.to_string()))?,
2663                )
2664            } else {
2665                None
2666            }
2667        } else {
2668            None
2669        };
2670
2671        let scope = if stored.scopes.is_empty() {
2672            None
2673        } else {
2674            Some(stored.scopes.join(" "))
2675        };
2676
2677        Ok(TokenResponse {
2678            access_token,
2679            token_type,
2680            expires_in: self.auth_config.access_token_lifetime_secs,
2681            refresh_token: Some(new_raw_refresh),
2682            scope,
2683            id_token,
2684        })
2685    }

2687    /// Revoke a token per RFC 7009.
2688    ///
2689    /// Always returns `Ok(())` — invalid tokens are silently ignored.
2690    pub async fn revoke_token(
2691        &self,
2692        tenant_id: Uuid,
2693        req: RevokeRequest,
2694        ctx: &TokenRequestContext,
2695    ) -> Result<(), OAuth2Error> {
2696        // Authenticate the client making the revocation request, by whichever
2697        // credential its REGISTRATION names (SEC-093) — not by whichever one
2698        // the request happened to carry.
2699        let ctx = ctx.clone().with_assertion(
2700            req.client_assertion.as_deref(),
2701            req.client_assertion_type.as_deref(),
2702        );
2703        self.authenticate_client(
2704            tenant_id,
2705            &req.client_id,
2706            req.client_secret.as_deref(),
2707            &ctx,
2708        )
2709        .await?;
2710
2711        // Try revoking as a refresh token (hash-based lookup).
2712        // For access tokens (short-lived JWTs), revocation is a no-op —
2713        // they expire naturally within minutes.
2714        let token_hash = hash_refresh_token(&req.token);
2715
2716        // Look up the token first to verify client ownership.
2717        // Only revoke if found and belongs to the requesting client;
2718        // otherwise treat as unknown per RFC 7009.
2719        if let Ok(stored) = self
2720            .refresh_token_repo
2721            .get_by_token_hash(tenant_id, &token_hash)
2722            .await
2723            && stored.client_id == req.client_id
2724        {
2725            self.refresh_token_repo
2726                .revoke(tenant_id, &token_hash)
2727                .await
2728                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
2729        }
2730
2731        Ok(())
2732    }

2734    /// Introspect a token per RFC 7662.
2735    pub async fn introspect_token(
2736        &self,
2737        tenant_id: Uuid,
2738        req: IntrospectRequest,
2739        ctx: &TokenRequestContext,
2740    ) -> Result<IntrospectionResponse, OAuth2Error> {
2741        // Authenticate the client making the introspection request, by
2742        // whichever credential its REGISTRATION names (SEC-093).
2743        let ctx = ctx.clone().with_assertion(
2744            req.client_assertion.as_deref(),
2745            req.client_assertion_type.as_deref(),
2746        );
2747        let client = self
2748            .authenticate_client(
2749                tenant_id,
2750                &req.client_id,
2751                req.client_secret.as_deref(),
2752                &ctx,
2753            )
2754            .await?;
2755
2756        // T21.2 — a public client may not introspect.
2757        //
2758        // RFC 7662 §2.1: "the endpoint MUST also require some form of
2759        // authorization to access this endpoint". A public client presents
2760        // nothing, so its `client_id` — a value that travels in every browser
2761        // redirect it makes — would be the whole of the authorization, and
2762        // this endpoint answers questions about tokens belonging to anyone in
2763        // the tenant. That is a token-metadata oracle for whoever reads one
2764        // authorization URL, so it is refused here rather than served.
2765        //
2766        // Revocation is deliberately NOT refused: RFC 7009 §2.1 contemplates
2767        // public clients explicitly, and `revoke_token` acts only on a token
2768        // the requesting client already owns, so the worst a caller who
2769        // guessed a public `client_id` can do is destroy a token they already
2770        // hold.
2771        if client.token_endpoint_auth_method.is_public() {
2772            return Err(OAuth2Error::InvalidClient(
2773                "a public client (token_endpoint_auth_method: none) cannot introspect tokens \
2774                 (RFC 7662 §2.1 requires an authenticated caller)"
2775                    .into(),
2776            ));
2777        }
2778
2779        // First try: decode as JWT access token.
2780        //
2781        // T21.3 — `decode_access_token_any_audience`, NOT
2782        // `validate_access_token`. The two differ in exactly one respect: this
2783        // one does not pin `aud` to AXIAM's own two audiences. Everything else
2784        // a token must satisfy — the EdDSA signature, the issuer, the expiry,
2785        // the tenant check two lines below — is unchanged.
2786        //
2787        // This is the one place in the codebase where that is the right call,
2788        // and I3 is why. AXIAM's own extractors and `validate_access_token`
2789        // stay pinned, so a token minted for an MCP server is refused at
2790        // `/users/me` and at `CheckAccess`. Introspection is not admitting a
2791        // caller, it is **describing a token to an already-authenticated
2792        // resource server** (RFC 7662), and a server that cannot be told what
2793        // a token's audience is cannot perform the audience check RFC 8707
2794        // exists to make possible. Refusing here would have made every
2795        // resource-bound token look *inactive* — the one answer that is
2796        // actively misleading, because `active: false` means "revoked,
2797        // expired or never issued" and this token is none of those.
2798        if let Ok(claims) =
2799            axiam_auth::token::decode_access_token_any_audience(&req.token, &self.auth_config)
2800        {
2801            let claims = &claims;
2802
2803            // Verify the token belongs to this tenant to prevent
2804            // cross-tenant introspection / metadata leaks.
2805            let token_tenant = claims.tenant_id.parse::<Uuid>().unwrap_or(Uuid::nil());
2806            if token_tenant != tenant_id {
2807                return Ok(IntrospectionResponse {
2808                    active: false,
2809                    ..Default::default()
2810                });
2811            }
2812
2813            return Ok(IntrospectionResponse {
2814                active: true,
2815                scope: claims.scope.clone(),
2816                // Access tokens don't carry an explicit client_id
2817                // claim; omit it to avoid confusing sub (user ID)
2818                // with client_id.
2819                client_id: None,
2820                sub: Some(claims.sub.clone()),
2821                exp: Some(claims.exp),
2822                iat: Some(claims.iat),
2823                token_type: Some("Bearer".into()),
2824                permissions: claims.permissions.clone(),
2825                // RFC 8705 §3.3 — echoed from the token, like `permissions`.
2826                // A resource server that introspects rather than validating
2827                // locally learns from this field, and only from this field,
2828                // that the token in its hand is not a bearer token.
2829                cnf: claims.cnf.clone(),
2830                // RFC 7662 §2.2 / RFC 8707 — echoed for the same reason, and
2831                // the reason this endpoint decodes without pinning the
2832                // audience in the first place.
2833                aud: claims.aud.clone(),
2834            });
2835        }
2836
2837        // Second try: look up as refresh token
2838        let token_hash = hash_refresh_token(&req.token);
2839        if let Ok(stored) = self
2840            .refresh_token_repo
2841            .get_by_token_hash(tenant_id, &token_hash)
2842            .await
2843        {
2844            // Only introspect tokens belonging to the requesting
2845            // client — prevent cross-client information leaks.
2846            if stored.client_id != req.client_id {
2847                return Ok(IntrospectionResponse {
2848                    active: false,
2849                    ..Default::default()
2850                });
2851            }
2852
2853            let scope = if stored.scopes.is_empty() {
2854                None
2855            } else {
2856                Some(stored.scopes.join(" "))
2857            };
2858            return Ok(IntrospectionResponse {
2859                active: true,
2860                scope,
2861                client_id: Some(stored.client_id),
2862                sub: stored.user_id.map(|u| u.to_string()),
2863                exp: Some(stored.expires_at.timestamp()),
2864                iat: Some(stored.created_at.timestamp()),
2865                token_type: Some("refresh_token".into()),
2866                // A refresh token is never an RPT: the uma-ticket grant issues
2867                // no refresh token, precisely so that an RPT cannot be renewed
2868                // past the ticket that authorised it.
2869                permissions: None,
2870                // A refresh token is not sender-constrained in AXIAM: it is an
2871                // opaque server-stored value, single-use with rotation, and it
2872                // is presented at the token endpoint where the client
2873                // re-authenticates. Its constraint is client authentication,
2874                // not `cnf`, and the certificate binding of the ACCESS token it
2875                // mints is re-derived from that fresh authentication (see
2876                // `handle_refresh_token`).
2877                cnf: None,
2878                // A refresh token is not addressed at anybody: it is redeemed
2879                // at AXIAM's token endpoint and nowhere else. The resource it
2880                // is *bound* to is deliberately not reported — it is not this
2881                // token's audience, it is the audience of a token this one
2882                // will mint, and RFC 7662's `aud` describes the token being
2883                // introspected.
2884                aud: None,
2885            });
2886        }
2887
2888        // Token is not recognised — return inactive per RFC 7662
2889        Ok(IntrospectionResponse {
2890            active: false,
2891            ..Default::default()
2892        })
2893    }

2895    /// Verify client credentials and return the authenticated registration.
2896    ///
2897    /// Shared by every client-authenticating endpoint that is not one of the
2898    /// three grants dispatched inside [`Self::exchange`]: revoke, introspect,
2899    /// PAR, the token-exchange grant and the uma-ticket grant. The first two
2900    /// discard the returned row; the others need it, because the client's
2901    /// registered scopes bound what an exchange may grant and its registered
2902    /// URIs bound which audiences it may address.
2903    ///
2904    /// # SEC-093 — the registration decides here too
2905    ///
2906    /// This used to call [`Self::verify_client_secret`] directly, so those
2907    /// five endpoints authenticated by shared secret **whatever
2908    /// `token_endpoint_auth_method` the client registered**. Because
2909    /// `SurrealOAuth2ClientRepository::create` mints a secret for every client
2910    /// unconditionally, a FAPI client registered for `tls_client_auth` — whose
2911    /// registration validation explicitly *forbids* shared-secret
2912    /// authentication — nonetheless held a live password-equivalent credential
2913    /// that worked at PAR, revoke, introspect, token-exchange and uma-ticket.
2914    /// The operator's model was "this client authenticates with a
2915    /// certificate"; the reality was an OR over two credentials, one of which
2916    /// they were told to discard at creation time.
2917    ///
2918    /// It now delegates to [`Self::authenticate_client_credential`], the same
2919    /// method the authorization_code / client_credentials / refresh_token
2920    /// grants use — deliberately *reused* rather than reimplemented, because a
2921    /// second copy of this rule is how the two drifted apart in the first
2922    /// place. `client_secret` is `Option` because a client registered for a
2923    /// strong method has none to present.
2924    pub async fn authenticate_client(
2925        &self,
2926        tenant_id: Uuid,
2927        client_id: &str,
2928        client_secret: Option<&str>,
2929        ctx: &TokenRequestContext,
2930    ) -> Result<OAuth2Client, OAuth2Error> {
2931        let client = self
2932            .client_repo
2933            .get_by_client_id(tenant_id, client_id)
2934            .await
2935            .map_err(|e| match e {
2936                // QUAL-03/D-11: only a genuinely-unknown client maps to
2937                // invalid_client. Any other error (e.g. a DB outage) must
2938                // surface as a distinct server error, never masquerade as
2939                // bad client credentials (error-oracle). The distinction is
2940                // internal only — the caller-visible description is
2941                // CLIENT_AUTH_FAILED either way, so it cannot be used to
2942                // probe which client ids exist (SEC-086).
2943                AxiamError::NotFound { .. } => {
2944                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
2945                }
2946                other => OAuth2Error::ServerError(other.to_string()),
2947            })?;
2948
2949        // Defence in depth. The lookup above is tenant-scoped in SQL, so this
2950        // can only fire if that predicate is dropped or a repository
2951        // implementation ignores `tenant_id` — which is exactly the
2952        // regression worth catching, since otherwise the cross-tenant
2953        // property lives entirely in a WHERE clause two crates away.
2954        if client.tenant_id != tenant_id {
2955            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
2956        }
2957
2958        self.authenticate_client_credential(tenant_id, &client, client_secret, ctx)
2959            .await?;
2960
2961        Ok(client)
2962    }
2963}
