1//! The CIBA grant at the token endpoint (G-7, CIBA Core §10–11).
2//!
3//! A child module of [`crate::token`] so that it issues through exactly the
4//! private machinery the other user grants use — client authentication by the
5//! registered method, the profile's request-time rules, sender-constraining,
6//! the `token.pre_issue` reactor hook, the refresh-token snapshot — rather
7//! than through a copy of it. The pending-request store is a parameter, not a
8//! ninth type parameter on `TokenService`, because only this grant reads it.
9//!
10//! # Order, and what each step may write
11//!
12//! 1. `auth_req_id` present — decidable from the request alone.
13//! 2. **The client authenticates**, by its registered method, and the
14//!    profile's request-time rules run (`enforce_token_request`, D-17). The
15//!    client must hold the grant.
16//! 3. The request is looked up **in this tenant, for this client**. Another
17//!    client's (or tenant's) `auth_req_id` is `invalid_grant`, exactly like an
18//!    unknown one — and nothing has been written for it.
19//! 4. Only now is the poll recorded (T-404's lesson: nothing is written for a
20//!    request before the request carrying it is validated), as a
21//!    compare-and-set on the `last_polled_at` read in step 3, and the back-off
22//!    decided: inside the interval is `slow_down`, whatever the state.
23//! 5. Expired → `expired_token` (and the row marked, conditionally).
24//!    Pending → `authorization_pending`, denied → `access_denied`, redeemed →
25//!    `invalid_grant`.
26//! 6. Approved → redeemed on the X6 two-layer arbiter, conditional on the
27//!    client; a lost race is `invalid_grant`.
28//! 7. The subject is re-read **after** redemption (so a refused account burns
29//!    the approval): an account that may no longer act, or is under
30//!    brute-force lockout, is `invalid_grant`.
31//! 8. Tokens are minted with the approval's evidence: the access token names
32//!    the approving session in `sid` (ending that session ends them), the ID
33//!    token carries `auth_time`, `acr` (the reported class, `report_acr`) and
34//!    `amr`, and a refresh token — only for a client holding `refresh_token` —
35//!    carries the same snapshot (D-9).
36

40use axiam_core::error::AxiamError;

50use super::{CLIENT_AUTH_FAILED, TokenRequest, TokenRequestContext, TokenResponse, TokenService};
51use crate::acr::{Acr, report_acr};

55impl<OC, AC, TR, RT, UR, SA, SR, AR> TokenService<OC, AC, TR, RT, UR, SA, SR, AR>
56where
57    OC: OAuth2ClientRepository,
58    SA: ServiceAccountRepository,
59    AC: AuthorizationCodeRepository,
60    TR: TenantRepository,
61    RT: RefreshTokenRepository,
62    UR: UserRepository,
63    SR: axiam_core::repository::SessionRepository,
64    AR: AuditLogRepository,
65{
66    /// `grant_type=urn:openid:params:grant-type:ciba` (CIBA Core §10.1).
67    ///
68    /// See the module documentation for the order of checks and what each
69    /// may write.
70    ///
71    /// # Errors
72    ///
73    /// The CIBA Core §11 / RFC 6749 §5.2 code for each refusal.
74    pub async fn exchange_ciba<CR: CibaRequestRepository>(
75        &self,
76        tenant_id: Uuid,
77        req: TokenRequest,
78        ctx: &TokenRequestContext,
79        requests: &CR,
80    ) -> Result<TokenResponse, OAuth2Error> {
81        // --- 1. the parameter --------------------------------------------
82        let auth_req_id = req
83            .auth_req_id
84            .as_deref()
85            .filter(|v| !v.is_empty())
86            .ok_or_else(|| {
87                OAuth2Error::InvalidRequest("auth_req_id is required for the CIBA grant".into())
88            })?;
89        let client_id = super::resolve_client_id(req.client_id.as_deref(), ctx)?;
90
91        // --- 2. the client -------------------------------------------------
92        //
93        // A CIBA client is never public (`ciba::validate_client_registration`),
94        // so a request carrying no credential is refused before the lookup —
95        // one answer whether or not the client exists (SEC-086's ordering).
96        let client_secret = req.client_secret.as_deref();
97        if ctx.carries_no_client_credential(client_secret) {
98            return Err(OAuth2Error::InvalidClient(
99                "client authentication is required".into(),
100            ));
101        }
102        let client = self
103            .client_repo
104            .get_by_client_id(tenant_id, client_id)
105            .await
106            .map_err(|e| match e {
107                AxiamError::NotFound { .. } => {
108                    OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into())
109                }
110                other => OAuth2Error::ServerError(other.to_string()),
111            })?;
112        if client.tenant_id != tenant_id {
113            return Err(OAuth2Error::InvalidClient(CLIENT_AUTH_FAILED.into()));
114        }
115        self.authenticate_client_credential(tenant_id, &client, client_secret, ctx)
116            .await?;
117        crate::fapi::enforce_token_request(&client, ctx.evidence())?;
118        if !holds_ciba_grant(&client.grant_types) {
119            return Err(OAuth2Error::UnauthorizedClient(
120                "client not authorized for the CIBA grant".into(),
121            ));
122        }
123
124        // --- 3. the request, for this client -------------------------------
125        let hash = hash_auth_req_id(auth_req_id);
126        let unknown = || OAuth2Error::InvalidGrant("auth_req_id is invalid".into());
127        let row = requests
128            .get_by_hash(tenant_id, &hash)
129            .await
130            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?
131            .filter(|r| r.client_id == client.client_id && r.tenant_id == tenant_id)
132            .ok_or_else(unknown)?;
133
134        // --- 4. the poll, recorded only now --------------------------------
135        let now = Utc::now();
136        let (too_fast, next_interval) = poll_backoff(row.interval_secs, row.last_polled_at, now);
137        let recorded = requests
138            .record_poll(tenant_id, row.id, row.last_polled_at, now, next_interval)
139            .await
140            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
141        if too_fast || !recorded {
142            // A lost compare-and-set is another token request for the same
143            // `auth_req_id` landing in between: polling too fast by definition.
144            return Err(OAuth2Error::SlowDown);
145        }
146
147        // --- 5. the state --------------------------------------------------
148        if row.status == CibaRequestStatus::Expired || row.is_expired_at(now) {
149            if matches!(
150                row.status,
151                CibaRequestStatus::Pending | CibaRequestStatus::Approved
152            ) && let Err(e) = requests.mark_expired(tenant_id, row.id, row.version).await
153            {
154                // Best effort: the sweep marks it too, and the answer below
155                // does not depend on the mark.
156                tracing::debug!(error = %e, "could not mark an expired CIBA request");
157            }
158            return Err(OAuth2Error::ExpiredToken);
159        }
160        // RFC 8707 — the target was decided at `bc-authorize`; a token request
161        // may repeat it, never change it. Checked before the state so a client
162        // mis-sending it learns on its first poll.
163        let resource =
164            crate::resource::resolve_bound(row.resource.as_deref(), req.resource.as_deref())?;
165        match row.status {
166            CibaRequestStatus::Pending => return Err(OAuth2Error::AuthorizationPending),
167            CibaRequestStatus::Denied => {
168                return Err(OAuth2Error::AccessDenied(
169                    "the user denied the request".into(),
170                ));
171            }
172            CibaRequestStatus::Redeemed => {
173                return Err(OAuth2Error::InvalidGrant(
174                    "auth_req_id has already been used".into(),
175                ));
176            }
177            CibaRequestStatus::Expired => return Err(OAuth2Error::ExpiredToken),
178            CibaRequestStatus::Approved => {}
179        }
180
181        // --- 6. single use, on the X6 arbiter ------------------------------
182        let redeemed = requests
183            .redeem(tenant_id, &hash, &client.client_id)
184            .await
185            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?
186            .ok_or_else(|| OAuth2Error::InvalidGrant("auth_req_id has already been used".into()))?;
187        let (Some(user_id), Some(approval)) = (redeemed.user_id, redeemed.approval.clone()) else {
188            // An approved request with no subject or no evidence is a datastore
189            // inconsistency: refuse rather than mint a token for nobody.
190            return Err(OAuth2Error::ServerError(
191                "approved CIBA request carries no subject".into(),
192            ));
193        };
194
195        // --- 7. the subject, re-read after the approval is spent ------------
196        let refused = || {
197            OAuth2Error::InvalidGrant(
198                "the account this grant was issued for can no longer sign in".into(),
199            )
200        };
201        let user = match self.user_repo.get_by_id(tenant_id, user_id).await {
202            Ok(user) => user,
203            Err(AxiamError::NotFound { .. }) => return Err(refused()),
204            Err(e) => return Err(OAuth2Error::ServerError(e.to_string())),
205        };
206        if !user_may_be_subject(&user) {
207            tracing::info!(
208                %tenant_id,
209                %user_id,
210                "refusing a CIBA redemption: the account may no longer sign in or is locked out"
211            );
212            return Err(refused());
213        }
214
215        // --- 8. issuance ---------------------------------------------------
216        let tenant = self
217            .tenant_repo
218            .get_by_id(tenant_id)
219            .await
220            .map_err(|e| match e {
221                AxiamError::NotFound { .. } => OAuth2Error::InvalidRequest("unknown tenant".into()),
222                other => OAuth2Error::ServerError(other.to_string()),
223            })?;
224        let cnf = self.certificate_binding_for(&client, ctx)?;
225        let token_type = crate::dpop::token_type_for(cnf.as_ref());
226        let ext = self
227            .pre_issue_ext_claims(
228                tenant_id,
229                &user_id.to_string(),
230                "user",
231                &client.client_id,
232                &redeemed.scopes,
233            )
234            .await?;
235        let audience = resource
236            .as_deref()
237            .unwrap_or(axiam_auth::token::AUD_USER)
238            .to_owned();
239        let minting = self.minting_config(ctx.issuer.as_deref());
240        let access_token = axiam_auth::token::AccessTokenSpec::user(
241            user_id,
242            tenant_id,
243            tenant.organization_id,
244            Uuid::new_v4().to_string(),
245        )
246        .aud(&audience)
247        .scopes(&redeemed.scopes)
248        .cnf(cnf)
249        .ext(ext)
250        .client_id(Some(&client.client_id))
251        .session(Some(approval.session_id))
252        .issue(&minting)
253        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
254
255        let refresh_token = if client.grant_types.iter().any(|g| g == "refresh_token") {
256            let raw_refresh = generate_refresh_token();
257            self.refresh_token_repo
258                .create(CreateRefreshToken {
259                    tenant_id,
260                    token_hash: hash_refresh_token(&raw_refresh),
261                    client_id: client.client_id.clone(),
262                    user_id: Some(user_id),
263                    scopes: redeemed.scopes.clone(),
264                    session_id: Some(approval.session_id),
265                    requested_userinfo_claims: Vec::new(),
266                    resource: resource.clone(),
267                    // D-9 — the approval's evidence travels with the grant, so
268                    // a refreshed ID token reports the original authentication.
269                    auth_time: Some(approval.auth_time),
270                    acr: Some(approval.acr.clone()),
271                    amr: approval.amr.clone(),
272                    expires_at: Utc::now()
273                        + chrono::Duration::seconds(self.refresh_token_lifetime_secs),
274                })
275                .await
276                .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
277            Some(raw_refresh)
278        } else {
279            None
280        };
281
282        // CIBA Core §10.1.1: an ID token is always returned (the scope always
283        // carries `openid`). It reports the approval: what was achieved, in the
284        // class the request asked for when the achievement satisfies it.
285        let achieved = Acr::from_wire(&approval.acr).unwrap_or(Acr::SingleFactor);
286        let evidence = IdTokenEvidence {
287            auth_time: Some(approval.auth_time.timestamp()),
288            acr: Some(report_acr(achieved, &redeemed.acr_values).to_owned()),
289            amr: axiam_core::models::session::Amr::encode_list(&approval.amr),
290        };
291        let id_token = issue_id_token(
292            user_id,
293            &client.client_id,
294            None,
295            Some(&user.username),
296            &redeemed.scopes,
297            &minting,
298            Some(approval.session_id),
299            &evidence,
300        )
301        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
302
303        tracing::debug!(
304            client_id = %client.client_id,
305            grant_type = CIBA_GRANT_TYPE,
306            "CIBA request redeemed"
307        );
308
309        Ok(TokenResponse {
310            access_token,
311            token_type,
312            expires_in: self.auth_config.access_token_lifetime_secs,
313            refresh_token,
314            scope: Some(redeemed.scopes.join(" ")),
315            id_token: Some(id_token),
316        })
317    }
