1//! OAuth2 Token Exchange (RFC 8693) — B3.
2//!
3//! Design and rationale: `claude_dev/token-exchange-design.md`. The short
4//! version, because it governs every function below:
5//!
6//! > **An exchange may only ever narrow.** There is no input, no
7//! > configuration and no client grant that causes the issued token to permit
8//! > something the subject token did not already permit.
9//!
10//! Each check here is that rule restated for one dimension — scopes,
11//! audience, lifetime, actor identity — and each has a test named after the
12//! thing it stops.
13//!
14//! Like the device grant, this is a **separate service** rather than another
15//! arm of [`crate::token::TokenService`]: it needs different collaborators,
16//! and the REST layer keeps the seam at one `match` on `grant_type`.
17

37/// The subject/actor token type for an AXIAM-issued access token.
38///
39/// Since X4 this is no longer the *only* admissible subject token type — see
40/// [`TOKEN_TYPE_JWT`] — but it remains the only type an exchange **issues**,
41/// and the only type an `actor_token` may be.
42pub const TOKEN_TYPE_ACCESS_TOKEN: &str = "urn:ietf:params:oauth:token-type:access_token";

44/// RFC 8693 §3 generic JWT subject-token type (X4).
45///
46/// Accepted only on the external path: a caller presenting a partner's token
47/// rarely knows whether that IdP would call it an "access token" in AXIAM's
48/// sense, and RFC 8693 provides this type for exactly that case. The token is
49/// still shape-checked — an ID token labelled `…:jwt` is refused just as
50/// firmly as one labelled `…:id_token`.
51pub const TOKEN_TYPE_JWT: &str = "urn:ietf:params:oauth:token-type:jwt";

53/// Token types a caller sometimes reaches for and never gets (X4).
54///
55/// Named individually rather than lumped into "unsupported" because a caller
56/// who sent one has made a *specific* mistake, and "unsupported
57/// subject_token_type" sends them looking for a config switch that does not
58/// exist. Refusing these is a security property, not a gap: a refresh token is
59/// a re-authentication credential and an ID token is an assertion to a client
60/// about a login — neither is a bearer credential for an API, and both are
61/// treated as lower-risk artefacts by the IdPs that issue them.
62pub const TOKEN_TYPE_REFRESH_TOKEN: &str = "urn:ietf:params:oauth:token-type:refresh_token";
63pub const TOKEN_TYPE_ID_TOKEN: &str = "urn:ietf:params:oauth:token-type:id_token";

65/// Grant-list entry that permits impersonation.
66///
67/// Encoded as a grant-type entry rather than a new column: the client
68/// registration already carries a list of what a client may do, adding to it
69/// needs no migration, and an operator reading `grant_types` sees this
70/// capability in the same place as every other one.
71pub const MAY_IMPERSONATE_GRANT: &str = "urn:axiam:params:oauth:grant-type:may-impersonate";

73/// `POST /oauth2/token` with `grant_type=…:token-exchange` (RFC 8693 §2.1).
74#[derive(Debug, Clone, Deserialize, utoipa::ToSchema)]
75pub struct TokenExchangeRequest {
76    pub subject_token: String,
77    pub subject_token_type: String,
78    /// RFC 8693 §2.1. Optional, and the AS picks the default — but a client
79    /// that *names* a type it does not get must be told, not handed an
80    /// access token and left to discover the difference. v1 issues access
81    /// tokens only, so any other explicit value is refused.
82    pub requested_token_type: Option<String>,
83    /// Presence of an actor token selects **delegation**; absence selects
84    /// **impersonation**, which is off unless the client is granted it.
85    pub actor_token: Option<String>,
86    pub actor_token_type: Option<String>,
87    /// Space-separated. Absent means "the subject's own scopes".
88    pub scope: Option<String>,
89    /// Target audience for the issued token. **The allow-list is the
90    /// client's `redirect_uris`** (SEC-089): a target is accepted if it
91    /// appears in the client's registered redirect URIs, or is one of
92    /// AXIAM's own built-in audiences. There is no separate audience field
93    /// in v1 — adding a redirect URI to a client also authorises it as a
94    /// token audience for that client. This allow-list is checked at two
95    /// call sites in this file (same-domain exchange, and the X4 external
96    /// exchange) — both are tagged `SEC-089` and must be touched together
97    /// (and together with this comment) if a future dedicated
98    /// `allowed_token_targets` field replaces this reuse. Full write-up in
99    /// `docs/api/token-exchange.md#audience`.
100    pub audience: Option<String>,
101    /// RFC 8707. Treated as a synonym of `audience`; if both are given they
102    /// must agree, because silently preferring one would make the request
103    /// mean something the caller did not write.
104    pub resource: Option<String>,
105}

188/// Answers "which of these scope names does this subject actually hold" (X4).
189///
190/// The port exists because a partner's token proves *authentication* and
191/// nothing else: what the resolved user may do is a question only the RBAC
192/// engine can answer, and `axiam-oauth2` cannot see the engine. Implemented in
193/// the composition root over `axiam-authz`.
194///
195/// The contract is one-directional and that is the whole point: an
196/// implementation may return **fewer** names than it was given, never more and
197/// never different ones. A caller treats the result as a filter, not as a
198/// source of scopes.
199pub trait SubjectScopeAuthority: Send + Sync {
200    fn held_scopes<'a>(
201        &'a self,
202        tenant_id: Uuid,
203        subject_id: Uuid,
204        candidates: &'a [String],
205    ) -> Pin<Box<dyn Future<Output = Result<Vec<String>, String>> + Send + 'a>>;
206}

208/// What an exchange did, for the audit record.
209///
210/// Impersonation is the reason this type exists. A delegated token carries an
211/// `act` claim naming who acted; an impersonated one deliberately does not, so
212/// the audit entry is the *only* surviving evidence that the acting party was
213/// not the subject.
214#[derive(Debug, Clone, Copy, PartialEq, Eq)]
215pub enum ExchangeKind {
216    Delegation,
217    Impersonation,
218    /// X4 — the subject token came from a trusted external IdP.
219    ///
220    /// A third kind rather than a flavour of impersonation, even though the
221    /// issued token also carries no `act` claim. The distinction is what the
222    /// acting party's authority rests on: an impersonating client asserts on
223    /// *its own* authority that it may be the user, while an external exchange
224    /// presents a trusted IdP's signed assertion that the user authenticated,
225    /// addressed to us. Different evidence, different gate (the per-provider
226    /// trust block, not `may_impersonate`), and an audit reader must be able
227    /// to tell them apart at a glance.
228    External,
229}
230
231impl ExchangeKind {
232    pub fn as_str(self) -> &'static str {
233        match self {
234            Self::Delegation => "delegation",
235            Self::Impersonation => "impersonation",
236            Self::External => "external",
237        }
238    }
239}

284/// Intersect the three scope sets, preserving the caller's requested order.
285///
286/// A free function, not a method, because it is the single most important
287/// thing in the feature and should be testable without a datastore, a client
288/// or a token. `requested` is already defaulted to the subject's scopes by the
289/// caller when the parameter was absent.
290///
291/// Returns `Err` naming the first requested scope that fails. **Refusing
292/// rather than silently dropping** is deliberate: a quietly-narrowed token
293/// works for some calls and not others, and the caller discovers that at the
294/// second service with no idea why. An error at the exchange is actionable.
295pub fn narrow_scopes(
296    requested: &[String],
297    subject_scopes: &[String],
298    client_scopes: &[String],
299) -> Result<Vec<String>, OAuth2Error> {
300    let mut granted: Vec<String> = Vec::with_capacity(requested.len());
301    for scope in requested {
302        if !subject_scopes.contains(scope) {
303            return Err(OAuth2Error::InvalidScope(format!(
304                "the subject token does not hold scope '{scope}'"
305            )));
306        }
307        // The client's own registration bounds the result even when the
308        // subject token is broader. This is what stops a compromised
309        // low-privilege service that has captured an admin's token from
310        // minting an admin token: it can only ever reach its own ceiling.
311        if !client_scopes.contains(scope) {
312            return Err(OAuth2Error::InvalidScope(format!(
313                "the exchanging client is not registered for scope '{scope}'"
314            )));
315        }
316        if !granted.contains(scope) {
317            granted.push(scope.clone());
318        }
319    }
320    if granted.is_empty() {
321        return Err(OAuth2Error::InvalidScope(
322            "the exchange would grant no scopes at all".into(),
323        ));
324    }
325    Ok(granted)
326}

347/// Token-exchange service.
348#[derive(Clone)]
349pub struct TokenExchangeService<TR> {
350    tenant_repo: TR,
351    auth_config: AuthConfig,
352    /// Independent ceiling on an exchanged token's lifetime, applied on top of
353    /// "never outlives its subject".
354    max_lifetime_secs: i64,
355    /// X4 — resolves external subject tokens. `None` disables the external
356    /// path entirely: a deployment that never wires this in cannot accept a
357    /// foreign token no matter how a provider row is configured, which is the
358    /// behaviour every pre-X4 test asserts and must keep asserting.
359    external_resolver: Option<std::sync::Arc<dyn ExternalSubjectResolver>>,
360    /// X4 — the RBAC engine's answer for the resolved user.
361    ///
362    /// Paired with `external_resolver` and checked as a pair: an external
363    /// exchange with no way to consult the engine would have to either fail or
364    /// skip the check, and skipping it is how `scope_map` output becomes an
365    /// authority in its own right.
366    scope_authority: Option<std::sync::Arc<dyn SubjectScopeAuthority>>,
367}

369impl<TR> TokenExchangeService<TR>
370where
371    TR: TenantRepository,
372{
373    pub fn new(tenant_repo: TR, auth_config: AuthConfig, max_lifetime_secs: i64) -> Self {
374        Self {
375            tenant_repo,
376            auth_config,
377            max_lifetime_secs,
378            external_resolver: None,
379            scope_authority: None,
380        }
381    }
382
383    /// Enable the X4 external path (both collaborators, together).
384    ///
385    /// Taken as a pair on purpose. There is no useful configuration in which
386    /// external subject tokens are accepted but the engine is not consulted,
387    /// and a builder that let the two be set independently would make that
388    /// configuration expressible.
389    pub fn with_external_subjects(
390        mut self,
391        resolver: std::sync::Arc<dyn ExternalSubjectResolver>,
392        authority: std::sync::Arc<dyn SubjectScopeAuthority>,
393    ) -> Self {
394        self.external_resolver = Some(resolver);
395        self.scope_authority = Some(authority);
396        self
397    }
398
399    /// Perform an exchange.
400    ///
401    /// The exchanging client is assumed already authenticated by the caller —
402    /// the token endpoint does that for every grant — and `client` is the row
403    /// that authentication produced.
404    ///
405    /// `cnf` is the confirmation claim the *exchanging client* earned on this
406    /// request, produced by `TokenService::certificate_binding_for` from the
407    /// certificate or DPoP proof it actually presented (SEC-096). It is
408    /// threaded in rather than computed here because the transport context
409    /// belongs to the HTTP layer and this service must not learn about it.
410    /// `None` — every client that registered no binding — issues exactly the
411    /// bytes this grant issued before SEC-096.
412    pub async fn exchange(
413        &self,
414        tenant_id: Uuid,
415        client: &OAuth2Client,
416        req: TokenExchangeRequest,
417        cnf: Option<axiam_auth::token::CnfClaim>,
418    ) -> Result<ExchangeOutcome, OAuth2Error> {
419        if !client
420            .grant_types
421            .iter()
422            .any(|g| g == TOKEN_EXCHANGE_GRANT_TYPE)
423        {
424            return Err(OAuth2Error::UnauthorizedClient(
425                "client is not registered for the token-exchange grant".into(),
426            ));
427        }
428
429        // Checked before anything is decoded: a client asking for a refresh
430        // token wants a different thing entirely, and answering it with an
431        // access token would satisfy the request shape while defeating the
432        // lifetime cap the whole grant is built around.
433        if let Some(want) = req.requested_token_type.as_deref()
434            && want != TOKEN_TYPE_ACCESS_TOKEN
435        {
436            return Err(OAuth2Error::InvalidRequest(format!(
437                "unsupported requested_token_type '{want}'; \
438                 v1 issues only {TOKEN_TYPE_ACCESS_TOKEN}"
439            )));
440        }
441
442        // Named refusals first (X4). A caller who sent a refresh or ID token
443        // type has made a specific mistake, and "unsupported
444        // subject_token_type" would send them hunting for a config switch that
445        // does not and will not exist.
446        match req.subject_token_type.as_str() {
447            TOKEN_TYPE_REFRESH_TOKEN => {
448                return Err(OAuth2Error::InvalidRequest(
449                    "a refresh token is a re-authentication credential, not a subject \
450                     token; present an access token"
451                        .into(),
452                ));
453            }
454            TOKEN_TYPE_ID_TOKEN => {
455                return Err(OAuth2Error::InvalidRequest(
456                    "an ID token asserts a login to a client, not authority at an API; \
457                     present an access token"
458                        .into(),
459                ));
460            }
461            TOKEN_TYPE_ACCESS_TOKEN | TOKEN_TYPE_JWT => {}
462            other => {
463                return Err(OAuth2Error::InvalidRequest(format!(
464                    "unsupported subject_token_type '{other}'; \
465                     accepted types are {TOKEN_TYPE_ACCESS_TOKEN} and {TOKEN_TYPE_JWT}"
466                )));
467            }
468        }
469
470        // --- internal or external? (X4) --------------------------------------
471        //
472        // Routed on the subject token's UNVERIFIED `iss`, which is safe
473        // precisely because neither destination trusts it: a token claiming to
474        // be ours is checked against our signing key, and a token claiming a
475        // partner's issuer is checked against that partner's JWKS. The claim
476        // chooses which key the token faces, never whether it faces one.
477        //
478        // A token with no readable `iss` takes the internal path, where
479        // `decode_access_token` refuses it — that is the same answer this grant
480        // gave before X4 existed, and it keeps "malformed" a single failure
481        // rather than two that differ by which branch noticed.
482        let claimed_issuer = unverified_issuer_of(&req.subject_token);
483        let is_external = match claimed_issuer.as_deref() {
484            Some(iss) => iss != self.auth_config.effective_issuer(),
485            None => false,
486        };
487
488        if is_external {
489            return self.exchange_external(tenant_id, client, req, cnf).await;
490        }
491
492        // From here down is B3, unchanged except for the transitivity check.
493        if req.subject_token_type == TOKEN_TYPE_JWT {
494            return Err(OAuth2Error::InvalidRequest(format!(
495                "subject_token_type {TOKEN_TYPE_JWT} is for tokens from external \
496                 issuers; an AXIAM-issued token must be presented as \
497                 {TOKEN_TYPE_ACCESS_TOKEN}"
498            )));
499        }
500
501        // --- the subject ---------------------------------------------------
502        let subject = decode_access_token(&req.subject_token, &self.auth_config)
503            .map_err(|_| OAuth2Error::InvalidGrant("subject token is not valid".into()))?;
504
505        // No re-exchange of a cross-domain token (X4), on this path too. The
506        // external path refuses the claim on the way in; this refuses it on
507        // the way back round. Without both, trust composes silently: a
508        // partner's token buys an AXIAM token, and that token buys another one
509        // whose provenance nobody can see.
510        if subject.ext_exchange.is_some() {
511            return Err(OAuth2Error::InvalidRequest(
512                "the subject token is already the product of a cross-domain exchange; \
513                 exchanges do not compose"
514                    .into(),
515            ));
516        }
517
518        // Cross-tenant is `invalid_grant`, not a distinct error: a caller
519        // learning that their token is valid SOMEWHERE ELSE is a
520        // tenant-enumeration signal.
521        if subject.tenant_id != tenant_id.to_string() {
522            return Err(OAuth2Error::InvalidGrant(
523                "subject token is not valid".into(),
524            ));
525        }
526
527        // No unbounded re-exchange. A token already carrying a full `act`
528        // chain cannot be exchanged again — otherwise the chain, and the
529        // token, grows one re-exchange at a time.
530        if let Some(act) = subject.act.as_ref()
531            && act.depth() >= MAX_ACT_CHAIN_DEPTH
532        {
533            return Err(OAuth2Error::InvalidRequest(format!(
534                "subject token already names {MAX_ACT_CHAIN_DEPTH} actors; \
535                 it cannot be exchanged again"
536            )));
537        }
538
539        // --- delegation or impersonation ------------------------------------
540        let (kind, actor_sub) = match req.actor_token.as_deref() {
541            Some(actor_token) => {
542                match req.actor_token_type.as_deref() {
543                    Some(t) if t == TOKEN_TYPE_ACCESS_TOKEN => {}
544                    _ => {
545                        return Err(OAuth2Error::InvalidRequest(
546                            "actor_token_type must be supplied and must be an access token".into(),
547                        ));
548                    }
549                }
550                let actor = decode_access_token(actor_token, &self.auth_config)
551                    .map_err(|_| OAuth2Error::InvalidGrant("actor token is not valid".into()))?;
552                if actor.tenant_id != tenant_id.to_string() {
553                    return Err(OAuth2Error::InvalidGrant("actor token is not valid".into()));
554                }
555                (ExchangeKind::Delegation, Some(actor.sub))
556            }
557            None => {
558                // Impersonation. The resulting token is indistinguishable from
559                // one the subject obtained directly, so it is off unless the
560                // client was explicitly granted it — and a client without the
561                // grant is REFUSED rather than silently downgraded to
562                // delegation, which would hand back a token that is not the
563                // one they asked for.
564                if !client_may_impersonate(client) {
565                    return Err(OAuth2Error::UnauthorizedClient(
566                        "impersonation requires the may_impersonate grant; \
567                         supply an actor_token to delegate instead"
568                            .into(),
569                    ));
570                }
571                (ExchangeKind::Impersonation, None)
572            }
573        };
574
575        // --- scopes ----------------------------------------------------------
576        let subject_scopes: Vec<String> = subject
577            .scope
578            .as_deref()
579            .map(|s| s.split_whitespace().map(str::to_owned).collect())
580            .unwrap_or_default();
581        let requested =
582            parse_scopes(req.scope.as_deref()).unwrap_or_else(|| subject_scopes.clone());
583        let granted = narrow_scopes(&requested, &subject_scopes, &client.scopes)?;
584
585        // --- audience ---------------------------------------------------------
586        let target = match (req.audience.as_deref(), req.resource.as_deref()) {
587            (Some(a), Some(r)) if a != r => {
588                return Err(OAuth2Error::InvalidRequest(
589                    "audience and resource disagree; supply one or make them equal".into(),
590                ));
591            }
592            (Some(a), _) => Some(a),
593            (None, Some(r)) => Some(r),
594            (None, None) => None,
595        };
596        let audience = match target {
597            Some(t) => {
598                // An unconstrained `aud` would let a service mint tokens
599                // addressed at systems it has no relationship with — the mesh
600                // equivalent of an open redirect. The client's registered URIs
601                // are its declared relationships.
602                //
603                // SEC-089: the allow-list *is* `client.redirect_uris` — there
604                // is no separate audience field in v1, so adding a redirect
605                // URI also authorises it as a token audience. Documented on
606                // `TokenExchangeRequest::audience` above and in
607                // `docs/api/token-exchange.md#audience`. The same check is
608                // repeated at the X4 external-exchange call site further
609                // down this file; a future `allowed_token_targets` field
610                // must replace both together.
611                if !is_builtin_audience(t) && !client.redirect_uris.iter().any(|u| u == t) {
612                    return Err(OAuth2Error::InvalidTarget(format!(
613                        "'{t}' is not a registered target for this client"
614                    )));
615                }
616                t.to_owned()
617            }
618            // Inherit. `aud` is optional on pre-Phase-4 tokens, which
619            // `decode_access_token` treats as user tokens.
620            None => subject.aud.clone().unwrap_or_else(|| AUD_USER.to_string()),
621        };
622
623        // --- lifetime ----------------------------------------------------------
624        let now = Utc::now().timestamp();
625        let subject_remaining = subject.exp - now;
626        if subject_remaining <= 0 {
627            return Err(OAuth2Error::InvalidGrant(
628                "subject token has expired".into(),
629            ));
630        }
631        // Never outlives its subject. Without this an exchange launders
632        // lifetime: hold a token for thirty seconds, exchange it, hold the
633        // result for the full access-token lifetime.
634        let lifetime = subject_remaining.min(self.max_lifetime_secs);
635        let expires_at = now + lifetime;
636
637        // --- issue --------------------------------------------------------------
638        let tenant = self
639            .tenant_repo
640            .get_by_id(tenant_id)
641            .await
642            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
643
644        let act = actor_sub.as_ref().map(|actor| ActClaim {
645            sub: actor.clone(),
646            act: subject.act.clone().map(Box::new),
647        });
648
649        // The subject kind is carried through UNCHANGED, including when the
650        // target is the machine audience (SEC-088).
651        //
652        // An earlier revision rewrote it to `OAuth2Client` whenever
653        // `audience == AUD_M2M`, while `sub` stayed the subject's own id. That
654        // produced the one combination the extractor contract says cannot
655        // occur — see `AuthenticatedServiceAccount::subject` in
656        // `axiam-api-rest`, which documents `sub_kind = oauth2_client` as
657        // meaning `sub` IS an OAuth2 client id (`oa_…`). A user UUID wearing
658        // that label is a type confusion aimed squarely at whoever writes the
659        // first consumer of that extractor.
660        //
661        // The audience change is the real and intended effect of narrowing a
662        // user token for a machine API. Relabelling the *subject* to match the
663        // audience answers a different question, and answers it wrongly: who
664        // the principal is did not change. A consumer that needs to know the
665        // token arrived by exchange has `act` and the audit record, which the
666        // impersonation path already relies on as its only evidence.
667        let access_token = issue_exchanged_token(
668            &subject.sub,
669            subject.sub_kind,
670            tenant_id,
671            tenant.organization_id,
672            &granted,
673            &self.auth_config,
674            Uuid::new_v4().to_string(),
675            &audience,
676            expires_at,
677            act,
678            // Same-domain: no foreign provenance to record.
679            None,
680            cnf.clone(),
681        )
682        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
683
684        Ok(ExchangeOutcome {
685            response: TokenExchangeResponse {
686                access_token,
687                issued_token_type: TOKEN_TYPE_ACCESS_TOKEN.to_string(),
688                // RFC 9449 §5: a DPoP-bound token is announced as such.
689                // Certificate-bound tokens stay `Bearer` (RFC 8705 leaves the
690                // HTTP scheme alone), so this is `Bearer` for every client
691                // that did not ask for DPoP binding.
692                token_type: crate::dpop::token_type_for(cnf.as_ref()),
693                expires_in: lifetime as u64,
694                scope: Some(granted.join(" ")),
695                // No refresh token, ever. One would let the holder outlive the
696                // subject token indefinitely — the lifetime cap above,
697                // defeated. Re-run the exchange.
698            },
699            kind,
700            subject: subject.sub,
701            actor: actor_sub,
702            granted_scopes: granted,
703            audience,
704            external: None,
705        })
706    }

708    /// X4 — the external branch of [`Self::exchange`].
709    ///
710    /// Separate function, not a `match` arm, because it shares almost nothing
711    /// with the internal path: the subject is resolved by a different
712    /// authority, the scopes come from a different source, and the answer to
713    /// "what may this token do" is decided by the RBAC engine rather than by
714    /// another token. Splicing the two would produce a function where every
715    /// line needs a "…but not on the external path" qualifier, which is the
716    /// shape mistakes hide in.
717    ///
718    /// The **order** of the checks below is the design doc's §"Validation
719    /// pipeline" and is normative.
720    async fn exchange_external(
721        &self,
722        tenant_id: Uuid,
723        client: &OAuth2Client,
724        req: TokenExchangeRequest,
725        cnf: Option<axiam_auth::token::CnfClaim>,
726    ) -> Result<ExchangeOutcome, OAuth2Error> {
727        // Not configured ⇒ the external path does not exist. Deliberately the
728        // same answer as "no provider trusts this issuer": whether a
729        // deployment has X4 wired in at all is not something an authenticated
730        // client needs to distinguish from a provider it did not configure.
731        let (Some(resolver), Some(authority)) = (
732            self.external_resolver.as_ref(),
733            self.scope_authority.as_ref(),
734        ) else {
735            return Err(OAuth2Error::InvalidGrant(ISSUER_NOT_TRUSTED.into()));
736        };
737
738        // Delegation across a trust boundary needs a second trust decision —
739        // "may this actor act for a subject *this IdP* vouched for" — that X4
740        // does not make. Refused before the token is verified so the answer
741        // does not depend on whether the subject token happened to be good.
742        if req.actor_token.is_some() || req.actor_token_type.is_some() {
743            return Err(OAuth2Error::InvalidRequest(
744                "actor_token is not supported for subject tokens from external \
745                 issuers in v1"
746                    .into(),
747            ));
748        }
749
750        let external = resolver
751            .resolve(tenant_id, &req.subject_token)
752            .await
753            .map_err(|e| match e {
754                ExternalSubjectRejection::IssuerNotTrusted => {
755                    OAuth2Error::InvalidGrant(ISSUER_NOT_TRUSTED.into())
756                }
757                // The precise reason goes to the audit record, not to the
758                // wire: which of a dozen checks refused a token is a map of
759                // our validation order, drawn one request at a time.
760                ExternalSubjectRejection::Rejected(_) => {
761                    OAuth2Error::InvalidGrant("subject token is not valid".into())
762                }
763                ExternalSubjectRejection::Server(m) => OAuth2Error::ServerError(m),
764            })?;
765
766        // --- scopes: four gates, and the partner's token is not one of them --
767        //
768        // `candidate_scopes` is what an AXIAM admin agreed this provider's
769        // assertions may map onto. It is not what the user holds, and it is
770        // certainly not what the partner said — deny-by-default mapping has
771        // already dropped everything unmapped.
772        let requested_explicitly = req.scope.is_some();
773        let requested =
774            parse_scopes(req.scope.as_deref()).unwrap_or_else(|| external.candidate_scopes.clone());
775
776        // Gate 1 + 2: the map's output and the exchanging client's ceiling.
777        // Reusing `narrow_scopes` is deliberate — the client-ceiling rule that
778        // stops a low-privilege service minting an admin token is exactly the
779        // same rule here, and a second implementation of it would be a second
780        // thing to keep correct.
781        let mapped_and_registered = if requested_explicitly {
782            narrow_scopes(&requested, &external.candidate_scopes, &client.scopes)?
783        } else {
784            // The default drops rather than refuses — see the design doc. A
785            // provider's `scope_map` is written for a provider, not a user, so
786            // a no-`scope` call that refused on the first name the caller
787            // could not have would fail for everyone but the most privileged
788            // user in the tenant.
789            let kept: Vec<String> = requested
790                .into_iter()
791                .filter(|s| external.candidate_scopes.contains(s) && client.scopes.contains(s))
792                .collect();
793            if kept.is_empty() {
794                return Err(OAuth2Error::InvalidScope(
795                    "no scope this provider maps to is registered for this client".into(),
796                ));
797            }
798            kept
799        };
800
801        // Gate 3: the RBAC engine, at mint time. Deny-override (B1) applies —
802        // see the authority's implementation. This is the check that makes an
803        // external token evidence of authentication rather than a grant of
804        // authorization: without it, an admin's `scope_map` would be an
805        // authority in its own right and every user the provider vouches for
806        // would hold the same scopes.
807        let held = authority
808            .held_scopes(tenant_id, external.user_id, &mapped_and_registered)
809            .await
810            .map_err(OAuth2Error::ServerError)?;
811
812        let granted: Vec<String> = if requested_explicitly {
813            if let Some(missing) = mapped_and_registered.iter().find(|s| !held.contains(s)) {
814                return Err(OAuth2Error::InvalidScope(format!(
815                    "the resolved user does not hold scope '{missing}'"
816                )));
817            }
818            mapped_and_registered
819        } else {
820            mapped_and_registered
821                .into_iter()
822                .filter(|s| held.contains(s))
823                .collect()
824        };
825        if granted.is_empty() {
826            return Err(OAuth2Error::InvalidScope(
827                "the exchange would grant no scopes at all".into(),
828            ));
829        }
830
831        // --- audience --------------------------------------------------------
832        //
833        // B3's rule, with one change that matters: when the request names no
834        // target the issued token gets AXIAM's own user audience. It never
835        // inherits the external token's `aud`, which names the *partner's*
836        // resource server and would be meaningless here — or, worse,
837        // coincidentally meaningful.
838        let target = match (req.audience.as_deref(), req.resource.as_deref()) {
839            (Some(a), Some(r)) if a != r => {
840                return Err(OAuth2Error::InvalidRequest(
841                    "audience and resource disagree; supply one or make them equal".into(),
842                ));
843            }
844            (Some(a), _) => Some(a),
845            (None, Some(r)) => Some(r),
846            (None, None) => None,
847        };
848        let audience = match target {
849            Some(t) => {
850                // SEC-089: same allow-list reuse as the same-domain exchange
851                // path above in this file — `client.redirect_uris` doubles
852                // as the audience allow-list, so adding a redirect URI also
853                // authorises it as a token audience. Documented on
854                // `TokenExchangeRequest::audience` and in
855                // `docs/api/token-exchange.md#audience`. Keep this check and
856                // the same-domain one in sync; a future
857                // `allowed_token_targets` field must replace both together.
858                if !is_builtin_audience(t) && !client.redirect_uris.iter().any(|u| u == t) {
859                    return Err(OAuth2Error::InvalidTarget(format!(
860                        "'{t}' is not a registered target for this client"
861                    )));
862                }
863                t.to_owned()
864            }
865            None => AUD_USER.to_string(),
866        };
867
868        // --- lifetime ---------------------------------------------------------
869        let now = Utc::now().timestamp();
870        let subject_remaining = external.subject_exp - now;
871        if subject_remaining <= 0 {
872            return Err(OAuth2Error::InvalidGrant(
873                "subject token has expired".into(),
874            ));
875        }
876        let mut lifetime = subject_remaining.min(self.max_lifetime_secs);
877        if let Some(provider_max) = external.max_lifetime_secs {
878            lifetime = lifetime.min(provider_max);
879        }
880        if lifetime <= 0 {
881            return Err(OAuth2Error::InvalidGrant(
882                "subject token has expired".into(),
883            ));
884        }
885
886        // --- issue --------------------------------------------------------------
887        let tenant = self
888            .tenant_repo
889            .get_by_id(tenant_id)
890            .await
891            .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
892
893        let access_token = issue_exchanged_token(
894            &external.user_id.to_string(),
895            // The subject is an AXIAM user — resolved through the federation
896            // link, which is what a federated login resolves through too.
897            SubjectKind::User,
898            tenant_id,
899            tenant.organization_id,
900            &granted,
901            &self.auth_config,
902            Uuid::new_v4().to_string(),
903            &audience,
904            now + lifetime,
905            // No `act`: nobody acted *for* the user. The IdP asserted that the
906            // user authenticated, which is a different statement, and the one
907            // `ext_exchange` records.
908            None,
909            Some(ExtExchangeClaim {
910                iss: external.issuer.clone(),
911            }),
912            cnf.clone(),
913        )
914        .map_err(|e| OAuth2Error::ServerError(e.to_string()))?;
915
916        Ok(ExchangeOutcome {
917            response: TokenExchangeResponse {
918                access_token,
919                issued_token_type: TOKEN_TYPE_ACCESS_TOKEN.to_string(),
920                // SEC-096, as on the same-domain path above.
921                token_type: crate::dpop::token_type_for(cnf.as_ref()),
922                expires_in: lifetime as u64,
923                scope: Some(granted.join(" ")),
924            },
925            kind: ExchangeKind::External,
926            subject: external.user_id.to_string(),
927            actor: None,
928            granted_scopes: granted,
929            audience,
930            external: Some(ExternalProvenance {
931                issuer: external.issuer,
932                provider_id: external.provider_id,
933                provider_name: external.provider_name,
934                external_subject: external.external_subject,
935                newly_provisioned: external.newly_provisioned,
936            }),
937        })
938    }
939}

1903        /// Exhaustive over the three gates the request itself does not set:
1904        /// whatever comes out is a subset of every one of them. The single
1905        /// most valuable test in the feature, mirroring B3's own property test
1906        /// one layer up.
1907        #[tokio::test]
1908        async fn granted_is_always_a_subset_of_map_client_and_engine() {
1909            let universe = ["a", "b", "c"];
1910            let subsets: Vec<Vec<String>> = (0u8..8)
1911                .map(|mask| {
1912                    universe
1913                        .iter()
1914                        .enumerate()
1915                        .filter(|(i, _)| mask & (1 << i) != 0)
1916                        .map(|(_, s)| (*s).to_string())
1917                        .collect()
1918                })
1919                .collect();
1920            let cfg = auth_config();
1921
1922            for candidates in &subsets {
1923                for client_scopes in &subsets {
1924                    for held in &subsets {
1925                        let mut subj = subject(&[]);
1926                        subj.candidate_scopes = candidates.clone();
1927                        let svc = service(
1928                            &cfg,
1929                            StubResolver::ok(subj),
1930                            Arc::new(StubAuthority(held.clone())),
1931                        );
1932                        let mut c = client(&[]);
1933                        c.scopes = client_scopes.clone();
1934
1935                        match svc
1936                            .exchange(
1937                                Uuid::new_v4(),
1938                                &c,
1939                                request(foreign_token(EXT_ISS), None),
1940                                None,
1941                            )
1942                            .await
1943                        {
1944                            Ok(outcome) => {
1945                                assert!(!outcome.granted_scopes.is_empty());
1946                                for s in &outcome.granted_scopes {
1947                                    assert!(candidates.contains(s), "escaped the scope map");
1948                                    assert!(client_scopes.contains(s), "escaped the client");
1949                                    assert!(held.contains(s), "escaped the engine");
1950                                }
1951                            }
1952                            Err(e) => assert_eq!(e.error_code(), "invalid_scope"),
1953                        }
1954                    }
1955                }
1956            }
