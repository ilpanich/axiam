1//! AXIAM's OPAQUE (RFC 9807) ciphersuite, key-stretching functions and client
2//! operations.
3//!
4//! # Why this crate exists
5//!
6//! AXIAM speaks OPAQUE from twelve places: the server, the React admin UI, and
7//! eleven client SDKs. Every one of them must agree on every byte, and OPAQUE
8//! is not the kind of protocol that can be hand-written twelve times and be
9//! expected to agree — it needs an oblivious PRF, `hash_to_curve`,
10//! `expand_message_xmd`, an envelope construction and a three-message AKE.
11//!
12//! The SRP implementation this replaces *was* hand-written eleven times,
13//! because SRP is modular arithmetic and every language has a bignum. That
14//! produced a bundled Montgomery modular exponentiation in the Swift SDK, a
15//! PHP SDK that reported itself unavailable without `ext-gmp`, and a
16//! cross-language conformance fixture that existed to catch the drift. None of
17//! that is a criticism of the implementations; it is what shipping eleven
18//! independent implementations of an authentication primitive costs.
19//!
20//! So this crate is the *only* implementation. It is compiled directly into
21//! the Rust SDK and the server, to WebAssembly for the TypeScript SDK and the
22//! admin UI, and behind a C ABI for the rest. That is also why it sits at
23//! layer 0 with no internal dependencies: anything it depended on would become
24//! a dependency of every SDK.
25//!
26//! # What is here, and what is not
27//!
28//! Here: the ciphersuite, the key-stretching functions, and the four
29//! **client**-side operations. The client half is what needs to exist in every
30//! language.
31//!
32//! Not here: the server half — key material, sealed sessions, credential
33//! identifiers, lockout attribution. That lives in `axiam-auth` and has no
34//! business being compiled into a client. The one thing the server borrows
35//! from this crate is [`AxiamOpaqueSuite`], so that there is exactly one
36//! definition of the suite and the two halves cannot drift.
37//!
38//! # Wire encoding
39//!
40//! Every protocol message crosses the AXIAM API as lowercase hex, so this
41//! crate's public surface is hex in and hex out. That is a deliberate
42//! narrowing: a `&str` API is expressible in every SDK language and across a C
43//! ABI without anyone inventing a byte-buffer convention, and hex has no
44//! variant spellings to disagree about the way base64 does.
45//!
46//! # Example
47//!
48//! ```
49//! use axiam_opaque::{AxiamKsf, ClientLoginState, ClientRegistrationState};
50//!
51//! // Whatever the user typed. Read from somewhere rather than written as a
52//! // literal, which is also what a real caller does.
53//! # let password = &format!("example-{}", std::process::id());
54//!
55//! // --- registration ---
56//! let ksf = AxiamKsf::argon2id(19456, 2, 1).unwrap();
57//! let (state, request) = ClientRegistrationState::start(password).unwrap();
58//! // ... server evaluates the OPRF and returns `registration_response` ...
59//! # let (setup, response) = axiam_opaque::testing::server_registration_start(&request);
60//! # let registration_response = response;
61//! let record = state.finish(password, &registration_response, &ksf).unwrap();
62//!
63//! // --- login ---
64//! let (state, ke1) = ClientLoginState::start(password).unwrap();
65//! # let ke2 = axiam_opaque::testing::server_login_start(&setup, &record.record, &ke1);
66//! // ... server returns `ke2` ...
67//! let finished = state.finish(password, &ke2, &ksf).unwrap();
68//! assert_eq!(finished.ke3.len(), 128); // 64 bytes, hex
69//! ```

83/// The `sha2` generation `opaque-ke` 4.0.1 is built against (`digest 0.10`),
84/// aliased so a workspace bump to `digest 0.11` cannot silently retype the
85/// ciphersuite out from under the wire format.
86use sha2_v10 as opaque_sha2;

88/// AXIAM's OPAQUE ciphersuite: `OPAQUE-3DH` over ristretto255 with SHA-512,
89/// HKDF-SHA-512 and HMAC-SHA-512 — the RFC 9807 recommended configuration.
90///
91/// One suite ships today. RFC 9807 also defines a P-256/SHA-256 profile, which
92/// is the slot a FIPS deployment would want; it is not shipped because every
93/// additional suite multiplies the cross-language conformance surface, which is
94/// the hardest part of this to keep correct.
95pub struct AxiamOpaqueSuite;
96
97impl CipherSuite for AxiamOpaqueSuite {
98    type OprfCs = opaque_ke::Ristretto255;
99    type KeyExchange = opaque_ke::TripleDh<opaque_ke::Ristretto255, opaque_sha2::Sha512>;
100    type Ksf = AxiamKsf;
101}

103/// Anything that can go wrong on the client side.
104///
105/// Deliberately coarse. A client that could distinguish "the server sent a
106/// malformed KE2" from "your password is wrong" would be reporting a fact it
107/// cannot actually establish — a wrong password and a hostile server both
108/// surface as an envelope that will not open.
109#[derive(Debug, thiserror::Error)]
110pub enum OpaqueError {
111    /// A hex input was not valid hex, or was not a valid protocol message.
112    #[error("malformed OPAQUE message: {0}")]
113    Malformed(&'static str),
114    /// The exchange failed. For a login this means the password is wrong, the
115    /// account has no record, or the server is not the one that holds it.
116    #[error("OPAQUE exchange failed")]
117    Failed,
118    /// KSF parameters outside the accepted band.
119    #[error("invalid KSF parameters: {0}")]
120    InvalidKsf(&'static str),
121}

123// ---------------------------------------------------------------------------
124// Key-stretching
125// ---------------------------------------------------------------------------

127/// A key-stretching function carrying its own cost parameters (RFC 9807 §3.1,
128/// `Stretch`).
129///
130/// Parameters live *in the value* rather than in the type because a credential
131/// enrolled under one cost must keep working after a tenant raises its policy.
132/// The server stores the parameters a record was created with and echoes them
133/// at login; a client that stretched with anything else derives a different
134/// randomized password and cannot open its own envelope.
135///
136/// Both variants are memory-hard. The SRP scheme this replaces offered
137/// PBKDF2-HMAC-SHA256 as a fallback, purely because three SDK languages had no
138/// usable Argon2id — a constraint that a single shared implementation removes,
139/// so there is no longer a reason to offer a KSF that a GPU farm enjoys.
140#[derive(Debug, Clone, Copy, PartialEq, Eq)]
141pub enum AxiamKsf {
142    /// Argon2id.
143    Argon2id {
144        /// Memory cost in KiB.
145        memory_kib: u32,
146        /// Time cost (number of passes).
147        iterations: u32,
148        /// Degree of parallelism.
149        parallelism: u32,
150    },
151    /// scrypt.
152    Scrypt {
153        /// CPU/memory cost, as log2(N).
154        log_n: u8,
155        /// Block size.
156        r: u32,
157        /// Parallelism.
158        p: u32,
159    },
160}
161
162impl AxiamKsf {
163    /// Argon2id with the given costs, range-checked.
164    ///
165    /// The bounds match `axiam_core::models::opaque::OpaqueKsfParams::validate`.
166    /// They are repeated here rather than shared because this crate is layer 0
167    /// and is compiled into every SDK: a client must be able to reject absurd
168    /// parameters handed to it by a server without linking the domain model.
169    pub fn argon2id(
170        memory_kib: u32,
171        iterations: u32,
172        parallelism: u32,
173    ) -> Result<Self, OpaqueError> {
174        if !(8192..=1_048_576).contains(&memory_kib) {
175            return Err(OpaqueError::InvalidKsf("argon2id memory_kib out of range"));
176        }
177        if !(1..=10).contains(&iterations) {
178            return Err(OpaqueError::InvalidKsf("argon2id iterations out of range"));
179        }
180        if !(1..=16).contains(&parallelism) {
181            return Err(OpaqueError::InvalidKsf("argon2id parallelism out of range"));
182        }
183        Ok(Self::Argon2id {
184            memory_kib,
185            iterations,
186            parallelism,
187        })
188    }
189
190    /// scrypt with the given costs, range-checked.
191    pub fn scrypt(log_n: u8, r: u32, p: u32) -> Result<Self, OpaqueError> {
192        if !(14..=20).contains(&log_n) {
193            return Err(OpaqueError::InvalidKsf("scrypt log_n out of range"));
194        }
195        if !(1..=16).contains(&r) {
196            return Err(OpaqueError::InvalidKsf("scrypt r out of range"));
197        }
198        if !(1..=16).contains(&p) {
199            return Err(OpaqueError::InvalidKsf("scrypt p out of range"));
200        }
201        Ok(Self::Scrypt { log_n, r, p })
202    }
203}
204
205impl Default for AxiamKsf {
206    /// The OWASP-aligned Argon2id defaults, matching AXIAM's server-side
207    /// password hashing cost.
208    ///
209    /// `Default` exists only because `opaque_ke::ksf::Ksf` requires it. Nothing
210    /// in AXIAM constructs a KSF this way — the parameters always come from the
211    /// server, per credential — and a client that silently fell back to a
212    /// default would derive a randomized password no record agrees with.
213    fn default() -> Self {
214        Self::Argon2id {
215            memory_kib: 19456,
216            iterations: 2,
217            parallelism: 1,
218        }
219    }
220}
221
222impl Ksf for AxiamKsf {
223    fn hash<L: ArrayLength<u8>>(
224        &self,
225        input: GenericArray<u8, L>,
226    ) -> Result<GenericArray<u8, L>, opaque_ke::errors::InternalError> {
227        let mut output = GenericArray::default();
228        match *self {
229            Self::Argon2id {
230                memory_kib,
231                iterations,
232                parallelism,
233            } => {
234                let params = argon2::Params::new(memory_kib, iterations, parallelism, None)
235                    .map_err(|_| opaque_ke::errors::InternalError::KsfError)?;
236                let argon2 = argon2::Argon2::new(
237                    argon2::Algorithm::Argon2id,
238                    argon2::Version::V0x13,
239                    params,
240                );
241                // RFC 9807 stretches a value that is already the output of an
242                // OPRF over the password, so it carries full entropy from the
243                // KSF's point of view and there is nothing for a salt to
244                // separate. A fixed all-zero salt is what the reference
245                // implementation uses; anything per-client here would be a
246                // value both sides would have to agree on for no benefit.
247                argon2
248                    .hash_password_into(&input, &[0u8; argon2::RECOMMENDED_SALT_LEN], &mut output)
249                    .map_err(|_| opaque_ke::errors::InternalError::KsfError)?;
250            }
251            Self::Scrypt { log_n, r, p } => {
252                let params = scrypt::Params::new(log_n, r, p, output.len())
253                    .map_err(|_| opaque_ke::errors::InternalError::KsfError)?;
254                scrypt::scrypt(&input, &[0u8; 32], &params, &mut output)
255                    .map_err(|_| opaque_ke::errors::InternalError::KsfError)?;
256            }
257        }
258        Ok(output)
259    }
260}

266/// A finished registration, ready to post to the server.
267#[derive(Debug, Clone)]
268pub struct RegistrationOutcome {
269    /// Lowercase-hex serialized RFC 9807 `RegistrationRecord`.
270    pub record: String,
271    /// The client's `export_key`, lowercase hex.
272    ///
273    /// RFC 9807 yields this as a by-product: a 64-byte key derived from the
274    /// password that the server never learns and cannot derive. AXIAM does not
275    /// use it, and no AXIAM endpoint accepts it. It is surfaced because
276    /// discarding it silently would remove an application's ability to build
277    /// end-to-end encryption on top of a login it already performs, and
278    /// re-deriving it later is impossible without the password.
279    pub export_key: String,
280}
281
282/// Client state between the two messages of a registration.
283///
284/// Not `Clone`, and consumed by [`Self::finish`]: the OPRF blind inside it is
285/// good for exactly one exchange.
286pub struct ClientRegistrationState {
287    inner: ClientRegistration<AxiamOpaqueSuite>,
288}
289
290impl ClientRegistrationState {
291    /// Blind the password. Returns the state and the lowercase-hex
292    /// `RegistrationRequest` to send to `/auth/opaque/register/start`.
293    pub fn start(password: &str) -> Result<(Self, String), OpaqueError> {
294        let mut rng = OsRng;
295        let started = ClientRegistration::<AxiamOpaqueSuite>::start(&mut rng, password.as_bytes())
296            .map_err(|_| OpaqueError::Failed)?;
297        Ok((
298            Self {
299                inner: started.state,
300            },
301            hex::encode(started.message.serialize()),
302        ))
303    }
304
305    /// Unblind the server's evaluation, stretch it, and seal the envelope.
306    ///
307    /// `password` must be the same string passed to [`Self::start`], and `ksf`
308    /// must carry the parameters the server named in its response.
309    pub fn finish(
310        self,
311        password: &str,
312        registration_response_hex: &str,
313        ksf: &AxiamKsf,
314    ) -> Result<RegistrationOutcome, OpaqueError> {
315        let bytes = hex::decode(registration_response_hex)
316            .map_err(|_| OpaqueError::Malformed("registration_response is not hex"))?;
317        let response = RegistrationResponse::<AxiamOpaqueSuite>::deserialize(&bytes)
318            .map_err(|_| OpaqueError::Malformed("registration_response"))?;
319
320        let mut rng = OsRng;
321        let finished = self
322            .inner
323            .finish(
324                &mut rng,
325                password.as_bytes(),
326                response,
327                ClientRegistrationFinishParameters::new(Default::default(), Some(ksf)),
328            )
329            .map_err(|_| OpaqueError::Failed)?;
330
331        Ok(RegistrationOutcome {
332            record: hex::encode(finished.message.serialize()),
333            export_key: hex::encode(finished.export_key),
334        })
335    }
336}

342/// A finished login, ready to post to the server.
343#[derive(Debug, Clone)]
344pub struct LoginOutcome {
345    /// Lowercase-hex serialized RFC 9807 `KE3`.
346    pub ke3: String,
347    /// The mutually authenticated session key, lowercase hex.
348    ///
349    /// Both sides derive it; neither transmits it. AXIAM issues ordinary
350    /// session cookies rather than binding anything to this, so no SDK is
351    /// required to do anything with it. Surfaced for the same reason as
352    /// [`RegistrationOutcome::export_key`].
353    pub session_key: String,
354    /// The client's `export_key`, lowercase hex. Identical to the value
355    /// produced at registration for the same password.
356    pub export_key: String,
357}
358
359/// Client state between the two messages of a login.
360pub struct ClientLoginState {
361    inner: ClientLogin<AxiamOpaqueSuite>,
362}
363
364impl ClientLoginState {
365    /// Blind the password and generate the client's ephemeral share. Returns
366    /// the state and the lowercase-hex `KE1` to send to
367    /// `/auth/opaque/login/start`.
368    pub fn start(password: &str) -> Result<(Self, String), OpaqueError> {
369        let mut rng = OsRng;
370        let started = ClientLogin::<AxiamOpaqueSuite>::start(&mut rng, password.as_bytes())
371            .map_err(|_| OpaqueError::Failed)?;
372        Ok((
373            Self {
374                inner: started.state,
375            },
376            hex::encode(started.message.serialize()),
377        ))
378    }
379
380    /// Open the envelope and produce `KE3`.
381    ///
382    /// An `Err` here is the *whole* of the client's authentication check, and
383    /// it covers both halves of the mutual authentication: the envelope only
384    /// opens under the right password, and `KE2`'s MAC only verifies if the
385    /// server actually holds the record. That is why the AXIAM login response
386    /// carries no server proof for the client to check afterwards — under SRP
387    /// it had to, and an SDK that forgot silently kept only half the protocol.
388    pub fn finish(
389        self,
390        password: &str,
391        ke2_hex: &str,
392        ksf: &AxiamKsf,
393    ) -> Result<LoginOutcome, OpaqueError> {
394        let bytes = hex::decode(ke2_hex).map_err(|_| OpaqueError::Malformed("ke2 is not hex"))?;
395        let ke2 = CredentialResponse::<AxiamOpaqueSuite>::deserialize(&bytes)
396            .map_err(|_| OpaqueError::Malformed("ke2"))?;
397
398        let mut rng = OsRng;
399        let finished = self
400            .inner
401            .finish(
402                &mut rng,
403                password.as_bytes(),
404                ke2,
405                ClientLoginFinishParameters::new(None, Default::default(), Some(ksf)),
406            )
407            .map_err(|_| OpaqueError::Failed)?;
408
409        Ok(LoginOutcome {
410            ke3: hex::encode(finished.message.serialize()),
411            session_key: hex::encode(finished.session_key),
412            export_key: hex::encode(finished.export_key),
413        })
414    }
415}

421/// Minimal server-side operations, for doctests and for `axiam-auth`'s and the
422/// SDKs' conformance fixtures.
423///
424/// **Not the AXIAM server implementation.** That lives in `axiam-auth` and adds
425/// everything that actually matters operationally: per-tenant key material
426/// encrypted at rest, sealed sessions, credential identifiers, decoy exchanges
427/// for identities that do not exist, and lockout attribution. What is here is
428/// the bare protocol, so that a test can complete an exchange without standing
429/// up a server.
430pub mod testing {
431    use super::*;
432    use opaque_ke::{
433        CredentialRequest, RegistrationRequest, RegistrationUpload, ServerLogin,
434        ServerLoginParameters, ServerRegistration, ServerSetup,
435    };
436
437    /// A throwaway server setup, hex-encoded.
438    pub type SetupHex = String;
439
440    /// Evaluate the OPRF over a client's blinded password.
441    ///
442    /// Returns `(setup, registration_response)`, both lowercase hex.
443    pub fn server_registration_start(request_hex: &str) -> (SetupHex, String) {
444        let mut rng = OsRng;
445        let setup = ServerSetup::<AxiamOpaqueSuite>::new(&mut rng);
446        let request = RegistrationRequest::<AxiamOpaqueSuite>::deserialize(
447            &hex::decode(request_hex).unwrap(),
448        )
449        .unwrap();
450        let started =
451            ServerRegistration::<AxiamOpaqueSuite>::start(&setup, request, b"test-credential")
452                .unwrap();
453        (
454            hex::encode(setup.serialize()),
455            hex::encode(started.message.serialize()),
456        )
457    }
458
459    /// Produce a `KE2` for a stored record. Returns lowercase hex.
460    pub fn server_login_start(setup_hex: &str, record_hex: &str, ke1_hex: &str) -> String {
461        let mut rng = OsRng;
462        let setup =
463            ServerSetup::<AxiamOpaqueSuite>::deserialize(&hex::decode(setup_hex).unwrap()).unwrap();
464        let record = ServerRegistration::<AxiamOpaqueSuite>::finish(
465            RegistrationUpload::deserialize(&hex::decode(record_hex).unwrap()).unwrap(),
466        );
467        let ke1 =
468            CredentialRequest::<AxiamOpaqueSuite>::deserialize(&hex::decode(ke1_hex).unwrap())
469                .unwrap();
470        let started = ServerLogin::<AxiamOpaqueSuite>::start(
471            &mut rng,
472            &setup,
473            Some(record),
474            ke1,
475            b"test-credential",
476            ServerLoginParameters::default(),
477        )
478        .unwrap();
479        hex::encode(started.message.serialize())
480    }
