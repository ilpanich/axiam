1//! The connector address guard: where a server an administrator names may be,
2//! and the one address a connection to it may use (T23.3.7, D-19, D-32;
3//! T-300, T-473).
4//!
5//! Two outbound connectors open raw TCP connections to a host an administrator
6//! chooses: the directory connector (`axiam-directory`, LDAP) and the SMTP
7//! email provider (`axiam-email`). Unchecked, either turns a configuration
8//! field into a connection AXIAM opens to an address of the administrator's
9//! choosing — its own loopback services, the cloud metadata endpoint, the pod
10//! network — and the way a failure looks tells an open port from a closed one.
11//! Both hold the host to this one rule; each keeps its own allow-list
12//! variable, because the networks a directory lives in and the networks a mail
13//! relay lives in are different decisions.
14//!
15//! # The rule ([`AddressPolicy::check`])
16//!
17//! The host is resolved (an IP-literal host is used as written) and **every**
18//! address it resolves to must pass, or the host is refused:
19//!
20//! | Class ([`axiam_core::ip_class`]) | Answer |
21//! |---|---|
22//! | global unicast | admitted |
23//! | loopback, unspecified, link-local (incl. `169.254.169.254`, `fe80::/10`), multicast, special-purpose | **refused, always** |
24//! | private — RFC 1918, CGNAT `100.64/10`, ULA `fc00::/7` | refused **unless** inside a network the operator listed for the connector |
25//! | a cloud metadata endpoint inside a private range (`fd00:ec2::254`, `100.100.100.200`) | refused, whatever the list says |
26//! | an address of this host on a port AXIAM itself listens on | refused |
27//!
28//! IPv4-mapped IPv6 addresses are judged as the IPv4 address they denote. The
29//! classification is the one [`crate::ssrf::guarded_fetch`] uses; only the
30//! policy differs — directories and mail relays live on private networks, so
31//! private ranges are admissible here where they are not for an outbound HTTP
32//! fetch, and only by the operator.
33//!
34//! # Pinning
35//!
36//! [`guard_host`] returns the vetted addresses, and the connector opens its TCP
37//! socket to one of **those** `SocketAddr`s; nothing resolves the name a second
38//! time. The TLS server name stays the configured host, so certificate
39//! verification is unchanged. DNS rebinding between the check and the connect
40//! is therefore impossible within one connection, and because the guard runs
41//! again at **every** connect, a name that is re-pointed after the
42//! configuration was saved is caught at the next connection.
43//!
44//! # The allow-list is the operator's, and is a residual
45//!
46//! It is deployment configuration, never tenant-editable. Inside the networks
47//! it lists, any administrator can point the connector at any host and port;
48//! what that buys is a TLS handshake (or a plaintext protocol greeting before
49//! one, for StartTLS) against a host that must then present a certificate the
50//! connector accepts before any credential is sent. Keep it to the networks
51//! the servers actually live in (decision D-32 says why it is a network list,
52//! not a host list as `AXIAM__PKI__SSRF_ALLOWED_HOSTS` is).
53//!
54//! Moved here from `axiam_directory::address` by #529 so the directory and the
55//! email provider share one implementation rather than two copies; this crate
56//! is the lowest both may depend on that already carries a runtime
57//! (`axiam-core` deliberately has none).

66/// Most addresses one host may resolve to before the guard stops looking: a
67/// directory or mail relay name is a handful of servers, not a CDN.
68pub const MAX_RESOLVED_ADDRESSES: usize = 32;

70/// Why an address, or a whole host, was refused.
71#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
72pub enum GuardError {
73    /// The configured address could not be parsed or names no host.
74    #[error("the address names no usable host")]
75    InvalidUrl,
76    /// The host is a bracketed IPv6 literal in a URL, which the directory
77    /// connector cannot name-check.
78    #[error("an IPv6-literal URL cannot be certificate-checked; name the server by a host name")]
79    Ipv6Literal,
80    /// The name did not resolve, or resolved to nothing.
81    #[error("the host could not be resolved")]
82    Unresolvable,
83    /// The name resolved to more addresses than [`MAX_RESOLVED_ADDRESSES`].
84    #[error("the host resolves to too many addresses")]
85    TooManyAddresses,
86    /// An address is loopback.
87    #[error("the host resolves to a loopback address")]
88    Loopback,
89    /// An address is unspecified (`0.0.0.0/8`, `::`).
90    #[error("the host resolves to an unspecified address")]
91    Unspecified,
92    /// An address is link-local (`169.254/16`, `fe80::/10`).
93    #[error("the host resolves to a link-local address")]
94    LinkLocal,
95    /// An address is multicast.
96    #[error("the host resolves to a multicast address")]
97    Multicast,
98    /// An address is in a special-purpose block nothing legitimate lives in.
99    #[error("the host resolves to a special-purpose address")]
100    SpecialPurpose,
101    /// An address is a cloud metadata endpoint inside a private range.
102    #[error("the host resolves to a cloud metadata address")]
103    Metadata,
104    /// An address is private and outside every network the operator listed.
105    #[error("the host resolves to a private address outside the deployment's allowed networks")]
106    PrivateNotAllowed,
107    /// An address is this host's own, on a port AXIAM listens on.
108    #[error("the host resolves to one of AXIAM's own listeners")]
109    OwnListener,
110}
111
112impl GuardError {
113    /// The machine-readable name of the rule, for an audit row or a log field.
114    #[must_use]
115    pub const fn rule(&self) -> &'static str {
116        match self {
117            Self::InvalidUrl => "address_guard.invalid_url",
118            Self::Ipv6Literal => "address_guard.ipv6_literal",
119            Self::Unresolvable => "address_guard.unresolvable",
120            Self::TooManyAddresses => "address_guard.too_many_addresses",
121            Self::Loopback => "address_guard.loopback",
122            Self::Unspecified => "address_guard.unspecified",
123            Self::LinkLocal => "address_guard.link_local",
124            Self::Multicast => "address_guard.multicast",
125            Self::SpecialPurpose => "address_guard.special_purpose",
126            Self::Metadata => "address_guard.metadata",
127            Self::PrivateNotAllowed => "address_guard.private_not_allowed",
128            Self::OwnListener => "address_guard.own_listener",
129        }
130    }
131
132    /// Whether the refusal is the configuration's fault (`true`) rather than a
133    /// resolver that could not answer (`false`, worth a retry later).
134    #[must_use]
135    pub const fn is_policy(&self) -> bool {
136        !matches!(self, Self::Unresolvable)
137    }
138}

140/// A deployment's rule for one connector's addresses. Built once at
141/// composition from deployment configuration; tenants cannot change it.
142#[derive(Debug, Clone, Default)]
143pub struct AddressPolicy {
144    allowed_private: Vec<IpNetwork>,
145    listener_ports: Vec<u16>,
146    admit_loopback: bool,
147}
148
149impl AddressPolicy {
150    /// The strict policy: no private network admitted, no listener known.
151    #[must_use]
152    pub fn new() -> Self {
153        Self::default()
154    }
155
156    /// Admit private addresses inside `networks` (the operator's allow-list
157    /// for this connector). Only addresses of the private class ever consult
158    /// the list: a network that also covers loopback, link-local or public
159    /// space admits nothing more than its private part, and never a metadata
160    /// endpoint.
161    #[must_use]
162    pub fn with_allowed_private_networks(mut self, networks: Vec<IpNetwork>) -> Self {
163        self.allowed_private = networks;
164        self
165    }
166
167    /// The ports AXIAM's own listeners are bound to (REST, gRPC). An address
168    /// of this host on one of them is refused, so a connector cannot be
169    /// pointed back at AXIAM itself.
170    #[must_use]
171    pub fn with_listener_ports(mut self, ports: impl IntoIterator<Item = u16>) -> Self {
172        self.listener_ports = ports.into_iter().collect();
173        self.listener_ports.sort_unstable();
174        self.listener_ports.dedup();
175        self
176    }
177
178    /// **Tests only.** Admit loopback addresses, so an in-process test server
179    /// on `127.0.0.1` can be reached; every other rule still applies
180    /// (link-local, metadata, private ranges, listener ports). Never called by
181    /// production code: the composition root builds its policies with
182    /// [`Self::new`], and a grep for this name finds every exception.
183    #[doc(hidden)]
184    #[must_use]
185    pub fn admitting_loopback_for_tests(mut self) -> Self {
186        self.admit_loopback = true;
187        self
188    }
189
190    /// The networks admitted.
191    #[must_use]
192    pub fn allowed_private_networks(&self) -> &[IpNetwork] {
193        &self.allowed_private
194    }
195
196    /// The listener ports refused on this host's addresses.
197    #[must_use]
198    pub fn listener_ports(&self) -> &[u16] {
199        &self.listener_ports
200    }
201
202    /// Judge one resolved address.
203    ///
204    /// # Errors
205    ///
206    /// The [`GuardError`] naming the rule `address` breaks.
207    pub fn check(&self, address: SocketAddr) -> Result<(), GuardError> {
208        let ip = canonical(address.ip());
209        match classify(ip) {
210            IpClass::Global => {}
211            IpClass::Loopback if self.admit_loopback => {}
212            IpClass::Loopback => return Err(GuardError::Loopback),
213            IpClass::Unspecified => return Err(GuardError::Unspecified),
214            IpClass::LinkLocal => return Err(GuardError::LinkLocal),
215            IpClass::Multicast => return Err(GuardError::Multicast),
216            IpClass::SpecialPurpose => return Err(GuardError::SpecialPurpose),
217            IpClass::Private => {
218                if is_never_allowed(ip) {
219                    return Err(GuardError::Metadata);
220                }
221                if !self.allowed_private.iter().any(|net| net.contains(ip)) {
222                    return Err(GuardError::PrivateNotAllowed);
223                }
224            }
225        }
226        if self.listener_ports.contains(&address.port()) && is_local_address(ip) {
227            return Err(GuardError::OwnListener);
228        }
229        Ok(())
230    }
231}

233/// Whether `ip` is an address of this host: loopback, or one an unprivileged
234/// socket can bind to (binding fails with `EADDRNOTAVAIL` for an address no
235/// local interface holds). No packet is sent. A host configured for non-local
236/// binds (`net.ipv4.ip_nonlocal_bind`) reads every address as local, which
237/// only widens the listener-port refusal — the safe direction.
238fn is_local_address(ip: IpAddr) -> bool {
239    ip.is_loopback() || std::net::UdpSocket::bind(SocketAddr::new(ip, 0)).is_ok()
240}

242/// Parse an allow-list variable's comma-separated form. Returns the blocks
243/// that parsed and the entries that did not (for the startup log; an entry
244/// that does not parse admits nothing, so a typo fails closed).
245#[must_use]
246pub fn parse_allowed_networks(raw: &str) -> (Vec<IpNetwork>, Vec<String>) {
247    let mut networks = Vec::new();
248    let mut rejected = Vec::new();
249    for entry in raw.split(',').map(str::trim).filter(|e| !e.is_empty()) {
250        match entry.parse::<IpNetwork>() {
251            Ok(network) if !networks.contains(&network) => networks.push(network),
252            Ok(_) => {}
253            Err(_) => rejected.push(entry.to_string()),
254        }
255    }
256    (networks, rejected)
257}

259/// A boxed resolution future.
260pub type ResolveFuture<'a> = Pin<Box<dyn Future<Output = io::Result<Vec<SocketAddr>>> + Send + 'a>>;
261
262/// Turns a host name into addresses. The production resolver is
263/// [`SystemResolver`]; tests script answers (including a name that answers
264/// differently each time, to prove a connector never resolves twice).
265pub trait Resolver: Send + Sync {
266    /// Resolve `host` (a domain name, never an IP literal), with `port` on
267    /// every address.
268    fn resolve<'a>(&'a self, host: &'a str, port: u16) -> ResolveFuture<'a>;
269}
270
271/// The operating system's resolver (`getaddrinfo`, through Tokio).
272#[derive(Debug, Clone, Copy, Default)]
273pub struct SystemResolver;
274
275impl Resolver for SystemResolver {
276    fn resolve<'a>(&'a self, host: &'a str, port: u16) -> ResolveFuture<'a> {
277        Box::pin(async move { Ok(tokio::net::lookup_host((host, port)).await?.collect()) })
278    }
279}

281/// A host that passed the guard: the name the certificate must carry and the
282/// addresses a connection may use.
283#[derive(Debug, Clone, PartialEq, Eq)]
284pub struct GuardedTarget {
285    /// The configured host, as the TLS server name.
286    pub host: String,
287    /// The port.
288    pub port: u16,
289    /// Every address the host resolved to, each one vetted. A connection uses
290    /// one of these and nothing else.
291    pub addresses: Vec<SocketAddr>,
292}
293
294/// Whether `host` is an IP literal (a bracketed IPv6 literal included), so a
295/// refusal that names the address's class reveals nothing the administrator
296/// did not type (F4 P23W3-04).
297#[must_use]
298pub fn is_ip_literal(host: &str) -> bool {
299    host.trim()
300        .trim_start_matches('[')
301        .trim_end_matches(']')
302        .parse::<IpAddr>()
303        .is_ok()
304}
305
306/// Resolve `host` once and judge every address under `policy`. Pure policy
307/// plus resolution: no protocol traffic, no socket opened. An IP-literal host
308/// (either family, bracketed or not) is used as written and never handed to
309/// the resolver.
310///
311/// # Errors
312///
313/// A [`GuardError`]: the host is empty, the name does not resolve, or any
314/// address it resolves to is refused.
315pub async fn guard_host(
316    host: &str,
317    port: u16,
318    policy: &AddressPolicy,
319    resolver: &dyn Resolver,
320) -> Result<GuardedTarget, GuardError> {
321    let host = host.trim();
322    if host.is_empty() {
323        return Err(GuardError::InvalidUrl);
324    }
325    let literal = host.trim_start_matches('[').trim_end_matches(']');
326    let addresses = match literal.parse::<IpAddr>() {
327        Ok(ip) => vec![SocketAddr::new(ip, port)],
328        Err(_) => resolver
329            .resolve(host, port)
330            .await
331            .map_err(|_| GuardError::Unresolvable)?,
332    };
333    if addresses.is_empty() {
334        return Err(GuardError::Unresolvable);
335    }
336    if addresses.len() > MAX_RESOLVED_ADDRESSES {
337        return Err(GuardError::TooManyAddresses);
338    }
339    let mut vetted = Vec::with_capacity(addresses.len());
340    for address in addresses {
341        policy.check(address)?;
342        // Pin the canonical form: `::ffff:a.b.c.d` is dialled as `a.b.c.d`,
343        // the address that was judged.
344        let pinned = SocketAddr::new(canonical(address.ip()), port);
345        if !vetted.contains(&pinned) {
346            vetted.push(pinned);
347        }
348    }
349    Ok(GuardedTarget {
350        host: host.to_string(),
351        port,
352        addresses: vetted,
353    })
354}

361    /// Answers each name from a table; counts the questions.
362    struct Table {
363        answers: Vec<(&'static str, Vec<IpAddr>)>,
364        asked: Mutex<usize>,
365    }
366
367    impl Resolver for Table {
368        fn resolve<'a>(&'a self, host: &'a str, port: u16) -> ResolveFuture<'a> {
369            *self.asked.lock().unwrap() += 1;
370            let found = self
371                .answers
372                .iter()
373                .find(|(name, _)| *name == host)
374                .map(|(_, ips)| ips.iter().map(|ip| SocketAddr::new(*ip, port)).collect());
375            Box::pin(async move {
376                found.ok_or_else(|| io::Error::new(io::ErrorKind::NotFound, "no such name"))
377            })
378        }
379    }
380
381    fn table(answers: Vec<(&'static str, Vec<&str>)>) -> Table {
382        Table {
383            answers: answers
384                .into_iter()
385                .map(|(name, ips)| (name, ips.iter().map(|ip| ip.parse().unwrap()).collect()))
386                .collect(),
387            asked: Mutex::new(0),
388        }
389    }
390
391    async fn verdict(
392        host: &str,
393        port: u16,
394        policy: &AddressPolicy,
395        resolver: &Table,
396    ) -> Result<(), GuardError> {
397        guard_host(host, port, policy, resolver).await.map(|_| ())
398    }
