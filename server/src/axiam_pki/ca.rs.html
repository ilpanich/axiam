42/// Maximum validity for CA certificates: 20 years (7300 days).
43///
44/// Aligns with NIST SP 800-57 Part 1 Rev 5 recommendations for root CA
45/// certificate lifetimes. Values above this are rejected to prevent
46/// chrono/time overflow and to enforce security best practice.
47pub const MAX_CA_VALIDITY_DAYS: u32 = 7300;

104    /// Generate a new CA certificate.
105    ///
106    /// Two shapes, chosen by the configured custodian rather than by the
107    /// caller, because the difference is a property of where keys live and not
108    /// of what was asked for:
109    ///
110    /// * **AXIAM generates it.** A self-signed root, whose private key is
111    ///   handed to the custodian and returned to the caller once. What every
112    ///   custodian but one does.
113    /// * **The custodian generates it.** Under `vault_pki`, Vault creates a
114    ///   root and a signing intermediate beneath it, and there is no private
115    ///   key to return because none ever existed here. [`GeneratedCaCertificate`]
116    ///   omits the field rather than sending an empty one.
117    pub async fn generate(
118        &self,
119        mut input: CreateCaCertificate,
120    ) -> AxiamResult<GeneratedCaCertificate> {
121        // DF-023. `subject` is a common name, not a distinguished name.
122        // Normalised here, at the top, rather than where the DN is built:
123        // this one value reaches the certificate, the `subject` column, and —
124        // through the default below — the intermediate's subject too, and the
125        // three have to agree. See [`crate::subject_common_name`].
126        input.subject = subject_common_name(&input.subject)?;
127        if let Some(intermediate) = input.intermediate_subject.take() {
128            input.intermediate_subject = Some(subject_common_name(&intermediate)?);
129        }
130
131        if input.validity_days == 0 || input.validity_days > MAX_CA_VALIDITY_DAYS {
132            return Err(AxiamError::Validation {
133                message: format!(
134                    "validity_days must be between 1 and {MAX_CA_VALIDITY_DAYS} \
135                     (NIST SP 800-57 max for CA certificates)"
136                ),
137            });
138        }
139        if let Some(days) = input.intermediate_validity_days
140            && (days == 0 || days > MAX_CA_VALIDITY_DAYS)
141        {
142            return Err(AxiamError::Validation {
143                message: format!(
144                    "intermediate_validity_days must be between 1 and {MAX_CA_VALIDITY_DAYS} \
145                     (NIST SP 800-57 max for CA certificates)"
146                ),
147            });
148        }
149
150        // The id is minted here rather than by the repository because a
151        // custodian outside the database addresses the key *by CA id*: the id
152        // has to exist before the key is stored, and the key has to be stored
153        // before the row. A custodian that refuses therefore stops the CA
154        // existing at all, rather than leaving a certificate whose key is
155        // nowhere and whose failure surfaces at the first issuance.
156        let ca_id = Uuid::new_v4();
157        let store = self.custodians.default_store()?;
158
159        if store.generates_cas() {
160            return self.generate_in_custodian(ca_id, input).await;
161        }
162
163        // CPU-bound: key generation + self-signing run in spawn_blocking behind semaphore (CQ-B02).
164        let _permit =
165            crate::permit::bounded_permit(&self.crypto_semaphore, self.permit_wait).await?;
166
167        let now = Utc::now();
168        let not_before = now;
169        let not_after = now
170            .checked_add_signed(Duration::days(i64::from(input.validity_days)))
171            .ok_or_else(|| AxiamError::Validation {
172                message: "validity_days produces a date out of range".into(),
173            })?;
174
175        let key_algorithm = input.key_algorithm.clone();
176        let subject = input.subject.clone();
177        let not_before_ts = not_before.timestamp();
178        let not_after_ts = not_after.timestamp();
179
180        let (private_key_pem, public_cert_pem, fingerprint) =
181            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String, String)> {
182                let key_pair = generate_keypair(&key_algorithm)?;
183                let private_key_pem = private_key_pem(&key_pair)?;
184
185                let mut params = CertificateParams::new(Vec::<String>::new())
186                    .map_err(|e| AxiamError::Certificate(e.to_string()))?;
187                params.distinguished_name.push(DnType::CommonName, &subject);
188                params.is_ca = IsCa::Ca(rcgen::BasicConstraints::Unconstrained);
189                // #653: rcgen's default is a hash of the public key.
190                params.serial_number = Some(crate::serial::random_serial());
191                params.not_before = time::OffsetDateTime::from_unix_timestamp(not_before_ts)
192                    .expect("valid timestamp");
193                params.not_after = time::OffsetDateTime::from_unix_timestamp(not_after_ts)
194                    .expect("valid timestamp");
195                let cert = params
196                    .self_signed(&key_pair)
197                    .map_err(|e| AxiamError::Certificate(e.to_string()))?;
198
199                let public_cert_pem = cert.pem();
200                let fingerprint = compute_fingerprint(cert.der());
201                Ok((private_key_pem, public_cert_pem, fingerprint))
202            })
203            .await
204            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
205
206        let stored = store
207            .store(input.organization_id, ca_id, &private_key_pem)
208            .await?;
209        let (encrypted_private_key, key_locator) = split_stored(stored);
210
211        let record = StoreCaCertificate {
212            id: ca_id,
213            organization_id: input.organization_id,
214            // An organization-level CA: the anchor, not something under a
215            // tenant, and with no parent inside AXIAM.
216            tenant_id: None,
217            parent_ca_id: None,
218            subject: input.subject,
219            public_cert_pem,
220            // A self-signed root is its own chain.
221            chain_pem: None,
222            fingerprint,
223            key_algorithm: input.key_algorithm,
224            not_before,
225            not_after,
226            encrypted_private_key,
227            key_custody: store.custody(),
228            key_locator,
229        };
230
231        let certificate = self.repo.create(record).await?;
232
233        Ok(GeneratedCaCertificate {
234            certificate,
235            private_key_pem: Some(private_key_pem),
236        })
237    }

239    /// The `vault_pki` path: Vault makes the key, the root and the intermediate.
240    ///
241    /// Nothing here is CPU-bound — no key is generated in this process — so it
242    /// deliberately does not take the crypto semaphore. Holding it across a
243    /// network round trip to Vault would let one slow CA generation block every
244    /// certificate issuance in the deployment, which is the opposite of what
245    /// the semaphore is for.
246    ///
247    /// Everything the row records is parsed back out of the certificate Vault
248    /// returned rather than taken from the request, for the same reason
249    /// [`Self::import`] does it: Vault caps a TTL to the mount's
250    /// `max_lease_ttl` and to the signing root's own expiry without failing the
251    /// call, so a row built from the request would claim a validity window the
252    /// certificate does not have.
253    async fn generate_in_custodian(
254        &self,
255        ca_id: Uuid,
256        input: CreateCaCertificate,
257    ) -> AxiamResult<GeneratedCaCertificate> {
258        let store = self.custodians.default_store()?;
259        let intermediate = if input.issue_from_root {
260            None
261        } else {
262            let subject = match &input.intermediate_subject {
263                // Already through `subject_common_name` in `generate`.
264                Some(given) => given.clone(),
265                None => {
266                    // #649 — the one name AXIAM composes itself. The root's
267                    // subject passed the 64-character bound; the root's subject
268                    // plus this suffix might not, and would otherwise reach the
269                    // custodian (and the row) as an over-long common name.
270                    let derived = format!("{} Intermediate Authority", input.subject);
271                    if derived.chars().count() > crate::subject::MAX_COMMON_NAME_CHARS {
272                        return Err(AxiamError::Validation {
273                            message: format!(
274                                "the default intermediate subject (the subject plus \" \
275                                 Intermediate Authority\") would exceed {} characters; give \
276                                 intermediate_subject",
277                                crate::subject::MAX_COMMON_NAME_CHARS
278                            ),
279                        });
280                    }
281                    derived
282                }
283            };
284            Some(IntermediateSpec {
285                subject,
286                validity_days: input
287                    .intermediate_validity_days
288                    .unwrap_or(input.validity_days),
289            })
290        };
291
292        let generated = store
293            .generate_ca(&CaGenerationRequest {
294                organization_id: input.organization_id,
295                ca_id,
296                subject: input.subject,
297                key_algorithm: input.key_algorithm,
298                validity_days: input.validity_days,
299                intermediate,
300            })
301            .await?;
302
303        let parsed = parse_ca_certificate(&generated.certificate_pem)?;
304        let record = StoreCaCertificate {
305            id: ca_id,
306            organization_id: input.organization_id,
307            tenant_id: None,
308            parent_ca_id: None,
309            subject: parsed.subject,
310            public_cert_pem: generated.certificate_pem,
311            chain_pem: (!generated.chain_pem.is_empty()).then(|| join_pem(&generated.chain_pem)),
312            fingerprint: parsed.fingerprint,
313            key_algorithm: parsed.key_algorithm,
314            not_before: parsed.not_before,
315            not_after: parsed.not_after,
316            encrypted_private_key: None,
317            key_custody: store.custody(),
318            key_locator: Some(generated.locator),
319        };
320
321        let certificate = self.repo.create(record).await?;
322        Ok(GeneratedCaCertificate {
323            certificate,
324            // There is nothing to return, and that is the point of this
325            // custodian rather than an omission.
326            private_key_pem: None,
327        })
328    }

330    /// Import a CA an organization already has (BYOK).
331    ///
332    /// The other way a CA comes to exist. Generation makes a key AXIAM chose;
333    /// this takes one from an offline root, an existing internal PKI or an HSM
334    /// ceremony, and puts AXIAM in the chain rather than at the top of it.
335    ///
336    /// Everything the row records — subject, validity window, key algorithm,
337    /// fingerprint — is parsed out of the supplied certificate rather than
338    /// taken from the request. A caller could otherwise claim a subject or a
339    /// validity window the certificate does not have, and AXIAM would enforce
340    /// the claim while every relying party read the certificate.
341    ///
342    /// A supplied private key is verified to belong to the certificate before
343    /// anything is written. Storing a mismatched pair would produce a CA that
344    /// looks issuable and fails at the first signature, with a message about
345    /// key parsing that names nothing an operator did.
346    pub async fn import(&self, input: ImportCaCertificate) -> AxiamResult<CaCertificate> {
347        let parsed = parse_ca_certificate(&input.public_cert_pem)?;
348
349        if let Some(ref key_pem) = input.private_key_pem {
350            verify_key_matches_certificate(key_pem, &input.public_cert_pem)?;
351        }
352
353        let ca_id = Uuid::new_v4();
354        let (encrypted_private_key, key_locator, custody) = match input.private_key_pem {
355            Some(ref key_pem) => {
356                let store = self.custodians.default_store()?;
357                // `import_ca` rather than `store`: a custodian that is itself a
358                // PKI — Vault's engine — takes the key and the certificate it
359                // belongs to as one bundle and has nowhere to put a key alone.
360                // Every other custodian ignores the certificate.
361                let stored = store
362                    .import_ca(
363                        input.organization_id,
364                        ca_id,
365                        &input.public_cert_pem,
366                        key_pem,
367                    )
368                    .await?;
369                let (inline, locator) = split_stored(stored);
370                (inline, locator, store.custody())
371            }
372            // No key: a trust anchor and nothing more. Recorded as `External`
373            // so the signing path can say why it cannot issue instead of
374            // failing on an absent ciphertext.
375            None => (None, None, CaKeyCustody::External),
376        };
377
378        let record = StoreCaCertificate {
379            id: ca_id,
380            organization_id: input.organization_id,
381            tenant_id: None,
382            parent_ca_id: None,
383            subject: parsed.subject,
384            public_cert_pem: input.public_cert_pem,
385            // Whatever chain an imported CA has is held by whoever imported it;
386            // AXIAM records what it was given and does not invent the rest.
387            chain_pem: None,
388            fingerprint: parsed.fingerprint,
389            key_algorithm: parsed.key_algorithm,
390            not_before: parsed.not_before,
391            not_after: parsed.not_after,
392            encrypted_private_key,
393            key_custody: custody,
394            key_locator,
395        };
396
397        self.repo.create(record).await
398    }

400    /// Create a tenant signing CA beneath one of the organization's CAs.
401    ///
402    /// The intermediate that stands between the organization's trust anchor and
403    /// a tenant's user, service and device certificates. AXIAM generates its
404    /// key — or the custodian does, under `vault_pki` — and the parent signs
405    /// it. The private key is returned exactly once, on the same terms as any
406    /// other generated key, and under `vault_pki` there is none to return.
407    ///
408    /// It is constrained to a path length of zero: it signs leaves and cannot
409    /// mint further CAs. Nothing in AXIAM issues a third tier, and an
410    /// unconstrained intermediate would be a capability every relying party
411    /// downstream had to trust for no gain.
412    pub async fn generate_intermediate(
413        &self,
414        mut input: CreateIntermediateCa,
415    ) -> AxiamResult<GeneratedCaCertificate> {
416        // DF-023, as in [`Self::generate`]: one common name, agreed between
417        // the certificate's DN and the row that describes it.
418        input.subject = subject_common_name(&input.subject)?;
419
420        if input.validity_days == 0 || input.validity_days > MAX_CA_VALIDITY_DAYS {
421            return Err(AxiamError::Validation {
422                message: format!(
423                    "validity_days must be between 1 and {MAX_CA_VALIDITY_DAYS} \
424                     (NIST SP 800-57 max for CA certificates)"
425                ),
426            });
427        }
428
429        let parent = self
430            .load_signing_parent(input.organization_id, input.parent_ca_id)
431            .await?;
432        let (not_before, not_after) = intermediate_window(&parent, input.validity_days)?;
433
434        let ca_id = Uuid::new_v4();
435        // Two custodians, and they are not always the same one. The parent's is
436        // where its key *is* — recorded per CA, so a deployment that adopted
437        // Vault after generating a root still finds that root's key sealed into
438        // its row. The default is where the deployment keeps keys *now*, and
439        // that is where this new key belongs: an intermediate generated today
440        // under a Vault-configured server goes to Vault, whatever the parent
441        // predates.
442        let parent_store = self.custodians.store_for(parent.key_custody)?;
443        let child_store = self.custodians.default_store()?;
444
445        // A custodian that signs on AXIAM's behalf also *creates* on AXIAM's
446        // behalf: asking it to sign a key generated here would defeat the
447        // property that makes it worth using.
448        if parent_store.signs_remotely() {
449            let parent_ref = Self::key_ref(&parent);
450            let created = parent_store
451                .generate_intermediate_ca(
452                    &parent_ref,
453                    &IntermediateCaRequest {
454                        organization_id: input.organization_id,
455                        ca_id,
456                        subject: input.subject.clone(),
457                        key_algorithm: input.key_algorithm,
458                        validity_days: window_days(not_before, not_after),
459                    },
460                )
461                .await?;
462
463            // What the custodian produced, not what was asked for — Vault caps
464            // a TTL to the mount's maximum and to the signer's own expiry
465            // without failing the call.
466            let parsed = parse_ca_certificate(&created.certificate_pem)?;
467            let certificate = self
468                .repo
469                .create(StoreCaCertificate {
470                    id: ca_id,
471                    organization_id: input.organization_id,
472                    tenant_id: Some(input.tenant_id),
473                    parent_ca_id: Some(parent.id),
474                    subject: parsed.subject,
475                    public_cert_pem: created.certificate_pem,
476                    chain_pem: (!created.chain_pem.is_empty())
477                        .then(|| join_pem(&created.chain_pem)),
478                    fingerprint: parsed.fingerprint,
479                    key_algorithm: parsed.key_algorithm,
480                    not_before: parsed.not_before,
481                    not_after: parsed.not_after,
482                    encrypted_private_key: None,
483                    key_custody: parent_store.custody(),
484                    key_locator: Some(created.locator),
485                })
486                .await?;
487            return Ok(GeneratedCaCertificate {
488                certificate,
489                private_key_pem: None,
490            });
491        }
492
493        let parent_ref = Self::key_ref(&parent);
494        let mut parent_key_pem = parent_store
495            .load(&parent_ref, parent.encrypted_private_key.as_deref())
496            .await?
497            .to_string();
498
499        let _permit =
500            crate::permit::bounded_permit(&self.crypto_semaphore, self.permit_wait).await?;
501
502        let parent_cert_pem = parent.public_cert_pem.clone();
503        let subject = input.subject.clone();
504        let key_algorithm = input.key_algorithm.clone();
505        let not_before_ts = not_before.timestamp();
506        let not_after_ts = not_after.timestamp();
507        let crl_points = self.crl_points_under(&parent);
508
509        let (private_key_pem, public_cert_pem, fingerprint) =
510            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String, String)> {
511                let parent_key = KeyPair::from_pem(&parent_key_pem).map_err(|e| {
512                    AxiamError::Certificate(format!("invalid parent CA private key: {e}"))
513                })?;
514                // Scrubbed as soon as it is parsed, the same as the leaf path.
515                parent_key_pem.zeroize();
516
517                // The issuer DN comes from the parent's stored certificate, not
518                // its mutable `subject` column, so what the child names as its
519                // issuer can never drift from what the parent actually is.
520                let issuer =
521                    Issuer::from_ca_cert_pem(&parent_cert_pem, parent_key).map_err(|e| {
522                        AxiamError::Certificate(format!("invalid parent CA certificate PEM: {e}"))
523                    })?;
524
525                let key_pair = generate_keypair(&key_algorithm)?;
526                let private_key_pem = private_key_pem(&key_pair)?;
527
528                let mut params = intermediate_params(&subject, not_before_ts, not_after_ts)?;
529                params.use_authority_key_identifier_extension = true;
530                params.crl_distribution_points = crl_points;
531
532                let cert = params.signed_by(&key_pair, &issuer).map_err(|e| {
533                    AxiamError::Certificate(format!("intermediate CA signing failed: {e}"))
534                })?;
535                let public_cert_pem = cert.pem();
536                let fingerprint = compute_fingerprint(cert.der());
537                Ok((private_key_pem, public_cert_pem, fingerprint))
538            })
539            .await
540            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
541
542        // `import_ca` rather than `store`, because the certificate and the key
543        // are both in hand and a custodian that is itself a PKI takes them
544        // together: under `vault_pki` this imports the intermediate into Vault
545        // as an issuer Vault can then sign leaves with, and under `vault` it
546        // falls through to `store` and the key goes to Vault's KV engine. The
547        // one custodian that keeps it here is `database`, and only because that
548        // is what the deployment asked for.
549        let stored = child_store
550            .import_ca(
551                input.organization_id,
552                ca_id,
553                &public_cert_pem,
554                &private_key_pem,
555            )
556            .await?;
557        let (encrypted_private_key, key_locator) = split_stored(stored);
558
559        let certificate = self
560            .repo
561            .create(StoreCaCertificate {
562                id: ca_id,
563                organization_id: input.organization_id,
564                tenant_id: Some(input.tenant_id),
565                parent_ca_id: Some(parent.id),
566                subject: input.subject,
567                public_cert_pem,
568                chain_pem: Some(chain_below(&parent)),
569                fingerprint,
570                key_algorithm: input.key_algorithm,
571                not_before,
572                not_after,
573                encrypted_private_key,
574                key_custody: child_store.custody(),
575                key_locator,
576            })
577            .await?;
578
579        Ok(GeneratedCaCertificate {
580            certificate,
581            private_key_pem: Some(private_key_pem),
582        })
583    }

585    /// Sign a tenant's own certificate signing request as a signing CA.
586    ///
587    /// The BYOK counterpart to [`Self::generate_intermediate`]. The key behind
588    /// the CSR was generated by whoever produced it and never reaches AXIAM, so
589    /// there is nothing to return once and nothing to take custody of: the row
590    /// records custody `External`, and issuance against this CA happens
591    /// wherever the key actually lives.
592    ///
593    /// The CSR's subject is honoured; its requested extensions are not. A
594    /// request that asked to be an unconstrained CA, or a TLS server, would
595    /// otherwise decide its own powers — AXIAM decides them, and they are
596    /// exactly "sign leaves".
597    pub async fn sign_intermediate_csr(
598        &self,
599        input: SignIntermediateCsr,
600    ) -> AxiamResult<CaCertificate> {
601        if input.validity_days == 0 || input.validity_days > MAX_CA_VALIDITY_DAYS {
602            return Err(AxiamError::Validation {
603                message: format!(
604                    "validity_days must be between 1 and {MAX_CA_VALIDITY_DAYS} \
605                     (NIST SP 800-57 max for CA certificates)"
606                ),
607            });
608        }
609
610        let parent = self
611            .load_signing_parent(input.organization_id, input.parent_ca_id)
612            .await?;
613        let (not_before, not_after) = intermediate_window(&parent, input.validity_days)?;
614
615        // Parsed here rather than inside the signing task so that a malformed
616        // CSR — by far the likeliest failure on this endpoint — is a 400 naming
617        // the request, not a 500 out of a custodian.
618        let csr_subject = csr_common_name(&input.csr_pem)?;
619
620        let ca_id = Uuid::new_v4();
621        let parent_store = self.custodians.store_for(parent.key_custody)?;
622
623        if parent_store.signs_remotely() {
624            let parent_ref = Self::key_ref(&parent);
625            let signed = parent_store
626                .sign_intermediate_csr(
627                    &parent_ref,
628                    &IntermediateSigningRequest {
629                        csr_pem: input.csr_pem,
630                        subject: csr_subject,
631                        ttl_seconds: (not_after - not_before).num_seconds(),
632                    },
633                )
634                .await?;
635
636            let parsed = parse_ca_certificate(&signed.certificate_pem)?;
637            return self
638                .repo
639                .create(StoreCaCertificate {
640                    id: ca_id,
641                    organization_id: input.organization_id,
642                    tenant_id: Some(input.tenant_id),
643                    parent_ca_id: Some(parent.id),
644                    subject: parsed.subject,
645                    public_cert_pem: signed.certificate_pem,
646                    chain_pem: (!signed.chain_pem.is_empty()).then(|| join_pem(&signed.chain_pem)),
647                    fingerprint: parsed.fingerprint,
648                    key_algorithm: parsed.key_algorithm,
649                    not_before: parsed.not_before,
650                    not_after: parsed.not_after,
651                    encrypted_private_key: None,
652                    // AXIAM never held this key and never will. `External` is
653                    // what stops the issuance path offering to sign with a key
654                    // that is not here.
655                    key_custody: CaKeyCustody::External,
656                    key_locator: None,
657                })
658                .await;
659        }
660
661        let parent_ref = Self::key_ref(&parent);
662        let mut parent_key_pem = parent_store
663            .load(&parent_ref, parent.encrypted_private_key.as_deref())
664            .await?
665            .to_string();
666
667        let _permit =
668            crate::permit::bounded_permit(&self.crypto_semaphore, self.permit_wait).await?;
669
670        let parent_cert_pem = parent.public_cert_pem.clone();
671        let csr_pem = input.csr_pem.clone();
672        let subject = csr_subject.clone();
673        let not_before_ts = not_before.timestamp();
674        let not_after_ts = not_after.timestamp();
675        let crl_points = self.crl_points_under(&parent);
676
677        let (public_cert_pem, fingerprint) =
678            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String)> {
679                let parent_key = KeyPair::from_pem(&parent_key_pem).map_err(|e| {
680                    AxiamError::Certificate(format!("invalid parent CA private key: {e}"))
681                })?;
682                parent_key_pem.zeroize();
683
684                let issuer =
685                    Issuer::from_ca_cert_pem(&parent_cert_pem, parent_key).map_err(|e| {
686                        AxiamError::Certificate(format!("invalid parent CA certificate PEM: {e}"))
687                    })?;
688
689                // `from_pem` verifies the request's own signature, which is the
690                // only proof that whoever sent it holds the matching private
691                // key. Without that check a caller could have a CA minted for
692                // somebody else's public key.
693                let mut request =
694                    CertificateSigningRequestParams::from_pem(&csr_pem).map_err(|e| {
695                        AxiamError::Validation {
696                            message: format!("the certificate signing request was rejected: {e}"),
697                        }
698                    })?;
699                // Everything the request asked to be, overwritten by what AXIAM
700                // has decided it is — the DN included, so the row and the
701                // certificate cannot disagree.
702                request.params = intermediate_params(&subject, not_before_ts, not_after_ts)?;
703                request.params.use_authority_key_identifier_extension = true;
704                request.params.crl_distribution_points = crl_points;
705
706                let cert = request.signed_by(&issuer).map_err(|e| {
707                    AxiamError::Certificate(format!("intermediate CA signing failed: {e}"))
708                })?;
709                let public_cert_pem = cert.pem();
710                let fingerprint = compute_fingerprint(cert.der());
711                Ok((public_cert_pem, fingerprint))
712            })
713            .await
714            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
715
716        // Read back out of the certificate that was produced: the key algorithm
717        // is the CSR's, not anything the caller stated, and the row must say
718        // what the certificate says.
719        let parsed = parse_ca_certificate(&public_cert_pem)?;
720
721        self.repo
722            .create(StoreCaCertificate {
723                id: ca_id,
724                organization_id: input.organization_id,
725                tenant_id: Some(input.tenant_id),
726                parent_ca_id: Some(parent.id),
727                subject: parsed.subject,
728                public_cert_pem,
729                chain_pem: Some(chain_below(&parent)),
730                fingerprint,
731                key_algorithm: parsed.key_algorithm,
732                not_before: parsed.not_before,
733                not_after: parsed.not_after,
734                encrypted_private_key: None,
735                key_custody: CaKeyCustody::External,
736                key_locator: None,
737            })
738            .await
739    }

741    /// Load the CA that is about to sign, refusing every state it cannot sign in.
742    ///
743    /// Each refusal exists because the alternative is a certificate that looks
744    /// issued and is not trusted: an expired parent produces a chain no
745    /// validator accepts, a revoked one produces a chain every validator
746    /// rejects on purpose, and a parent AXIAM holds no key for produces a
747    /// failure several layers down that names a decryption error rather than
748    /// the missing key.
749    async fn load_signing_parent(
750        &self,
751        organization_id: Uuid,
752        parent_ca_id: Uuid,
753    ) -> AxiamResult<CaCertificate> {
754        let parent = self.repo.get_by_id(organization_id, parent_ca_id).await?;
755
756        if parent.tenant_id.is_some() {
757            return Err(AxiamError::Validation {
758                message: "that CA is itself a tenant signing CA, which is constrained to a \
759                          path length of zero and cannot sign another CA: choose an \
760                          organization CA as the parent"
761                    .into(),
762            });
763        }
764        if parent.status != CertificateStatus::Active {
765            return Err(AxiamError::Validation {
766                message: "the parent CA is not active".into(),
767            });
768        }
769        let now = Utc::now();
770        if now < parent.not_before || now > parent.not_after {
771            return Err(AxiamError::Validation {
772                message: "the parent CA is expired or not yet valid".into(),
773            });
774        }
775        if parent.key_custody == CaKeyCustody::External {
776            return Err(AxiamError::Validation {
777                message: "AXIAM holds no private key for that CA — it was imported as a trust \
778                          anchor only, so it can be trusted but cannot sign"
779                    .into(),
780            });
781        }
782        Ok(parent)
783    }

821    /// Move a CA's signing key to the deployment's currently configured
822    /// custodian.
823    ///
824    /// # Why this exists
825    ///
826    /// Custody is recorded per CA, not read from configuration, and that is
827    /// deliberate: a deployment that adopts Vault keeps issuing from the CAs it
828    /// already has, whose keys are still sealed into their rows. The cost of
829    /// that design is that adopting Vault changes nothing about the keys you
830    /// already have — they stay in the database, forever, unless something
831    /// moves them. This is that something.
832    ///
833    /// Without it the only route from database custody to Vault is to generate
834    /// a new CA and re-issue every leaf certificate beneath it, which for a
835    /// trust anchor means touching every relying party. That is a real
836    /// migration for a real reason; it is not a reasonable price for changing
837    /// where a key is filed.
838    ///
839    /// # Order of operations
840    ///
841    /// Copy first, record second, delete last:
842    ///
843    /// 1. Load the key from the custodian the **row** names.
844    /// 2. Hand it to the configured custodian.
845    /// 3. Update the row — new custody, new locator, and the row's own
846    ///    ciphertext cleared in the same statement.
847    /// 4. Best-effort release from the old custodian.
848    ///
849    /// A failure at step 1 or 2 leaves the CA exactly as it was: the row still
850    /// names the old custodian, which still holds the key, and issuance is
851    /// unaffected. A failure at step 4 leaves an orphaned secret behind — a
852    /// cleanup task, not a security failure, and the same trade-off revocation
853    /// already makes. What must never happen is the reverse order, which would
854    /// delete a key the new custodian had not accepted.
855    ///
856    /// # Errors
857    ///
858    /// - No custodian configured, or the CA is already under the configured one.
859    /// - The CA is under `external` custody: AXIAM holds no key to move.
860    /// - The CA is under a custodian that will not hand its key over
861    ///   (`vault_pki`). That is the stronger property, not a limitation to work
862    ///   around, and moving *out* of it would weaken the CA.
863    pub async fn migrate_key_custody(
864        &self,
865        organization_id: Uuid,
866        id: Uuid,
867    ) -> AxiamResult<CaCertificate> {
868        let certificate = self.repo.get_by_id(organization_id, id).await?;
869        let target = self.custodians.default_store()?;
870
871        if certificate.key_custody == target.custody() {
872            return Err(AxiamError::Validation {
873                message: format!(
874                    "this CA's key is already held by the `{}` custodian",
875                    target.custody()
876                ),
877            });
878        }
879        if certificate.key_custody == CaKeyCustody::External {
880            return Err(AxiamError::Validation {
881                message: "AXIAM holds no private key for this CA, so there is nothing to \
882                          move. It was imported as a trust anchor only."
883                    .into(),
884            });
885        }
886
887        let source = self.custodians.store_for(certificate.key_custody)?;
888        if source.signs_remotely() {
889            return Err(AxiamError::Validation {
890                message: format!(
891                    "this CA's key is held by the `{}` custodian, which never hands it \
892                     over — that is a stronger property than the target offers, and \
893                     moving out of it would weaken the CA",
894                    certificate.key_custody
895                ),
896            });
897        }
898
899        let key_ref = Self::key_ref(&certificate);
900        let private_key_pem = source
901            .load(&key_ref, certificate.encrypted_private_key.as_deref())
902            .await?;
903
904        // `import_ca` rather than `store`: a custodian that is a PKI rather
905        // than a safe needs the certificate alongside the key, and every other
906        // custodian's default body ignores it and calls `store`.
907        let stored = target
908            .import_ca(
909                organization_id,
910                id,
911                &certificate.public_cert_pem,
912                &private_key_pem,
913            )
914            .await?;
915        // Both halves, not just the locator: an inlining custodian (database)
916        // hands back ciphertext and no locator, and the row's column is the
917        // only place it has to live. Discarding it here wrote `custody =
918        // database` beside an empty key column and then released the source
919        // copy, which destroyed the signing key.
920        let (ciphertext, locator) = split_stored(stored);
921
922        let updated = self
923            .repo
924            .update_key_custody(organization_id, id, target.custody(), locator, ciphertext)
925            .await?;
926
927        if let Err(e) = source.delete(&key_ref).await {
928            tracing::warn!(
929                error = %e,
930                %organization_id,
931                ca_id = %id,
932                from = %certificate.key_custody,
933                "could not release the CA key from its previous custodian after \
934                 migration; the key has moved and this leaves an orphaned copy to \
935                 clean up"
936            );
937        }
938
939        tracing::info!(
940            %organization_id,
941            ca_id = %id,
942            from = %certificate.key_custody,
943            to = %updated.key_custody,
944            "CA signing key custody migrated"
945        );
946        Ok(updated)
947    }

949    /// Revoke a CA: the row first, then its key at its custodian, then — for a
950    /// subordinate CA its parent signed through Vault's PKI engine — the CA's
951    /// serial on the parent's list in Vault (#612).
952    ///
953    /// AXIAM's revocation stands whatever the custodians answer: the row is
954    /// what AXIAM signs with and what its own list for the parent reads (#565).
955    /// The forward is the leaf path's (T-470): within
956    /// [`CUSTODIAN_REVOKE_ATTEMPTS`] attempts and [`CUSTODIAN_REVOKE_BUDGET`],
957    /// and a refusal or an unreachable Vault is answered as
958    /// [`CustodianRevocation::Pending`], for the caller to record, and retried by
959    /// the cleanup job's `vault_revocation` sweep from the row — which keeps the
960    /// certificate and `parent_ca_id`, everything the retry needs, after the
961    /// key is gone.
962    ///
963    /// The forward comes **after** the key is released, not before. Vault
964    /// refuses `revoke` for a serial that is still an issuer in the same mount
965    /// ("adding issuer to its own CRL is not allowed"), and a parent whose
966    /// locator records no root (imported, or created with `issue_from_root`
967    /// and imported into the issuing mount) signs its subordinates from the
968    /// very mount that holds their issuers. Once the subordinate's issuer is
969    /// deleted, its serial is a stored certificate like a leaf's, and the
970    /// parent's mount accepts it; where the mounts differ (the usual two-tier
971    /// organization CA) the order changes nothing.
972    ///
973    /// A release that failed here (logged and swallowed) or never ran (the
974    /// process died after the revocation) leaves that issuer in the parent's
975    /// mount, and Vault then refuses every forward. The sweep recognises the
976    /// refusal and runs the release again before it forwards
977    /// ([`forward_ca_revocation_releasing`], R101W1-04).
978    pub async fn revoke(
979        &self,
980        organization_id: Uuid,
981        id: Uuid,
982    ) -> AxiamResult<CustodianRevocation> {
983        self.revoke_within(organization_id, id, CUSTODIAN_REVOKE_ATTEMPTS, None)
984            .await
985    }
986
987    /// [`Self::revoke`], forwarding within `attempts` and before `deadline` —
988    /// `None` for [`CUSTODIAN_REVOKE_BUDGET`] from the moment the key is
989    /// released, so a slow release does not eat the forward's budget.
990    async fn revoke_within(
991        &self,
992        organization_id: Uuid,
993        id: Uuid,
994        attempts: u32,
995        deadline: Option<tokio::time::Instant>,
996    ) -> AxiamResult<CustodianRevocation> {
997        let certificate = self.repo.get_by_id(organization_id, id).await?;
998        self.repo.revoke(organization_id, id).await?;
999
1000        // Release custody after the revocation, and never before: the row is
1001        // the record of what exists, and a custodian that could not be reached
1002        // must not leave a CA marked active whose key is gone. What a failure
1003        // here leaves behind is an orphaned secret — a cleanup task, not a
1004        // security failure, because AXIAM will not sign with a revoked CA
1005        // whatever the custodian still holds.
1006        if let Ok(store) = self.custodians.store_for(certificate.key_custody) {
1007            let key_ref = Self::key_ref(&certificate);
1008            if let Err(e) = store.delete(&key_ref).await {
1009                tracing::warn!(
1010                    ca_id = %id,
1011                    custody = %certificate.key_custody,
1012                    error = %e,
1013                    "CA revoked, but its key could not be removed from the custodian"
1014                );
1015            }
1016        }
1017
1018        // The row as read before the revocation: neither the revocation nor
1019        // the release changes what the forward reads (the certificate, the
1020        // custody, the parent).
1021        let deadline =
1022            deadline.unwrap_or_else(|| tokio::time::Instant::now() + CUSTODIAN_REVOKE_BUDGET);
1023        Ok(forward_ca_revocation(
1024            &self.repo,
1025            &self.custodians,
1026            &certificate,
1027            attempts,
1028            deadline,
1029        )
1030        .await)
1031    }

1033    /// Revoke every active signing CA of a tenant that is being deleted, each
1034    /// through [`Self::revoke`] — so its custodian releases the key — and
1035    /// return how many (R1W1-01).
1036    ///
1037    /// A deleted tenant's CA is disowned with it: it goes on its parent's
1038    /// revocation list, so every leaf it signed stops chaining for relying
1039    /// parties outside AXIAM, instead of staying valid until it expires. Its row
1040    /// outlives the tenant until it expires (`tenant_purge` keeps a revoked,
1041    /// unexpired CA, without its key), so the parent's list keeps naming it.
1042    ///
1043    /// Each subordinate CA's forward to its parent's list in Vault (#612) takes
1044    /// one attempt, inside one [`CUSTODIAN_REVOKE_BUDGET`] for all of them, as
1045    /// the tenant's leaves do (`CertService::revoke_tenant`); one Vault did not
1046    /// take is counted in [`TenantCasRevoked::vault_pending`] and retried by
1047    /// the cleanup job's sweep.
1048    pub async fn revoke_tenant_cas(
1049        &self,
1050        organization_id: Uuid,
1051        tenant_id: Uuid,
1052    ) -> AxiamResult<TenantCasRevoked> {
1053        const PAGE: u64 = 100;
1054        let mut active = Vec::new();
1055        let mut offset = 0;
1056        loop {
1057            let page = self
1058                .repo
1059                .list_by_tenant(
1060                    organization_id,
1061                    tenant_id,
1062                    Pagination {
1063                        offset,
1064                        limit: PAGE,
1065                        search: None,
1066                    },
1067                )
1068                .await?;
1069            let read = page.items.len() as u64;
1070            active.extend(
1071                page.items
1072                    .into_iter()
1073                    .filter(|ca| ca.status == CertificateStatus::Active)
1074                    .map(|ca| ca.id),
1075            );
1076            offset += read;
1077            if read < PAGE || offset >= page.total {
1078                break;
1079            }
1080        }
1081        let deadline = tokio::time::Instant::now() + CUSTODIAN_REVOKE_BUDGET;
1082        let mut vault_pending = 0usize;
1083        for id in &active {
1084            if let CustodianRevocation::Pending { .. } = self
1085                .revoke_within(organization_id, *id, 1, Some(deadline))
1086                .await?
1087            {
1088                vault_pending += 1;
1089            }
1090        }
1091        Ok(TenantCasRevoked {
1092            revoked: active.len(),
1093            vault_pending,
1094        })
1095    }

1118/// Tell Vault that `revoked`, a revoked subordinate CA, is revoked — on the
1119/// list of the parent that signed it, when that parent's key Vault's PKI engine
1120/// holds (#612). Never fails: the outcome is the answer.
1121///
1122/// [`CustodianRevocation::NotNeeded`] for a CA with no parent, a parent that is
1123/// gone, not `vault_pki` (its list is AXIAM's, which reads the row) or itself
1124/// revoked (revoking it deleted its issuers from Vault, so there is no list
1125/// there left to name the child), and for a CA Vault never signed: one in
1126/// `database` or `vault` custody under a `vault_pki` parent was signed in
1127/// AXIAM before the parent's key moved into Vault. On success the row records
1128/// `vault_revoked_at` (schema v96).
1129///
1130/// Shared by [`CaService::revoke`] and the cleanup job's sweep
1131/// (`CertService::forward_pending_revocations`), which reads only the row: the
1132/// certificate (its serial) and `parent_ca_id`, both kept by a revoked CA's row
1133/// after its key is released and until eight days after it expires
1134/// (`tenant_purge`), past the sweep's own window (`not_after`).
1135pub(crate) async fn forward_ca_revocation<R: CaCertificateRepository>(
1136    repo: &R,
1137    custodians: &CaKeyCustodians,
1138    revoked: &CaCertificate,
1139    attempts: u32,
1140    deadline: tokio::time::Instant,
1141) -> CustodianRevocation {
1142    let pending = |reason: String| {
1143        tracing::error!(
1144            ca_id = %revoked.id,
1145            organization_id = %revoked.organization_id,
1146            parent_ca_id = ?revoked.parent_ca_id,
1147            reason = %crate::vault_forward::sanitise_reason(&reason),
1148            "CA revoked in AXIAM but not yet on its parent's revocation list in Vault, which \
1149             therefore does not name it; the cleanup job retries"
1150        );
1151        CustodianRevocation::Pending { reason }
1152    };
1153
1154    let Some(parent_id) = revoked.parent_ca_id else {
1155        return CustodianRevocation::NotNeeded;
1156    };
1157    if !matches!(
1158        revoked.key_custody,
1159        CaKeyCustody::VaultPki | CaKeyCustody::External
1160    ) {
1161        return CustodianRevocation::NotNeeded;
1162    }
1163    let parent = match repo.get_by_issuer_id(parent_id).await {
1164        Ok(parent) => parent,
1165        Err(AxiamError::NotFound { .. }) => return CustodianRevocation::NotNeeded,
1166        Err(e) => return pending(format!("the parent CA could not be read: {e}")),
1167    };
1168    if parent.key_custody != CaKeyCustody::VaultPki || parent.status != CertificateStatus::Active {
1169        return CustodianRevocation::NotNeeded;
1170    }
1171    let store = match custodians.store_for(parent.key_custody) {
1172        Ok(store) if store.signs_remotely() => store,
1173        Ok(_) => return CustodianRevocation::NotNeeded,
1174        Err(e) => return pending(e.to_string()),
1175    };
1176    let serial = match serial_hex(&revoked.public_cert_pem) {
1177        Ok(serial) => serial,
1178        Err(e) => return pending(e.to_string()),
1179    };
1180
1181    let parent_ref = CaService::<R>::key_ref(&parent);
1182    if let Err(reason) = attempt_custodian(attempts, deadline, || {
1183        store.revoke_signed_ca(&parent_ref, &serial)
1184    })
1185    .await
1186    {
1187        return pending(reason);
1188    }
1189    if let Err(e) = repo
1190        .mark_revocation_forwarded(revoked.organization_id, revoked.id)
1191        .await
1192    {
1193        // Vault has it; the next sweep forwards it again, which Vault answers
1194        // with success.
1195        tracing::warn!(
1196            ca_id = %revoked.id,
1197            error = %e,
1198            "Vault accepted the CA's revocation, but recording that failed"
1199        );
1200    }
1201    CustodianRevocation::Forwarded
1202}

1204/// The refusal Vault's PKI engine answers `revoke` with for a serial that is
1205/// still an issuer of the mount it is revoked in: "adding issuer (id: …) to its
1206/// own CRL is not allowed" (issuers are revoked through `issuer/<ref>/revoke`).
1207/// Matched case-insensitively and without the issuer's id, which is the part
1208/// that varies.
1209const ISSUER_ON_ITS_OWN_CRL: &str = "to its own crl is not allowed";

1211/// Whether `reason`, the text of a failed forward, is Vault saying the revoked
1212/// CA is still an issuer in the mount it is being revoked in (R101W1-04).
1213///
1214/// That is the single-mount layout — a parent that signs from the mount that
1215/// also holds its subordinates' issuers — and it means the release
1216/// [`CaService::revoke`] makes after the revocation did not happen: the delete
1217/// failed (logged and swallowed there) or the process died between the two.
1218pub(crate) fn is_issuer_on_its_own_crl_refusal(reason: &str) -> bool {
1219    reason.to_ascii_lowercase().contains(ISSUER_ON_ITS_OWN_CRL)
1220}

1222/// Run the custodian's release of a revoked subordinate CA again — the
1223/// issuer(s) and key its own locator names, and nothing else — and say whether
1224/// the custodian accepted it (R101W1-04).
1225///
1226/// For the sweep, when Vault refused a forward because the CA is still an issuer
1227/// in the parent's mount ([`is_issuer_on_its_own_crl_refusal`]). Idempotent: the
1228/// Vault store treats an issuer or key that is already gone as released, so a
1229/// release that had in fact succeeded costs two `404`s and no error. Never
1230/// releases a CA that is not revoked, nor one whose key is not in Vault's PKI
1231/// engine (an `external` CA's key is not Vault's to delete), and gives up at
1232/// `deadline`. A failure is logged and answered `false`; the forward then
1233/// stays pending as it was.
1234pub(crate) async fn release_revoked_ca<R: CaCertificateRepository>(
1235    custodians: &CaKeyCustodians,
1236    revoked: &CaCertificate,
1237    deadline: tokio::time::Instant,
1238) -> bool {
1239    if revoked.status != CertificateStatus::Revoked || revoked.key_custody != CaKeyCustody::VaultPki
1240    {
1241        return false;
1242    }
1243    let store = match custodians.store_for(revoked.key_custody) {
1244        Ok(store) => store,
1245        Err(e) => {
1246            tracing::warn!(ca_id = %revoked.id, error = %e, "the revoked CA's key custodian is not available");
1247            return false;
1248        }
1249    };
1250    let key_ref = CaService::<R>::key_ref(revoked);
1251    match tokio::time::timeout_at(deadline, store.delete(&key_ref)).await {
1252        Ok(Ok(())) => true,
1253        Ok(Err(e)) => {
1254            tracing::warn!(
1255                ca_id = %revoked.id,
1256                error = %e,
1257                "a revoked CA's issuer could not be removed from Vault again"
1258            );
1259            false
1260        }
1261        Err(_) => {
1262            tracing::warn!(ca_id = %revoked.id, "Vault did not answer the removal of a revoked CA's issuer in time");
1263            false
1264        }
1265    }
1266}

1268/// [`forward_ca_revocation`] for the cleanup job's sweep, which also finishes
1269/// the release a revocation left undone (R101W1-04).
1270///
1271/// Where the parent signs from the mount that holds its subordinates' issuers,
1272/// Vault refuses to revoke a serial that is still an issuer there, so a CA whose
1273/// release failed is refused at every forward. `last_error` is the sanitised
1274/// reason the row recorded at its previous failed attempt. When it names that
1275/// refusal the release is run again before the forward; when it does not (the
1276/// request that revoked the CA records nothing on the row, and the reason is cut
1277/// to 256 characters), a forward that is refused that way is followed by the
1278/// release and one more forward, inside the same claim and `deadline`. Once
1279/// per attempt either way: the row's claim, backoff and failure count are the
1280/// caller's and unchanged.
1281pub(crate) async fn forward_ca_revocation_releasing<R: CaCertificateRepository>(
1282    repo: &R,
1283    custodians: &CaKeyCustodians,
1284    revoked: &CaCertificate,
1285    last_error: Option<&str>,
1286    attempts: u32,
1287    deadline: tokio::time::Instant,
1288) -> CustodianRevocation {
1289    let refused_before = last_error.is_some_and(is_issuer_on_its_own_crl_refusal);
1290    if refused_before {
1291        release_revoked_ca::<R>(custodians, revoked, deadline).await;
1292    }
1293    let outcome = forward_ca_revocation(repo, custodians, revoked, attempts, deadline).await;
1294    if !refused_before
1295        && matches!(&outcome, CustodianRevocation::Pending { reason } if is_issuer_on_its_own_crl_refusal(reason))
1296        && release_revoked_ca::<R>(custodians, revoked, deadline).await
1297    {
1298        return forward_ca_revocation(repo, custodians, revoked, attempts, deadline).await;
1299    }
1300    outcome
1301}

1332/// The certificate a tenant signing CA gets, whoever generated its key.
1333///
1334/// One place, so the generate path and the sign-a-CSR path cannot produce two
1335/// differently-powered CAs from the same endpoint pair. Path length zero and
1336/// `keyCertSign`/`cRLSign` and nothing else: it signs leaves and revokes them,

1338/// critical on a CA, which rcgen does whenever the list is non-empty — leaving
1339/// it empty omits the extension entirely and produces a CA whose powers are
1340/// unstated.
1341fn intermediate_params(
1342    subject: &str,
1343    not_before_ts: i64,
1344    not_after_ts: i64,
1345) -> AxiamResult<CertificateParams> {
1346    let mut params = CertificateParams::new(Vec::<String>::new())
1347        .map_err(|e| AxiamError::Certificate(e.to_string()))?;
1348    params.distinguished_name.push(DnType::CommonName, subject);
1349    params.is_ca = IsCa::Ca(BasicConstraints::Constrained(0));
1350    // #653: rcgen's default is a hash of the public key, so two intermediates on
1351    // one key would share a serial.
1352    params.serial_number = Some(crate::serial::random_serial());
1353    params.key_usages = vec![KeyUsagePurpose::KeyCertSign, KeyUsagePurpose::CrlSign];

1355        .map_err(|e| AxiamError::Certificate(format!("invalid notBefore: {e}")))?;
1356    params.not_after = time::OffsetDateTime::from_unix_timestamp(not_after_ts)
1357        .map_err(|e| AxiamError::Certificate(format!("invalid notAfter: {e}")))?;
1358    Ok(params)

1371/// The validity window for a certificate signed by `parent`.
1372///
1373/// Capped to the parent's own expiry, because a certificate that outlives its
1374/// issuer is one every validator rejects for the whole of its extra life —
1375/// silently, and long after anybody remembers asking for it.
1376fn intermediate_window(
1377    parent: &CaCertificate,
1378    validity_days: u32,
1379) -> AxiamResult<(DateTime<Utc>, DateTime<Utc>)> {
1380    let not_before = Utc::now();
1381    let requested = not_before
1382        .checked_add_signed(Duration::days(i64::from(validity_days)))
1383        .ok_or_else(|| AxiamError::Validation {
1384            message: "validity_days produces a date out of range".into(),
1385        })?;
1386    // Same rule, and the same reasoning, as the leaf path in `cert.rs`: an
1387    // intermediate that outlives its parent stops validating the day the parent
1388    // expires, so the surplus is time the operator thinks they have and does
1389    // not. This used to silently truncate to the parent's notAfter; refusing and
1390    // naming the achievable number is what lets them pick a real one.
1391    if requested > parent.not_after {
1392        let available = crate::cert::issuer_bounded_validity_days(not_before, parent.not_after);
1393        return Err(AxiamError::Validation {
1394            message: format!(
1395                "validity_days is {validity_days} but the parent CA expires on {} — an \
1396                 intermediate cannot outlive the CA that signs it. The most this parent can \
1397                 grant today is {available} day{}.",
1398                parent.not_after.format("%Y-%m-%d"),
1399                if available == 1 { "" } else { "s" },
1400            ),
1401        });
1402    }
1403    Ok((not_before, requested))
1404}

1406/// A window expressed back in whole days, for a custodian whose API takes a TTL.
1407///
1408/// Rounded up, and floored at one: a window shorter than a day would otherwise
1409/// become a TTL of zero, which Vault reads as "the mount's default" rather than
1410/// "no time at all".
1411fn window_days(not_before: DateTime<Utc>, not_after: DateTime<Utc>) -> u32 {
1412    let hours = (not_after - not_before).num_hours().max(1);
1413    u32::try_from((hours + 23) / 24).unwrap_or(u32::MAX).max(1)
1414}

1416/// What a certificate signing request says about itself, once its signature has
1417/// been checked.
1418///
1419/// Everything here comes from a **single** parse and a **single** signature
1420/// verification. Reading the common name in one function, the key in another
1421/// and the requested extensions in a third would verify possession up to three
1422/// times on the way to three different answers, and would let a future caller
1423/// reach one of the facts without having verified anything at all.
1424#[derive(Debug)]
1425pub(crate) struct CsrFacts {
1426    /// The common name the request asks for. Non-empty: a subject is what
1427    /// distinguishes one certificate from another in every list that shows
1428    /// them.
1429    pub(crate) common_name: String,
1430    /// The key algorithm, read from the request's own public key rather than
1431    /// taken from the caller.
1432    pub(crate) key_algorithm: KeyAlgorithm,
1433    /// The RSA modulus size in bits, or `None` for a key that is not RSA.
1434    ///
1435    /// Separate from `key_algorithm` because [`KeyAlgorithm`] has no way to say
1436    /// "RSA, 2048" — the variant is named `Rsa4096` — and a caller enforcing a
1437    /// modulus floor must see the real number rather than the label.
1438    pub(crate) rsa_modulus_bits: Option<usize>,
1439    /// The names of the requested extensions AXIAM refuses to sign over, in
1440    /// the order they appear. Empty when the request asks for none of them.
1441    ///
1442    /// Exactly `subjectAltName`, `keyUsage` and `extendedKeyUsage` — the three
1443    /// a leaf caller can put in a CSR that would otherwise reach the
1444    /// certificate, and the three that must be refused rather than dropped.
1445    ///
1446    /// Refused, and not silently stripped, for two reasons. A caller who put
1447    /// SANs in a leaf CSR meant them, and a certificate issued without them
1448    /// and with nothing said is a credential that fails where it is deployed.
1449    /// And a caller who put a key usage in one **would get it** under
1450    /// `vault_pki` custody: Vault's `sign-verbatim` discards the `key_usage`
1451    /// and `ext_key_usage` request parameters when the CSR carries those
1452    /// extensions and issues what the CSR asked for. Dropping them would
1453    /// therefore be a promise AXIAM can keep on one custodian and not the
1454    /// other, which is worse than a rule that holds everywhere.
1455    ///
1456    /// `basicConstraints` is deliberately **not** here: the DB path overwrites
1457    /// the whole parameter set before signing, and Vault ignores a CSR's basic
1458    /// constraints outright (warning that it did). A CSR asking to be a CA is
1459    /// signed as a leaf on both, which is the test
1460    /// `a_csr_that_asked_to_be_a_ca_does_not_get_to_be_one` asserts.
1461    pub(crate) refused_extensions: Vec<&'static str>,
1462}

1464/// Read a CSR, refusing anything that does not prove possession of its own key.
1465///
1466/// Called outside the signing task so a malformed request — by far the likeliest
1467/// failure on an endpoint whose input is pasted by hand — is a `400` naming the
1468/// CSR rather than a `500` out of a custodian.
1469pub(crate) fn inspect_csr(pem: &str) -> AxiamResult<CsrFacts> {
1470    let block = pem::parse(pem).map_err(|e| AxiamError::Validation {
1471        message: format!("the certificate signing request is not PEM-encoded: {e}"),
1472    })?;
1473    let (_, csr) =
1474        x509_parser::certification_request::X509CertificationRequest::from_der(block.contents())
1475            .map_err(|e| AxiamError::Validation {
1476                message: format!("the certificate signing request could not be parsed: {e}"),
1477            })?;
1478    // The whole point of the endpoint: this is the only evidence that whoever
1479    // sent the request holds the matching private key. Without it a caller
1480    // could have a certificate minted for somebody else's public key.
1481    csr.verify_signature().map_err(|_| AxiamError::Validation {
1482        message: "the certificate signing request's signature does not verify against the \
1483                  public key it carries, so nothing proves the sender holds the matching \
1484                  private key"
1485            .into(),
1486    })?;
1487
1488    let subject = &csr.certification_request_info.subject;
1489    // R110W1-02 — the name is single-valued. A `vault_pki` CA sends this very
1490    // CSR to `sign-verbatim`, which copies the whole subject, and a relying
1491    // party that reads the *last* common name (Go's `crypto/x509` overwrites
1492    // `Subject.CommonName` with each in turn) would authenticate the holder
1493    // under a name the row, which records the first, does not show.
1494    if subject.iter_common_name().nth(1).is_some() {
1495        return Err(AxiamError::Validation {
1496            message: "the certificate signing request's subject has more than one common name"
1497                .into(),
1498        });
1499    }
1500    let common_name = subject
1501        .iter_common_name()
1502        .next()
1503        .and_then(|cn| cn.as_str().ok())
1504        .map(str::to_string)
1505        .filter(|cn| !cn.trim().is_empty())
1506        .ok_or_else(|| AxiamError::Validation {
1507            message: "the certificate signing request has no common name in its subject".into(),
1508        })?;
1509    // #649 — the other way a name reaches a certificate. The name is the CSR's
1510    // and is not rewritten (a remote signer signs these very bytes), so it is
1511    // refused, not repaired, by the rules a request's `subject` meets.
1512    crate::subject::check_common_name_value(
1513        &common_name,
1514        "the certificate signing request's common name",
1515    )?;
1516    // R110W1-02 — and so is every other attribute of the subject. The local
1517    // signing paths keep nothing of the subject but the common name (the DN is
1518    // rebuilt from it), so an `O=`/`OU=` is dropped there; but `sign-verbatim`
1519    // writes them into the certificate as given, so a newline or a bidi
1520    // override in an `OU` would reach an issued subject exactly as #649 closed
1521    // for the CN. Refused on every custodian, so the same CSR is not accepted
1522    // on one and refused on another, by the same value rules and not by a
1523    // narrower list: an ordinary `openssl req -subj /C=…/O=…/CN=…` request,
1524    // which is signed today, keeps being signed. A value that is not a string
1525    // type cannot be checked and is refused (fail closed).
1526    for attr in subject.iter().flat_map(|rdn| rdn.iter()) {
1527        if attr.attr_type() == &x509_parser::oid_registry::OID_X509_COMMON_NAME {
1528            continue;
1529        }
1530        let what = format!(
1531            "the certificate signing request's subject attribute {}",
1532            attr.attr_type().to_id_string()
1533        );
1534        let value = attr.as_str().map_err(|_| AxiamError::Validation {
1535            message: format!("{what} is not a text string"),
1536        })?;
1537        crate::subject::check_common_name_value(value, &what)?;
1538    }
1539
1540    let spki = &csr.certification_request_info.subject_pki;
1541    let (key_algorithm, rsa_modulus_bits) = match spki.algorithm.algorithm.to_id_string().as_str() {
1542        // RFC 8410 §3 — id-Ed25519.
1543        "1.3.101.112" => (KeyAlgorithm::Ed25519, None),
1544        // RFC 8017 — rsaEncryption. The modulus is read here and checked by the
1545        // caller: `KeyAlgorithm::Rsa4096` is a label and not a measurement, and
1546        // `parse_ca_certificate` above maps *any* RSA OID onto it deliberately
1547        // (an imported root of another size is still a usable root). A path
1548        // that enforces a floor cannot reuse that mapping.
1549        "1.2.840.113549.1.1.1" => {
1550            let bits = match spki.parsed() {
1551                Ok(x509_parser::public_key::PublicKey::RSA(rsa)) => rsa.key_size(),
1552                // A key whose OID says RSA but whose bytes will not parse as an
1553                // RSA key reports zero, which fails every floor. Fail closed:
1554                // the alternative is treating "unreadable" as "large enough".
1555                _ => 0,
1556            };
1557            (KeyAlgorithm::Rsa4096, Some(bits))
1558        }
1559        other => {
1560            return Err(AxiamError::Validation {
1561                message: format!(
1562                    "the certificate signing request carries a {other} public key; AXIAM signs \
1563                     Ed25519 and RSA keys only"
1564                ),
1565            });
1566        }
1567    };
1568
1569    let mut refused_extensions = Vec::new();
1570    if let Some(exts) = csr.requested_extensions() {
1571        use x509_parser::extensions::ParsedExtension;
1572        for ext in exts {
1573            match ext {
1574                ParsedExtension::SubjectAlternativeName(_) => {
1575                    refused_extensions.push("subjectAltName")
1576                }
1577                ParsedExtension::KeyUsage(_) => refused_extensions.push("keyUsage"),
1578                ParsedExtension::ExtendedKeyUsage(_) => refused_extensions.push("extendedKeyUsage"),
1579                _ => {}
1580            }
1581        }
1582        refused_extensions.dedup();
1583    }
1584
1585    Ok(CsrFacts {
1586        common_name,
1587        key_algorithm,
1588        rsa_modulus_bits,
1589        refused_extensions,
1590    })
1591}

1597/// the point: two possession checks are two places for one to be forgotten.
1598fn csr_common_name(pem: &str) -> AxiamResult<String> {

1612/// Read an imported CA certificate, refusing anything AXIAM cannot use as one.
1613///
1614/// Three checks, and each exists because skipping it produces a CA that fails
1615/// later in a way that names nothing:
1616///
1617/// 1. It parses as X.509 at all.
1618/// 2. Its Basic Constraints say `CA:TRUE`. A leaf certificate imported as a CA
1619///    would be offered in the issuing-CA dropdown and rejected by every relying
1620///    party that checked the chain.
1621/// 3. Its public key is one AXIAM can issue against. `KeyAlgorithm` has two
1622///    variants, and a P-256 CA stored as "Ed25519" would mislead every reader
1623///    of the row.
1624fn parse_ca_certificate(pem: &str) -> AxiamResult<ParsedCa> {
1625    let (_, der) =
1626        x509_parser::pem::parse_x509_pem(pem.as_bytes()).map_err(|e| AxiamError::Validation {
1627            message: format!(
1628                "the supplied CA certificate is not a PEM-encoded X.509 certificate: {e}"
1629            ),
1630        })?;
1631    let (_, cert) =
1632        X509Certificate::from_der(&der.contents).map_err(|e| AxiamError::Validation {
1633            message: format!("the supplied CA certificate could not be parsed: {e}"),
1634        })?;
1635
1636    let is_ca = cert
1637        .basic_constraints()
1638        .ok()
1639        .flatten()
1640        .map(|bc| bc.value.ca)
1641        .unwrap_or(false);
1642    if !is_ca {
1643        return Err(AxiamError::Validation {
1644            message: "the supplied certificate is not a CA: its Basic Constraints extension does \
1645             not say CA:TRUE, so nothing that validates a chain would accept a \
1646             certificate it signed"
1647                .into(),
1648        });
1649    }
1650
1651    let key_algorithm = match cert
1652        .public_key()
1653        .algorithm
1654        .algorithm
1655        .to_id_string()
1656        .as_str()
1657    {
1658        // RFC 8410 §3 — id-Ed25519.
1659        "1.3.101.112" => KeyAlgorithm::Ed25519,
1660        // RFC 8017 — rsaEncryption. Import is the only way an RSA CA reaches
1661        // AXIAM: rcgen's `ring` backend cannot *generate* RSA keys (see
1662        // `crypto::generate_keypair`), so `POST .../ca-certificates` with
1663        // `Rsa4096` fails, while an RSA root from an existing PKI is exactly
1664        // the thing this endpoint is for.
1665        //
1666        // The variant is named for the size AXIAM would generate; an imported
1667        // RSA CA of another size is still RSA and still usable, and refusing it
1668        // would reject a perfectly good existing root over a label.
1669        "1.2.840.113549.1.1.1" => KeyAlgorithm::Rsa4096,
1670        other => {
1671            return Err(AxiamError::Validation {
1672                message: format!(
1673                    "the supplied CA uses public key algorithm {other}, which AXIAM cannot \
1674                 issue against — only Ed25519 and RSA are supported"
1675                ),
1676            });
1677        }
1678    };
1679
1680    let subject = cert
1681        .subject()
1682        .iter_common_name()
1683        .next()
1684        .and_then(|cn| cn.as_str().ok())
1685        .map(str::to_string)
1686        // A CA with no CN is unusual but legal; the full DN is a truthful
1687        // stand-in and better than refusing the import over a display string.
1688        .unwrap_or_else(|| cert.subject().to_string());
1689
1690    let to_utc = |t: ASN1Time, what: &str| -> AxiamResult<DateTime<Utc>> {
1691        DateTime::from_timestamp(t.timestamp(), 0).ok_or_else(|| AxiamError::Validation {
1692            message: format!("the supplied CA's {what} is out of range"),
1693        })
1694    };
1695    let not_before = to_utc(cert.validity().not_before, "notBefore")?;
1696    let not_after = to_utc(cert.validity().not_after, "notAfter")?;
1697    if not_after <= not_before {
1698        return Err(AxiamError::Validation {
1699            message: "the supplied CA's validity window ends before it starts".into(),
1700        });
1701    }
1702    if not_after <= Utc::now() {
1703        return Err(AxiamError::Validation {
1704            message: "the supplied CA has already expired: it could be imported but could never \
1705             issue a certificate, since leaf validity is capped to the CA's"
1706                .into(),
1707        });
1708    }
1709
1710    Ok(ParsedCa {
1711        subject,
1712        fingerprint: compute_fingerprint(&der.contents),
1713        key_algorithm,
1714        not_before,
1715        not_after,
1716    })
1717}
1718
1719/// Prove the supplied private key is the one this certificate was issued with.
1720///
1721/// By comparing SubjectPublicKeyInfo: the DER rcgen derives from the private
1722/// key against the DER the certificate carries. A private key determines its
1723/// public key, so equal SPKI is exactly the question being asked.
1724///
1725/// Note what this deliberately does *not* rely on:
1726/// `rcgen::Issuer::from_ca_cert_pem` accepts a key and a certificate that have
1727/// nothing to do with each other — it parses the certificate and pairs it with
1728/// whatever key it was handed, deferring the mismatch to a signature that no
1729/// relying party will verify. Using it as the check here would have looked
1730/// right and passed everything.
1731///
1732/// Without this, a mismatched pair produces a CA that lists as issuable and
1733/// dies at the first signature, with an error about key parsing that names
1734/// nothing the operator did.
1735fn verify_key_matches_certificate(key_pem: &str, cert_pem: &str) -> AxiamResult<()> {
1736    let key_pair = rcgen::KeyPair::from_pem(key_pem).map_err(|e| AxiamError::Validation {
1737        message: format!("the supplied private key is not a PEM-encoded key AXIAM can read: {e}"),
1738    })?;
1739
1740    let (_, pem) = x509_parser::pem::parse_x509_pem(cert_pem.as_bytes()).map_err(|e| {
1741        AxiamError::Validation {
1742            message: format!(
1743                "the supplied CA certificate is not a PEM-encoded X.509 certificate: {e}"
1744            ),
1745        }
1746    })?;
1747    let (_, cert) =
1748        X509Certificate::from_der(&pem.contents).map_err(|e| AxiamError::Validation {
1749            message: format!("the supplied CA certificate could not be parsed: {e}"),
1750        })?;
1751
1752    // rcgen exposes the SPKI as PEM; `x509_parser` gives the certificate's as
1753    // DER. Decode the former rather than re-encoding the latter, so the
1754    // comparison is over the bytes a parser produced on both sides.
1755    let key_spki = pem::parse(key_pair.public_key_pem()).map_err(|e| {
1756        AxiamError::Internal(format!(
1757            "could not re-encode the supplied key's public half: {e}"
1758        ))
1759    })?;
1760
1761    if key_spki.contents() != cert.public_key().raw {
1762        return Err(AxiamError::Validation {
1763            message: "the supplied private key does not match the supplied CA certificate".into(),
1764        });
1765    }
1766    Ok(())
1767}

1774    /// Build a self-signed CA the way an external PKI would hand one over.

1776        let key_pair = KeyPair::generate_for(&rcgen::PKCS_ED25519).unwrap();
1777        let mut params = CertificateParams::new(Vec::<String>::new()).unwrap();

1784    }
