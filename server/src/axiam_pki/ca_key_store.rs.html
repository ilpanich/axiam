1//! Custodians for CA signing keys — the implementations behind
2//! [`axiam_core::ca_keys::CaKeyStore`].
3//!
4//! Three of them today:
5//!
6//! * [`DatabaseCaKeyStore`] — AES-256-GCM ciphertext in the `ca_certificate`
7//!   row, sealed under the process-wide `pki_encryption_key`. What AXIAM has
8//!   always done, and still the default.
9//! * [`VaultCaKeyStore`] — a HashiCorp Vault KV v2 secret. The row holds a path
10//!   and no key material.
11//! * [`VaultPkiCaKeyStore`](crate::vault_pki::VaultPkiCaKeyStore) — Vault's PKI
12//!   secrets engine, where the key is *generated* and never leaves. Its own
13//!   module, because it is the one custodian that also generates CAs and signs
14//!   with them.
15//!
16//! A KMS is the next one, and the port is shaped so it slots in beside these
17//! rather than replacing them: a deployment that adopts a new custodian keeps
18//! the CAs it already had, because each row records which custodian holds its
19//! key.
20

40/// The CA key sealed into its own row with AES-256-GCM.
41///
42/// The control this gives is real but bounded, and worth stating plainly: the
43/// key and the thing that opens it are in the same blast radius. Whoever holds
44/// a database dump *and* one process's `pki_encryption_key` holds every CA in
45/// the deployment, and no record anywhere says they read it.
46#[derive(Clone)]
47pub struct DatabaseCaKeyStore {
48    encryption_key: [u8; 32],
49}
50
51impl DatabaseCaKeyStore {
52    pub fn new(encryption_key: [u8; 32]) -> Self {
53        Self { encryption_key }
54    }
55}
56
57impl CaKeyStore for DatabaseCaKeyStore {
58    fn store<'a>(
59        &'a self,
60        _organization_id: Uuid,
61        _ca_id: Uuid,
62        private_key_pem: &'a str,
63    ) -> Pin<Box<dyn Future<Output = AxiamResult<StoredCaKey>> + Send + 'a>> {
64        Box::pin(async move {
65            let ciphertext = encrypt_secret(private_key_pem.as_bytes(), &self.encryption_key)?;
66            Ok(StoredCaKey::Inline(ciphertext))
67        })
68    }
69
70    fn load<'a>(
71        &'a self,
72        _key_ref: &'a CaKeyRef,
73        inline: Option<&'a [u8]>,
74    ) -> Pin<Box<dyn Future<Output = AxiamResult<Zeroizing<String>>> + Send + 'a>> {
75        Box::pin(async move {
76            let ciphertext = inline.ok_or_else(|| {
77                AxiamError::Certificate("CA certificate has no stored private key".into())
78            })?;
79            let plaintext = decrypt_secret(ciphertext, &self.encryption_key)?;
80            let pem = String::from_utf8(plaintext).map_err(|_| {
81                // A wrong key produces an authentication failure inside
82                // `decrypt_secret`, not this — so reaching here means the
83                // ciphertext decrypted to bytes that are not a PEM, which is
84                // corruption rather than a key mix-up.
85                AxiamError::Crypto("decrypted CA private key is not valid UTF-8".into())
86            })?;
87            Ok(Zeroizing::new(pem))
88        })
89    }
90
91    fn delete<'a>(
92        &'a self,
93        _key_ref: &'a CaKeyRef,
94    ) -> Pin<Box<dyn Future<Output = AxiamResult<()>> + Send + 'a>> {
95        // The material lives in the row; removing the row removes it.
96        Box::pin(async { Ok(()) })
97    }
98
99    fn custody(&self) -> CaKeyCustody {
100        CaKeyCustody::Database
101    }
102
103    fn describe(&self) -> &'static str {
104        "database"
105    }
106}

112/// How to reach Vault for CA key custody.
113///
114/// Deliberately separate from `axiam_auth::secrets::VaultConfig` even though
115/// both address a KV v2 engine. That one names *one path* holding the
116/// deployment's fixed startup secrets as fields; this one names a *prefix*
117/// under which one secret per CA is created at runtime. Sharing a type would
118/// mean one `path` field meaning two different things, and a policy that had to
119/// grant both uses to one token.
120///
121/// `Debug` is hand-written to redact the token — this struct is exactly the
122/// kind of thing that reaches a startup log line or a panic message.
123#[derive(Clone, PartialEq, Eq)]
124pub struct VaultCaKeyConfig {
125    /// Base address, e.g. `https://vault.internal:8200`.
126    pub address: String,
127    /// A token with create/read/delete on `mount`/`prefix`/*.
128    pub token: String,
129    /// KV v2 mount point. Vault's default is `secret`.
130    pub mount: String,
131    /// Path prefix under the mount. One secret is written per CA beneath it.
132    pub prefix: String,
133    /// PEM bundle for the CA that issued Vault's listener certificate.
134    ///
135    /// `None` verifies against the built-in roots, which is right only for a
136    /// Vault fronted by a publicly-trusted certificate. reqwest is built with
137    /// `rustls-tls`, whose roots are compiled in, so an internally-issued
138    /// listener certificate has no `SSL_CERT_FILE` to fall back on and the
139    /// handshake simply fails.
140    pub ca_cert_path: Option<std::path::PathBuf>,
141}
142
143impl std::fmt::Debug for VaultCaKeyConfig {
144    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
145        f.debug_struct("VaultCaKeyConfig")
146            .field("address", &self.address)
147            .field("token", &"<redacted>")
148            .field("mount", &self.mount)
149            .field("prefix", &self.prefix)
150            .field("ca_cert_path", &self.ca_cert_path)
151            .finish()
152    }
153}
154
155/// The field name inside the Vault secret. One field, one key.
156const VAULT_KEY_FIELD: &str = "private_key_pem";
157
158/// How long to wait on Vault before giving up.
159///
160/// Bounded because this runs on the request that issues a certificate, not at
161/// startup like the secret provider: an unbounded wait would turn a Vault
162/// outage into hung request handlers rather than a refused issuance.
163const VAULT_TIMEOUT: Duration = Duration::from_secs(10);
164
165/// CA keys held in HashiCorp Vault's KV v2 engine.
166///
167/// Each CA gets its own secret at `<mount>/data/<prefix>/<org_id>/<ca_id>`,
168/// with the PEM in a single `private_key_pem` field. One secret per CA rather
169/// than one secret with a field per CA, because Vault policy paths are the unit
170/// of access control: an operator who wants to grant read on one organization's
171/// CAs and not another's can write that as a path glob, and could not write it
172/// at all if every key shared one secret.
173///
174/// The key still reaches AXIAM's memory to sign with — see the port's module
175/// docs for why that is custody rather than the stronger property, and what
176/// would be needed for the stronger one.
177#[derive(Clone)]
178pub struct VaultCaKeyStore {
179    client: Client,
180    config: VaultCaKeyConfig,
181}
182
183impl std::fmt::Debug for VaultCaKeyStore {
184    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
185        f.debug_struct("VaultCaKeyStore")
186            .field("config", &self.config)
187            .finish()
188    }
189}

191impl VaultCaKeyStore {
192    /// Build a store, including the TLS trust anchor if one is configured.
193    ///
194    /// Fails rather than falling back to the built-in roots when the anchor is
195    /// unreadable: silently verifying against a different trust store than the
196    /// operator asked for is how a misconfiguration becomes a connection to
197    /// something that is not their Vault.
198    pub fn new(config: VaultCaKeyConfig) -> AxiamResult<Self> {
199        let mut builder = Client::builder().timeout(VAULT_TIMEOUT);
200
201        if let Some(ref path) = config.ca_cert_path {
202            let pem = std::fs::read(path).map_err(|e| {
203                AxiamError::Internal(format!(
204                    "vault CA key store: could not read the trust anchor at {}: {e}",
205                    path.display()
206                ))
207            })?;
208            let cert = reqwest::Certificate::from_pem(&pem).map_err(|e| {
209                AxiamError::Internal(format!(
210                    "vault CA key store: {} is not a PEM certificate: {e}",
211                    path.display()
212                ))
213            })?;
214            builder = builder.add_root_certificate(cert);
215        }
216
217        let client = builder.build().map_err(|e| {
218            AxiamError::Internal(format!(
219                "vault CA key store: could not build HTTP client: {e}"
220            ))
221        })?;
222
223        Ok(Self { client, config })
224    }
225
226    /// Build from an already-configured client. For tests, and for a caller
227    /// that has its own client policy.
228    pub fn with_client(client: Client, config: VaultCaKeyConfig) -> Self {
229        Self { client, config }
230    }
231
232    /// The KV v2 *data* URL for one CA's secret.
233    ///
234    /// Ids are UUIDs and cannot contain a path separator, so there is nothing
235    /// to escape — but the locator is built here once and stored, so a future
236    /// change to the layout cannot strand keys written under the old one.
237    fn data_url(&self, organization_id: Uuid, ca_id: Uuid) -> String {
238        format!(
239            "{}/v1/{}/data/{}/{organization_id}/{ca_id}",
240            self.config.address.trim_end_matches('/'),
241            self.config.mount.trim_matches('/'),
242            self.config.prefix.trim_matches('/'),
243        )
244    }
245
246    /// The KV v2 *metadata* URL, whose DELETE removes every version.
247    ///
248    /// A KV v2 `DELETE` on the data path only marks the latest version deleted
249    /// and leaves it readable by anyone who can ask for a version number. For a
250    /// CA signing key that is not deletion.
251    fn metadata_url(&self, locator: &str) -> String {
252        format!(
253            "{}/v1/{}/metadata/{}",
254            self.config.address.trim_end_matches('/'),
255            self.config.mount.trim_matches('/'),
256            locator.trim_matches('/'),
257        )
258    }
259
260    /// The stored locator: the path *within the mount*, no address, no mount.
261    ///
262    /// Storing it that way means moving Vault to a new address, or remounting
263    /// the engine, does not invalidate every CA row.
264    fn locator(&self, organization_id: Uuid, ca_id: Uuid) -> String {
265        format!(
266            "{}/{organization_id}/{ca_id}",
267            self.config.prefix.trim_matches('/')
268        )
269    }
270
271    fn read_url(&self, locator: &str) -> String {
272        format!(
273            "{}/v1/{}/data/{}",
274            self.config.address.trim_end_matches('/'),
275            self.config.mount.trim_matches('/'),
276            locator.trim_matches('/'),
277        )
278    }
279}

281impl CaKeyStore for VaultCaKeyStore {
282    fn store<'a>(
283        &'a self,
284        organization_id: Uuid,
285        ca_id: Uuid,
286        private_key_pem: &'a str,
287    ) -> Pin<Box<dyn Future<Output = AxiamResult<StoredCaKey>> + Send + 'a>> {
288        Box::pin(async move {
289            let url = self.data_url(organization_id, ca_id);
290            let body = serde_json::json!({
291                "data": { VAULT_KEY_FIELD: private_key_pem },
292                // Refuse to overwrite: a CA id is freshly generated, so an
293                // existing secret at this path means either an id collision or
294                // a retry of a call that already succeeded. Silently replacing
295                // would destroy a live signing key.
296                "options": { "cas": 0 },
297            });
298
299            let response = self
300                .client
301                .post(&url)
302                .header("X-Vault-Token", &self.config.token)
303                .json(&body)
304                .send()
305                .await
306                // Deliberately carries neither the token nor the key.
307                .map_err(|e| {
308                    AxiamError::Internal(format!("vault: writing the CA key to {url} failed: {e}"))
309                })?;
310
311            if !response.status().is_success() {
312                return Err(AxiamError::Internal(format!(
313                    "vault: {url} answered {} on write — check the token's policy \
314                     has create and update on this path",
315                    response.status()
316                )));
317            }
318
319            Ok(StoredCaKey::Referenced(
320                self.locator(organization_id, ca_id),
321            ))
322        })
323    }
324
325    fn load<'a>(
326        &'a self,
327        key_ref: &'a CaKeyRef,
328        _inline: Option<&'a [u8]>,
329    ) -> Pin<Box<dyn Future<Output = AxiamResult<Zeroizing<String>>> + Send + 'a>> {
330        Box::pin(async move {
331            let url = self.read_url(&key_ref.locator);
332            let response = self
333                .client
334                .get(&url)
335                .header("X-Vault-Token", &self.config.token)
336                .send()
337                .await
338                .map_err(|e| {
339                    AxiamError::Internal(format!(
340                        "vault: reading the CA key from {url} failed: {e}"
341                    ))
342                })?;
343
344            if response.status() == reqwest::StatusCode::NOT_FOUND {
345                // Distinguished from a transport failure because it means
346                // something different: the CA row points at a secret that is
347                // not there, which is a custody problem an operator has to fix,
348                // not something a retry will resolve.
349                return Err(AxiamError::Certificate(format!(
350                    "vault holds no key for this CA at {}: it was deleted, or the \
351                     mount was changed after the CA was created",
352                    key_ref.locator
353                )));
354            }
355            if !response.status().is_success() {
356                return Err(AxiamError::Internal(format!(
357                    "vault: {url} answered {} on read — check the token's policy \
358                     has read on this path",
359                    response.status()
360                )));
361            }
362
363            let body: serde_json::Value = response.json().await.map_err(|e| {
364                AxiamError::Internal(format!("vault: {url} returned non-JSON: {e}"))
365            })?;
366
367            // KV v2 nests under data.data; KV v1 does not. Accept both, as the
368            // startup secret provider does, so a deployment on the older engine
369            // is not told its key is absent.
370            let pem = body
371                .get("data")
372                .and_then(|d| d.get("data").or(Some(d)))
373                .and_then(|d| d.get(VAULT_KEY_FIELD))
374                .and_then(|v| v.as_str())
375                .ok_or_else(|| {
376                    AxiamError::Certificate(format!(
377                        "vault: the secret at {} has no `{VAULT_KEY_FIELD}` field",
378                        key_ref.locator
379                    ))
380                })?;
381
382            Ok(Zeroizing::new(pem.to_string()))
383        })
384    }
385
386    fn delete<'a>(
387        &'a self,
388        key_ref: &'a CaKeyRef,
389    ) -> Pin<Box<dyn Future<Output = AxiamResult<()>> + Send + 'a>> {
390        Box::pin(async move {
391            let url = self.metadata_url(&key_ref.locator);
392            let response = self
393                .client
394                .delete(&url)
395                .header("X-Vault-Token", &self.config.token)
396                .send()
397                .await
398                .map_err(|e| {
399                    AxiamError::Internal(format!("vault: deleting the CA key at {url} failed: {e}"))
400                })?;
401
402            // Already gone is success: this is called on revocation, and a
403            // second revocation must not fail because the first one worked.
404            if response.status().is_success() || response.status() == reqwest::StatusCode::NOT_FOUND
405            {
406                return Ok(());
407            }
408            Err(AxiamError::Internal(format!(
409                "vault: {url} answered {} on delete",
410                response.status()
411            )))
412        })
413    }
414
415    fn custody(&self) -> CaKeyCustody {
416        CaKeyCustody::Vault
417    }
418
419    fn describe(&self) -> &'static str {
420        "vault"
421    }
422}

428/// A CA whose private key AXIAM does not hold.
429///
430/// Not a custodian so much as the absence of one, expressed as a type so the
431/// signing path can ask the same question of every CA and get an answer that
432/// says *why* rather than a bare "no key".
433#[derive(Clone, Debug, Default)]
434pub struct ExternalCaKeyStore;
435
436impl CaKeyStore for ExternalCaKeyStore {
437    fn store<'a>(
438        &'a self,
439        _organization_id: Uuid,
440        _ca_id: Uuid,
441        _private_key_pem: &'a str,
442    ) -> Pin<Box<dyn Future<Output = AxiamResult<StoredCaKey>> + Send + 'a>> {
443        Box::pin(async {
444            Err(AxiamError::Validation {
445                message: "this CA's private key is held outside AXIAM and cannot be \
446                          stored here"
447                    .into(),
448            })
449        })
450    }
451
452    fn load<'a>(
453        &'a self,
454        _key_ref: &'a CaKeyRef,
455        _inline: Option<&'a [u8]>,
456    ) -> Pin<Box<dyn Future<Output = AxiamResult<Zeroizing<String>>> + Send + 'a>> {
457        Box::pin(async {
458            // `Validation`, not `Certificate`: this is a statement about the
459            // request — the caller picked a CA that holds no key — and it has
460            // to reach them. `Certificate` maps to a generic 500 whose body
461            // says only "an internal error occurred".
462            Err(AxiamError::Validation {
463                message: "this CA was imported without a private key: AXIAM can trust \
464                          certificates it signed but cannot issue new ones against it"
465                    .into(),
466            })
467        })
468    }
469
470    fn delete<'a>(
471        &'a self,
472        _key_ref: &'a CaKeyRef,
473    ) -> Pin<Box<dyn Future<Output = AxiamResult<()>> + Send + 'a>> {
474        Box::pin(async { Ok(()) })
475    }
476
477    fn custody(&self) -> CaKeyCustody {
478        CaKeyCustody::External
479    }
480
481    fn describe(&self) -> &'static str {
482        "external"
483    }
484}

686/// Every custodian this process can reach, dispatching on what a CA row says.
687///
688/// The reason this exists rather than the composition root simply choosing one
689/// store: a deployment that adopts Vault does not thereby move the CAs it
690/// already has. Those rows still say `database`, their keys are still sealed
691/// into them, and a signing path that consulted only the *current* setting
692/// would fail to find them.
693///
694/// So: new CAs go to [`Self::default_store`], the one configured as current,
695/// and reads go to whichever custodian the row names. Switching custodian is
696/// then a decision about new CAs only, and migrating an old one is a deliberate
697/// separate act rather than something that happens by restarting with a
698/// different environment.
699#[derive(Clone)]
700pub struct CaKeyCustodians {
701    database: Option<DatabaseCaKeyStore>,
702    vault: Option<VaultCaKeyStore>,
703    vault_pki: Option<VaultPkiCaKeyStore>,
704    external: ExternalCaKeyStore,
705    /// Whether the Vault custodians were built from the secret provider's
706    /// address and token rather than the PKI-specific pair.
707    ///
708    /// Reported by the composition root at startup so an operator who never set
709    /// `AXIAM__PKI__VAULT_ADDR` can see *why* their CA keys are in Vault, and
710    /// so the reverse case — a deployment expecting Vault and getting the
711    /// database — is a line in the log rather than a discovery months later.
712    pub(crate) vault_inherited: bool,
713    /// `None` when a deployment has configured no custodian at all.
714    ///
715    /// That is a legitimate state, not a misconfiguration: a deployment that
716    /// issues no certificates — no CAs, no mTLS — has no reason to hold a PKI
717    /// key, and refusing to start would break it for a feature it does not
718    /// use. Generating a CA is then what fails, with a message naming what to
719    /// set, which is where the operator actually is when they need it.
720    default_custody: Option<CaKeyCustody>,
721}
722
723impl std::fmt::Debug for CaKeyCustodians {
724    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
725        f.debug_struct("CaKeyCustodians")
726            .field("database", &self.database.is_some())
727            .field("vault", &self.vault.is_some())
728            .field("vault_pki", &self.vault_pki.is_some())
729            .field("default", &self.default_custody)
730            .finish()
731    }
732}
733
734impl CaKeyCustodians {
735    /// Build the set, choosing `default_custody` for new CAs.
736    ///
737    /// `None` means nothing is configured, which is allowed and is what a
738    /// deployment that issues no certificates has. `Some(..)` is an operator
739    /// asking for a specific custodian, and *that* fails here when the
740    /// custodian is not actually available — an operator who configured Vault
741    /// and silently got database custody would have exactly the property they
742    /// were trying to stop having.
743    ///
744    /// The distinction matters more than it looks. Making "nothing configured"
745    /// a startup failure refuses to boot every deployment that has no CAs at
746    /// all, for a feature it never touches; making an *explicit* choice fail
747    /// silently is how a security control quietly stops applying. Only the
748    /// second is worth stopping the process for.
749    pub fn new(
750        database: Option<DatabaseCaKeyStore>,
751        vault: Option<VaultCaKeyStore>,
752        vault_pki: Option<VaultPkiCaKeyStore>,
753        default_custody: Option<CaKeyCustody>,
754    ) -> AxiamResult<Self> {
755        if let Some(requested) = default_custody {
756            let available = match requested {
757                CaKeyCustody::Database => database.is_some(),
758                CaKeyCustody::Vault => vault.is_some(),
759                CaKeyCustody::VaultPki => vault_pki.is_some(),
760                // Never a default: it would mean generating CAs whose keys are
761                // discarded, which is a CA that can sign nothing.
762                CaKeyCustody::External => false,
763            };
764            if !available {
765                return Err(AxiamError::Internal(format!(
766                    "CA key custody is set to `{requested}` but that custodian is not \
767                     configured: database custody needs AXIAM__PKI__ENCRYPTION_KEY, vault \
768                     and vault_pki custody need AXIAM__PKI__VAULT_ADDR and \
769                     AXIAM__PKI__VAULT_TOKEN"
770                )));
771            }
772        }
773        Ok(Self {
774            database,
775            vault,
776            vault_pki,
777            external: ExternalCaKeyStore,
778            vault_inherited: false,
779            default_custody,
780        })
781    }
782
783    /// Which custodian new CAs are created with, or `None` if none is configured.
784    pub fn default_custody(&self) -> Option<CaKeyCustody> {
785        self.default_custody
786    }
787
788    /// Whether the Vault custodians were inherited from the secret provider's
789    /// configuration rather than named by the PKI-specific variables.
790    pub fn vault_inherited(&self) -> bool {
791        self.vault_inherited
792    }
793
794    /// Whether new CA signing keys are being sealed into the database while a
795    /// Vault custodian was actually available to hold them.
796    ///
797    /// The one arrangement nobody chooses on purpose, and the one this codebase
798    /// shipped a beta with. It can only happen now by an operator writing
799    /// `AXIAM__PKI__CA_KEY_STORE=database` next to a working Vault, so it is a
800    /// warning rather than a refusal — but a loud one, because the difference
801    /// between the two is whether a database dump hands over every CA in the
802    /// deployment.
803    pub fn database_custody_despite_vault(&self) -> bool {
804        self.default_custody == Some(CaKeyCustody::Database) && self.vault.is_some()
805    }
806
807    /// The custodian for a new CA.
808    ///
809    /// The failure an operator meets when they generate their first CA on a
810    /// deployment that never configured PKI — which is where the message about
811    /// what to set is most use, and long after the process would have refused
812    /// to start.
813    pub fn default_store(&self) -> AxiamResult<&dyn CaKeyStore> {
814        let custody = self.default_custody.ok_or_else(|| {
815            AxiamError::Internal(
816                "no CA key custodian is configured, so a CA signing key cannot be \
817                 stored: set AXIAM__PKI__ENCRYPTION_KEY to keep it encrypted in the \
818                 database, or AXIAM__PKI__VAULT_ADDR and AXIAM__PKI__VAULT_TOKEN to \
819                 keep it in Vault"
820                    .into(),
821            )
822        })?;
823        self.store_for(custody)
824    }
825
826    /// The custodian a given CA row names.
827    ///
828    /// A row naming a custodian this process cannot reach is an error that says
829    /// which one and what configures it — the shape an operator meets after
830    /// restoring a database into a deployment configured differently, and the
831    /// one moment a clear message saves an afternoon.
832    pub fn store_for(&self, custody: CaKeyCustody) -> AxiamResult<&dyn CaKeyStore> {
833        match custody {
834            CaKeyCustody::Database => self
835                .database
836                .as_ref()
837                .map(|s| s as &dyn CaKeyStore)
838                .ok_or_else(|| {
839                    AxiamError::Internal(
840                        "this CA's key is sealed into its row, but AXIAM__PKI__ENCRYPTION_KEY \
841                         is not set — the key cannot be opened"
842                            .into(),
843                    )
844                }),
845            CaKeyCustody::Vault => self
846                .vault
847                .as_ref()
848                .map(|s| s as &dyn CaKeyStore)
849                .ok_or_else(|| {
850                    AxiamError::Internal(
851                        "this CA's key is in Vault, but no Vault is configured — set \
852                         AXIAM__PKI__VAULT_ADDR and AXIAM__PKI__VAULT_TOKEN"
853                            .into(),
854                    )
855                }),
856            CaKeyCustody::VaultPki => self
857                .vault_pki
858                .as_ref()
859                .map(|s| s as &dyn CaKeyStore)
860                .ok_or_else(|| {
861                    AxiamError::Internal(
862                        "this CA's key is in Vault's PKI engine, but no Vault is \
863                         configured — set AXIAM__PKI__VAULT_ADDR and \
864                         AXIAM__PKI__VAULT_TOKEN"
865                            .into(),
866                    )
867                }),
868            CaKeyCustody::External => Ok(&self.external),
869        }
870    }
871}

1306/// The Vault a deployment reaches for CA key custody, and where that was said.
1307struct VaultEndpoint {
1308    address: Option<String>,
1309    token: Option<String>,
1310    ca_cert_path: Option<std::path::PathBuf>,
1311    /// True when the endpoint came from the secret provider's variables rather
1312    /// than the PKI-specific ones. Carried so the composition root can say so
1313    /// in its startup log: an operator who never set `AXIAM__PKI__VAULT_ADDR`
1314    /// and now finds their CA keys in Vault should be able to read why.
1315    inherited: bool,
1316}
1317
1318impl std::fmt::Debug for VaultEndpoint {
1319    /// Redacts the token.
1320    ///
1321    /// Hand-written rather than derived for the usual reason and one specific
1322    /// one: this type exists to be assembled at startup and asserted on in
1323    /// tests, so every `unwrap`, `assert_eq!` and panic message is a place a
1324    /// derived impl would print a live Vault token into a log an operator is
1325    /// about to paste into an issue.
1326    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
1327        f.debug_struct("VaultEndpoint")
1328            .field("address", &self.address)
1329            .field("token", &self.token.as_ref().map(|_| "[REDACTED]"))
1330            .field("ca_cert_path", &self.ca_cert_path)
1331            .field("inherited", &self.inherited)
1332            .finish()
1333    }
1334}
1335
1336/// Resolve the Vault endpoint for CA key custody.
1337///
1338/// The PKI-specific pair wins outright. Otherwise the secret provider's pair is
1339/// inherited, because one Vault is the overwhelmingly common deployment and
1340/// making an operator address it twice produced a beta in which every CA
1341/// signing key sat in the database behind a log line that said `vault`.
1342///
1343/// A half-configured pair is refused rather than ignored, and the message names
1344/// the pair the operator actually touched — being told to set
1345/// `AXIAM__PKI__VAULT_TOKEN` when you set `AXIAM__AUTH__VAULT_ADDR` is a worse
1346/// error than none.
1347///
1348/// Takes a lookup rather than reading the environment directly so the
1349/// precedence rules can be tested as a function of their inputs. Driving them
1350/// through the real environment would mean mutating process-global state from
1351/// tests the harness runs on many threads at once — which is how a suite
1352/// starts passing or failing by run order.
1353fn vault_endpoint_from(lookup: &dyn Fn(&str) -> Option<String>) -> AxiamResult<VaultEndpoint> {
1354    let read = |name: &str| lookup(name).filter(|v| !v.trim().is_empty());
1355    let path = |name: &str| read(name).map(std::path::PathBuf::from);
1356
1357    // `(addr, token, cert, inherited, addr_env, token_env)` for whichever pair
1358    // is in play.
1359    let (address, token, ca_cert_path, inherited, addr_env, token_env) =
1360        match (read(CA_VAULT_ADDR_ENV), read(CA_VAULT_TOKEN_ENV)) {
1361            // Nothing PKI-specific at all: inherit the secret provider's Vault.
1362            (None, None) => (
1363                read(AUTH_VAULT_ADDR_ENV),
1364                read(AUTH_VAULT_TOKEN_ENV),
1365                path(AUTH_VAULT_CA_CERT_ENV),
1366                true,
1367                AUTH_VAULT_ADDR_ENV,
1368                AUTH_VAULT_TOKEN_ENV,
1369            ),
1370            // A PKI-specific Vault, complete or half-said. Its own CA-cert
1371            // variable falls back to the secret provider's, since a separate
1372            // token against the same listener still needs the same trust anchor.
1373            (addr, tok) => (
1374                addr,
1375                tok,
1376                path(CA_VAULT_CA_CERT_ENV).or_else(|| path(AUTH_VAULT_CA_CERT_ENV)),
1377                false,
1378                CA_VAULT_ADDR_ENV,
1379                CA_VAULT_TOKEN_ENV,
1380            ),
1381        };
1382
1383    match (&address, &token) {
1384        (Some(_), None) => {
1385            return Err(AxiamError::Internal(format!(
1386                "{addr_env} is set but {token_env} is not"
1387            )));
1388        }
1389        (None, Some(_)) => {
1390            return Err(AxiamError::Internal(format!(
1391                "{token_env} is set but {addr_env} is not"
1392            )));
1393        }
1394        _ => {}
1395    }
1396
1397    Ok(VaultEndpoint {
1398        address,
1399        token,
1400        ca_cert_path,
1401        inherited,
1402    })
1403}

1405pub fn custodians_from_env(encryption_key: Option<[u8; 32]>) -> AxiamResult<CaKeyCustodians> {
1406    custodians_from(encryption_key, &|name| std::env::var(name).ok())
1407}
1408
1409/// [`custodians_from_env`] against an arbitrary lookup.
1410///
1411/// `pub` because it is also the honest entry point for an embedder whose
1412/// configuration does not live in the process environment.
1413///
1414/// `&dyn Fn` rather than `impl Fn`: a generic seam is monomorphised once per
1415/// closure type, and coverage instrumentation counts every instantiation's
1416/// lines separately. A test suite calling this with a dozen different closures
1417/// would multiply this function's line count by twelve and its *uncovered*
1418/// lines with it — the tests would measurably worsen the very number they
1419/// exist to improve. One instantiation, one count.
1420///
1421/// The same seam, and for the same reason, as [`vault_endpoint_from`]: this
1422/// function decides which custodian a deployment gets, and that decision is
1423/// the one the 1.0.0-beta01 report caught getting silently wrong. Driving it
1424/// through the real environment would mean mutating process-global state from
1425/// a multi-threaded harness, so the rules are tested as a function of their
1426/// inputs instead.
1427pub fn custodians_from(
1428    encryption_key: Option<[u8; 32]>,
1429    lookup: &dyn Fn(&str) -> Option<String>,
1430) -> AxiamResult<CaKeyCustodians> {
1431    let env = |name: &str| lookup(name).filter(|v| !v.is_empty());
1432    let database = encryption_key.map(DatabaseCaKeyStore::new);
1433    let VaultEndpoint {
1434        address: vault_addr,
1435        token: vault_token,
1436        ca_cert_path: vault_ca_cert,
1437        inherited: vault_inherited,
1438    } = vault_endpoint_from(lookup)?;
1439
1440    let vault = match (vault_addr.clone(), vault_token.clone()) {
1441        (Some(address), Some(token)) => Some(VaultCaKeyStore::new(VaultCaKeyConfig {
1442            address,
1443            token,
1444            mount: env(CA_VAULT_MOUNT_ENV).unwrap_or_else(|| DEFAULT_CA_VAULT_MOUNT.to_string()),
1445            prefix: env(CA_VAULT_PREFIX_ENV).unwrap_or_else(|| DEFAULT_CA_VAULT_PREFIX.to_string()),
1446            ca_cert_path: vault_ca_cert.clone(),
1447        })?),
1448        // An address with no token, or the reverse, is a half-finished
1449        // configuration and worth saying so — it is otherwise indistinguishable
1450        // from not having configured Vault at all, right up until the default
1451        // custodian check fails with a message about a variable that *is* set.
1452        (Some(_), None) | (None, Some(_)) => {
1453            // Unreachable: `vault_endpoint_from` already rejects a
1454            // half-configured pair, naming whichever pair the operator set.
1455            unreachable!("vault_endpoint_from rejects half-configured pairs")
1456        }
1457        (None, None) => None,
1458    };
1459
1460    // The PKI engine reaches the *same* Vault: an operator running one Vault
1461    // should not configure its address twice. Only the mounts differ, and both
1462    // have Vault's own conventional defaults — so adopting this custodian is one
1463    // variable, `CA_KEY_STORE=vault_pki`, on a deployment that already had
1464    // Vault custody working.
1465    let vault_pki = match (vault_addr, vault_token) {
1466        (Some(address), Some(token)) => Some(VaultPkiCaKeyStore::new(VaultPkiConfig {
1467            address,
1468            token,
1469            root_mount: env(CA_VAULT_PKI_ROOT_MOUNT_ENV)
1470                .unwrap_or_else(|| DEFAULT_PKI_ROOT_MOUNT.to_string()),
1471            int_mount: env(CA_VAULT_PKI_INT_MOUNT_ENV)
1472                .unwrap_or_else(|| DEFAULT_PKI_INT_MOUNT.to_string()),
1473            ca_cert_path: vault_ca_cert,
1474        })?),
1475        // The half-configured cases are already a hard error above, on the KV
1476        // store built from the same two variables.
1477        _ => None,
1478    };
1479
1480    let default_custody = match lookup(CA_KEY_STORE_ENV) {
1481        // An explicit request. `CaKeyCustodians::new` refuses it when the named
1482        // custodian is not actually configured — the one case worth stopping
1483        // the process for.
1484        Some(raw) if !raw.trim().is_empty() => {
1485            Some(raw.trim().parse::<CaKeyCustody>().map_err(|e| {
1486                AxiamError::Internal(format!(
1487                    "{CA_KEY_STORE_ENV}: {e} — expected `database`, `vault` or `vault_pki`"
1488                ))
1489            })?)
1490        }
1491        // Vault configured and no explicit choice means Vault: an operator who
1492        // wired up an address and a token did so in order to use it, and
1493        // making them also set a third variable is a step whose only outcome is
1494        // being forgotten.
1495        _ if vault.is_some() => Some(CaKeyCustody::Vault),
1496        _ if database.is_some() => Some(CaKeyCustody::Database),
1497        // Nothing configured at all. Not an error: a deployment that issues no
1498        // certificates has no reason to hold a PKI key, and refusing to start
1499        // would break it for a feature it never uses.
1500        _ => None,
1501    };
1502
1503    let mut custodians = CaKeyCustodians::new(database, vault, vault_pki, default_custody)?;
1504    custodians.vault_inherited = vault_inherited;
1505    Ok(custodians)
1506}
