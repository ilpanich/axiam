31/// Who is asking for a leaf, as far as the issuing CA is concerned.
32///
33/// A signing CA row carries the tenant it signs for
34/// ([`axiam_core::models::certificate::CaCertificate::tenant_id`]), and before
35/// this type nothing read it: a CA id plus `certificates:generate` was enough
36/// to mint a leaf under **any** CA of the organization, including another
37/// tenant's. The scope is resolved from the caller's own record — the tenant it
38/// lives in — and never from request input, which is why it is a separate
39/// argument and not a field on the request body types.
40///
41/// It says what the caller *is*, not what it may do: the permission check is
42/// still the REST layer's, and an organization principal without
43/// `certificates:generate` mints nothing.
44#[derive(Debug, Clone, Copy, PartialEq, Eq)]
45pub enum IssuingScope {
46    /// An ordinary tenant principal.
47    ///
48    /// May issue under the signing CA of the tenant it is acting on, and under
49    /// nothing else — not another tenant's CA, and not the organization CA
50    /// above them, which is the anchor the tenant tier exists to keep leaves
51    /// away from.
52    Tenant,
53    /// A principal whose own record lives in the organization's reserved scope.
54    ///
55    /// May additionally issue directly under an organization-level CA. Acting
56    /// on a particular tenant's CA still means naming that tenant — through
57    /// `X-Axiam-Tenant` on the REST path — because the acting tenant is what
58    /// the CA is matched against for everyone.
59    Organization,
60}
61
62/// The refusal text for an issuing CA that is revoked or otherwise not active.
63///
64/// A constant because [`crate::saml_signing`] recognises it: the SAML signing
65/// credential's route turns it into a `400`, where the certificate routes have
66/// always answered it as a `Certificate` error.
67pub(crate) const CA_NOT_ACTIVE: &str = "CA certificate is not active";
68/// The refusal text for an issuing CA outside its validity window. See
69/// [`CA_NOT_ACTIVE`].
70pub(crate) const CA_NOT_VALID: &str = "CA certificate is expired or not yet valid";
71
72/// Hard cap for leaf certificate validity: 825 days (~27 months).
73///
74/// Aligns with CA/Browser Forum Baseline Requirements and Apple/Mozilla
75/// root program policies. Internal PKI may allow up to this limit;
76/// tenants can configure a lower per-tenant maximum via the
77/// `max_certificate_validity_days` key in their metadata.
78pub const MAX_LEAF_CERT_VALIDITY_DAYS: u32 = 825;
79
80/// Default leaf certificate validity when no tenant override is set: 365 days.
81pub const DEFAULT_LEAF_CERT_VALIDITY_DAYS: u32 = 365;
82
83/// The longest validity, in whole days, an issuer expiring at `issuer_not_after`
84/// can still grant to something issued at `now`.
85///
86/// A certificate must not outlive the CA that signed it — past the issuer's
87/// notAfter the chain stops validating, so the extra days are not merely
88/// useless, they are days the holder believes they have and does not. Both the
89/// leaf path ([`CertService::generate`]) and the intermediate path
90/// (`ca::generate_intermediate`) refuse a request above this number and quote it
91/// back, and the certificates API returns it per CA so the admin UI can cap its
92/// own input rather than discovering the limit on submit.
93///
94/// Rounded **down** to whole days, and floored at zero: an issuer with 36 hours
95/// left can grant one day, not two, and an expired issuer can grant nothing.
96/// (Callers reject an expired or not-yet-valid issuer before reaching here; the
97/// zero is a total function's answer, not a code path anyone rides.)
98///
99/// # Examples
100///
101/// ```
102/// use axiam_pki::cert::issuer_bounded_validity_days;
103/// use chrono::{Duration, Utc};
104///
105/// let now = Utc::now();
106/// // A CA with 90 days and 12 hours left grants 90 days, never 91.
107/// let ca_expiry = now + Duration::days(90) + Duration::hours(12);
108/// assert_eq!(issuer_bounded_validity_days(now, ca_expiry), 90);
109///
110/// // An issuer already past its notAfter grants nothing.
111/// assert_eq!(issuer_bounded_validity_days(now, now - Duration::days(1)), 0);
112/// ```
113pub fn issuer_bounded_validity_days(now: DateTime<Utc>, issuer_not_after: DateTime<Utc>) -> u32 {
114    let remaining = (issuer_not_after - now).num_days();
115    u32::try_from(remaining.max(0)).unwrap_or(u32::MAX)
116}

118/// What [`CertService::prepare_leaf_issuance`] establishes for both leaf paths.

123struct LeafIssuance<'a> {

127    not_after: DateTime<Utc>,
128}

158    pub fn new(
159        ca_repo: CA,
160        cert_repo: CR,
161        config: PkiConfig,
162        crypto_semaphore: Arc<Semaphore>,
163        custodians: Arc<CaKeyCustodians>,
164    ) -> Self {

174    /// Everything both leaf paths must establish before a certificate can be
175    /// signed: the validity bounds, the issuing CA, and which custodian holds
176    /// its key.
177    ///
178    /// Factored out of [`Self::generate`] when [`Self::sign_csr`] arrived,
179    /// rather than copied into it. Every check here is a refusal that must read
180    /// the same on both paths — a CSR-signed certificate that could be issued
181    /// under a revoked CA, or outlive its issuer, while a generated one could
182    /// not, would be a hole with the shape of a second implementation.
183    /// `generate`'s own tests are the regression net for the move.
184    async fn prepare_leaf_issuance(
185        &self,
186        org_id: Uuid,
187        acting_tenant_id: Uuid,
188        scope: IssuingScope,
189        issuer_ca_id: Uuid,
190        validity_days: u32,
191        max_validity_days: Option<u32>,
192    ) -> AxiamResult<LeafIssuance<'_>> {
193        // Enforce validity_days bounds: > 0 and <= tenant/hard cap
194        let effective_max = max_validity_days
195            .unwrap_or(DEFAULT_LEAF_CERT_VALIDITY_DAYS)
196            .min(MAX_LEAF_CERT_VALIDITY_DAYS);
197        if validity_days == 0 || validity_days > effective_max {
198            return Err(AxiamError::Validation {
199                message: format!(
200                    "validity_days must be between 1 and {effective_max} \
201                     (CA/Browser Forum BR hard cap: {MAX_LEAF_CERT_VALIDITY_DAYS} days)"
202                ),
203            });
204        }
205
206        // Scoped to the organization, so a CA id from another organization is
207        // not found rather than usable (T-98).
208        let ca_cert = self.ca_repo.get_by_id(org_id, issuer_ca_id).await?;
209
210        // ...and scoped to the *tenant* as well, which the organization scope
211        // alone never was (DF-017 / DF-025): `get_by_id` asks only
212        // `WHERE organization_id = $org_id`, so every CA of the organization —
213        // every other tenant's signing CA, and the organization anchor above
214        // them — was usable by any tenant administrator holding
215        // `certificates:generate`. The leaf was then recorded under the
216        // caller's tenant with another tenant's issuer, and chained to the same
217        // root every relying party in the organization trusts.
218        //
219        // Refused as `NotFound`, following the cross-organization precedent
220        // (`a_ca_in_another_organization_is_not_found`): a CA the caller may
221        // not use is a CA the caller cannot see, and the refusal must not tell
222        // an outsider whether it exists, is revoked, or has expired. Which is
223        // why it sits here, ahead of the status and window checks below.
224        let usable = match ca_cert.tenant_id {
225            // A tenant signing CA: usable by whoever is acting on that tenant,
226            // organization principal or not.
227            Some(ca_tenant) => ca_tenant == acting_tenant_id,
228            // An organization-level CA: the trust anchor. Issuing leaves
229            // directly under it is the thing the tenant tier exists to prevent,
230            // so only a principal that lives in the organization scope may.
231            None => scope == IssuingScope::Organization,
232        };
233        if !usable {
234            return Err(AxiamError::NotFound {
235                entity: "CA certificate".into(),
236                id: issuer_ca_id.to_string(),
237            });
238        }
239
240        if ca_cert.status != CertificateStatus::Active {
241            return Err(AxiamError::Certificate(CA_NOT_ACTIVE.into()));
242        }
243
244        // Validate CA certificate validity window
245        let now = Utc::now();
246        if now < ca_cert.not_before || now > ca_cert.not_after {
247            return Err(AxiamError::Certificate(CA_NOT_VALID.into()));
248        }
249
250        // Fetch the signing key from whichever custodian this CA's row names —
251        // not from whichever is currently configured. A deployment that adopted
252        // Vault still has CAs whose keys are sealed into their rows, and asking
253        // the current setting about them would fail to find a key that is
254        // perfectly present.
255        //
256        // An imported CA with no key at all resolves to the `External`
257        // custodian, whose error says it has no private key and therefore
258        // cannot issue — rather than the bare "no stored private key" this used
259        // to produce for both that case and a genuine decryption failure.
260        let store = self.custodians.store_for(ca_cert.key_custody)?;
261
262        let not_before = now;
263        let requested_not_after = now
264            .checked_add_signed(Duration::days(i64::from(validity_days)))
265            .ok_or_else(|| AxiamError::Validation {
266                message: "validity_days produces a date out of range".into(),
267            })?;
268        // A leaf must not outlive the CA that signed it: a relying party stops
269        // trusting the chain the moment the issuer expires, so the extra days
270        // buy nothing and the certificate would fail every handshake made after
271        // the issuer's notAfter.
272        //
273        // This used to be a silent `min(requested, ca_cert.not_after)`. Silent
274        // was the wrong call: an operator asking for two years against a CA with
275        // three months left got a three-month certificate, no warning, and a
276        // renewal calendar built on a date the certificate does not carry. Say
277        // no and name the number they can actually have — see
278        // `issuer_bounded_validity_days`, which the API also exposes so the
279        // form can cap its own input before anyone submits it.
280        if requested_not_after > ca_cert.not_after {
281            let available = issuer_bounded_validity_days(now, ca_cert.not_after);
282            return Err(AxiamError::Validation {
283                message: format!(
284                    "validity_days is {} but the issuing CA expires on {} — a certificate \
285                     cannot outlive its issuer. The most this CA can grant today is {} day{}.",
286                    validity_days,
287                    ca_cert.not_after.format("%Y-%m-%d"),
288                    available,
289                    if available == 1 { "" } else { "s" },
290                ),
291            });
292        }
293
294        Ok(LeafIssuance {
295            ca_cert,
296            store,
297            not_before,
298            not_after: requested_not_after,
299        })
300    }

302    /// Generate a new certificate signed by a CA of this organization.
303    ///
304    /// `org_id` is required to look up the CA certificate and its encrypted
305    /// private key for signing.
306    ///
307    /// `scope` says what the caller is — see [`IssuingScope`]. Together with
308    /// `input.tenant_id`, which is the tenant being acted on and comes from the
309    /// authenticated context rather than the body, it decides which CAs of the
310    /// organization this call can reach at all.
311    ///
312    /// `max_validity_days` is the tenant-level cap (from tenant metadata).
313    /// Pass `None` to use the default ([`DEFAULT_LEAF_CERT_VALIDITY_DAYS`]).
314    /// The hard cap ([`MAX_LEAF_CERT_VALIDITY_DAYS`], 825 days) is always
315    /// enforced per CA/Browser Forum Baseline Requirements.
316    ///
317    /// `server_names` is the acting tenant's **effective**
318    /// `server_cert_allowed_names` (S-7), resolved by the caller from the
319    /// settings hierarchy. It is a required argument rather than something
320    /// this service looks up so that no caller can issue without having
321    /// answered the question; an empty slice is the shipped answer, and it
322    /// refuses every `Server` request. See
323    /// [`axiam_core::models::server_names::check_leaf_names`].
324    pub async fn generate(
325        &self,
326        org_id: Uuid,
327        scope: IssuingScope,
328        mut input: CreateCertificate,
329        max_validity_days: Option<u32>,
330        server_names: &[String],
331    ) -> AxiamResult<GeneratedCertificate> {
332        // D-21. The SAML signing leaf is not an inventory certificate: it is
333        // issued by `crate::saml_signing` and kept in the tenant's `saml_idp_credential` row. Refused here by
334        // name rather than left to the schema's `cert_type` assertion, which
335        // would refuse it only after the key was generated and the leaf signed.
336        if input.cert_type == CertificateType::SamlSigning {
337            return Err(AxiamError::Validation {
338                message: "a SamlSigning certificate is not issued through the certificate \
339                          inventory"
340                    .into(),
341            });
342        }
343
344        // DF-023. Normalised before anything reads it, so the locally signed
345        // path, the remote-custodian path and the stored row all carry the
346        // same common name. See [`crate::subject_common_name`].
347        input.subject = subject_common_name(&input.subject)?;
348
349        let material = self
350            .issue_leaf_material(org_id, scope, &input, max_validity_days, server_names)
351            .await?;
352
353        let store = StoreCertificate {
354            tenant_id: input.tenant_id,
355            issuer_ca_id: input.issuer_ca_id,
356            subject: input.subject,
357            public_cert_pem: material.public_cert_pem,
358            fingerprint: material.fingerprint,
359            cert_type: input.cert_type,
360            key_algorithm: input.key_algorithm,
361            not_before: material.not_before,
362            not_after: material.not_after,
363            metadata: input.metadata.unwrap_or(serde_json::json!({})),
364        };
365
366        let certificate = self.cert_repo.create(store).await?;
367
368        Ok(GeneratedCertificate {
369            certificate,
370            private_key_pem: material.private_key_pem,
371            chain_pem: material.chain_pem,
372        })
373    }

375    /// Everything [`Self::generate`] does up to, and not including, writing the
376    /// inventory row: the name fence, the issuing-CA checks, the key pair and
377    /// the signature, under whichever custodian holds the CA's key.
378    ///
379    /// `input.subject` must already be normalised.
380    pub(crate) async fn issue_leaf_material(
381        &self,
382        org_id: Uuid,
383        scope: IssuingScope,
384        input: &CreateCertificate,
385        max_validity_days: Option<u32>,
386        server_names: &[String],
387    ) -> AxiamResult<LeafMaterial> {
388        // S-7 — the name fence, before the CA is even looked up: a name the
389        // tenant may not use is refused the same way whichever CA it named
390        // and whichever custodian holds that CA's key.
391        let sans = check_leaf_names(
392            &input.cert_type,
393            &input.subject,
394            &input.subject_alt_names,
395            server_names,
396        )?;
397        let profile = LeafProfile::for_leaf(&input.cert_type, &input.key_algorithm);
398
399        let LeafIssuance {
400            ca_cert,
401            store,
402            not_before,
403            not_after,
404        } = self
405            .prepare_leaf_issuance(
406                org_id,
407                input.tenant_id,
408                scope,
409                input.issuer_ca_id,
410                input.validity_days,
411                max_validity_days,
412            )
413            .await?;
414
415        // A custodian that signs on AXIAM's behalf never hands the key over, so
416        // there is nothing to load and nothing to sign with here: the leaf key
417        // is generated locally, its public half goes out in a CSR, and a
418        // certificate comes back. See [`axiam_core::ca_keys`].
419        if store.signs_remotely() {
420            return self
421                .generate_remotely(&ca_cert, input, not_before, not_after, sans, profile)
422                .await;
423        }
424
425        // Fetch the signing key from whichever custodian this CA's row named.
426        let key_ref = CaService::<CA>::key_ref(&ca_cert);
427        let mut ca_private_key_pem = store
428            .load(&key_ref, ca_cert.encrypted_private_key.as_deref())
429            .await?
430            .to_string();
431
432        // CPU-bound: key generation + certificate signing run in spawn_blocking behind semaphore (CQ-B02).
433        let _permit = self
434            .crypto_semaphore
435            .acquire()
436            .await
437            .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
438
439        let ca_cert_pem = ca_cert.public_cert_pem.clone();
440        let ee_subject = input.subject.clone();
441        let key_algorithm = input.key_algorithm.clone();
442        let not_before_ts = not_before.timestamp();
443        let not_after_ts = not_after.timestamp();
444
445        let (private_key_pem, public_cert_pem, fingerprint) =
446            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String, String)> {
447                let ca_key_pair = KeyPair::from_pem(&ca_private_key_pem)
448                    .map_err(|e| AxiamError::Certificate(format!("invalid CA private key: {e}")))?;
449                // Scrub the decrypted CA private-key PEM from memory as soon as
450                // the KeyPair is parsed — it is not needed past this point and
451                // must not linger in the heap buffer (defense-in-depth).
452                ca_private_key_pem.zeroize();
453
454                // Reconstruct the signing CA issuer from its real, stored certificate
455                // PEM — NOT from the (mutable) `subject` field — so the issuer DN
456                // embedded in every leaf cert can never drift from the CA's actual
457                // Subject DN (QUAL-05/D-08, T-29-11). rcgen 0.14 moved
458                // `from_ca_cert_pem` onto `Issuer`, which now owns the signing key and
459                // is passed directly to `signed_by`.
460                let ca_issuer =
461                    Issuer::from_ca_cert_pem(&ca_cert_pem, ca_key_pair).map_err(|e| {
462                        AxiamError::Certificate(format!("invalid CA certificate PEM: {e}"))
463                    })?;
464
465                // Generate end-entity key pair.
466                let ee_key_pair = generate_keypair(&key_algorithm)?;
467                let private_key_pem = ee_key_pair.serialize_pem();
468
469                // Build end-entity certificate request — from the same
470                // function `sign_csr` overwrites a caller's parameters with, so
471                // the two paths cannot issue different shapes.
472                let ee_params =
473                    leaf_params(&ee_subject, not_before_ts, not_after_ts, &sans, &profile)?;
474
475                let cert = ee_params.signed_by(&ee_key_pair, &ca_issuer).map_err(|e| {
476                    AxiamError::Certificate(format!("certificate signing failed: {e}"))
477                })?;
478
479                let public_cert_pem = cert.pem();
480                let fingerprint = compute_fingerprint(cert.der());
481                Ok((private_key_pem, public_cert_pem, fingerprint))
482            })
483            .await
484            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
485
486        Ok(LeafMaterial {
487            private_key_pem,
488            public_cert_pem,
489            // Signed in-process by a CA whose certificate the caller can fetch;
490            // there is no chain here they cannot already assemble.
491            chain_pem: None,
492            fingerprint,
493            not_before,
494            not_after,
495        })
496    }

498    /// Issue against a CA whose key lives in — and stays in — its custodian.
499    ///
500    /// The end-entity key is still generated here and still returned once: what
501    /// moves to the custodian is the *signature*, not the subscriber's key. The
502    /// CSR carries the public half and nothing else.
503    async fn generate_remotely(
504        &self,
505        ca_cert: &axiam_core::models::certificate::CaCertificate,
506        input: &CreateCertificate,
507        not_before: DateTime<Utc>,
508        not_after: DateTime<Utc>,
509        sans: Vec<RequestedName>,
510        profile: LeafProfile,
511    ) -> AxiamResult<LeafMaterial> {
512        // Only the keygen and CSR are CPU-bound. The permit is dropped before
513        // the call to the custodian: holding it across a network round trip
514        // would let one slow signer block every other issuance in the
515        // deployment, which is the opposite of what the semaphore is for.
516        let (private_key_pem, csr_pem) = {
517            let _permit = self
518                .crypto_semaphore
519                .acquire()
520                .await
521                .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
522
523            let ee_subject = input.subject.clone();
524            let key_algorithm = input.key_algorithm.clone();
525            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String)> {
526                let ee_key_pair = generate_keypair(&key_algorithm)?;
527                let private_key_pem = ee_key_pair.serialize_pem();
528
529                let mut params = CertificateParams::new(Vec::<String>::new())
530                    .map_err(|e| AxiamError::Certificate(e.to_string()))?;
531                params
532                    .distinguished_name
533                    .push(DnType::CommonName, &ee_subject);
534                params.is_ca = IsCa::NoCa;
535                // S-7. The SANs go *in the CSR*, because that is the only place
536                // a verbatim signer reads them from: Vault's `sign-verbatim`
537                // ignores `alt_names` and `ip_sans` (measured on 1.18.3). This
538                // CSR is AXIAM's own, built from names the fence just admitted,
539                // so nothing a caller wrote reaches the list. The usages are
540                // not put here — they travel as `profile`, which the signer
541                // applies because this CSR requests none.
542                params.subject_alt_names = san_types(&sans)?;
543                // No validity window: a PKCS#10 request cannot carry one. The
544                // signer is told the lifetime out of band, as a TTL.
545                let csr = params.serialize_request(&ee_key_pair).map_err(|e| {
546                    AxiamError::Certificate(format!("certificate request failed: {e}"))
547                })?;
548                let csr_pem = csr
549                    .pem()
550                    .map_err(|e| AxiamError::Certificate(format!("CSR encoding failed: {e}")))?;
551                Ok((private_key_pem, csr_pem))
552            })
553            .await
554            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??
555        };
556
557        let key_ref = CaService::<CA>::key_ref(ca_cert);
558        let store = self.custodians.store_for(ca_cert.key_custody)?;
559        let signed = store
560            .sign_csr(
561                &key_ref,
562                &LeafSigningRequest {
563                    csr_pem,
564                    // AXIAM built this request, three lines up, from a key it
565                    // just generated: there is nothing in it a caller chose.
566                    csr_is_caller_supplied: false,
567                    ttl_seconds: (not_after - not_before).num_seconds(),
568                    profile,
569                },
570            )
571            .await?;
572
573        // What the signer produced, not what was asked for. A remote signer
574        // caps a TTL to its mount's own maximum without failing the call, so a
575        // row built from the request would claim a window the certificate does
576        // not have — and the fingerprint has to be of the real bytes or every
577        // lookup by fingerprint misses.
578        let issued = parse_issued_leaf(&signed.certificate_pem)?;
579
580        Ok(LeafMaterial {
581            private_key_pem,
582            public_cert_pem: signed.certificate_pem,
583            chain_pem: (!signed.chain_pem.is_empty()).then(|| join_pem(&signed.chain_pem)),
584            fingerprint: issued.fingerprint,
585            not_before: issued.not_before,
586            not_after: issued.not_after,
587        })
588    }
589
590    /// The minimum RSA modulus AXIAM will sign a certificate over, in bits.
591    ///
592    /// Stated here rather than inferred from [`KeyAlgorithm::Rsa4096`]'s name,
593    /// because that variant is a label and not a measurement:
594    /// `ca::parse_ca_certificate` maps *any* RSA OID onto it, deliberately, so
595    /// that an imported root of some other size is still a usable root. A path
596    /// that enforces a floor cannot reuse that mapping, and an RSA-2048 CSR
597    /// signed and then recorded as `Rsa4096` is the exact shape of T-96.
598    const MIN_CSR_RSA_MODULUS_BITS: usize = 4096;
599
600    /// Sign a caller's certificate signing request, issuing an end-entity
601    /// certificate for a key AXIAM never sees.
602    ///
603    /// The BYOK counterpart to [`Self::generate`]. The subscriber's private key
604    /// is generated by whoever made the CSR and never crosses the wire in
605    /// either direction — which is why the answer is a [`Certificate`] and not
606    /// a `GeneratedCertificate`: there is no key to return.
607    ///
608    /// # What the caller decides, and what AXIAM decides
609    ///
610    /// The caller decides the key and the subject. AXIAM decides everything
611    /// else, and the difference is enforced rather than documented:
612    ///
613    /// - **Possession** is proved by the request's own self-signature, checked
614    ///   by [`ca::inspect_csr`] before anything else happens. Without it a
615    ///   caller could have a certificate minted over somebody else's public key.
616    /// - **The key** must be Ed25519, or RSA with a modulus of at least
617    ///   [`Self::MIN_CSR_RSA_MODULUS_BITS`] bits. The modulus is measured, not
618    ///   taken from a label.
619    /// - **The extensions** are AXIAM's. A CSR requesting `subjectAltName`,
620    ///   `keyUsage` or `extendedKeyUsage` is refused by name rather than
621    ///   silently stripped; every other requested extension is discarded when
622    ///   the parameter set is overwritten with the leaf parameters
623    ///   [`Self::generate`] builds. A CSR asking to be an unconstrained CA
624    ///   comes back a leaf.
625    /// - **The validity, the issuer and the tenant** come from
626    ///   [`Self::prepare_leaf_issuance`], the same function `generate` uses, so
627    ///   the two paths cannot drift on a revoked CA, an expired one, a CA
628    ///   belonging to another organization, or a certificate that would outlive
629    ///   its issuer.
630    ///
631    /// # Why `keyUsage` is refused rather than dropped
632    ///
633    /// On the in-process path it could be dropped: `request.params` is replaced
634    /// wholesale before signing and nothing the caller asked for survives. Under
635    /// [`CaKeyCustody::VaultPki`] it could not. Vault's `sign-verbatim`
636    /// discards the `key_usage` and `ext_key_usage` request parameters when the
637    /// CSR carries those extensions, and issues what the CSR asked for — so a
638    /// silent strip would be a promise AXIAM keeps on one custodian and breaks
639    /// on the other, for the same CSR. One rule that holds everywhere is worth
640    /// more than a convenience that holds in one deployment.
641    ///
642    /// `basicConstraints` needs no such rule: the in-process path overwrites it
643    /// and Vault ignores a CSR's basic constraints outright.
644    ///
645    /// # Names (S-7)
646    ///
647    /// `subjectAltName` stays refused *in the CSR*. A `Server` certificate's
648    /// names come from [`SignCertificateCsr::subject_alt_names`], through the
649    /// same fence [`Self::generate`] runs, with `server_names` as the acting
650    /// tenant's effective allow-list. Under a custodian that signs remotely a
651    /// `Server` request is refused: such a signer signs the caller's CSR
652    /// verbatim, takes SANs only from it, and the CSR may not carry any — so
653    /// there is no channel left through which the explicit names could reach
654    /// the certificate. Use [`Self::generate`] under such a CA.
655    pub async fn sign_csr(
656        &self,
657        org_id: Uuid,
658        scope: IssuingScope,
659        input: SignCertificateCsr,
660        max_validity_days: Option<u32>,
661        server_names: &[String],
662    ) -> AxiamResult<Certificate> {
663        // D-21: see `generate`. A caller's CSR is the opposite of what the SAML
664        // credential is — a key AXIAM never sees — so it can never be one.
665        if input.cert_type == CertificateType::SamlSigning {
666            return Err(AxiamError::Validation {
667                message: "a SamlSigning certificate is not issued through the certificate \
668                          inventory"
669                    .into(),
670            });
671        }
672
673        // Parsed before the CA is even looked up, and outside the blocking
674        // task: a malformed or unsigned request is by far the likeliest failure
675        // on an endpoint whose input is pasted by hand, and it must be a 400
676        // naming the CSR rather than a 500 out of a custodian.
677        let facts = ca::inspect_csr(&input.csr_pem)?;
678
679        if let Some(bits) = facts.rsa_modulus_bits
680            && bits < Self::MIN_CSR_RSA_MODULUS_BITS
681        {
682            return Err(AxiamError::Validation {
683                message: format!(
684                    "the certificate signing request carries an RSA key with a {bits}-bit \
685                     modulus; AXIAM signs Ed25519 keys and RSA keys of at least {} bits",
686                    Self::MIN_CSR_RSA_MODULUS_BITS
687                ),
688            });
689        }
690
691        if !facts.refused_extensions.is_empty() {
692            return Err(AxiamError::Validation {
693                message: format!(
694                    "the certificate signing request asks for {}, which AXIAM does not sign \
695                     over on an end-entity certificate. It is refused rather than dropped \
696                     because a certificate issued without what you asked for, and with \
697                     nothing said, would fail where you deploy it. Remove the extension \
698                     request and submit the CSR again.",
699                    facts.refused_extensions.join(" and ")
700                ),
701            });
702        }
703
704        // S-7 — the same fence `generate` runs, over the CSR's own common name
705        // and the request's explicit SANs.
706        let sans = check_leaf_names(
707            &input.cert_type,
708            &facts.common_name,
709            &input.subject_alt_names,
710            server_names,
711        )?;
712        let profile = LeafProfile::for_leaf(&input.cert_type, &facts.key_algorithm);
713
714        let LeafIssuance {
715            ca_cert,
716            store,
717            not_before,
718            not_after,
719        } = self
720            .prepare_leaf_issuance(
721                org_id,
722                input.tenant_id,
723                scope,
724                input.issuer_ca_id,
725                input.validity_days,
726                max_validity_days,
727            )
728            .await?;
729
730        if store.signs_remotely() && !sans.is_empty() {
731            return Err(AxiamError::Validation {
732                message: "a Server certificate cannot be issued from a caller's CSR under a CA \
733                          whose key is held by a remote signer: the signer copies the CSR's \
734                          subjectAltName verbatim and ignores any other, and AXIAM refuses a CSR \
735                          that carries one. Issue it with POST /api/v1/certificates under this \
736                          CA, or sign the CSR under a CA whose key AXIAM holds"
737                    .into(),
738            });
739        }
740
741        let (public_cert_pem, fingerprint, chain_pem, issued_window) = if store.signs_remotely() {
742            let key_ref = CaService::<CA>::key_ref(&ca_cert);
743            let signed = store
744                .sign_csr(
745                    &key_ref,
746                    &LeafSigningRequest {
747                        csr_pem: input.csr_pem.clone(),
748                        ttl_seconds: (not_after - not_before).num_seconds(),
749                        // These bytes are the caller's. See the custodian's own
750                        // doc comment for what it does about that.
751                        csr_is_caller_supplied: true,
752                        profile,
753                    },
754                )
755                .await?;
756            // What the signer produced, not what was asked for: a remote signer
757            // caps a TTL to its mount's own maximum without failing the call.
758            let issued = parse_issued_leaf(&signed.certificate_pem)?;
759            let chain = (!signed.chain_pem.is_empty()).then(|| join_pem(&signed.chain_pem));
760            (
761                signed.certificate_pem,
762                issued.fingerprint,
763                chain,
764                Some((issued.not_before, issued.not_after)),
765            )
766        } else {
767            let key_ref = CaService::<CA>::key_ref(&ca_cert);
768            let mut ca_private_key_pem = store
769                .load(&key_ref, ca_cert.encrypted_private_key.as_deref())
770                .await?
771                .to_string();
772
773            let _permit = self
774                .crypto_semaphore
775                .acquire()
776                .await
777                .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
778
779            let ca_cert_pem = ca_cert.public_cert_pem.clone();
780            let csr_pem = input.csr_pem.clone();
781            let subject = facts.common_name.clone();
782            let not_before_ts = not_before.timestamp();
783            let not_after_ts = not_after.timestamp();
784
785            let (pem, fingerprint) =
786                tokio::task::spawn_blocking(move || -> AxiamResult<(String, String)> {
787                    let ca_key_pair = KeyPair::from_pem(&ca_private_key_pem).map_err(|e| {
788                        AxiamError::Certificate(format!("invalid CA private key: {e}"))
789                    })?;
790                    ca_private_key_pem.zeroize();
791
792                    let ca_issuer =
793                        Issuer::from_ca_cert_pem(&ca_cert_pem, ca_key_pair).map_err(|e| {
794                            AxiamError::Certificate(format!("invalid CA certificate PEM: {e}"))
795                        })?;
796
797                    // Parsed a second time, inside the task, because rcgen's
798                    // signing API needs its own representation. The answer that
799                    // matters — whether the signature verifies — was already
800                    // established above, on the same bytes; this is not a
801                    // second opinion about it.
802                    let mut request = rcgen::CertificateSigningRequestParams::from_pem(&csr_pem)
803                        .map_err(|e| AxiamError::Validation {
804                            message: format!("the certificate signing request was rejected: {e}"),
805                        })?;
806                    // Everything the request asked to be, overwritten by what
807                    // AXIAM has decided it is — the same leaf parameters
808                    // `generate` builds, so the two paths issue the same shape
809                    // of certificate for the same request.
810                    request.params =
811                        leaf_params(&subject, not_before_ts, not_after_ts, &sans, &profile)?;
812                    // Deliberately **not** `use_authority_key_identifier_extension`,
813                    // which the intermediate CSR path sets. Setting it here put
814                    // an authorityKeyIdentifier on a CSR-signed leaf that a
815                    // generated leaf does not carry, which is exactly the
816                    // divergence D-5 forbids: two ways of asking for the same
817                    // certificate producing two different certificates. An AKI
818                    // on leaves is worth having and is a change to make on both
819                    // paths at once — the KU/EKU profile (S-7) was made that
820                    // way — not a side effect of how the key was made.
821
822                    let cert = request.signed_by(&ca_issuer).map_err(|e| {
823                        AxiamError::Certificate(format!("certificate signing failed: {e}"))
824                    })?;
825                    let pem = cert.pem();
826                    let fingerprint = compute_fingerprint(cert.der());
827                    Ok((pem, fingerprint))
828                })
829                .await
830                .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
831
832            // Signed in-process by a CA whose certificate the caller can fetch;
833            // there is no chain here they cannot already assemble.
834            (pem, fingerprint, None, None)
835        };
836
837        let (row_not_before, row_not_after) = issued_window.unwrap_or((not_before, not_after));
838
839        let certificate = self
840            .cert_repo
841            .create(StoreCertificate {
842                tenant_id: input.tenant_id,
843                issuer_ca_id: input.issuer_ca_id,
844                // What the certificate says, never what the caller said
845                // separately — there is no separate field for them to say it in
846                // (T-194: the row and the certificate cannot disagree).
847                subject: facts.common_name,
848                public_cert_pem,
849                fingerprint,
850                cert_type: input.cert_type,
851                // Read off the CSR's own public key, not asserted by the caller.
852                key_algorithm: facts.key_algorithm,
853                not_before: row_not_before,
854                not_after: row_not_after,
855                metadata: input.metadata.unwrap_or(serde_json::json!({})),
856            })
857            .await?;
858
859        // `chain_pem` has nowhere to go on a `Certificate` and is not dropped
860        // silently: under the custodians that return one, the chain is the
861        // issuing CA's own certificate, which `GET /certificates/{id}` already
862        // resolves through `issuer_ca_id`. A caller holding the certificate can
863        // assemble the chain from data the API already gives them.
864        let _ = chain_pem;
865
866        Ok(certificate)
867    }

896/// What a certificate a remote signer returned actually says about itself.
897struct IssuedLeaf {
898    fingerprint: String,
899    not_before: DateTime<Utc>,
900    not_after: DateTime<Utc>,
901}
902
903/// The parameters every AXIAM leaf certificate is built from: the common name,
904/// the validity window, `CA:FALSE`, the SANs the name fence admitted, and the
905/// per-type usage profile (S-7).
906///
907/// The SANs are empty for every type but `Server`, and are only ever the names
908/// [`check_leaf_names`] returned — never anything from a CSR. The profile is
909/// [`LeafProfile::for_leaf`]; before S-7 a leaf carried no keyUsage and no
910/// extendedKeyUsage, which RFC 5280 reads as "any usage", so adding them only
911/// narrows. Leaves issued earlier keep carrying neither.
912///
913/// Shared by [`CertService::generate`], which builds a key to go with it, and
914/// by [`CertService::sign_csr`], which overwrites a caller's requested
915/// parameters with it. That sharing is the enforcement: "a CSR-signed leaf is
916/// the same shape as a generated one" is true because there is one function
917/// that says what the shape is.
918fn leaf_params(
919    subject: &str,
920    not_before_ts: i64,
921    not_after_ts: i64,
922    sans: &[RequestedName],
923    profile: &LeafProfile,
924) -> AxiamResult<CertificateParams> {
925    let mut params = CertificateParams::new(Vec::<String>::new())
926        .map_err(|e| AxiamError::Certificate(e.to_string()))?;
927    params.distinguished_name.push(DnType::CommonName, subject);
928    params.is_ca = IsCa::NoCa;
929    params.subject_alt_names = san_types(sans)?;
930    params.key_usages = profile
931        .key_usage
932        .iter()
933        .map(|u| match u {
934            LeafKeyUsage::DigitalSignature => KeyUsagePurpose::DigitalSignature,
935            LeafKeyUsage::KeyEncipherment => KeyUsagePurpose::KeyEncipherment,
936        })
937        .collect();
938    params.extended_key_usages = profile
939        .extended_key_usage
940        .iter()
941        .map(|u| match u {
942            LeafExtendedKeyUsage::ClientAuth => ExtendedKeyUsagePurpose::ClientAuth,
943            LeafExtendedKeyUsage::ServerAuth => ExtendedKeyUsagePurpose::ServerAuth,
944            LeafExtendedKeyUsage::DocumentSigning => ExtendedKeyUsagePurpose::Other(
945                LeafExtendedKeyUsage::DocumentSigning.oid_arcs().to_vec(),
946            ),
947        })
948        .collect();
949    params.not_before = time::OffsetDateTime::from_unix_timestamp(not_before_ts)
950        .map_err(|e| AxiamError::Certificate(format!("invalid notBefore: {e}")))?;
951    params.not_after = time::OffsetDateTime::from_unix_timestamp(not_after_ts)
952        .map_err(|e| AxiamError::Certificate(format!("invalid notAfter: {e}")))?;
953    Ok(params)
954}
955
956/// The fence's names in rcgen's form.
957fn san_types(sans: &[RequestedName]) -> AxiamResult<Vec<SanType>> {
958    sans.iter()
959        .map(|n| match n {
960            RequestedName::Dns(d) => rcgen::string::Ia5String::try_from(d.as_str())
961                .map(SanType::DnsName)
962                .map_err(|e| AxiamError::Certificate(format!("invalid DNS name: {e}"))),
963            RequestedName::Ip(ip) => Ok(SanType::IpAddress(*ip)),
964        })
965        .collect()
966}
967
968/// Read back a signed leaf, so the row describes the certificate rather than
969/// the request that produced it.
970fn parse_issued_leaf(pem: &str) -> AxiamResult<IssuedLeaf> {
971    let (_, block) = x509_parser::pem::parse_x509_pem(pem.as_bytes()).map_err(|e| {
972        AxiamError::Certificate(format!(
973            "the signer returned something that is not a PEM certificate: {e}"
974        ))
975    })?;
976    let (_, cert) = X509Certificate::from_der(&block.contents).map_err(|e| {
977        AxiamError::Certificate(format!("the signed certificate could not be parsed: {e}"))
978    })?;
979
980    let to_utc = |t: i64, what: &str| -> AxiamResult<DateTime<Utc>> {
981        DateTime::from_timestamp(t, 0).ok_or_else(|| {
982            AxiamError::Certificate(format!("the signed certificate's {what} is out of range"))
983        })
984    };
985
986    Ok(IssuedLeaf {
987        fingerprint: compute_fingerprint(&block.contents),
988        not_before: to_utc(cert.validity().not_before.timestamp(), "notBefore")?,
989        not_after: to_utc(cert.validity().not_after.timestamp(), "notAfter")?,
990    })
991}
