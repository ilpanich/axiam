31/// Who is asking for a leaf, as far as the issuing CA is concerned.
32///
33/// A signing CA row carries the tenant it signs for
34/// ([`axiam_core::models::certificate::CaCertificate::tenant_id`]), and before
35/// this type nothing read it: a CA id plus `certificates:generate` was enough
36/// to mint a leaf under **any** CA of the organization, including another
37/// tenant's. The scope is resolved from the caller's own record — the tenant it
38/// lives in — and never from request input, which is why it is a separate
39/// argument and not a field on the request body types.
40///
41/// It says what the caller *is*, not what it may do: the permission check is
42/// still the REST layer's, and an organization principal without
43/// `certificates:generate` mints nothing.
44#[derive(Debug, Clone, Copy, PartialEq, Eq)]
45pub enum IssuingScope {
46    /// An ordinary tenant principal.
47    ///
48    /// May issue under the signing CA of the tenant it is acting on, and under
49    /// nothing else — not another tenant's CA, and not the organization CA
50    /// above them, which is the anchor the tenant tier exists to keep leaves
51    /// away from.
52    Tenant,
53    /// A principal whose own record lives in the organization's reserved scope.
54    ///
55    /// May additionally issue directly under an organization-level CA. Acting
56    /// on a particular tenant's CA still means naming that tenant — through
57    /// `X-Axiam-Tenant` on the REST path — because the acting tenant is what
58    /// the CA is matched against for everyone.
59    Organization,
60}
61
62/// Hard cap for leaf certificate validity: 825 days (~27 months).
63///
64/// Aligns with CA/Browser Forum Baseline Requirements and Apple/Mozilla
65/// root program policies. Internal PKI may allow up to this limit;
66/// tenants can configure a lower per-tenant maximum via the
67/// `max_certificate_validity_days` key in their metadata.
68pub const MAX_LEAF_CERT_VALIDITY_DAYS: u32 = 825;
69
70/// Default leaf certificate validity when no tenant override is set: 365 days.
71pub const DEFAULT_LEAF_CERT_VALIDITY_DAYS: u32 = 365;
72
73/// The longest validity, in whole days, an issuer expiring at `issuer_not_after`
74/// can still grant to something issued at `now`.
75///
76/// A certificate must not outlive the CA that signed it — past the issuer's
77/// notAfter the chain stops validating, so the extra days are not merely
78/// useless, they are days the holder believes they have and does not. Both the
79/// leaf path ([`CertService::generate`]) and the intermediate path
80/// (`ca::generate_intermediate`) refuse a request above this number and quote it
81/// back, and the certificates API returns it per CA so the admin UI can cap its
82/// own input rather than discovering the limit on submit.
83///
84/// Rounded **down** to whole days, and floored at zero: an issuer with 36 hours
85/// left can grant one day, not two, and an expired issuer can grant nothing.
86/// (Callers reject an expired or not-yet-valid issuer before reaching here; the
87/// zero is a total function's answer, not a code path anyone rides.)
88///
89/// # Examples
90///
91/// ```
92/// use axiam_pki::cert::issuer_bounded_validity_days;
93/// use chrono::{Duration, Utc};
94///
95/// let now = Utc::now();
96/// // A CA with 90 days and 12 hours left grants 90 days, never 91.
97/// let ca_expiry = now + Duration::days(90) + Duration::hours(12);
98/// assert_eq!(issuer_bounded_validity_days(now, ca_expiry), 90);
99///
100/// // An issuer already past its notAfter grants nothing.
101/// assert_eq!(issuer_bounded_validity_days(now, now - Duration::days(1)), 0);
102/// ```
103pub fn issuer_bounded_validity_days(now: DateTime<Utc>, issuer_not_after: DateTime<Utc>) -> u32 {
104    let remaining = (issuer_not_after - now).num_days();
105    u32::try_from(remaining.max(0)).unwrap_or(u32::MAX)
106}

108/// What [`CertService::prepare_leaf_issuance`] establishes for both leaf paths.
109///
110/// Borrowed rather than owned for `store`: [`CaKeyCustodians::store_for`] hands
111/// back a reference into the service's own custodian table, and cloning a
112/// custodian to avoid one lifetime would duplicate a Vault client per issuance.
113struct LeafIssuance<'a> {
114    ca_cert: axiam_core::models::certificate::CaCertificate,
115    store: &'a dyn axiam_core::ca_keys::CaKeyStore,
116    not_before: DateTime<Utc>,
117    not_after: DateTime<Utc>,
118}

134    pub fn new(
135        ca_repo: CA,
136        cert_repo: CR,
137        config: PkiConfig,
138        crypto_semaphore: Arc<Semaphore>,
139        custodians: Arc<CaKeyCustodians>,
140    ) -> Self {
141        Self {
142            ca_repo,
143            cert_repo,
144            config,
145            crypto_semaphore,
146            custodians,
147        }

150    /// Everything both leaf paths must establish before a certificate can be
151    /// signed: the validity bounds, the issuing CA, and which custodian holds
152    /// its key.
153    ///
154    /// Factored out of [`Self::generate`] when [`Self::sign_csr`] arrived,
155    /// rather than copied into it. Every check here is a refusal that must read
156    /// the same on both paths — a CSR-signed certificate that could be issued
157    /// under a revoked CA, or outlive its issuer, while a generated one could
158    /// not, would be a hole with the shape of a second implementation.
159    /// `generate`'s own tests are the regression net for the move.
160    async fn prepare_leaf_issuance(
161        &self,
162        org_id: Uuid,
163        acting_tenant_id: Uuid,
164        scope: IssuingScope,
165        issuer_ca_id: Uuid,
166        validity_days: u32,
167        max_validity_days: Option<u32>,
168    ) -> AxiamResult<LeafIssuance<'_>> {
169        // Enforce validity_days bounds: > 0 and <= tenant/hard cap
170        let effective_max = max_validity_days
171            .unwrap_or(DEFAULT_LEAF_CERT_VALIDITY_DAYS)
172            .min(MAX_LEAF_CERT_VALIDITY_DAYS);
173        if validity_days == 0 || validity_days > effective_max {
174            return Err(AxiamError::Validation {
175                message: format!(
176                    "validity_days must be between 1 and {effective_max} \
177                     (CA/Browser Forum BR hard cap: {MAX_LEAF_CERT_VALIDITY_DAYS} days)"
178                ),
179            });
180        }
181
182        // Scoped to the organization, so a CA id from another organization is
183        // not found rather than usable (T-98).
184        let ca_cert = self.ca_repo.get_by_id(org_id, issuer_ca_id).await?;
185
186        // ...and scoped to the *tenant* as well, which the organization scope
187        // alone never was (DF-017 / DF-025): `get_by_id` asks only
188        // `WHERE organization_id = $org_id`, so every CA of the organization —
189        // every other tenant's signing CA, and the organization anchor above
190        // them — was usable by any tenant administrator holding
191        // `certificates:generate`. The leaf was then recorded under the
192        // caller's tenant with another tenant's issuer, and chained to the same
193        // root every relying party in the organization trusts.
194        //
195        // Refused as `NotFound`, following the cross-organization precedent
196        // (`a_ca_in_another_organization_is_not_found`): a CA the caller may
197        // not use is a CA the caller cannot see, and the refusal must not tell
198        // an outsider whether it exists, is revoked, or has expired. Which is
199        // why it sits here, ahead of the status and window checks below.
200        let usable = match ca_cert.tenant_id {
201            // A tenant signing CA: usable by whoever is acting on that tenant,
202            // organization principal or not.
203            Some(ca_tenant) => ca_tenant == acting_tenant_id,
204            // An organization-level CA: the trust anchor. Issuing leaves
205            // directly under it is the thing the tenant tier exists to prevent,
206            // so only a principal that lives in the organization scope may.
207            None => scope == IssuingScope::Organization,
208        };
209        if !usable {
210            return Err(AxiamError::NotFound {
211                entity: "CA certificate".into(),
212                id: issuer_ca_id.to_string(),
213            });
214        }
215
216        if ca_cert.status != CertificateStatus::Active {
217            return Err(AxiamError::Certificate(
218                "CA certificate is not active".into(),
219            ));
220        }
221
222        // Validate CA certificate validity window
223        let now = Utc::now();
224        if now < ca_cert.not_before || now > ca_cert.not_after {
225            return Err(AxiamError::Certificate(
226                "CA certificate is expired or not yet valid".into(),
227            ));
228        }
229
230        // Fetch the signing key from whichever custodian this CA's row names —
231        // not from whichever is currently configured. A deployment that adopted
232        // Vault still has CAs whose keys are sealed into their rows, and asking
233        // the current setting about them would fail to find a key that is
234        // perfectly present.
235        //
236        // An imported CA with no key at all resolves to the `External`
237        // custodian, whose error says it has no private key and therefore
238        // cannot issue — rather than the bare "no stored private key" this used
239        // to produce for both that case and a genuine decryption failure.
240        let store = self.custodians.store_for(ca_cert.key_custody)?;
241
242        let not_before = now;
243        let requested_not_after = now
244            .checked_add_signed(Duration::days(i64::from(validity_days)))
245            .ok_or_else(|| AxiamError::Validation {
246                message: "validity_days produces a date out of range".into(),
247            })?;
248        // A leaf must not outlive the CA that signed it: a relying party stops
249        // trusting the chain the moment the issuer expires, so the extra days
250        // buy nothing and the certificate would fail every handshake made after
251        // the issuer's notAfter.
252        //
253        // This used to be a silent `min(requested, ca_cert.not_after)`. Silent
254        // was the wrong call: an operator asking for two years against a CA with
255        // three months left got a three-month certificate, no warning, and a
256        // renewal calendar built on a date the certificate does not carry. Say
257        // no and name the number they can actually have — see
258        // `issuer_bounded_validity_days`, which the API also exposes so the
259        // form can cap its own input before anyone submits it.
260        if requested_not_after > ca_cert.not_after {
261            let available = issuer_bounded_validity_days(now, ca_cert.not_after);
262            return Err(AxiamError::Validation {
263                message: format!(
264                    "validity_days is {} but the issuing CA expires on {} — a certificate \
265                     cannot outlive its issuer. The most this CA can grant today is {} day{}.",
266                    validity_days,
267                    ca_cert.not_after.format("%Y-%m-%d"),
268                    available,
269                    if available == 1 { "" } else { "s" },
270                ),
271            });
272        }
273
274        Ok(LeafIssuance {
275            ca_cert,
276            store,
277            not_before,
278            not_after: requested_not_after,
279        })
280    }

282    /// Generate a new certificate signed by a CA of this organization.
283    ///
284    /// `org_id` is required to look up the CA certificate and its encrypted
285    /// private key for signing.
286    ///
287    /// `scope` says what the caller is — see [`IssuingScope`]. Together with
288    /// `input.tenant_id`, which is the tenant being acted on and comes from the
289    /// authenticated context rather than the body, it decides which CAs of the
290    /// organization this call can reach at all.
291    ///
292    /// `max_validity_days` is the tenant-level cap (from tenant metadata).
293    /// Pass `None` to use the default ([`DEFAULT_LEAF_CERT_VALIDITY_DAYS`]).
294    /// The hard cap ([`MAX_LEAF_CERT_VALIDITY_DAYS`], 825 days) is always
295    /// enforced per CA/Browser Forum Baseline Requirements.
296    ///
297    /// `server_names` is the acting tenant's **effective**
298    /// `server_cert_allowed_names` (S-7), resolved by the caller from the
299    /// settings hierarchy. It is a required argument rather than something
300    /// this service looks up so that no caller can issue without having
301    /// answered the question; an empty slice is the shipped answer, and it
302    /// refuses every `Server` request. See
303    /// [`axiam_core::models::server_names::check_leaf_names`].
304    pub async fn generate(
305        &self,
306        org_id: Uuid,
307        scope: IssuingScope,
308        mut input: CreateCertificate,
309        max_validity_days: Option<u32>,
310        server_names: &[String],
311    ) -> AxiamResult<GeneratedCertificate> {
312        // DF-023. Normalised before anything reads it, so the locally signed
313        // path, the remote-custodian path and the stored row all carry the
314        // same common name. See [`crate::subject_common_name`].
315        input.subject = subject_common_name(&input.subject)?;
316
317        // S-7 — the name fence, before the CA is even looked up: a name the
318        // tenant may not use is refused the same way whichever CA it named
319        // and whichever custodian holds that CA's key.
320        let sans = check_leaf_names(
321            &input.cert_type,
322            &input.subject,
323            &input.subject_alt_names,
324            server_names,
325        )?;
326        let profile = LeafProfile::for_leaf(&input.cert_type, &input.key_algorithm);
327
328        let LeafIssuance {
329            ca_cert,
330            store,
331            not_before,
332            not_after,
333        } = self
334            .prepare_leaf_issuance(
335                org_id,
336                input.tenant_id,
337                scope,
338                input.issuer_ca_id,
339                input.validity_days,
340                max_validity_days,
341            )
342            .await?;
343
344        // A custodian that signs on AXIAM's behalf never hands the key over, so
345        // there is nothing to load and nothing to sign with here: the leaf key
346        // is generated locally, its public half goes out in a CSR, and a
347        // certificate comes back. See [`axiam_core::ca_keys`].
348        if store.signs_remotely() {
349            return self
350                .generate_remotely(&ca_cert, input, not_before, not_after, sans, profile)
351                .await;
352        }
353
354        // Fetch the signing key from whichever custodian this CA's row named.
355        let key_ref = CaService::<CA>::key_ref(&ca_cert);
356        let mut ca_private_key_pem = store
357            .load(&key_ref, ca_cert.encrypted_private_key.as_deref())
358            .await?
359            .to_string();
360
361        // CPU-bound: key generation + certificate signing run in spawn_blocking behind semaphore (CQ-B02).
362        let _permit = self
363            .crypto_semaphore
364            .acquire()
365            .await
366            .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
367
368        let ca_cert_pem = ca_cert.public_cert_pem.clone();
369        let ee_subject = input.subject.clone();
370        let key_algorithm = input.key_algorithm.clone();
371        let not_before_ts = not_before.timestamp();
372        let not_after_ts = not_after.timestamp();
373
374        let (private_key_pem, public_cert_pem, fingerprint) =
375            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String, String)> {
376                let ca_key_pair = KeyPair::from_pem(&ca_private_key_pem)
377                    .map_err(|e| AxiamError::Certificate(format!("invalid CA private key: {e}")))?;
378                // Scrub the decrypted CA private-key PEM from memory as soon as
379                // the KeyPair is parsed — it is not needed past this point and
380                // must not linger in the heap buffer (defense-in-depth).
381                ca_private_key_pem.zeroize();
382
383                // Reconstruct the signing CA issuer from its real, stored certificate
384                // PEM — NOT from the (mutable) `subject` field — so the issuer DN
385                // embedded in every leaf cert can never drift from the CA's actual
386                // Subject DN (QUAL-05/D-08, T-29-11). rcgen 0.14 moved
387                // `from_ca_cert_pem` onto `Issuer`, which now owns the signing key and
388                // is passed directly to `signed_by`.
389                let ca_issuer =
390                    Issuer::from_ca_cert_pem(&ca_cert_pem, ca_key_pair).map_err(|e| {
391                        AxiamError::Certificate(format!("invalid CA certificate PEM: {e}"))
392                    })?;
393
394                // Generate end-entity key pair.
395                let ee_key_pair = generate_keypair(&key_algorithm)?;
396                let private_key_pem = ee_key_pair.serialize_pem();
397
398                // Build end-entity certificate request — from the same
399                // function `sign_csr` overwrites a caller's parameters with, so
400                // the two paths cannot issue different shapes.
401                let ee_params =
402                    leaf_params(&ee_subject, not_before_ts, not_after_ts, &sans, &profile)?;
403
404                let cert = ee_params.signed_by(&ee_key_pair, &ca_issuer).map_err(|e| {
405                    AxiamError::Certificate(format!("certificate signing failed: {e}"))
406                })?;
407
408                let public_cert_pem = cert.pem();
409                let fingerprint = compute_fingerprint(cert.der());
410                Ok((private_key_pem, public_cert_pem, fingerprint))
411            })
412            .await
413            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
414
415        let store = StoreCertificate {
416            tenant_id: input.tenant_id,
417            issuer_ca_id: input.issuer_ca_id,
418            subject: input.subject,
419            public_cert_pem,
420            fingerprint,
421            cert_type: input.cert_type,
422            key_algorithm: input.key_algorithm,
423            not_before,
424            not_after,
425            metadata: input.metadata.unwrap_or(serde_json::json!({})),
426        };
427
428        let certificate = self.cert_repo.create(store).await?;
429
430        Ok(GeneratedCertificate {
431            certificate,
432            private_key_pem,
433            // Signed in-process by a CA whose certificate the caller can fetch;
434            // there is no chain here they cannot already assemble.
435            chain_pem: None,
436        })
437    }
438
439    /// Issue against a CA whose key lives in — and stays in — its custodian.
440    ///
441    /// The end-entity key is still generated here and still returned once: what
442    /// moves to the custodian is the *signature*, not the subscriber's key. The
443    /// CSR carries the public half and nothing else.
444    async fn generate_remotely(
445        &self,
446        ca_cert: &axiam_core::models::certificate::CaCertificate,
447        input: CreateCertificate,
448        not_before: DateTime<Utc>,
449        not_after: DateTime<Utc>,
450        sans: Vec<RequestedName>,
451        profile: LeafProfile,
452    ) -> AxiamResult<GeneratedCertificate> {
453        // Only the keygen and CSR are CPU-bound. The permit is dropped before
454        // the call to the custodian: holding it across a network round trip
455        // would let one slow signer block every other issuance in the
456        // deployment, which is the opposite of what the semaphore is for.
457        let (private_key_pem, csr_pem) = {
458            let _permit = self
459                .crypto_semaphore
460                .acquire()
461                .await
462                .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
463
464            let ee_subject = input.subject.clone();
465            let key_algorithm = input.key_algorithm.clone();
466            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String)> {
467                let ee_key_pair = generate_keypair(&key_algorithm)?;
468                let private_key_pem = ee_key_pair.serialize_pem();
469
470                let mut params = CertificateParams::new(Vec::<String>::new())
471                    .map_err(|e| AxiamError::Certificate(e.to_string()))?;
472                params
473                    .distinguished_name
474                    .push(DnType::CommonName, &ee_subject);
475                params.is_ca = IsCa::NoCa;
476                // S-7. The SANs go *in the CSR*, because that is the only place
477                // a verbatim signer reads them from: Vault's `sign-verbatim`
478                // ignores `alt_names` and `ip_sans` (measured on 1.18.3). This
479                // CSR is AXIAM's own, built from names the fence just admitted,
480                // so nothing a caller wrote reaches the list. The usages are
481                // not put here — they travel as `profile`, which the signer
482                // applies because this CSR requests none.
483                params.subject_alt_names = san_types(&sans)?;
484                // No validity window: a PKCS#10 request cannot carry one. The
485                // signer is told the lifetime out of band, as a TTL.
486                let csr = params.serialize_request(&ee_key_pair).map_err(|e| {
487                    AxiamError::Certificate(format!("certificate request failed: {e}"))
488                })?;
489                let csr_pem = csr
490                    .pem()
491                    .map_err(|e| AxiamError::Certificate(format!("CSR encoding failed: {e}")))?;
492                Ok((private_key_pem, csr_pem))
493            })
494            .await
495            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??
496        };
497
498        let key_ref = CaService::<CA>::key_ref(ca_cert);
499        let store = self.custodians.store_for(ca_cert.key_custody)?;
500        let signed = store
501            .sign_csr(
502                &key_ref,
503                &LeafSigningRequest {
504                    csr_pem,
505                    // AXIAM built this request, three lines up, from a key it
506                    // just generated: there is nothing in it a caller chose.
507                    csr_is_caller_supplied: false,
508                    ttl_seconds: (not_after - not_before).num_seconds(),
509                    profile,
510                },
511            )
512            .await?;
513
514        // What the signer produced, not what was asked for. A remote signer
515        // caps a TTL to its mount's own maximum without failing the call, so a
516        // row built from the request would claim a window the certificate does
517        // not have — and the fingerprint has to be of the real bytes or every
518        // lookup by fingerprint misses.
519        let issued = parse_issued_leaf(&signed.certificate_pem)?;
520
521        let store_record = StoreCertificate {
522            tenant_id: input.tenant_id,
523            issuer_ca_id: input.issuer_ca_id,
524            subject: input.subject,
525            public_cert_pem: signed.certificate_pem,
526            fingerprint: issued.fingerprint,
527            cert_type: input.cert_type,
528            key_algorithm: input.key_algorithm,
529            not_before: issued.not_before,
530            not_after: issued.not_after,
531            metadata: input.metadata.unwrap_or(serde_json::json!({})),
532        };
533
534        let certificate = self.cert_repo.create(store_record).await?;
535
536        Ok(GeneratedCertificate {
537            certificate,
538            private_key_pem,
539            chain_pem: (!signed.chain_pem.is_empty()).then(|| join_pem(&signed.chain_pem)),
540        })
541    }

543    /// The minimum RSA modulus AXIAM will sign a certificate over, in bits.
544    ///
545    /// Stated here rather than inferred from [`KeyAlgorithm::Rsa4096`]'s name,
546    /// because that variant is a label and not a measurement:
547    /// `ca::parse_ca_certificate` maps *any* RSA OID onto it, deliberately, so
548    /// that an imported root of some other size is still a usable root. A path
549    /// that enforces a floor cannot reuse that mapping, and an RSA-2048 CSR
550    /// signed and then recorded as `Rsa4096` is the exact shape of T-96.
551    const MIN_CSR_RSA_MODULUS_BITS: usize = 4096;
552
553    /// Sign a caller's certificate signing request, issuing an end-entity
554    /// certificate for a key AXIAM never sees.
555    ///
556    /// The BYOK counterpart to [`Self::generate`]. The subscriber's private key
557    /// is generated by whoever made the CSR and never crosses the wire in
558    /// either direction — which is why the answer is a [`Certificate`] and not
559    /// a `GeneratedCertificate`: there is no key to return.
560    ///
561    /// # What the caller decides, and what AXIAM decides
562    ///
563    /// The caller decides the key and the subject. AXIAM decides everything
564    /// else, and the difference is enforced rather than documented:
565    ///
566    /// - **Possession** is proved by the request's own self-signature, checked
567    ///   by [`ca::inspect_csr`] before anything else happens. Without it a
568    ///   caller could have a certificate minted over somebody else's public key.
569    /// - **The key** must be Ed25519, or RSA with a modulus of at least
570    ///   [`Self::MIN_CSR_RSA_MODULUS_BITS`] bits. The modulus is measured, not
571    ///   taken from a label.
572    /// - **The extensions** are AXIAM's. A CSR requesting `subjectAltName`,
573    ///   `keyUsage` or `extendedKeyUsage` is refused by name rather than
574    ///   silently stripped; every other requested extension is discarded when
575    ///   the parameter set is overwritten with the leaf parameters
576    ///   [`Self::generate`] builds. A CSR asking to be an unconstrained CA
577    ///   comes back a leaf.
578    /// - **The validity, the issuer and the tenant** come from
579    ///   [`Self::prepare_leaf_issuance`], the same function `generate` uses, so
580    ///   the two paths cannot drift on a revoked CA, an expired one, a CA
581    ///   belonging to another organization, or a certificate that would outlive
582    ///   its issuer.
583    ///
584    /// # Why `keyUsage` is refused rather than dropped
585    ///
586    /// On the in-process path it could be dropped: `request.params` is replaced
587    /// wholesale before signing and nothing the caller asked for survives. Under
588    /// [`CaKeyCustody::VaultPki`] it could not. Vault's `sign-verbatim`
589    /// discards the `key_usage` and `ext_key_usage` request parameters when the
590    /// CSR carries those extensions, and issues what the CSR asked for — so a
591    /// silent strip would be a promise AXIAM keeps on one custodian and breaks
592    /// on the other, for the same CSR. One rule that holds everywhere is worth
593    /// more than a convenience that holds in one deployment.
594    ///
595    /// `basicConstraints` needs no such rule: the in-process path overwrites it
596    /// and Vault ignores a CSR's basic constraints outright.
597    ///
598    /// # Names (S-7)
599    ///
600    /// `subjectAltName` stays refused *in the CSR*. A `Server` certificate's
601    /// names come from [`SignCertificateCsr::subject_alt_names`], through the
602    /// same fence [`Self::generate`] runs, with `server_names` as the acting
603    /// tenant's effective allow-list. Under a custodian that signs remotely a
604    /// `Server` request is refused: such a signer signs the caller's CSR
605    /// verbatim, takes SANs only from it, and the CSR may not carry any — so
606    /// there is no channel left through which the explicit names could reach
607    /// the certificate. Use [`Self::generate`] under such a CA.
608    pub async fn sign_csr(
609        &self,
610        org_id: Uuid,
611        scope: IssuingScope,
612        input: SignCertificateCsr,
613        max_validity_days: Option<u32>,
614        server_names: &[String],
615    ) -> AxiamResult<Certificate> {
616        // Parsed before the CA is even looked up, and outside the blocking
617        // task: a malformed or unsigned request is by far the likeliest failure
618        // on an endpoint whose input is pasted by hand, and it must be a 400
619        // naming the CSR rather than a 500 out of a custodian.
620        let facts = ca::inspect_csr(&input.csr_pem)?;
621
622        if let Some(bits) = facts.rsa_modulus_bits
623            && bits < Self::MIN_CSR_RSA_MODULUS_BITS
624        {
625            return Err(AxiamError::Validation {
626                message: format!(
627                    "the certificate signing request carries an RSA key with a {bits}-bit \
628                     modulus; AXIAM signs Ed25519 keys and RSA keys of at least {} bits",
629                    Self::MIN_CSR_RSA_MODULUS_BITS
630                ),
631            });
632        }
633
634        if !facts.refused_extensions.is_empty() {
635            return Err(AxiamError::Validation {
636                message: format!(
637                    "the certificate signing request asks for {}, which AXIAM does not sign \
638                     over on an end-entity certificate. It is refused rather than dropped \
639                     because a certificate issued without what you asked for, and with \
640                     nothing said, would fail where you deploy it. Remove the extension \
641                     request and submit the CSR again.",
642                    facts.refused_extensions.join(" and ")
643                ),
644            });
645        }
646
647        // S-7 — the same fence `generate` runs, over the CSR's own common name
648        // and the request's explicit SANs.
649        let sans = check_leaf_names(
650            &input.cert_type,
651            &facts.common_name,
652            &input.subject_alt_names,
653            server_names,
654        )?;
655        let profile = LeafProfile::for_leaf(&input.cert_type, &facts.key_algorithm);
656
657        let LeafIssuance {
658            ca_cert,
659            store,
660            not_before,
661            not_after,
662        } = self
663            .prepare_leaf_issuance(
664                org_id,
665                input.tenant_id,
666                scope,
667                input.issuer_ca_id,
668                input.validity_days,
669                max_validity_days,
670            )
671            .await?;
672
673        if store.signs_remotely() && !sans.is_empty() {
674            return Err(AxiamError::Validation {
675                message: "a Server certificate cannot be issued from a caller's CSR under a CA \
676                          whose key is held by a remote signer: the signer copies the CSR's \
677                          subjectAltName verbatim and ignores any other, and AXIAM refuses a CSR \
678                          that carries one. Issue it with POST /api/v1/certificates under this \
679                          CA, or sign the CSR under a CA whose key AXIAM holds"
680                    .into(),
681            });
682        }
683
684        let (public_cert_pem, fingerprint, chain_pem, issued_window) = if store.signs_remotely() {
685            let key_ref = CaService::<CA>::key_ref(&ca_cert);
686            let signed = store
687                .sign_csr(
688                    &key_ref,
689                    &LeafSigningRequest {
690                        csr_pem: input.csr_pem.clone(),
691                        ttl_seconds: (not_after - not_before).num_seconds(),
692                        // These bytes are the caller's. See the custodian's own
693                        // doc comment for what it does about that.
694                        csr_is_caller_supplied: true,
695                        profile,
696                    },
697                )
698                .await?;
699            // What the signer produced, not what was asked for: a remote signer
700            // caps a TTL to its mount's own maximum without failing the call.
701            let issued = parse_issued_leaf(&signed.certificate_pem)?;
702            let chain = (!signed.chain_pem.is_empty()).then(|| join_pem(&signed.chain_pem));
703            (
704                signed.certificate_pem,
705                issued.fingerprint,
706                chain,
707                Some((issued.not_before, issued.not_after)),
708            )
709        } else {
710            let key_ref = CaService::<CA>::key_ref(&ca_cert);
711            let mut ca_private_key_pem = store
712                .load(&key_ref, ca_cert.encrypted_private_key.as_deref())
713                .await?
714                .to_string();
715
716            let _permit = self
717                .crypto_semaphore
718                .acquire()
719                .await
720                .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
721
722            let ca_cert_pem = ca_cert.public_cert_pem.clone();
723            let csr_pem = input.csr_pem.clone();
724            let subject = facts.common_name.clone();
725            let not_before_ts = not_before.timestamp();
726            let not_after_ts = not_after.timestamp();
727
728            let (pem, fingerprint) =
729                tokio::task::spawn_blocking(move || -> AxiamResult<(String, String)> {
730                    let ca_key_pair = KeyPair::from_pem(&ca_private_key_pem).map_err(|e| {
731                        AxiamError::Certificate(format!("invalid CA private key: {e}"))
732                    })?;
733                    ca_private_key_pem.zeroize();
734
735                    let ca_issuer =
736                        Issuer::from_ca_cert_pem(&ca_cert_pem, ca_key_pair).map_err(|e| {
737                            AxiamError::Certificate(format!("invalid CA certificate PEM: {e}"))
738                        })?;
739
740                    // Parsed a second time, inside the task, because rcgen's
741                    // signing API needs its own representation. The answer that
742                    // matters — whether the signature verifies — was already
743                    // established above, on the same bytes; this is not a
744                    // second opinion about it.
745                    let mut request = rcgen::CertificateSigningRequestParams::from_pem(&csr_pem)
746                        .map_err(|e| AxiamError::Validation {
747                            message: format!("the certificate signing request was rejected: {e}"),
748                        })?;
749                    // Everything the request asked to be, overwritten by what
750                    // AXIAM has decided it is — the same leaf parameters
751                    // `generate` builds, so the two paths issue the same shape
752                    // of certificate for the same request.
753                    request.params =
754                        leaf_params(&subject, not_before_ts, not_after_ts, &sans, &profile)?;
755                    // Deliberately **not** `use_authority_key_identifier_extension`,
756                    // which the intermediate CSR path sets. Setting it here put
757                    // an authorityKeyIdentifier on a CSR-signed leaf that a
758                    // generated leaf does not carry, which is exactly the
759                    // divergence D-5 forbids: two ways of asking for the same
760                    // certificate producing two different certificates. An AKI
761                    // on leaves is worth having and is a change to make on both
762                    // paths at once — the KU/EKU profile (S-7) was made that
763                    // way — not a side effect of how the key was made.
764
765                    let cert = request.signed_by(&ca_issuer).map_err(|e| {
766                        AxiamError::Certificate(format!("certificate signing failed: {e}"))
767                    })?;
768                    let pem = cert.pem();
769                    let fingerprint = compute_fingerprint(cert.der());
770                    Ok((pem, fingerprint))
771                })
772                .await
773                .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
774
775            // Signed in-process by a CA whose certificate the caller can fetch;
776            // there is no chain here they cannot already assemble.
777            (pem, fingerprint, None, None)
778        };
779
780        let (row_not_before, row_not_after) = issued_window.unwrap_or((not_before, not_after));
781
782        let certificate = self
783            .cert_repo
784            .create(StoreCertificate {
785                tenant_id: input.tenant_id,
786                issuer_ca_id: input.issuer_ca_id,
787                // What the certificate says, never what the caller said
788                // separately — there is no separate field for them to say it in
789                // (T-194: the row and the certificate cannot disagree).
790                subject: facts.common_name,
791                public_cert_pem,
792                fingerprint,
793                cert_type: input.cert_type,
794                // Read off the CSR's own public key, not asserted by the caller.
795                key_algorithm: facts.key_algorithm,
796                not_before: row_not_before,
797                not_after: row_not_after,
798                metadata: input.metadata.unwrap_or(serde_json::json!({})),
799            })
800            .await?;
801
802        // `chain_pem` has nowhere to go on a `Certificate` and is not dropped
803        // silently: under the custodians that return one, the chain is the
804        // issuing CA's own certificate, which `GET /certificates/{id}` already
805        // resolves through `issuer_ca_id`. A caller holding the certificate can
806        // assemble the chain from data the API already gives them.
807        let _ = chain_pem;
808
809        Ok(certificate)
810    }

839/// What a certificate a remote signer returned actually says about itself.
840struct IssuedLeaf {
841    fingerprint: String,
842    not_before: DateTime<Utc>,
843    not_after: DateTime<Utc>,
844}
845
846/// The parameters every AXIAM leaf certificate is built from: the common name,
847/// the validity window, `CA:FALSE`, the SANs the name fence admitted, and the
848/// per-type usage profile (S-7).
849///
850/// The SANs are empty for every type but `Server`, and are only ever the names
851/// [`check_leaf_names`] returned — never anything from a CSR. The profile is
852/// [`LeafProfile::for_leaf`]; before S-7 a leaf carried no keyUsage and no
853/// extendedKeyUsage, which RFC 5280 reads as "any usage", so adding them only
854/// narrows. Leaves issued earlier keep carrying neither.
855///
856/// Shared by [`CertService::generate`], which builds a key to go with it, and
857/// by [`CertService::sign_csr`], which overwrites a caller's requested
858/// parameters with it. That sharing is the enforcement: "a CSR-signed leaf is
859/// the same shape as a generated one" is true because there is one function
860/// that says what the shape is.
861fn leaf_params(
862    subject: &str,
863    not_before_ts: i64,
864    not_after_ts: i64,
865    sans: &[RequestedName],
866    profile: &LeafProfile,
867) -> AxiamResult<CertificateParams> {
868    let mut params = CertificateParams::new(Vec::<String>::new())
869        .map_err(|e| AxiamError::Certificate(e.to_string()))?;
870    params.distinguished_name.push(DnType::CommonName, subject);
871    params.is_ca = IsCa::NoCa;
872    params.subject_alt_names = san_types(sans)?;
873    params.key_usages = profile
874        .key_usage
875        .iter()
876        .map(|u| match u {
877            LeafKeyUsage::DigitalSignature => KeyUsagePurpose::DigitalSignature,
878            LeafKeyUsage::KeyEncipherment => KeyUsagePurpose::KeyEncipherment,
879        })
880        .collect();
881    params.extended_key_usages = profile
882        .extended_key_usage
883        .iter()
884        .map(|u| match u {
885            LeafExtendedKeyUsage::ClientAuth => ExtendedKeyUsagePurpose::ClientAuth,
886            LeafExtendedKeyUsage::ServerAuth => ExtendedKeyUsagePurpose::ServerAuth,
887        })
888        .collect();
889    params.not_before = time::OffsetDateTime::from_unix_timestamp(not_before_ts)
890        .map_err(|e| AxiamError::Certificate(format!("invalid notBefore: {e}")))?;
891    params.not_after = time::OffsetDateTime::from_unix_timestamp(not_after_ts)
892        .map_err(|e| AxiamError::Certificate(format!("invalid notAfter: {e}")))?;
893    Ok(params)
894}
895
896/// The fence's names in rcgen's form.
897fn san_types(sans: &[RequestedName]) -> AxiamResult<Vec<SanType>> {
898    sans.iter()
899        .map(|n| match n {
900            RequestedName::Dns(d) => rcgen::string::Ia5String::try_from(d.as_str())
901                .map(SanType::DnsName)
902                .map_err(|e| AxiamError::Certificate(format!("invalid DNS name: {e}"))),
903            RequestedName::Ip(ip) => Ok(SanType::IpAddress(*ip)),
904        })
905        .collect()
906}
907
908/// Read back a signed leaf, so the row describes the certificate rather than
909/// the request that produced it.
910fn parse_issued_leaf(pem: &str) -> AxiamResult<IssuedLeaf> {
911    let (_, block) = x509_parser::pem::parse_x509_pem(pem.as_bytes()).map_err(|e| {
912        AxiamError::Certificate(format!(
913            "the signer returned something that is not a PEM certificate: {e}"
914        ))
915    })?;
916    let (_, cert) = X509Certificate::from_der(&block.contents).map_err(|e| {
917        AxiamError::Certificate(format!("the signed certificate could not be parsed: {e}"))
918    })?;
919
920    let to_utc = |t: i64, what: &str| -> AxiamResult<DateTime<Utc>> {
921        DateTime::from_timestamp(t, 0).ok_or_else(|| {
922            AxiamError::Certificate(format!("the signed certificate's {what} is out of range"))
923        })
924    };
925
926    Ok(IssuedLeaf {
927        fingerprint: compute_fingerprint(&block.contents),
928        not_before: to_utc(cert.validity().not_before.timestamp(), "notBefore")?,
929        not_after: to_utc(cert.validity().not_after.timestamp(), "notAfter")?,
930    })
931}
