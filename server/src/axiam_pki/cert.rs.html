25/// Hard cap for leaf certificate validity: 825 days (~27 months).
26///
27/// Aligns with CA/Browser Forum Baseline Requirements and Apple/Mozilla
28/// root program policies. Internal PKI may allow up to this limit;
29/// tenants can configure a lower per-tenant maximum via the
30/// `max_certificate_validity_days` key in their metadata.
31pub const MAX_LEAF_CERT_VALIDITY_DAYS: u32 = 825;
32
33/// Default leaf certificate validity when no tenant override is set: 365 days.
34pub const DEFAULT_LEAF_CERT_VALIDITY_DAYS: u32 = 365;
35
36/// Service for tenant-level certificate operations.
37#[derive(Clone)]
38pub struct CertService<CA, CR> {
39    ca_repo: CA,
40    cert_repo: CR,
41    #[allow(dead_code)]
42    config: PkiConfig,
43    /// Shared bounding semaphore for CPU-bound crypto (CQ-B02).
44    crypto_semaphore: Arc<Semaphore>,
45    /// Who holds the CA signing keys. See [`crate::ca_key_store`].
46    custodians: Arc<CaKeyCustodians>,
47}
48
49impl<CA: CaCertificateRepository, CR: CertificateRepository> CertService<CA, CR> {
50    pub fn new(
51        ca_repo: CA,
52        cert_repo: CR,
53        config: PkiConfig,
54        crypto_semaphore: Arc<Semaphore>,
55        custodians: Arc<CaKeyCustodians>,
56    ) -> Self {
57        Self {
58            ca_repo,
59            cert_repo,
60            config,
61            crypto_semaphore,
62            custodians,
63        }
64    }

66    /// Generate a new certificate signed by an organization CA.
67    ///
68    /// `org_id` is required to look up the CA certificate and its encrypted
69    /// private key for signing.
70    ///
71    /// `max_validity_days` is the tenant-level cap (from tenant metadata).
72    /// Pass `None` to use the default ([`DEFAULT_LEAF_CERT_VALIDITY_DAYS`]).
73    /// The hard cap ([`MAX_LEAF_CERT_VALIDITY_DAYS`], 825 days) is always
74    /// enforced per CA/Browser Forum Baseline Requirements.
75    pub async fn generate(
76        &self,
77        org_id: Uuid,
78        input: CreateCertificate,
79        max_validity_days: Option<u32>,
80    ) -> AxiamResult<GeneratedCertificate> {
81        // Enforce validity_days bounds: > 0 and <= tenant/hard cap
82        let effective_max = max_validity_days
83            .unwrap_or(DEFAULT_LEAF_CERT_VALIDITY_DAYS)
84            .min(MAX_LEAF_CERT_VALIDITY_DAYS);
85        if input.validity_days == 0 || input.validity_days > effective_max {
86            return Err(AxiamError::Validation {
87                message: format!(
88                    "validity_days must be between 1 and {effective_max} \
89                     (CA/Browser Forum BR hard cap: {MAX_LEAF_CERT_VALIDITY_DAYS} days)"
90                ),
91            });
92        }
93
94        // Fetch the CA cert to get the encrypted private key.
95        let ca_cert = self.ca_repo.get_by_id(org_id, input.issuer_ca_id).await?;
96
97        if ca_cert.status != CertificateStatus::Active {
98            return Err(AxiamError::Certificate(
99                "CA certificate is not active".into(),
100            ));
101        }
102
103        // Validate CA certificate validity window
104        let now = Utc::now();
105        if now < ca_cert.not_before || now > ca_cert.not_after {
106            return Err(AxiamError::Certificate(
107                "CA certificate is expired or not yet valid".into(),
108            ));
109        }
110
111        // Fetch the signing key from whichever custodian this CA's row names —
112        // not from whichever is currently configured. A deployment that adopted
113        // Vault still has CAs whose keys are sealed into their rows, and asking
114        // the current setting about them would fail to find a key that is
115        // perfectly present.
116        //
117        // An imported CA with no key at all resolves to the `External`
118        // custodian, whose error says it has no private key and therefore
119        // cannot issue — rather than the bare "no stored private key" this used
120        // to produce for both that case and a genuine decryption failure.
121        let store = self.custodians.store_for(ca_cert.key_custody)?;
122
123        let not_before = now;
124        let requested_not_after = now
125            .checked_add_signed(Duration::days(i64::from(input.validity_days)))
126            .ok_or_else(|| AxiamError::Validation {
127                message: "validity_days produces a date out of range".into(),
128            })?;
129        // Cap leaf validity to CA validity window
130        let not_after = std::cmp::min(requested_not_after, ca_cert.not_after);
131
132        // A custodian that signs on AXIAM's behalf never hands the key over, so
133        // there is nothing to load and nothing to sign with here: the leaf key
134        // is generated locally, its public half goes out in a CSR, and a
135        // certificate comes back. See [`axiam_core::ca_keys`].
136        if store.signs_remotely() {
137            return self
138                .generate_remotely(&ca_cert, input, not_before, not_after)
139                .await;
140        }
141
142        // Fetch the signing key from whichever custodian this CA's row named.
143        let key_ref = CaService::<CA>::key_ref(&ca_cert);
144        let mut ca_private_key_pem = store
145            .load(&key_ref, ca_cert.encrypted_private_key.as_deref())
146            .await?
147            .to_string();
148
149        // CPU-bound: key generation + certificate signing run in spawn_blocking behind semaphore (CQ-B02).
150        let _permit = self
151            .crypto_semaphore
152            .acquire()
153            .await
154            .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
155
156        let ca_cert_pem = ca_cert.public_cert_pem.clone();
157        let ee_subject = input.subject.clone();
158        let key_algorithm = input.key_algorithm.clone();
159        let not_before_ts = not_before.timestamp();
160        let not_after_ts = not_after.timestamp();
161
162        let (private_key_pem, public_cert_pem, fingerprint) =
163            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String, String)> {
164                let ca_key_pair = KeyPair::from_pem(&ca_private_key_pem)
165                    .map_err(|e| AxiamError::Certificate(format!("invalid CA private key: {e}")))?;
166                // Scrub the decrypted CA private-key PEM from memory as soon as
167                // the KeyPair is parsed — it is not needed past this point and
168                // must not linger in the heap buffer (defense-in-depth).
169                ca_private_key_pem.zeroize();
170
171                // Reconstruct the signing CA issuer from its real, stored certificate
172                // PEM — NOT from the (mutable) `subject` field — so the issuer DN
173                // embedded in every leaf cert can never drift from the CA's actual
174                // Subject DN (QUAL-05/D-08, T-29-11). rcgen 0.14 moved
175                // `from_ca_cert_pem` onto `Issuer`, which now owns the signing key and
176                // is passed directly to `signed_by`.
177                let ca_issuer =
178                    Issuer::from_ca_cert_pem(&ca_cert_pem, ca_key_pair).map_err(|e| {
179                        AxiamError::Certificate(format!("invalid CA certificate PEM: {e}"))
180                    })?;
181
182                // Generate end-entity key pair.
183                let ee_key_pair = generate_keypair(&key_algorithm)?;
184                let private_key_pem = ee_key_pair.serialize_pem();
185
186                // Build end-entity certificate request.
187                let mut ee_params = CertificateParams::new(Vec::<String>::new())
188                    .map_err(|e| AxiamError::Certificate(e.to_string()))?;
189                ee_params
190                    .distinguished_name
191                    .push(DnType::CommonName, &ee_subject);
192                ee_params.is_ca = IsCa::NoCa;
193                ee_params.not_before = time::OffsetDateTime::from_unix_timestamp(not_before_ts)
194                    .expect("valid timestamp");
195                ee_params.not_after = time::OffsetDateTime::from_unix_timestamp(not_after_ts)
196                    .expect("valid timestamp");
197
198                let cert = ee_params.signed_by(&ee_key_pair, &ca_issuer).map_err(|e| {
199                    AxiamError::Certificate(format!("certificate signing failed: {e}"))
200                })?;
201
202                let public_cert_pem = cert.pem();
203                let fingerprint = compute_fingerprint(cert.der());
204                Ok((private_key_pem, public_cert_pem, fingerprint))
205            })
206            .await
207            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
208
209        let store = StoreCertificate {
210            tenant_id: input.tenant_id,
211            issuer_ca_id: input.issuer_ca_id,
212            subject: input.subject,
213            public_cert_pem,
214            fingerprint,
215            cert_type: input.cert_type,
216            key_algorithm: input.key_algorithm,
217            not_before,
218            not_after,
219            metadata: input.metadata.unwrap_or(serde_json::json!({})),
220        };
221
222        let certificate = self.cert_repo.create(store).await?;
223
224        Ok(GeneratedCertificate {
225            certificate,
226            private_key_pem,
227            // Signed in-process by a CA whose certificate the caller can fetch;
228            // there is no chain here they cannot already assemble.
229            chain_pem: None,
230        })
231    }
232
233    /// Issue against a CA whose key lives in — and stays in — its custodian.
234    ///
235    /// The end-entity key is still generated here and still returned once: what
236    /// moves to the custodian is the *signature*, not the subscriber's key. The
237    /// CSR carries the public half and nothing else.
238    async fn generate_remotely(
239        &self,
240        ca_cert: &axiam_core::models::certificate::CaCertificate,
241        input: CreateCertificate,
242        not_before: DateTime<Utc>,
243        not_after: DateTime<Utc>,
244    ) -> AxiamResult<GeneratedCertificate> {
245        // Only the keygen and CSR are CPU-bound. The permit is dropped before
246        // the call to the custodian: holding it across a network round trip
247        // would let one slow signer block every other issuance in the
248        // deployment, which is the opposite of what the semaphore is for.
249        let (private_key_pem, csr_pem) = {
250            let _permit = self
251                .crypto_semaphore
252                .acquire()
253                .await
254                .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
255
256            let ee_subject = input.subject.clone();
257            let key_algorithm = input.key_algorithm.clone();
258            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String)> {
259                let ee_key_pair = generate_keypair(&key_algorithm)?;
260                let private_key_pem = ee_key_pair.serialize_pem();
261
262                let mut params = CertificateParams::new(Vec::<String>::new())
263                    .map_err(|e| AxiamError::Certificate(e.to_string()))?;
264                params
265                    .distinguished_name
266                    .push(DnType::CommonName, &ee_subject);
267                params.is_ca = IsCa::NoCa;
268                // No validity window: a PKCS#10 request cannot carry one. The
269                // signer is told the lifetime out of band, as a TTL.
270                let csr = params.serialize_request(&ee_key_pair).map_err(|e| {
271                    AxiamError::Certificate(format!("certificate request failed: {e}"))
272                })?;
273                let csr_pem = csr
274                    .pem()
275                    .map_err(|e| AxiamError::Certificate(format!("CSR encoding failed: {e}")))?;
276                Ok((private_key_pem, csr_pem))
277            })
278            .await
279            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??
280        };
281
282        let key_ref = CaService::<CA>::key_ref(ca_cert);
283        let store = self.custodians.store_for(ca_cert.key_custody)?;
284        let signed = store
285            .sign_csr(
286                &key_ref,
287                &LeafSigningRequest {
288                    csr_pem,
289                    ttl_seconds: (not_after - not_before).num_seconds(),
290                },
291            )
292            .await?;
293
294        // What the signer produced, not what was asked for. A remote signer
295        // caps a TTL to its mount's own maximum without failing the call, so a
296        // row built from the request would claim a window the certificate does
297        // not have — and the fingerprint has to be of the real bytes or every
298        // lookup by fingerprint misses.
299        let issued = parse_issued_leaf(&signed.certificate_pem)?;
300
301        let store_record = StoreCertificate {
302            tenant_id: input.tenant_id,
303            issuer_ca_id: input.issuer_ca_id,
304            subject: input.subject,
305            public_cert_pem: signed.certificate_pem,
306            fingerprint: issued.fingerprint,
307            cert_type: input.cert_type,
308            key_algorithm: input.key_algorithm,
309            not_before: issued.not_before,
310            not_after: issued.not_after,
311            metadata: input.metadata.unwrap_or(serde_json::json!({})),
312        };
313
314        let certificate = self.cert_repo.create(store_record).await?;
315
316        Ok(GeneratedCertificate {
317            certificate,
318            private_key_pem,
319            chain_pem: (!signed.chain_pem.is_empty()).then(|| join_pem(&signed.chain_pem)),
320        })
321    }
322

350/// What a certificate a remote signer returned actually says about itself.
351struct IssuedLeaf {
352    fingerprint: String,
353    not_before: DateTime<Utc>,
354    not_after: DateTime<Utc>,
355}
356
357/// Read back a signed leaf, so the row describes the certificate rather than
358/// the request that produced it.
359fn parse_issued_leaf(pem: &str) -> AxiamResult<IssuedLeaf> {
360    let (_, block) = x509_parser::pem::parse_x509_pem(pem.as_bytes()).map_err(|e| {
361        AxiamError::Certificate(format!(
362            "the signer returned something that is not a PEM certificate: {e}"
363        ))
364    })?;
365    let (_, cert) = X509Certificate::from_der(&block.contents).map_err(|e| {
366        AxiamError::Certificate(format!("the signed certificate could not be parsed: {e}"))
367    })?;
368
369    let to_utc = |t: i64, what: &str| -> AxiamResult<DateTime<Utc>> {
370        DateTime::from_timestamp(t, 0).ok_or_else(|| {
371            AxiamError::Certificate(format!("the signed certificate's {what} is out of range"))
372        })
373    };
374
375    Ok(IssuedLeaf {
376        fingerprint: compute_fingerprint(&block.contents),
377        not_before: to_utc(cert.validity().not_before.timestamp(), "notBefore")?,
378        not_after: to_utc(cert.validity().not_after.timestamp(), "notAfter")?,
379    })
380}
