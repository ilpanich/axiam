26/// Hard cap for leaf certificate validity: 825 days (~27 months).
27///
28/// Aligns with CA/Browser Forum Baseline Requirements and Apple/Mozilla
29/// root program policies. Internal PKI may allow up to this limit;
30/// tenants can configure a lower per-tenant maximum via the
31/// `max_certificate_validity_days` key in their metadata.
32pub const MAX_LEAF_CERT_VALIDITY_DAYS: u32 = 825;
33
34/// Default leaf certificate validity when no tenant override is set: 365 days.
35pub const DEFAULT_LEAF_CERT_VALIDITY_DAYS: u32 = 365;
36
37/// The longest validity, in whole days, an issuer expiring at `issuer_not_after`
38/// can still grant to something issued at `now`.
39///
40/// A certificate must not outlive the CA that signed it — past the issuer's
41/// notAfter the chain stops validating, so the extra days are not merely
42/// useless, they are days the holder believes they have and does not. Both the
43/// leaf path ([`CertService::generate`]) and the intermediate path
44/// (`ca::generate_intermediate`) refuse a request above this number and quote it
45/// back, and the certificates API returns it per CA so the admin UI can cap its
46/// own input rather than discovering the limit on submit.
47///
48/// Rounded **down** to whole days, and floored at zero: an issuer with 36 hours
49/// left can grant one day, not two, and an expired issuer can grant nothing.
50/// (Callers reject an expired or not-yet-valid issuer before reaching here; the
51/// zero is a total function's answer, not a code path anyone rides.)
52///
53/// # Examples
54///
55/// ```
56/// use axiam_pki::cert::issuer_bounded_validity_days;
57/// use chrono::{Duration, Utc};
58///
59/// let now = Utc::now();
60/// // A CA with 90 days and 12 hours left grants 90 days, never 91.
61/// let ca_expiry = now + Duration::days(90) + Duration::hours(12);
62/// assert_eq!(issuer_bounded_validity_days(now, ca_expiry), 90);
63///
64/// // An issuer already past its notAfter grants nothing.
65/// assert_eq!(issuer_bounded_validity_days(now, now - Duration::days(1)), 0);
66/// ```
67pub fn issuer_bounded_validity_days(now: DateTime<Utc>, issuer_not_after: DateTime<Utc>) -> u32 {
68    let remaining = (issuer_not_after - now).num_days();
69    u32::try_from(remaining.max(0)).unwrap_or(u32::MAX)
70}

72/// What [`CertService::prepare_leaf_issuance`] establishes for both leaf paths.
73///
74/// Borrowed rather than owned for `store`: [`CaKeyCustodians::store_for`] hands
75/// back a reference into the service's own custodian table, and cloning a
76/// custodian to avoid one lifetime would duplicate a Vault client per issuance.
77struct LeafIssuance<'a> {
78    ca_cert: axiam_core::models::certificate::CaCertificate,
79    store: &'a dyn axiam_core::ca_keys::CaKeyStore,
80    not_before: DateTime<Utc>,
81    not_after: DateTime<Utc>,
82}

99        ca_repo: CA,

101        config: PkiConfig,
102        crypto_semaphore: Arc<Semaphore>,
103        custodians: Arc<CaKeyCustodians>,
104    ) -> Self {
105        Self {
106            ca_repo,
107            cert_repo,
108            config,
109            crypto_semaphore,
110            custodians,
111        }

114    /// Everything both leaf paths must establish before a certificate can be
115    /// signed: the validity bounds, the issuing CA, and which custodian holds
116    /// its key.
117    ///
118    /// Factored out of [`Self::generate`] when [`Self::sign_csr`] arrived,
119    /// rather than copied into it. Every check here is a refusal that must read
120    /// the same on both paths — a CSR-signed certificate that could be issued
121    /// under a revoked CA, or outlive its issuer, while a generated one could
122    /// not, would be a hole with the shape of a second implementation.
123    /// `generate`'s own tests are the regression net for the move.
124    async fn prepare_leaf_issuance(
125        &self,
126        org_id: Uuid,
127        issuer_ca_id: Uuid,
128        validity_days: u32,
129        max_validity_days: Option<u32>,
130    ) -> AxiamResult<LeafIssuance<'_>> {
131        // Enforce validity_days bounds: > 0 and <= tenant/hard cap
132        let effective_max = max_validity_days
133            .unwrap_or(DEFAULT_LEAF_CERT_VALIDITY_DAYS)
134            .min(MAX_LEAF_CERT_VALIDITY_DAYS);
135        if validity_days == 0 || validity_days > effective_max {
136            return Err(AxiamError::Validation {
137                message: format!(
138                    "validity_days must be between 1 and {effective_max} \
139                     (CA/Browser Forum BR hard cap: {MAX_LEAF_CERT_VALIDITY_DAYS} days)"
140                ),
141            });
142        }
143
144        // Scoped to the organization, so a CA id from another organization is
145        // not found rather than usable (T-98).
146        let ca_cert = self.ca_repo.get_by_id(org_id, issuer_ca_id).await?;
147
148        if ca_cert.status != CertificateStatus::Active {
149            return Err(AxiamError::Certificate(
150                "CA certificate is not active".into(),
151            ));
152        }
153
154        // Validate CA certificate validity window
155        let now = Utc::now();
156        if now < ca_cert.not_before || now > ca_cert.not_after {
157            return Err(AxiamError::Certificate(
158                "CA certificate is expired or not yet valid".into(),
159            ));
160        }
161
162        // Fetch the signing key from whichever custodian this CA's row names —
163        // not from whichever is currently configured. A deployment that adopted
164        // Vault still has CAs whose keys are sealed into their rows, and asking
165        // the current setting about them would fail to find a key that is
166        // perfectly present.
167        //
168        // An imported CA with no key at all resolves to the `External`
169        // custodian, whose error says it has no private key and therefore
170        // cannot issue — rather than the bare "no stored private key" this used
171        // to produce for both that case and a genuine decryption failure.
172        let store = self.custodians.store_for(ca_cert.key_custody)?;
173
174        let not_before = now;
175        let requested_not_after = now
176            .checked_add_signed(Duration::days(i64::from(validity_days)))
177            .ok_or_else(|| AxiamError::Validation {
178                message: "validity_days produces a date out of range".into(),
179            })?;
180        // A leaf must not outlive the CA that signed it: a relying party stops
181        // trusting the chain the moment the issuer expires, so the extra days
182        // buy nothing and the certificate would fail every handshake made after
183        // the issuer's notAfter.
184        //
185        // This used to be a silent `min(requested, ca_cert.not_after)`. Silent
186        // was the wrong call: an operator asking for two years against a CA with
187        // three months left got a three-month certificate, no warning, and a
188        // renewal calendar built on a date the certificate does not carry. Say
189        // no and name the number they can actually have — see
190        // `issuer_bounded_validity_days`, which the API also exposes so the
191        // form can cap its own input before anyone submits it.
192        if requested_not_after > ca_cert.not_after {
193            let available = issuer_bounded_validity_days(now, ca_cert.not_after);
194            return Err(AxiamError::Validation {
195                message: format!(
196                    "validity_days is {} but the issuing CA expires on {} — a certificate \
197                     cannot outlive its issuer. The most this CA can grant today is {} day{}.",
198                    validity_days,
199                    ca_cert.not_after.format("%Y-%m-%d"),
200                    available,
201                    if available == 1 { "" } else { "s" },
202                ),
203            });
204        }
205
206        Ok(LeafIssuance {
207            ca_cert,
208            store,
209            not_before,
210            not_after: requested_not_after,
211        })
212    }

214    /// Generate a new certificate signed by an organization CA.
215    ///
216    /// `org_id` is required to look up the CA certificate and its encrypted
217    /// private key for signing.
218    ///
219    /// `max_validity_days` is the tenant-level cap (from tenant metadata).
220    /// Pass `None` to use the default ([`DEFAULT_LEAF_CERT_VALIDITY_DAYS`]).
221    /// The hard cap ([`MAX_LEAF_CERT_VALIDITY_DAYS`], 825 days) is always
222    /// enforced per CA/Browser Forum Baseline Requirements.
223    pub async fn generate(
224        &self,
225        org_id: Uuid,
226        input: CreateCertificate,
227        max_validity_days: Option<u32>,
228    ) -> AxiamResult<GeneratedCertificate> {
229        let LeafIssuance {
230            ca_cert,
231            store,
232            not_before,
233            not_after,
234        } = self
235            .prepare_leaf_issuance(
236                org_id,
237                input.issuer_ca_id,
238                input.validity_days,
239                max_validity_days,
240            )
241            .await?;
242
243        // A custodian that signs on AXIAM's behalf never hands the key over, so
244        // there is nothing to load and nothing to sign with here: the leaf key
245        // is generated locally, its public half goes out in a CSR, and a
246        // certificate comes back. See [`axiam_core::ca_keys`].
247        if store.signs_remotely() {
248            return self
249                .generate_remotely(&ca_cert, input, not_before, not_after)
250                .await;
251        }
252
253        // Fetch the signing key from whichever custodian this CA's row named.
254        let key_ref = CaService::<CA>::key_ref(&ca_cert);
255        let mut ca_private_key_pem = store
256            .load(&key_ref, ca_cert.encrypted_private_key.as_deref())
257            .await?
258            .to_string();
259
260        // CPU-bound: key generation + certificate signing run in spawn_blocking behind semaphore (CQ-B02).
261        let _permit = self
262            .crypto_semaphore
263            .acquire()
264            .await
265            .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
266
267        let ca_cert_pem = ca_cert.public_cert_pem.clone();
268        let ee_subject = input.subject.clone();
269        let key_algorithm = input.key_algorithm.clone();
270        let not_before_ts = not_before.timestamp();
271        let not_after_ts = not_after.timestamp();
272
273        let (private_key_pem, public_cert_pem, fingerprint) =
274            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String, String)> {
275                let ca_key_pair = KeyPair::from_pem(&ca_private_key_pem)
276                    .map_err(|e| AxiamError::Certificate(format!("invalid CA private key: {e}")))?;
277                // Scrub the decrypted CA private-key PEM from memory as soon as
278                // the KeyPair is parsed — it is not needed past this point and
279                // must not linger in the heap buffer (defense-in-depth).
280                ca_private_key_pem.zeroize();
281
282                // Reconstruct the signing CA issuer from its real, stored certificate
283                // PEM — NOT from the (mutable) `subject` field — so the issuer DN
284                // embedded in every leaf cert can never drift from the CA's actual
285                // Subject DN (QUAL-05/D-08, T-29-11). rcgen 0.14 moved
286                // `from_ca_cert_pem` onto `Issuer`, which now owns the signing key and
287                // is passed directly to `signed_by`.
288                let ca_issuer =
289                    Issuer::from_ca_cert_pem(&ca_cert_pem, ca_key_pair).map_err(|e| {
290                        AxiamError::Certificate(format!("invalid CA certificate PEM: {e}"))
291                    })?;
292
293                // Generate end-entity key pair.
294                let ee_key_pair = generate_keypair(&key_algorithm)?;
295                let private_key_pem = ee_key_pair.serialize_pem();
296
297                // Build end-entity certificate request — from the same
298                // function `sign_csr` overwrites a caller's parameters with, so
299                // the two paths cannot issue different shapes.
300                let ee_params = leaf_params(&ee_subject, not_before_ts, not_after_ts)?;
301
302                let cert = ee_params.signed_by(&ee_key_pair, &ca_issuer).map_err(|e| {
303                    AxiamError::Certificate(format!("certificate signing failed: {e}"))
304                })?;
305
306                let public_cert_pem = cert.pem();
307                let fingerprint = compute_fingerprint(cert.der());
308                Ok((private_key_pem, public_cert_pem, fingerprint))
309            })
310            .await
311            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
312
313        let store = StoreCertificate {
314            tenant_id: input.tenant_id,
315            issuer_ca_id: input.issuer_ca_id,
316            subject: input.subject,
317            public_cert_pem,
318            fingerprint,
319            cert_type: input.cert_type,
320            key_algorithm: input.key_algorithm,
321            not_before,
322            not_after,
323            metadata: input.metadata.unwrap_or(serde_json::json!({})),
324        };
325
326        let certificate = self.cert_repo.create(store).await?;
327
328        Ok(GeneratedCertificate {
329            certificate,
330            private_key_pem,
331            // Signed in-process by a CA whose certificate the caller can fetch;
332            // there is no chain here they cannot already assemble.
333            chain_pem: None,
334        })
335    }
336
337    /// Issue against a CA whose key lives in — and stays in — its custodian.
338    ///
339    /// The end-entity key is still generated here and still returned once: what
340    /// moves to the custodian is the *signature*, not the subscriber's key. The
341    /// CSR carries the public half and nothing else.
342    async fn generate_remotely(
343        &self,
344        ca_cert: &axiam_core::models::certificate::CaCertificate,
345        input: CreateCertificate,
346        not_before: DateTime<Utc>,
347        not_after: DateTime<Utc>,
348    ) -> AxiamResult<GeneratedCertificate> {
349        // Only the keygen and CSR are CPU-bound. The permit is dropped before
350        // the call to the custodian: holding it across a network round trip
351        // would let one slow signer block every other issuance in the
352        // deployment, which is the opposite of what the semaphore is for.
353        let (private_key_pem, csr_pem) = {
354            let _permit = self
355                .crypto_semaphore
356                .acquire()
357                .await
358                .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
359
360            let ee_subject = input.subject.clone();
361            let key_algorithm = input.key_algorithm.clone();
362            tokio::task::spawn_blocking(move || -> AxiamResult<(String, String)> {
363                let ee_key_pair = generate_keypair(&key_algorithm)?;
364                let private_key_pem = ee_key_pair.serialize_pem();
365
366                let mut params = CertificateParams::new(Vec::<String>::new())
367                    .map_err(|e| AxiamError::Certificate(e.to_string()))?;
368                params
369                    .distinguished_name
370                    .push(DnType::CommonName, &ee_subject);
371                params.is_ca = IsCa::NoCa;
372                // No validity window: a PKCS#10 request cannot carry one. The
373                // signer is told the lifetime out of band, as a TTL.
374                let csr = params.serialize_request(&ee_key_pair).map_err(|e| {
375                    AxiamError::Certificate(format!("certificate request failed: {e}"))
376                })?;
377                let csr_pem = csr
378                    .pem()
379                    .map_err(|e| AxiamError::Certificate(format!("CSR encoding failed: {e}")))?;
380                Ok((private_key_pem, csr_pem))
381            })
382            .await
383            .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??
384        };
385
386        let key_ref = CaService::<CA>::key_ref(ca_cert);
387        let store = self.custodians.store_for(ca_cert.key_custody)?;
388        let signed = store
389            .sign_csr(
390                &key_ref,
391                &LeafSigningRequest {
392                    csr_pem,
393                    // AXIAM built this request, three lines up, from a key it
394                    // just generated: there is nothing in it a caller chose.
395                    csr_is_caller_supplied: false,
396                    ttl_seconds: (not_after - not_before).num_seconds(),
397                },
398            )
399            .await?;
400
401        // What the signer produced, not what was asked for. A remote signer
402        // caps a TTL to its mount's own maximum without failing the call, so a
403        // row built from the request would claim a window the certificate does
404        // not have — and the fingerprint has to be of the real bytes or every
405        // lookup by fingerprint misses.
406        let issued = parse_issued_leaf(&signed.certificate_pem)?;
407
408        let store_record = StoreCertificate {
409            tenant_id: input.tenant_id,
410            issuer_ca_id: input.issuer_ca_id,
411            subject: input.subject,
412            public_cert_pem: signed.certificate_pem,
413            fingerprint: issued.fingerprint,
414            cert_type: input.cert_type,
415            key_algorithm: input.key_algorithm,
416            not_before: issued.not_before,
417            not_after: issued.not_after,
418            metadata: input.metadata.unwrap_or(serde_json::json!({})),
419        };
420
421        let certificate = self.cert_repo.create(store_record).await?;
422
423        Ok(GeneratedCertificate {
424            certificate,
425            private_key_pem,
426            chain_pem: (!signed.chain_pem.is_empty()).then(|| join_pem(&signed.chain_pem)),
427        })
428    }
429
430    /// The minimum RSA modulus AXIAM will sign a certificate over, in bits.
431    ///
432    /// Stated here rather than inferred from [`KeyAlgorithm::Rsa4096`]'s name,
433    /// because that variant is a label and not a measurement:
434    /// `ca::parse_ca_certificate` maps *any* RSA OID onto it, deliberately, so
435    /// that an imported root of some other size is still a usable root. A path
436    /// that enforces a floor cannot reuse that mapping, and an RSA-2048 CSR
437    /// signed and then recorded as `Rsa4096` is the exact shape of T-96.
438    const MIN_CSR_RSA_MODULUS_BITS: usize = 4096;
439
440    /// Sign a caller's certificate signing request, issuing an end-entity
441    /// certificate for a key AXIAM never sees.
442    ///
443    /// The BYOK counterpart to [`Self::generate`]. The subscriber's private key
444    /// is generated by whoever made the CSR and never crosses the wire in
445    /// either direction — which is why the answer is a [`Certificate`] and not
446    /// a `GeneratedCertificate`: there is no key to return.
447    ///
448    /// # What the caller decides, and what AXIAM decides
449    ///
450    /// The caller decides the key and the subject. AXIAM decides everything
451    /// else, and the difference is enforced rather than documented:
452    ///
453    /// - **Possession** is proved by the request's own self-signature, checked
454    ///   by [`ca::inspect_csr`] before anything else happens. Without it a
455    ///   caller could have a certificate minted over somebody else's public key.
456    /// - **The key** must be Ed25519, or RSA with a modulus of at least
457    ///   [`Self::MIN_CSR_RSA_MODULUS_BITS`] bits. The modulus is measured, not
458    ///   taken from a label.
459    /// - **The extensions** are AXIAM's. A CSR requesting `subjectAltName`,
460    ///   `keyUsage` or `extendedKeyUsage` is refused by name rather than
461    ///   silently stripped; every other requested extension is discarded when
462    ///   the parameter set is overwritten with the leaf parameters
463    ///   [`Self::generate`] builds. A CSR asking to be an unconstrained CA
464    ///   comes back a leaf.
465    /// - **The validity, the issuer and the tenant** come from
466    ///   [`Self::prepare_leaf_issuance`], the same function `generate` uses, so
467    ///   the two paths cannot drift on a revoked CA, an expired one, a CA
468    ///   belonging to another organization, or a certificate that would outlive
469    ///   its issuer.
470    ///
471    /// # Why `keyUsage` is refused rather than dropped
472    ///
473    /// On the in-process path it could be dropped: `request.params` is replaced
474    /// wholesale before signing and nothing the caller asked for survives. Under
475    /// [`CaKeyCustody::VaultPki`] it could not. Vault's `sign-verbatim`
476    /// discards the `key_usage` and `ext_key_usage` request parameters when the
477    /// CSR carries those extensions, and issues what the CSR asked for — so a
478    /// silent strip would be a promise AXIAM keeps on one custodian and breaks
479    /// on the other, for the same CSR. One rule that holds everywhere is worth
480    /// more than a convenience that holds in one deployment.
481    ///
482    /// `basicConstraints` needs no such rule: the in-process path overwrites it
483    /// and Vault ignores a CSR's basic constraints outright.
484    pub async fn sign_csr(
485        &self,
486        org_id: Uuid,
487        input: SignCertificateCsr,
488        max_validity_days: Option<u32>,
489    ) -> AxiamResult<Certificate> {
490        // Parsed before the CA is even looked up, and outside the blocking
491        // task: a malformed or unsigned request is by far the likeliest failure
492        // on an endpoint whose input is pasted by hand, and it must be a 400
493        // naming the CSR rather than a 500 out of a custodian.
494        let facts = ca::inspect_csr(&input.csr_pem)?;
495
496        if let Some(bits) = facts.rsa_modulus_bits
497            && bits < Self::MIN_CSR_RSA_MODULUS_BITS
498        {
499            return Err(AxiamError::Validation {
500                message: format!(
501                    "the certificate signing request carries an RSA key with a {bits}-bit \
502                     modulus; AXIAM signs Ed25519 keys and RSA keys of at least {} bits",
503                    Self::MIN_CSR_RSA_MODULUS_BITS
504                ),
505            });
506        }
507
508        if !facts.refused_extensions.is_empty() {
509            return Err(AxiamError::Validation {
510                message: format!(
511                    "the certificate signing request asks for {}, which AXIAM does not sign \
512                     over on an end-entity certificate. It is refused rather than dropped \
513                     because a certificate issued without what you asked for, and with \
514                     nothing said, would fail where you deploy it. Remove the extension \
515                     request and submit the CSR again.",
516                    facts.refused_extensions.join(" and ")
517                ),
518            });
519        }
520
521        let LeafIssuance {
522            ca_cert,
523            store,
524            not_before,
525            not_after,
526        } = self
527            .prepare_leaf_issuance(
528                org_id,
529                input.issuer_ca_id,
530                input.validity_days,
531                max_validity_days,
532            )
533            .await?;
534
535        let (public_cert_pem, fingerprint, chain_pem, issued_window) = if store.signs_remotely() {
536            let key_ref = CaService::<CA>::key_ref(&ca_cert);
537            let signed = store
538                .sign_csr(
539                    &key_ref,
540                    &LeafSigningRequest {
541                        csr_pem: input.csr_pem.clone(),
542                        ttl_seconds: (not_after - not_before).num_seconds(),
543                        // These bytes are the caller's. See the custodian's own
544                        // doc comment for what it does about that.
545                        csr_is_caller_supplied: true,
546                    },
547                )
548                .await?;
549            // What the signer produced, not what was asked for: a remote signer
550            // caps a TTL to its mount's own maximum without failing the call.
551            let issued = parse_issued_leaf(&signed.certificate_pem)?;
552            let chain = (!signed.chain_pem.is_empty()).then(|| join_pem(&signed.chain_pem));
553            (
554                signed.certificate_pem,
555                issued.fingerprint,
556                chain,
557                Some((issued.not_before, issued.not_after)),
558            )
559        } else {
560            let key_ref = CaService::<CA>::key_ref(&ca_cert);
561            let mut ca_private_key_pem = store
562                .load(&key_ref, ca_cert.encrypted_private_key.as_deref())
563                .await?
564                .to_string();
565
566            let _permit = self
567                .crypto_semaphore
568                .acquire()
569                .await
570                .map_err(|_| AxiamError::Internal("crypto semaphore closed".into()))?;
571
572            let ca_cert_pem = ca_cert.public_cert_pem.clone();
573            let csr_pem = input.csr_pem.clone();
574            let subject = facts.common_name.clone();
575            let not_before_ts = not_before.timestamp();
576            let not_after_ts = not_after.timestamp();
577
578            let (pem, fingerprint) =
579                tokio::task::spawn_blocking(move || -> AxiamResult<(String, String)> {
580                    let ca_key_pair = KeyPair::from_pem(&ca_private_key_pem).map_err(|e| {
581                        AxiamError::Certificate(format!("invalid CA private key: {e}"))
582                    })?;
583                    ca_private_key_pem.zeroize();
584
585                    let ca_issuer =
586                        Issuer::from_ca_cert_pem(&ca_cert_pem, ca_key_pair).map_err(|e| {
587                            AxiamError::Certificate(format!("invalid CA certificate PEM: {e}"))
588                        })?;
589
590                    // Parsed a second time, inside the task, because rcgen's
591                    // signing API needs its own representation. The answer that
592                    // matters — whether the signature verifies — was already
593                    // established above, on the same bytes; this is not a
594                    // second opinion about it.
595                    let mut request = rcgen::CertificateSigningRequestParams::from_pem(&csr_pem)
596                        .map_err(|e| AxiamError::Validation {
597                            message: format!("the certificate signing request was rejected: {e}"),
598                        })?;
599                    // Everything the request asked to be, overwritten by what
600                    // AXIAM has decided it is — the same leaf parameters
601                    // `generate` builds, so the two paths issue the same shape
602                    // of certificate for the same request.
603                    request.params = leaf_params(&subject, not_before_ts, not_after_ts)?;
604                    // Deliberately **not** `use_authority_key_identifier_extension`,
605                    // which the intermediate CSR path sets. Setting it here put
606                    // an authorityKeyIdentifier on a CSR-signed leaf that a
607                    // generated leaf does not carry, which is exactly the
608                    // divergence D-5 forbids: two ways of asking for the same
609                    // certificate producing two different certificates. An AKI
610                    // on leaves is worth having and is a change to make on both
611                    // paths at once, with the KU/EKU profiles, not a side
612                    // effect of how the key was made.
613
614                    let cert = request.signed_by(&ca_issuer).map_err(|e| {
615                        AxiamError::Certificate(format!("certificate signing failed: {e}"))
616                    })?;
617                    let pem = cert.pem();
618                    let fingerprint = compute_fingerprint(cert.der());
619                    Ok((pem, fingerprint))
620                })
621                .await
622                .map_err(|e| AxiamError::Internal(format!("spawn_blocking join error: {e}")))??;
623
624            // Signed in-process by a CA whose certificate the caller can fetch;
625            // there is no chain here they cannot already assemble.
626            (pem, fingerprint, None, None)
627        };
628
629        let (row_not_before, row_not_after) = issued_window.unwrap_or((not_before, not_after));
630
631        let certificate = self
632            .cert_repo
633            .create(StoreCertificate {
634                tenant_id: input.tenant_id,
635                issuer_ca_id: input.issuer_ca_id,
636                // What the certificate says, never what the caller said
637                // separately — there is no separate field for them to say it in
638                // (T-194: the row and the certificate cannot disagree).
639                subject: facts.common_name,
640                public_cert_pem,
641                fingerprint,
642                cert_type: input.cert_type,
643                // Read off the CSR's own public key, not asserted by the caller.
644                key_algorithm: facts.key_algorithm,
645                not_before: row_not_before,
646                not_after: row_not_after,
647                metadata: input.metadata.unwrap_or(serde_json::json!({})),
648            })
649            .await?;
650
651        // `chain_pem` has nowhere to go on a `Certificate` and is not dropped
652        // silently: under the custodians that return one, the chain is the
653        // issuing CA's own certificate, which `GET /certificates/{id}` already
654        // resolves through `issuer_ca_id`. A caller holding the certificate can
655        // assemble the chain from data the API already gives them.
656        let _ = chain_pem;
657
658        Ok(certificate)
659    }

688/// What a certificate a remote signer returned actually says about itself.
689struct IssuedLeaf {
690    fingerprint: String,
691    not_before: DateTime<Utc>,
692    not_after: DateTime<Utc>,
693}
694
695/// Read back a signed leaf, so the row describes the certificate rather than
696/// the request that produced it.
697/// The parameters every AXIAM leaf certificate is built from: the common name,
698/// the validity window, and `CA:FALSE`.
699///
700/// No subjectAltName, no key usage, no extended key usage — deliberately, and
701/// identically on both leaf paths. Adding a usage profile to one of them only
702/// would be a worse outcome than the status quo: two ways of asking for the
703/// same certificate would produce two different certificates. Giving *both*
704/// paths a per-`cert_type` profile is a follow-up to be decided once, for both,
705/// with a migration note for everything already issued without one.
706///
707/// Shared by [`CertService::generate`], which builds a key to go with it, and
708/// by [`CertService::sign_csr`], which overwrites a caller's requested
709/// parameters with it. That sharing is the enforcement: "a CSR-signed leaf is
710/// the same shape as a generated one" is true because there is one function
711/// that says what the shape is.
712fn leaf_params(
713    subject: &str,
714    not_before_ts: i64,
715    not_after_ts: i64,
716) -> AxiamResult<CertificateParams> {
717    let mut params = CertificateParams::new(Vec::<String>::new())
718        .map_err(|e| AxiamError::Certificate(e.to_string()))?;
719    params.distinguished_name.push(DnType::CommonName, subject);
720    params.is_ca = IsCa::NoCa;
721    params.not_before = time::OffsetDateTime::from_unix_timestamp(not_before_ts)
722        .map_err(|e| AxiamError::Certificate(format!("invalid notBefore: {e}")))?;
723    params.not_after = time::OffsetDateTime::from_unix_timestamp(not_after_ts)
724        .map_err(|e| AxiamError::Certificate(format!("invalid notAfter: {e}")))?;
725    Ok(params)
726}
727
728fn parse_issued_leaf(pem: &str) -> AxiamResult<IssuedLeaf> {
729    let (_, block) = x509_parser::pem::parse_x509_pem(pem.as_bytes()).map_err(|e| {
730        AxiamError::Certificate(format!(
731            "the signer returned something that is not a PEM certificate: {e}"
732        ))
733    })?;
734    let (_, cert) = X509Certificate::from_der(&block.contents).map_err(|e| {
735        AxiamError::Certificate(format!("the signed certificate could not be parsed: {e}"))
736    })?;
737
738    let to_utc = |t: i64, what: &str| -> AxiamResult<DateTime<Utc>> {
739        DateTime::from_timestamp(t, 0).ok_or_else(|| {
740            AxiamError::Certificate(format!("the signed certificate's {what} is out of range"))
741        })
742    };
743
744    Ok(IssuedLeaf {
745        fingerprint: compute_fingerprint(&block.contents),
746        not_before: to_utc(cert.validity().not_before.timestamp(), "notBefore")?,
747        not_after: to_utc(cert.validity().not_after.timestamp(), "notAfter")?,
748    })
749}
