1//! PKI configuration types.
2
3/// PKI configuration — holds the AES-256-GCM key for encrypting CA private keys,
4/// plus the FIDO MDS3 ingestion settings (D10, X3).
5///
6/// `encryption_key` is `Option` — absent means the operator has not configured
7/// the key yet. Operations that encrypt private key material (CA generation,
8/// PGP AuditSigning key generation) will return an error rather than silently
9/// encrypting with a known-zero key (SEC-012).
10///
11/// MDS ingestion is opt-in and makes **no** outbound network calls unless
12/// `mds_enabled` is explicitly set (D10): the master switch defaults to
13/// `false`, so upgrading to a build that includes this feature changes no
14/// deployment's runtime behavior until an operator turns it on.
15#[derive(Clone)]
16pub struct PkiConfig {
17    pub encryption_key: Option<[u8; 32]>,
18
19    // -----------------------------------------------------------------
20    // FIDO MDS3 ingestion (D10)
21    // -----------------------------------------------------------------
22    /// Master switch. `false` (the default) means zero outbound calls —
23    /// no background refresh job is spawned and the admin-triggered refresh
24    /// endpoint refuses to run. Env: `AXIAM__PKI__MDS_ENABLED`.
25    pub mds_enabled: bool,
26    /// FIDO MDS3 BLOB fetch source. Env: `AXIAM__PKI__MDS_BLOB_URL`.
27    pub mds_blob_url: String,
28    /// Local BLOB file path for air-gapped deployments. When set, it wins
29    /// over `mds_blob_url` and no network fetch happens at all.
30    /// Env: `AXIAM__PKI__MDS_BLOB_PATH`.
31    pub mds_blob_path: Option<std::path::PathBuf>,
32    /// Background refresh interval, in seconds. `0` disables the background
33    /// job (the admin-triggered endpoint still works). Default: 604800
34    /// (weekly). Env: `AXIAM__PKI__MDS_REFRESH_INTERVAL_SECS`.
35    pub mds_refresh_interval_secs: u64,
36    /// Expected DNS SAN (or CN fallback) on the MDS BLOB's leaf signing
37    /// certificate (D4 step 5) — configurable so a legitimate FIDO hostname
38    /// change is an ops action, not a code release.
39    /// Env: `AXIAM__PKI__MDS_LEAF_DNS`.
40    pub mds_leaf_dns: String,
41    /// T-153: refuse attested registration once the ingested MDS BLOB is more
42    /// than this many days past its `nextUpdate`. `0` (the default) disables
43    /// the check and keeps the fail-open behaviour described below.
44    /// Env: `AXIAM__PKI__MDS_MAX_STALE_DAYS`.
45    ///
46    /// Staleness deliberately never hard-fails *ingestion*: a transient outage
47    /// at the FIDO Alliance must not brick WebAuthn registration for everyone.
48    /// The cost of that choice is T-153 — an authenticator model revoked since
49    /// the last successful refresh keeps passing attestation policy, because
50    /// the revocation is in a BLOB this deployment has not seen.
51    ///
52    /// This knob lets an operator put a bound on that window, and it is opt-in
53    /// rather than defaulted to a value because the right bound is a property
54    /// of the deployment: a high-assurance tenant may want days, an air-gapped
55    /// one (`mds_blob_path`, no automatic refresh path at all) would be taken
56    /// offline by anything short of months.
57    ///
58    /// Scope is attestation only. An unattested ceremony under
59    /// `AttestationMode::None` consults no metadata, so stale metadata cannot
60    /// have misled it and refusing it would deny a registration for a reason
61    /// that does not apply to it.
62    pub mds_max_stale_days: u32,
63}
64
65/// Default FIDO MDS3 BLOB fetch source (D10).
66pub const DEFAULT_MDS_BLOB_URL: &str = "https://mds3.fidoalliance.org/";
67
68/// Default background MDS refresh interval: weekly (D10).
69pub const DEFAULT_MDS_REFRESH_INTERVAL_SECS: u64 = 604_800;
70
71/// Default expected leaf DNS name on the MDS BLOB signing certificate (D4/D10).
72pub const DEFAULT_MDS_LEAF_DNS: &str = "mds.fidoalliance.org";
73
74impl Default for PkiConfig {
75    /// Matches today's behavior byte-for-byte: no encryption key configured,
76    /// and MDS ingestion fully disabled (no outbound calls).
77    fn default() -> Self {
78        Self {
79            encryption_key: None,
80            mds_enabled: false,
81            mds_blob_url: DEFAULT_MDS_BLOB_URL.to_string(),
82            mds_blob_path: None,
83            mds_refresh_interval_secs: DEFAULT_MDS_REFRESH_INTERVAL_SECS,
84            mds_leaf_dns: DEFAULT_MDS_LEAF_DNS.to_string(),
85            mds_max_stale_days: 0,
86        }
87    }
88}
