1//! PKI configuration types.
2
3/// PKI configuration — holds the AES-256-GCM key for encrypting CA private keys,
4/// plus the FIDO MDS3 ingestion settings (D10, X3).
5///
6/// `encryption_key` is `Option` — absent means the operator has not configured
7/// the key yet. Operations that encrypt private key material (CA generation,
8/// PGP AuditSigning key generation) will return an error rather than silently
9/// encrypting with a known-zero key (SEC-012).
10///
11/// MDS ingestion is opt-in and makes **no** outbound network calls unless
12/// `mds_enabled` is explicitly set (D10): the master switch defaults to
13/// `false`, so upgrading to a build that includes this feature changes no
14/// deployment's runtime behavior until an operator turns it on.
15#[derive(Clone)]
16pub struct PkiConfig {
17    pub encryption_key: Option<[u8; 32]>,
18
19    // -----------------------------------------------------------------
20    // FIDO MDS3 ingestion (D10)
21    // -----------------------------------------------------------------
22    /// Master switch. `false` (the default) means zero outbound calls —
23    /// no background refresh job is spawned and the admin-triggered refresh
24    /// endpoint refuses to run. Env: `AXIAM__PKI__MDS_ENABLED`.
25    pub mds_enabled: bool,
26    /// FIDO MDS3 BLOB fetch source. Env: `AXIAM__PKI__MDS_BLOB_URL`.
27    pub mds_blob_url: String,
28    /// Local BLOB file path for air-gapped deployments. When set, it wins
29    /// over `mds_blob_url` and no network fetch happens at all.
30    /// Env: `AXIAM__PKI__MDS_BLOB_PATH`.
31    pub mds_blob_path: Option<std::path::PathBuf>,
32    /// Background refresh interval, in seconds. `0` disables the background
33    /// job (the admin-triggered endpoint still works). Default: 604800
34    /// (weekly). Env: `AXIAM__PKI__MDS_REFRESH_INTERVAL_SECS`.
35    pub mds_refresh_interval_secs: u64,
36    /// Expected DNS SAN (or CN fallback) on the MDS BLOB's leaf signing
37    /// certificate (D4 step 5) — configurable so a legitimate FIDO hostname
38    /// change is an ops action, not a code release.
39    /// Env: `AXIAM__PKI__MDS_LEAF_DNS`.
40    pub mds_leaf_dns: String,
41}
42
43/// Default FIDO MDS3 BLOB fetch source (D10).
44pub const DEFAULT_MDS_BLOB_URL: &str = "https://mds3.fidoalliance.org/";
45
46/// Default background MDS refresh interval: weekly (D10).
47pub const DEFAULT_MDS_REFRESH_INTERVAL_SECS: u64 = 604_800;
48
49/// Default expected leaf DNS name on the MDS BLOB signing certificate (D4/D10).
50pub const DEFAULT_MDS_LEAF_DNS: &str = "mds.fidoalliance.org";
51
52impl Default for PkiConfig {
53    /// Matches today's behavior byte-for-byte: no encryption key configured,
54    /// and MDS ingestion fully disabled (no outbound calls).
55    fn default() -> Self {
56        Self {
57            encryption_key: None,
58            mds_enabled: false,
59            mds_blob_url: DEFAULT_MDS_BLOB_URL.to_string(),
60            mds_blob_path: None,
61            mds_refresh_interval_secs: DEFAULT_MDS_REFRESH_INTERVAL_SECS,
62            mds_leaf_dns: DEFAULT_MDS_LEAF_DNS.to_string(),
63        }
64    }
65}
