1//! Shared cryptographic helpers for CA/certificate key generation, fingerprinting,
2//! and AES-256-GCM secret encryption — consolidated from the byte-identical
3//! implementations previously triplicated across `ca.rs`, `cert.rs`, and `pgp.rs`
4//! (QUAL-05/D-08).
5//!
6//! `pgp.rs` intentionally does NOT use [`generate_keypair`] here: its keypair
7//! generation produces a distinct `PgpKeyAlgorithm` + `user_id` -> `SignedSecretKey`
8//! type that is not X.509/rcgen-based and must not be merged into this module.
9

20/// Modulus size, in bits, of the RSA keys AXIAM issues.
21///
22/// Named rather than inlined because it also documents the lower bound *ring*
23/// will accept when it later loads the key back for signing: ring refuses a
24/// PKCS#8 RSA key below 2048 bits outright, so a future "RSA-2048 for
25/// constrained devices" option is a change to this constant plus a new
26/// [`KeyAlgorithm`] variant, not a change to the mechanism below.
27const RSA_MODULUS_BITS: usize = 4096;
28
29/// Generate an X.509 key pair for the given algorithm.
30///
31/// # Why RSA takes the long way round
32///
33/// rcgen delegates its crypto to *ring*, and ring deliberately implements no
34/// RSA key **generation** — only signing and verification. `KeyPair::generate_for`
35/// therefore answered every RSA-4096 request with
36/// `"There is no support for generating keys for the given algorithm"`, which
37/// reached the operator as a 500 on an algorithm the API advertises. Ring *can*
38/// sign with an RSA key it is handed (see rcgen's `PKCS_RSA_SHA256` arm of
39/// `from_pkcs8_der_and_sign_algo`), so the missing half is generation alone.
40///
41/// The `rsa` crate supplies exactly that half: generate the key, serialize it as
42/// PKCS#8 DER, and hand it to rcgen with the signature algorithm named
43/// explicitly. The resulting [`KeyPair`] is indistinguishable from one rcgen
44/// generated itself — same type, same `serialize_pem`, same signing path — so no
45/// caller needs to know which algorithm it asked for.
46///
47/// Switching rcgen to its `aws_lc_rs` backend would also work and is the more
48/// usual answer, but it swaps the crypto provider under *every* algorithm and
49/// pulls a C toolchain (cmake, and NASM on Windows) into the build. This keeps
50/// the provider, the build, and the Ed25519 path exactly as they were.
51///
52/// # Cost
53///
54/// RSA-4096 generation is seconds of CPU, not milliseconds, and it is a
55/// rejection-sampling loop whose duration varies run to run. Every call site in
56/// this crate already runs inside `tokio::task::spawn_blocking`, which is what
57/// keeps that off the async runtime's worker threads; a new call site must do
58/// the same.
59///
60/// # Examples
61///
62/// ```ignore
63/// // Inside a blocking context — never directly on an async worker.
64/// let key_pair = tokio::task::spawn_blocking(move || {
65///     generate_keypair(&KeyAlgorithm::Rsa4096)
66/// })
67/// .await??;
68/// assert!(key_pair.serialize_pem().contains("PRIVATE KEY"));
69/// ```
70pub(crate) fn generate_keypair(algorithm: &KeyAlgorithm) -> AxiamResult<KeyPair> {
71    match algorithm {
72        KeyAlgorithm::Ed25519 => KeyPair::generate_for(&rcgen::PKCS_ED25519)
73            .map_err(|e| AxiamError::Certificate(format!("Ed25519 keygen failed: {e}"))),
74        KeyAlgorithm::Rsa4096 => generate_rsa_keypair(),
75    }
76}

78/// Generate an RSA-4096 key pair and load it into rcgen for signing.
79fn generate_rsa_keypair() -> AxiamResult<KeyPair> {
80    let mut rng = rand_core::OsRng;
81    let private_key = RsaPrivateKey::new(&mut rng, RSA_MODULUS_BITS)
82        .map_err(|e| AxiamError::Certificate(format!("RSA-4096 keygen failed: {e}")))?;
83
84    // PKCS#8 PEM rather than DER purely to keep the hand-off inside rcgen's own
85    // public API: `from_pkcs8_der_and_sign_algo` takes a `PrivatePkcs8KeyDer`,
86    // a rustls-pki-types type rcgen does not re-export, so reaching it would
87    // mean depending on rustls-pki-types directly and pinning its version
88    // against rcgen's. `pkcs8::LineEnding` is already in the tree via `rsa`.
89    //
90    // `to_pkcs8_pem` returns a `Zeroizing<String>`, so the intermediate copy of
91    // the private key is wiped when this scope ends. rcgen keeps its own copy
92    // inside the returned `KeyPair` for as long as it needs to sign.
93    let pkcs8_pem = private_key
94        .to_pkcs8_pem(rsa::pkcs8::LineEnding::LF)
95        .map_err(|e| {
96            AxiamError::Certificate(format!("RSA-4096 key could not be encoded as PKCS#8: {e}"))
97        })?;
98
99    KeyPair::from_pkcs8_pem_and_sign_algo(&pkcs8_pem, &PKCS_RSA_SHA256).map_err(|e| {
100        AxiamError::Certificate(format!("RSA-4096 key was rejected by the signer: {e}"))
101    })
102}

104/// Compute SHA-256 fingerprint from DER-encoded certificate bytes.
105pub(crate) fn compute_fingerprint(der: &[u8]) -> String {
106    let hash = Sha256::digest(der);
107    hex::encode(hash)
108}

110/// Encrypt data with AES-256-GCM. The 12-byte nonce is prepended to the
111/// ciphertext so the caller doesn't need to store it separately.
112pub(crate) fn encrypt_secret(plaintext: &[u8], key_bytes: &[u8; 32]) -> AxiamResult<Vec<u8>> {
113    let key = Key::<Aes256Gcm>::from(*key_bytes);
114    let cipher = Aes256Gcm::new(&key);
115    let nonce_bytes: [u8; 12] = Generate::generate();
116    let nonce = Nonce::<U12>::from(nonce_bytes);
117    let ciphertext = cipher
118        .encrypt(&nonce, plaintext)
119        .map_err(|e| AxiamError::Crypto(format!("AES-256-GCM encryption failed: {e}")))?;
120
121    let mut result = Vec::with_capacity(nonce.len() + ciphertext.len());
122    result.extend_from_slice(&nonce);
123    result.extend_from_slice(&ciphertext);
124    Ok(result)
125}
126
127/// Decrypt AES-256-GCM encrypted data (12-byte nonce prepended to ciphertext).
128pub(crate) fn decrypt_secret(data: &[u8], key_bytes: &[u8; 32]) -> AxiamResult<Vec<u8>> {
129    if data.len() < 12 {
130        return Err(AxiamError::Crypto(
131            "encrypted data too short (missing nonce)".into(),
132        ));
133    }
134    let (nonce_bytes, ciphertext) = data.split_at(12);
135    let key = Key::<Aes256Gcm>::from(*key_bytes);
136    let cipher = Aes256Gcm::new(&key);
137    let nonce = Nonce::<U12>::try_from(nonce_bytes)
138        .map_err(|_| AxiamError::Crypto("invalid nonce length".into()))?;
139    cipher
140        .decrypt(&nonce, ciphertext)
141        .map_err(|e| AxiamError::Crypto(format!("AES-256-GCM decryption failed: {e}")))
142}

148    #[test]
149    fn generate_keypair_ed25519_produces_usable_pem() {
150        let kp = generate_keypair(&KeyAlgorithm::Ed25519).expect("ed25519 keygen must succeed");
151        let pem = kp.serialize_pem();
152        assert!(pem.contains("PRIVATE KEY"));
153    }

155    #[test]
156    fn generate_keypair_rsa4096_produces_a_usable_signing_key() {
157        // The regression this pins: rcgen's `ring` backend cannot *generate*
158        // RSA keys, so this arm used to fail with "There is no support for
159        // generating keys for the given algorithm" and surface as a 500 on an
160        // algorithm the certificate API advertises. Generating with the `rsa`
161        // crate and handing ring the PKCS#8 gets a key ring will sign with.
162        let kp = generate_keypair(&KeyAlgorithm::Rsa4096).expect("RSA-4096 keygen must succeed");
163        let pem = kp.serialize_pem();
164        assert!(
165            pem.contains("PRIVATE KEY"),
166            "expected a PKCS#8 PEM, got: {pem}"
167        );
168        assert!(
169            KeyPair::from_pem(&pem).is_ok(),
170            "the serialized key must round-trip back through rcgen"
171        );
172    }

174    #[test]
175    fn generate_keypair_rsa4096_signs_a_self_signed_certificate() {
176        // Loading the key is not the same fact as ring accepting it for
177        // signature generation: `from_pkcs8_der_and_sign_algo` only parses.
178        // Actually signing something is what proves the whole path works.
179        let kp = generate_keypair(&KeyAlgorithm::Rsa4096).expect("RSA-4096 keygen must succeed");
180        let mut params =
181            rcgen::CertificateParams::new(Vec::<String>::new()).expect("empty SAN list is valid");
182        params
183            .distinguished_name
184            .push(rcgen::DnType::CommonName, "rsa-keygen-test");
185        let cert = params
186            .self_signed(&kp)
187            .expect("an RSA-4096 key must be able to self-sign");
188        assert!(cert.pem().contains("BEGIN CERTIFICATE"));
189    }

191    #[test]
192    fn generate_keypair_rsa4096_is_not_deterministic() {
193        // A generator that returned the same key twice would be catastrophic
194        // and is exactly the shape a stubbed-out implementation takes.
195        let a = generate_keypair(&KeyAlgorithm::Rsa4096)

263    fn decrypt_secret_rejects_tampered_ciphertext() {
264        let key = [5u8; 32];
265        let mut ciphertext = encrypt_secret(b"authentic data", &key).unwrap();
266        // Flip a bit in the ciphertext body (after the 12-byte nonce) to break the
267        // AES-GCM authentication tag.
268        let last = ciphertext.len() - 1;
269        ciphertext[last] ^= 0xFF;
270        let err = decrypt_secret(&ciphertext, &key).unwrap_err();
271        let msg = format!("{err:?}");
272        assert!(msg.contains("decryption failed"), "got: {msg}");
273    }
