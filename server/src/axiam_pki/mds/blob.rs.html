1//! FIDO MDS3 BLOB verification and parsing (D4).
2//!
3//! [`verify_and_parse`] implements D4 steps 1-9 in the pinned order. Every
4//! step fails closed: there is no code path that returns a parsed
5//! [`MdsBlob`] without every prior step having succeeded.

20/// Header `alg` values accepted for an MDS3 BLOB (D4 step 2). Exactly one —
21/// `none` and every other algorithm are rejected with a distinct error, and
22/// this is the *only* place `alg` is ever read from: it never influences
23/// parsing before this check, and the payload is never consulted for it.
24const ALLOWED_ALG: &str = "RS256";
25
26/// Maximum accepted number of certificates in the JWT header's `x5c` claim
27/// (D4 step 3) — bounds the chain-walk below against a pathologically long
28/// (attacker-controlled) header.
29const MAX_X5C_LEN: usize = 8;
30
31/// Assert that a certificate is allowed to *issue* other certificates.
32///
33/// This is not a nicety, it is the control that makes the vendored anchor
34/// mean anything. GlobalSign Root CA – R3 is a **public** CA root: it sits
35/// above the certificates of the entire public web, not just the FIDO
36/// Alliance's. Verifying signatures alone — "each `x5c[i]` verifies under
37/// `x5c[i+1]`, and the last one verifies under our root" — is therefore
38/// satisfied by a chain an attacker can assemble from an *ordinary*
39/// end-entity certificate:
40///
41/// ```text
42///   x5c = [ forged leaf (SAN mds.fidoalliance.org, signed by the attacker),
43///           attacker's genuine EE cert (genuinely issued under R3),
44///           genuine GlobalSign intermediate ]
45/// ```
46///
47/// Every signature in that chain verifies, and the forged leaf carries
48/// whatever SAN the attacker chose — so leaf-identity pinning does not stop
49/// it either. What stops it is refusing to treat a non-CA certificate as an
50/// issuer: an ordinary EE certificate carries `basicConstraints: CA=false`
51/// (or omits the extension), so it can never be `x5c[i+1]`.
52///
53/// Requires, for every issuing position in the chain:
54/// - `basicConstraints` present with `CA=true`; and
55/// - if `keyUsage` is present, `keyCertSign` asserted (absent `keyUsage` is
56///   permitted — it is optional in RFC 5280 and some older CA certificates
57///   omit it — but a present-and-contradictory one is fatal).
58fn assert_is_issuer(cert: &X509Certificate<'_>) -> Result<(), MdsError> {
59    let bc = cert
60        .basic_constraints()
61        .map_err(|e| MdsError::InvalidCertificate(e.to_string()))?;
62    match bc {
63        Some(ext) if ext.value.ca => {}
64        _ => return Err(MdsError::IssuerNotCa),
65    }
66
67    if let Some(ku) = cert
68        .key_usage()
69        .map_err(|e| MdsError::InvalidCertificate(e.to_string()))?
70        && !ku.value.key_cert_sign()
71    {
72        return Err(MdsError::IssuerNotCa);
73    }
74
75    Ok(())
76}
77
78/// Assert that the chain's path-length constraints are respected.
79///
80/// `pathLenConstraint` on a CA certificate bounds how many intermediates may
81/// appear *below* it. Ignoring it would let an attacker who controls a
82/// deliberately-constrained sub-CA extend the chain further than its issuer
83/// permitted. `depth_below` is the number of non-self-issued CA certificates
84/// between this certificate and the leaf.
85fn assert_path_len(cert: &X509Certificate<'_>, depth_below: usize) -> Result<(), MdsError> {
86    let bc = cert
87        .basic_constraints()
88        .map_err(|e| MdsError::InvalidCertificate(e.to_string()))?;
89    if let Some(ext) = bc
90        && let Some(max) = ext.value.path_len_constraint
91        && depth_below as u64 > max as u64
92    {
93        return Err(MdsError::PathLenExceeded);
94    }
95    Ok(())
96}

106// ---------------------------------------------------------------------------
107// Raw payload shape (D4 step 7) — mirrors the FIDO MDS3 JSON schema exactly,
108// including its camelCase field names and the split between entry-level and
109// metadataStatement-level fields. Converted into the flat, storage-shaped
110// `axiam_core::models::mds::MdsEntry` by `convert_entries` below.
111// ---------------------------------------------------------------------------
112
113#[derive(Debug, Deserialize)]
114struct RawMdsPayload {
115    no: i64,
116    #[serde(rename = "nextUpdate")]
117    next_update: String,
118    #[serde(default)]
119    entries: Vec<RawMdsEntry>,
120}
121
122#[derive(Debug, Deserialize)]
123struct RawMdsEntry {
124    /// Present for FIDO2/WebAuthn authenticators. Absent for legacy UAF/U2F
125    /// entries (keyed by `aaid`/`attestationCertificateKeyIdentifiers`
126    /// instead) — those are skipped by `convert_entries` (D1).
127    #[serde(default)]
128    aaguid: Option<String>,
129    #[serde(default, rename = "metadataStatement")]
130    metadata_statement: Option<RawMetadataStatement>,
131    #[serde(default, rename = "statusReports")]
132    status_reports: Vec<MdsStatusReport>,
133    #[serde(default, rename = "timeOfLastStatusChange")]
134    time_of_last_status_change: Option<String>,
135}
136
137#[derive(Debug, Deserialize)]
138struct RawMetadataStatement {
139    #[serde(default)]
140    description: Option<String>,
141    #[serde(default, rename = "attestationRootCertificates")]
142    attestation_root_certificates: Vec<String>,
143}
144
145/// Convert raw payload entries into storage-shaped [`MdsEntry`] values,
146/// skipping (and counting) entries with no `aaguid` — UAF/U2F entries, and
147/// defensively any entry whose `aaguid` fails to parse as a UUID (D1).
148fn convert_entries(raw: Vec<RawMdsEntry>) -> (Vec<MdsEntry>, usize) {
149    let mut entries = Vec::with_capacity(raw.len());
150    let mut skipped = 0usize;
151
152    for e in raw {
153        let aaguid = match e.aaguid.as_deref().map(Uuid::parse_str) {
154            Some(Ok(id)) => id,
155            _ => {
156                skipped += 1;
157                continue;
158            }
159        };
160
161        let (description, attestation_root_certificates) = match e.metadata_statement {
162            Some(ms) => (ms.description, ms.attestation_root_certificates),
163            None => (None, Vec::new()),
164        };
165
166        entries.push(MdsEntry {
167            aaguid,
168            description,
169            attestation_root_certificates,
170            status_reports: e.status_reports,
171            time_of_last_status_change: e.time_of_last_status_change,
172        });
173    }
174
175    (entries, skipped)
176}

182/// Verify a FIDO MDS3 BLOB's signature chain back to the vendored root
183/// anchor and parse its payload, in the exact order pinned by D4:
184///
185/// 1. Split into exactly 3 dot-separated segments; decode the header.
186/// 2. Header `alg` must be exactly `RS256` — reject `none` and anything else.
187/// 3. `x5c` must be present, non-empty, and ≤ 8 entries (standard base64 DER).
188/// 4. Build the chain: every cert's validity window covers `now`; each
189///    `x5c[i]` verifies under `x5c[i+1]`'s key; the last `x5c` cert verifies
190///    under the vendored root's key (or, if it *is* the vendored root by DER
191///    equality, the cert before it does — via the same per-`i` chain check).
192/// 5. The leaf (`x5c[0]`) must carry `expected_leaf_dns` as a SAN DNS entry
193///    (falling back to CN only when there is no SAN extension at all).
194/// 6. Verify the JWT signature over `header.payload` with the leaf's RSA
195///    public key.
196/// 7. Only now does the payload JSON get parsed.
197/// 8. Rollback protection is **not** performed here — see
198///    [`super::decide_ingest_outcome`] and its doc comment for why.
199/// 9. `nextUpdate < now` marks the result `stale`, logged at WARN; ingestion
200///    still succeeds — staleness never hard-fails, policy decides.
201///
202/// Fails closed at every step: no branch returns `Ok` without every prior
203/// check having passed.
204pub fn verify_and_parse(
205    blob: &str,
206    now: DateTime<Utc>,
207    expected_leaf_dns: &str,
208) -> Result<MdsBlob, MdsError> {
209    let root_der = load_root_anchor()?;
210    verify_and_parse_with_anchor(blob, now, expected_leaf_dns, &root_der)
211}

213/// [`verify_and_parse`] with the trust anchor supplied by the caller instead
214/// of loaded from the vendored PEM.
215///
216/// This is the single implementation — [`verify_and_parse`] is a thin wrapper
217/// that passes the vendored anchor. The parameterization exists so the test
218/// suite can drive **this exact code path** against a throwaway root, rather
219/// than against a copy of it: a test that re-implements the logic it is
220/// meant to guard silently stops guarding it the moment the two diverge, and
221/// this pipeline is precisely where a silent divergence is most expensive.
222pub fn verify_and_parse_with_anchor(
223    blob: &str,
224    now: DateTime<Utc>,
225    expected_leaf_dns: &str,
226    root_der: &[u8],
227) -> Result<MdsBlob, MdsError> {
228    // --- Step 1: exactly 3 segments; decode header ---
229    let parts: Vec<&str> = blob.split('.').collect();
230    if parts.len() != 3 {
231        return Err(MdsError::MalformedJwt);
232    }
233    let header_b64 = parts[0];
234
235    let header_bytes = URL_SAFE_NO_PAD
236        .decode(header_b64)
237        .map_err(|_| MdsError::InvalidHeaderEncoding)?;
238    let header: serde_json::Value =
239        serde_json::from_slice(&header_bytes).map_err(|_| MdsError::InvalidHeaderJson)?;
240
241    // --- Step 2: alg must be exactly RS256. Never read from the payload. ---
242    let alg = header.get("alg").and_then(|v| v.as_str()).unwrap_or("");
243    if alg != ALLOWED_ALG {
244        return Err(MdsError::UnsupportedAlgorithm(alg.to_string()));
245    }
246
247    // --- Step 3: x5c present, non-empty, <= 8 entries, standard base64 DER ---
248    let x5c_json = header.get("x5c").and_then(|v| v.as_array());
249    let x5c_json = match x5c_json {
250        Some(arr) if !arr.is_empty() => arr,
251        _ => return Err(MdsError::MissingX5c),
252    };
253    if x5c_json.len() > MAX_X5C_LEN {
254        return Err(MdsError::X5cTooLong);
255    }
256
257    let mut cert_ders: Vec<Vec<u8>> = Vec::with_capacity(x5c_json.len());
258    for v in x5c_json {
259        let s = v.as_str().ok_or(MdsError::InvalidX5cEncoding)?;
260        let der = STANDARD
261            .decode(s)
262            .map_err(|_| MdsError::InvalidX5cEncoding)?;
263        cert_ders.push(der);
264    }
265
266    let mut certs: Vec<X509Certificate<'_>> = Vec::with_capacity(cert_ders.len());
267    for der in &cert_ders {
268        let (_, cert) = X509Certificate::from_der(der)
269            .map_err(|e| MdsError::InvalidCertificate(e.to_string()))?;
270        certs.push(cert);
271    }
272
273    // --- Step 4: chain build ---
274    let now_ts = now.timestamp();
275    for cert in &certs {
276        let validity = cert.validity();
277        if now_ts < validity.not_before.timestamp() || now_ts > validity.not_after.timestamp() {
278            return Err(MdsError::CertificateExpired);
279        }
280    }
281
282    let n = certs.len();
283    for i in 0..n.saturating_sub(1) {
284        // `certs[i + 1]` is being used as an ISSUER here, so it must actually
285        // be permitted to issue — see `assert_is_issuer` for why signature
286        // verification alone is not enough under a public CA root.
287        assert_is_issuer(&certs[i + 1])?;
288        assert_path_len(&certs[i + 1], i)?;
289        certs[i]
290            .verify_signature(Some(certs[i + 1].public_key()))
291            .map_err(|_| MdsError::ChainVerifyFailed)?;
292    }
293
294    let (_, root_cert) = X509Certificate::from_der(root_der)
295        .map_err(|_| MdsError::InvalidCertificate("root anchor".into()))?;
296    let last = certs.last().expect("x5c is non-empty, checked above");
297    let last_is_root = last.as_raw() == root_der;
298    if !last_is_root {
299        // The last x5c entry is not our root itself, so it must be directly
300        // signed BY our root (the common real-world shape: x5c carries only
301        // leaf + intermediate, no root).
302        assert_is_issuer(&root_cert)?;
303        assert_path_len(&root_cert, n.saturating_sub(1))?;
304        last.verify_signature(Some(root_cert.public_key()))
305            .map_err(|_| MdsError::ChainVerifyFailed)?;
306    }
307    // If `last_is_root`, the per-`i` loop above already verified `certs[n-2]`
308    // (the cert before it, if any) against `certs[n-1]`'s public key — and
309    // `certs[n-1]` IS byte-identical to our trusted anchor, so that check
310    // already used the real trusted key, and `assert_is_issuer` already ran
311    // against it in that same iteration. No further action needed.
312
313    // The leaf must NOT itself be a CA: a chain that ends in an issuing
314    // certificate would let whoever holds it mint further certificates that
315    // this verifier would then accept.
316    if certs[0].is_ca() {
317        return Err(MdsError::LeafIsCa);
318    }
319
320    // --- Step 5: leaf identity pinning ---
321    let leaf = &certs[0];
322    let leaf_matches = match leaf
323        .subject_alternative_name()
324        .map_err(|e| MdsError::InvalidCertificate(e.to_string()))?
325    {
326        Some(ext) => ext.value.general_names.iter().any(
327            |gn| matches!(gn, GeneralName::DNSName(d) if d.eq_ignore_ascii_case(expected_leaf_dns)),
328        ),
329        // No SAN extension at all -> fall back to CN (D4 step 5).
330        None => leaf
331            .subject()
332            .iter_common_name()
333            .filter_map(|atv| atv.as_str().ok())
334            .any(|cn| cn.eq_ignore_ascii_case(expected_leaf_dns)),
335    };
336    if !leaf_matches {
337        return Err(MdsError::LeafIdentityMismatch);
338    }
339
340    // --- Step 6: verify the JWT signature with the leaf's RSA public key ---
341    let spki = leaf.public_key();
342    let decoding_key = DecodingKey::from_rsa_der(spki.subject_public_key.data.as_ref());
343    let mut validation = Validation::new(Algorithm::RS256);
344    validation.validate_exp = false;
345    validation.validate_aud = false;
346    validation.required_spec_claims.clear();
347
348    let token_data = jsonwebtoken::decode::<serde_json::Value>(blob, &decoding_key, &validation)
349        .map_err(|_| MdsError::SignatureInvalid)?;
350
351    // --- Step 7: only now does the payload get parsed ---
352    let raw_payload: RawMdsPayload = serde_json::from_value(token_data.claims)
353        .map_err(|e| MdsError::InvalidPayload(e.to_string()))?;
354
355    let (entries, skipped) = convert_entries(raw_payload.entries);
356    if skipped > 0 {
357        tracing::info!(
358            skipped_entries = skipped,
359            "MDS BLOB: skipped entries with no (or unparsable) aaguid — UAF/U2F entries"
360        );
361    }
362
363    // --- Step 9: staleness (step 8, rollback, is handled by the caller —
364    // see super::decide_ingest_outcome) ---
365    let next_update = NaiveDate::parse_from_str(&raw_payload.next_update, "%Y-%m-%d")
366        .map_err(|e| MdsError::InvalidNextUpdate(e.to_string()))?;
367    let stale = next_update < now.date_naive();
368    if stale {
369        tracing::warn!(
370            next_update = %next_update,
371            now = %now,
372            "MDS BLOB is stale (nextUpdate has passed) — ingesting anyway; policy decides"
373        );
374    }
375
376    let meta = MdsBlobMeta {
377        no: raw_payload.no,
378        next_update,
379        entry_count: entries.len() as u64,
380        last_refreshed_at: now,
381        stale,
382    };
383
384    Ok(MdsBlob { meta, entries })
385}

390//
391// These are *synthetic* test vectors: a real FIDO MDS3 BLOB is a signed JWT
392// from a live FIDO Alliance private key over a real CA chain, and is
393// deliberately not committed to this repo (it's ~10 MB, and there is no way
394// to "fake" a vendor-signed artifact honestly). Instead, this suite builds
395// its own throwaway root -> intermediate -> leaf PKI with `rcgen` and signs
396// a small synthetic payload with a leaf-embedded RSA key, verifying through
397// a seam (`verify_and_parse_with_root`) that accepts an injected root DER
398// instead of the real vendored anchor. `verify_and_parse` itself is only
399// exercised indirectly through that seam for the positive/negative-chain
400// cases; the digest-pin and real-anchor-loading behavior is covered
401// separately in `super::tests` (`root_anchor_digest_matches_pin`).
402//
403// A `#[ignore]`d test further down runs the *actual* `verify_and_parse`
404// (against the real vendored anchor) over a real downloaded BLOB when
405// `AXIAM_MDS_BLOB_PATH` is set — see its doc comment for how to run it.

412    /// A throwaway 2048-bit RSA key, generated fresh on first use.
413    ///
414    /// The JWT under test is always RS256, so these vectors need an RSA key
415    /// to sign with, and `rcgen` cannot generate one (`ring` has no RSA
416    /// keygen). It is generated here rather than embedded as a PEM literal:
417    /// a committed private key is indistinguishable from a leaked one to

419    /// fixture" is a claim a reader has to take on trust. Generating it

503    /// Build a compact JWT: `header.payload.signature`, RS256-signed with
504    /// `leaf_rsa_pkcs8_pem`, header carrying `alg`/`x5c` as given.
505    fn build_jwt(
506        alg: &str,
507        x5c: &[String],
508        payload_json: &serde_json::Value,
509        leaf_rsa_pkcs8_pem: &str,
510    ) -> String {
511        let header = serde_json::json!({ "alg": alg, "typ": "JWT", "x5c": x5c });
512        let header_b64 = URL_SAFE_NO_PAD.encode(serde_json::to_vec(&header).unwrap());
513        let payload_b64 = URL_SAFE_NO_PAD.encode(serde_json::to_vec(payload_json).unwrap());
514        let signing_input = format!("{header_b64}.{payload_b64}");
515
516        if alg != "RS256" {
517            // Callers testing alg rejection don't need (and, for "none",
518            // conceptually shouldn't have) a real signature.
519            return format!("{signing_input}.");
520        }
521
522        let encoding_key =
523            jsonwebtoken::EncodingKey::from_rsa_pem(leaf_rsa_pkcs8_pem.as_bytes()).unwrap();
524        let mut header_struct = jsonwebtoken::Header::new(jsonwebtoken::Algorithm::RS256);
525        header_struct.x5c = Some(x5c.to_vec());
526        // Signed via jsonwebtoken's own encoder (not the hand-built
527        // header_b64/signing_input above) to guarantee the header JSON that
528        // gets signed is exactly what jsonwebtoken itself produces.
529        jsonwebtoken::encode(&header_struct, payload_json, &encoding_key).unwrap()
530    }

561    /// ([`super::verify_and_parse_with_anchor`]) against a throwaway root.
562    /// There is deliberately no re-implementation of the verification logic
563    /// in this module: a test that copies the code it guards stops guarding

571        super::verify_and_parse_with_anchor(blob, now, expected_leaf_dns, root_der)
572    }

826    // --- the public-root splice attack (assert_is_issuer) ---

828    #[test]
829    fn end_entity_cert_cannot_be_used_as_an_issuer() {
830        // The attack this closes: the vendored anchor is a *public* CA root,
831        // so an attacker can legitimately obtain an ordinary end-entity
832        // certificate beneath it. If issuer certificates were not required
833        // to be CAs, the attacker could self-mint a leaf carrying
834        // `mds.fidoalliance.org`, sign it with their own EE key, and present
835        //
836        //     x5c = [ forged leaf, attacker EE cert, genuine intermediate ]
837        //
838        // — a chain in which every signature verifies and the leaf carries
839        // the pinned name. Only the CA check stops it.
840        let root_key = KeyPair::generate_for(&rcgen::PKCS_ED25519).unwrap();
841        let mut root_params = CertificateParams::new(Vec::<String>::new()).unwrap();
842        root_params
843            .distinguished_name
844            .push(DnType::CommonName, "Public Root CA");
845        root_params.is_ca = IsCa::Ca(rcgen::BasicConstraints::Unconstrained);
846        let root_cert = root_params.self_signed(&root_key).unwrap();
847        let root_der = root_cert.der().to_vec();
848        let root_issuer = Issuer::from_params(&root_params, root_key);
849
850        // An ordinary, genuinely-issued END-ENTITY certificate: CA=false.
851        let attacker_key = KeyPair::generate_for(&rcgen::PKCS_ED25519).unwrap();
852        let mut attacker_params = CertificateParams::new(vec!["attacker.example".to_string()])
853            .expect("attacker EE params");
854        attacker_params
855            .distinguished_name
856            .push(DnType::CommonName, "attacker.example");
857        attacker_params.is_ca = IsCa::NoCa;
858        let attacker_cert = attacker_params
859            .signed_by(&attacker_key, &root_issuer)
860            .unwrap();
861        let attacker_der = attacker_cert.der().to_vec();
862        let attacker_issuer = Issuer::from_params(&attacker_params, attacker_key);
863
864        // The attacker signs a leaf for the pinned MDS hostname with their
865        // own EE key. Cryptographically valid; semantically forbidden.
866        let leaf_key = KeyPair::from_pkcs8_pem_and_sign_algo(
867            &TEST_LEAF_RSA_PKCS8_PEM,
868            &rcgen::PKCS_RSA_SHA256,
869        )
870        .unwrap();
871        let mut leaf_params = CertificateParams::new(vec![LEAF_DNS.to_string()]).unwrap();
872        leaf_params
873            .distinguished_name
874            .push(DnType::CommonName, LEAF_DNS);
875        leaf_params.is_ca = IsCa::NoCa;
876        let leaf_cert = leaf_params.signed_by(&leaf_key, &attacker_issuer).unwrap();
877
878        let x5c = vec![
879            STANDARD.encode(leaf_cert.der()),
880            STANDARD.encode(&attacker_der),
881        ];
882        let payload = synthetic_payload(999, "2099-01-01");
883        let jwt = build_jwt("RS256", &x5c, &payload, &TEST_LEAF_RSA_PKCS8_PEM);
884
885        let result = verify_and_parse_with_root(&jwt, now(), LEAF_DNS, &root_der);
886        assert!(
887            matches!(result, Err(MdsError::IssuerNotCa)),
888            "an end-entity certificate must never be accepted as an issuer, got {result:?}"
889        );
890    }
