1//! Error type for FIDO MDS3 BLOB ingestion (X3).
2
3/// Errors from fetching, verifying, and parsing a FIDO MDS3 BLOB.
4///
5/// Every variant here corresponds to a fail-closed rejection point in
6/// [`crate::mds::blob::verify_and_parse`] or the fetch/load wrappers around
7/// it — there is no variant that means "trust this anyway".
8#[derive(Debug, thiserror::Error)]
9pub enum MdsError {
10    // -----------------------------------------------------------------
11    // D3: vendored root anchor
12    // -----------------------------------------------------------------
13    #[error("vendored FIDO MDS root anchor PEM could not be parsed")]
14    RootAnchorUnparsable,
15
16    /// The vendored anchor's SHA-256 digest does not match the pinned
17    /// constant. This must never happen for an untampered checkout — if it
18    /// does, treat the build as compromised (see the module docs on
19    /// `FIDO_MDS_ROOT_SHA256_HEX`).
20    #[error(
21        "vendored FIDO MDS root anchor digest mismatch — the anchor file may have been tampered with"
22    )]
23    RootAnchorDigestMismatch,
24
25    // -----------------------------------------------------------------
26    // D4 steps 1-3: JWT structure / header
27    // -----------------------------------------------------------------
28    #[error("malformed JWT: expected exactly 3 dot-separated segments")]
29    MalformedJwt,
30
31    #[error("JWT header is not valid base64url")]
32    InvalidHeaderEncoding,
33
34    #[error("JWT header is not valid JSON")]
35    InvalidHeaderJson,
36
37    /// The header's `alg` is not exactly `RS256`. Carries the rejected
38    /// value (or `"none"`/absent) for diagnostics — never used to decide
39    /// anything, only to report.
40    #[error("unsupported JWT algorithm {0:?} — only RS256 is accepted")]
41    UnsupportedAlgorithm(String),
42
43    #[error("JWT header x5c claim is missing or empty")]
44    MissingX5c,
45
46    #[error("JWT header x5c claim has more than 8 entries")]
47    X5cTooLong,
48
49    #[error("JWT header x5c entry is not valid base64")]
50    InvalidX5cEncoding,
51
52    #[error("x5c certificate entry could not be parsed as DER: {0}")]
53    InvalidCertificate(String),
54
55    // -----------------------------------------------------------------
56    // D4 step 4: chain build
57    // -----------------------------------------------------------------
58    #[error("a certificate in the x5c chain is outside its validity window")]
59    CertificateExpired,
60
61    #[error("x5c certificate chain signature verification failed")]
62    ChainVerifyFailed,
63
64    /// A certificate used as an issuer in the chain is not permitted to issue
65    /// certificates (`basicConstraints` absent or `CA=false`, or `keyUsage`
66    /// present without `keyCertSign`).
67    ///
68    /// This is the check that keeps the vendored anchor meaningful: without
69    /// it, any holder of an ordinary end-entity certificate issued under the
70    /// same public root could splice a self-minted leaf beneath their own
71    /// certificate and produce a chain that verifies. See
72    /// `blob::assert_is_issuer`.
73    #[error("a certificate used as an issuer in the x5c chain is not a CA")]
74    IssuerNotCa,
75
76    #[error("x5c chain is deeper than an issuing certificate's pathLenConstraint permits")]
77    PathLenExceeded,
78
79    #[error("the x5c leaf certificate is itself a CA certificate")]
80    LeafIsCa,
81
82    // -----------------------------------------------------------------
83    // D4 step 5: leaf identity pinning
84    // -----------------------------------------------------------------
85    #[error(
86        "leaf certificate identity does not match the expected MDS hostname (checked SAN DNS names, falling back to CN only when no SAN extension is present)"
87    )]
88    LeafIdentityMismatch,
89
90    // -----------------------------------------------------------------
91    // D4 step 6: JWT signature
92    // -----------------------------------------------------------------
93    #[error("JWT signature verification failed")]
94    SignatureInvalid,
95
96    // -----------------------------------------------------------------
97    // D4 steps 7-9: payload
98    // -----------------------------------------------------------------
99    #[error("BLOB payload is not valid JSON matching the expected MDS schema: {0}")]
100    InvalidPayload(String),
101
102    #[error("nextUpdate field could not be parsed as YYYY-MM-DD: {0}")]
103    InvalidNextUpdate(String),
104
105    // -----------------------------------------------------------------
106    // Fetch / local-file paths (D10)
107    // -----------------------------------------------------------------
108    #[error("MDS BLOB fetch failed: {0}")]
109    FetchFailed(String),
110
111    #[error("MDS BLOB response exceeded the {0}-byte cap")]
112    ResponseTooLarge(usize),
113
114    #[error("MDS BLOB response was not valid UTF-8")]
115    InvalidEncoding,
116
117    #[error("local MDS BLOB file could not be read: {0}")]
118    LocalFileRead(String),
119
120    #[error("MDS ingestion is disabled (AXIAM__PKI__MDS_ENABLED=false)")]
121    IngestionDisabled,
122}
123
124impl From<MdsError> for axiam_core::error::AxiamError {
125    fn from(err: MdsError) -> Self {
126        use axiam_core::error::AxiamError;
127        match err {
128            // Upstream/network problems — not our bug, mirrors
129            // axiam-federation's FederationError -> AxiamError mapping for
130            // the same class of failure.
131            MdsError::FetchFailed(msg) => AxiamError::ServiceUnavailable(msg),
132            MdsError::IngestionDisabled => AxiamError::Validation {
133                message: err.to_string(),
134            },
135            // Everything else is a verification/parsing failure: the BLOB
136            // (or its vendored anchor) failed a security-relevant check.
137            // None of these should ever be shown to an end user verbatim —
138            // callers log/audit the specific variant and surface a fixed
139            // message, same posture as WebauthnAttestationDenied (D11).
140            other => AxiamError::Internal(other.to_string()),
141        }
142    }
143}
