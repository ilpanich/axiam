1//! FIDO Alliance Metadata Service (MDS3) ingestion (X3).
2//!
3//! `axiam-pki` owns this because MDS ingestion is fundamentally a
4//! trust-anchor concern — the same category of thing as CA certificate
5//! management — not a WebAuthn concern: it verifies a signed BLOB's chain
6//! of custody back to a vendored root, exactly like `axiam-pki::mtls` does
7//! for mTLS client certificates, before anything is allowed to consult it.
8//!
9//! ## Module layout
10//!
11//! - [`error::MdsError`] — this crate's local error type for the whole
12//!   ingestion path (fetch, verify, parse).
13//! - [`blob::verify_and_parse`] — the fail-closed BLOB verification pipeline
14//!   (D4), and the [`blob::MdsBlob`] result type.
15//! - This module (`mod.rs`) — the vendored trust anchor (D3), the
16//!   fetch/local-file entry points (D10), and the rollback decision (D4
17//!   step 8).
18//!
19//! ## Why the vendored anchor is digest-pinned, not just "the file in git"
20//!
21//! [`FIDO_MDS_ROOT_SHA256_HEX`] is the root of trust for **every**
22//! attestation decision the policy engine (`axiam_core::models::webauthn_policy`)
23//! makes: if the vendored anchor were ever silently swapped for a different
24//! certificate, "only FIDO-certified authenticators may register" quietly
25//! becomes "any authenticator an attacker can mint an attestation chain
26//! for" — a security regression that would produce no test failure and no
27//! error, only a bad root key. Digest-pinning turns that into a loud,
28//! fail-closed startup error instead: [`load_root_anchor`] recomputes the
29//! SHA-256 of the vendored PEM's DER bytes on every use and refuses to
30//! proceed on any mismatch. The vendored `.pem` file is *not* re-fetched
31//! from anywhere at runtime — matching the digest against the pinned
32//! constant **is** the check.
33//!
34//! **Anchor update procedure** (for the admin guide, and for whoever needs
35//! to rotate this): obtain the new GlobalSign Root CA – R3 (or successor)
36//! certificate from a trusted, independently-verifiable source (e.g.
37//! GlobalSign's own published fingerprint page, not just the download URL),
38//! compute its SHA-256 over the DER encoding, update both the `.pem` file
39//! and [`FIDO_MDS_ROOT_SHA256_HEX`] in the same commit, and get that commit
40//! reviewed with the digest change called out explicitly — the digest
41//! constant is the actual security control; the PEM file is inert without it
42//! matching.
43

54/// The vendored FIDO MDS3 root anchor: GlobalSign Root CA – R3.
55///
56/// `include_str!`'d at compile time so the anchor ships inside the binary —
57/// there is no runtime file dependency and no way to point this at a
58/// different file without recompiling.
59const ROOT_ANCHOR_PEM: &str = include_str!("fido-mds-root-ca-r3.pem");
60
61/// SHA-256 (hex) of the vendored anchor's DER encoding. Independently
62/// verified against GlobalSign's published fingerprint for "GlobalSign Root
63/// CA - R3" at the time this was vendored (see the module docs for the
64/// update procedure). This is the actual security control — see module docs.
65pub const FIDO_MDS_ROOT_SHA256_HEX: &str =
66    "cbb522d7b7f127ad6a0113865bdf1cd4102e7d0759af635a7cf4720dc963c53b";
67
68/// Maximum accepted size for a fetched/loaded MDS3 BLOB response body.
69///
70/// The real BLOB is ~10 MB; this is deliberately raised well above the SSRF
71/// guard's default 5 MiB cap (`crate::ssrf::MAX_RESPONSE_BYTES`, private to
72/// that module) via [`crate::ssrf::guarded_fetch_with_cap`] — a named,
73/// MDS-specific constant rather than a global cap increase (D2/D10).
74pub const MDS_MAX_BLOB_BYTES: usize = 32 * 1024 * 1024;
75
76/// Parse the vendored root anchor PEM, verify its digest against
77/// [`FIDO_MDS_ROOT_SHA256_HEX`], and return its DER bytes.
78///
79/// Fails closed: any parse error or digest mismatch is an error, never a
80/// silent fallback.
81pub fn load_root_anchor() -> Result<Vec<u8>, MdsError> {
82    let parsed = pem::parse(ROOT_ANCHOR_PEM).map_err(|_| MdsError::RootAnchorUnparsable)?;
83    let der = parsed.contents();
84
85    let digest = Sha256::digest(der);
86    let digest_hex = hex::encode(digest);
87
88    if digest_hex != FIDO_MDS_ROOT_SHA256_HEX {
89        return Err(MdsError::RootAnchorDigestMismatch);
90    }
91
92    Ok(der.to_vec())
93}

99/// Fetch, verify, and parse the MDS3 BLOB from `url` through the shared SSRF
100/// guard ([`crate::ssrf::guarded_fetch_with_cap`], capped at
101/// [`MDS_MAX_BLOB_BYTES`]).
102///
103/// `allow_private` exists solely for integration tests that point this at a
104/// loopback mock server — production callers must pass `false`.
105pub async fn fetch_verified_blob(
106    url: &str,
107    expected_leaf_dns: &str,
108    allow_private: bool,
109) -> Result<MdsBlob, MdsError> {
110    let response =
111        crate::ssrf::guarded_fetch_with_cap(url, allow_private, MDS_MAX_BLOB_BYTES, |c, u| {
112            c.get(u)
113        })
114        .await
115        .map_err(|e| MdsError::FetchFailed(e.to_string()))?;
116
117    let bytes = crate::ssrf::read_capped_body(response, MDS_MAX_BLOB_BYTES)
118        .await
119        .map_err(|e| match e {
120            crate::ssrf::SsrfError::ResponseTooLarge(cap) => MdsError::ResponseTooLarge(cap),
121            other => MdsError::FetchFailed(other.to_string()),
122        })?;
123
124    let text = String::from_utf8(bytes).map_err(|_| MdsError::InvalidEncoding)?;
125    verify_and_parse(&text, Utc::now(), expected_leaf_dns)
126}
127
128/// Load, verify, and parse the MDS3 BLOB from a local file — the air-gapped
129/// deployment path (`AXIAM__PKI__MDS_BLOB_PATH`, D10). No network I/O
130/// happens on this path at all.
131pub fn load_verified_blob_from_file(
132    path: &std::path::Path,
133    expected_leaf_dns: &str,
134) -> Result<MdsBlob, MdsError> {
135    let text = std::fs::read_to_string(path).map_err(|e| MdsError::LocalFileRead(e.to_string()))?;
136    verify_and_parse(&text, Utc::now(), expected_leaf_dns)
137}

139// ---------------------------------------------------------------------------
140// D4 step 8: rollback protection
141// ---------------------------------------------------------------------------
142
143/// The outcome of comparing a freshly-verified BLOB's `no` (serial number)
144/// against the last stored serial (D4 step 8).
145///
146/// This is a **pure decision function**, deliberately kept separate from
147/// [`verify_and_parse`]: the pinned `verify_and_parse` signature
148/// (`blob, now, expected_leaf_dns`) has no way to receive "the currently
149/// stored `no`" — that value lives in storage (`MdsRepository::get_meta`,
150/// `axiam-core`), which is out of this wave's scope (`axiam-db` wiring is a
151/// later wave). The ingestion orchestrator that wires storage in is
152/// responsible for calling this after `verify_and_parse` succeeds and before
153/// calling `MdsRepository::replace_entries`.
154#[derive(Debug, Clone, Copy, PartialEq, Eq)]
155pub enum BlobIngestOutcome {
156    /// No stored serial exists yet — first-ever ingestion.
157    Initial,
158    /// `no` is strictly greater than the stored serial — replace.
159    Replace,
160    /// `no` equals the stored serial — a no-op refresh: only
161    /// `last_refreshed_at` should be bumped, entries are already current.
162    NoOpRefresh,
163    /// `no` is lower than the stored serial — a rollback attempt. Reject:
164    /// never replace stored entries with an older BLOB.
165    RollbackRejected,
166}
167
168/// D4 step 8: decide what a freshly-verified BLOB's serial number means
169/// relative to what's already stored.
170pub fn decide_ingest_outcome(new_no: i64, stored_no: Option<i64>) -> BlobIngestOutcome {
171    match stored_no {
172        None => BlobIngestOutcome::Initial,
173        Some(stored) if new_no > stored => BlobIngestOutcome::Replace,
174        Some(stored) if new_no == stored => BlobIngestOutcome::NoOpRefresh,
175        Some(_) => BlobIngestOutcome::RollbackRejected,
176    }
177}

183#[cfg(test)]
184mod tests {
185    use super::*;
186
187    #[test]
188    fn root_anchor_digest_matches_pin() {
189        let der = load_root_anchor().expect("vendored anchor must parse and match its pin");
190        let digest_hex = hex::encode(Sha256::digest(&der));
191        assert_eq!(digest_hex, FIDO_MDS_ROOT_SHA256_HEX);
192    }
193
194    #[test]
195    fn root_anchor_digest_mismatch_fails_closed() {
196        // Simulate a tampered anchor: a different (but validly-PEM-encoded)
197        // certificate would parse fine but must not match the pin. We can't
198        // swap the embedded `include_str!` at runtime, so this test instead
199        // proves the *mechanism* — recomputing the digest of arbitrary bytes
200        // and comparing against the pinned constant — rejects a mismatch.
201        let not_the_root_der = b"this is not the vendored root certificate";
202        let digest_hex = hex::encode(Sha256::digest(not_the_root_der));
203        assert_ne!(
204            digest_hex, FIDO_MDS_ROOT_SHA256_HEX,
205            "sanity: arbitrary bytes must not coincidentally match the pin"
206        );
207    }
208
209    #[test]
210    fn decide_ingest_outcome_initial_when_nothing_stored() {
211        assert_eq!(decide_ingest_outcome(100, None), BlobIngestOutcome::Initial);
212    }
213
214    #[test]
215    fn decide_ingest_outcome_replace_when_newer() {
216        assert_eq!(
217            decide_ingest_outcome(101, Some(100)),
218            BlobIngestOutcome::Replace
219        );
220    }
221
222    #[test]
223    fn decide_ingest_outcome_noop_refresh_when_equal() {
224        assert_eq!(
225            decide_ingest_outcome(100, Some(100)),
226            BlobIngestOutcome::NoOpRefresh
227        );
228    }
229
230    #[test]
231    fn decide_ingest_outcome_rejects_rollback() {
232        assert_eq!(
233            decide_ingest_outcome(99, Some(100)),
234            BlobIngestOutcome::RollbackRejected
235        );
236    }
237
238    #[tokio::test]
239    async fn fetch_verified_blob_rejects_non_https_without_seam() {
240        let result =
241            fetch_verified_blob("http://example.com/blob.jwt", "mds.fidoalliance.org", false).await;
242        assert!(matches!(result, Err(MdsError::FetchFailed(_))));
243    }
244
245    #[test]
246    fn load_verified_blob_from_file_reports_missing_file() {
247        let result = load_verified_blob_from_file(
248            std::path::Path::new("/nonexistent/path/to/blob.jwt"),
249            "mds.fidoalliance.org",
250        );
251        assert!(matches!(result, Err(MdsError::LocalFileRead(_))));
252    }
