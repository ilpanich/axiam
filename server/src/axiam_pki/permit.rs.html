1//! The bounded wait for the shared crypto semaphore (#671, T-102).
2//!
3//! `axiam-pki` cannot call `axiam_auth::crypto_gate::acquire_hash_permit`
4//! (the two crates share a layer), so the PKI services carry the same bound:
5//! the deployment's `hash_acquire_timeout_secs`, set at the composition root
6//! with each service's `with_permit_wait`.
7
8use std::time::Duration;
9
10use axiam_core::error::{AxiamError, AxiamResult};
11use tokio::sync::{Semaphore, SemaphorePermit};
12
13/// The wait a service uses until the composition root replaces it; equal to
14/// the default of `hash_acquire_timeout_secs`.
15pub const DEFAULT_PERMIT_WAIT: Duration = Duration::from_secs(5);
16
17/// A permit of `semaphore`, or `ServiceUnavailable` (HTTP 503) once `wait` has
18/// passed. A closed semaphore is `Internal`, as the untimed call was.
19pub(crate) async fn bounded_permit(
20    semaphore: &Semaphore,
21    wait: Duration,
22) -> AxiamResult<SemaphorePermit<'_>> {
23    match tokio::time::timeout(wait, semaphore.acquire()).await {
24        Ok(Ok(permit)) => Ok(permit),
25        Ok(Err(_closed)) => Err(AxiamError::Internal("crypto semaphore closed".into())),
26        Err(_elapsed) => Err(AxiamError::ServiceUnavailable(
27            "the service is at capacity; please retry shortly".into(),
28        )),
29    }
30}
