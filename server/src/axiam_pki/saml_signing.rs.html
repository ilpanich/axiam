1//! The tenant's SAML identity-provider signing credential — issuance, sealing,
2//! lookup and retirement (G-2, T23.2.1, D-21).
3//!
4//! One RSA-4096 leaf per tenant, signed by a signing CA of the tenant's
5//! organization, carrying
6//! [`CertificateType::SamlSigning`](axiam_core::models::certificate::CertificateType::SamlSigning)'s
7//! profile (`digitalSignature` only, `documentSigning`, no SANs). The signing is
8//! [`CertService`](crate::cert::CertService)'s own — the same CA lookup, tenant
9//! fence, issuer-bounded validity, custodian resolution and leaf construction,
10//! through `CertService::issue_leaf_material` — and what differs is only
11//! where the result goes: **not** a `certificate` row (D-21), but a
12//! `saml_idp_credential` row whose private key is sealed AES-256-GCM under
13//! `pki_encryption_key` by [`DatabaseCaKeyStore`](crate::DatabaseCaKeyStore).
14//!
15//! # What this does not do
16//!
17//! * **No lazy creation.** Nothing issues a credential as a side effect of
18//!   serving metadata or an SSO request: an SP pins the metadata certificate, so
19//!   it changes only by an administrator's act, and [`SamlIdpCredentialService::issue`]
20//!   is that act.
21//! * **No automatic rotation.** The `next` slot exists so rotation needs no
22//!   schema change; only an administrator's
23//!   [`promote`](SamlIdpCredentialService::promote) (T23.2.5, D-42) moves it.
24//!
25//! # The key
26//!
27//! Never returned by any API and never in a `Debug`. [`SamlIdpSigningKey`] is the
28//! one place plaintext exists: a [`Zeroizing`] buffer, returned by
29//! [`SamlIdpCredentialService::get_active_signing_key`] for the signer.
30

53/// The longest a SAML IdP signing certificate may be valid: two years, in days.
54///
55/// Shorter than the 825-day cap on generic leaves, because this key signs every
56/// assertion the tenant issues and an SP pins the certificate: the window a
57/// leaked key stays useful for is the window it is valid for. Also bounded by
58/// the issuing CA's own `not_after`, as for every leaf.
59pub const MAX_SAML_IDP_CREDENTIAL_VALIDITY_DAYS: u32 = 730;
60
61/// The algorithm of the SAML signing key (D-21): RSA-4096 with `rsa-sha256`,
62/// the XML-DSig pair every service provider accepts.
63const KEY_ALGORITHM: KeyAlgorithm = KeyAlgorithm::Rsa4096;
64
65/// The credential's private key in the clear, for the signer.
66///
67/// A [`Zeroizing`] buffer, so the PEM is overwritten when this is dropped, and a
68/// `Debug` that prints the public facts and no key.
69pub struct SamlIdpSigningKey {
70    /// The credential the key belongs to.
71    pub credential: SamlIdpCredential,
72    /// PKCS#8 PEM of the RSA-4096 private key.
73    pub private_key_pem: Zeroizing<String>,
74}
75
76impl std::fmt::Debug for SamlIdpSigningKey {
77    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
78        f.debug_struct("SamlIdpSigningKey")
79            .field("credential", &self.credential)
80            .field("private_key_pem", &"[REDACTED]")
81            .finish()
82    }
83}

85/// Issues, seals, looks up and retires a tenant's SAML signing credentials.
86///
87/// Holds a [`CertService`] only for its issuing path — it never calls the
88/// service's `generate`, so no `certificate` row is written — and the
89/// custodians the database custodian is resolved from.
90#[derive(Clone)]
91pub struct SamlIdpCredentialService<CA, CR, SR> {
92    leaf_issuer: CertService<CA, CR>,
93    custodians: Arc<CaKeyCustodians>,
94    repo: SR,
95}
96
97impl<CA, CR, SR> SamlIdpCredentialService<CA, CR, SR>
98where
99    CA: CaCertificateRepository,
100    CR: CertificateRepository,
101    SR: SamlIdpCredentialRepository,
102{
103    /// Build the service over the certificate service whose issuing path it
104    /// reuses, the CA key custodians the rest of the PKI already shares, and the
105    /// credential repository.
106    pub fn new(
107        leaf_issuer: CertService<CA, CR>,
108        custodians: Arc<CaKeyCustodians>,
109        repo: SR,
110    ) -> Self {
111        Self {
112            leaf_issuer,
113            custodians,
114            repo,
115        }
116    }
117
118    /// Issue the tenant's SAML signing credential from the named CA, and store
119    /// it with its key sealed.
120    ///
121    /// `issuer_ca_id` must name an **active signing CA** the caller may use: a CA
122    /// of another organization is `NotFound`; one of another tenant, or the
123    /// organization anchor for a [`IssuingScope::Tenant`] caller, is `NotFound`
124    /// too (the rule `CertService` applies to every leaf); a revoked one, an
125    /// expired one, or one whose key AXIAM does not hold (an imported CA, which
126    /// cannot sign) is refused. `validity_days` is 1 to
127    /// [`MAX_SAML_IDP_CREDENTIAL_VALIDITY_DAYS`] and cannot outlive the CA.
128    ///
129    /// `status` is [`SamlIdpCredentialStatus::Active`] or
130    /// [`SamlIdpCredentialStatus::Next`]. A tenant already holding one in that
131    /// slot is `AlreadyExists` — checked first so an occupied slot does not cost
132    /// an RSA-4096 key generation, and enforced by the database for the race the
133    /// check cannot see.
134    pub async fn issue(
135        &self,
136        org_id: Uuid,
137        tenant_id: Uuid,
138        scope: IssuingScope,
139        issuer_ca_id: Uuid,
140        validity_days: u32,
141        status: SamlIdpCredentialStatus,
142    ) -> AxiamResult<SamlIdpCredential> {
143        if status == SamlIdpCredentialStatus::Retired {
144            return Err(AxiamError::Validation {
145                message: "a SAML IdP credential is issued active or next, never retired".into(),
146            });
147        }
148        if self
149            .repo
150            .list(tenant_id)
151            .await?
152            .iter()
153            .any(|c| c.status == status)
154        {
155            return Err(AxiamError::AlreadyExists {
156                entity: "saml_idp_credential".into(),
157            });
158        }
159
160        let id = new_id();
161        let subject = subject_common_name(&format!("AXIAM SAML IdP {tenant_id}"))?;
162
163        // The same issuing path `CertService::generate` runs, stopped before it
164        // writes an inventory row. `generate` itself refuses `SamlSigning` by
165        // name; this is the one caller that may name it.
166        let mut leaf = self
167            .leaf_issuer
168            .issue_leaf_material(
169                org_id,
170                scope,
171                &CreateCertificate {
172                    tenant_id,
173                    issuer_ca_id,
174                    subject,
175                    cert_type: CertificateType::SamlSigning,
176                    key_algorithm: KEY_ALGORITHM,
177                    validity_days,
178                    metadata: None,
179                    // A SAML signing leaf names no host.
180                    subject_alt_names: Vec::new(),
181                },
182                Some(MAX_SAML_IDP_CREDENTIAL_VALIDITY_DAYS),
183                &[],
184            )
185            .await
186            .map_err(ca_that_cannot_sign)?;
187        // Into a zeroizing buffer at once. The move leaves no second copy.
188        let private_key_pem = Zeroizing::new(std::mem::take(&mut leaf.private_key_pem));
189        let serial = leaf_serial_hex(&leaf.public_cert_pem)?;
190
191        // D-21: sealed through the database custodian, explicitly — not
192        // through whichever custodian is the deployment's default. The key
193        // signs on every assertion, so a Vault round trip per sign-in is the
194        // wrong latency, and `VaultPki` cannot hold an exported key at all.
195        let custodian = self.custodians.store_for(CaKeyCustody::Database)?;
196        let key = match custodian.store(org_id, id, &private_key_pem).await? {
197            StoredCaKey::Inline(ciphertext) => SealedSamlIdpKey {
198                custody: custodian.custody(),
199                locator: None,
200                ciphertext: Some(ciphertext),
201            },
202            StoredCaKey::Referenced(locator) => SealedSamlIdpKey {
203                custody: custodian.custody(),
204                locator: Some(locator),
205                ciphertext: None,
206            },
207        };
208
209        self.repo
210            .create(StoreSamlIdpCredential {
211                id,
212                tenant_id,
213                issuer_ca_id,
214                certificate_pem: leaf.public_cert_pem,
215                serial,
216                fingerprint: leaf.fingerprint,
217                not_before: leaf.not_before,
218                not_after: leaf.not_after,
219                status,
220                key,
221            })
222            .await
223    }

225    /// The tenant's active credential with its private key opened, for the
226    /// assertion signer; `None` when the tenant has no active credential.
227    ///
228    /// The key is decrypted into a [`Zeroizing`] buffer and nowhere else. This
229    /// does **not** refuse an expired credential: whether to sign with a
230    /// certificate past its `not_after` is the signer's decision, made where it
231    /// can say why.
232    pub async fn get_active_signing_key(
233        &self,
234        org_id: Uuid,
235        tenant_id: Uuid,
236    ) -> AxiamResult<Option<SamlIdpSigningKey>> {
237        let Some(sealed) = self.repo.get_active_sealed(tenant_id).await? else {
238            return Ok(None);
239        };
240        let key_ref = CaKeyRef {
241            organization_id: org_id,
242            ca_id: sealed.credential.id,
243            custody: sealed.key.custody,
244            locator: sealed.key.locator.clone().unwrap_or_default(),
245        };
246        let private_key_pem = self
247            .custodians
248            .store_for(sealed.key.custody)?
249            .load(&key_ref, sealed.key.ciphertext.as_deref())
250            .await?;
251        Ok(Some(SamlIdpSigningKey {
252            credential: sealed.credential,
253            private_key_pem,
254        }))
255    }

257    /// The tenant's active credential, without its key; `None` when there is
258    /// none.
259    pub async fn get_active(&self, tenant_id: Uuid) -> AxiamResult<Option<SamlIdpCredential>> {
260        self.repo.get_active(tenant_id).await
261    }
262
263    /// Every credential of the tenant, oldest first, without key material.
264    pub async fn list(&self, tenant_id: Uuid) -> AxiamResult<Vec<SamlIdpCredential>> {
265        self.repo.list(tenant_id).await
266    }
267
268    /// Promote the tenant's `next` credential to `active`, retiring the one that
269    /// was (its key destroyed), in one transaction (D-42).
270    ///
271    /// `id` must be the tenant's current `next` credential and `now` inside its
272    /// validity window, or the answer is `Conflict` and nothing changes;
273    /// `NotFound` for an id that is not this tenant's. See
274    /// [`SamlIdpCredentialRepository::promote`].
275    pub async fn promote(
276        &self,
277        tenant_id: Uuid,
278        id: Uuid,
279    ) -> AxiamResult<SamlIdpCredentialPromotion> {
280        self.repo.promote(tenant_id, id, chrono::Utc::now()).await
281    }
282
283    /// Retire a credential: out of its slot, out of metadata, key destroyed.
284    ///
285    /// The sealed key is cleared in the same write that takes the row out of its
286    /// slot, so database custody — the only custody [`Self::issue`] uses — leaves
287    /// nothing behind. No CRL entry is made: service providers pin the metadata
288    /// certificate, so removing it from metadata is what ends their trust. A
289    /// custodian that kept the key outside the row would need its own delete
290    /// here; none is reachable today, and the row records the custody so adding
291    /// one is a code change and not a migration.
292    ///
293    /// Retiring a retired credential returns it unchanged. `NotFound` for an id
294    /// that is not this tenant's.
295    pub async fn retire(&self, tenant_id: Uuid, id: Uuid) -> AxiamResult<SamlIdpCredential> {
296        self.repo.retire(tenant_id, id).await
297    }
298}

300/// A CA that is revoked, expired or not yet valid is a statement about the
301/// request — the caller named a CA that cannot sign — so it is a `Validation`
302/// the caller reads, not the `Certificate` error the generic issuing paths give
303/// it (which reaches a client as a bare 500). Everything else passes through.
304fn ca_that_cannot_sign(error: AxiamError) -> AxiamError {
305    match error {
306        AxiamError::Certificate(message) if message == CA_NOT_ACTIVE => AxiamError::Validation {
307            message: "the issuing CA is not active (revoked or retired) and cannot sign".into(),
308        },
309        AxiamError::Certificate(message) if message == CA_NOT_VALID => AxiamError::Validation {
310            message: "the issuing CA is expired or not yet valid and cannot sign".into(),
311        },
312        other => other,
313    }
314}

316/// The serial number of a PEM certificate as lower-case hex.
317fn leaf_serial_hex(pem: &str) -> AxiamResult<String> {
318    let (_, block) = x509_parser::pem::parse_x509_pem(pem.as_bytes()).map_err(|e| {
319        AxiamError::Certificate(format!("the signed certificate is not a PEM: {e}"))
320    })?;
321    let (_, cert) = X509Certificate::from_der(&block.contents).map_err(|e| {
322        AxiamError::Certificate(format!("the signed certificate could not be parsed: {e}"))
323    })?;
324    Ok(hex::encode(cert.raw_serial()))
