1//! Shared SSRF guard — resolve-once-and-pin outbound fetch helper.
2//!
3//! Generalizes the byte-identical guard logic previously duplicated in
4//! `jwks_cache::is_private_jwks_ip`/`validate_jwks_url` and
5//! `axiam-api-rest::webhook::is_private_ip`/`resolve_and_validate_host`
6//! (SECHRD-02 / D-01a DRY) into one reusable module, and — critically —
7//! adds IP **pinning**, which neither prior guard did: both validated the
8//! resolved `IpAddr` and then let `reqwest` re-resolve DNS independently at
9//! send time, leaving a DNS-rebind TOCTOU window open between validation and
10//! connect (D-01c closes this).
11//!
12//! Use [`guarded_fetch`] for every outbound fetch to an admin/IdP-supplied
13//! URL (JWKS, OIDC discovery, OIDC token exchange, SAML metadata, webhook
14//! delivery). It:
15//!
16//! 1. Resolves the host (A + AAAA) fresh — no cross-request DNS caching
17//!    (D-01c).
18//! 2. Rejects the fetch if ANY resolved address is non-globally-routable
19//!    (D-01a) — unless the caller opted into the `allow_private` test seam
20//!    (see below). [`is_disallowed_ip`] enumerates the families and, since
21//!    SEC-094, canonicalises the IPv4-in-IPv6 encodings first: an `AAAA`
22//!    record carrying `::ffff:169.254.169.254` used to pass this step and
23//!    then be *pinned* by step 3.
24//! 3. Pins the exact validated `IpAddr` into a fresh, single-use
25//!    `reqwest::Client` via `ClientBuilder::resolve()`, so the socket that
26//!    is actually opened is the one that was validated — not a second,
27//!    independently-resolved address (D-01c).
28//! 4. Disables `reqwest`'s automatic redirect following and instead
29//!    manually re-runs the FULL guard (resolve → validate → pin → send)
30//!    against the `Location` target, bounded to [`MAX_HOPS`] hops (D-01b).
31//!
32//! ## The `allow_private` test seam only applies to the first hop
33//!
34//! `allow_private` exists solely so integration tests can point a guarded
35//! fetch at a loopback mock server (mirrors the pre-existing
36//! `JwksCache::new_allow_private_networks` seam). It is honored **only for
37//! the very first hop** of [`guarded_fetch`] — every redirect hop after that
38//! is always validated with the strict (production) check, regardless of
39//! `allow_private`. A `Location` header is attacker-influenced response
40//! data, not the admin-configured URL the caller opted to trust; holding it
41//! to the same relaxed standard as the test seam would silently defeat the
42//! redirect-bypass defense this module exists to provide (D-01b: "re-run
43//! the full SSRF guard against the redirect target").
44

50/// Maximum number of redirect hops [`guarded_fetch`] will follow before

54/// Errors produced by the shared SSRF guard.
55#[derive(Debug, thiserror::Error)]
56pub enum SsrfError {
57    #[error("invalid URL")]
58    InvalidUrl,
59    #[error("failed to resolve host")]
60    ResolveFailed,
61    #[error("SSRF blocked: resolved IP is private/loopback/link-local/unspecified")]
62    Blocked,
63    #[error("failed to build HTTP client")]

65    #[error("HTTP request failed: {0}")]
66    RequestFailed(String),
67    #[error("too many redirects")]
68    TooManyRedirects,
69    #[error("SSRF blocked: non-HTTPS scheme not permitted for IdP fetches")]
70    InsecureScheme,

76/// Discovery/metadata/token/JWKS documents are small; a multi-GB body is a

93static ALLOWED_HOSTS: OnceLock<BTreeSet<String>> = OnceLock::new();

95/// Install the operator's SSRF host exception list (SEC-107).
96///
97/// # Why an exception exists at all
98///
99/// The address rule is a *network-topology* control being used as a *trust*
100/// control, and in real deployments the two diverge: a Kubernetes-internal
101/// Keycloak or Entra proxy at `10.x` is a perfectly legitimate IdP, and so is
102/// an internal webhook consumer, an internal MDS mirror, or an air-gapped
103/// deployment where *everything* is RFC1918. With no exception at all, an
104/// operator's only recourse is to run AXIAM outside the guard's assumptions or
105/// to patch the binary — and a guard that gets patched around protects nobody.
106/// It has already cost this project measurable work (R5.4's cross-vendor
107/// Keycloak test cannot use the containerized server for exactly this reason).
108///
109/// # Why it is a host list and not a switch or a CIDR range
110///
111/// A boolean `SSRF_ALLOW_PRIVATE=true` is the `verify_peer: false` of this
112/// module: it appears in a dev compose file, works, and travels unchanged into
113/// production. A CIDR exception can be *widened by a DNS answer* — allow
114/// `10.0.0.0/8` and any hostname an admin can set now reaches anything in it.
115/// A host exception cannot: the operator names the exact destination they
116/// intend to reach, and a DNS answer for some *other* host is still blocked.
117///
118/// # The four properties that keep it from being a hole
119///
120/// 1. **Default-empty.** Nothing is exempt unless this is called with a
121///    non-empty list. It is a `OnceLock`, so it is set once at composition and
122///    cannot be re-armed later by anything.
123/// 2. **Exact match only.** ASCII-lowercased equality — no wildcards, no
124///    suffix matching, no "ends with .internal". `*.corp` cannot be spelled.
125/// 3. **First hop only.** Redirect targets are always validated strictly, for
126///    the reason the module docs give about `allow_private`: a `Location`
127///    header is attacker-influenced response data, not the URL the operator
128///    chose to trust.
129/// 4. **The metadata services stay unreachable** — [`is_never_allowed`]. An
130///    operator asking for their `10.x` IdP is not asking for
131///    `169.254.169.254`, and an allowlisted host whose DNS is poisoned onto a
132///    metadata endpoint is exactly the attack this control would otherwise
133///    re-open. That includes SEC-094's IPv4-mapped spelling, because the
134///    canonicalisation runs before this check, not after it.
135///
136/// Returns the number of entries installed. Calling it twice is a no-op on the
137/// second call (the `OnceLock` keeps the first list), which is deliberate:
138/// re-arming a security exception at runtime is not a feature.
139pub fn set_allowed_hosts<I: IntoIterator<Item = String>>(hosts: I) -> usize {
140    let set: BTreeSet<String> = hosts
141        .into_iter()
142        .map(|h| h.trim().to_ascii_lowercase())
143        .filter(|h| !h.is_empty())
144        .collect();
145    let _ = ALLOWED_HOSTS.set(set);
146    allowed_hosts().len()
147}

149/// Parse the comma-separated [`ALLOWED_HOSTS_ENV`] form.
150pub fn parse_allowed_hosts(raw: &str) -> Vec<String> {
151    raw.split(',')
152        .map(|h| h.trim().to_ascii_lowercase())
153        .filter(|h| !h.is_empty())
154        .collect()
155}

157/// The installed exception list. Empty until [`set_allowed_hosts`] is called.
158pub fn allowed_hosts() -> &'static BTreeSet<String> {
159    static EMPTY: OnceLock<BTreeSet<String>> = OnceLock::new();
160    ALLOWED_HOSTS
161        .get()
162        .unwrap_or_else(|| EMPTY.get_or_init(BTreeSet::new))
163}

165/// Whether `host` is one the operator explicitly named (SEC-107).
166fn host_is_allowlisted(host: &str) -> bool {
167    let set = allowed_hosts();
168    !set.is_empty() && set.contains(&host.trim().to_ascii_lowercase())
169}

171/// Addresses that stay blocked **even for an allowlisted host** (SEC-107).
172///
173/// The exception exists so an operator can reach their own internal IdP. It
174/// does not exist so a poisoned DNS answer for that IdP's name can reach the
175/// cloud metadata service, which is the one destination on the private-address
176/// map that turns SSRF into credential theft. Each entry below is a metadata
177/// endpoint or a transition encoding that can name one:
178///
179/// | Address | Who |
180/// |---|---|
181/// | `169.254.0.0/16` | AWS / GCP / Azure IMDS (`169.254.169.254`) — the whole link-local range, because `169.254.170.2` (ECS task roles) is in it too |
182/// | `fe80::/10` | the IPv6 link-local equivalent |
183/// | `fd00:ec2::254` | AWS IMDS over IPv6 |
184/// | `100.100.100.200` | Alibaba Cloud metadata |
185/// | `192.0.0.192` | Oracle Cloud metadata |
186/// | `::/96` | the deprecated IPv4-compatible encoding — nothing legitimate resolves to it, and it is a second spelling of everything above |
187///
188/// Canonicalisation happens in [`is_disallowed_ip`]'s caller path *before*
189/// this runs, so `::ffff:169.254.169.254` is classified as the v4 address it
190/// denotes and is refused here too — SEC-094 is not re-opened by SEC-107.
191fn is_never_allowed(ip: IpAddr) -> bool {
192    let canonical = match ip {
193        IpAddr::V6(v6) => match v6.to_ipv4_mapped() {
194            Some(v4) => IpAddr::V4(v4),
195            None => ip,
196        },
197        v4 => v4,
198    };
199    match canonical {
200        IpAddr::V4(v4) => {
201            v4.is_link_local()
202                || v4 == Ipv4Addr::new(100, 100, 100, 200)
203                || v4 == Ipv4Addr::new(192, 0, 0, 192)
204        }
205        IpAddr::V6(v6) => {
206            let s = v6.segments();
207            // fe80::/10
208            (s[0] & 0xffc0 == 0xfe80)
209                // ::/96, including :: and ::1 — the deprecated
210                // IPv4-compatible form, which `to_ipv4_mapped` does not fold.
211                || is_ipv4_compatible_v6(v6)
212                // fd00:ec2::254 — AWS IMDS over IPv6, exactly.
213                || v6 == "fd00:ec2::254".parse::<Ipv6Addr>().expect("literal")
214        }
215    }
216}

218/// Returns `true` for IPv4 addresses that must never be contacted from a
219/// server-side outbound fetch (SEC-094).
220///
221/// | Range | RFC | Why |
222/// |---|---|---|
223/// | `0.0.0.0/8` | RFC 1122 | "this network"; `0.0.0.0` reaches localhost on Linux |
224/// | `10/8`, `172.16/12`, `192.168/16` | RFC 1918 | private |
225/// | `100.64.0.0/10` | RFC 6598 | CGNAT shared address space — carrier-internal |
226/// | `127.0.0.0/8` | RFC 1122 | loopback (the whole /8, not just `127.0.0.1`) |
227/// | `169.254.0.0/16` | RFC 3927 | link-local — **the cloud metadata service** |
228/// | `192.0.0.0/24` | RFC 6890 | IETF protocol assignments (incl. `192.0.0.8` etc.) |
229/// | `192.0.2.0/24`, `198.51.100.0/24`, `203.0.113.0/24` | RFC 5737 | documentation |
230/// | `192.88.99.0/24` | RFC 7526 | deprecated 6to4 relay anycast |
231/// | `198.18.0.0/15` | RFC 2544 | benchmarking |
232/// | `224.0.0.0/4` | RFC 5771 | multicast |
233/// | `240.0.0.0/4` | RFC 1112 | reserved, incl. `255.255.255.255` broadcast |
234///
235/// `Ipv4Addr::is_shared`, `is_documentation`, `is_benchmarking` and
236/// `is_reserved` are all still unstable (`#![feature(ip)]`), hence the
237/// hand-rolled octet arithmetic.
238fn is_disallowed_ipv4(v4: Ipv4Addr) -> bool {
239    let o = v4.octets();
240    v4.is_loopback()
241        || v4.is_private()
242        || v4.is_link_local()
243        || v4.is_broadcast()
244        || v4.is_multicast()
245        || o[0] == 0 // 0.0.0.0/8 — not merely `is_unspecified()`
246        || (o[0] == 100 && (o[1] & 0xc0) == 0x40) // 100.64.0.0/10 CGNAT
247        || (o[0] == 192 && o[1] == 0 && o[2] == 0) // 192.0.0.0/24
248        || (o[0] == 192 && o[1] == 0 && o[2] == 2) // 192.0.2.0/24
249        || (o[0] == 192 && o[1] == 88 && o[2] == 99) // 192.88.99.0/24
250        || (o[0] == 198 && (o[1] & 0xfe) == 18) // 198.18.0.0/15
251        || (o[0] == 198 && o[1] == 51 && o[2] == 100) // 198.51.100.0/24
252        || (o[0] == 203 && o[1] == 0 && o[2] == 113) // 203.0.113.0/24
253        || o[0] >= 240 // 240.0.0.0/4 reserved + broadcast
254}

256/// Returns `true` for IPv6 addresses that must never be contacted.
257///
258/// Callers reach this only via [`is_disallowed_ip`], which has already dealt
259/// with the two `::`-prefixed IPv4-in-IPv6 forms — `::ffff:a.b.c.d` is
260/// canonicalised to [`is_disallowed_ipv4`], `::a.b.c.d` is rejected outright.
261/// What is left here is either genuinely-v6 or a *transition* encoding, and the
262/// transition encodings are handled explicitly below because each one can name
263/// an internal v4 host.
264fn is_disallowed_ipv6(v6: Ipv6Addr) -> bool {
265    let s = v6.segments();
266
267    // 6to4, 2002::/16 (RFC 3056): bits 16..48 are the IPv4 address of the
268    // 6to4 site. `2002:7f00:0001::1` is 127.0.0.1; `2002:a9fe:a9fe::1` is the
269    // metadata service. The prefix as a whole cannot be blocked (it maps the
270    // ENTIRE public IPv4 space), so classify the address it embeds.
271    if s[0] == 0x2002 {
272        let embedded = Ipv4Addr::new(
273            (s[1] >> 8) as u8,
274            (s[1] & 0xff) as u8,
275            (s[2] >> 8) as u8,
276            (s[2] & 0xff) as u8,
277        );
278        if is_disallowed_ipv4(embedded) {
279            return true;
280        }
281    }
282
283    // NAT64 well-known prefix, 64:ff9b::/96 (RFC 6052): low 32 bits are the
284    // IPv4 destination. Same reasoning as 6to4 — classify what it embeds.
285    if s[0] == 0x0064 && s[1] == 0xff9b && s[2] == 0 && s[3] == 0 && s[4] == 0 && s[5] == 0 {
286        let embedded = Ipv4Addr::new(
287            (s[6] >> 8) as u8,
288            (s[6] & 0xff) as u8,
289            (s[7] >> 8) as u8,
290            (s[7] & 0xff) as u8,
291        );
292        if is_disallowed_ipv4(embedded) {
293            return true;
294        }
295    }
296
297    v6.is_loopback()                            // ::1
298        || v6.is_unspecified()                  // ::
299        || v6.is_multicast()                    // ff00::/8
300        || (s[0] & 0xffc0) == 0xfe80            // fe80::/10 link-local
301        || (s[0] & 0xffc0) == 0xfec0            // fec0::/10 site-local (deprecated, RFC3879)
302        || (s[0] & 0xfe00) == 0xfc00            // fc00::/7 unique-local
303        || (s[0] == 0x0100 && s[1] == 0 && s[2] == 0 && s[3] == 0) // 100::/64 discard-only
304        || (s[0] == 0x0064 && s[1] == 0xff9b && s[2] == 0x0001)    // 64:ff9b:1::/48 local NAT64
305        // 2001::/23 IETF protocol assignments (RFC 2928): Teredo 2001::/32,
306        // benchmarking 2001:2::/48, ORCHIDv2 2001:20::/28. Teredo in
307        // particular embeds BOTH a server and an (obfuscated) client IPv4
308        // address; the whole /23 is non-global-unicast, so block it outright.
309        || (s[0] == 0x2001 && (s[1] & 0xfe00) == 0x0000)
310        || (s[0] == 0x2001 && s[1] == 0x0db8)   // 2001:db8::/32 documentation
311        || (s[0] == 0x3fff && (s[1] & 0xf000) == 0) // 3fff::/20 documentation (RFC 9637)
312}

314/// Returns `true` for IP addresses that must never be contacted from a
315/// server-side outbound fetch to an admin/IdP-supplied URL.
316///
317/// # SEC-094 — canonicalisation happens FIRST
318///
319/// The predicate this replaced matched on the `IpAddr` variant as it arrived
320/// from `getaddrinfo`. An `AAAA` record may legally contain an IPv4-mapped
321/// address (`::ffff:169.254.169.254`), whose `segments()[0]` is `0x0000`: it
322/// matched none of the v6 arms, and the v4 arms were never consulted because
323/// the value was an `IpAddr::V6`. The address then went straight into
324/// `ClientBuilder::resolve()`, so the pinning that closes the DNS-rebind
325/// window (D-01c) *guaranteed* the attacker's address was the one dialled —
326/// and on a dual-stack host `connect()` to `::ffff:a.b.c.d` reaches the IPv4
327/// destination.
328///
329/// Both IPv4-in-IPv6 embeddings are folded before classification:
330///
331/// * `::ffff:0:0/96` — IPv4-mapped. Legitimate: `getaddrinfo` with
332///   `AI_V4MAPPED` returns it for real v4 hosts, so it is *canonicalised*
333///   (checked as the v4 address it denotes) rather than rejected outright.
334/// * `::/96` — IPv4-compatible, deprecated by RFC 4291 §2.5.5.1. Rejected
335///   outright; nothing legitimate resolves to it. Note that
336///   `IpAddr::to_canonical` does **not** fold this form (it delegates to
337///   `to_ipv4_mapped`, not `to_ipv4`), which is why it is handled here by
338///   hand.
339pub fn is_disallowed_ip(ip: IpAddr) -> bool {
340    match ip {
341        IpAddr::V4(v4) => is_disallowed_ipv4(v4),
342        IpAddr::V6(v6) => match v6.to_ipv4_mapped() {
343            // ::ffff:a.b.c.d — classify as the v4 address it denotes.
344            Some(v4) => is_disallowed_ipv4(v4),
345            None => {
346                // ::/96 (IPv4-compatible, deprecated) — always rejected. `::`
347                // and `::1` fall in this range too and are disallowed anyway.
348                if is_ipv4_compatible_v6(v6) {
349                    return true;
350                }
351                is_disallowed_ipv6(v6)
352            }
353        },
354    }
355}

357/// `::/96` — the deprecated IPv4-compatible form, plus `::` and `::1`.
358fn is_ipv4_compatible_v6(v6: Ipv6Addr) -> bool {
359    let s = v6.segments();
360    s[0] == 0 && s[1] == 0 && s[2] == 0 && s[3] == 0 && s[4] == 0 && s[5] == 0
361}

363/// Resolve `host:port` (A + AAAA), reject if ANY resolved address is
364/// disallowed, and return ONE validated address to pin into the connection.
365///
366/// `allow_private`, when `true`, skips the disallow check entirely. This
367/// exists solely to preserve the pre-existing loopback mock-server
368/// integration-test seam (mirrors `JwksCache::new_allow_private_networks`);
369/// it MUST be `false` in production code paths.
370///
371/// SEC-107: a host the operator named through [`set_allowed_hosts`] is also
372/// exempt from the address rule — but only from the *address* rule, only on
373/// this hop, and never for a metadata endpoint ([`is_never_allowed`]). Every
374/// use of the exception is logged at WARN with the host and the address it
375/// resolved to, so a poisoned answer for an allowlisted name leaves a record
376/// rather than passing silently.
377pub async fn resolve_and_pick(
378    host: &str,
379    port: u16,
380    allow_private: bool,
381) -> Result<IpAddr, SsrfError> {
382    let addrs: Vec<IpAddr> = tokio::net::lookup_host((host, port))
383        .await
384        .map_err(|_| SsrfError::ResolveFailed)?
385        .map(|a| a.ip())
386        .collect();
387
388    if addrs.is_empty() {
389        return Err(SsrfError::ResolveFailed);
390    }
391
392    if allow_private {
393        return Ok(addrs[0]);
394    }
395
396    if let Some(bad) = addrs.iter().find(|ip| is_disallowed_ip(**ip)) {
397        // The exception is evaluated only once the strict rule has already
398        // said no, so an allowlisted host that resolves to a public address
399        // costs nothing and logs nothing.
400        if !host_is_allowlisted(host) {
401            return Err(SsrfError::Blocked);
402        }
403        if let Some(forbidden) = addrs.iter().find(|ip| is_never_allowed(**ip)) {
404            tracing::error!(
405                host,
406                address = %forbidden,
407                "SSRF: {host} is on {ALLOWED_HOSTS_ENV} but resolved to a cloud metadata \
408                 endpoint; refusing. An allowlist entry exempts a host from the \
409                 private-address rule, never from this one — treat this as a possible \
410                 DNS poisoning attempt against a trusted name."
411            );
412            return Err(SsrfError::Blocked);
413        }
414        tracing::warn!(
415            host,
416            address = %bad,
417            "SSRF: allowing a non-routable address because {host} is named in \
418             {ALLOWED_HOSTS_ENV}"
419        );
420    }
421
422    Ok(addrs[0])
423}

425/// Build a fresh, single-use client pinned to `ip` for `host`.
426///
427/// No connection pooling/caching across requests — a new `Client` is built
428/// per guarded fetch (D-01c: "fresh per request", so a rebind between two
429/// calls minutes apart can never reuse a stale pinned connection). Automatic
430/// redirect following is disabled (D-01b) — [`guarded_fetch`] re-validates
431/// and re-issues each hop explicitly instead of trusting `reqwest` to follow
432/// a `Location` header unchecked.
433pub fn pinned_client(host: &str, ip: IpAddr, port: u16) -> Result<reqwest::Client, SsrfError> {
434    reqwest::Client::builder()
435        .resolve(host, SocketAddr::new(ip, port))
436        .redirect(reqwest::redirect::Policy::none())
437        .timeout(Duration::from_secs(10))
438        .build()
439        .map_err(|_| SsrfError::ClientBuildFailed)
440}

442/// Orchestrates resolve + pin + fetch + bounded manual redirect
443/// re-validation (D-01b), using the default [`MAX_RESPONSE_BYTES`] cap.
444///
445/// `allow_private` is honored only for the first hop — see the module docs
446/// for why redirect targets are always strictly validated regardless of the
447/// caller's test-seam opt-in.
448///
449/// `build_request` builds the actual request (e.g. `|c, u| c.get(u)` or
450/// `|c, u| c.post(u).form(&params)`) against the freshly pinned client for
451/// the current hop's URL.
452pub async fn guarded_fetch(
453    url: &str,
454    allow_private: bool,
455    build_request: impl Fn(&reqwest::Client, &str) -> reqwest::RequestBuilder,
456) -> Result<reqwest::Response, SsrfError> {
457    guarded_fetch_with_cap(url, allow_private, MAX_RESPONSE_BYTES, build_request).await
458}

460/// Same as [`guarded_fetch`], but with an explicit Content-Length cap
461/// instead of the hard-coded default [`MAX_RESPONSE_BYTES`] (SEC-069).
462///
463/// Added for the FIDO MDS3 BLOB fetch path (D2/D10, X3): the BLOB is
464/// legitimately ~10 MB, well over the 5 MiB default every other guarded
465/// fetch type (JWKS, OIDC discovery, token, SAML metadata, webhook
466/// delivery) uses. This does **not** change the cap for any of those
467/// existing callers — [`guarded_fetch`] still always uses
468/// `MAX_RESPONSE_BYTES`; only a caller that deliberately opts in by naming
469/// its own cap (e.g. `axiam_pki::mds::MDS_MAX_BLOB_BYTES`) gets a larger one.
470pub async fn guarded_fetch_with_cap(
471    url: &str,
472    allow_private: bool,
473    max_response_bytes: usize,
474    build_request: impl Fn(&reqwest::Client, &str) -> reqwest::RequestBuilder,
475) -> Result<reqwest::Response, SsrfError> {
476    let mut current = url.to_string();
477
478    for hop in 0..MAX_HOPS {
479        let parsed = url::Url::parse(&current).map_err(|_| SsrfError::InvalidUrl)?;
480        let host = parsed.host_str().ok_or(SsrfError::InvalidUrl)?.to_string();
481        let port = parsed.port_or_known_default().unwrap_or(443);
482
483        // Only the first hop honors the caller's test seam; every redirect
484        // hop thereafter is always strictly validated (module docs above).
485        let hop_allow_private = allow_private && hop == 0;
486
487        // SEC-069: enforce HTTPS for every hop. A plaintext `http://` IdP
488        // endpoint (admin-misconfigured, or an attacker-supplied redirect)
489        // would carry the decrypted client_secret / bearer material in the
490        // clear. `http` is tolerated only behind the private-network test seam
491        // (loopback/dev), exactly like the address checks.
492        if parsed.scheme() != "https" && !hop_allow_private {
493            return Err(SsrfError::InsecureScheme);
494        }
495
496        let ip = resolve_and_pick(&host, port, hop_allow_private).await?;
497        let client = pinned_client(&host, ip, port)?;
498
499        let resp = build_request(&client, &current)
500            .send()
501            .await
502            .map_err(|e| SsrfError::RequestFailed(e.to_string()))?;
503
504        if resp.status().is_redirection() {
505            let location = resp
506                .headers()
507                .get("location")
508                .and_then(|v| v.to_str().ok())
509                .ok_or(SsrfError::InvalidUrl)?
510                .to_string();
511            current = parsed
512                .join(&location)
513                .map_err(|_| SsrfError::InvalidUrl)?
514                .to_string();
515            continue;
516        }
517
518        // SEC-069: reject an over-large advertised body before the caller
519        // buffers it. This is the coarse Content-Length gate; body readers that
520        // need a hard guarantee against a lying/chunked response still apply
521        // their own streaming cap (JWKS: 512 KiB).
522        if let Some(len) = resp.content_length()
523            && len > max_response_bytes as u64
524        {
525            return Err(SsrfError::ResponseTooLarge(max_response_bytes));
526        }
527
528        return Ok(resp);
529    }
530
531    Err(SsrfError::TooManyRedirects)
532}
533
534/// Read a response body with a hard streaming cap, aborting as soon as `cap`
535/// is exceeded — WITHOUT buffering the rest of the body first (CQ-B23).
536///
537/// This replaces the previous "buffer the whole body via `.bytes()`, then
538/// check `.len()` against the cap" pattern used by discovery/token-exchange
539/// reads. That pattern still let a malicious or misconfigured endpoint force
540/// full in-memory buffering of an arbitrarily large response before the
541/// existing size check ever ran (the coarse `Content-Length`-based check in
542/// [`guarded_fetch`] above only catches endpoints that both send the header
543/// AND tell the truth about it — a chunked, no-`Content-Length` response
544/// bypasses it entirely). Reading chunk-by-chunk with a running byte count
545/// bounds peak memory use to ~`cap` bytes regardless.
546///
547/// Uses `reqwest::Response::chunk()` (always available, unlike
548/// `bytes_stream()` which needs the `stream` cargo feature this workspace
549/// does not enable) so no new dependency/feature is required.
550pub async fn read_capped_body(
551    mut response: reqwest::Response,
552    cap: usize,
553) -> Result<Vec<u8>, SsrfError> {
554    let mut buf = Vec::with_capacity(cap.min(64 * 1024));
555    while let Some(chunk) = response
556        .chunk()
557        .await
558        .map_err(|e| SsrfError::RequestFailed(e.to_string()))?
559    {
560        buf.extend_from_slice(&chunk);
561        if buf.len() > cap {
562            return Err(SsrfError::ResponseTooLarge(cap));
563        }
564    }
565    Ok(buf)
566}

1041    /// CQ-B23: a body exceeding the cap is rejected — via a chunked,
1042    /// no-`Content-Length` response so the ONLY thing that can catch it is
1043    /// the streaming running-byte-count check, not the coarse
1044    /// `Content-Length` gate in [`guarded_fetch`] (which this test bypasses
1045    /// by calling `read_capped_body` directly on a plain `reqwest` response).
1046    #[tokio::test]
1047    async fn read_capped_body_rejects_body_over_cap() {
1048        use tokio::io::{AsyncReadExt, AsyncWriteExt};
1049        use tokio::net::TcpListener;
1050
1051        const CAP: usize = 16;
1052
1053        let listener = TcpListener::bind("127.0.0.1:0")
1054            .await
1055            .expect("bind mock server");
1056        let addr = listener.local_addr().expect("local_addr");
1057
1058        tokio::spawn(async move {
1059            if let Ok((mut stream, _)) = listener.accept().await {
1060                let mut buf = [0u8; 1024];
1061                let _ = stream.read(&mut buf).await;
1062                // Chunked transfer-encoding, no Content-Length: a body far
1063                // larger than CAP, split across multiple chunks so the
1064                // reader must actually stream (not just look at one read).
1065                let _ = stream
1066                    .write_all(
1067                        b"HTTP/1.1 200 OK\r\nTransfer-Encoding: chunked\r\nConnection: close\r\n\r\n",
1068                    )
1069                    .await;
1070                let chunk = "x".repeat(32);
1071                for _ in 0..8 {
1072                    let framed = format!("{:x}\r\n{}\r\n", chunk.len(), chunk);
1073                    let _ = stream.write_all(framed.as_bytes()).await;
1074                }
1075                let _ = stream.write_all(b"0\r\n\r\n").await;
1076            }
1077        });
1078
1079        let url = format!("http://127.0.0.1:{}/big", addr.port());
1080        let response = reqwest::Client::new().get(&url).send().await.unwrap();
1081        let result = read_capped_body(response, CAP).await;
1082
1083        assert!(
1084            matches!(result, Err(SsrfError::ResponseTooLarge(CAP))),
1085            "expected ResponseTooLarge({CAP}), got: {result:?}"
1086        );
1087    }
