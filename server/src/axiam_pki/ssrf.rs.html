1//! Shared SSRF guard — resolve-once-and-pin outbound fetch helper.
2//!
3//! Generalizes the byte-identical guard logic previously duplicated in
4//! `jwks_cache::is_private_jwks_ip`/`validate_jwks_url` and
5//! `axiam-api-rest::webhook::is_private_ip`/`resolve_and_validate_host`
6//! (SECHRD-02 / D-01a DRY) into one reusable module, and — critically —
7//! adds IP **pinning**, which neither prior guard did: both validated the
8//! resolved `IpAddr` and then let `reqwest` re-resolve DNS independently at
9//! send time, leaving a DNS-rebind TOCTOU window open between validation and
10//! connect (D-01c closes this).
11//!
12//! Use [`guarded_fetch`] for every outbound fetch to an admin/IdP-supplied
13//! URL (JWKS, OIDC discovery, OIDC token exchange, SAML metadata, webhook
14//! delivery). It:
15//!
16//! 1. Resolves the host (A + AAAA) fresh — no cross-request DNS caching
17//!    (D-01c).
18//! 2. Rejects the fetch if ANY resolved address is non-globally-routable
19//!    (D-01a) — unless the caller opted into the `allow_private` test seam
20//!    (see below). [`is_disallowed_ip`] enumerates the families and, since
21//!    SEC-094, canonicalises the IPv4-in-IPv6 encodings first: an `AAAA`
22//!    record carrying `::ffff:169.254.169.254` used to pass this step and
23//!    then be *pinned* by step 3.
24//! 3. Pins the exact validated `IpAddr` into a fresh, single-use
25//!    `reqwest::Client` via `ClientBuilder::resolve()`, so the socket that
26//!    is actually opened is the one that was validated — not a second,
27//!    independently-resolved address (D-01c).
28//! 4. Disables `reqwest`'s automatic redirect following and instead
29//!    manually re-runs the FULL guard (resolve → validate → pin → send)
30//!    against the `Location` target, bounded to [`MAX_HOPS`] hops (D-01b).
31//!
32//! ## The `allow_private` test seam only applies to the first hop
33//!
34//! `allow_private` exists solely so integration tests can point a guarded
35//! fetch at a loopback mock server (mirrors the pre-existing
36//! `JwksCache::new_allow_private_networks` seam). It is honored **only for
37//! the very first hop** of [`guarded_fetch`] — every redirect hop after that
38//! is always validated with the strict (production) check, regardless of
39//! `allow_private`. A `Location` header is attacker-influenced response
40//! data, not the admin-configured URL the caller opted to trust; holding it
41//! to the same relaxed standard as the test seam would silently defeat the
42//! redirect-bypass defense this module exists to provide (D-01b: "re-run
43//! the full SSRF guard against the redirect target").
44

50/// Maximum number of redirect hops [`guarded_fetch`] will follow before
51/// giving up. Each hop re-runs the full guard against the `Location` target.
52const MAX_HOPS: u8 = 3;

54/// Errors produced by the shared SSRF guard.
55#[derive(Debug, thiserror::Error)]
56pub enum SsrfError {
57    #[error("invalid URL")]
58    InvalidUrl,
59    #[error("failed to resolve host")]
60    ResolveFailed,
61    #[error("SSRF blocked: resolved IP is private/loopback/link-local/unspecified")]
62    Blocked,
63    #[error("failed to build HTTP client")]
64    ClientBuildFailed,
65    #[error("HTTP request failed: {0}")]
66    RequestFailed(String),
67    #[error("too many redirects")]
68    TooManyRedirects,
69    #[error("SSRF blocked: non-HTTPS scheme not permitted for IdP fetches")]
70    InsecureScheme,
71    #[error("SSRF blocked: response body exceeds the {0}-byte cap")]
72    ResponseTooLarge(usize),
73}

75/// Maximum acceptable `Content-Length` for a guarded IdP response (SEC-069).
76/// Discovery/metadata/token/JWKS documents are small; a multi-GB body is a
77/// memory-exhaustion DoS vector. JWKS additionally applies its own 512 KiB
78/// read cap downstream; this is the coarse first line of defence for all four
79/// federation fetch types.
80const MAX_RESPONSE_BYTES: usize = 5 * 1024 * 1024;

93static ALLOWED_HOSTS: OnceLock<BTreeSet<String>> = OnceLock::new();

95/// Install the operator's SSRF host exception list (SEC-107).
96///
97/// # Why an exception exists at all
98///
99/// The address rule is a *network-topology* control being used as a *trust*
100/// control, and in real deployments the two diverge: a Kubernetes-internal
101/// Keycloak or Entra proxy at `10.x` is a perfectly legitimate IdP, and so is
102/// an internal webhook consumer, an internal MDS mirror, or an air-gapped
103/// deployment where *everything* is RFC1918. With no exception at all, an
104/// operator's only recourse is to run AXIAM outside the guard's assumptions or
105/// to patch the binary — and a guard that gets patched around protects nobody.
106/// It has already cost this project measurable work (R5.4's cross-vendor
107/// Keycloak test cannot use the containerized server for exactly this reason).
108///
109/// # Why it is a host list and not a switch or a CIDR range
110///
111/// A boolean `SSRF_ALLOW_PRIVATE=true` is the `verify_peer: false` of this
112/// module: it appears in a dev compose file, works, and travels unchanged into
113/// production. A CIDR exception can be *widened by a DNS answer* — allow
114/// `10.0.0.0/8` and any hostname an admin can set now reaches anything in it.
115/// A host exception cannot: the operator names the exact destination they
116/// intend to reach, and a DNS answer for some *other* host is still blocked.
117///
118/// # The four properties that keep it from being a hole
119///
120/// 1. **Default-empty.** Nothing is exempt unless this is called with a
121///    non-empty list. It is a `OnceLock`, so it is set once at composition and
122///    cannot be re-armed later by anything.
123/// 2. **Exact match only.** ASCII-lowercased equality — no wildcards, no
124///    suffix matching, no "ends with .internal". `*.corp` cannot be spelled.
125/// 3. **First hop only.** Redirect targets are always validated strictly, for
126///    the reason the module docs give about `allow_private`: a `Location`
127///    header is attacker-influenced response data, not the URL the operator
128///    chose to trust.
129/// 4. **The metadata services stay unreachable** — [`is_never_allowed`]. An
130///    operator asking for their `10.x` IdP is not asking for
131///    `169.254.169.254`, and an allowlisted host whose DNS is poisoned onto a
132///    metadata endpoint is exactly the attack this control would otherwise
133///    re-open. That includes SEC-094's IPv4-mapped spelling, because the
134///    canonicalisation runs before this check, not after it.
135///
136/// Returns the number of entries installed. Calling it twice is a no-op on the
137/// second call (the `OnceLock` keeps the first list), which is deliberate:
138/// re-arming a security exception at runtime is not a feature.
139pub fn set_allowed_hosts<I: IntoIterator<Item = String>>(hosts: I) -> usize {
140    let set: BTreeSet<String> = hosts
141        .into_iter()
142        .map(|h| h.trim().to_ascii_lowercase())
143        .filter(|h| !h.is_empty())
144        .collect();
145    let _ = ALLOWED_HOSTS.set(set);
146    allowed_hosts().len()
147}

149/// Parse the comma-separated [`ALLOWED_HOSTS_ENV`] form.
150pub fn parse_allowed_hosts(raw: &str) -> Vec<String> {
151    raw.split(',')
152        .map(|h| h.trim().to_ascii_lowercase())
153        .filter(|h| !h.is_empty())
154        .collect()
155}

157/// The installed exception list. Empty until [`set_allowed_hosts`] is called.
158pub fn allowed_hosts() -> &'static BTreeSet<String> {
159    static EMPTY: OnceLock<BTreeSet<String>> = OnceLock::new();
160    ALLOWED_HOSTS
161        .get()
162        .unwrap_or_else(|| EMPTY.get_or_init(BTreeSet::new))
163}

165/// Whether `host` is one the operator explicitly named (SEC-107).
166fn host_is_allowlisted(host: &str) -> bool {
167    let set = allowed_hosts();
168    !set.is_empty() && set.contains(&host.trim().to_ascii_lowercase())
169}

171// The address classification — which families are loopback, link-local,
172// private, special-purpose or global, SEC-094's canonicalisation of the
173// IPv4-in-IPv6 encodings, and SEC-107's never-allowed metadata endpoints — lives
174// in `axiam_core::ip_class` since T23.3.7, so the directory connector's address
175// guard (T-300) classifies exactly as this one does. The policy stays here.
176// `is_disallowed_ip` is re-exported because callers outside this crate (the
177// webhook tests among them) have always reached it as `ssrf::is_disallowed_ip`.
178pub use axiam_core::ip_class::is_disallowed_ip;
179use axiam_core::ip_class::is_never_allowed;

181/// Resolve `host:port` (A + AAAA), reject if ANY resolved address is
182/// disallowed, and return ONE validated address to pin into the connection.
183///
184/// `allow_private`, when `true`, skips the disallow check entirely. This
185/// exists solely to preserve the pre-existing loopback mock-server
186/// integration-test seam (mirrors `JwksCache::new_allow_private_networks`);
187/// it MUST be `false` in production code paths.
188///
189/// SEC-107: a host the operator named through [`set_allowed_hosts`] is also
190/// exempt from the address rule — but only from the *address* rule, only on
191/// this hop, and never for a metadata endpoint ([`is_never_allowed`]). Every
192/// use of the exception is logged at WARN with the host and the address it
193/// resolved to, so a poisoned answer for an allowlisted name leaves a record
194/// rather than passing silently.
195pub async fn resolve_and_pick(
196    host: &str,
197    port: u16,
198    allow_private: bool,
199) -> Result<IpAddr, SsrfError> {
200    let addrs: Vec<IpAddr> = tokio::net::lookup_host((host, port))
201        .await
202        .map_err(|_| SsrfError::ResolveFailed)?
203        .map(|a| a.ip())
204        .collect();
205
206    if addrs.is_empty() {
207        return Err(SsrfError::ResolveFailed);
208    }
209
210    if allow_private {
211        return Ok(addrs[0]);
212    }
213
214    if let Some(bad) = addrs.iter().find(|ip| is_disallowed_ip(**ip)) {
215        // The exception is evaluated only once the strict rule has already
216        // said no, so an allowlisted host that resolves to a public address
217        // costs nothing and logs nothing.
218        if !host_is_allowlisted(host) {
219            return Err(SsrfError::Blocked);
220        }
221        if let Some(forbidden) = addrs.iter().find(|ip| is_never_allowed(**ip)) {
222            tracing::error!(
223                host,
224                address = %forbidden,
225                "SSRF: {host} is on {ALLOWED_HOSTS_ENV} but resolved to a cloud metadata \
226                 endpoint; refusing. An allowlist entry exempts a host from the \
227                 private-address rule, never from this one — treat this as a possible \
228                 DNS poisoning attempt against a trusted name."
229            );
230            return Err(SsrfError::Blocked);
231        }
232        tracing::warn!(
233            host,
234            address = %bad,
235            "SSRF: allowing a non-routable address because {host} is named in \
236             {ALLOWED_HOSTS_ENV}"
237        );
238    }
239
240    Ok(addrs[0])
241}

243/// Build a fresh, single-use client pinned to `ip` for `host`.
244///
245/// No connection pooling/caching across requests — a new `Client` is built
246/// per guarded fetch (D-01c: "fresh per request", so a rebind between two
247/// calls minutes apart can never reuse a stale pinned connection). Automatic
248/// redirect following is disabled (D-01b) — [`guarded_fetch`] re-validates
249/// and re-issues each hop explicitly instead of trusting `reqwest` to follow
250/// a `Location` header unchecked.
251pub fn pinned_client(host: &str, ip: IpAddr, port: u16) -> Result<reqwest::Client, SsrfError> {
252    reqwest::Client::builder()
253        .resolve(host, SocketAddr::new(ip, port))
254        .redirect(reqwest::redirect::Policy::none())
255        .timeout(Duration::from_secs(10))
256        .build()
257        .map_err(|_| SsrfError::ClientBuildFailed)
258}

260/// Orchestrates resolve + pin + fetch + bounded manual redirect
261/// re-validation (D-01b), using the default [`MAX_RESPONSE_BYTES`] cap.
262///
263/// `allow_private` is honored only for the first hop — see the module docs
264/// for why redirect targets are always strictly validated regardless of the
265/// caller's test-seam opt-in.
266///
267/// `build_request` builds the actual request (e.g. `|c, u| c.get(u)` or
268/// `|c, u| c.post(u).form(&params)`) against the freshly pinned client for
269/// the current hop's URL.
270pub async fn guarded_fetch(
271    url: &str,
272    allow_private: bool,
273    build_request: impl Fn(&reqwest::Client, &str) -> reqwest::RequestBuilder,
274) -> Result<reqwest::Response, SsrfError> {
275    guarded_fetch_with_cap(url, allow_private, MAX_RESPONSE_BYTES, build_request).await
276}

278/// Same as [`guarded_fetch`], but with an explicit Content-Length cap
279/// instead of the hard-coded default [`MAX_RESPONSE_BYTES`] (SEC-069).
280///
281/// Added for the FIDO MDS3 BLOB fetch path (D2/D10, X3): the BLOB is
282/// legitimately ~10 MB, well over the 5 MiB default every other guarded
283/// fetch type (JWKS, OIDC discovery, token, SAML metadata, webhook
284/// delivery) uses. This does **not** change the cap for any of those
285/// existing callers — [`guarded_fetch`] still always uses
286/// `MAX_RESPONSE_BYTES`; only a caller that deliberately opts in by naming
287/// its own cap (e.g. `axiam_pki::mds::MDS_MAX_BLOB_BYTES`) gets a larger one.
288pub async fn guarded_fetch_with_cap(
289    url: &str,
290    allow_private: bool,
291    max_response_bytes: usize,
292    build_request: impl Fn(&reqwest::Client, &str) -> reqwest::RequestBuilder,
293) -> Result<reqwest::Response, SsrfError> {
294    let mut current = url.to_string();
295
296    for hop in 0..MAX_HOPS {
297        let parsed = url::Url::parse(&current).map_err(|_| SsrfError::InvalidUrl)?;
298        let host = parsed.host_str().ok_or(SsrfError::InvalidUrl)?.to_string();
299        let port = parsed.port_or_known_default().unwrap_or(443);
300
301        // Only the first hop honors the caller's test seam; every redirect
302        // hop thereafter is always strictly validated (module docs above).
303        let hop_allow_private = allow_private && hop == 0;
304
305        // SEC-069: enforce HTTPS for every hop. A plaintext `http://` IdP
306        // endpoint (admin-misconfigured, or an attacker-supplied redirect)
307        // would carry the decrypted client_secret / bearer material in the
308        // clear. `http` is tolerated only behind the private-network test seam
309        // (loopback/dev), exactly like the address checks.
310        if parsed.scheme() != "https" && !hop_allow_private {
311            return Err(SsrfError::InsecureScheme);
312        }
313
314        let ip = resolve_and_pick(&host, port, hop_allow_private).await?;
315        let client = pinned_client(&host, ip, port)?;
316
317        let resp = build_request(&client, &current)
318            .send()
319            .await
320            .map_err(|e| SsrfError::RequestFailed(e.to_string()))?;
321
322        if resp.status().is_redirection() {
323            let location = resp
324                .headers()
325                .get("location")
326                .and_then(|v| v.to_str().ok())
327                .ok_or(SsrfError::InvalidUrl)?
328                .to_string();
329            current = parsed
330                .join(&location)
331                .map_err(|_| SsrfError::InvalidUrl)?
332                .to_string();
333            continue;
334        }
335
336        // SEC-069: reject an over-large advertised body before the caller
337        // buffers it. This is the coarse Content-Length gate; body readers that
338        // need a hard guarantee against a lying/chunked response still apply
339        // their own streaming cap (JWKS: 512 KiB).
340        if let Some(len) = resp.content_length()
341            && len > max_response_bytes as u64
342        {
343            return Err(SsrfError::ResponseTooLarge(max_response_bytes));
344        }
345
346        return Ok(resp);
347    }
348
349    Err(SsrfError::TooManyRedirects)
350}

352/// One guarded hop that **never follows a redirect** (G-5, T23.5.3, D-53 (9)).
353///
354/// The same resolve (A + AAAA, fresh), refusal of any non-global address, HTTPS
355/// requirement (waived only with `allow_private`, the test seam), IP pinning and
356/// `Content-Length` cap as [`guarded_fetch`], on the one URL it is given. A `3xx`
357/// is **returned to the caller as a response** — its `Location` is not resolved,
358/// not validated and not fetched — so a caller that must not send a body or a
359/// credential anywhere but the address it was given (an SSF push, whose
360/// `Authorization` header would otherwise follow a redirect) can refuse a
361/// redirect by construction instead of by inspecting a later hop.
362///
363/// [`guarded_fetch`] and [`guarded_fetch_with_cap`] are unchanged.
364pub async fn guarded_fetch_no_redirect(
365    url: &str,
366    allow_private: bool,
367    build_request: impl Fn(&reqwest::Client, &str) -> reqwest::RequestBuilder,
368) -> Result<reqwest::Response, SsrfError> {
369    let parsed = url::Url::parse(url).map_err(|_| SsrfError::InvalidUrl)?;
370    let host = parsed.host_str().ok_or(SsrfError::InvalidUrl)?.to_string();
371    let port = parsed.port_or_known_default().unwrap_or(443);
372
373    // SEC-069, as on the first hop of `guarded_fetch`.
374    if parsed.scheme() != "https" && !allow_private {
375        return Err(SsrfError::InsecureScheme);
376    }
377
378    let ip = resolve_and_pick(&host, port, allow_private).await?;
379    let client = pinned_client(&host, ip, port)?;
380    let resp = build_request(&client, url)
381        .send()
382        .await
383        .map_err(|e| SsrfError::RequestFailed(e.to_string()))?;
384
385    if let Some(len) = resp.content_length()
386        && len > MAX_RESPONSE_BYTES as u64
387    {
388        return Err(SsrfError::ResponseTooLarge(MAX_RESPONSE_BYTES));
389    }
390    Ok(resp)
391}

393/// Read a response body with a hard streaming cap, aborting as soon as `cap`
394/// is exceeded — WITHOUT buffering the rest of the body first (CQ-B23).
395///
396/// This replaces the previous "buffer the whole body via `.bytes()`, then
397/// check `.len()` against the cap" pattern used by discovery/token-exchange
398/// reads. That pattern still let a malicious or misconfigured endpoint force
399/// full in-memory buffering of an arbitrarily large response before the
400/// existing size check ever ran (the coarse `Content-Length`-based check in
401/// [`guarded_fetch`] above only catches endpoints that both send the header
402/// AND tell the truth about it — a chunked, no-`Content-Length` response
403/// bypasses it entirely). Reading chunk-by-chunk with a running byte count
404/// bounds peak memory use to ~`cap` bytes regardless.
405///
406/// Uses `reqwest::Response::chunk()` (always available, unlike
407/// `bytes_stream()` which needs the `stream` cargo feature this workspace
408/// does not enable) so no new dependency/feature is required.
409pub async fn read_capped_body(
410    mut response: reqwest::Response,
411    cap: usize,
412) -> Result<Vec<u8>, SsrfError> {
413    let mut buf = Vec::with_capacity(cap.min(64 * 1024));
414    while let Some(chunk) = response
415        .chunk()
416        .await
417        .map_err(|e| SsrfError::RequestFailed(e.to_string()))?
418    {
419        buf.extend_from_slice(&chunk);
420        if buf.len() > cap {
421            return Err(SsrfError::ResponseTooLarge(cap));
422        }
423    }
424    Ok(buf)
425}

642    /// D2/D10 (X3): `guarded_fetch_with_cap` accepts a response whose
643    /// advertised Content-Length exceeds the default [`MAX_RESPONSE_BYTES`]
644    /// but is within the caller-supplied cap — proving the MDS BLOB fetch
645    /// path (~10 MB, opting into a 32 MiB cap) is not silently rejected by
646    /// the coarse gate the default-cap `guarded_fetch` would apply.
647    #[tokio::test]
648    async fn guarded_fetch_with_cap_allows_body_over_default_cap_but_within_custom_cap() {
649        use tokio::io::{AsyncReadExt, AsyncWriteExt};
650        use tokio::net::TcpListener;
651
652        let listener = TcpListener::bind("127.0.0.1:0")
653            .await
654            .expect("bind mock server");
655        let addr = listener.local_addr().expect("local_addr");
656        // Bigger than MAX_RESPONSE_BYTES (5 MiB) would allow via `guarded_fetch`.
657        let big_len = MAX_RESPONSE_BYTES + 1024;
658
659        tokio::spawn(async move {
660            if let Ok((mut stream, _)) = listener.accept().await {
661                let mut buf = [0u8; 1024];
662                let _ = stream.read(&mut buf).await;
663                // Only the headers matter for this test — `guarded_fetch_with_cap`
664                // decides from Content-Length before the body is ever read.
665                let response = format!(
666                    "HTTP/1.1 200 OK\r\nContent-Length: {big_len}\r\nConnection: close\r\n\r\n"
667                );
668                let _ = stream.write_all(response.as_bytes()).await;
669            }
670        });
671
672        let url = format!("http://127.0.0.1:{}/big", addr.port());
673
674        // Default cap: rejected.
675        let default_result = guarded_fetch(&url, true, |c, u| c.get(u)).await;
676        assert!(
677            matches!(default_result, Err(SsrfError::ResponseTooLarge(cap)) if cap == MAX_RESPONSE_BYTES),
678            "expected the default cap to reject a body this large, got: {default_result:?}"
679        );
680
681        // MDS-sized cap: accepted (the coarse Content-Length gate passes;
682        // the connection is dropped afterward so `.send()` may itself race
683        // with `Connection: close`, which is irrelevant to what this test
684        // is asserting — that the cap comparison itself used our larger
685        // value, not the default).
686        let mds_cap = MAX_RESPONSE_BYTES + 2 * 1024 * 1024;
687        let capped_result = guarded_fetch_with_cap(&url, true, mds_cap, |c, u| c.get(u)).await;
688        assert!(
689            !matches!(capped_result, Err(SsrfError::ResponseTooLarge(_))),
690            "expected a body within the custom cap to pass the Content-Length gate, got: {capped_result:?}"
691        );
692    }

726    /// CQ-B23: a body exceeding the cap is rejected — via a chunked,
727    /// no-`Content-Length` response so the ONLY thing that can catch it is
728    /// the streaming running-byte-count check, not the coarse
729    /// `Content-Length` gate in [`guarded_fetch`] (which this test bypasses
730    /// by calling `read_capped_body` directly on a plain `reqwest` response).
731    #[tokio::test]
732    async fn read_capped_body_rejects_body_over_cap() {
733        use tokio::io::{AsyncReadExt, AsyncWriteExt};
734        use tokio::net::TcpListener;
735
736        const CAP: usize = 16;
737
738        let listener = TcpListener::bind("127.0.0.1:0")
739            .await
740            .expect("bind mock server");
741        let addr = listener.local_addr().expect("local_addr");
742
743        tokio::spawn(async move {
744            if let Ok((mut stream, _)) = listener.accept().await {
745                let mut buf = [0u8; 1024];
746                let _ = stream.read(&mut buf).await;
747                // Chunked transfer-encoding, no Content-Length: a body far
748                // larger than CAP, split across multiple chunks so the
749                // reader must actually stream (not just look at one read).
750                let _ = stream
751                    .write_all(
752                        b"HTTP/1.1 200 OK\r\nTransfer-Encoding: chunked\r\nConnection: close\r\n\r\n",
753                    )
754                    .await;
755                let chunk = "x".repeat(32);
756                for _ in 0..8 {
757                    let framed = format!("{:x}\r\n{}\r\n", chunk.len(), chunk);
758                    let _ = stream.write_all(framed.as_bytes()).await;
759                }
760                let _ = stream.write_all(b"0\r\n\r\n").await;
761            }
762        });
763
764        let url = format!("http://127.0.0.1:{}/big", addr.port());
765        let response = reqwest::Client::new().get(&url).send().await.unwrap();
766        let result = read_capped_body(response, CAP).await;
767
768        assert!(
769            matches!(result, Err(SsrfError::ResponseTooLarge(CAP))),
770            "expected ResponseTooLarge({CAP}), got: {result:?}"
771        );
772    }
