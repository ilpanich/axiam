1//! The retry policy of the Vault revocation sweep (T-470, #603).
2//!
3//! A revocation Vault refuses is not retried at every tick: each refusal pushes
4//! the row's next attempt out (1 minute, doubling, to 6 hours), a replica claims
5//! a row before it attempts it, and a row that keeps failing past
6//! [`FAILING_AFTER_ATTEMPTS`] is reported as its own job-health error. The
7//! constants and the two pure functions live here, apart from the sweep, so the
8//! next consumer of the same bookkeeping (a CA certificate's revocation) does
9//! not copy them.
10
11use std::time::Duration;
12
13/// The wait after a row's first refusal.
14pub const BACKOFF_BASE: Duration = Duration::from_secs(60);
15
16/// The longest wait between two attempts at one row.
17pub const BACKOFF_CAP: Duration = Duration::from_secs(6 * 60 * 60);
18
19/// Refusals after which a row counts as failing for good, and `/health/jobs`
20/// reports it (as `vault_revocation_failing`) instead of failing the pass that
21/// retries it. Twelve attempts span about a day: the first nine back off from
22/// one minute to a little over four hours, the rest wait the six-hour cap.
23pub const FAILING_AFTER_ATTEMPTS: u32 = 12;
24
25/// How long a replica holds a claimed row. Longer than one sweep pass
26/// ([`crate::cert`] bounds it to 30 seconds), so a row being attempted is not
27/// due for anyone else; a replica that dies mid-attempt loses the row for this
28/// long and no longer, with nothing counted against the row.
29pub const CLAIM_LEASE: Duration = Duration::from_secs(120);
30
31/// The longest reason a row keeps, in characters: a status and a short message.
32pub const MAX_REASON_CHARS: usize = 256;
33
34/// How long to wait before the next attempt at a row that has now failed
35/// `attempts` times (1 for the first refusal): [`BACKOFF_BASE`] doubled for each
36/// earlier refusal, capped at [`BACKOFF_CAP`].
37pub fn backoff(attempts: u32) -> Duration {
38    let doublings = attempts.saturating_sub(1).min(20);
39    BACKOFF_BASE
40        .saturating_mul(1u32 << doublings)
41        .min(BACKOFF_CAP)
42}
43
44/// A custodian's error as a row may keep it: control characters and runs of
45/// whitespace collapsed, anything that reads like a Vault token replaced, and
46/// the whole cut to [`MAX_REASON_CHARS`].
47///
48/// The stored reason is read by operators and may reach a log or an export, so
49/// it carries neither a credential nor a response body. The Vault client never
50/// puts its token in an error; this is the second line, for a message a server
51/// echoed back or a future change.
52pub fn sanitise_reason(raw: &str) -> String {
53    let words: Vec<&str> = raw.split_whitespace().collect();
54    let mut out = String::new();
55    for word in words {
56        if !out.is_empty() {
57            out.push(' ');
58        }
59        let word: String = word.chars().filter(|c| !c.is_control()).collect();
60        out.push_str(&redact_tokens(&word));
61    }
62    if out.chars().count() > MAX_REASON_CHARS {
63        let mut cut: String = out.chars().take(MAX_REASON_CHARS - 1).collect();
64        cut.push('…');
65        return cut;
66    }
67    out
68}
69
70/// The shortest body a token must have after its prefix.
71const MIN_TOKEN_BODY: usize = 8;
72
73/// `word` with every Vault token in it replaced, the token's span only.
74///
75/// A token starts at any position whose previous character is not a token
76/// character (`=`, `"`, `:`, `'`, or the start), so `token=hvs.…` and
77/// `{"token":"hvs.…"}` are masked and the text around them is kept.
78fn redact_tokens(word: &str) -> String {
79    let chars: Vec<char> = word.chars().collect();
80    let mut out = String::with_capacity(word.len());
81    let mut i = 0;
82    while i < chars.len() {
83        let at_boundary = i == 0 || !is_token_char(chars[i - 1]);
84        match at_boundary.then(|| token_len(&chars[i..])).flatten() {
85            Some(len) => {
86                out.push_str("[redacted]");
87                i += len;
88            }
89            None => {
90                out.push(chars[i]);
91                i += 1;
92            }
93        }
94    }
95    out
96}

98/// The length of the Vault token that `s` starts with, if it starts with one.
99///
100/// `hvs.`, `hvb.` and `hvr.` take a body of token characters (which include
101/// `.`: the modern body is base64url and may carry the separator). The older
102/// `s.` (service) and `b.` (batch) take a body of base62/base64url characters
103/// without `.`, so a host name such as `b.vault-internal.example.com` is not a
104/// token. A `.` that ends the match is punctuation and stays out of the span.
105fn token_len(s: &[char]) -> Option<usize> {
106    let starts =
107        |prefix: &str| prefix.len() <= s.len() && prefix.chars().zip(s).all(|(p, c)| p == *c);
108    let run = |from: usize, accept: fn(char) -> bool| {
109        s[from..].iter().take_while(|c| accept(**c)).count()
110    };
111    if ["hvs.", "hvb.", "hvr."].iter().any(|p| starts(p)) {
112        let body = s[4..].iter().take_while(|c| is_token_char(**c)).count();
113        let body = s[4..4 + body]
114            .iter()
115            .rposition(|c| *c != '.')
116            .map_or(0, |last| last + 1);
117        return (body >= MIN_TOKEN_BODY).then_some(4 + body);
118    }
119    if starts("s.") || starts("b.") {
120        let body = run(2, is_legacy_body_char);
121        let continues = s.get(2 + body) == Some(&'.')
122            && s.get(3 + body)
123                .is_some_and(|c| is_token_char(*c) && *c != '.');
124        return (body >= MIN_TOKEN_BODY && !continues).then_some(2 + body);
125    }
126    None
127}
128
129fn is_token_char(c: char) -> bool {
130    c.is_ascii_alphanumeric() || c == '_' || c == '-' || c == '.'
131}
132
133fn is_legacy_body_char(c: char) -> bool {
134    is_token_char(c) && c != '.'
135}

137#[cfg(test)]
138mod tests {

141    /// `prefix` and a low-entropy body of `12 * parts` token characters, built
142    /// at run time: a literal looks like a credential to secret scanners.
143    fn fake_token(prefix: &str, parts: usize) -> String {
144        format!(
145            "{prefix}{}",
146            ["fake", "body", "0123"].concat().repeat(parts)
147        )
148    }
149
150    #[test]
151    fn backoff_doubles_from_a_minute_to_the_cap() {
152        let minutes: Vec<u64> = (1..=11).map(|n| backoff(n).as_secs() / 60).collect();
153        assert_eq!(minutes, [1, 2, 4, 8, 16, 32, 64, 128, 256, 360, 360]);
154        assert_eq!(backoff(0), BACKOFF_BASE, "no failure yet waits the base");
155        assert_eq!(
156            backoff(u32::MAX),
157            BACKOFF_CAP,
158            "a huge count cannot overflow"
159        );
160    }
161
162    #[test]
163    fn a_reason_is_one_bounded_line_without_a_token() {
164        let reason = sanitise_reason(&format!(
165            "vault pki: http://v:8200/v1/pki/revoke answered 403 Forbidden:\n\tpermission denied \
166             for {}",
167            fake_token("hvs.", 3)
168        ));
169        assert!(!reason.contains('\n') && !reason.contains('\t'));
170        assert!(reason.contains("403 Forbidden") && reason.contains("permission denied"));
171        assert!(!reason.contains("hvs."), "{reason}");
172        assert!(reason.contains("[redacted]"));
173
174        let long = sanitise_reason(&"x".repeat(10_000));
175        assert_eq!(long.chars().count(), MAX_REASON_CHARS);
176        // Multi-byte text is cut on a character, not a byte.
177        let wide = sanitise_reason(&"é".repeat(1_000));
178        assert_eq!(wide.chars().count(), MAX_REASON_CHARS);
179        // Ordinary words that merely start like a prefix are left alone.
180        assert_eq!(sanitise_reason("s.o.s. hvs.x"), "s.o.s. hvs.x");
181        assert_eq!(sanitise_reason("b.c. b.short"), "b.c. b.short");
182    }

184    /// #665 (R110W1-05): a token is masked wherever a non-token character
185    /// precedes it, the span only; a host name that starts like a legacy
186    /// prefix is left alone.
187    #[test]
188    fn a_token_is_masked_at_any_boundary_and_the_text_around_it_stays() {
189        let tok = fake_token("hvs.", 3);
190        let tok = tok.as_str();
191        assert_eq!(sanitise_reason(&format!("token={tok}")), "token=[redacted]");
192        assert_eq!(
193            sanitise_reason(&format!("{{\"token\":\"{tok}\"}}")),
194            "{\"token\":\"[redacted]\"}"
195        );
196        assert_eq!(sanitise_reason(&format!("\"{tok}\".")), "\"[redacted]\".");
197        assert_eq!(sanitise_reason(&format!("'{tok}'.")), "'[redacted]'.");
198        assert_eq!(sanitise_reason(&format!("x:{tok}.")), "x:[redacted].");
199        assert_eq!(
200            sanitise_reason(&format!("token={},", fake_token("b.", 1))),
201            "token=[redacted],"
202        );
203        // Preceded by a token character it is part of a longer word.
204        assert_eq!(sanitise_reason(&format!("abc{tok}")), format!("abc{tok}"));
205
206        assert_eq!(
207            sanitise_reason("b.vault-internal.example.com"),
208            "b.vault-internal.example.com"
209        );
210        assert_eq!(
211            sanitise_reason("s.vault-1.eu-west.internal"),
212            "s.vault-1.eu-west.internal"
213        );
214    }

216    /// R101W1-06: a legacy batch token (`b.` and a body) is masked as `s.` is.
217    #[test]
218    fn a_legacy_batch_token_is_masked() {
219        let (long, short) = (fake_token("b.", 2), fake_token("b.", 1));
220        let reason = sanitise_reason(&format!("denied for {long} and \"{short}\""));
221        assert!(!reason.contains(&short), "{reason}");
222        assert_eq!(reason.matches("[redacted]").count(), 2, "{reason}");
223        assert!(reason.starts_with("denied for "));
224    }

226    /// The row, the audit append and the log each apply the sanitiser to what
227    /// the sweep or the handler already sanitised: a second pass changes nothing.
228    #[test]
229    fn sanitising_is_idempotent() {
230        let tokens = format!(
231            "denied for {} {}",
232            fake_token("hvs.", 3),
233            fake_token("b.", 1)
234        );
235        for raw in [&tokens, &"y ".repeat(5_000), &"é".repeat(1_000)] {
236            let once = sanitise_reason(raw);
237            assert_eq!(sanitise_reason(&once), once);
238        }
239    }
