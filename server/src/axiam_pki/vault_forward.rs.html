1//! The retry policy of the Vault revocation sweep (T-470, #603).
2//!
3//! A revocation Vault refuses is not retried at every tick: each refusal pushes
4//! the row's next attempt out (1 minute, doubling, to 6 hours), a replica claims
5//! a row before it attempts it, and a row that keeps failing past
6//! [`FAILING_AFTER_ATTEMPTS`] is reported as its own job-health error. The
7//! constants and the two pure functions live here, apart from the sweep, so the
8//! next consumer of the same bookkeeping (a CA certificate's revocation) does
9//! not copy them.
10
11use std::time::Duration;
12
13/// The wait after a row's first refusal.
14pub const BACKOFF_BASE: Duration = Duration::from_secs(60);
15
16/// The longest wait between two attempts at one row.
17pub const BACKOFF_CAP: Duration = Duration::from_secs(6 * 60 * 60);
18
19/// Refusals after which a row counts as failing for good, and `/health/jobs`
20/// reports it (as `vault_revocation_failing`) instead of failing the pass that
21/// retries it. Twelve attempts span about a day: the first nine back off from
22/// one minute to a little over four hours, the rest wait the six-hour cap.
23pub const FAILING_AFTER_ATTEMPTS: u32 = 12;
24
25/// How long a replica holds a claimed row. Longer than one sweep pass
26/// ([`crate::cert`] bounds it to 30 seconds), so a row being attempted is not
27/// due for anyone else; a replica that dies mid-attempt loses the row for this
28/// long and no longer, with nothing counted against the row.
29pub const CLAIM_LEASE: Duration = Duration::from_secs(120);
30
31/// The longest reason a row keeps, in characters: a status and a short message.
32pub const MAX_REASON_CHARS: usize = 256;
33
34/// How long to wait before the next attempt at a row that has now failed
35/// `attempts` times (1 for the first refusal): [`BACKOFF_BASE`] doubled for each
36/// earlier refusal, capped at [`BACKOFF_CAP`].
37pub fn backoff(attempts: u32) -> Duration {
38    let doublings = attempts.saturating_sub(1).min(20);
39    BACKOFF_BASE
40        .saturating_mul(1u32 << doublings)
41        .min(BACKOFF_CAP)
42}

44/// A custodian's error as a row may keep it: control characters and runs of
45/// whitespace collapsed, anything that reads like a Vault token replaced, and
46/// the whole cut to [`MAX_REASON_CHARS`].
47///
48/// The stored reason is read by operators and may reach a log or an export, so
49/// it carries neither a credential nor a response body. The Vault client never
50/// puts its token in an error; this is the second line, for a message a server
51/// echoed back or a future change.
52pub fn sanitise_reason(raw: &str) -> String {
53    let words: Vec<&str> = raw.split_whitespace().collect();
54    let mut out = String::new();
55    for word in words {
56        if !out.is_empty() {
57            out.push(' ');
58        }
59        let word: String = word.chars().filter(|c| !c.is_control()).collect();
60        out.push_str(&redact_tokens(&word));
61    }
62    if out.chars().count() > MAX_REASON_CHARS {
63        let mut cut: String = out.chars().take(MAX_REASON_CHARS - 1).collect();
64        cut.push('…');
65        return cut;
66    }
67    out
68}

70/// The shortest body a token must have after its prefix.
71const MIN_TOKEN_BODY: usize = 8;
72
73/// `word` with every Vault token in it replaced, the token's span only.
74///
75/// A token starts at any position whose previous character is not a token
76/// character (`=`, `"`, `:`, `'`, or the start), so `token=hvs.…` and
77/// `{"token":"hvs.…"}` are masked and the text around them is kept.
78///
79/// A modern (`hvs.`/`hvb.`/`hvr.`) token also starts right after a recognised
80/// escape (`\n`, `\t`, `\r`, `\"`, `\uXXXX`, `%XX`), whose last character is
81/// itself a token character: the escape is kept, the token masked. The legacy
82/// prefixes get no such boundary (T-470). Each position looks back at most
83/// six characters (`\uXXXX`) and a match is skipped whole, so the scan stays
84/// linear in the word: nothing is re-read after a mask, and a failed attempt
85/// stops at the first non-token character.
86fn redact_tokens(word: &str) -> String {
87    let chars: Vec<char> = word.chars().collect();
88    let mut out = String::with_capacity(word.len());
89    let mut i = 0;
90    while i < chars.len() {
91        let at_boundary = i == 0 || !is_token_char(chars[i - 1]);
92        let found = if at_boundary {
93            token_len(&chars[i..])
94        } else if ends_with_escape(&chars[..i]) {
95            modern_token_len(&chars[i..])
96        } else {
97            None
98        };
99        match found {
100            Some(len) => {
101                out.push_str("[redacted]");
102                i += len;
103            }
104            None => {
105                out.push(chars[i]);
106                i += 1;
107            }
108        }
109    }
110    out
111}

113/// Whether `before` ends with a recognised escape sequence.
114fn ends_with_escape(before: &[char]) -> bool {
115    let n = before.len();
116    let at = |back: usize| (back <= n).then(|| before[n - back]);
117    let hex =
118        |from: usize, to: usize| (from..=to).all(|b| at(b).is_some_and(|c| c.is_ascii_hexdigit()));
119    (at(2) == Some('\\') && matches!(at(1), Some('n' | 't' | 'r' | '"')))
120        || (at(6) == Some('\\') && at(5) == Some('u') && hex(1, 4))
121        || (at(3) == Some('%') && hex(1, 2))
122}

124/// The length of the Vault token that `s` starts with, if it starts with one.
125///
126/// `hvs.`, `hvb.` and `hvr.` take a body of token characters (which include
127/// `.`: the modern body is base64url and may carry the separator). The older
128/// `s.` (service) and `b.` (batch) take a body of base62/base64url characters
129/// without `.`, so a host name such as `b.vault-internal.example.com` is not a
130/// token. A `.` that ends the match is punctuation and stays out of the span.
131fn token_len(s: &[char]) -> Option<usize> {
132    let starts =
133        |prefix: &str| prefix.len() <= s.len() && prefix.chars().zip(s).all(|(p, c)| p == *c);
134    let run = |from: usize, accept: fn(char) -> bool| {
135        s[from..].iter().take_while(|c| accept(**c)).count()
136    };
137    if let Some(len) = modern_token_len(s) {
138        return Some(len);
139    }
140    if starts("s.") || starts("b.") {
141        let body = run(2, is_legacy_body_char);
142        let continues = s.get(2 + body) == Some(&'.')
143            && s.get(3 + body)
144                .is_some_and(|c| is_token_char(*c) && *c != '.');
145        return (body >= MIN_TOKEN_BODY && !continues).then_some(2 + body);
146    }
147    None
148}
149
150/// The length of the modern (`hvs.`, `hvb.`, `hvr.`) token `s` starts with.
151fn modern_token_len(s: &[char]) -> Option<usize> {
152    let prefixed = ["hvs.", "hvb.", "hvr."]
153        .iter()
154        .any(|p| s.len() >= p.len() && p.chars().zip(s).all(|(p, c)| p == *c));
155    if !prefixed {
156        return None;
157    }
158    let body = s[4..].iter().take_while(|c| is_token_char(**c)).count();
159    let body = s[4..4 + body]
160        .iter()
161        .rposition(|c| *c != '.')
162        .map_or(0, |last| last + 1);
163    (body >= MIN_TOKEN_BODY).then_some(4 + body)
164}
165
166fn is_token_char(c: char) -> bool {
167    c.is_ascii_alphanumeric() || c == '_' || c == '-' || c == '.'
168}
169
170fn is_legacy_body_char(c: char) -> bool {
171    is_token_char(c) && c != '.'
172}

178    /// `prefix` and a low-entropy body of `12 * parts` token characters, built
179    /// at run time: a literal looks like a credential to secret scanners.

181        format!(
182            "{prefix}{}",
183            ["fake", "body", "0123"].concat().repeat(parts)

187    #[test]
188    fn backoff_doubles_from_a_minute_to_the_cap() {
189        let minutes: Vec<u64> = (1..=11).map(|n| backoff(n).as_secs() / 60).collect();
190        assert_eq!(minutes, [1, 2, 4, 8, 16, 32, 64, 128, 256, 360, 360]);
191        assert_eq!(backoff(0), BACKOFF_BASE, "no failure yet waits the base");
192        assert_eq!(
193            backoff(u32::MAX),
194            BACKOFF_CAP,
195            "a huge count cannot overflow"
196        );
197    }

199    #[test]
200    fn a_reason_is_one_bounded_line_without_a_token() {
201        let reason = sanitise_reason(&format!(
202            "vault pki: http://v:8200/v1/pki/revoke answered 403 Forbidden:\n\tpermission denied \
203             for {}",
204            fake_token("hvs.", 3)
205        ));
206        assert!(!reason.contains('\n') && !reason.contains('\t'));
207        assert!(reason.contains("403 Forbidden") && reason.contains("permission denied"));
208        assert!(!reason.contains("hvs."), "{reason}");
209        assert!(reason.contains("[redacted]"));
210
211        let long = sanitise_reason(&"x".repeat(10_000));
212        assert_eq!(long.chars().count(), MAX_REASON_CHARS);
213        // Multi-byte text is cut on a character, not a byte.
214        let wide = sanitise_reason(&"é".repeat(1_000));
215        assert_eq!(wide.chars().count(), MAX_REASON_CHARS);
216        // Ordinary words that merely start like a prefix are left alone.
217        assert_eq!(sanitise_reason("s.o.s. hvs.x"), "s.o.s. hvs.x");
218        assert_eq!(sanitise_reason("b.c. b.short"), "b.c. b.short");
219    }

221    /// #665 (R110W1-05): a token is masked wherever a non-token character
222    /// precedes it, the span only; a host name that starts like a legacy
223    /// prefix is left alone.
224    #[test]
225    fn a_token_is_masked_at_any_boundary_and_the_text_around_it_stays() {
226        let tok = fake_token("hvs.", 3);
227        let tok = tok.as_str();
228        assert_eq!(sanitise_reason(&format!("token={tok}")), "token=[redacted]");
229        assert_eq!(
230            sanitise_reason(&format!("{{\"token\":\"{tok}\"}}")),

232        );
233        assert_eq!(sanitise_reason(&format!("\"{tok}\".")), "\"[redacted]\".");
234        assert_eq!(sanitise_reason(&format!("'{tok}'.")), "'[redacted]'.");
235        assert_eq!(sanitise_reason(&format!("x:{tok}.")), "x:[redacted].");
236        assert_eq!(
237            sanitise_reason(&format!("token={},", fake_token("b.", 1))),
238            "token=[redacted],"
239        );
240        // Preceded by a token character it is part of a longer word.
241        assert_eq!(sanitise_reason(&format!("abc{tok}")), format!("abc{tok}"));
242
243        assert_eq!(
244            sanitise_reason("b.vault-internal.example.com"),
245            "b.vault-internal.example.com"
246        );
247        assert_eq!(
248            sanitise_reason("s.vault-1.eu-west.internal"),
249            "s.vault-1.eu-west.internal"
250        );

253    /// #740 (R111W1-06): a modern token right after an escape is masked, the
254    /// escape kept; the legacy trade-off and the host names stay as T-470 has them.
255    #[test]
256    fn a_token_after_an_escape_is_masked_and_the_escape_stays() {
257        let tok = fake_token("hvs.", 3);
258        for esc in [
259            "\\n", "\\t", "\\r", "\\\"", "\\u0022", "\\u00aF", "%3D", "%0a",
260        ] {
261            for prefix in ["hvs.", "hvb.", "hvr."] {
262                let tok = fake_token(prefix, 3);
263                assert_eq!(
264                    sanitise_reason(&format!("msg{esc}{tok}")),
265                    format!("msg{esc}[redacted]"),
266                    "{esc} {prefix}"
267                );
268            }
269        }
270        assert_eq!(
271            sanitise_reason(&format!("token%3D{tok}")),
272            "token%3D[redacted]"
273        );
274        // Not an escape: a token character, or a malformed escape, still joins the word.
275        for glued in ["abc", "\\x", "\\u12", "%G1", "%1"] {
276            assert_eq!(
277                sanitise_reason(&format!("{glued}{tok}")),
278                format!("{glued}{tok}")
279            );
280        }
281        // A short body after an escape is not a token.
282        assert_eq!(sanitise_reason("a\\nhvs.short"), "a\\nhvs.short");
283        // The legacy trade-off (T-470): `s.` + body + `.word` reads as a host, unmasked.
284        let legacy = ["s.", "AbCdEfGh", "1234", ".next"].concat();
285        assert_eq!(sanitise_reason(&legacy), legacy);
286        // The legacy prefixes get no escape boundary.
287        let after_escape = format!("msg\\n{}", fake_token("s.", 1));
288        assert_eq!(sanitise_reason(&after_escape), after_escape);
289        for host in ["b.vault-internal.example.com", "s.vault-1.eu-west.internal"] {
290            assert_eq!(sanitise_reason(host), host);
291            let escaped = format!("x\\n{host}");
292            assert_eq!(sanitise_reason(&escaped), escaped);
293        }
294        // Idempotent.
295        let once = sanitise_reason(&format!("a\\n{tok} b%3D{tok}"));
296        assert_eq!(sanitise_reason(&once), once);
297    }

299    /// The look-back is a constant six characters and a mask skips its span, so
300    /// a long run of escape-and-prefix pairs is read once, not once per pair.
301    #[test]
302    fn the_escape_scan_stays_linear() {
303        let unit = "\\nhvs.abc";
304        let (small, large) = (unit.repeat(2_000), unit.repeat(200_000));
305        let time = |s: &str| {
306            let t = std::time::Instant::now();
307            let out = sanitise_reason(s);
308            (t.elapsed(), out)
309        };
310        let (t1, _) = time(&small);
311        let (t2, _) = time(&large);
312        // 100x the input; a quadratic scan would take ~10_000x. Generous bound.
313        assert!(
314            t2 < t1.max(std::time::Duration::from_millis(5)) * 1_000,
315            "{t1:?} {t2:?}"
316        );
317        let pairs = format!("{}{}", "\\n", fake_token("hvs.", 1)).repeat(50_000);
318        let (_, out) = time(&pairs);
319        assert!(!out.contains("fake"));
320    }
