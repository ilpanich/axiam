1//! The retry policy of the Vault revocation sweep (T-470, #603).
2//!
3//! A revocation Vault refuses is not retried at every tick: each refusal pushes
4//! the row's next attempt out (1 minute, doubling, to 6 hours), a replica claims
5//! a row before it attempts it, and a row that keeps failing past
6//! [`FAILING_AFTER_ATTEMPTS`] is reported as its own job-health error. The
7//! constants and the two pure functions live here, apart from the sweep, so the
8//! next consumer of the same bookkeeping (a CA certificate's revocation) does
9//! not copy them.
10
11use std::time::Duration;
12
13/// The wait after a row's first refusal.
14pub const BACKOFF_BASE: Duration = Duration::from_secs(60);
15
16/// The longest wait between two attempts at one row.
17pub const BACKOFF_CAP: Duration = Duration::from_secs(6 * 60 * 60);
18
19/// Refusals after which a row counts as failing for good, and `/health/jobs`
20/// reports it (as `vault_revocation_failing`) instead of failing the pass that
21/// retries it. Twelve attempts span about a day: the first nine back off from
22/// one minute to a little over four hours, the rest wait the six-hour cap.
23pub const FAILING_AFTER_ATTEMPTS: u32 = 12;
24
25/// How long a replica holds a claimed row. Longer than one sweep pass
26/// ([`crate::cert`] bounds it to 30 seconds), so a row being attempted is not
27/// due for anyone else; a replica that dies mid-attempt loses the row for this
28/// long and no longer, with nothing counted against the row.
29pub const CLAIM_LEASE: Duration = Duration::from_secs(120);
30
31/// The longest reason a row keeps, in characters: a status and a short message.
32pub const MAX_REASON_CHARS: usize = 256;
33
34/// How long to wait before the next attempt at a row that has now failed
35/// `attempts` times (1 for the first refusal): [`BACKOFF_BASE`] doubled for each
36/// earlier refusal, capped at [`BACKOFF_CAP`].
37pub fn backoff(attempts: u32) -> Duration {
38    let doublings = attempts.saturating_sub(1).min(20);
39    BACKOFF_BASE
40        .saturating_mul(1u32 << doublings)
41        .min(BACKOFF_CAP)
42}
43
44/// A custodian's error as a row may keep it: control characters and runs of
45/// whitespace collapsed, anything that reads like a Vault token replaced, and
46/// the whole cut to [`MAX_REASON_CHARS`].
47///
48/// The stored reason is read by operators and may reach a log or an export, so
49/// it carries neither a credential nor a response body. The Vault client never
50/// puts its token in an error; this is the second line, for a message a server
51/// echoed back or a future change.
52pub fn sanitise_reason(raw: &str) -> String {
53    let words: Vec<&str> = raw.split_whitespace().collect();
54    let mut out = String::new();
55    for word in words {
56        if !out.is_empty() {
57            out.push(' ');
58        }
59        let word: String = word.chars().filter(|c| !c.is_control()).collect();
60        if looks_like_a_token(&word) {
61            out.push_str("[redacted]");
62        } else {
63            out.push_str(&word);
64        }
65    }
66    if out.chars().count() > MAX_REASON_CHARS {
67        let mut cut: String = out.chars().take(MAX_REASON_CHARS - 1).collect();
68        cut.push('…');
69        return cut;
70    }
71    out
72}

74/// A Vault token as Vault spells one: `hvs.`, `hvb.` or `hvr.` and a body, or
75/// the older `s.` (service) and `b.` (batch) followed by a long run of token
76/// characters.
77fn looks_like_a_token(word: &str) -> bool {
78    let word =
79        word.trim_matches(|c: char| !c.is_ascii_alphanumeric() && c != '.' && c != '_' && c != '-');
80    let body = |prefix: &str| word.strip_prefix(prefix).filter(|rest| rest.len() >= 8);
81    ["hvs.", "hvb.", "hvr.", "s.", "b."]
82        .iter()
83        .any(|prefix| body(prefix).is_some_and(|rest| rest.chars().all(is_token_char)))
84}
85
86fn is_token_char(c: char) -> bool {
87    c.is_ascii_alphanumeric() || c == '_' || c == '-' || c == '.'
88}

90#[cfg(test)]
91mod tests {
92    use super::*;
93
94    #[test]
95    fn backoff_doubles_from_a_minute_to_the_cap() {
96        let minutes: Vec<u64> = (1..=11).map(|n| backoff(n).as_secs() / 60).collect();
97        assert_eq!(minutes, [1, 2, 4, 8, 16, 32, 64, 128, 256, 360, 360]);
98        assert_eq!(backoff(0), BACKOFF_BASE, "no failure yet waits the base");
99        assert_eq!(
100            backoff(u32::MAX),
101            BACKOFF_CAP,
102            "a huge count cannot overflow"
103        );
104    }
105
106    #[test]
107    fn a_reason_is_one_bounded_line_without_a_token() {
108        let reason = sanitise_reason(
109            "vault pki: http://v:8200/v1/pki/revoke answered 403 Forbidden:\n\tpermission denied \
110             for hvs.CAESIAbcdefghijklmnop012345",
111        );
112        assert!(!reason.contains('\n') && !reason.contains('\t'));
113        assert!(reason.contains("403 Forbidden") && reason.contains("permission denied"));
114        assert!(!reason.contains("hvs."), "{reason}");
115        assert!(reason.contains("[redacted]"));
116
117        let long = sanitise_reason(&"x".repeat(10_000));
118        assert_eq!(long.chars().count(), MAX_REASON_CHARS);
119        // Multi-byte text is cut on a character, not a byte.
120        let wide = sanitise_reason(&"é".repeat(1_000));
121        assert_eq!(wide.chars().count(), MAX_REASON_CHARS);
122        // Ordinary words that merely start like a prefix are left alone.
123        assert_eq!(sanitise_reason("s.o.s. hvs.x"), "s.o.s. hvs.x");
124        assert_eq!(sanitise_reason("b.c. b.short"), "b.c. b.short");
125    }
126
127    /// R101W1-06: a legacy batch token (`b.` and a body) is masked as `s.` is.
128    #[test]
129    fn a_legacy_batch_token_is_masked() {
130        let reason =
131            sanitise_reason("denied for b.AAAAAQJ0aGlzX2lzX2Ff and \"b.AAAAAQJ0aGlzX2Ff\"");
132        assert!(!reason.contains("b.AAAA"), "{reason}");
133        assert_eq!(reason.matches("[redacted]").count(), 2, "{reason}");
134        assert!(reason.starts_with("denied for "));
135    }
136
137    /// The row, the audit append and the log each apply the sanitiser to what
138    /// the sweep or the handler already sanitised: a second pass changes nothing.
139    #[test]
140    fn sanitising_is_idempotent() {
141        for raw in [
142            "denied for hvs.CAESIAbcdefghijklmnop012345 b.AAAAAQJ0aGlzX2Ff",
143            &"y ".repeat(5_000),
144            &"é".repeat(1_000),
145        ] {
146            let once = sanitise_reason(raw);
147            assert_eq!(sanitise_reason(&once), once);
148        }
149    }
