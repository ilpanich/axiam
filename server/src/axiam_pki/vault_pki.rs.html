1//! CA keys that are born inside HashiCorp Vault and never leave it.
2//!
3//! [`VaultCaKeyStore`](crate::ca_key_store::VaultCaKeyStore) puts a key AXIAM
4//! generated into Vault's KV engine. This is the other arrangement, and the
5//! difference is not a degree: Vault's **PKI secrets engine** generates the key
6//! itself, exposes no API that exports it, and signs on AXIAM's behalf. The
7//! private key of a CA under this custodian has never existed in this process,
8//! so a compromise of AXIAM — memory dump, malicious build, an operator with a
9//! shell — yields certificates that Vault's audit log records, and no key.
10//!
11//! # The shape, and where it comes from
12//!
13//! HashiCorp's own PKI walkthrough is the model, and this implements it
14//! literally:
15//!
16//! 1. `POST <root_mount>/root/generate/internal` — a root whose key stays in
17//!    Vault. `internal` (rather than `exported`) is what makes that true.
18//! 2. `POST <int_mount>/intermediate/generate/internal` — a second key, also
19//!    kept, and a CSR.
20//! 3. `POST <root_mount>/issuer/<root>/sign-intermediate` — the root signs it.
21//! 4. `POST <int_mount>/intermediate/set-signed` — the signed certificate goes
22//!    back to the mount holding its key, which is now an issuer.
23//! 5. `POST <int_mount>/issuer/<int>/sign-verbatim` — every leaf, thereafter.
24//!
25//! The two-tier structure is not ceremony. A root that signs exactly one
26//! intermediate and nothing else can have that intermediate revoked and
27//! replaced without redistributing the trust anchor, which is the one PKI
28//! operation that is otherwise impossible to perform quietly.
29//!
30//! # What this does *not* remove
31//!
32//! On the BYOK path — [`CaKeyStore::import_ca`] — the key an operator supplies
33//! passes through AXIAM's memory on its way to `issuers/import/bundle`. It has
34//! to: AXIAM is the thing receiving the HTTP request. What the operator gets is
35//! that AXIAM never *stores* it, and every use of it afterwards is Vault's.
36//!
37//! # Mounts, not a mount per CA
38//!
39//! Both mounts are enabled by the operator and shared by every CA. The
40//! alternative — AXIAM enabling a mount per CA — needs `sys/mounts` write,
41//! which is a permission that can rewrite the whole Vault. Sharing two mounts
42//! and addressing issuers by id keeps the token's policy to the PKI paths
43//! themselves. Vault has supported multiple issuers per mount since 1.11.
44

60/// How long to wait on Vault before giving up.
61///
62/// Longer than the KV store's ten seconds because these calls do real work:
63/// generating an RSA-4096 root inside Vault is seconds of CPU, not a lookup.
64/// Still bounded, because this runs on the request that generates a CA or
65/// issues a certificate, and an unbounded wait turns a Vault outage into hung
66/// request handlers rather than a refused issuance.
67const VAULT_PKI_TIMEOUT: Duration = Duration::from_secs(60);
68

78/// How to reach the two PKI mounts.
79///
80/// `Debug` is hand-written to redact the token — this struct is exactly the
81/// kind of thing that reaches a startup log line or a panic message.
82#[derive(Clone, PartialEq, Eq)]
83pub struct VaultPkiConfig {
84    /// Base address, e.g. `https://vault.internal:8200`.
85    pub address: String,
86    /// A token whose policy covers the paths in this module's header.
87    pub token: String,
88    /// PKI mount holding the roots AXIAM generates.
89    pub root_mount: String,
90    /// PKI mount holding the issuing intermediates, and the issuer that signs
91    /// every leaf.
92    pub int_mount: String,
93    /// PEM bundle for the CA that issued Vault's listener certificate.
94    ///
95    /// `None` verifies against the built-in roots, which is right only for a
96    /// Vault fronted by a publicly-trusted certificate: reqwest is built with
97    /// `rustls-tls`, whose roots are compiled in, so an internally-issued
98    /// listener certificate has no `SSL_CERT_FILE` to fall back on.
99    pub ca_cert_path: Option<std::path::PathBuf>,
100}

118/// One issuer inside one mount, addressed the way Vault addresses it.
119#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
120pub struct VaultPkiIssuer {
121    /// The PKI mount, without leading or trailing slashes.
122    pub mount: String,
123    /// Vault's `issuer_id` — a UUID it minted, not a name.
124    pub issuer: String,
125    /// Vault's `key_id` for the key beneath that issuer.
126    pub key: String,
127}

129/// Everything needed to sign with, and later delete, a `vault_pki` CA.
130///
131/// Serialised into the CA row's single `key_locator` string, because the row
132/// has one column for "where the custodian put it" and this custodian put it in
133/// four places. JSON rather than a delimited form so that adding a field later
134/// — a namespace, a second intermediate — does not have to guess how many
135/// separators an old row had.
136///
137/// The mounts are recorded rather than re-read from configuration for the same
138/// reason the custodian itself is recorded per CA: an operator who changes
139/// `AXIAM__PKI__VAULT_PKI_INT_MOUNT` has not moved the issuers that already
140/// exist, and a signing path that consulted the current setting would look in
141/// the wrong mount.
142#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
143pub struct VaultPkiLocator {
144    /// The issuer that signs leaves.
145    pub issuing: VaultPkiIssuer,
146    /// The root the issuing intermediate was created beneath.
147    ///
148    /// `None` when the issuing certificate *is* the root, and when the CA was
149    /// imported rather than generated — in both cases there is no second issuer
150    /// to clean up on revocation.
151    #[serde(default, skip_serializing_if = "Option::is_none")]
152    pub root: Option<VaultPkiIssuer>,
153}
154
155impl VaultPkiLocator {
156    /// Read a locator back out of a CA row.
157    pub fn parse(locator: &str) -> AxiamResult<Self> {
158        serde_json::from_str(locator).map_err(|e| {
159            AxiamError::Certificate(format!(
160                "this CA's key locator is not one this custodian wrote: {e}"
161            ))
162        })
163    }
164
165    fn encode(&self) -> AxiamResult<String> {
166        serde_json::to_string(self).map_err(|e| {
167            AxiamError::Internal(format!("vault pki: could not encode the key locator: {e}"))
168        })
169    }
170
171    /// The issuer to sign a *CA* with, which is not the one that signs leaves.
172    ///
173    /// `issuing` is created with `max_path_length: 0` — it may sign leaves and
174    /// nothing else, so a certificate it issued to another CA would be rejected
175    /// by every validator that checked the chain. The root above it has no such
176    /// constraint, and is what a new tier hangs from.
177    ///
178    /// `None` root means there is no second tier: the CA was created with
179    /// `issue_from_root`, or imported, and `issuing` is the top of what AXIAM
180    /// has. Whatever constraints that certificate carries are then the
181    /// operator's own, and Vault will refuse the signature if they forbid it.
182    fn signing_issuer(&self) -> &VaultPkiIssuer {
183        self.root.as_ref().unwrap_or(&self.issuing)
184    }
185}

206impl VaultPkiCaKeyStore {
207    /// Build a store, including the TLS trust anchor if one is configured.
208    ///
209    /// Fails rather than falling back to the built-in roots when the anchor is
210    /// unreadable: silently verifying against a different trust store than the
211    /// operator asked for is how a misconfiguration becomes a connection to
212    /// something that is not their Vault.
213    pub fn new(config: VaultPkiConfig) -> AxiamResult<Self> {
214        let mut builder = Client::builder().timeout(VAULT_PKI_TIMEOUT);
215
216        if let Some(ref path) = config.ca_cert_path {
217            let pem = std::fs::read(path).map_err(|e| {
218                AxiamError::Internal(format!(
219                    "vault PKI CA key store: could not read the trust anchor at {}: {e}",
220                    path.display()
221                ))
222            })?;
223            let cert = reqwest::Certificate::from_pem(&pem).map_err(|e| {
224                AxiamError::Internal(format!(
225                    "vault PKI CA key store: {} is not a PEM certificate: {e}",
226                    path.display()
227                ))
228            })?;
229            builder = builder.add_root_certificate(cert);
230        }
231
232        let client = builder.build().map_err(|e| {
233            AxiamError::Internal(format!(
234                "vault PKI CA key store: could not build HTTP client: {e}"
235            ))
236        })?;
237
238        Ok(Self { client, config })
239    }
240
241    /// Build from an already-configured client. For tests, and for a caller
242    /// that has its own client policy.
243    pub fn with_client(client: Client, config: VaultPkiConfig) -> Self {
244        Self { client, config }
245    }

247    fn url(&self, mount: &str, path: &str) -> String {
248        format!(
249            "{}/v1/{}/{}",
250            self.config.address.trim_end_matches('/'),
251            mount.trim_matches('/'),
252            path.trim_start_matches('/'),
253        )
254    }
255
256    /// `POST` a JSON body and return the response's `data` object.
257    ///
258    /// Vault's warnings are logged rather than swallowed. They are how it tells
259    /// you a TTL was capped to the mount's `max_lease_ttl` — a 30-day default
260    /// that silently turns a requested ten-year root into a month-long one, and
261    /// the single most likely way a working configuration surprises an operator.
262    async fn post(&self, url: &str, body: serde_json::Value) -> AxiamResult<serde_json::Value> {
263        let response = self
264            .client
265            .post(url)
266            .header("X-Vault-Token", &self.config.token)
267            .json(&body)
268            // Deliberately carries neither the token nor the request body.
269            .send()
270            .await
271            .map_err(|e| AxiamError::Internal(format!("vault pki: {url} could not be reached: {e}")))?;
272
273        let status = response.status();
274        let payload: serde_json::Value = response.json().await.unwrap_or(serde_json::Value::Null);
275
276        if !status.is_success() {
277            return Err(AxiamError::Certificate(format!(
278                "vault pki: {url} answered {status}: {}",
279                vault_errors(&payload)
280            )));
281        }
282
283        log_warnings(url, &payload);
284
285        payload
286            .get("data")
287            .cloned()
288            .filter(|d| !d.is_null())
289            .ok_or_else(|| {
290                AxiamError::Internal(format!("vault pki: {url} answered {status} with no `data`"))
291            })
292    }
293
294    /// `DELETE` a path, treating "already gone" as success.
295    async fn delete(&self, url: &str) -> AxiamResult<()> {
296        let response = self
297            .client
298            .delete(url)
299            .header("X-Vault-Token", &self.config.token)
300            .send()
301            .await
302            .map_err(|e| AxiamError::Internal(format!("vault pki: deleting {url} failed: {e}")))?;
303
304        // Already gone is success: this runs on revocation, and a second
305        // revocation must not fail because the first one worked.
306        if response.status().is_success() || response.status() == reqwest::StatusCode::NOT_FOUND {
307            return Ok(());
308        }
309        Err(AxiamError::Internal(format!(
310            "vault pki: {url} answered {} on delete",
311            response.status()
312        )))
313    }

315    /// Generate a root whose key stays inside Vault.
316    async fn generate_root(
317        &self,
318        request: &CaGenerationRequest,
319    ) -> AxiamResult<(String, VaultPkiIssuer)> {
320        let (key_type, key_bits) = key_params(&request.key_algorithm);
321        let url = self.url(&self.config.root_mount, "root/generate/internal");
322        let data = self
323            .post(
324                &url,
325                serde_json::json!({
326                    "common_name": request.subject,
327                    "ttl": ttl_days(request.validity_days),
328                    "key_type": key_type,
329                    "key_bits": key_bits,
330                    // Names are for whoever opens Vault's UI; the locator
331                    // addresses by id, so a rename cannot strand a CA row.
332                    "issuer_name": format!("axiam-root-{}", request.ca_id),
333                    "key_name": format!("axiam-root-{}", request.ca_id),
334                    "format": "pem",
335                }),
336            )
337            .await?;
338
339        let certificate = string_field(&data, "certificate", &url)?;
340        Ok((
341            certificate,
342            VaultPkiIssuer {
343                mount: self.config.root_mount.trim_matches('/').to_string(),
344                issuer: string_field(&data, "issuer_id", &url)?,
345                key: string_field(&data, "key_id", &url)?,
346            },
347        ))
348    }

350    /// `POST issuer/{id}/sign-intermediate` — the one call that mints a CA.
351    ///
352    /// Shared by the three paths that need it: creating the organization CA's
353    /// own intermediate, creating a tenant signing CA, and signing a CSR a
354    /// tenant produced elsewhere. The constraints are stated here once so those
355    /// three cannot drift into issuing differently-powered CAs.
356    ///
357    /// Returns the certificate and, when Vault named one, the issuing CA's own
358    /// certificate — the chain entry a relying party needs and cannot get from
359    /// anywhere else, because the issuer's certificate lives inside Vault.
360    async fn sign_as_intermediate(
361        &self,
362        signer: &VaultPkiIssuer,
363        csr_pem: &str,
364        subject: &str,
365        ttl: String,
366    ) -> AxiamResult<(String, Option<String>)> {
367        let url = self.url(
368            &signer.mount,
369            &format!("issuer/{}/sign-intermediate", signer.issuer),
370        );
371        let signed = self
372            .post(
373                &url,
374                serde_json::json!({
375                    "csr": csr_pem,
376                    "common_name": subject,
377                    "ttl": ttl,
378                    "format": "pem",
379                    // Signs leaves and nothing else. Without this it inherits an
380                    // unconstrained path length and could mint further CAs — a
381                    // capability nothing in AXIAM uses and everything downstream
382                    // would have to trust.
383                    "max_path_length": 0,
384                    // The CSR's own subject and extensions are ignored in favour
385                    // of what is named here: `use_csr_values` would let the
386                    // request describe its own powers, and AXIAM has already
387                    // decided what they are.
388                    "use_csr_values": false,
389                }),
390            )
391            .await?;
392
393        let certificate = string_field(&signed, "certificate", &url)?;
394        let issuing_ca = signed
395            .get("issuing_ca")
396            .and_then(|v| v.as_str())
397            .filter(|s| !s.is_empty() && s.trim() != certificate.trim())
398            .map(str::to_string);
399        Ok((certificate, issuing_ca))
400    }
401
402    /// Create the intermediate key and CSR, have the root sign it, and hand the
403    /// signed certificate back to the mount that holds its key.
404    async fn generate_intermediate(
405        &self,
406        request: &CaGenerationRequest,
407        subject: &str,
408        validity_days: u32,
409        root: &VaultPkiIssuer,
410    ) -> AxiamResult<(String, VaultPkiIssuer)> {
411        let (key_type, key_bits) = key_params(&request.key_algorithm);
412
413        let generate_url = self.url(&self.config.int_mount, "intermediate/generate/internal");
414        let generated = self
415            .post(
416                &generate_url,
417                serde_json::json!({
418                    "common_name": subject,
419                    "key_type": key_type,
420                    "key_bits": key_bits,
421                    "key_name": format!("axiam-int-{}", request.ca_id),
422                    "format": "pem",
423                }),
424            )
425            .await?;
426        let csr = string_field(&generated, "csr", &generate_url)?;
427
428        let (certificate, _issuing_ca) = self
429            .sign_as_intermediate(root, &csr, subject, ttl_days(validity_days))
430            .await?;
431
432        let set_url = self.url(&self.config.int_mount, "intermediate/set-signed");
433        let imported = self
434            .post(&set_url, serde_json::json!({ "certificate": certificate }))
435            .await?;
436        let issuer = first_imported_issuer(&imported, &set_url)?;
437        // The key was created by `intermediate/generate/internal`, so
438        // `set-signed` imports no key and `imported_keys` is empty; the id we
439        // want is the one that call already returned.
440        let key = string_field(&generated, "key_id", &generate_url)?;
441
442        Ok((
443            certificate,
444            VaultPkiIssuer {
445                mount: self.config.int_mount.trim_matches('/').to_string(),
446                issuer,
447                key,
448            },
449        ))
450    }

453impl CaKeyStore for VaultPkiCaKeyStore {
454    /// Refused, and the refusal is the feature.
455    ///
456    /// Reaching here means a caller generated a CA key in this process and
457    /// asked Vault's PKI engine to keep it — which is
458    /// [`crate::ca_key_store::VaultCaKeyStore`]'s job, under custody `vault`.
459    /// The PKI engine has nowhere to put a key that has no certificate.
460    fn store<'a>(
461        &'a self,
462        _organization_id: Uuid,
463        _ca_id: Uuid,
464        _private_key_pem: &'a str,
465    ) -> Pin<Box<dyn Future<Output = AxiamResult<StoredCaKey>> + Send + 'a>> {
466        Box::pin(async {
467            Err(AxiamError::Validation {
468                message: "Vault's PKI engine stores a key only together with the \
469                          certificate it belongs to: generate the CA in Vault, or import \
470                          the certificate and key together"
471                    .into(),
472            })
473        })
474    }
475
476    /// Refused: there is no API that exports this key, by design.
477    ///
478    /// `Validation` rather than `Certificate` so the sentence reaches the
479    /// caller — `Certificate` maps to a 500 whose body says only that an
480    /// internal error occurred, and this is not an error at all but a
481    /// statement about how the CA was made.
482    fn load<'a>(
483        &'a self,
484        _key_ref: &'a CaKeyRef,
485        _inline: Option<&'a [u8]>,
486    ) -> Pin<Box<dyn Future<Output = AxiamResult<Zeroizing<String>>> + Send + 'a>> {
487        Box::pin(async {
488            Err(AxiamError::Validation {
489                message: "this CA's private key was generated inside Vault's PKI engine and \
490                          cannot be exported: AXIAM signs through Vault instead of holding \
491                          the key"
492                    .into(),
493            })
494        })
495    }
496
497    /// Remove the issuer, then the key it used — in that order, because Vault
498    /// refuses to delete a key an issuer still references.
499    ///
500    /// Both tiers go: the root exists to sign one intermediate, and a root
501    /// whose only intermediate has been revoked signs nothing anyone should
502    /// accept.
503    fn delete<'a>(
504        &'a self,
505        key_ref: &'a CaKeyRef,
506    ) -> Pin<Box<dyn Future<Output = AxiamResult<()>> + Send + 'a>> {
507        Box::pin(async move {
508            let locator = VaultPkiLocator::parse(&key_ref.locator)?;
509            let mut issuers = vec![locator.issuing];
510            issuers.extend(locator.root);
511            for issuer in issuers {
512                self.delete(&self.url(&issuer.mount, &format!("issuer/{}", issuer.issuer)))
513                    .await?;
514                self.delete(&self.url(&issuer.mount, &format!("key/{}", issuer.key)))
515                    .await?;
516            }
517            Ok(())
518        })
519    }
520
521    fn generates_cas(&self) -> bool {
522        true
523    }
524
525    fn generate_ca<'a>(
526        &'a self,
527        request: &'a CaGenerationRequest,
528    ) -> Pin<Box<dyn Future<Output = AxiamResult<GeneratedCa>> + Send + 'a>> {
529        Box::pin(async move {
530            let (root_pem, root) = self.generate_root(request).await?;
531
532            let (certificate_pem, chain_pem, locator) = match request.intermediate {
533                Some(ref spec) => {
534                    let (int_pem, issuing) = self
535                        .generate_intermediate(request, &spec.subject, spec.validity_days, &root)
536                        .await?;
537                    (
538                        int_pem,
539                        // Built here rather than taken from Vault's `ca_chain`,
540                        // whose contents differ between endpoints and versions.
541                        // The root's certificate came back from the call that
542                        // created it, and that is the only copy anything
543                        // outside Vault will ever see.
544                        vec![root_pem],
545                        VaultPkiLocator {
546                            issuing,
547                            root: Some(root),
548                        },
549                    )
550                }
551                None => (
552                    root_pem,
553                    Vec::new(),
554                    VaultPkiLocator {
555                        issuing: root,
556                        root: None,
557                    },
558                ),
559            };
560
561            Ok(GeneratedCa {
562                certificate_pem,
563                chain_pem,
564                locator: locator.encode()?,
565            })
566        })
567    }
568
569    /// BYOK: hand Vault a key and the certificate it belongs to, together.
570    ///
571    /// The imported issuer goes into the *issuing* mount, because that is the
572    /// mount leaves are signed from and an imported CA is by definition the one
573    /// doing the signing — there is no root above it that AXIAM created.
574    fn import_ca<'a>(
575        &'a self,
576        _organization_id: Uuid,
577        _ca_id: Uuid,
578        certificate_pem: &'a str,
579        private_key_pem: &'a str,
580    ) -> Pin<Box<dyn Future<Output = AxiamResult<StoredCaKey>> + Send + 'a>> {
581        Box::pin(async move {
582            let url = self.url(&self.config.int_mount, "issuers/import/bundle");
583            // Key first: Vault reads the bundle as a stream of PEM blocks and
584            // does not care about order, but a human reading a failed request
585            // in an audit log does.
586            let bundle = format!(
587                "{}\n{}",
588                private_key_pem.trim_end(),
589                certificate_pem.trim_end()
590            );
591            let data = self
592                .post(&url, serde_json::json!({ "pem_bundle": bundle }))
593                .await?;
594
595            let issuer = first_imported_issuer(&data, &url)?;
596            let key = data
597                .get("mapping")
598                .and_then(|m| m.get(&issuer))
599                .and_then(|v| v.as_str())
600                .filter(|k| !k.is_empty())
601                .ok_or_else(|| {
602                    // Vault maps issuer -> key and leaves the value empty for an
603                    // issuer it holds no key for. That means the bundle carried
604                    // a certificate and no usable key, which would produce a CA
605                    // that lists as issuable and fails at the first signature.
606                    AxiamError::Validation {
607                        message: "Vault imported the certificate but no private key for it: \
608                                  check the bundle contains the key that matches this CA"
609                            .into(),
610                    }
611                })?
612                .to_string();
613
614            let locator = VaultPkiLocator {
615                issuing: VaultPkiIssuer {
616                    mount: self.config.int_mount.trim_matches('/').to_string(),
617                    issuer,
618                    key,
619                },
620                root: None,
621            };
622            Ok(StoredCaKey::Referenced(locator.encode()?))
623        })
624    }

626    fn signs_remotely(&self) -> bool {
627        true
628    }

630    /// Sign a leaf with `sign-verbatim`, which takes the CSR's subject and
631    /// extensions as given.
632    ///
633    /// Verbatim rather than a Vault role because AXIAM has already decided what
634    /// the certificate says — subject, validity, key algorithm, and the tenant
635    /// policy behind them. A role would restate a subset of those rules in a
636    /// second place, and the two would drift.
637    ///
638    /// The cost is that the token's policy on this path is what stands between
639    /// a compromised AXIAM and a certificate for any name at all. An operator
640    /// who wants Vault to enforce names as well can point the mount's role at
641    /// the same issuer; what they cannot do is have neither.
642    fn sign_csr<'a>(
643        &'a self,
644        key_ref: &'a CaKeyRef,
645        request: &'a LeafSigningRequest,
646    ) -> Pin<Box<dyn Future<Output = AxiamResult<SignedLeaf>> + Send + 'a>> {
647        Box::pin(async move {
648            let locator = VaultPkiLocator::parse(&key_ref.locator)?;
649            let url = self.url(
650                &locator.issuing.mount,
651                &format!("issuer/{}/sign-verbatim", locator.issuing.issuer),
652            );
653            let data = self
654                .post(
655                    &url,
656                    serde_json::json!({
657                        "csr": request.csr_pem,
658                        "ttl": format!("{}s", request.ttl_seconds.max(1)),
659                        "format": "pem",
660                    }),
661                )
662                .await?;
663
664            let certificate_pem = string_field(&data, "certificate", &url)?;
665            // `ca_chain` means different things at different Vault versions —
666            // sometimes including the leaf. Drop anything equal to the
667            // certificate itself rather than trusting the shape.
668            let chain_pem = data
669                .get("ca_chain")
670                .and_then(|c| c.as_array())
671                .map(|entries| {
672                    entries
673                        .iter()
674                        .filter_map(|v| v.as_str())
675                        .filter(|pem| pem.trim() != certificate_pem.trim())
676                        .map(str::to_string)
677                        .collect()
678                })
679                .unwrap_or_default();
680
681            Ok(SignedLeaf {
682                certificate_pem,
683                chain_pem,
684            })
685        })
686    }

688    /// Create a tenant signing CA whose key is born in — and stays in — Vault.
689    ///
690    /// The same three calls as [`Self::generate_intermediate`], pointed at an
691    /// issuer that already exists rather than one this call just made: Vault
692    /// generates the key and a CSR in the issuing mount, the parent's signing
693    /// issuer signs it, and `intermediate/set-signed` hands the certificate
694    /// back to the mount that holds the key. Vault can then issue this tenant's
695    /// leaves itself, which is the whole point — the key never exists here, so
696    /// there is nothing for AXIAM to lose.
697    fn generate_intermediate_ca<'a>(
698        &'a self,
699        parent: &'a CaKeyRef,
700        request: &'a IntermediateCaRequest,
701    ) -> Pin<Box<dyn Future<Output = AxiamResult<CustodiedIntermediate>> + Send + 'a>> {
702        Box::pin(async move {
703            let parent_locator = VaultPkiLocator::parse(&parent.locator)?;
704            let signer = parent_locator.signing_issuer();
705            let (key_type, key_bits) = key_params(&request.key_algorithm);
706
707            let generate_url = self.url(&self.config.int_mount, "intermediate/generate/internal");
708            let generated = self
709                .post(
710                    &generate_url,
711                    serde_json::json!({
712                        "common_name": request.subject,
713                        "key_type": key_type,
714                        "key_bits": key_bits,
715                        "key_name": format!("axiam-tenant-int-{}", request.ca_id),
716                        "format": "pem",
717                    }),
718                )
719                .await?;
720            let csr = string_field(&generated, "csr", &generate_url)?;
721
722            let (certificate, issuing_ca) = self
723                .sign_as_intermediate(
724                    signer,
725                    &csr,
726                    &request.subject,
727                    ttl_days(request.validity_days),
728                )
729                .await?;
730
731            let set_url = self.url(&self.config.int_mount, "intermediate/set-signed");
732            let imported = self
733                .post(&set_url, serde_json::json!({ "certificate": certificate }))
734                .await?;
735            let issuer = first_imported_issuer(&imported, &set_url)?;
736            // The key came from `intermediate/generate/internal`, so
737            // `set-signed` imports no key and `imported_keys` is empty; the id
738            // wanted here is the one that call already returned.
739            let key = string_field(&generated, "key_id", &generate_url)?;
740
741            Ok(CustodiedIntermediate {
742                certificate_pem: certificate,
743                chain_pem: issuing_ca.into_iter().collect(),
744                locator: VaultPkiLocator {
745                    issuing: VaultPkiIssuer {
746                        mount: self.config.int_mount.trim_matches('/').to_string(),
747                        issuer,
748                        key,
749                    },
750                    // No root to clean up on revocation: the root above this one
751                    // belongs to the organization CA and outlives it.
752                    root: None,
753                }
754                .encode()?,
755            })
756        })
757    }
758
759    /// Sign somebody else's CSR as a CA, with `sign-intermediate`.
760    ///
761    /// Not `sign-verbatim`, which is what leaves use: verbatim takes the CSR's
762    /// own extensions, and a request that asked to be an unconstrained CA would
763    /// get to be one. `sign-intermediate` with `use_csr_values: false` means
764    /// AXIAM states what the certificate is and Vault ignores what the request
765    /// wanted.
766    fn sign_intermediate_csr<'a>(
767        &'a self,
768        parent: &'a CaKeyRef,
769        request: &'a IntermediateSigningRequest,
770    ) -> Pin<Box<dyn Future<Output = AxiamResult<SignedLeaf>> + Send + 'a>> {
771        Box::pin(async move {
772            let parent_locator = VaultPkiLocator::parse(&parent.locator)?;
773            let (certificate_pem, issuing_ca) = self
774                .sign_as_intermediate(
775                    parent_locator.signing_issuer(),
776                    &request.csr_pem,
777                    &request.subject,
778                    format!("{}s", request.ttl_seconds.max(1)),
779                )
780                .await?;
781            Ok(SignedLeaf {
782                certificate_pem,
783                chain_pem: issuing_ca.into_iter().collect(),
784            })
785        })
786    }

801/// Vault's `key_type` and `key_bits` for one of AXIAM's algorithms.
802///
803/// Note what this makes possible: Vault generates RSA keys, and rcgen's `ring`
804/// backend does not. `Rsa4096` is refused by `POST .../ca-certificates` under
805/// every other custodian and works under this one.
806fn key_params(algorithm: &KeyAlgorithm) -> (&'static str, u32) {
807    match algorithm {
808        KeyAlgorithm::Ed25519 => ("ed25519", 0),
809        KeyAlgorithm::Rsa4096 => ("rsa", 4096),
810    }
811}
812
813/// Days as a Vault duration string.
814fn ttl_days(days: u32) -> String {
815    format!("{}h", u64::from(days) * 24)
816}
817
818/// Pull a non-empty string field out of a Vault `data` object.
819fn string_field(data: &serde_json::Value, field: &str, url: &str) -> AxiamResult<String> {
820    data.get(field)
821        .and_then(|v| v.as_str())
822        .filter(|s| !s.is_empty())
823        .map(str::to_string)
824        .ok_or_else(|| {
825            AxiamError::Internal(format!(
826                "vault pki: {url} returned no `{field}` — check the mount is a PKI engine"
827            ))
828        })
829}
830
831/// The one issuer a `set-signed` or `import/bundle` call created.
832///
833/// More than one means the bundle carried a chain, which AXIAM does not send:
834/// taking the first would pick an issuer arbitrarily and record a locator that
835/// signs with something other than the certificate in the row.
836fn first_imported_issuer(data: &serde_json::Value, url: &str) -> AxiamResult<String> {
837    let imported: Vec<&str> = data
838        .get("imported_issuers")
839        .and_then(|v| v.as_array())
840        .map(|a| a.iter().filter_map(|v| v.as_str()).collect())
841        .unwrap_or_default();
842
843    match imported.as_slice() {
844        [one] => Ok((*one).to_string()),
845        [] => Err(AxiamError::Validation {
846            message: format!(
847                "vault pki: {url} imported no issuer — Vault already holds this \
848                 certificate, so it belongs to a CA that exists"
849            ),
850        }),
851        many => Err(AxiamError::Validation {
852            message: format!(
853                "vault pki: {url} imported {} issuers — send one certificate, not a chain",
854                many.len()
855            ),
856        }),
857    }
858}
859
860/// Vault's `errors` array, flattened for a message.
861fn vault_errors(payload: &serde_json::Value) -> String {
862    let errors: Vec<&str> = payload
863        .get("errors")
864        .and_then(|v| v.as_array())
865        .map(|a| a.iter().filter_map(|v| v.as_str()).collect())
866        .unwrap_or_default();
867    if errors.is_empty() {
868        "no error detail".to_string()
869    } else {
870        errors.join("; ")
871    }
872}
873
874/// Surface Vault's warnings, which are how a capped TTL is reported.
875fn log_warnings(url: &str, payload: &serde_json::Value) {
876    if let Some(warnings) = payload.get("warnings").and_then(|v| v.as_array()) {
877        for warning in warnings.iter().filter_map(|v| v.as_str()) {
878            tracing::warn!(%url, warning, "vault pki returned a warning");
879        }
880    }
881}
