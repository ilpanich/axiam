1//! SCIM error responses (RFC 7644 §3.12).
2//!
3//! A SCIM error body is NOT the same shape as AXIAM's ordinary REST error
4//! envelope (`axiam_api_rest::error::AxiamApiError`) — it has its own
5//! `schemas`/`status`/`scimType`/`detail` fields, mandated by the spec so
6//! generic SCIM clients (Okta, Entra, ...) can parse failures uniformly
7//! across every vendor's implementation. [`ScimError`] is that shape;
8//! `From<AxiamError>`/`From<AxiamApiError>` translate AXIAM's own error
9//! taxonomy into it so handlers can freely `?`-propagate repository and
10//! authorization errors without hand-mapping each call site.
11
12use actix_web::http::StatusCode;
13use actix_web::{HttpResponse, ResponseError};
14use axiam_api_rest::AxiamApiError;
15use axiam_core::error::AxiamError;
16use serde::Serialize;
17
18/// `urn:ietf:params:scim:api:messages:2.0:Error` (RFC 7644 §3.12).
19pub const SCIM_ERROR_SCHEMA: &str = "urn:ietf:params:scim:api:messages:2.0:Error";
20
21#[derive(Debug)]
22pub struct ScimError {
23    pub status: StatusCode,
24    /// RFC 7644 §3.12 `scimType` — a detail code for 400/409 responses.
25    /// `None` for status codes the RFC doesn't define a `scimType` table for
26    /// (401/403/404/5xx).
27    pub scim_type: Option<&'static str>,
28    pub detail: String,
29}
30
31impl ScimError {
32    pub fn new(status: StatusCode, detail: impl Into<String>) -> Self {
33        Self {
34            status,
35            scim_type: None,
36            detail: detail.into(),
37        }
38    }
39
40    pub fn with_type(
41        status: StatusCode,
42        scim_type: &'static str,
43        detail: impl Into<String>,
44    ) -> Self {
45        Self {
46            status,
47            scim_type: Some(scim_type),
48            detail: detail.into(),
49        }
50    }
51
52    pub fn not_found(resource_type: &str, id: &str) -> Self {
53        Self::new(
54            StatusCode::NOT_FOUND,
55            format!("{resource_type} {id} not found"),
56        )
57    }
58
59    /// RFC 7644 §3.12: 400 with `scimType: invalidFilter` — the caller sent a
60    /// filter expression outside the supported subset (`userName eq`,
61    /// `externalId eq`). Complex filters are explicitly out of scope (B4).
62    pub fn invalid_filter(detail: impl Into<String>) -> Self {
63        Self::with_type(StatusCode::BAD_REQUEST, "invalidFilter", detail)
64    }
65
66    /// RFC 7644 §3.12: 400 with `scimType: invalidPath` — an unsupported (or
67    /// malformed) PATCH `path`. B4 restricts PATCH to the standard attribute
68    /// paths Okta/Entra actually send; anything else lands here.
69    pub fn invalid_path(detail: impl Into<String>) -> Self {
70        Self::with_type(StatusCode::BAD_REQUEST, "invalidPath", detail)
71    }
72
73    /// RFC 7644 §3.12: 400 with `scimType: invalidValue`.
74    pub fn invalid_value(detail: impl Into<String>) -> Self {
75        Self::with_type(StatusCode::BAD_REQUEST, "invalidValue", detail)
76    }
77
78    /// RFC 7644 §3.12: 400 with `scimType: mutability` — an attempt to write
79    /// a read-only/immutable attribute.
80    pub fn mutability(detail: impl Into<String>) -> Self {
81        Self::with_type(StatusCode::BAD_REQUEST, "mutability", detail)
82    }
83
84    /// RFC 7644 §3.12: 501 Not Implemented — bulk operations and complex
85    /// filters are explicitly out of scope for this crate (B4). The RFC does
86    /// not define a `scimType` for 501, so `scim_type` stays `None`.
87    pub fn not_implemented(detail: impl Into<String>) -> Self {
88        Self::new(StatusCode::NOT_IMPLEMENTED, detail)
89    }
90}
91
92impl std::fmt::Display for ScimError {
93    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
94        write!(f, "SCIM error {}: {}", self.status, self.detail)
95    }
96}

100#[derive(Serialize)]
101struct ScimErrorBody {
102    schemas: [&'static str; 1],
103    status: String,
104    #[serde(skip_serializing_if = "Option::is_none", rename = "scimType")]
105    scim_type: Option<&'static str>,
106    detail: String,
107}
108
109impl ResponseError for ScimError {
110    fn status_code(&self) -> StatusCode {
111        self.status
112    }
113
114    fn error_response(&self) -> HttpResponse {
115        if self.status.is_server_error() {
116            // SEC-011/CQ-B33 parity with AxiamApiError: never leak internal
117            // detail (DB strings, etc.) in a 5xx body — log it instead.
118            tracing::error!(status = %self.status, detail = %self.detail, "SCIM internal error");
119            return HttpResponse::build(self.status).json(ScimErrorBody {
120                schemas: [SCIM_ERROR_SCHEMA],
121                status: self.status.as_u16().to_string(),
122                scim_type: None,
123                detail: "An internal error occurred".to_string(),
124            });
125        }
126        HttpResponse::build(self.status).json(ScimErrorBody {
127            schemas: [SCIM_ERROR_SCHEMA],
128            status: self.status.as_u16().to_string(),
129            scim_type: self.scim_type,
130            detail: self.detail.clone(),
131        })
132    }
133}

135/// Translate AXIAM's own error taxonomy into the SCIM error shape so
136/// handlers can `?`-propagate repository calls directly.
137impl From<AxiamError> for ScimError {
138    fn from(err: AxiamError) -> Self {
139        match &err {
140            AxiamError::NotFound { entity, id } => {
141                Self::new(StatusCode::NOT_FOUND, format!("{entity} {id} not found"))
142            }
143            AxiamError::AlreadyExists { entity } => Self::with_type(
144                StatusCode::CONFLICT,
145                "uniqueness",
146                format!("{entity} already exists"),
147            ),
148            AxiamError::AuthenticationFailed { reason } => {
149                Self::new(StatusCode::UNAUTHORIZED, reason.clone())
150            }
151            AxiamError::AuthorizationDenied { reason, .. } => {
152                Self::new(StatusCode::FORBIDDEN, reason.clone())
153            }
154            AxiamError::Validation { message } => Self::invalid_value(message.clone()),
155            AxiamError::PasswordPolicy { message } => Self::invalid_value(message.clone()),
156            AxiamError::TenantContext => {
157                Self::new(StatusCode::BAD_REQUEST, "tenant context missing or invalid")
158            }
159            AxiamError::RateLimited => {
160                Self::new(StatusCode::TOO_MANY_REQUESTS, "rate limit exceeded")
161            }
162            AxiamError::ServiceUnavailable(msg) => {
163                Self::new(StatusCode::SERVICE_UNAVAILABLE, msg.clone())
164            }
165            // 5xx-shaped variants: message content is dropped by
166            // `error_response`'s server-error branch above regardless of what
167            // we put in `detail`, so a plain `to_string()` here is fine.
168            _ => Self::new(StatusCode::INTERNAL_SERVER_ERROR, err.to_string()),
169        }
170    }
171}

173/// `RequirePermission::check` returns `AxiamApiError` (axiam-api-rest's own
174/// newtype). Unwrap it back to `AxiamError` and reuse the mapping above so
175/// SCIM authorization denials get a proper SCIM error body instead of
176/// AXIAM's ordinary REST envelope.
177impl From<AxiamApiError> for ScimError {
178    fn from(err: AxiamApiError) -> Self {
179        err.0.into()
180    }
181}

183#[cfg(test)]
184mod tests {
185    use super::*;
186
187    /// The whole point of this module: a SCIM client must never receive AXIAM's
188    /// ordinary REST envelope. Each arm below pins the STATUS and the RFC 7644
189    /// `scimType` a generic SCIM client (Okta, Entra) dispatches on — getting
190    /// the status right but the `scimType` wrong is a silent interop failure,
191    /// because those clients branch on `scimType` to decide whether to retry,
192    /// re-map an attribute, or surface a conflict to an administrator.
193    #[test]
194    fn axiam_errors_map_to_the_rfc_7644_status_and_scim_type() {
195        let cases: Vec<(AxiamError, StatusCode, Option<&'static str>)> = vec![
196            (
197                AxiamError::NotFound {
198                    entity: "User".into(),
199                    id: "abc".into(),
200                },
201                StatusCode::NOT_FOUND,
202                None,
203            ),
204            // `uniqueness` is the one a provisioning client acts on: it means
205            // "this user already exists", not "retry later".
206            (
207                AxiamError::AlreadyExists {
208                    entity: "User".into(),
209                },
210                StatusCode::CONFLICT,
211                Some("uniqueness"),
212            ),
213            (
214                AxiamError::AuthenticationFailed {
215                    reason: "bad token".into(),
216                },
217                StatusCode::UNAUTHORIZED,
218                None,
219            ),
220            (
221                AxiamError::AuthorizationDenied {
222                    reason: "no grant".into(),
223                    action: Some("users:create".into()),
224                    resource_id: None,
225                },
226                StatusCode::FORBIDDEN,
227                None,
228            ),
229            (
230                AxiamError::Validation {
231                    message: "userName is required".into(),
232                },
233                StatusCode::BAD_REQUEST,
234                Some("invalidValue"),
235            ),
236            (
237                AxiamError::PasswordPolicy {
238                    message: "too short".into(),
239                },
240                StatusCode::BAD_REQUEST,
241                Some("invalidValue"),
242            ),
243            (AxiamError::TenantContext, StatusCode::BAD_REQUEST, None),
244            (AxiamError::RateLimited, StatusCode::TOO_MANY_REQUESTS, None),
245            (
246                AxiamError::ServiceUnavailable("db down".into()),
247                StatusCode::SERVICE_UNAVAILABLE,
248                None,
249            ),
250            // The catch-all: every remaining variant is 5xx-shaped.
251            (
252                AxiamError::Database("connection reset".into()),
253                StatusCode::INTERNAL_SERVER_ERROR,
254                None,
255            ),
256            (
257                AxiamError::Internal("boom".into()),
258                StatusCode::INTERNAL_SERVER_ERROR,
259                None,
260            ),
261        ];
262
263        for (err, want_status, want_type) in cases {
264            let rendered = format!("{err}");
265            let scim: ScimError = err.into();
266            assert_eq!(scim.status, want_status, "status for {rendered}");
267            assert_eq!(scim.scim_type, want_type, "scimType for {rendered}");
268        }
269    }
270
271    #[test]
272    fn not_found_detail_names_the_entity_and_id() {
273        let scim: ScimError = AxiamError::NotFound {
274            entity: "Group".into(),
275            id: "g-1".into(),
276        }
277        .into();
278        assert!(scim.detail.contains("Group"), "detail: {}", scim.detail);
279        assert!(scim.detail.contains("g-1"), "detail: {}", scim.detail);
280    }

282    #[test]
283    fn status_code_is_the_status_the_error_carries() {
284        // ResponseError::status_code is what Actix uses to build the response
285        // line; a constructor that set `status` without it being reflected here
286        // would send a 200 with an error body.
287        let e = ScimError::not_implemented("bulk is out of scope");
288        assert_eq!(ResponseError::status_code(&e), StatusCode::NOT_IMPLEMENTED);
289        assert_eq!(e.scim_type, None);
290
291        let m = ScimError::mutability("readOnly attribute");
292        assert_eq!(ResponseError::status_code(&m), StatusCode::BAD_REQUEST);
293        assert_eq!(m.scim_type, Some("mutability"));
294    }

296    #[test]
297    fn display_carries_the_status_and_detail_for_logs() {
298        let e = ScimError::new(StatusCode::NOT_FOUND, "User 42 not found");
299        let s = e.to_string();
300        assert!(s.contains("404"), "{s}");
301        assert!(s.contains("User 42 not found"), "{s}");
302    }

304    /// SEC-011: a 5xx body must not echo the internal detail — the DB string
305    /// that produced it goes to the log, not to a SCIM client. This is the
306    /// asymmetry worth pinning, since the 4xx branch DOES echo detail.
307    #[test]
308    fn server_errors_do_not_leak_detail_into_the_body() {
309        let internal: ScimError = AxiamError::Database("host=db-1 user=axiam".into()).into();
310        let resp = ResponseError::error_response(&internal);
311        assert_eq!(resp.status(), StatusCode::INTERNAL_SERVER_ERROR);
312
313        let client_error = ScimError::invalid_value("userName is required");
314        let resp = ResponseError::error_response(&client_error);
315        assert_eq!(resp.status(), StatusCode::BAD_REQUEST);
316    }
