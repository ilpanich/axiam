1//! SCIM error responses (RFC 7644 §3.12).
2//!
3//! A SCIM error body is NOT the same shape as AXIAM's ordinary REST error
4//! envelope (`axiam_api_rest::error::AxiamApiError`) — it has its own
5//! `schemas`/`status`/`scimType`/`detail` fields, mandated by the spec so
6//! generic SCIM clients (Okta, Entra, ...) can parse failures uniformly
7//! across every vendor's implementation. [`ScimError`] is that shape;
8//! `From<AxiamError>`/`From<AxiamApiError>` translate AXIAM's own error
9//! taxonomy into it so handlers can freely `?`-propagate repository and
10//! authorization errors without hand-mapping each call site.
11
12use actix_web::http::{StatusCode, header};
13use actix_web::{HttpResponse, ResponseError};
14use axiam_api_rest::AxiamApiError;
15use axiam_core::error::AxiamError;
16use serde::Serialize;

18/// `urn:ietf:params:scim:api:messages:2.0:Error` (RFC 7644 §3.12).
19pub const SCIM_ERROR_SCHEMA: &str = "urn:ietf:params:scim:api:messages:2.0:Error";
20
21#[derive(Debug)]
22pub struct ScimError {
23    pub status: StatusCode,
24    /// RFC 7644 §3.12 `scimType` — a detail code for 400/409 responses.
25    /// `None` for status codes the RFC doesn't define a `scimType` table for
26    /// (401/403/404/5xx).
27    pub scim_type: Option<&'static str>,
28    pub detail: String,
29    /// Seconds to put in `Retry-After`, set only on a *retry advisory* — a
30    /// transient failure the caller is being told to repeat (T-262 / R-4).
31    ///
32    /// Its presence does two things in [`ResponseError::error_response`]: it
33    /// emits the header, and it exempts the body from the blanket 5xx
34    /// redaction. Set it only from a fixed, payload-free `detail`; see
35    /// [`ScimError::retry_later`].
36    pub retry_after: Option<u32>,
37}
38
39impl ScimError {
40    pub fn new(status: StatusCode, detail: impl Into<String>) -> Self {
41        Self {
42            status,
43            scim_type: None,
44            detail: detail.into(),
45            retry_after: None,
46        }
47    }
48
49    pub fn with_type(
50        status: StatusCode,
51        scim_type: &'static str,
52        detail: impl Into<String>,
53    ) -> Self {
54        Self {
55            status,
56            scim_type: Some(scim_type),
57            detail: detail.into(),
58            retry_after: None,
59        }
60    }
61
62    /// A transient failure the caller is being told to repeat: `503` plus the
63    /// `Retry-After: secs` that makes the status actionable (T-262 / R-4).
64    ///
65    /// Unlike every other 5xx this type produces, the `detail` reaches the
66    /// client verbatim — so it MUST be a fixed sentence carrying no payload.
67    /// Nothing derived from a datastore message, a query, or a caller's input
68    /// may be interpolated into it.
69    pub fn retry_later(detail: impl Into<String>, secs: u32) -> Self {
70        Self {
71            status: StatusCode::SERVICE_UNAVAILABLE,
72            // RFC 7644 §3.12 defines no `scimType` for 503.
73            scim_type: None,
74            detail: detail.into(),
75            retry_after: Some(secs),
76        }
77    }
78
79    pub fn not_found(resource_type: &str, id: &str) -> Self {
80        Self::new(
81            StatusCode::NOT_FOUND,
82            format!("{resource_type} {id} not found"),
83        )
84    }
85
86    /// RFC 7644 §3.12: 400 with `scimType: invalidFilter` — the caller sent a
87    /// filter expression outside the supported subset (`userName eq`,
88    /// `externalId eq`). Complex filters are explicitly out of scope (B4).
89    pub fn invalid_filter(detail: impl Into<String>) -> Self {
90        Self::with_type(StatusCode::BAD_REQUEST, "invalidFilter", detail)
91    }
92
93    /// RFC 7644 §3.12: 400 with `scimType: invalidPath` — an unsupported (or
94    /// malformed) PATCH `path`. B4 restricts PATCH to the standard attribute
95    /// paths Okta/Entra actually send; anything else lands here.
96    pub fn invalid_path(detail: impl Into<String>) -> Self {
97        Self::with_type(StatusCode::BAD_REQUEST, "invalidPath", detail)
98    }
99
100    /// RFC 7644 §3.12: 400 with `scimType: invalidValue`.
101    pub fn invalid_value(detail: impl Into<String>) -> Self {
102        Self::with_type(StatusCode::BAD_REQUEST, "invalidValue", detail)
103    }
104
105    /// RFC 7644 §3.12: 400 with `scimType: mutability` — an attempt to write
106    /// a read-only/immutable attribute.
107    pub fn mutability(detail: impl Into<String>) -> Self {
108        Self::with_type(StatusCode::BAD_REQUEST, "mutability", detail)
109    }
110
111    /// RFC 7644 §3.12: 501 Not Implemented — bulk operations and complex
112    /// filters are explicitly out of scope for this crate (B4). The RFC does
113    /// not define a `scimType` for 501, so `scim_type` stays `None`.
114    pub fn not_implemented(detail: impl Into<String>) -> Self {
115        Self::new(StatusCode::NOT_IMPLEMENTED, detail)
116    }
117}
118
119impl std::fmt::Display for ScimError {
120    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
121        write!(f, "SCIM error {}: {}", self.status, self.detail)
122    }
123}

127#[derive(Serialize)]
128struct ScimErrorBody {
129    schemas: [&'static str; 1],
130    status: String,
131    #[serde(skip_serializing_if = "Option::is_none", rename = "scimType")]
132    scim_type: Option<&'static str>,
133    detail: String,
134}
135
136impl ResponseError for ScimError {
137    fn status_code(&self) -> StatusCode {
138        self.status
139    }
140
141    fn error_response(&self) -> HttpResponse {
142        let mut builder = HttpResponse::build(self.status);
143        // T-262 / R-4: the half of the contract that makes a 503 useful.
144        // CONTRACT §16.1 makes every SDK honour `Retry-After` as a **floor**,
145        // so the caller's own backoff still governs the wait. Set from the
146        // same field the redaction carve-out below reads, so the header and
147        // the body it explains cannot drift apart.
148        if let Some(secs) = self.retry_after {
149            builder.insert_header((header::RETRY_AFTER, secs.to_string()));
150        }
151
152        if self.status.is_server_error() {
153            if self.retry_after.is_none() {
154                // SEC-011/CQ-B33 parity with AxiamApiError: never leak internal
155                // detail (DB strings, etc.) in a 5xx body — log it instead.
156                tracing::error!(status = %self.status, detail = %self.detail, "SCIM internal error");
157                return builder.json(ScimErrorBody {
158                    schemas: [SCIM_ERROR_SCHEMA],
159                    status: self.status.as_u16().to_string(),
160                    scim_type: None,
161                    detail: "An internal error occurred".to_string(),
162                });
163            }
164            // A retry advisory is the one 5xx whose detail is echoed: it is a
165            // fixed constant (see `retry_later`), and redacting it would turn
166            // "retry this request" into "the server broke" — a dead end for
167            // the provisioning IdP the `Retry-After` above is aimed at. Still
168            // logged, at WARN rather than ERROR: sustained contention is worth
169            // an operator's attention, but it is not a fault.
170            tracing::warn!(status = %self.status, detail = %self.detail, "SCIM transient failure");
171        }
172
173        builder.json(ScimErrorBody {
174            schemas: [SCIM_ERROR_SCHEMA],
175            status: self.status.as_u16().to_string(),
176            scim_type: self.scim_type,
177            detail: self.detail.clone(),
178        })
179    }
180}

182/// Translate AXIAM's own error taxonomy into the SCIM error shape so
183/// handlers can `?`-propagate repository calls directly.
184impl From<AxiamError> for ScimError {
185    fn from(err: AxiamError) -> Self {
186        match &err {
187            AxiamError::NotFound { entity, id } => {
188                Self::new(StatusCode::NOT_FOUND, format!("{entity} {id} not found"))
189            }
190            AxiamError::AlreadyExists { entity } => Self::with_type(
191                StatusCode::CONFLICT,
192                "uniqueness",
193                format!("{entity} already exists"),
194            ),
195            AxiamError::AuthenticationFailed { reason } => {
196                Self::new(StatusCode::UNAUTHORIZED, reason.clone())
197            }
198            AxiamError::AuthorizationDenied { reason, .. } => {
199                Self::new(StatusCode::FORBIDDEN, reason.clone())
200            }
201            AxiamError::Validation { message } => Self::invalid_value(message.clone()),
202            AxiamError::PasswordPolicy { message } => Self::invalid_value(message.clone()),
203            AxiamError::TenantContext => {
204                Self::new(StatusCode::BAD_REQUEST, "tenant context missing or invalid")
205            }
206            AxiamError::RateLimited => {
207                Self::new(StatusCode::TOO_MANY_REQUESTS, "rate limit exceeded")
208            }
209            AxiamError::ServiceUnavailable(msg) => {
210                Self::new(StatusCode::SERVICE_UNAVAILABLE, msg.clone())
211            }
212            // T-262 / R-4, completing `3ccef6a30`, which wired this variant
213            // into REST and gRPC but not here. A SCIM caller that lost an
214            // optimistic-concurrency race must be told to repeat the request,
215            // not that the server broke: an IdP reading a 500 marks the sync
216            // failed and re-sends the whole record. `err.to_string()` is safe
217            // to echo because the variant carries no payload — the engine's
218            // own words stay on `DbError::Conflict`, in the log.
219            //
220            // Not a 409/`uniqueness`: RFC 7644 §3.12 reads that as "your
221            // request conflicts with the resource's state", which would have
222            // the client change the request rather than resend it.
223            AxiamError::WriteContention => Self::retry_later(err.to_string(), 1),
224            // 5xx-shaped variants: message content is dropped by
225            // `error_response`'s server-error branch above regardless of what
226            // we put in `detail`, so a plain `to_string()` here is fine.
227            _ => Self::new(StatusCode::INTERNAL_SERVER_ERROR, err.to_string()),
228        }
229    }
230}

232/// `RequirePermission::check` returns `AxiamApiError` (axiam-api-rest's own
233/// newtype). Unwrap it back to `AxiamError` and reuse the mapping above so
234/// SCIM authorization denials get a proper SCIM error body instead of
235/// AXIAM's ordinary REST envelope.
236impl From<AxiamApiError> for ScimError {
237    fn from(err: AxiamApiError) -> Self {
238        err.0.into()
239    }
240}

248    /// The whole point of this module: a SCIM client must never receive AXIAM's
249    /// ordinary REST envelope. Each arm below pins the STATUS and the RFC 7644
250    /// `scimType` a generic SCIM client (Okta, Entra) dispatches on — getting
251    /// the status right but the `scimType` wrong is a silent interop failure,
252    /// because those clients branch on `scimType` to decide whether to retry,
253    /// re-map an attribute, or surface a conflict to an administrator.
254    #[test]
255    fn axiam_errors_map_to_the_rfc_7644_status_and_scim_type() {
256        let cases: Vec<(AxiamError, StatusCode, Option<&'static str>)> = vec![
257            (
258                AxiamError::NotFound {
259                    entity: "User".into(),
260                    id: "abc".into(),
261                },
262                StatusCode::NOT_FOUND,
263                None,
264            ),
265            // `uniqueness` is the one a provisioning client acts on: it means
266            // "this user already exists", not "retry later".
267            (
268                AxiamError::AlreadyExists {
269                    entity: "User".into(),
270                },
271                StatusCode::CONFLICT,
272                Some("uniqueness"),
273            ),
274            (
275                AxiamError::AuthenticationFailed {
276                    reason: "bad token".into(),
277                },
278                StatusCode::UNAUTHORIZED,
279                None,
280            ),
281            (
282                AxiamError::AuthorizationDenied {
283                    reason: "no grant".into(),
284                    action: Some("users:create".into()),
285                    resource_id: None,
286                },
287                StatusCode::FORBIDDEN,
288                None,
289            ),
290            (
291                AxiamError::Validation {
292                    message: "userName is required".into(),
293                },
294                StatusCode::BAD_REQUEST,
295                Some("invalidValue"),
296            ),
297            (
298                AxiamError::PasswordPolicy {
299                    message: "too short".into(),
300                },
301                StatusCode::BAD_REQUEST,
302                Some("invalidValue"),
303            ),
304            (AxiamError::TenantContext, StatusCode::BAD_REQUEST, None),
305            (AxiamError::RateLimited, StatusCode::TOO_MANY_REQUESTS, None),
306            (
307                AxiamError::ServiceUnavailable("db down".into()),
308                StatusCode::SERVICE_UNAVAILABLE,
309                None,
310            ),
311            // T-262 / R-4: 503, NOT the 500 the catch-all used to give it, and
312            // not the 409 `uniqueness` that would have the client rewrite the
313            // request instead of repeating it.
314            (
315                AxiamError::WriteContention,
316                StatusCode::SERVICE_UNAVAILABLE,
317                None,
318            ),
319            // The catch-all: every remaining variant is 5xx-shaped.
320            (
321                AxiamError::Database("connection reset".into()),
322                StatusCode::INTERNAL_SERVER_ERROR,
323                None,
324            ),
325            (
326                AxiamError::Internal("boom".into()),
327                StatusCode::INTERNAL_SERVER_ERROR,
328                None,
329            ),
330        ];
331
332        for (err, want_status, want_type) in cases {
333            let rendered = format!("{err}");
334            let scim: ScimError = err.into();
335            assert_eq!(scim.status, want_status, "status for {rendered}");
336            assert_eq!(scim.scim_type, want_type, "scimType for {rendered}");
337        }
338    }

373    /// Render an error the way Actix will, so a test can read the status, the
374    /// `Retry-After` header and the JSON body a SCIM client actually receives
375    /// — the three things the T-262 contract is written in terms of.
376    fn render(err: ScimError) -> (StatusCode, Option<String>, serde_json::Value) {
377        let response = ResponseError::error_response(&err);
378        let status = response.status();
379        let retry_after = response
380            .headers()
381            .get(header::RETRY_AFTER)
382            .map(|v| v.to_str().unwrap().to_owned());
383        let bytes = response.into_body().try_into_bytes().unwrap();
384        (status, retry_after, serde_json::from_slice(&bytes).unwrap())
385    }

387    /// SEC-011: a 5xx body must not echo the internal detail — the DB string
388    /// that produced it goes to the log, not to a SCIM client. This is the
389    /// asymmetry worth pinning, since the 4xx branch DOES echo detail.
390    ///
391    /// It is also the **control** for the `WriteContention` carve-out below:
392    /// that carve-out must stay one variant wide, so this asserts the actual
393    /// body, not merely the status.
394    #[test]
395    fn server_errors_do_not_leak_detail_into_the_body() {
396        let (status, retry_after, body) =
397            render(AxiamError::Database("host=db-1 user=axiam".into()).into());
398        assert_eq!(status, StatusCode::INTERNAL_SERVER_ERROR);
399        assert_eq!(retry_after, None, "a 500 must not advertise a retry");
400        assert_eq!(body["detail"], "An internal error occurred");
401        let rendered = body.to_string();
402        for leak in ["host=db-1", "user=axiam"] {
403            assert!(
404                !rendered.contains(leak),
405                "{leak:?} must not appear in the response body: {rendered}"
406            );
407        }
408
409        let client_error = ScimError::invalid_value("userName is required");
410        let resp = ResponseError::error_response(&client_error);
411        assert_eq!(resp.status(), StatusCode::BAD_REQUEST);
412    }

414    /// T-262 / R-4, completing `3ccef6a30`. A SCIM caller that loses an
415    /// optimistic-concurrency race is told to come back, not that the server
416    /// broke — the whole point being that an IdP (Okta, Entra) can tell a
417    /// repeatable request from a failed sync it must re-send in full.
418    #[test]
419    fn a_contended_write_answers_503_with_retry_after() {
420        let (status, retry_after, body) = render(AxiamError::WriteContention.into());
421        assert_eq!(status, StatusCode::SERVICE_UNAVAILABLE);
422        assert_ne!(
423            status,
424            StatusCode::INTERNAL_SERVER_ERROR,
425            "the catch-all must no longer swallow this variant"
426        );
427        assert_eq!(retry_after.as_deref(), Some("1"));
428        assert_eq!(body["status"], "503");
429        // RFC 7644 §3.12 defines no `scimType` for 5xx.
430        assert_eq!(body.get("scimType"), None);
431    }

433    /// The half a naïve match arm alone would still get wrong: the blanket 5xx
434    /// redaction would replace the one detail that is worth sending, leaving a
435    /// client with a `Retry-After` and a body saying the server is broken.
436    #[test]
437    fn the_retry_advisory_detail_survives_the_5xx_redaction() {
438        let (_, _, body) = render(AxiamError::WriteContention.into());
439        assert_ne!(
440            body["detail"], "An internal error occurred",
441            "the carve-out is what makes the 503 actionable"
442        );
443        assert_eq!(body["detail"], AxiamError::WriteContention.to_string());
444    }

446    /// The variant carries no payload, so the engine's own words cannot reach
447    /// the body by accident — only by somebody giving it a payload "for
448    /// debugging". That is what this pins, now that the detail is echoed.
449    #[test]
450    fn the_retry_advisory_never_carries_the_engines_message() {

455                !rendered.contains(leak),
456                "{leak:?} must not appear in the response body: {rendered}"

458        }
