1//! The provisioning **source**: [`ScimProvisioner`] turns a repository's
2//! committed-change report into one reference per enabled target (D-57).
3//!
4//! What travels is `{resource_type, axiam_id}` and nothing else. The deliverer
5//! computes what to send from the data as it is at the attempt, so a queued
6//! message — or a dead-lettered one, which lives for seven days — names a
7//! resource by id and tells nobody anything about the person.
8
9use std::sync::Arc;
10
11use axiam_core::models::scim_target::{ScimResourceType, ScimTarget, ScimTargetScope};
12use axiam_core::outbound::{OutboundKind, OutboundMessage, OutboundPublisher};
13use axiam_core::provisioning::{ProvisioningFuture, ProvisioningSink};
14use axiam_core::repository::ScimTargetRepository;
15use serde_json::json;
16use uuid::Uuid;
17
18/// The `event_type` of a user reference.
19pub const USER_EVENT_TYPE: &str = "scim.user";
20/// The `event_type` of a group reference.
21pub const GROUP_EVENT_TYPE: &str = "scim.group";
22
23/// The message that tells a target's deliverer to re-sync one resource.
24///
25/// A fresh `delivery_id` per message; `attempt` is `0`, as for every producer
26/// of the dispatcher (the consumer loop owns the counter).
27#[must_use]
28pub fn reference_message(
29    tenant_id: Uuid,
30    target_id: Uuid,
31    resource_type: ScimResourceType,
32    axiam_id: Uuid,
33) -> OutboundMessage {
34    OutboundMessage {
35        kind: OutboundKind::ScimPush,
36        tenant_id,
37        target_id,
38        delivery_id: Uuid::new_v4(),
39        event_type: match resource_type {
40            ScimResourceType::User => USER_EVENT_TYPE,
41            ScimResourceType::Group => GROUP_EVENT_TYPE,
42        }
43        .to_owned(),
44        payload: json!({
45            "resource_type": resource_type.as_str(),
46            "axiam_id": axiam_id.to_string(),
47        }),
48        attempt: 0,
49    }
50}
51
52/// Whether `target` pushes `group_id`: `push_groups` is on, and the group is in
53/// scope (every group for `all_users`, the listed ones otherwise — D-57).
54#[must_use]
55pub fn group_in_scope(target: &ScimTarget, group_id: Uuid) -> bool {
56    target.push_groups
57        && match &target.scope {
58            ScimTargetScope::AllUsers => true,
59            ScimTargetScope::Groups(listed) => listed.contains(&group_id),
60        }
61}

63/// The [`ProvisioningSink`] a deployment binds: for each **enabled** target of
64/// the tenant, one reference per affected resource on the shared outbound
65/// dispatcher.
66///
67/// | Report | Enqueued, per enabled target |
68/// |---|---|
69/// | `user_changed` | the user |
70/// | `group_changed` | the group, when the target pushes it (`push_groups`, in scope) |
71/// | `membership_changed` | the user (its scope may have changed) and, when the target pushes it, the group |
72///
73/// A failure to list the targets or to enqueue is logged once per report and
74/// never reaches the repository that reported: the write it follows has
75/// committed, and the next change of the resource (or reconciliation) converges.
76pub struct ScimProvisioner<T> {
77    targets: T,
78    publisher: Arc<dyn OutboundPublisher>,
79}
80
81impl<T: ScimTargetRepository + 'static> ScimProvisioner<T> {
82    /// A provisioner over the target registry and the dispatcher's publisher.
83    pub fn new(targets: T, publisher: Arc<dyn OutboundPublisher>) -> Self {
84        Self { targets, publisher }
85    }
86
87    /// Enqueue `wanted(target)` for every enabled target of the tenant.
88    async fn fan_out(
89        &self,
90        tenant_id: Uuid,
91        wanted: impl Fn(&ScimTarget) -> Vec<(ScimResourceType, Uuid)>,
92    ) {
93        let targets = match self.targets.list_enabled(tenant_id).await {
94            Ok(targets) => targets,
95            Err(error) => {
96                tracing::warn!(
97                    target: "axiam::scim_push",
98                    %tenant_id,
99                    %error,
100                    "the SCIM targets could not be listed; a provisioning change was not queued"
101                );
102                return;
103            }
104        };
105        let mut failed = 0usize;
106        for target in &targets {
107            for (resource_type, axiam_id) in wanted(target) {
108                let message = reference_message(tenant_id, target.id, resource_type, axiam_id);
109                if self.publisher.enqueue(&message).await.is_err() {
110                    failed += 1;
111                }
112            }
113        }
114        if failed > 0 {
115            // Once per report, not once per message: a broker outage must not
116            // write a line for every target of every change.
117            tracing::warn!(
118                target: "axiam::scim_push",
119                %tenant_id,
120                failed,
121                "SCIM provisioning references could not be enqueued"
122            );
123        }
124    }
125}
126
127impl<T: ScimTargetRepository + 'static> ProvisioningSink for ScimProvisioner<T> {
128    fn user_changed(&self, tenant_id: Uuid, user_id: Uuid) -> ProvisioningFuture<'_, ()> {
129        Box::pin(async move {
130            self.fan_out(tenant_id, |_| vec![(ScimResourceType::User, user_id)])
131                .await;
132        })
133    }
134
135    fn group_changed(&self, tenant_id: Uuid, group_id: Uuid) -> ProvisioningFuture<'_, ()> {
136        Box::pin(async move {
137            self.fan_out(tenant_id, |target| {
138                if group_in_scope(target, group_id) {
139                    vec![(ScimResourceType::Group, group_id)]
140                } else {
141                    Vec::new()
142                }
143            })
144            .await;
145        })
146    }
147
148    fn membership_changed(
149        &self,
150        tenant_id: Uuid,
151        group_id: Uuid,
152        user_id: Uuid,
153    ) -> ProvisioningFuture<'_, ()> {
154        Box::pin(async move {
155            self.fan_out(tenant_id, |target| {
156                let mut refs = vec![(ScimResourceType::User, user_id)];
157                if group_in_scope(target, group_id) {
158                    refs.push((ScimResourceType::Group, group_id));
159                }
160                refs
161            })
162            .await;
163        })
164    }
165}
