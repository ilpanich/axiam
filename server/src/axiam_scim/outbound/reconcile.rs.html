1//! Reconciliation: the nightly (and on-demand) repair of a target's drift
2//! (G-6, T23.6.3, D-58).
3//!
4//! Delivery is level-triggered and fed by change reports, so what it cannot see
5//! is a change nobody reported: a downstream administrator who edited or
6//! removed an account, a broker outage that dropped a reference, an erasure
7//! `DELETE` that was refused. Reconciliation is the second source of the same
8//! references, plus a read of the downstream.
9//!
10//! # One run, for one target
11//!
12//! 0. **Claim.** `claim_reconciliation` is a conditional write on the target's
13//!    delivery state: it succeeds for one caller per interval, on any replica,
14//!    and stamps the start of the run. There is no completion write. The
15//!    scheduled job passes [`RECONCILE_INTERVAL`] (24 h); the on-demand entry
16//!    passes [`RECONCILE_WALL_CLOCK`], the longest a run may take.
17//! 1. **Enqueue.** One reference per in-scope user, per pushed group and per
18//!    linked resource, on the shared dispatcher; the deliverer converges each.
19//!    A link left `erase_pending` by a refused erasure `DELETE` is a linked
20//!    resource like any other: this is what retries it.
21//! 2. **Read the downstream.** `GET /Users` (and `/Groups` when the target
22//!    pushes groups), `startIndex`/`count = 100`, every request through the
23//!    deliverer's own guarded, no-redirect, credential-checked path
24//!    (`ScimPushDeliverer::call`; there is no second HTTP path), with a page
25//!    budget and a wall-clock budget. For each downstream resource:
26//!    * **linked** → when what AXIAM would send now differs from it, the
27//!      link's digest is cleared (so the next sync sends it) and the resource
28//!      is queued again;
29//!    * **not linked, `externalId` is the id of an AXIAM resource of this
30//!      tenant that should not be there** (out of scope, disabled, erased) → the
31//!      link is adopted and a reference queued, and the deliverer deprovisions
32//!      it as it would have had the change been reported;
33//!    * **anything else** (no `externalId`, one that is not an AXIAM id of this
34//!      tenant, an id of another tenant) → **never touched**: it is an account
35//!      the downstream's own application made.
36//!
37//!    When a collection was listed to its end, a link whose downstream
38//!    resource was **not** listed is dropped and the resource queued again
39//!    (re-created); a link made after the run began is left alone.
40//!
41//! The run's findings are logged once, as one line, and returned as a
42//! [`ReconcileReport`]. Reasons are a fixed vocabulary: never a URL, a body or
43//! a value.

73/// How long after a run began the scheduled job runs it again: once a day.
74pub const RECONCILE_INTERVAL: Duration = Duration::from_secs(24 * 60 * 60);
75
76/// The longest one run may take. Also the interval an on-demand run is claimed
77/// for: a second request while the first may still be running is "already
78/// claimed".
79pub const RECONCILE_WALL_CLOCK: Duration = Duration::from_secs(5 * 60);
80
81/// Downstream pages read per collection (`/Users`, `/Groups`) per run. With
82/// [`LIST_PAGE_SIZE`] that is at most 10 000 resources of each kind; a larger
83/// downstream is audited in part (the run says so) and never has a link
84/// dropped for being "missing" from a listing that was cut short.
85pub const RECONCILE_MAX_PAGES: u32 = 100;
86
87/// `count` of a downstream list request (RFC 7644 §3.4.2.4).
88pub const LIST_PAGE_SIZE: u32 = 100;
89
90/// The page size of AXIAM's own reads while enqueueing.
91const AXIAM_PAGE: u64 = 100;

93/// How far a run got in reading the downstream.
94#[derive(Debug, Clone, PartialEq, Eq)]
95pub enum Listing {
96    /// Every collection was read to its end.
97    Complete,
98    /// A budget (pages or wall clock) stopped the listing: what was read was
99    /// audited, nothing was concluded from what was not.
100    BudgetExhausted,
101    /// The downstream could not be read; the reason is a fixed phrase.
102    Failed(String),
103}
104
105/// What one reconciliation run found and did.
106#[derive(Debug, Clone, PartialEq, Eq)]
107pub struct ReconcileReport {
108    /// References queued in step 1 and for repairs (a resource queued twice
109    /// counts twice).
110    pub enqueued: u64,
111    /// References that could not be queued (a broker outage).
112    pub enqueue_failed: u64,
113    /// Linked resources left `erase_pending` that were queued for another
114    /// `DELETE`.
115    pub erase_retried: u64,
116    /// Links whose digest was cleared because the downstream differs.
117    pub drift_cleared: u64,
118    /// Links dropped because their downstream resource is gone.
119    pub links_dropped: u64,
120    /// Downstream resources of this tenant that should not be there, adopted
121    /// and queued for deprovisioning.
122    pub deprovisioned: u64,
123    /// Downstream list pages read.
124    pub pages: u32,
125    /// How far the downstream listing got.
126    pub listing: Listing,
127}

151/// What a request to reconcile a target came to.
152#[derive(Debug, Clone, PartialEq, Eq)]
153pub enum ReconcileOutcome {
154    /// The caller held the claim and the run was made.
155    Ran(ReconcileReport),
156    /// Another run holds the claim (or ran within the interval): nothing was

170    /// Another run holds the claim (or ran within the on-demand window).

186/// The scheduled job's view of reconciliation, object-safe so that the cleanup
187/// loop holds one without being generic over the deliverer's repositories.
188pub trait ScimReconciliation: Send + Sync {
189    /// Reconcile every enabled target whose last run is older than
190    /// [`RECONCILE_INTERVAL`] (the claim decides), stopping between targets

192    ///
193    /// # Errors
194    ///
195    /// The enabled targets could not be listed.
196    fn run_due<'a>(
197        &'a self,
198        should_stop: &'a (dyn Fn() -> bool + Send + Sync),
199    ) -> Pin<Box<dyn Future<Output = Result<ReconcileSweep, AxiamError>> + Send + 'a>>;
200}

202/// An `Exit` from a step that reads AXIAM's own datastore: only a datastore
203/// failure can come out of those.
204fn datastore(exit: Exit) -> AxiamError {
205    match exit {
206        Exit::Retry(reason) | Exit::Dead(reason) | Exit::Fail(reason) => {
207            AxiamError::Internal(reason)
208        }
209    }
210}

212/// The reason an `Exit` is reported with: its fixed phrase.

214    match exit {
215        Exit::Retry(reason) | Exit::Dead(reason) | Exit::Fail(reason) => reason,
216    }

227    /// Reconcile one target **on demand**: the same claim as the scheduled
228    /// job, taken for [`RECONCILE_WALL_CLOCK`], so that a request made while a
229    /// run is under way (or has only just finished) is
230    /// [`ReconcileOutcome::AlreadyClaimed`].
231    ///
232    /// # Errors
233    ///
234    /// `NotFound` when the target does not exist in the tenant; any other
235    /// failure of AXIAM's own datastore.
236    pub async fn reconcile_now(
237        &self,
238        tenant_id: Uuid,
239        target_id: Uuid,
240    ) -> Result<ReconcileOutcome, AxiamError> {
241        self.reconcile(tenant_id, target_id, RECONCILE_WALL_CLOCK)
242            .await
243    }

245    /// Reconcile one target **on demand, in the background**: the same claim as
246    /// [`Self::reconcile_now`], but the run is made on a spawned task and the
247    /// answer is given as soon as the claim is decided — what the management
248    /// API's `202` / `409` is (T23.6.4). The run's findings are logged once, as
249    /// they are for every run; nobody awaits them.
250    ///
251    /// # Errors
252    ///
253    /// `NotFound` when the target does not exist in the tenant; any other
254    /// failure of AXIAM's own datastore while claiming.
255    pub async fn start_reconcile_now(
256        self: &Arc<Self>,
257        tenant_id: Uuid,
258        target_id: Uuid,
259    ) -> Result<ReconcileStart, AxiamError> {
260        let seconds = i64::try_from(RECONCILE_WALL_CLOCK.as_secs()).unwrap_or(i64::MAX);
261        match self.claim(tenant_id, target_id, seconds).await? {
262            Err(ReconcileOutcome::TargetDisabled) => Ok(ReconcileStart::TargetDisabled),
263            Err(_) => Ok(ReconcileStart::AlreadyClaimed),
264            Ok((target, started_at)) => {
265                let this = Arc::clone(self);
266                tokio::spawn(async move {
267                    // Findings are logged inside; there is nobody to return them to.
268                    let _ = this.run_and_log(target, started_at).await;
269                });
270                Ok(ReconcileStart::Started)
271            }
272        }
273    }

275    /// Reconcile one target if its last claim is older than `min_interval`.
276    async fn reconcile(
277        &self,
278        tenant_id: Uuid,
279        target_id: Uuid,
280        min_interval: Duration,
281    ) -> Result<ReconcileOutcome, AxiamError> {
282        let seconds = i64::try_from(min_interval.as_secs()).unwrap_or(i64::MAX);
283        match self.claim(tenant_id, target_id, seconds).await? {
284            Err(refused) => Ok(refused),
285            Ok((target, started_at)) => self
286                .run_and_log(target, started_at)
287                .await
288                .map(ReconcileOutcome::Ran),
289        }
290    }

292    /// Read the target and take the claim: `Ok(Ok(..))` when the caller holds
293    /// it, `Ok(Err(outcome))` when nothing was done (disabled, or claimed).
294    async fn claim(
295        &self,
296        tenant_id: Uuid,
297        target_id: Uuid,
298        min_interval_secs: i64,
299    ) -> Result<Result<(ScimTarget, DateTime<Utc>), ReconcileOutcome>, AxiamError> {
300        let target = self.targets.get(tenant_id, target_id).await?;
301        if !target.enabled {
302            return Ok(Err(ReconcileOutcome::TargetDisabled));
303        }
304        let started_at = Utc::now();
305        if !self
306            .state
307            .claim_reconciliation(tenant_id, target_id, started_at, min_interval_secs)
308            .await?
309        {
310            return Ok(Err(ReconcileOutcome::AlreadyClaimed));
311        }
312        Ok(Ok((target, started_at)))
313    }

315    /// Make the run the caller claimed, and log its findings once.
316    async fn run_and_log(
317        &self,
318        target: ScimTarget,
319        started_at: DateTime<Utc>,
320    ) -> Result<ReconcileReport, AxiamError> {
321        let (tenant_id, target_id) = (target.tenant_id, target.id);
322        let deadline = Instant::now() + RECONCILE_WALL_CLOCK;
323        let mut report = ReconcileReport::new();
324        let outcome = self
325            .run_claimed(target, started_at, deadline, &mut report)
326            .await;
327        match &outcome {
328            Ok(()) => tracing::info!(
329                target: "axiam::scim_reconcile",
330                %tenant_id,
331                %target_id,
332                enqueued = report.enqueued,
333                enqueue_failed = report.enqueue_failed,
334                erase_retried = report.erase_retried,
335                drift_cleared = report.drift_cleared,
336                links_dropped = report.links_dropped,
337                deprovisioned = report.deprovisioned,
338                pages = report.pages,
339                listing = match &report.listing {
340                    Listing::Complete => "complete",
341                    Listing::BudgetExhausted => "budget exhausted",
342                    Listing::Failed(reason) => reason.as_str(),
343                },
344                "SCIM reconciliation finished"
345            ),
346            Err(_) => tracing::warn!(
347                target: "axiam::scim_reconcile",
348                %tenant_id,
349                %target_id,
350                "SCIM reconciliation stopped: AXIAM's datastore could not be read"
351            ),
352        }
353        outcome.map(|()| report)
354    }

356    async fn run_claimed(
357        &self,
358        target: ScimTarget,
359        started_at: DateTime<Utc>,
360        deadline: Instant,
361        report: &mut ReconcileReport,
362    ) -> Result<(), AxiamError> {
363        let mut queued = HashSet::new();
364        self.enqueue_everything(&target, &mut queued, report)
365            .await?;
366
367        let mut run = Run::new(target);
368        let mut worst = Listing::Complete;
369        let mut collections = vec![ScimResourceType::User];
370        if run.target.push_groups {
371            collections.push(ScimResourceType::Group);
372        }
373        for resource_type in collections {
374            let listing = self
375                .audit_collection(&mut run, resource_type, started_at, deadline, report)
376                .await?;
377            // A failure outranks a budget, which outranks completeness.
378            worst = match (worst, listing) {
379                (Listing::Failed(reason), _) | (_, Listing::Failed(reason)) => {
380                    Listing::Failed(reason)
381                }
382                (Listing::BudgetExhausted, _) | (_, Listing::BudgetExhausted) => {
383                    Listing::BudgetExhausted
384                }
385                _ => Listing::Complete,
386            };
387            if matches!(worst, Listing::Failed(_)) {
388                break;
389            }
390        }
391        report.listing = worst;
392        Ok(())
393    }

399    /// One reference per in-scope user, per pushed group and per linked
400    /// resource, each queued once.
401    async fn enqueue_everything(
402        &self,
403        target: &ScimTarget,
404        queued: &mut HashSet<(ScimResourceType, Uuid)>,
405        report: &mut ReconcileReport,
406    ) -> Result<(), AxiamError> {
407        let tenant_id = target.tenant_id;
408
409        // Users in scope. An erased or deleted account with a link is queued
410        // below with the links; one without a link has nothing to remove.
411        match &target.scope {
412            ScimTargetScope::AllUsers => {
413                let mut offset = 0;
414                loop {
415                    let page = self.users.list(tenant_id, axiam_page(offset)).await?;
416                    for user in &page.items {
417                        if !matches!(user.status, UserStatus::Deleted | UserStatus::Anonymized) {
418                            self.queue_once(
419                                target,
420                                ScimResourceType::User,
421                                user.id,
422                                queued,
423                                report,
424                            )
425                            .await;
426                        }
427                    }
428                    offset += page.items.len() as u64;
429                    if page.items.is_empty() || offset >= page.total {
430                        break;
431                    }
432                }
433            }
434            ScimTargetScope::Groups(listed) => {
435                for group_id in listed {
436                    let mut offset = 0;
437                    loop {
438                        let page = match self
439                            .groups
440                            .get_members(tenant_id, *group_id, axiam_page(offset))
441                            .await
442                        {
443                            Ok(page) => page,
444                            // A listed group that no longer exists has no members.
445                            Err(AxiamError::NotFound { .. }) => break,
446                            Err(error) => return Err(error),
447                        };
448                        for user in &page.items {
449                            self.queue_once(
450                                target,
451                                ScimResourceType::User,
452                                user.id,
453                                queued,
454                                report,
455                            )
456                            .await;
457                        }
458                        offset += page.items.len() as u64;
459                        if page.items.is_empty() || offset >= page.total {
460                            break;
461                        }
462                    }
463                }
464            }
465        }
466
467        // Groups the target pushes.
468        if target.push_groups {
469            match &target.scope {
470                ScimTargetScope::AllUsers => {
471                    let mut offset = 0;
472                    loop {
473                        let page = self.groups.list(tenant_id, axiam_page(offset)).await?;
474                        for group in &page.items {
475                            self.queue_once(
476                                target,
477                                ScimResourceType::Group,
478                                group.id,
479                                queued,
480                                report,
481                            )
482                            .await;
483                        }
484                        offset += page.items.len() as u64;
485                        if page.items.is_empty() || offset >= page.total {
486                            break;
487                        }
488                    }
489                }
490                ScimTargetScope::Groups(listed) => {
491                    for group_id in listed {
492                        self.queue_once(target, ScimResourceType::Group, *group_id, queued, report)
493                            .await;
494                    }
495                }
496            }
497        }
498
499        // Everything already linked, wherever it stands now: this is what
500        // removes a resource that left scope, was erased or was deleted, and
501        // what retries an erasure `DELETE` that was refused.
502        let mut offset = 0;
503        loop {
504            let page = self
505                .links
506                .list_by_target(tenant_id, target.id, None, axiam_page(offset))
507                .await?;
508            for link in &page.items {
509                if link.erase_pending {
510                    report.erase_retried += 1;
511                }
512                self.queue_once(target, link.resource_type, link.axiam_id, queued, report)
513                    .await;
514            }
515            offset += page.items.len() as u64;
516            if page.items.is_empty() || offset >= page.total {
517                break;
518            }
519        }
520        Ok(())
521    }

523    /// Queue a reference unless this run already queued it.
524    async fn queue_once(
525        &self,
526        target: &ScimTarget,
527        resource_type: ScimResourceType,
528        axiam_id: Uuid,
529        queued: &mut HashSet<(ScimResourceType, Uuid)>,
530        report: &mut ReconcileReport,
531    ) {
532        if queued.insert((resource_type, axiam_id)) {
533            self.queue_again(target, resource_type, axiam_id, report)
534                .await;
535        }
536    }

538    /// Queue a reference, whether or not this run queued it before: a repair
539    /// made after the first reference may have been delivered needs another.
540    async fn queue_again(
541        &self,
542        target: &ScimTarget,
543        resource_type: ScimResourceType,
544        axiam_id: Uuid,
545        report: &mut ReconcileReport,
546    ) {
547        let message = reference_message(target.tenant_id, target.id, resource_type, axiam_id);
548        match self.publisher.enqueue(&message).await {
549            Ok(()) => report.enqueued += 1,
550            Err(_) => report.enqueue_failed += 1,
551        }
552    }

558    /// Page one downstream collection and repair what it shows. `Err` only for
559    /// a failure of AXIAM's own datastore; whatever the downstream does is a
560    /// [`Listing`].
561    async fn audit_collection(
562        &self,
563        run: &mut Run,
564        resource_type: ScimResourceType,
565        started_at: DateTime<Utc>,
566        deadline: Instant,
567        report: &mut ReconcileReport,
568    ) -> Result<Listing, AxiamError> {
569        let target = run.target.clone();
570        let mut seen: HashSet<String> = HashSet::new();
571        let mut start_index: u64 = 1;
572        let mut pages = 0u32;
573        loop {
574            if pages >= RECONCILE_MAX_PAGES || Instant::now() >= deadline {
575                return Ok(Listing::BudgetExhausted);
576            }
577            let url = list_url(&target.base_url, resource_type, start_index).map_err(datastore)?;
578            let response = match self
579                .call(run, Method::GET, &url, Payload::scim(None), LIST_BODY_CAP)
580                .await
581            {
582                Ok(response) => response,
583                Err(exit) => return Ok(Listing::Failed(reason_of(exit))),
584            };
585            if !(200..=299).contains(&response.status) {
586                return Ok(Listing::Failed(reason_of(
587                    self.reject(run, response.status),
588                )));
589            }
590            let Ok(document) = serde_json::from_slice::<Value>(&response.body) else {
591                return Ok(Listing::Failed(
592                    "the receiver's list response is malformed".to_owned(),
593                ));
594            };
595            pages += 1;
596            report.pages += 1;
597
598            let resources = document
599                .get("Resources")
600                .and_then(Value::as_array)
601                .map_or(&[][..], Vec::as_slice);
602            for resource in resources {
603                self.audit_resource(run, resource_type, resource, &mut seen, report)
604                    .await?;
605            }
606
607            let consumed = resources.len() as u64;
608            if consumed == 0 {
609                break;
610            }
611            start_index += consumed;
612            let listed = start_index - 1;
613            match document.get("totalResults").and_then(Value::as_u64) {
614                Some(total) if listed >= total => break,
615                Some(_) => {}
616                // No total: a short page is the last.
617                None if consumed < u64::from(LIST_PAGE_SIZE) => break,
618                None => {}
619            }
620        }
621
622        // Only a listing that reached its end may conclude that a link's
623        // resource is gone: every early exit above returned.
624        self.drop_missing_links(&target, resource_type, &seen, started_at, report)
625            .await?;
626        Ok(Listing::Complete)
627    }

629    /// One downstream resource, judged.
630    async fn audit_resource(
631        &self,
632        run: &Run,
633        resource_type: ScimResourceType,
634        resource: &Value,
635        seen: &mut HashSet<String>,
636        report: &mut ReconcileReport,
637    ) -> Result<(), AxiamError> {
638        let target = &run.target;
639        let Some(downstream_id) = resource
640            .get("id")
641            .and_then(Value::as_str)
642            .filter(|id| usable_downstream_id(id))
643        else {
644            return Ok(());
645        };
646        seen.insert(downstream_id.to_owned());
647
648        let linked = self
649            .links
650            .get_by_downstream_id(target.tenant_id, target.id, resource_type, downstream_id)
651            .await?;
652        if let Some(link) = linked {
653            return self.audit_linked(target, &link, resource, report).await;
654        }
655
656        // Not ours to touch unless `externalId` is the id of an AXIAM resource
657        // of this tenant, and then only if that resource should not be there.
658        let Some(axiam_id) = resource
659            .get("externalId")
660            .and_then(Value::as_str)
661            .and_then(|id| Uuid::parse_str(id).ok())
662        else {
663            return Ok(());
664        };
665        let unwanted = match resource_type {
666            ScimResourceType::User => self.user_should_be_gone(target, axiam_id, resource).await?,
667            ScimResourceType::Group => self.group_should_be_gone(target, axiam_id).await?,
668        };
669        if !unwanted {
670            return Ok(());
671        }
672        let adopted = self
673            .links
674            .create(NewScimTargetLink {
675                tenant_id: target.tenant_id,
676                target_id: target.id,
677                resource_type,
678                axiam_id,
679                downstream_id: downstream_id.to_owned(),
680            })
681            .await;
682        match adopted {
683            Ok(_) => {
684                report.deprovisioned += 1;
685                self.queue_again(target, resource_type, axiam_id, report)
686                    .await;
687                Ok(())
688            }
689            // Linked meanwhile (another replica, a delivery): not ours to redo.
690            Err(AxiamError::AlreadyExists { .. }) => Ok(()),
691            Err(error) => Err(error),
692        }
693    }
694
695    /// A linked resource: clear its digest when the downstream differs from
696    /// what would be sent now. A resource that is to be removed is not drift:
697    /// its reference is already queued.
698    async fn audit_linked(
699        &self,
700        target: &ScimTarget,
701        link: &ScimTargetLink,
702        resource: &Value,
703        report: &mut ReconcileReport,
704    ) -> Result<(), AxiamError> {
705        let drifted = match link.resource_type {
706            ScimResourceType::User => {
707                let user = match self.users.get_by_id(target.tenant_id, link.axiam_id).await {
708                    Ok(user) => Some(user),
709                    Err(AxiamError::NotFound { .. }) => None,
710                    Err(error) => return Err(error),
711                };
712                match self.desired_user(target, user).await.map_err(datastore)? {
713                    DesiredUser::Present { user, active } => {
714                        UserRepresentation::new(&user, target, active).differs_from(resource)
715                    }
716                    DesiredUser::Remove { .. } => false,
717                }
718            }
719            ScimResourceType::Group => {
720                let group = match self.groups.get_by_id(target.tenant_id, link.axiam_id).await {
721                    Ok(group) if group_in_scope(target, group.id) => group,
722                    Ok(_) | Err(AxiamError::NotFound { .. }) => return Ok(()),
723                    Err(error) => return Err(error),
724                };
725                match self.linked_member_ids(target, &group).await {
726                    Ok(members) => GroupRepresentation::new(&group, members).differs_from(resource),
727                    // Too large to push: nothing to compare.
728                    Err(Exit::Dead(_)) => false,
729                    Err(exit) => return Err(datastore(exit)),
730                }
731            }
732        };
733        if !drifted {
734            return Ok(());
735        }
736        self.links
737            .set_digest(
738                link.tenant_id,
739                link.target_id,
740                link.resource_type,
741                link.axiam_id,
742                None,
743            )
744            .await?;
745        report.drift_cleared += 1;
746        // The reference queued in step 1 may already have been delivered, with
747        // the digest still set: queue another, behind the clearing.
748        self.queue_again(target, link.resource_type, link.axiam_id, report)
749            .await;
750        Ok(())
751    }
752
753    /// Whether the downstream account named by `axiam_id` is one AXIAM would
754    /// not provision now: the user exists **in this tenant** and is erased,
755    /// deleted, out of scope or not active, and — for a deactivation policy —
756    /// the downstream still has it active. A user of another tenant, and an id
757    /// that is no user at all, are not found here and are left alone.
758    async fn user_should_be_gone(
759        &self,
760        target: &ScimTarget,
761        axiam_id: Uuid,
762        resource: &Value,
763    ) -> Result<bool, AxiamError> {
764        let user = match self.users.get_by_id(target.tenant_id, axiam_id).await {
765            Ok(user) => user,
766            Err(AxiamError::NotFound { .. }) => return Ok(false),
767            Err(error) => return Err(error),
768        };
769        Ok(
770            match self
771                .desired_user(target, Some(user))
772                .await
773                .map_err(datastore)?
774            {
775                DesiredUser::Remove { .. } => true,
776                DesiredUser::Present { active: false, .. } => resource
777                    .get("active")
778                    .and_then(Value::as_bool)
779                    .unwrap_or(true),
780                DesiredUser::Present { active: true, .. } => false,
781            },
782        )
783    }
784
785    /// Whether the downstream group named by `axiam_id` is a group of this
786    /// tenant that the target no longer pushes.
787    async fn group_should_be_gone(
788        &self,
789        target: &ScimTarget,
790        axiam_id: Uuid,
791    ) -> Result<bool, AxiamError> {
792        match self.groups.get_by_id(target.tenant_id, axiam_id).await {
793            Ok(group) => Ok(!group_in_scope(target, group.id)),
794            Err(AxiamError::NotFound { .. }) => Ok(false),
795            Err(error) => Err(error),
796        }
797    }
798
799    /// After a collection was listed to its end: forget the links whose
800    /// downstream resource was not in it, and queue those resources again so
801    /// that they are re-created. A link made after the run began is not judged
802    /// by a listing that may predate it.
803    async fn drop_missing_links(
804        &self,
805        target: &ScimTarget,
806        resource_type: ScimResourceType,
807        seen: &HashSet<String>,
808        started_at: DateTime<Utc>,
809        report: &mut ReconcileReport,
810    ) -> Result<(), AxiamError> {
811        let mut missing: Vec<ScimTargetLink> = Vec::new();
812        let mut offset = 0;
813        loop {
814            let page = self
815                .links
816                .list_by_target(
817                    target.tenant_id,
818                    target.id,
819                    Some(resource_type),
820                    axiam_page(offset),
821                )
822                .await?;
823            missing.extend(
824                page.items
825                    .iter()
826                    .filter(|link| {
827                        link.created_at < started_at && !seen.contains(&link.downstream_id)
828                    })
829                    .cloned(),
830            );
831            offset += page.items.len() as u64;
832            if page.items.is_empty() || offset >= page.total {
833                break;
834            }
835        }
836        for link in missing {
837            let removed = self
838                .links
839                .delete(
840                    link.tenant_id,
841                    link.target_id,
842                    link.resource_type,
843                    link.axiam_id,
844                )
845                .await?;
846            if removed {
847                report.links_dropped += 1;
848                self.queue_again(target, link.resource_type, link.axiam_id, report)
849                    .await;
850            }
851        }
852        Ok(())
853    }
854}

856impl<T, L, S, U, G> ScimReconciliation for ScimPushDeliverer<T, L, S, U, G>
857where
858    T: ScimTargetRepository + 'static,
859    L: ScimTargetLinkRepository + 'static,
860    S: ScimTargetStateRepository + 'static,
861    U: UserRepository + 'static,
862    G: GroupRepository + 'static,
863{
864    fn run_due<'a>(
865        &'a self,
866        should_stop: &'a (dyn Fn() -> bool + Send + Sync),
867    ) -> Pin<Box<dyn Future<Output = Result<ReconcileSweep, AxiamError>> + Send + 'a>> {
868        Box::pin(async move {
869            let targets = self.targets.list_all_enabled().await?;
870            let mut sweep = ReconcileSweep::default();
871            for target in targets {
872                if should_stop() {
873                    break;
874                }
875                match self
876                    .reconcile(target.tenant_id, target.id, RECONCILE_INTERVAL)
877                    .await
878                {
879                    Ok(ReconcileOutcome::Ran(report)) => {
880                        sweep.reconciled += 1;
881                        if report.is_failure() {
882                            sweep.failed += 1;
883                        }
884                    }
885                    // Not due, claimed by another replica, or switched off
886                    // since the listing: the common case at every tick.
887                    Ok(ReconcileOutcome::AlreadyClaimed | ReconcileOutcome::TargetDisabled)
888                    | Err(AxiamError::NotFound { .. }) => {}
889                    Err(_) => {
890                        // `reconcile` logged the one line for this target.
891                        sweep.failed += 1;
892                    }
893                }
894            }
895            Ok(sweep)
896        })
897    }
898}

908/// `<base>/<collection>?startIndex=<n>&count=100`.
909fn list_url(
910    base_url: &str,
911    resource_type: ScimResourceType,
912    start_index: u64,
913) -> Result<String, Exit> {
914    let plain = resource_url(base_url, collection_of(resource_type), None, None)?;
915    let mut url =
916        Url::parse(&plain).map_err(|_| Exit::dead("the target's base URL is not valid"))?;
917    url.query_pairs_mut()
918        .append_pair("startIndex", &start_index.to_string())
919        .append_pair("count", &LIST_PAGE_SIZE.to_string());
920    Ok(url.to_string())
921}

923/// [`axiam_api_rest::state::bundles::ScimReconcileTrigger`] over the deliverer:
924/// what the management API's `POST /api/v1/scim-targets/{id}/reconcile` calls.
925pub struct ReconcileLauncher<T, L, S, U, G> {
926    deliverer: Arc<ScimPushDeliverer<T, L, S, U, G>>,
927}

929impl<T, L, S, U, G> ReconcileLauncher<T, L, S, U, G> {
930    /// Launch runs on `deliverer`, the instance the scheduled job and the
931    /// consumer use, so that all of them share its credential path and its
932    /// access-token cache.
933    #[must_use]
934    pub fn new(deliverer: Arc<ScimPushDeliverer<T, L, S, U, G>>) -> Self {
935        Self { deliverer }
936    }
937}

939impl<T, L, S, U, G> axiam_api_rest::state::bundles::ScimReconcileTrigger
940    for ReconcileLauncher<T, L, S, U, G>
941where
942    T: ScimTargetRepository + 'static,
943    L: ScimTargetLinkRepository + 'static,
944    S: ScimTargetStateRepository + 'static,
945    U: UserRepository + 'static,
946    G: GroupRepository + 'static,
947{
948    fn start<'a>(
949        &'a self,
950        tenant_id: Uuid,
951        target_id: Uuid,
952    ) -> Pin<
953        Box<
954            dyn Future<
955                    Output = Result<axiam_api_rest::state::bundles::ScimReconcileStart, AxiamError>,
956                > + Send
957                + 'a,
958        >,
959    > {
960        use axiam_api_rest::state::bundles::ScimReconcileStart as Api;
961        Box::pin(async move {
962            Ok(
963                match self
964                    .deliverer
965                    .start_reconcile_now(tenant_id, target_id)
966                    .await?
967                {
968                    ReconcileStart::Started => Api::Started,
969                    ReconcileStart::AlreadyClaimed => Api::AlreadyClaimed,
970                    ReconcileStart::TargetDisabled => Api::TargetDisabled,
971                },
972            )
973        })
974    }
975}
