1//! The documents the outbound client sends (RFC 7643 §4 resources, RFC 7644
2//! §3.5.2 `PatchOp`) and the digest of a representation.
3//!
4//! These are `Serialize`-only types of their own: the inbound wire types of
5//! this crate are `Deserialize`-only and private to their handlers, and bending
6//! them to serve both directions would couple the two.
7//!
8//! A representation is built from the AXIAM [`User`] or [`Group`] and the
9//! target's mapping — a **fixed attribute set**, not a mapping language:
10//!
11//! | Downstream | From |
12//! |---|---|
13//! | `userName` | the username, or the email when the target says `user_name_from = email` |
14//! | `name.givenName`, `name.familyName`, `displayName` | `metadata.scim.givenName`, `…familyName`, `…formatted` — where the inbound SCIM server stores them ([`crate::scim_metadata`]) |
15//! | `emails` | the primary email |
16//! | `active` | the desired state computed by the deliverer |
17//! | `externalId` | the AXIAM id |
18//!
19//! An attribute AXIAM does not hold is **omitted**, never sent empty and never
20//! removed: the downstream may own it.
21

75    /// Whether the account is enabled downstream.
76    pub active: bool,

79impl UserRepresentation {
80    /// The representation of `user` under `target`'s mapping, with `active` as
81    /// the deliverer decided it.
82    #[must_use]
83    pub fn new(user: &User, target: &ScimTarget, active: bool) -> Self {
84        let user_name = match target.user_name_from {
85            UserNameSource::Email if !user.email.is_empty() => user.email.clone(),
86            _ => user.username.clone(),
87        };
88        let given_name = scim_metadata::get_str(&user.metadata, "givenName");
89        let family_name = scim_metadata::get_str(&user.metadata, "familyName");
90        let name = (given_name.is_some() || family_name.is_some()).then_some(NameRepresentation {
91            given_name,
92            family_name,
93        });
94        Self {
95            schemas: [USER_SCHEMA],
96            external_id: user.id.to_string(),
97            user_name,
98            name,
99            display_name: scim_metadata::get_str(&user.metadata, "formatted"),
100            emails: if user.email.is_empty() {
101                Vec::new()
102            } else {
103                vec![EmailRepresentation {
104                    value: user.email.clone(),
105                    primary: true,
106                }]
107            },
108            active,
109        }
110    }
111
112    /// The `replace` operations of a `PATCH` for this representation: the
113    /// mapped attributes only (`userName`, `name.givenName`, `name.familyName`,
114    /// `displayName`, the primary `emails` value, `active`, `externalId`).
115    #[must_use]
116    pub fn patch(&self) -> PatchRequest {
117        let mut operations = vec![PatchOperation::replace("userName", json!(self.user_name))];
118        if let Some(name) = &self.name {
119            if let Some(given) = &name.given_name {
120                operations.push(PatchOperation::replace("name.givenName", json!(given)));
121            }
122            if let Some(family) = &name.family_name {
123                operations.push(PatchOperation::replace("name.familyName", json!(family)));
124            }
125        }
126        if let Some(display) = &self.display_name {
127            operations.push(PatchOperation::replace("displayName", json!(display)));
128        }
129        if !self.emails.is_empty() {
130            operations.push(PatchOperation::replace("emails", json!(self.emails)));
131        }
132        operations.push(PatchOperation::replace("active", json!(self.active)));
133        operations.push(PatchOperation::replace(
134            "externalId",
135            json!(self.external_id),
136        ));
137        PatchRequest::new(operations)
138    }
139}

141/// A string member of a downstream resource.
142fn text_at<'a>(resource: &'a Value, path: &[&str]) -> Option<&'a str> {
143    let mut cursor = resource;
144    for part in path {
145        cursor = cursor.get(part)?;
146    }
147    cursor.as_str()
148}

150impl UserRepresentation {
151    /// Whether the downstream's `resource` differs from this representation in
152    /// any attribute the mapping sets (the fixed set of the module
153    /// documentation): reconciliation clears the link's digest when it does, so
154    /// that the next sync sends the representation again.
155    ///
156    /// An attribute AXIAM does not hold is not compared (the downstream may own
157    /// it). `userName` and the email are compared without regard to ASCII case
158    /// (RFC 7643 §4.1.1: `userName` is case-insensitive, and receivers
159    /// commonly fold both), `active` defaults to `true` when the downstream
160    /// states nothing (§4.1.1), and the email is found among the downstream's
161    /// `emails` whichever entry carries it.
162    #[must_use]
163    pub fn differs_from(&self, resource: &Value) -> bool {
164        let folded = |a: Option<&str>, b: &str| !a.is_some_and(|a| a.eq_ignore_ascii_case(b));
165        if folded(text_at(resource, &["userName"]), &self.user_name) {
166            return true;
167        }
168        if text_at(resource, &["externalId"]) != Some(self.external_id.as_str()) {
169            return true;
170        }
171        if resource
172            .get("active")
173            .and_then(Value::as_bool)
174            .unwrap_or(true)
175            != self.active
176        {
177            return true;
178        }
179        if let Some(name) = &self.name {
180            if let Some(given) = &name.given_name
181                && text_at(resource, &["name", "givenName"]) != Some(given.as_str())
182            {
183                return true;
184            }
185            if let Some(family) = &name.family_name
186                && text_at(resource, &["name", "familyName"]) != Some(family.as_str())
187            {
188                return true;
189            }
190        }
191        if let Some(display) = &self.display_name
192            && text_at(resource, &["displayName"]) != Some(display.as_str())
193        {
194            return true;
195        }
196        if let Some(email) = self.emails.first() {
197            let present = resource
198                .get("emails")
199                .and_then(Value::as_array)
200                .into_iter()
201                .flatten()
202                .any(|entry| {
203                    entry
204                        .get("value")
205                        .and_then(Value::as_str)
206                        .is_some_and(|v| v.eq_ignore_ascii_case(&email.value))
207                });
208            if !present {
209                return true;
210            }
211        }
212        false
213    }
214}

216impl GroupRepresentation {
217    /// Whether the downstream's `resource` differs from this representation:
218    /// `displayName`, `externalId`, or the set of `members` (the whole list is
219    /// what a `PATCH` sends, so the whole list is what is compared).
220    #[must_use]
221    pub fn differs_from(&self, resource: &Value) -> bool {
222        if text_at(resource, &["displayName"]) != Some(self.display_name.as_str())
223            || text_at(resource, &["externalId"]) != Some(self.external_id.as_str())
224        {
225            return true;
226        }
227        let mut downstream: Vec<&str> = resource
228            .get("members")
229            .and_then(Value::as_array)
230            .into_iter()
231            .flatten()
232            .filter_map(|member| member.get("value").and_then(Value::as_str))
233            .collect();
234        downstream.sort_unstable();
235        downstream.dedup();
236        let ours: Vec<&str> = self.members.iter().map(|m| m.value.as_str()).collect();
237        downstream != ours
238    }
239}

248/// What `POST /Groups` carries and what a group `PATCH` is derived from.
249#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
250pub struct GroupRepresentation {
251    schemas: [&'static str; 1],
252    /// The AXIAM group id.
253    #[serde(rename = "externalId")]
254    pub external_id: String,
255    /// The group's name.
256    #[serde(rename = "displayName")]
257    pub display_name: String,
258    /// The downstream ids of the group's **linked** members, sorted (so the
259    /// digest does not depend on the order the datastore returned them in).
260    pub members: Vec<MemberRepresentation>,
261}
262
263impl GroupRepresentation {
264    /// The representation of `group` with `member_ids` (downstream user ids).
265    #[must_use]
266    pub fn new(group: &Group, mut member_ids: Vec<String>) -> Self {
267        member_ids.sort_unstable();
268        member_ids.dedup();
269        Self {
270            schemas: [GROUP_SCHEMA],
271            external_id: group.id.to_string(),
272            display_name: group.name.clone(),
273            members: member_ids
274                .into_iter()
275                .map(|value| MemberRepresentation { value })
276                .collect(),
277        }
278    }
279
280    /// The `replace` operations of a `PATCH`: `displayName`, `externalId` and
281    /// `members` (the whole list, so a removed member leaves).
282    #[must_use]
283    pub fn patch(&self) -> PatchRequest {
284        PatchRequest::new(vec![
285            PatchOperation::replace("displayName", json!(self.display_name)),
286            PatchOperation::replace("externalId", json!(self.external_id)),
287            PatchOperation::replace("members", json!(self.members)),
288        ])
289    }
290}

292/// One RFC 7644 §3.5.2 operation. Only `replace` is ever sent: a `PUT` would
293/// clobber attributes the downstream owns, and an `add` of a single-valued
294/// attribute is a `replace` by another name.
295#[derive(Debug, Clone, PartialEq, Serialize)]
296pub struct PatchOperation {
297    op: &'static str,
298    path: &'static str,
299    value: Value,
300}
301
302impl PatchOperation {
303    fn replace(path: &'static str, value: Value) -> Self {
304        Self {
305            op: "replace",
306            path,
307            value,
308        }
309    }
310
311    /// The attribute path the operation names.
312    #[must_use]
313    pub fn path(&self) -> &'static str {
314        self.path
315    }
316}

318/// An RFC 7644 §3.5.2 `PatchOp` message.
319#[derive(Debug, Clone, PartialEq, Serialize)]
320pub struct PatchRequest {
321    schemas: [&'static str; 1],
322    #[serde(rename = "Operations")]
323    operations: Vec<PatchOperation>,
324}
325
326impl PatchRequest {
327    fn new(operations: Vec<PatchOperation>) -> Self {
328        Self {
329            schemas: [PATCH_OP_SCHEMA],
330            operations,
331        }
332    }
333
334    /// The operations, in the order they are sent.
335    #[must_use]
336    pub fn operations(&self) -> &[PatchOperation] {
337        &self.operations
338    }
339}

341/// SHA-256 (lower-case hex) over the canonical serialisation of a
342/// representation: serde writes a struct's members in declaration order, so the
343/// bytes depend on the content and on nothing else.
344#[must_use]
345pub fn digest_of<T: Serialize>(representation: &T) -> String {
346    // A struct of strings, booleans and vectors of them cannot fail to
347    // serialise; the empty fallback would only ever change a digest, which
348    // re-sends, never loses an update.
349    let bytes = serde_json::to_vec(representation).unwrap_or_default();
350    hex::encode(Sha256::digest(&bytes))
351}

515    #[test]
516    fn drift_is_a_difference_in_a_mapped_attribute_the_mapping_holds() {
517        let u = user();
518        let t = target(UserNameSource::Username);
519        let repr = UserRepresentation::new(&u, &t, true);
520        // What a receiver that stored exactly what was sent answers with.
521        let mut downstream = serde_json::to_value(&repr).unwrap();
522        downstream["id"] = json!("d-1");
523        assert!(!repr.differs_from(&downstream));
524
525        // Attributes the downstream owns, and case folding, are not drift.
526        let mut owned = downstream.clone();
527        owned["title"] = json!("Engineer");
528        owned["userName"] = json!("ALICE");
529        owned["emails"] = json!([{"value": "other@example.com"}, {"value": "Alice@Example.com"}]);
530        assert!(!repr.differs_from(&owned));
531        let mut implicit = downstream.clone();
532        implicit.as_object_mut().unwrap().remove("active");
533        assert!(!repr.differs_from(&implicit), "active defaults to true");
534
535        for (path, value) in [
536            ("userName", json!("mallory")),
537            ("externalId", json!("someone-else")),
538            ("active", json!(false)),
539            ("displayName", json!("Renamed")),
540            (
541                "name",
542                json!({"givenName": "Alice", "familyName": "Changed"}),
543            ),
544            ("emails", json!([{"value": "changed@example.com"}])),
545        ] {
546            let mut edited = downstream.clone();
547            edited[path] = value;
548            assert!(repr.differs_from(&edited), "{path}");
549        }
550        let mut gone = downstream.clone();
551        gone.as_object_mut().unwrap().remove("userName");
552        assert!(repr.differs_from(&gone));
553    }
