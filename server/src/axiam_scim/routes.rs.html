14/// Bucket name for the shared (cross-replica) rate-limit counter. Must be
15/// unique across every rate-limited resource in the deployment — see
16/// [`RateLimitShared`]'s docs.
17const SCIM_BUCKET: &str = "scim";
18
19/// Register every `/scim/v2/*` route with the shipped rate-limit posture.
20///
21/// Equivalent to [`scim_routes_with_rate_limits`] with
22/// [`RateLimitConfig::default`], mirroring `axiam_api_rest::api_v1_routes`
23/// versus `register_api_v1_routes`. The composition root
24/// (`axiam-server::main`) uses the configured form; tests that do not care
25/// about the limiter use this one.
26pub fn scim_routes<C: Connection + Clone>(cfg: &mut web::ServiceConfig) {
27    scim_routes_with_rate_limits::<C>(cfg, &RateLimitConfig::default());
28}
29
30/// Register every `/scim/v2/*` route on the given `ServiceConfig`.
31///
32/// Wrapped in the same [`AuthzMiddleware`] every other authenticated AXIAM
33/// REST scope uses (`server.rs`'s `/api/v1/*` scopes) — SCIM clients always
34/// carry `Authorization: Bearer <token>` (never the browser session cookie),
35/// so it behaves here exactly as it does on any bearer-only route: 401 on a
36/// wholly missing credential, then defer the actual `scim:provision` check
37/// to [`crate::auth::require_scim_provision`] inside each handler.
38///
39/// # Rate limiting (R5.2)
40///
41/// Wrapped in the SAME two layers every rate-limited family in `server.rs`
42/// uses, in the same order — [`RateLimitShared`] (cross-replica, fail-open)
43/// as the OUTER wrap, [`build_governor`] (per-replica, in-memory GCRA) just
44/// inside it. Because `.wrap()` applies last-registered-outermost, the calls
45/// below read bottom-up: `RateLimitShared` first, then the governor, then
46/// `AuthzMiddleware`.
47///
48/// The limiters sit OUTSIDE `AuthzMiddleware` deliberately: an unauthenticated
49/// `/scim/v2` flood should be shed before it reaches the credential check,
50/// which is also why the discovery endpoints are covered by the same bucket
51/// rather than left unmetered.
52///
53/// One bucket spans the whole scope — see
54/// [`RateLimitConfig::scim_per_min`] for why (and for where the 600/min
55/// default comes from).
56pub fn scim_routes_with_rate_limits<C: Connection + Clone>(
57    cfg: &mut web::ServiceConfig,
58    rate_limit_cfg: &RateLimitConfig,
59) {
60    let scim_per_min = rate_limit_cfg.scim_per_min;
61    cfg.service(
62        web::scope("/scim/v2")
63            .wrap(AuthzMiddleware)
64            .wrap(build_governor(scim_per_min))
65            .wrap(RateLimitShared::<C>::new(SCIM_BUCKET, scim_per_min))
66            // Service-discovery endpoints (RFC 7643 §5-7). Same auth
67            // requirement as every other SCIM route — nothing here is
68            // registered as an `AuthzMiddleware` public path, since even
69            // these mostly-static bodies still describe an AXIAM-tenant-
70            // specific deployment surface.
71            .route(
72                "/ServiceProviderConfig",
73                web::get().to(schema::service_provider_config),
74            )
75            .route("/ResourceTypes", web::get().to(schema::resource_types))
76            .route("/ResourceTypes/{name}", web::get().to(schema::resource_type))
77            .route("/Schemas", web::get().to(schema::schemas))
78            .route("/Schemas/{id}", web::get().to(schema::schema_by_id))
79            // B4: bulk operations are explicitly out of scope — return the
80            // correct SCIM error (501) rather than a generic 404.
81            .route("/Bulk", web::post().to(bulk_not_supported))
82            .service(
83                web::resource("/Users")
84                    .route(web::get().to(users::list::<C>))
85                    .route(web::post().to(users::create::<C>)),
86            )
87            .service(
88                web::resource("/Users/{id}")
89                    .route(web::get().to(users::get::<C>))
90                    .route(web::put().to(users::replace::<C>))
91                    .route(web::patch().to(users::patch::<C>))
92                    .route(web::delete().to(users::delete::<C>)),
93            )
94            .service(
95                web::resource("/Groups")
96                    .route(web::get().to(groups::list::<C>))
97                    .route(web::post().to(groups::create::<C>)),
98            )
99            .service(
100                web::resource("/Groups/{id}")
101                    .route(web::get().to(groups::get::<C>))
102                    .route(web::put().to(groups::replace::<C>))
103                    .route(web::patch().to(groups::patch::<C>))
104                    .route(web::delete().to(groups::delete::<C>)),
105            ),
106    );
107}
108
109async fn bulk_not_supported() -> HttpResponse {
110    ScimError::not_implemented("Bulk operations are not supported (B4: explicitly out of scope)")
111        .error_response()
