1//! Storage for SCIM-only fields (`externalId`, `name.*`) inside the existing
2//! generic `metadata: serde_json::Value` column AXIAM's `User`/`Group`
3//! models already carry, instead of adding new schema (B4: "map onto the
4//! EXISTING user/group repositories — do not create parallel storage").
5//!
6//! Everything SCIM writes lives under a single top-level `"scim"` key so it
7//! can never collide with metadata some other feature (or the admin UI)
8//! writes into the same column, and a PUT/PATCH that touches only SCIM
9//! fields never disturbs those other keys.
10
11use serde_json::{Map, Value};
12
13const SCIM_KEY: &str = "scim";
14
15/// Read `metadata.scim.<field>` as a string, or `None` if absent/not a string.
16pub fn get_str(metadata: &Value, field: &str) -> Option<String> {
17    metadata
18        .get(SCIM_KEY)?
19        .get(field)?
20        .as_str()
21        .map(str::to_owned)
22}
23
24/// Set (`Some`) or remove (`None`) `metadata.scim.<field>`, leaving every
25/// other key — under `"scim"` or at the top level — untouched. Creates the
26/// `"scim"` sub-object (and, if `metadata` wasn't already a JSON object, the
27/// object itself) on demand.
28pub fn set_str(metadata: &mut Value, field: &str, value: Option<String>) {
29    if !metadata.is_object() {
30        *metadata = Value::Object(Map::new());
31    }
32    let obj = metadata.as_object_mut().expect("just ensured object above");
33    let scim = obj
34        .entry(SCIM_KEY)
35        .or_insert_with(|| Value::Object(Map::new()));
36    if !scim.is_object() {
37        *scim = Value::Object(Map::new());
38    }
39    let scim_obj = scim.as_object_mut().expect("just ensured object above");
40    match value {
41        Some(v) => {
42            scim_obj.insert(field.to_string(), Value::String(v));
43        }
44        None => {
45            scim_obj.remove(field);
46        }
47    }
48}

50#[cfg(test)]
51mod tests {
52    use super::*;
53    use serde_json::json;
54
55    #[test]
56    fn set_then_get_round_trips() {
57        let mut meta = json!({});
58        set_str(&mut meta, "externalId", Some("ext-1".into()));
59        assert_eq!(get_str(&meta, "externalId"), Some("ext-1".to_string()));
60    }
61
62    #[test]
63    fn set_none_removes_field() {
64        let mut meta = json!({"scim": {"externalId": "ext-1"}});
65        set_str(&mut meta, "externalId", None);
66        assert_eq!(get_str(&meta, "externalId"), None);
67        // The (now-empty) "scim" object stays — harmless, and simpler than
68        // pruning it back out.
69        assert!(meta.get("scim").is_some());
70    }
71
72    #[test]
73    fn preserves_unrelated_top_level_and_scim_keys() {
74        let mut meta = json!({"app_note": "keep me", "scim": {"givenName": "Alice"}});
75        set_str(&mut meta, "externalId", Some("ext-1".into()));
76        assert_eq!(meta["app_note"], "keep me");
77        assert_eq!(get_str(&meta, "givenName"), Some("Alice".to_string()));
78        assert_eq!(get_str(&meta, "externalId"), Some("ext-1".to_string()));
79    }
80
81    #[test]
82    fn non_object_metadata_is_replaced_not_panicked_on() {
83        let mut meta = Value::Null;
84        set_str(&mut meta, "externalId", Some("ext-1".into()));
85        assert_eq!(get_str(&meta, "externalId"), Some("ext-1".to_string()));
86    }
87}
