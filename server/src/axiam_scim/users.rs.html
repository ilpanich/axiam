1//! SCIM `/Users` resource (RFC 7643 §4.1, RFC 7644 §3.2-3.6).
2//!
3//! Maps onto the EXISTING `UserRepository` (B4) — no parallel storage.
4//! AXIAM's `User` model has no `externalId`/`name.*` columns, so those live
5//! inside the existing generic `metadata` JSON column under a `"scim"`
6//! sub-key (see [`crate::scim_metadata`]); everything else (`userName` <->
7//! `username`, `active` <-> `status == Active`, `emails[primary]` <->
8//! `email`) is a direct field mapping.
9//!
10//! Every handler here mirrors the native `POST/GET/PUT/DELETE
11//! /api/v1/users` handlers (`axiam_api_rest::handlers::users`) for the parts
12//! B4 requires parity on: same `create_with_consent` GDPR path, the same
13//! `emit_webhook` event names (`user.created`/`user.updated`/
14//! `user.deleted`), and the same D7 `invalidate_subject` cache flush on a
15//! status-narrowing update.
16//!
17//! # ETag (optional-v2)
18//!
19//! Not implemented. `ServiceProviderConfig` advertises `etag.supported:
20//! false`. Emitting a weak ETag (`W/"<updated_at millis>"`) on every
21//! response, and honoring `If-Match` on PUT/PATCH/DELETE, is cheap enough to
22//! be a reasonable follow-on — it just wasn't judged in-scope for B4's
23//! "implement if cheap, otherwise leave a documented TODO."
24

169impl ScimUser {
170    fn from_user(user: &User, req: &HttpRequest) -> Self {
171        let id = user.id.to_string();
172        Self {
173            schemas: [USER_SCHEMA],
174            external_id: scim_metadata::get_str(&user.metadata, "externalId"),
175            user_name: user.username.clone(),
176            name: ScimName::from_metadata(&user.metadata),
177            emails: if user.email.is_empty() {
178                vec![]
179            } else {
180                vec![ScimEmail {
181                    value: user.email.clone(),
182                    primary: true,
183                }]
184            },
185            active: user.status == UserStatus::Active,
186            meta: ScimMeta {
187                resource_type: "User",
188                created: user.created_at,
189                last_modified: user.updated_at,
190                location: location_url(req, "Users", &id),
191            },
192            id,
193        }
194    }
195}

232/// Full-tenant scan cap for the `externalId eq` filter (B4's filtering
233/// subset has no indexed lookup for it — only `userName` has a dedicated
234/// repository method). Documented limitation, not a silent truncation: a
235/// tenant with more users than this returns a filtered result computed over
236/// only the first [`EXTERNAL_ID_SCAN_CAP`] (by `created_at`), which is the
237/// same trade-off `axiam-api-rest`'s own admin-UI search boxes make today.
238const EXTERNAL_ID_SCAN_CAP: u64 = 1000;
239
240pub(crate) fn start_index_to_offset(start_index: Option<u64>) -> u64 {
241    start_index.unwrap_or(1).max(1) - 1
242}
243
244pub(crate) fn clamp_count(count: Option<u64>) -> u64 {
245    count.unwrap_or(50).min(200)
246}

252/// `GET /scim/v2/Users`
253pub async fn list<C: Connection + Clone>(
254    http_req: HttpRequest,
255    user: ScimPrincipal,
256    authz: AuthzData,
257    state: web::Data<AppState<C>>,
258    query: web::Query<ScimListQuery>,
259) -> Result<HttpResponse, ScimError> {
260    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
261    let query = query.into_inner();
262    let start_index = query.start_index.unwrap_or(1).max(1);
263    let offset = start_index_to_offset(query.start_index);
264    let limit = clamp_count(query.count);
265
266    let Some(raw_filter) = query.filter else {
267        return list_unfiltered(&http_req, &user, &state, offset, limit, start_index).await;
268    };
269
270    let filter = crate::filter::parse_eq_filter(&raw_filter, &["username", "externalid"])?;
271    match filter.attr.as_str() {
272        "username" => {
273            let (items, total) = match state
274                .user_repo
275                .get_by_username(user.tenant_id(), &filter.value)
276                .await
277            {
278                Ok(u) if offset == 0 && limit > 0 => (vec![u], 1),
279                Ok(_) => (vec![], 1),
280                Err(AxiamError::NotFound { .. }) => (vec![], 0),
281                Err(e) => return Err(e.into()),
282            };
283            let resources: Vec<ScimUser> = items
284                .iter()
285                .map(|u| ScimUser::from_user(u, &http_req))
286                .collect();
287            Ok(HttpResponse::Ok().json(ScimListResponse::new(resources, total, start_index)))
288        }
289        "externalid" => {
290            let all = state
291                .user_repo
292                .list(
293                    user.tenant_id(),
294                    Pagination {
295                        offset: 0,
296                        limit: EXTERNAL_ID_SCAN_CAP,
297                    },
298                )
299                .await?;
300            let matched: Vec<&User> = all
301                .items
302                .iter()
303                .filter(|u| {
304                    scim_metadata::get_str(&u.metadata, "externalId").as_deref()
305                        == Some(filter.value.as_str())
306                })
307                .collect();
308            let total = matched.len() as u64;
309            let page: Vec<ScimUser> = matched
310                .into_iter()
311                .skip(offset as usize)
312                .take(limit as usize)
313                .map(|u| ScimUser::from_user(u, &http_req))
314                .collect();
315            Ok(HttpResponse::Ok().json(ScimListResponse::new(page, total, start_index)))
316        }
317        // parse_eq_filter already rejects anything not in the allow-list.
318        _ => unreachable!("parse_eq_filter enforces the allowed-attribute list"),
319    }
320}

322async fn list_unfiltered<C: Connection + Clone>(
323    http_req: &HttpRequest,
324    user: &ScimPrincipal,
325    state: &web::Data<AppState<C>>,
326    offset: u64,
327    limit: u64,
328    start_index: u64,
329) -> Result<HttpResponse, ScimError> {
330    if limit == 0 {
331        // Still need an accurate totalResults; LIMIT 1 gets it cheaply.
332        let probe = state
333            .user_repo
334            .list(
335                user.tenant_id(),
336                Pagination {
337                    offset: 0,
338                    limit: 1,
339                },
340            )
341            .await?;
342        return Ok(HttpResponse::Ok().json(ScimListResponse::<ScimUser>::new(
343            vec![],
344            probe.total,
345            start_index,
346        )));
347    }
348    let result = state
349        .user_repo
350        .list(user.tenant_id(), Pagination { offset, limit })
351        .await?;
352    let resources: Vec<ScimUser> = result
353        .items
354        .iter()
355        .map(|u| ScimUser::from_user(u, http_req))
356        .collect();
357    Ok(HttpResponse::Ok().json(ScimListResponse::new(resources, result.total, start_index)))
358}

376fn build_scim_metadata(external_id: &Option<String>, name: &Option<ScimNameInput>) -> Value {
377    let mut metadata = json!({});
378    if let Some(ext) = external_id {
379        scim_metadata::set_str(&mut metadata, "externalId", Some(ext.clone()));
380    }
381    if let Some(name) = name {
382        if let Some(g) = &name.given_name {
383            scim_metadata::set_str(&mut metadata, "givenName", Some(g.clone()));
384        }
385        if let Some(f) = &name.family_name {
386            scim_metadata::set_str(&mut metadata, "familyName", Some(f.clone()));
387        }
388        if let Some(fmt) = &name.formatted {
389            scim_metadata::set_str(&mut metadata, "formatted", Some(fmt.clone()));
390        }
391    }
392    metadata
393}

402/// `POST /scim/v2/Users`
403pub async fn create<C: Connection + Clone>(
404    http_req: HttpRequest,
405    user: ScimPrincipal,
406    authz: AuthzData,
407    state: web::Data<AppState<C>>,
408    body: web::Json<ScimUserWrite>,
409) -> Result<HttpResponse, ScimError> {
410    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
411    let req = body.into_inner();
412
413    let email = primary_email(&req.emails).ok_or_else(|| {
414        ScimError::invalid_value(
415            "\"emails\" with at least one entry is required to provision an AXIAM user",
416        )
417    })?;
418
419    let metadata = build_scim_metadata(&req.external_id, &req.name);
420    let password = req.password.clone().unwrap_or_else(random_password);
421
422    let input = CreateUser {
423        tenant_id: user.tenant_id(),
424        username: req.user_name.clone(),
425        email,
426        password,
427        metadata: Some(metadata),
428    };
429
430    // Mirrors handlers::users::create exactly (axiam-api-rest): the
431    // Art. 7 proof-of-consent row is written in the SAME transaction as the
432    // user, so a SCIM-provisioned user can never exist without it either.
433    let ip_address = client_ip(&http_req);
434    let ua = user_agent(&http_req);
435    let created = state
436        .user_repo
437        .create_with_consent(input, "terms_of_service", "current", ip_address, ua)
438        .await?;
439
440    // CreateUser has no `status` field — every new user starts
441    // PendingVerification regardless of what was requested, which SCIM's
442    // `active` (== status == Active) would read back as `false` even for a
443    // fixture that explicitly asked for `active: true`. Set the status
444    // explicitly with a follow-up update so `active` in the create response
445    // matches what was requested, in both directions.
446    let final_status = if req.active {
447        UserStatus::Active
448    } else {
449        UserStatus::Inactive
450    };
451    let final_user = state
452        .user_repo
453        .update(
454            user.tenant_id(),
455            created.id,
456            UpdateUser {
457                status: Some(final_status),
458                ..Default::default()
459            },
460        )
461        .await?;
462
463    // Same webhook event/payload shape as native user creation.
464    state
465        .emit_webhook(
466            final_user.tenant_id,
467            "user.created",
468            json!({ "id": final_user.id, "username": final_user.username }),
469        )
470        .await;
471
472    let scim_user = ScimUser::from_user(&final_user, &http_req);
473    Ok(HttpResponse::Created()
474        .append_header(("Location", scim_user.meta.location.clone()))
475        .json(scim_user))
476}

478/// `PUT /scim/v2/Users/{id}` — full replace (RFC 7644 §3.5.1).
479pub async fn replace<C: Connection + Clone>(
480    http_req: HttpRequest,
481    user: ScimPrincipal,
482    authz: AuthzData,
483    state: web::Data<AppState<C>>,
484    path: web::Path<Uuid>,
485    body: web::Json<ScimUserWrite>,
486) -> Result<HttpResponse, ScimError> {
487    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
488    let id = path.into_inner();
489    let current = state.user_repo.get_by_id(user.tenant_id(), id).await?;
490    let req = body.into_inner();
491
492    let email = primary_email(&req.emails).ok_or_else(|| {
493        ScimError::invalid_value("\"emails\" with at least one entry is required")
494    })?;
495
496    // PUT is a full replace of the SCIM-visible representation: rebuild the
497    // "scim" metadata sub-object from scratch, but leave any OTHER top-level
498    // metadata key (written by some other feature) exactly as it was.
499    let mut metadata = current.metadata.clone();
500    if let Some(obj) = metadata.as_object_mut() {
501        obj.remove("scim");
502    }
503    let scim_fields = build_scim_metadata(&req.external_id, &req.name);
504    if let Some(scim_obj) = scim_fields.get("scim") {
505        if !metadata.is_object() {
506            metadata = json!({});
507        }
508        metadata
509            .as_object_mut()
510            .expect("just ensured object above")
511            .insert("scim".to_string(), scim_obj.clone());
512    }
513
514    let status = if req.active {
515        UserStatus::Active
516    } else {
517        UserStatus::Inactive
518    };
519
520    let updated = state
521        .user_repo
522        .update(
523            user.tenant_id(),
524            id,
525            UpdateUser {
526                username: Some(req.user_name.clone()),
527                email: Some(email),
528                status: Some(status.clone()),
529                metadata: Some(metadata),
530                ..Default::default()
531            },
532        )
533        .await?;
534
535    // D7 parity with handlers::users::update: a PUT can narrow access via
536    // `status`, so flush this subject's cached authz decisions.
537    authz
538        .get_ref()
539        .as_ref()
540        .invalidate_subject(user.tenant_id(), id)
541        .await?;
542
543    // SEC-098: a PUT carrying `active: false` is the RFC 7644 §3.5.1 spelling
544    // of a deactivation and must revoke on the same terms as the PATCH one.
545    if status == UserStatus::Inactive {
546        revoke_live_credentials(&state, user.tenant_id(), id, "scim.deactivated").await;
547    }
548
549    state
550        .emit_webhook(
551            updated.tenant_id,
552            "user.updated",
553            json!({ "id": updated.id, "username": updated.username }),
554        )
555        .await;
556
557    Ok(HttpResponse::Ok().json(ScimUser::from_user(&updated, &http_req)))
558}

560/// Revoke every live credential a user holds (SEC-098).
561///
562/// Both chokepoints, exactly as `AuthService::revoke_all_sessions` does them:
563/// the session table (which backs the session-flow refresh token) *and* the
564/// OAuth2 refresh-token table. They are two tables and a credential change
565/// that hits only one leaves the other spendable.
566///
567/// # Why SCIM needs this and the native handlers were allowed not to
568///
569/// Flushing `invalidate_subject` — which every SCIM write already did — clears
570/// the *authorization decision* cache. It does not touch a single credential:
571/// the attacker's access token still validates until it expires, and their
572/// refresh token still mints new ones. So a password rotated through SCIM to
573/// lock out a compromised account, or an `active: false` written by an IdP's
574/// offboarding job, left the session it was meant to kill alive.
575///
576/// SCIM is the path an IdP drives *deprovisioning* through. RFC 7644's
577/// consumers — Okta, Entra — treat `active: false` and `DELETE` as "this
578/// person is gone as of now", and the immediacy is the whole reason the
579/// integration exists.
580///
581/// Best-effort by design: a revocation failure is logged at ERROR and does not
582/// fail the SCIM write. The write itself is the durable half (the password is
583/// changed, the account is deactivated, and the refresh path re-reads
584/// `check_user_status`); refusing the whole operation because the session
585/// table blipped would leave the IdP retrying a deprovisioning that has in
586/// fact already been applied.
587async fn revoke_live_credentials<C: Connection + Clone>(
588    state: &AppState<C>,
589    tenant_id: Uuid,
590    user_id: Uuid,
591    reason: &'static str,
592) {
593    use axiam_core::repository::{RefreshTokenRepository, SessionRepository};
594
595    if let Err(e) = state
596        .session_repo
597        .invalidate_user_sessions(tenant_id, user_id)
598        .await
599    {
600        tracing::error!(
601            %tenant_id, %user_id, reason, error = %e,
602            "SCIM: could not invalidate sessions after a credential-affecting write"
603        );
604    }
605    match state
606        .refresh_token_repo
607        .revoke_all_for_user(tenant_id, user_id)
608        .await
609    {
610        Ok(revoked) => tracing::info!(
611            target: "axiam::audit",
612            event = "scim.credentials_revoked",
613            %tenant_id, %user_id, reason, oauth2_tokens_revoked = revoked,
614            "SCIM write revoked every live session and OAuth2 refresh token for the user"
615        ),
616        Err(e) => tracing::error!(
617            %tenant_id, %user_id, reason, error = %e,
618            "SCIM: could not revoke OAuth2 refresh tokens after a credential-affecting write"
619        ),
620    }
621}

631fn apply_user_delta_metadata(current: &Value, delta: &UserPatchDelta) -> Option<Value> {
632    let touches_metadata = delta.external_id.is_some()
633        || delta.given_name.is_some()
634        || delta.family_name.is_some()
635        || delta.formatted.is_some();
636    if !touches_metadata {
637        return None;
638    }
639    let mut metadata = current.clone();
640    if let Some(ext) = &delta.external_id {
641        scim_metadata::set_str(&mut metadata, "externalId", ext.clone());
642    }
643    if let Some(g) = &delta.given_name {
644        scim_metadata::set_str(&mut metadata, "givenName", g.clone());
645    }
646    if let Some(f) = &delta.family_name {
647        scim_metadata::set_str(&mut metadata, "familyName", f.clone());
648    }
649    if let Some(fmt) = &delta.formatted {
650        scim_metadata::set_str(&mut metadata, "formatted", fmt.clone());
651    }
652    Some(metadata)
653}

655/// `PATCH /scim/v2/Users/{id}` (RFC 7644 §3.5.2) — op subset in
656/// [`crate::patch`].
657pub async fn patch<C: Connection + Clone>(
658    http_req: HttpRequest,
659    user: ScimPrincipal,
660    authz: AuthzData,
661    state: web::Data<AppState<C>>,
662    path: web::Path<Uuid>,
663    body: web::Json<PatchRequest>,
664) -> Result<HttpResponse, ScimError> {
665    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
666    let id = path.into_inner();
667    let current = state.user_repo.get_by_id(user.tenant_id(), id).await?;
668    let delta = parse_user_patch(&body.into_inner())?;
669
670    // Hash with the server-configured pepper, exactly as the bootstrap and login
671    // paths do (see `axiam_api_rest::handlers::bootstrap`). Hashing with `None`
672    // here would store a hash computed WITHOUT the pepper while login verifies
673    // WITH it, so every SCIM-provisioned user would fail authentication.
674    let pepper = state.auth_config.pepper.as_ref().map(|p| p.expose_secret());
675    let password_hash = match &delta.password {
676        Some(pw) => Some(password::hash_password(pw, pepper).map_err(|e| {
677            ScimError::new(
678                StatusCode::INTERNAL_SERVER_ERROR,
679                format!("password hashing failed: {e}"),
680            )
681        })?),
682        None => None,
683    };
684
685    let update = UpdateUser {
686        username: delta.username.clone(),
687        email: delta.email.clone(),
688        status: delta.active.map(|a| {
689            if a {
690                UserStatus::Active
691            } else {
692                UserStatus::Inactive
693            }
694        }),
695        metadata: apply_user_delta_metadata(&current.metadata, &delta),
696        password_hash,
697        ..Default::default()
698    };
699
700    // SEC-098: read before `update` is moved. A password write and a
701    // deactivation are the two PATCH shapes that must not leave a live
702    // credential behind.
703    let revocation_reason = match (
704        update.password_hash.is_some(),
705        update.status == Some(UserStatus::Inactive),
706    ) {
707        (true, _) => Some("scim.password_set"),
708        (false, true) => Some("scim.deactivated"),
709        (false, false) => None,
710    };
711
712    let updated = if user_patch_is_noop(&update) {
713        current
714    } else {
715        let u = state.user_repo.update(user.tenant_id(), id, update).await?;
716        authz
717            .get_ref()
718            .as_ref()
719            .invalidate_subject(user.tenant_id(), id)
720            .await?;
721        if let Some(reason) = revocation_reason {
722            revoke_live_credentials(&state, user.tenant_id(), id, reason).await;
723        }
724        state
725            .emit_webhook(
726                u.tenant_id,
727                "user.updated",
728                json!({ "id": u.id, "username": u.username }),
729            )
730            .await;
731        u
732    };
733
734    Ok(HttpResponse::Ok().json(ScimUser::from_user(&updated, &http_req)))
735}

737/// `DELETE /scim/v2/Users/{id}`
738pub async fn delete<C: Connection + Clone>(
739    user: ScimPrincipal,
740    authz: AuthzData,
741    state: web::Data<AppState<C>>,
742    path: web::Path<Uuid>,
743) -> Result<HttpResponse, ScimError> {
744    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
745    let id = path.into_inner();
746    state.user_repo.delete(user.tenant_id(), id).await?;
747
748    authz
749        .get_ref()
750        .as_ref()
751        .invalidate_subject(user.tenant_id(), id)
752        .await?;
753
754    // SEC-098: `DELETE /Users/{id}` is a soft delete to `Inactive`, so without
755    // this the offboarded account keeps a live session and a spendable refresh
756    // token. This is the endpoint an IdP calls when someone leaves.
757    revoke_live_credentials(&state, user.tenant_id(), id, "scim.deleted").await;
758
759    state
760        .emit_webhook(user.tenant_id(), "user.deleted", json!({ "id": id }))
761        .await;
762
763    Ok(HttpResponse::NoContent().finish())
764}
