1//! SCIM `/Users` resource (RFC 7643 §4.1, RFC 7644 §3.2-3.6).
2//!
3//! Maps onto the EXISTING `UserRepository` (B4) — no parallel storage.
4//! AXIAM's `User` model has no `externalId`/`name.*` columns, so those live
5//! inside the existing generic `metadata` JSON column under a `"scim"`
6//! sub-key (see [`crate::scim_metadata`]); everything else (`userName` <->
7//! `username`, `active` <-> `status == Active`, `emails[primary]` <->
8//! `email`) is a direct field mapping.
9//!
10//! # `phoneNumbers` and `addresses` (X7 G8 / W7)
11//!
12//! Both were dropped on the floor until W7 — parsed by nobody, stored nowhere,
13//! and absent from every response. They now map onto real columns
14//! (`user.phone_number`, `user.address`), and the mapping is a rename rather
15//! than a translation because RFC 7643 §4.1.2's address members and OIDC
16//! Core §5.1.1's are the same six fields.
17//!
18//! **AXIAM keeps one of each.** OIDC Core §5.1 defines one `phone_number`
19//! claim and one `address` claim, so a multi-valued attribute would be storage
20//! with nothing to release it through; the entry marked `primary` wins, else
21//! the first, which is the rule `emails` already follows. `type` (`"work"`,
22//! `"home"`) is accepted and not stored, for the reason Art. 5(1)(c) gives:
23//! it is personal data nothing in this system reads.
24//!
25//! Provisioning them does **not** release them. What reaches a relying party
26//! is decided four gates later, at UserInfo, by the tenant switch, the
27//! client's registered scopes, the request's scopes and the subject's consent
28//! record — see `axiam_oauth2::sensitive`. An identity provider pushing a
29//! postal address is populating a profile, not authorising a disclosure.
30//!
31//! Every handler here mirrors the native `POST/GET/PUT/DELETE
32//! /api/v1/users` handlers (`axiam_api_rest::handlers::users`) for the parts
33//! B4 requires parity on: same `create_with_consent` GDPR path, the same
34//! `emit_webhook` event names (`user.created`/`user.updated`/
35//! `user.deleted`), and the same D7 `invalidate_subject` cache flush on a
36//! status-narrowing update.
37//!
38//! # ETag (optional-v2)
39//!
40//! Not implemented. `ServiceProviderConfig` advertises `etag.supported:
41//! false`. Emitting a weak ETag (`W/"<updated_at millis>"`) on every
42//! response, and honoring `If-Match` on PUT/PATCH/DELETE, is cheap enough to
43//! be a reasonable follow-on — it just wasn't judged in-scope for B4's
44//! "implement if cheap, otherwise leave a documented TODO."
45

137/// Body shape for both `POST /Users` (create) and `PUT /Users/{id}`
138/// (full replace) — RFC 7644 §3.3/§3.5.1 use the same resource
139/// representation for both.
140#[derive(Debug, Deserialize)]
141pub struct ScimUserWrite {

170fn primary_email(emails: &[ScimEmailInput]) -> Option<String> {
171    emails
172        .iter()
173        .find(|e| e.primary == Some(true))
174        .or_else(|| emails.first())
175        .map(|e| e.value.clone())
176}
177
178/// The one telephone number AXIAM keeps: the entry marked primary, else the
179/// first. The same rule [`primary_email`] uses, deliberately — a provisioning
180/// client that learns how AXIAM picks an email should not have to learn a
181/// second rule for a telephone number.
182///
183/// A blank `value` selects nothing rather than storing an empty string: an
184/// empty telephone number is an absent one, and storing it would make
185/// `phone_number_verified: false` appear for a claim with no value.
186fn primary_phone(phones: &[ScimPhoneInput]) -> Option<String> {
187    phones
188        .iter()
189        .find(|p| p.primary == Some(true))
190        .or_else(|| phones.first())
191        .map(|p| p.value.trim())
192        .filter(|v| !v.is_empty())
193        .map(str::to_owned)
194}
195
196/// The one address AXIAM keeps, chosen by the same rule, and dropped when
197/// every member of it is blank.
198fn primary_address(addresses: &[ScimAddressInput]) -> Option<Address> {
199    let chosen = addresses
200        .iter()
201        .find(|a| a.primary == Some(true))
202        .or_else(|| addresses.first())?;
203    let trim = |v: &Option<String>| {
204        v.as_deref()
205            .map(str::trim)
206            .filter(|s| !s.is_empty())
207            .map(str::to_owned)
208    };
209    let address = Address {
210        formatted: trim(&chosen.formatted),
211        street_address: trim(&chosen.street_address),
212        locality: trim(&chosen.locality),
213        region: trim(&chosen.region),
214        postal_code: trim(&chosen.postal_code),
215        country: trim(&chosen.country),
216    };
217    (!address.is_empty()).then_some(address)
218}

351impl ScimUser {
352    fn from_user(user: &User, req: &HttpRequest) -> Self {
353        let id = user.id.to_string();
354        Self {
355            schemas: [USER_SCHEMA],
356            external_id: scim_metadata::get_str(&user.metadata, "externalId"),
357            user_name: user.username.clone(),
358            name: ScimName::from_metadata(&user.metadata),
359            emails: if user.email.is_empty() {
360                vec![]
361            } else {
362                vec![ScimEmail {
363                    value: user.email.clone(),
364                    primary: true,
365                }]
366            },
367            phone_numbers: user
368                .phone_number
369                .iter()
370                .map(|value| ScimPhone {
371                    value: value.clone(),
372                    primary: true,
373                })
374                .collect(),
375            addresses: user.address.iter().map(ScimAddress::from).collect(),
376            active: user.status == UserStatus::Active,
377            meta: ScimMeta {
378                resource_type: "User",
379                created: user.created_at,
380                last_modified: user.updated_at,
381                location: location_url(req, "Users", &id),
382            },
383            id,
384        }
385    }
386}

423/// Full-tenant scan cap for the `externalId eq` filter (B4's filtering
424/// subset has no indexed lookup for it — only `userName` has a dedicated
425/// repository method). Documented limitation, not a silent truncation: a
426/// tenant with more users than this returns a filtered result computed over
427/// only the first [`EXTERNAL_ID_SCAN_CAP`] (by `created_at`), which is the
428/// same trade-off `axiam-api-rest`'s own admin-UI search boxes make today.
429const EXTERNAL_ID_SCAN_CAP: u64 = 1000;
430
431pub(crate) fn start_index_to_offset(start_index: Option<u64>) -> u64 {
432    start_index.unwrap_or(1).max(1) - 1
433}
434
435pub(crate) fn clamp_count(count: Option<u64>) -> u64 {
436    count.unwrap_or(50).min(200)
437}

443/// `GET /scim/v2/Users`
444pub async fn list<C: Connection + Clone>(
445    http_req: HttpRequest,
446    user: ScimPrincipal,
447    authz: AuthzData,
448    state: web::Data<AppState<C>>,
449    query: web::Query<ScimListQuery>,
450) -> Result<HttpResponse, ScimError> {
451    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
452    let query = query.into_inner();
453    let start_index = query.start_index.unwrap_or(1).max(1);
454    let offset = start_index_to_offset(query.start_index);
455    let limit = clamp_count(query.count);
456
457    let Some(raw_filter) = query.filter else {
458        return list_unfiltered(&http_req, &user, &state, offset, limit, start_index).await;
459    };
460
461    let filter = crate::filter::parse_eq_filter(&raw_filter, &["username", "externalid"])?;
462    match filter.attr.as_str() {
463        "username" => {
464            let (items, total) = match state
465                .user_repo
466                .get_by_username(user.tenant_id(), &filter.value)
467                .await
468            {
469                Ok(u) if offset == 0 && limit > 0 => (vec![u], 1),
470                Ok(_) => (vec![], 1),
471                Err(AxiamError::NotFound { .. }) => (vec![], 0),
472                Err(e) => return Err(e.into()),
473            };
474            let resources: Vec<ScimUser> = items
475                .iter()
476                .map(|u| ScimUser::from_user(u, &http_req))
477                .collect();
478            Ok(HttpResponse::Ok().json(ScimListResponse::new(resources, total, start_index)))
479        }
480        "externalid" => {
481            let all = state
482                .user_repo
483                .list(
484                    user.tenant_id(),
485                    Pagination {
486                        offset: 0,
487                        limit: EXTERNAL_ID_SCAN_CAP,
488                        search: None,
489                    },
490                )
491                .await?;
492            let matched: Vec<&User> = all
493                .items
494                .iter()
495                .filter(|u| {
496                    scim_metadata::get_str(&u.metadata, "externalId").as_deref()
497                        == Some(filter.value.as_str())
498                })
499                .collect();
500            let total = matched.len() as u64;
501            let page: Vec<ScimUser> = matched
502                .into_iter()
503                .skip(offset as usize)
504                .take(limit as usize)
505                .map(|u| ScimUser::from_user(u, &http_req))
506                .collect();
507            Ok(HttpResponse::Ok().json(ScimListResponse::new(page, total, start_index)))
508        }
509        // parse_eq_filter already rejects anything not in the allow-list.
510        _ => unreachable!("parse_eq_filter enforces the allowed-attribute list"),
511    }
512}
513
514async fn list_unfiltered<C: Connection + Clone>(
515    http_req: &HttpRequest,
516    user: &ScimPrincipal,
517    state: &web::Data<AppState<C>>,
518    offset: u64,
519    limit: u64,
520    start_index: u64,
521) -> Result<HttpResponse, ScimError> {
522    if limit == 0 {
523        // Still need an accurate totalResults; LIMIT 1 gets it cheaply.
524        let probe = state
525            .user_repo
526            .list(
527                user.tenant_id(),
528                Pagination {
529                    offset: 0,
530                    limit: 1,
531                    search: None,
532                },
533            )
534            .await?;
535        return Ok(HttpResponse::Ok().json(ScimListResponse::<ScimUser>::new(
536            vec![],
537            probe.total,
538            start_index,
539        )));
540    }
541    let result = state
542        .user_repo
543        .list(
544            user.tenant_id(),
545            Pagination {
546                offset,
547                limit,
548                search: None,
549            },
550        )
551        .await?;
552    let resources: Vec<ScimUser> = result
553        .items
554        .iter()
555        .map(|u| ScimUser::from_user(u, http_req))
556        .collect();
557    Ok(HttpResponse::Ok().json(ScimListResponse::new(resources, result.total, start_index)))
558}

576fn build_scim_metadata(external_id: &Option<String>, name: &Option<ScimNameInput>) -> Value {
577    let mut metadata = json!({});
578    if let Some(ext) = external_id {
579        scim_metadata::set_str(&mut metadata, "externalId", Some(ext.clone()));
580    }
581    if let Some(name) = name {
582        if let Some(g) = &name.given_name {
583            scim_metadata::set_str(&mut metadata, "givenName", Some(g.clone()));
584        }
585        if let Some(f) = &name.family_name {
586            scim_metadata::set_str(&mut metadata, "familyName", Some(f.clone()));
587        }
588        if let Some(fmt) = &name.formatted {
589            scim_metadata::set_str(&mut metadata, "formatted", Some(fmt.clone()));
590        }
591    }
592    metadata
593}

595/// A random, high-entropy password for SCIM-provisioned users that didn't

602/// `POST /scim/v2/Users`
603pub async fn create<C: Connection + Clone>(
604    http_req: HttpRequest,
605    user: ScimPrincipal,
606    authz: AuthzData,
607    state: web::Data<AppState<C>>,
608    body: web::Json<ScimUserWrite>,
609) -> Result<HttpResponse, ScimError> {
610    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
611    let req = body.into_inner();
612
613    let email = primary_email(&req.emails).ok_or_else(|| {
614        ScimError::invalid_value(
615            "\"emails\" with at least one entry is required to provision an AXIAM user",
616        )
617    })?;
618
619    let metadata = build_scim_metadata(&req.external_id, &req.name);
620    let password = req.password.clone().unwrap_or_else(random_password);
621
622    let input = CreateUser {
623        tenant_id: user.tenant_id(),
624        username: req.user_name.clone(),
625        email,
626        password,
627        metadata: Some(metadata),
628    };
629
630    // Mirrors handlers::users::create exactly (axiam-api-rest): the
631    // Art. 7 proof-of-consent row is written in the SAME transaction as the
632    // user, so a SCIM-provisioned user can never exist without it either.
633    let ip_address = client_ip(&http_req);
634    let ua = user_agent(&http_req);
635    let created = state
636        .user_repo
637        .create_with_consent(input, "terms_of_service", "current", ip_address, ua)
638        .await?;
639
640    // CreateUser has no `status` field — every new user starts
641    // PendingVerification regardless of what was requested, which SCIM's
642    // `active` (== status == Active) would read back as `false` even for a
643    // fixture that explicitly asked for `active: true`. Set the status
644    // explicitly with a follow-up update so `active` in the create response
645    // matches what was requested, in both directions.
646    let final_status = if req.active {
647        UserStatus::Active
648    } else {
649        UserStatus::Inactive
650    };
651    let final_user = state
652        .user_repo
653        .update(
654            user.tenant_id(),
655            created.id,
656            UpdateUser {
657                status: Some(final_status),
658                // W7 — written on the same follow-up update the status already
659                // needed, so provisioning a user with a telephone number costs
660                // no extra round trip. `Some(None)` when the client sent none,
661                // because a create is a statement about the whole resource.
662                phone_number: Some(primary_phone(&req.phone_numbers)),
663                address: Some(primary_address(&req.addresses)),
664                ..Default::default()
665            },
666        )
667        .await?;
668
669    // Same webhook event/payload shape as native user creation.
670    state
671        .emit_webhook(
672            final_user.tenant_id,
673            "user.created",
674            json!({ "id": final_user.id, "username": final_user.username }),
675        )
676        .await;
677
678    let scim_user = ScimUser::from_user(&final_user, &http_req);
679    Ok(HttpResponse::Created()
680        .append_header(("Location", scim_user.meta.location.clone()))
681        .json(scim_user))
682}

684/// `PUT /scim/v2/Users/{id}` — full replace (RFC 7644 §3.5.1).
685pub async fn replace<C: Connection + Clone>(
686    http_req: HttpRequest,
687    user: ScimPrincipal,
688    authz: AuthzData,
689    state: web::Data<AppState<C>>,
690    path: web::Path<Uuid>,
691    body: web::Json<ScimUserWrite>,
692) -> Result<HttpResponse, ScimError> {
693    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
694    let id = path.into_inner();
695    let current = state.user_repo.get_by_id(user.tenant_id(), id).await?;
696    let req = body.into_inner();
697
698    let email = primary_email(&req.emails).ok_or_else(|| {
699        ScimError::invalid_value("\"emails\" with at least one entry is required")
700    })?;
701
702    // PUT is a full replace of the SCIM-visible representation: rebuild the
703    // "scim" metadata sub-object from scratch, but leave any OTHER top-level
704    // metadata key (written by some other feature) exactly as it was.
705    let mut metadata = current.metadata.clone();
706    if let Some(obj) = metadata.as_object_mut() {
707        obj.remove("scim");
708    }
709    let scim_fields = build_scim_metadata(&req.external_id, &req.name);
710    if let Some(scim_obj) = scim_fields.get("scim") {
711        if !metadata.is_object() {
712            metadata = json!({});
713        }
714        metadata
715            .as_object_mut()
716            .expect("just ensured object above")
717            .insert("scim".to_string(), scim_obj.clone());
718    }
719
720    let status = if req.active {
721        UserStatus::Active
722    } else {
723        UserStatus::Inactive
724    };
725
726    let updated = state
727        .user_repo
728        .update(
729            user.tenant_id(),
730            id,
731            UpdateUser {
732                username: Some(req.user_name.clone()),
733                email: Some(email),
734                status: Some(status.clone()),
735                metadata: Some(metadata),
736                // W7 — `PUT` replaces the resource (RFC 7644 §3.5.1), so an
737                // omitted `phoneNumbers` clears the stored number rather than
738                // leaving it. That is the behaviour a provisioning client
739                // relies on to *remove* an attribute, and it is also what
740                // makes an identity provider the source of truth rather than
741                // one of two.
742                phone_number: Some(primary_phone(&req.phone_numbers)),
743                address: Some(primary_address(&req.addresses)),
744                ..Default::default()
745            },
746        )
747        .await?;
748
749    // D7 parity with handlers::users::update: a PUT can narrow access via
750    // `status`, so flush this subject's cached authz decisions.
751    authz
752        .get_ref()
753        .as_ref()
754        .invalidate_subject(user.tenant_id(), id)
755        .await?;
756
757    // SEC-098: a PUT carrying `active: false` is the RFC 7644 §3.5.1 spelling
758    // of a deactivation and must revoke on the same terms as the PATCH one.
759    if status == UserStatus::Inactive {
760        revoke_live_credentials(&state, user.tenant_id(), id, "scim.deactivated").await;
761    }
762
763    state
764        .emit_webhook(
765            updated.tenant_id,
766            "user.updated",
767            json!({ "id": updated.id, "username": updated.username }),
768        )
769        .await;
770
771    Ok(HttpResponse::Ok().json(ScimUser::from_user(&updated, &http_req)))
772}

774/// Revoke every live credential a user holds (SEC-098).
775///
776/// Both chokepoints, exactly as `AuthService::revoke_all_sessions` does them:
777/// the session table (which backs the session-flow refresh token) *and* the
778/// OAuth2 refresh-token table. They are two tables and a credential change
779/// that hits only one leaves the other spendable.
780///
781/// # Why SCIM needs this and the native handlers were allowed not to
782///
783/// Flushing `invalidate_subject` — which every SCIM write already did — clears
784/// the *authorization decision* cache. It does not touch a single credential:
785/// the attacker's access token still validates until it expires, and their
786/// refresh token still mints new ones. So a password rotated through SCIM to
787/// lock out a compromised account, or an `active: false` written by an IdP's
788/// offboarding job, left the session it was meant to kill alive.
789///
790/// SCIM is the path an IdP drives *deprovisioning* through. RFC 7644's
791/// consumers — Okta, Entra — treat `active: false` and `DELETE` as "this
792/// person is gone as of now", and the immediacy is the whole reason the
793/// integration exists.
794///
795/// Best-effort by design: a revocation failure is logged at ERROR and does not
796/// fail the SCIM write. The write itself is the durable half (the password is
797/// changed, the account is deactivated, and the refresh path re-reads
798/// `check_user_status`); refusing the whole operation because the session
799/// table blipped would leave the IdP retrying a deprovisioning that has in
800/// fact already been applied.
801async fn revoke_live_credentials<C: Connection + Clone>(
802    state: &AppState<C>,
803    tenant_id: Uuid,
804    user_id: Uuid,
805    reason: &'static str,
806) {
807    use axiam_core::repository::{RefreshTokenRepository, SessionRepository};
808
809    if let Err(e) = state
810        .session_repo
811        .invalidate_user_sessions(tenant_id, user_id)
812        .await
813    {
814        tracing::error!(
815            %tenant_id, %user_id, reason, error = %e,
816            "SCIM: could not invalidate sessions after a credential-affecting write"
817        );
818    }
819    match state
820        .refresh_token_repo
821        .revoke_all_for_user(tenant_id, user_id)
822        .await
823    {
824        Ok(revoked) => tracing::info!(
825            target: "axiam::audit",
826            event = "scim.credentials_revoked",
827            %tenant_id, %user_id, reason, oauth2_tokens_revoked = revoked,
828            "SCIM write revoked every live session and OAuth2 refresh token for the user"
829        ),
830        Err(e) => tracing::error!(
831            %tenant_id, %user_id, reason, error = %e,
832            "SCIM: could not revoke OAuth2 refresh tokens after a credential-affecting write"
833        ),
834    }
835}

837/// Whether a parsed PATCH asks for no change at all.
838///
839/// # Every settable field has to appear here
840///
841/// This is the guard that decides whether `update` is called, so a field the
842/// list forgets is a field a PATCH cannot write — and it fails in the worst
843/// possible way, with `200 OK` and the unchanged resource echoed back. A caller
844/// that sets only that attribute is told it succeeded.
845///
846/// `phone_number` and `address` (X7 G8 / W7) were added to `UpdateUser` and to
847/// the PATCH parser and not to this list, so `PATCH /scim/v2/Users/{id}` with
848/// `phoneNumbers` and `addresses` — nothing else — answered 200, wrote nothing,
849/// and the OIDF `oidcc-scope-address` and `oidcc-scope-phone` modules then
850/// reported UserInfo withholding claims for a user who did not in fact have
851/// them. Four release gates were examined before the missing line was found,
852/// because every one of them was working.
853fn user_patch_is_noop(u: &UpdateUser) -> bool {
854    // Destructured rather than field-tested, so that adding a field to
855    // `UpdateUser` fails to compile here instead of silently becoming
856    // unwritable. The `..` cases are the ones SCIM never sets.
857    let UpdateUser {
858        username,
859        email,
860        status,
861        metadata,
862        password_hash,
863        phone_number,
864        address,
865        ..
866    } = u;
867    username.is_none()
868        && email.is_none()
869        && status.is_none()
870        && metadata.is_none()
871        && password_hash.is_none()
872        && phone_number.is_none()
873        && address.is_none()
874}
875
876fn apply_user_delta_metadata(current: &Value, delta: &UserPatchDelta) -> Option<Value> {
877    let touches_metadata = delta.external_id.is_some()
878        || delta.given_name.is_some()
879        || delta.family_name.is_some()
880        || delta.formatted.is_some();
881    if !touches_metadata {
882        return None;
883    }
884    let mut metadata = current.clone();
885    if let Some(ext) = &delta.external_id {
886        scim_metadata::set_str(&mut metadata, "externalId", ext.clone());
887    }
888    if let Some(g) = &delta.given_name {
889        scim_metadata::set_str(&mut metadata, "givenName", g.clone());
890    }
891    if let Some(f) = &delta.family_name {
892        scim_metadata::set_str(&mut metadata, "familyName", f.clone());
893    }
894    if let Some(fmt) = &delta.formatted {
895        scim_metadata::set_str(&mut metadata, "formatted", fmt.clone());
896    }
897    Some(metadata)
898}

900/// `PATCH /scim/v2/Users/{id}` (RFC 7644 §3.5.2) — op subset in
901/// [`crate::patch`].
902pub async fn patch<C: Connection + Clone>(
903    http_req: HttpRequest,
904    user: ScimPrincipal,
905    authz: AuthzData,
906    state: web::Data<AppState<C>>,
907    path: web::Path<Uuid>,
908    body: web::Json<PatchRequest>,
909) -> Result<HttpResponse, ScimError> {
910    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
911    let id = path.into_inner();
912    let current = state.user_repo.get_by_id(user.tenant_id(), id).await?;
913    let delta = parse_user_patch(&body.into_inner())?;
914
915    // Hash with the server-configured pepper, exactly as the bootstrap and login
916    // paths do (see `axiam_api_rest::handlers::bootstrap`). Hashing with `None`
917    // here would store a hash computed WITHOUT the pepper while login verifies
918    // WITH it, so every SCIM-provisioned user would fail authentication.
919    let pepper = state.auth_config.pepper.as_ref().map(|p| p.expose_secret());
920    let password_hash = match &delta.password {
921        Some(pw) => Some(password::hash_password(pw, pepper).map_err(|e| {
922            ScimError::new(
923                StatusCode::INTERNAL_SERVER_ERROR,
924                format!("password hashing failed: {e}"),
925            )
926        })?),
927        None => None,
928    };
929
930    let update = UpdateUser {
931        username: delta.username.clone(),
932        email: delta.email.clone(),
933        status: delta.active.map(|a| {
934            if a {
935                UserStatus::Active
936            } else {
937                UserStatus::Inactive
938            }
939        }),
940        metadata: apply_user_delta_metadata(&current.metadata, &delta),
941        password_hash,
942        // W7 — `None` when the PATCH said nothing about the attribute, which
943        // is the whole difference between PATCH and PUT: a partial update
944        // leaves an unmentioned telephone number alone, where a `PUT` that
945        // omits it clears it.
946        phone_number: delta.phone_number.clone(),
947        address: delta.address.clone(),
948        ..Default::default()
949    };
950
951    // SEC-098: read before `update` is moved. A password write and a
952    // deactivation are the two PATCH shapes that must not leave a live
953    // credential behind.
954    let revocation_reason = match (
955        update.password_hash.is_some(),
956        update.status == Some(UserStatus::Inactive),
957    ) {
958        (true, _) => Some("scim.password_set"),
959        (false, true) => Some("scim.deactivated"),
960        (false, false) => None,
961    };
962
963    let updated = if user_patch_is_noop(&update) {
964        current
965    } else {
966        let u = state.user_repo.update(user.tenant_id(), id, update).await?;
967        authz
968            .get_ref()
969            .as_ref()
970            .invalidate_subject(user.tenant_id(), id)
971            .await?;
972        if let Some(reason) = revocation_reason {
973            revoke_live_credentials(&state, user.tenant_id(), id, reason).await;
974        }
975        state
976            .emit_webhook(
977                u.tenant_id,
978                "user.updated",
979                json!({ "id": u.id, "username": u.username }),
980            )
981            .await;
982        u
983    };
984
985    Ok(HttpResponse::Ok().json(ScimUser::from_user(&updated, &http_req)))
986}

988/// `DELETE /scim/v2/Users/{id}`
989pub async fn delete<C: Connection + Clone>(
990    user: ScimPrincipal,
991    authz: AuthzData,
992    state: web::Data<AppState<C>>,
993    path: web::Path<Uuid>,
994) -> Result<HttpResponse, ScimError> {
995    require_scim_provision(&user, authz.get_ref().as_ref()).await?;
996    let id = path.into_inner();
997    state.user_repo.delete(user.tenant_id(), id).await?;
998
999    authz
1000        .get_ref()
1001        .as_ref()
1002        .invalidate_subject(user.tenant_id(), id)
1003        .await?;
1004
1005    // SEC-098: `DELETE /Users/{id}` is a soft delete to `Inactive`, so without
1006    // this the offboarded account keeps a live session and a spendable refresh
1007    // token. This is the endpoint an IdP calls when someone leaves.
1008    revoke_live_credentials(&state, user.tenant_id(), id, "scim.deleted").await;
1009
1010    state
1011        .emit_webhook(user.tenant_id(), "user.deleted", json!({ "id": id }))
1012        .await;
1013
1014    Ok(HttpResponse::NoContent().finish())
1015}
