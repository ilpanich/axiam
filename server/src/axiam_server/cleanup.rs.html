1//! Periodic cleanup task for expired federation rows, GDPR purges, and export jobs.
2//!
3//! SurrealDB v3 does not support native TTL on rows (RESEARCH §7), so this task
4//! periodically sweeps various tables:
5//! - `saml_assertion_replay` and `federation_login_state`: expired rows
6//! - `user`: accounts past their scheduled purge date (D-05/D-06/D-08)
7//! - `export_job`: queued jobs waiting to have their encrypted blob generated (D-12)
8//! - every tenant-scoped table: the rows of a deleted (tombstoned) tenant, then
9//!   its tenant row (#523, D-4)

11//! The task shuts down cleanly when the caller sends `true` through the watch

209/// How often the `tenant_purge` sweep also looks for orphaned tenant ids.
210///
211/// The look reads every tenant-scoped table (one `GROUP BY` each), which is not
212/// something to do every minute for a residue that only an upgrade can have: a
213/// deletion made by this version tombstones, so it never leaves an orphan. Once
214/// at process start, which is when an upgraded deployment first runs this
215/// code, and once a day after that.
216pub const ORPHAN_TENANT_SCAN_INTERVAL: Duration = Duration::from_secs(24 * 60 * 60);
217
218/// How long a replica's claim on a tombstoned tenant's purge keeps the other
219/// replicas off it (#610); a claim older than this is taken over.
220///
221/// The purge is not batched: it is one `DELETE … WHERE tenant_id = $id` per
222/// tenant-scoped table, about seventy statements in sequence, each removing
223/// all of the tenant's rows of its table at once (the certificate tables three
224/// statements each), so even a large tenant's purge is expected to take
225/// minutes; an hour leaves an order of magnitude of headroom before a purge
226/// still running could be overlapped — and an overlap is safe in any case: the
227/// steps are idempotent and only one purge's final `DELETE` finds the tenant
228/// row, so the purge is recorded once. What the lease costs is the delay
229/// after a replica dies holding a claim: the tenant waits at most an hour for
230/// another replica, an erasure of data already out of every read path. Clock
231/// skew between replicas, which the stamp is written by, is seconds against an
232/// hour. A purge that does not finish releases its claim — one that fails, one
233/// given up because the process is stopping, and (#655) one aborted or
234/// panicking, from a drop guard — so the next tick, or the next start, resumes
235/// it at once; only a process that ends before the release is written leaves
236/// the tenant to the lease: a `SIGKILL`, or a unit cut at the stop's bound,
237/// whose guard releases the claim from a spawned task that the process exit may
238/// cancel (#687).
239pub const TENANT_PURGE_LEASE: Duration = Duration::from_secs(60 * 60);
240
241/// How long a unit's claim on its pending-record marker (#689) keeps a sweep
242/// from recording the work itself: the tenant purge's lease, for the same
243/// reasons. A unit holds the claim for the handful of statements between its
244/// point of no return and its record, which the teardown bounds at seconds; a
245/// unit that is cut releases it from a drop guard, so the next start records
246/// the work at once, and only a process that ends before the release is
247/// written — a `SIGKILL`, or a unit cut at the stop's bound, whose spawned
248/// release the process exit may cancel (#687) — leaves the marker to this lease.
249pub const PENDING_RECORD_LEASE: Duration = TENANT_PURGE_LEASE;

255const fn longest(a: u64, b: u64) -> u64 {
256    if a > b { a } else { b }
257}

259/// The longest fixed window any writer of `rate_limit_bucket` counts in, in
260/// seconds. Derived from the writers, not chosen:
261///
262/// - the REST and gRPC shared counters, whose window is
263///   [`axiam_db::rate_limit_counter::DEFAULT_WINDOW_SECS`] (the REST layer's
264///   private `WINDOW_SECS` and the gRPC layer's are each asserted equal to it
265///   by their own tests; the gRPC one is read here as well);
266/// - the CIBA hint cap, [`axiam_oauth2::ciba::HINT_CAP_WINDOW_SECS`].
267///
268/// `rg -n "increment_by" crates/` finds no other writer; a new one with a
269/// longer window must be added to this expression, or the sweep deletes its
270/// rows early (see [`RATE_LIMIT_BUCKET_RETENTION`] for what that costs).
271pub const RATE_LIMIT_BUCKET_LONGEST_WINDOW_SECS: u64 = longest(
272    axiam_oauth2::ciba::HINT_CAP_WINDOW_SECS,
273    longest(
274        axiam_db::rate_limit_counter::DEFAULT_WINDOW_SECS,
275        axiam_api_grpc::middleware::rate_limit::WINDOW_SECS as u64,
276    ),
277);

279/// Slack on top of two windows, covering the clock offset between a writer, a
280/// reader and the datastore: one minute, far beyond what NTP leaves.
281pub const RATE_LIMIT_BUCKET_CLOCK_MARGIN: Duration = Duration::from_secs(60);
282
283/// How long after its last write a `rate_limit_bucket` row is kept: twice
284/// [`RATE_LIMIT_BUCKET_LONGEST_WINDOW_SECS`], plus
285/// [`RATE_LIMIT_BUCKET_CLOCK_MARGIN`].
286///
287/// # Why no live limiter reads a row this old
288///
289/// A bucket row holds the count of one fixed window `[s, s + W)`, `W` its
290/// writer's window. A write lands at some `u >= s`: it is made while the
291/// window is open, and a write-behind flush of the window comes later still.
292/// A limiter reads a stored count in two places only:
293///
294/// 1. the window it belongs to, at a time `t < s + W` (the shared counters'
295///    flush returns it, and the hint cap adds to it);
296/// 2. the hint cap's *previous* window, at `t < s + 2W` (the one later window
297///    that adds it to its own count).
298///
299/// From `t >= s + 2W` on, nothing reads the stored count as a count: an
300/// upsert for any later window finds `window_start` older than its own and
301/// starts from its delta, and a peek finds the same and answers `0` — both
302/// exactly what an absent row gives. So a row is dead once `s + 2W` has
303/// passed, and `s <= u`.
304///
305/// The sweep deletes a row only when `updated_at` (the datastore's clock at
306/// the last write, `u`) is more than `2L + margin` in the past of the
307/// *datastore's* clock (`time::now()`, evaluated in the selection and in the
308/// `DELETE`, #676), `L >= W` being the longest window in use. The sweeping
309/// replica's own clock takes no part in the comparison, so its offset from the
310/// datastore is not an input. A reader at `t` (its own clock) after the sweep
311/// has `t >= u + 2L + margin - offset >= s + 2W` as long as the offset between
312/// the *reader's* clock and the datastore's is under the margin. No row a
313/// limiter still reads is ever deleted.
314///
315/// If a reader's clock were off by more than the margin, the cost is a count
316/// read as `0` for the rest of one window — the same thing a window boundary
317/// does. For the hint cap that is a stretch that may admit up to twice the
318/// allowance, so its "exactly" holds under the assumption that readers and
319/// the datastore agree to within the margin (T-424).
320///
321/// The table is bounded by what was written in the last `RETENTION`; the
322/// sweep makes the growth of a table that gained a row per hint, per client
323/// IP and per endpoint, forever, a steady state.
324pub const RATE_LIMIT_BUCKET_RETENTION: Duration = Duration::from_secs(
325    2 * RATE_LIMIT_BUCKET_LONGEST_WINDOW_SECS + RATE_LIMIT_BUCKET_CLOCK_MARGIN.as_secs(),
326);

328/// Rows one `rate_limit_bucket` delete removes at most.
329pub const RATE_LIMIT_BUCKET_SWEEP_BATCH: u32 = 500;

331/// Deletes in one tick: with [`RATE_LIMIT_BUCKET_SWEEP_BATCH`], 50 000 rows,
332/// after which the next tick continues. A tick is never held open by a table
333/// that grew for months before this sweep existed.
334pub const RATE_LIMIT_BUCKET_SWEEP_MAX_BATCHES: u32 = 100;
335
336/// One pass of the `rate_limit_bucket` job (#638): delete the buckets whose
337/// last write is older than [`RATE_LIMIT_BUCKET_RETENTION`] on the datastore's
338/// clock, in batches; returns how many were removed.
339///
340/// Safe for any number of replicas sweeping at once: each delete re-checks
341/// `updated_at` on the row it removes, so a row is removed once, and a row a
342/// limiter has just written is kept. A public free function, for the reason
343/// [`run_erasure_pipeline`] is.
344pub async fn sweep_rate_limit_buckets<C: Connection + Send + Sync + 'static>(
345    repo: &axiam_db::SurrealRateLimitBucketRepository<C>,
346) -> Result<u64, AxiamError> {
347    let mut removed = 0u64;
348    for _ in 0..RATE_LIMIT_BUCKET_SWEEP_MAX_BATCHES {
349        let n = repo
350            .delete_stale(RATE_LIMIT_BUCKET_RETENTION, RATE_LIMIT_BUCKET_SWEEP_BATCH)
351            .await?;
352        removed += n;
353        if n < u64::from(RATE_LIMIT_BUCKET_SWEEP_BATCH) {
354            break;
355        }
356    }
357    Ok(removed)
358}

368/// One pass of the `tenant_purge` job: purge every tombstoned tenant and, when
369/// `scan_orphans`, every orphaned tenant id.
370///
371/// A free function, and public, for the reason [`run_erasure_pipeline`] is
372/// one. Each tenant is purged by
373/// [`SurrealTenantRepository::purge_tombstoned_rows`] — every tenant-scoped
374/// table in the order user erasure uses, of its audit trail the rows at or
375/// before the time of the export the deletion required (T-118, R1W1-04) — and
376/// then [`SurrealTenantRepository::remove_purged_tenant`], its tenant row; or,
377/// for an orphan, under its claim marker, by
378/// [`SurrealTenantRepository::purge_orphan_rows`], which leaves the audit
379/// trail to the retention sweep, and then by the same removal of the marker.
380/// Audit rows the purge keeps name a tenant that no longer exists and are
381/// deleted by the audit retention sweep (T-119) like any other. Each completed purge is recorded in the **system**
382/// log as `tenants.purged`, beside the `tenants.deleted` record the deletion
383/// wrote, with the export time used and the audit rows deleted and kept;
384/// neither record is ever purged.
385///
386/// Every replica runs this sweep (#610). A tombstoned tenant is purged only by
387/// the replica whose [`SurrealTenantRepository::claim_purge`] wins it; the
388/// others skip it for [`TENANT_PURGE_LEASE`]. A tenant the purge finds already
389/// purged — gone before the claim, or removed by an overlapping purge after a
390/// lease ran out — is success, not a failure, and is not recorded again: only
391/// the purge that removed the tenant row writes `tenants.purged`. An orphaned
392/// id is claimed the same way, by a marker row
393/// ([`SurrealTenantRepository::claim_orphan_purge`], R101W1-09 (a)): replicas
394/// whose daily scans coincide purge and record it once, and a scan that
395/// predates another replica's finished purge finds nothing left, writes no
396/// marker and records nothing (#714).
397///
398/// Each tenant's purge, from its claim to its record, is one unit of
399/// [`AuditedWork`] (#655): a task of its own, which aborting the cleanup task
400/// does not cut. The sweep starts no purge once the process is stopping
401/// (checked between tenants); a purge under way gives itself up at the stop as
402/// long as its tenant row is still there, releasing its claim, and once the row
403/// is removed it runs on to its record.
404///
405/// The removal leaves a `tenant_purge_pending` marker in its own transaction,
406/// which the unit clears once it has recorded the purge (#689). A unit cut in
407/// between leaves the marker, and every pass starts by recording what the
408/// markers left standing ([`record_pending_purges`]), so the next start writes
409/// the `tenants.purged` the cut unit did not, once.
410///
411/// * `Ok(n)` is the number of tenants (and orphaned ids) this call purged.
412/// * One tenant's failure does not stop the others; any failure makes the sweep
413///   fail, naming how many failed and never a tenant's data. The failed tenant
414///   stays tombstoned, its claim is released, and the next pass resumes it
415///   (every step is idempotent).
416/// * A stop ends the pass early with `Ok`: the tenants left stay tombstoned and
417///   unclaimed, and the next start purges them.
418///
419/// # Errors
420///
421/// [`AxiamError::Internal`] as above, or the datastore error that kept the
422/// tombstoned tenants or the orphans from being listed (the tombstoned tenants
423/// are still purged when only the orphans could not be).
424pub async fn sweep_tenant_purge<C, S>(
425    tenant_repo: &Arc<SurrealTenantRepository<C>>,
426    audit: &Arc<S>,
427    scan_orphans: bool,
428    work: &AuditedWork,
429) -> Result<u64, AxiamError>
430where
431    C: Connection,
432    S: AuditWriteSink + 'static,
433{
434    let lease = chrono::TimeDelta::from_std(TENANT_PURGE_LEASE)
435        .map_err(|e| AxiamError::Internal(format!("tenant purge lease: {e}")))?;
436    // #689: first, the purges a unit removed and was cut before recording. A
437    // failure here does not hold back the purges; it is reported after them.
438    let record_lease = chrono::TimeDelta::from_std(PENDING_RECORD_LEASE)
439        .map_err(|e| AxiamError::Internal(format!("pending record lease: {e}")))?;
440    let pending_error = record_pending_purges(tenant_repo, audit.as_ref(), record_lease, work)
441        .await
442        .err();
443
444    let mut due: Vec<(Uuid, bool)> = tenant_repo
445        .list_tombstoned()
446        .await?
447        .into_iter()
448        .map(|id| (id, false))
449        .collect();
450    // A failed look for orphans does not hold back the tombstoned tenants: it
451    // is reported after them, and the next pass looks again.
452    let mut scan_error = None;
453    if scan_orphans {
454        match tenant_repo.orphaned_tenant_ids().await {
455            Ok(ids) => due.extend(ids.into_iter().map(|id| (id, true))),
456            Err(e) => scan_error = Some(e),
457        }
458    }
459
460    let (mut purged, mut failed) = (0u64, 0u64);
461    let due_count = due.len();
462    for (started, (tenant_id, orphan)) in due.into_iter().enumerate() {
463        // #655: no purge starts once the process is stopping; the tenants left
464        // stay tombstoned, and the next start purges them.
465        if work.stopping() {
466            tracing::info!(
467                purged,
468                left = due_count - started,
469                "tenant purge sweep stopped: the process is stopping (the rest stay tombstoned)"
470            );
471            break;
472        }
473        let (tenants, sink, unit) = (Arc::clone(tenant_repo), Arc::clone(audit), work.clone());
474        let outcome = if orphan {
475            work.run(purge_orphan_unit(tenants, sink, tenant_id, unit))
476                .await
477        } else {
478            work.run(purge_tombstoned_unit(tenants, sink, tenant_id, lease, unit))
479                .await
480        };
481        match outcome {
482            Some(Ok(PurgeOutcome::Purged)) => purged += 1,
483            // Another replica holds it, or it is already gone.
484            Some(Ok(PurgeOutcome::Skipped)) => {}
485            // Given up at the stop (or still running, and left to the
486            // teardown's join): the loop's check above ends the pass.
487            Some(Ok(PurgeOutcome::Abandoned)) | None if work.stopping() => {}
488            Some(Ok(PurgeOutcome::Abandoned)) => {}
489            // The unit panicked; its claim guard released the claim.
490            None => {
491                failed += 1;
492                tracing::warn!(
493                    %tenant_id,
494                    orphan,
495                    "tenant purge ended without an outcome; the tenant stays tombstoned and the \
496                     next sweep resumes it"
497                );
498            }
499            Some(Err(e)) => {
500                failed += 1;
501                tracing::warn!(
502                    error = %e,
503                    %tenant_id,
504                    orphan,
505                    "tenant purge incomplete; the tenant stays tombstoned and the next sweep resumes it"
506                );
507            }
508        }
509    }
510
511    if failed > 0 {
512        return Err(AxiamError::Internal(format!(
513            "tenant purge was incomplete for {failed} of {} tenant(s)",
514            failed + purged
515        )));
516    }
517    if let Some(e) = scan_error.or(pending_error) {
518        return Err(e);
519    }
520    Ok(purged)
521}

523/// How one tenant's purge unit ended, when it did not fail.
524enum PurgeOutcome {
525    /// This unit purged the tenant and wrote `tenants.purged`.
526    Purged,
527    /// Another replica holds the claim, or the tenant is already purged (for
528    /// an orphaned id: a scan that predates its finished purge).
529    Skipped,
530    /// Given up because the process is stopping, before the point of no
531    /// return; the tenant stays tombstoned and unclaimed.
532    Abandoned,
533}

535/// One tombstoned tenant's purge, from its claim to its `tenants.purged`
536/// record: one unit of [`AuditedWork`] (#655).
537///
538/// The point of no return is the removal of the tenant row: before it every
539/// step is idempotent and the tenant stays tombstoned, so the unit gives up at
540/// the stop and releases its claim; it never removes the row once the process
541/// is stopping. After it, the record is the only thing left, and the unit
542/// writes it whatever the stop says.
543///
544/// The tombstoned row may be an orphan purge's claim marker whose claimant
545/// died or gave it up (R101W1-09 (a)): its rows are then purged as an orphan's
546/// and the record says `orphan: true` (the report's `orphan`).
547async fn purge_tombstoned_unit<C, S>(
548    tenants: Arc<SurrealTenantRepository<C>>,
549    audit: Arc<S>,
550    tenant_id: Uuid,
551    lease: chrono::TimeDelta,
552    work: AuditedWork,
553) -> Result<PurgeOutcome, AxiamError>
554where
555    C: Connection,
556    S: AuditWriteSink,
557{
558    let claimed_at = Utc::now();
559    if !tenants.claim_purge(tenant_id, claimed_at, lease).await? {
560        return Ok(PurgeOutcome::Skipped);
561    }
562    // From here every exit but a removed tenant row releases the claim — the
563    // awaited release below, or, for an abort or a panic, the guard's drop.
564    let claim = PurgeClaim {
565        tenants: Arc::clone(&tenants),
566        tenant_id,
567        claimed_at,
568        held: true,
569    };
570    let rows = work
571        .until_stopped(async {
572            let report = tenants.purge_tombstoned_rows(tenant_id).await?;
573            work.step(UnitStep::TenantRowsPurged(tenant_id)).await;
574            Ok::<_, AxiamError>(report)
575        })
576        .await;
577    let report = match rows {
578        Some(Ok(report)) => report,
579        // Purged by an overlapping replica: done, and recorded by the purge
580        // that removed the row.
581        Some(Err(AxiamError::NotFound { .. })) => {
582            claim.settle();
583            return Ok(PurgeOutcome::Skipped);
584        }
585        // Let the next tick (any replica's) retry it now rather than after
586        // the lease.
587        Some(Err(e)) => {
588            claim.release().await;
589            return Err(e);
590        }
591        None => {
592            claim.release().await;
593            return Ok(PurgeOutcome::Abandoned);
594        }
595    };
596    // Once the process is stopping, no new crossing of the point of no return
597    // starts (a crossing already being sent completes).
598    if work.stopping() {
599        claim.release().await;
600        return Ok(PurgeOutcome::Abandoned);
601    }
602    let Some(record) = cross_point_of_no_return(&tenants, tenant_id, &report, claim).await? else {
603        return Ok(PurgeOutcome::Skipped);
604    };
605    work.step(UnitStep::TenantRowRemoved(tenant_id)).await;
606    record_tenant_purged(audit.as_ref(), tenant_id, false, &report).await;
607    work.step(UnitStep::Recorded(tenant_id)).await;
608    record.settle().await;
609    Ok(PurgeOutcome::Purged)
610}

612/// A purge unit's point of no return (#655): remove the tenant row (or the
613/// orphan's marker) and, in the same transaction, write its
614/// `tenant_purge_pending` marker (#689), then settle the purge claim and hold
615/// the marker's in its place. `Ok(None)` when an overlapping purge removed the
616/// row first (it records the purge).
617///
618/// The marker carries a stamp of its own, taken here, rather than the purge
619/// claim's: the claim may be as old as the purge, and a stale stamp would let
620/// another replica's sweep take the marker while this unit records it.
621async fn cross_point_of_no_return<C: Connection>(
622    tenants: &Arc<SurrealTenantRepository<C>>,
623    tenant_id: Uuid,
624    report: &axiam_db::repository::tenant_purge::TenantPurgeReport,
625    claim: PurgeClaim<C>,
626) -> Result<Option<RecordClaim<C>>, AxiamError> {
627    let stamp = Utc::now();
628    match tenants.remove_purged_tenant(tenant_id, report, stamp).await {
629        Ok(()) => {
630            claim.settle();
631            Ok(Some(RecordClaim {
632                marker: PendingMarker::TenantPurge {
633                    tenants: Arc::clone(tenants),
634                    tenant_id,
635                },
636                stamp,
637                held: true,
638            }))
639        }
640        Err(AxiamError::NotFound { .. }) => {
641            claim.settle();
642            Ok(None)
643        }
644        Err(e) => {
645            // The transaction may have committed before the error reached us:
646            // release a marker stamped with ours, if there is one, so the next
647            // sweep records the purge at once. Nothing matches otherwise.
648            if let Err(release) = tenants.release_purge_record(tenant_id, stamp).await {
649                tracing::warn!(
650                    error = %release,
651                    %tenant_id,
652                    "could not release a tenant purge's pending record; it lapses with its lease"
653                );
654            }
655            claim.release().await;
656            Err(e)
657        }
658    }
659}

661/// Record the tenant purges whose `tenant_purge_pending` marker (#689) still
662/// stands — a unit cut between the removal of the tenant row and the clearing
663/// of its marker — and clear each marker. The first step of every
664/// [`sweep_tenant_purge`] pass; none starts once the process is stopping.
665///
666/// Each marker is claimed first
667/// ([`SurrealTenantRepository::claim_pending_purge_records`]): only one released
668/// or older than `lease`, and conditionally on the claim read, so a unit still
669/// recording and another replica's sweep are never both its owner. The record
670/// is written unless the system log already holds a `tenants.purged` for the
671/// id written since the removal (a unit cut after its record and before the
672/// clearing), with what the removal stored: the record is the one the unit
673/// would have written. The claim is held by the same guard a unit holds, so a
674/// pass cut here releases it.
675///
676/// `Ok(n)` is the number of records this call wrote.
677///
678/// # Errors
679///
680/// The datastore error that kept the markers from being read or claimed, or
681/// [`AxiamError::Internal`] naming how many could not be checked (each of
682/// those is released for the next pass).
683pub async fn record_pending_purges<C, S>(
684    tenants: &Arc<SurrealTenantRepository<C>>,
685    audit: &S,
686    lease: chrono::TimeDelta,
687    work: &AuditedWork,
688) -> Result<u64, AxiamError>
689where
690    C: Connection,
691    S: AuditWriteSink,
692{
693    if work.stopping() {
694        return Ok(0);
695    }
696    let pending = tenants
697        .claim_pending_purge_records(Utc::now(), lease)
698        .await?;
699    let (mut recorded, mut failed) = (0u64, 0u64);
700    for marker in pending {
701        let claim = RecordClaim {
702            marker: PendingMarker::TenantPurge {
703                tenants: Arc::clone(tenants),
704                tenant_id: marker.tenant_id,
705            },
706            stamp: marker.claimed_at,
707            held: true,
708        };
709        if work.stopping() {
710            claim.release().await;
711            continue;
712        }
713        match tenants
714            .purge_recorded_since(
715                marker.tenant_id,
716                axiam_api_rest::handlers::tenants::TENANT_PURGED_ACTION,
717                marker.purged_at,
718            )
719            .await
720        {
721            Ok(true) => {}
722            Ok(false) => {
723                record_tenant_purged(audit, marker.tenant_id, false, &marker.report).await;
724                recorded += 1;
725                tracing::warn!(
726                    tenant_id = %marker.tenant_id,
727                    "recorded a tenant purge whose unit was cut before its record (#689)"
728                );
729                work.step(UnitStep::Recorded(marker.tenant_id)).await;
730            }
731            Err(e) => {
732                failed += 1;
733                tracing::warn!(
734                    error = %e,
735                    tenant_id = %marker.tenant_id,
736                    "could not check a pending tenant purge record; the next pass retries it"
737                );
738                claim.release().await;
739                continue;
740            }
741        }
742        claim.settle().await;
743    }
744    if failed > 0 {
745        return Err(AxiamError::Internal(format!(
746            "{failed} pending tenant purge record(s) could not be checked"
747        )));
748    }
749    Ok(recorded)
750}

752/// One orphaned tenant id's purge and its `tenants.purged` record: one unit of
753/// [`AuditedWork`] (#655), claimed by a marker row (R101W1-09 (a)).
754///
755/// The unit arms its [`PurgeClaim`] **before**
756/// [`SurrealTenantRepository::claim_orphan_purge`] writes the marker, keyed by
757/// the stamp the marker will carry, so the marker is guarded from the instant
758/// it exists (#714): a cut anywhere from the `CREATE` on releases the claim,
759/// and the release is conditional on the stamp, so it changes nothing when the
760/// `CREATE` did not land. The claim writes the marker only while the id still
761/// owns a row to purge: another replica's claim, or a scan that predates a
762/// finished purge, is skipped, writes nothing and records nothing. Then the
763/// unit runs as [`purge_tombstoned_unit`] does: the rows
764/// ([`SurrealTenantRepository::purge_orphan_rows`]), then the removal of the
765/// marker ([`SurrealTenantRepository::remove_purged_tenant`]) as the point of
766/// no return, and the record only when this unit removed it. Given up at the
767/// stop, failed, or cut, the marker stays tombstoned with its claim released,
768/// and the tombstoned-tenant sweep purges it at once as the orphan it is, and
769/// records it — a marker always stands for a purge nobody has recorded.
770async fn purge_orphan_unit<C, S>(
771    tenants: Arc<SurrealTenantRepository<C>>,
772    audit: Arc<S>,
773    tenant_id: Uuid,
774    work: AuditedWork,
775) -> Result<PurgeOutcome, AxiamError>
776where
777    C: Connection,
778    S: AuditWriteSink,
779{
780    let claimed_at = Utc::now();
781    // Armed before the marker exists (#714): every exit from here but a
782    // removed marker — or a claim that wrote none — releases the claim, so the
783    // marker is the tombstoned-tenant sweep's at once.
784    let claim = PurgeClaim {
785        tenants: Arc::clone(&tenants),
786        tenant_id,
787        claimed_at,
788        held: true,
789    };
790    match tenants.claim_orphan_purge(tenant_id, claimed_at).await {
791        Ok(true) => {}
792        // No marker of ours: another replica's, nothing left, or no orphan.
793        Ok(false) => {
794            claim.settle();
795            tenants.orphan_claim_refusal(tenant_id).await?;
796            return Ok(PurgeOutcome::Skipped);
797        }
798        // The marker may have landed: release it, by its stamp.
799        Err(e) => {
800            claim.release().await;
801            return Err(e);
802        }
803    }
804    work.step(UnitStep::OrphanClaimed(tenant_id)).await;
805    let report = match work
806        .until_stopped(tenants.purge_orphan_rows(tenant_id))
807        .await
808    {
809        Some(Ok(report)) => report,
810        Some(Err(e)) => {
811            claim.release().await;
812            return Err(e);
813        }
814        None => {
815            claim.release().await;
816            return Ok(PurgeOutcome::Abandoned);
817        }
818    };
819    // Once the process is stopping, no new crossing of the point of no return
820    // starts (a crossing already being sent completes).
821    if work.stopping() {
822        claim.release().await;
823        return Ok(PurgeOutcome::Abandoned);
824    }
825    // `None`: the marker was taken over and removed by another purge, which
826    // records it.
827    let Some(record) = cross_point_of_no_return(&tenants, tenant_id, &report, claim).await? else {
828        return Ok(PurgeOutcome::Skipped);
829    };
830    work.step(UnitStep::TenantRowRemoved(tenant_id)).await;
831    record_tenant_purged(audit.as_ref(), tenant_id, true, &report).await;
832    work.step(UnitStep::Recorded(tenant_id)).await;
833    record.settle().await;
834    Ok(PurgeOutcome::Purged)
835}

874/// A replica's claim on a tombstoned tenant's purge (#610), released on every
875/// exit but a removed tenant row (#655). An orphan purge arms it before its
876/// claim marker is written, so the marker is guarded from the instant it
877/// exists (#714).
878///
879/// The purge releases it itself, awaited, where it can — a failure, or giving
880/// up at the stop. Every other exit — the unit aborted at the teardown's bound,
881/// a panic, a release that was itself cut — drops the guard still holding the
882/// claim, and the drop spawns the release on the runtime. Without that, the
883/// next start's sweep found a claim younger than [`TENANT_PURGE_LEASE`] and
884/// skipped the tenant for up to an hour. That release is a spawned task, and
885/// nothing waits for it: a unit cut at the teardown's bound (the one exit where
886/// the guard is the only release, typically a datastore too slow to answer the
887/// awaited one) is dropped inside [`AuditedWork::join`], and the process exit
888/// that follows drops the runtime, which cancels the task unless it has already
889/// landed. The tenant then waits out [`TENANT_PURGE_LEASE`] (#687; awaiting the
890/// release at the bound was declined, since it needs a budget line of its own).
891struct PurgeClaim<C: Connection> {
892    tenants: Arc<SurrealTenantRepository<C>>,
893    tenant_id: Uuid,
894    claimed_at: chrono::DateTime<Utc>,
895    held: bool,
896}
897
898impl<C: Connection> PurgeClaim<C> {
899    /// Release the claim, awaited. A release cut mid-way is retried by the
900    /// drop, since the claim counts as held until it has answered.
901    async fn release(mut self) {
902        if let Err(e) = self
903            .tenants
904            .release_purge_claim(self.tenant_id, self.claimed_at)
905            .await
906        {
907            tracing::warn!(
908                error = %e,
909                tenant_id = %self.tenant_id,
910                "could not release the tenant purge claim; it lapses with its lease"
911            );
912        }
913        self.held = false;
914    }
915
916    /// The tenant row is gone (this purge removed it, or another did): there is
917    /// nothing left to claim.
918    fn settle(mut self) {
919        self.held = false;
920    }
921}
922
923impl<C: Connection> Drop for PurgeClaim<C> {
924    fn drop(&mut self) {
925        if !self.held {
926            return;
927        }
928        let (tenants, tenant_id, claimed_at) =
929            (Arc::clone(&self.tenants), self.tenant_id, self.claimed_at);
930        match tokio::runtime::Handle::try_current() {
931            Ok(runtime) => {
932                runtime.spawn(async move {
933                    if let Err(e) = tenants.release_purge_claim(tenant_id, claimed_at).await {
934                        tracing::warn!(
935                            error = %e,
936                            %tenant_id,
937                            "could not release the claim of a tenant purge that was cut; it \
938                             lapses with its lease"
939                        );
940                    }
941                });
942            }
943            Err(_) => tracing::warn!(
944                %tenant_id,
945                "the claim of a tenant purge that was cut outside a runtime lapses with its lease"
946            ),
947        }
948    }
949}

1008            }
1009        }
1010    }

1013        match self {

1018                deletions,
1019                tenant_id,
1020                user_id,
1021            } => {
1022                deletions
1023                    .release_erasure_record(*tenant_id, *user_id, stamp)
1024                    .await
1025            }
1026        }
1027    }

1030/// A claim on a pending-record marker (#689), held from the point of no return
1031/// (or a sweep's claim) to the record: settled — the marker cleared, for an
1032/// erasure with its deletion request's completion — once the record is
1033/// written, and released on every other exit, so the next sweep records the
1034/// work at once rather than after [`PENDING_RECORD_LEASE`].
1035///
1036/// Like [`PurgeClaim`], an exit that cannot await — the unit aborted at the
1037/// teardown's bound, a panic — drops the guard still holding the claim, and the
1038/// drop spawns the release on the runtime. A unit cut at the teardown's bound
1039/// drops it just before the process exit drops the runtime, which cancels a
1040/// release that has not landed (see [`PurgeClaim`]); the marker then lapses with
1041/// the lease (#687).
1042/// Every write is conditional on `stamp`, so a marker another replica has
1043/// taken over is left to it.
1044struct RecordClaim<C: Connection> {
1045    marker: PendingMarker<C>,
1046    stamp: chrono::DateTime<Utc>,
1047    held: bool,
1048}
1049
1050impl<C: Connection> RecordClaim<C> {
1051    /// The record is written: clear the marker. A failed clearing leaves it for
1052    /// the next sweep, which finds the record already written and clears it.
1053    async fn settle(mut self) {
1054        if let Err(e) = self.marker.settle(self.stamp).await {
1055            tracing::warn!(
1056                error = %e,
1057                id = %self.marker.id(),
1058                "could not clear a pending-record marker; the next sweep finds the record \
1059                 written and clears it"
1060            );
1061        }
1062        self.held = false;
1063    }
1064
1065    /// Give the claim up, awaited, without recording.
1066    async fn release(mut self) {

1068            tracing::warn!(
1069                error = %e,
1070                id = %self.marker.id(),
1071                "could not release a pending-record marker; it lapses with its lease"
1072            );
1073        }
1074        self.held = false;
1075    }
1076}

1079    fn drop(&mut self) {
1080        if !self.held {
1081            return;
1082        }
1083        let (marker, stamp) = (self.marker.clone(), self.stamp);
1084        match tokio::runtime::Handle::try_current() {
1085            Ok(runtime) => {
1086                runtime.spawn(async move {
1087                    if let Err(e) = marker.release(stamp).await {
1088                        tracing::warn!(
1089                            error = %e,
1090                            id = %marker.id(),
1091                            "could not release the pending-record marker of a unit that was cut; \
1092                             it lapses with its lease"
1093                        );
1094                    }
1095                });
1096            }
1097            Err(_) => tracing::warn!(
1098                id = %self.marker.id(),
1099                "the pending-record marker of a unit cut outside a runtime lapses with its lease"
1100            ),
1101        }
1102    }

1147/// The cleanup task's units of work that must end in an audit row (#655,
1148/// #624): a tombstoned tenant's purge through its `tenants.purged` record, an
1149/// orphaned tenant id's purge through its record, and one account's erasure
1150/// through its `account_deletion` completion and its `gdpr.user_pseudonymized`
1151/// row.
1152///
1153/// The cleanup task runs them one at a time, each as a `tokio::spawn`ed task of
1154/// its own ([`Self::run`]), so aborting the cleanup task — the composition
1155/// root's bounded join — does not cut one. The root then waits for the unit in
1156/// flight with [`Self::join`], within the same bound, and aborts it past that,
1157/// so a replica that lost its lease writes nothing of its own code after its
1158/// stop's budget (a statement already sent may still commit).
1159///
1160/// Each unit has a **point of no return** — the write after which no later
1161/// sweep finds the work again: the removal of the tenant row; the anonymization
1162/// of the account. Before it, the unit gives itself up as soon as the process
1163/// is stopping (the tenant stays tombstoned, its claim released; the account
1164/// stays due), and it starts no new crossing once the stop is requested (a
1165/// crossing already being sent completes). So the only unit a stop can find
1166/// past its point of no return began crossing it before the stop was requested, and it has the whole of the teardown's window for its last
1167/// statements — from the stop signal through the gRPC stop, the lease release
1168/// and the outbound and audit drains to the end of the cleanup bound: 5 to 18 s
1169/// for `SIGTERM`, 2 to 13 s after a lost lease, whose stop signal comes at the
1170/// loss (#628). One that outlasts even that is cut; the marker its point of
1171/// no return left in the same transaction (#689) makes the next start's sweep
1172/// write the record it owed, once ([`record_owed_erasures`],
1173/// [`record_pending_purges`]; T-444).
1174#[derive(Clone)]
1175pub struct AuditedWork {
1176    stop: watch::Receiver<bool>,
1177    in_flight: Arc<std::sync::Mutex<Option<tokio::task::JoinHandle<()>>>>,
1178    step_for_tests: Option<UnitStepForTests>,
1179}
1180
1181impl AuditedWork {
1182    /// Units that give themselves up when `stop` reads `true` — the cleanup
1183    /// task's shutdown signal. A `stop` whose sender is gone while it reads
1184    /// `false` never stops anything.
1185    #[must_use]
1186    pub fn new(stop: watch::Receiver<bool>) -> Self {
1187        Self {
1188            stop,
1189            in_flight: Arc::default(),
1190            step_for_tests: None,
1191        }
1192    }
1193
1194    /// **Test seam, never set in production** (#655, #624): run `step` at each
1195    /// [`UnitStep`], so a test can hold a unit there across a stop.
1196    #[doc(hidden)]
1197    #[must_use]
1198    pub fn with_step_for_tests(mut self, step: Option<UnitStepForTests>) -> Self {
1199        self.step_for_tests = step;
1200        self
1201    }
1202
1203    /// Whether the process has asked the cleanup task to stop.
1204    #[must_use]
1205    pub fn stopping(&self) -> bool {
1206        *self.stop.borrow()
1207    }
1208
1209    /// Resolves once the process is stopping; never, when it cannot be.
1210    async fn stopped(&self) {
1211        let mut stop = self.stop.clone();
1212        if stop.wait_for(|stopping| *stopping).await.is_err() {
1213            std::future::pending::<()>().await;
1214        }
1215    }
1216
1217    /// `work`'s output, or `None` when the process is stopping first: `work`
1218    /// is then dropped where it is. For the steps before a unit's point of no
1219    /// return only.
1220    async fn until_stopped<F: std::future::Future>(&self, work: F) -> Option<F::Output> {
1221        tokio::select! {
1222            biased;
1223            output = work => Some(output),
1224            () = self.stopped() => None,
1225        }
1226    }
1227
1228    /// The test step at `at`, if one is set.
1229    async fn step(&self, at: UnitStep) {
1230        if let Some(step) = &self.step_for_tests {
1231            step(at).await;
1232        }
1233    }
1234
1235    fn slot(&self) -> std::sync::MutexGuard<'_, Option<tokio::task::JoinHandle<()>>> {
1236        self.in_flight
1237            .lock()
1238            .unwrap_or_else(std::sync::PoisonError::into_inner)
1239    }
1240
1241    /// Spawn `unit` as a task of its own and wait for its output.
1242    ///
1243    /// `None` when the process is stopping — before the unit was started, or
1244    /// while it runs: it is then left running for [`Self::join`] — or when the
1245    /// unit panicked (logged here). One unit at a time: the caller awaits this
1246    /// before it starts the next.
1247    pub async fn run<T: Send + 'static>(
1248        &self,
1249        unit: impl std::future::Future<Output = T> + Send + 'static,
1250    ) -> Option<T> {
1251        if self.stopping() {
1252            return None;
1253        }
1254        let (tx, rx) = tokio::sync::oneshot::channel();
1255        *self.slot() = Some(tokio::spawn(async move {
1256            let _ = tx.send(unit.await);
1257        }));
1258        tokio::select! {
1259            biased;
1260            output = rx => {
1261                let handle = self.slot().take();
1262                match output {
1263                    Ok(output) => Some(output),
1264                    Err(_) => {
1265                        if let Some(handle) = handle
1266                            && let Err(e) = handle.await
1267                        {
1268                            tracing::error!(error = %e, "a cleanup unit ended without an outcome");
1269                        }
1270                        None
1271                    }
1272                }
1273            }
1274            // Left to the teardown, which waits for it.
1275            () = self.stopped() => None,
1276        }
1277    }
1278
1279    /// Wait until `deadline` for the unit in flight, if any; past it, abort the
1280    /// unit — it is cut where it is — and say so with `false`. The composition
1281    /// root calls this once the cleanup task itself has ended or been aborted
1282    /// (#655, #624).
1283    pub async fn join(&self, deadline: tokio::time::Instant) -> bool {
1284        let Some(mut handle) = self.slot().take() else {
1285            return true;
1286        };
1287        match tokio::time::timeout_at(deadline, &mut handle).await {
1288            Ok(Ok(())) => true,
1289            Ok(Err(e)) => {
1290                tracing::error!(error = %e, "a cleanup unit ended without an outcome");
1291                true
1292            }
1293            Err(_) => {
1294                handle.abort();
1295                // Once this returns the unit has been dropped: it writes
1296                // nothing more, and its claim guard has run.
1297                let _ = handle.await;
1298                false
1299            }
1300        }
1301    }
1302}

1344/// The `certificate_retention` job (R101W1-05, #618): delete certificate rows

1348/// A revoked row is kept for as long as its issuer's revocation list can name

1351/// row in a live tenant before, so a revoked one — and the PEM, subject and
1352/// `metadata` in it, of an account the Art. 17 purge had erased — outlived any
1353/// purpose for the tenant's lifetime. A deleted tenant's rows are the tenant
1354/// purge's, not this sweep's. Two replicas may run it at once: each batch is
1355/// chosen and deleted by id, and a row the other replica removed first is not
1356/// counted. Touches no audit row.
1357///
1358/// Runs at most [`CERTIFICATE_RETENTION_MAX_BATCHES`] batches of
1359/// [`CERTIFICATE_RETENTION_BATCH`] rows, and stops between batches on a

1361/// the reason [`run_erasure_pipeline`] is.
1362///
1363/// # Errors
1364///

1366/// stay deleted.
1367pub async fn sweep_certificate_retention<CR>(

1369    shutdown: &watch::Receiver<bool>,
1370) -> Result<u64, AxiamError>
1371where
1372    CR: axiam_core::repository::CertificateRepository,
1373{
1374    let mut removed = 0_u64;
1375    for _ in 0..CERTIFICATE_RETENTION_MAX_BATCHES {
1376        if *shutdown.borrow() {
1377            break;
1378        }
1379        let batch = cert_repo
1380            .delete_past_retention(CERTIFICATE_RETENTION_BATCH)
1381            .await?;
1382        removed += batch;
1383        if batch < u64::from(CERTIFICATE_RETENTION_BATCH) {
1384            break;
1385        }
1386    }
1387    Ok(removed)
1388}

1394/// One pass of the directory sync job, as a sweep the scheduler can record.
1395///
1396/// A free function, and public, for the reason [`run_erasure_pipeline`] is one:
1397/// the loop that calls it is awkward to construct in a test, and the conversion
1398/// of the job's outcome into job health is the part worth testing directly.
1399///
1400/// * `Ok(n)` is the number of accounts the pass changed (deactivated, updated or
1401///   re-mapped); a pass that found nothing to do is `Ok(0)`.
1402/// * **Any tenant's failure makes the sweep fail** — an unreachable directory, a
1403///   refused search, the safety valve — with a message that names how many
1404///   tenants failed and the fixed tag of each, never a tenant's data. The other
1405///   tenants still ran; one tenant's outage is not another's.
1406/// * A shutdown signal abandons the pass (`Ok(0)`): every step the job takes is
1407///   idempotent or a compare-and-set, and the next process starts it again.
1408///
1409/// # Errors
1410///
1411/// [`AxiamError::Internal`] when the pass failed, as described above.
1412pub async fn sweep_directories<C: Connection + Send + Sync + 'static>(
1413    sync: &DirectorySyncJob<C>,
1414    mut shutdown: watch::Receiver<bool>,
1415) -> Result<u64, AxiamError> {
1416    let outcome = tokio::select! {
1417        // A shutdown request wins over starting work: polled first.
1418        biased;
1419        _ = async {
1420            // Resolves on a shutdown request; a dropped sender never does.
1421            loop {
1422                if *shutdown.borrow() {
1423                    return;
1424                }
1425                if shutdown.changed().await.is_err() {
1426                    std::future::pending::<()>().await;
1427                }
1428            }
1429        } => {
1430            tracing::info!("directory sync abandoned: the process is shutting down");
1431            return Ok(0);
1432        }
1433        outcome = sync.run_due() => outcome,
1434    };
1435    match outcome {
1436        Ok(summary) => match summary.failure_message() {
1437            Some(message) => Err(AxiamError::Internal(message)),
1438            None => Ok(summary.changed()),
1439        },
1440        Err(error) => Err(AxiamError::Internal(format!(
1441            "directory sync could not list the tenants' directories ({})",
1442            error.tag()
1443        ))),
1444    }
1445}

1451/// Whether a self-registered client is due to be swept (T21.4).
1452///
1453/// A free function, and public, for the reason [`run_erasure_pipeline`] is
1454/// one: the loop that calls it is awkward to construct in a test, and the
1455/// decision it makes — *delete somebody's client registration* — is the part
1456/// worth testing directly.
1457///
1458/// The clock it reads is `last_authorized_at` when the client has ever been
1459/// authorized and `created_at` when it has not.
1460///
1461/// `DcrSweepWindow::days == 0` is never due. Zero means "never sweep", which
1462/// an operator who prunes out of band may legitimately want, and reading it as
1463/// "sweep everything immediately" would delete a tenant's whole client table
1464/// on the next tick.
1465///
1466/// # The second clock (T21.8 / MCP-05)
1467///
1468/// A row that has **never** been authorized, in a tenant whose effective
1469/// `dynamic_registration` is `anonymous`, is measured against
1470/// [`axiam_core::models::settings::DCR_UNAUTHORIZED_CLIENT_TTL_SECS`] instead
1471/// — an hour. That constant's own documentation says why it is a constant and
1472/// what promoting it to a tenant setting would cost.
1473///
1474/// Those two conditions are the whole of the change, and each is load-bearing.
1475///
1476/// **Never authorized.** The thirty-day window is sized, in
1477/// `DEFAULT_DCR_UNUSED_CLIENT_TTL_DAYS`'s own doc comment, for "a client
1478/// somebody uses monthly". A client registered and never authorized is not
1479/// that client: every MCP client this phase exists to serve authorizes within
1480/// seconds of registering, because registration is the first step of the same
1481/// flow. One TTL served two situations that have nothing in common, which is
1482/// what made `dcr_max_clients` an availability budget a stranger could spend
1483/// for a month. The sweeper could always tell the two apart — the row carries
1484/// `last_authorized_at: None` — so the fix is a second clock, not a bigger
1485/// quota.
1486///
1487/// **`anonymous` only.** In `initial_access_token` mode the row exists because
1488/// an administrator minted a handle for it and somebody redeemed it. There is
1489/// no unauthenticated exposure to bound, and the thirty-day clock is the right
1490/// one: an operator who hands somebody a registration token on Friday should
1491/// not find the registration gone on Monday.
1492///
1493/// Every tenant on `disabled` or `initial_access_token` — which is every
1494/// tenant that did not opt into `anonymous` behind D3 — takes byte for byte
1495/// the path it took before T21.8.
1496pub fn dcr_client_is_due_for_sweep(
1497    client: &axiam_core::models::oauth2_client::OAuth2Client,
1498    ttl: DcrSweepWindow,
1499    now: chrono::DateTime<Utc>,
1500) -> bool {
1501    if client.last_authorized_at.is_none() && ttl.mode == DynamicRegistrationMode::Anonymous {
1502        return now - client.created_at
1503            > chrono::Duration::seconds(i64::from(DCR_UNAUTHORIZED_CLIENT_TTL_SECS));
1504    }
1505    if ttl.days == 0 {
1506        return false;
1507    }
1508    now - dcr_client_last_seen(client) > chrono::Duration::days(i64::from(ttl.days))
1509}

1649/// Delete externally registered clients that have gone unseen for longer than
1650/// their tenant's `dcr_unused_client_ttl_days` (T21.4, widened to `cimd` by
1651/// T21.8 / MCP-04).
1652///
1653/// One `managed_by` per call, so `/health/jobs` can distinguish the two
1654/// sweeps: they delete different things for different reasons and an operator
1655/// debugging one should not have to read the other's counter.
1656///
1657/// # What it will not touch
1658///
1659/// Never `admin`. An administrator's client is never swept, however long it
1660/// sits unused, because somebody decided it should exist and nothing here is
1661/// entitled to reverse that. The repository query filters on the value rather
1662/// than on "not admin", so a fourth provenance added later is opted **in** by
1663/// somebody writing it down.
1664///
1665/// # Why `cimd` is swept now, when T21.4 argued it should not be
1666///
1667/// T21.4's reasoning, which this function used to carry: a CIMD shadow row is
1668/// a cache of a document the client publishes, so deleting it would be
1669/// re-materialised on the next request and the TTL would mean nothing.
1670///
1671/// That is right about TTL semantics and says nothing about storage, which is
1672/// what MCP-04 is about: a cache that is never evicted is not a cache. An
1673/// inert row is still a row — listed on the OAuth2 clients page, counted in
1674/// every `list_all_by_managed_by` this function runs, and a permanent write a
1675/// stranger made at the cost of one unauthenticated request. And eviction on
1676/// last-seen is *consistent* with the old argument rather than against it: a
1677/// row deleted while its document is still published is re-materialised on the
1678/// next request, which is exactly what a cache should do. What it must not do
1679/// is delete a row that is still in use, which is what
1680/// [`cimd_client_last_seen`] is for — every resolve moves `updated_at`, so a
1681/// document presented once a day is never due under a 30-day TTL.
1682///
1683/// # Per-tenant TTL, resolved per row
1684///
1685/// The window is a tenant setting, so each row's tenant is resolved and its
1686/// effective settings read. Deployment-wide sweeps elsewhere in this file
1687/// (audit retention) deliberately do **not** work this way, and the difference
1688/// is who owns the decision: audit retention is a property of the datastore an
1689/// operator is responsible for, where this is a property of the tenant's
1690/// relationship with the clients it lets register.
1691///
1692/// The settings read is cached per tenant for the duration of one sweep —
1693/// `dcr_max_clients` bounds the rows per tenant, so a hundred clients in one
1694/// tenant is one read rather than a hundred.
1695///
1696/// A tenant whose settings cannot be read is skipped: the fail-closed
1697/// direction for a sweep that **deletes** is to delete nothing.
1698pub async fn sweep_unused_external_clients<CR, TR, SR, RT, AC, PR, DG, CB>(
1699    client_repo: &CR,
1700    tenant_repo: &TR,
1701    settings_repo: &SR,
1702    grants: &ClientGrantStores<RT, AC, PR, DG, CB>,
1703    managed_by: axiam_core::models::oauth2_client::ManagedBy,
1704    now: chrono::DateTime<Utc>,
1705) -> Result<u64, AxiamError>
1706where
1707    CR: axiam_core::repository::OAuth2ClientRepository,
1708    TR: TenantRepository,
1709    SR: axiam_core::repository::SettingsRepository,
1710    RT: axiam_core::repository::RefreshTokenRepository,
1711    AC: axiam_core::repository::AuthorizationCodeRepository,
1712    PR: axiam_core::repository::PushedAuthRequestRepository,
1713    DG: axiam_core::repository::DeviceGrantRepository,
1714    CB: axiam_core::repository::CibaRequestRepository,
1715{
1716    use axiam_core::models::oauth2_client::ManagedBy;
1717    use axiam_core::repository::SettingsRepository;
1718
1719    type DuePredicate = fn(
1720        &axiam_core::models::oauth2_client::OAuth2Client,
1721        DcrSweepWindow,
1722        chrono::DateTime<Utc>,
1723    ) -> bool;
1724    let (job, is_due): (&str, DuePredicate) = match managed_by {
1725        ManagedBy::Dcr => ("dcr_unused_clients", dcr_client_is_due_for_sweep),
1726        ManagedBy::Cimd => ("cimd_unused_clients", cimd_client_is_due_for_sweep),
1727        // An administrator's client is never swept. Answered here rather than
1728        // by trusting every caller, because the argument for it is a security
1729        // one and belongs next to the query.
1730        ManagedBy::Admin => return Ok(0),
1731    };
1732
1733    let clients = client_repo.list_all_by_managed_by(managed_by).await?;
1734    if clients.is_empty() {
1735        return Ok(0);
1736    }
1737
1738    let mut ttl_by_tenant: std::collections::HashMap<Uuid, Option<DcrSweepWindow>> =
1739        std::collections::HashMap::new();
1740    let mut removed = 0u64;
1741
1742    for client in clients {
1743        let ttl = match ttl_by_tenant.get(&client.tenant_id) {
1744            Some(cached) => *cached,
1745            None => {
1746                let resolved = match tenant_repo.get_by_id(client.tenant_id).await {
1747                    Ok(tenant) => SettingsRepository::get_effective_settings(
1748                        settings_repo,
1749                        tenant.organization_id,
1750                        client.tenant_id,
1751                    )
1752                    .await
1753                    .ok()
1754                    .map(|s| DcrSweepWindow {
1755                        days: s.oidc.dcr_unused_client_ttl_days,
1756                        mode: s.oidc.dynamic_registration,
1757                    }),
1758                    Err(_) => None,
1759                };
1760                ttl_by_tenant.insert(client.tenant_id, resolved);
1761                resolved
1762            }
1763        };
1764        // `None` is an unreadable tenant or an unreadable settings row.
1765        let Some(ttl) = ttl else {
1766            continue;
1767        };
1768        if !is_due(&client, ttl, now) {
1769            continue;
1770        }
1771
1772        // #517 — the client's grants go before the row, as on an
1773        // administrator's delete: a `cimd` row re-materialises under the same
1774        // `client_id`, and a refresh token, code or pushed request left behind
1775        // would work again against it. A revocation that fails keeps the row
1776        // for the next sweep rather than leave a row-less client with live
1777        // grants.
1778        if let Err(e) = grants.revoke(client.tenant_id, &client.client_id).await {
1779            tracing::warn!(
1780                job,
1781                error = %e,
1782                client_id = %client.client_id,
1783                "could not revoke an unused externally registered client's grants; not \
1784                 deleting it this sweep"
1785            );
1786            continue;
1787        }
1788        if let Err(e) = client_repo.delete(client.tenant_id, client.id).await {
1789            // One failure must not stop the sweep: the next row may be a
1790            // different tenant entirely.
1791            tracing::warn!(
1792                job,
1793                error = %e,
1794                client_id = %client.client_id,
1795                "could not delete an unused externally registered client"
1796            );
1797            continue;
1798        }
1799        // A request that read the row before the delete may have written a
1800        // grant between the two writes above; one more pass leaves none.
1801        if let Err(e) = grants.revoke(client.tenant_id, &client.client_id).await {
1802            tracing::warn!(
1803                job,
1804                error = %e,
1805                client_id = %client.client_id,
1806                "a swept client's grants could not be revoked a second time"
1807            );
1808        }
1809        removed += 1;
1810        tracing::info!(
1811            job,
1812            tenant_id = %client.tenant_id,
1813            client_id = %client.client_id,
1814            ttl_days = ttl.days,
1815            ever_authorized = client.last_authorized_at.is_some(),
1816            mode = %ttl.mode,
1817            "deleted an externally registered client that has gone unseen within its \
1818             tenant's TTL"
1819        );
1820    }
1821
1822    Ok(removed)
1823}

1947/// The three steps of [`run_erasure_pipeline`], with the SSF report (G-5, D-52)
1948/// between the second and the third.
1949///
1950/// The report follows the anonymization because that is the write that makes the
1951/// account gone — whether or not the erasure proof is then written. A proof that
1952/// fails to write leaves an anonymized account that no sweep selects again, so a
1953/// report that waited for the proof would never be sent.
1954///
1955/// With `work` (#624), the steps give up before the anonymization — the point of
1956/// no return — once the process is stopping, and answer `Ok(false)`: the audit
1957/// scrub is idempotent and the account stays due. `Ok(true)` once the account
1958/// is anonymized and its proof written.
1959#[allow(clippy::too_many_arguments)]
1960async fn erasure_steps<A, EP, U>(
1961    audit_repo: &A,
1962    erasure_proof_repo: &EP,
1963    user_repo: &U,
1964    tenant_id: Uuid,
1965    user_id: Uuid,
1966    pseudonym: &str,
1967    email_hash: &str,
1968    report: Option<(&dyn SsfSystemAccountSink, axiam_core::models::user::User)>,
1969    work: Option<&AuditedWork>,
1970) -> Result<bool, AxiamError>
1971where
1972    A: AuditLogRepository,
1973    EP: ErasureProofRepository,
1974    U: UserRepository,
1975{
1976    // FATAL now (was: `if let Err(e) = ... { tracing::warn!(...) }`).
1977    let scrub = audit_repo.pseudonymize_actor(tenant_id, user_id, pseudonym);
1978    match work {
1979        Some(work) => match work.until_stopped(scrub).await {
1980            Some(scrubbed) => {
1981                scrubbed?;
1982            }
1983            None => return Ok(false),
1984        },
1985        None => {
1986            scrub.await?;
1987        }
1988    }
1989    // #624: the anonymization is never started once the process is stopping.
1990    if work.is_some_and(AuditedWork::stopping) {
1991        return Ok(false);
1992    }
1993
1994    // Clears `deletion_pending` — the re-selection anchor. Must run before
1995    // the proof so a failure here (unreachable in practice since this is
1996    // now the second of three fallible steps) still leaves the user due.
1997    user_repo
1998        .anonymize_user(tenant_id, user_id, email_hash, pseudonym)
1999        .await?;
2000
2001    // G-5 (D-52): the account is gone; tell the receivers what it was.
2002    if let Some((sink, user)) = report {
2003        sink.account_purged(tenant_id, &user).await;
2004    }
2005
2006    // Written STRICTLY LAST — only reached once every step above succeeded.
2007    erasure_proof_repo
2008        .create(CreateErasureProof {
2009            pseudonym: pseudonym.to_string(),
2010            tenant_id,
2011            user_id,
2012            erased_at: Utc::now(),
2013        })
2014        .await?;
2015
2016    Ok(true)
2017}

2121/// An erasure through its point of no return to its record (#624): the erasure
2122/// pipeline ([`run_erasure_pipeline_reporting`]: audit scrub, anonymization,
2123/// proof), then the `gdpr.user_pseudonymized` audit row, then the
2124/// `account_deletion` row marked completed (#689 (i): the record first). Run

2126/// reach it.
2127///
2128/// The anonymization is the point of no return: it clears `deletion_pending`,
2129/// so no later sweep selects the account again, and in the same transaction
2130/// sets the account's record marker (#689), which this call holds from there
2131/// (released by a guard if it is cut) and clears with the completion once the
2132/// audit row is written; a marker left set is recorded by the next sweep
2133/// ([`record_owed_erasures`]). Until then — the audit scrub — the call gives
2134/// up once `work` is stopping and answers `Ok(false)`: the account stays due.
2135/// Past it, it runs on to the audit row whatever the stop says, and answers
2136/// `Ok(true)`. A free function, and public, for the reason
2137/// [`run_erasure_pipeline`] is one.
2138///
2139/// # Errors
2140///
2141/// Those of [`run_erasure_pipeline_reporting`], before anything is recorded.
2142/// The completion and the audit row are not fallible here: a failed completion
2143/// is logged, and a failed audit write is dead-lettered (SECHRD-12).
2144pub async fn erase_and_record<A, EP, U, CR, C>(
2145    audit_repo: &A,
2146    erasure_proof_repo: &EP,
2147    user_repo: &U,
2148    cert_repo: &CR,
2149    account_deletion_repo: &SurrealAccountDeletionRepository<C>,
2150    sink: Option<&dyn SsfSystemAccountSink>,
2151    subject: ErasureSubject<'_>,
2152    work: &AuditedWork,
2153) -> Result<bool, AxiamError>
2154where
2155    A: AuditLogRepository + AuditWriteSink,
2156    EP: ErasureProofRepository,
2157    U: UserRepository,
2158    CR: axiam_core::repository::CertificateRepository,
2159    C: Connection,
2160{
2161    let ErasureSubject {
2162        tenant_id,
2163        user_id,
2164        pseudonym,
2165        email_hash,
2166        certificates_revoked,
2167    } = subject;
2168    work.step(UnitStep::ErasurePrepared(user_id)).await;
2169    // (d)/(e)/(g) Run the erasure pipeline: pseudonymize audit actor refs
2170    // (now FATAL, was swallowed) -> anonymize the user row -> write the
2171    // erasure proof STRICTLY LAST (SECHRD-06 / D-03a, T-25-13). A failure
2172    // anywhere in this call propagates with `?` and aborts the purge:
2173    // `anonymize_user` (the only step that clears `deletion_pending`)
2174    // never ran, so the user stays due for a retry via
2175    // `find_due_for_purge`, and NO erasure proof is ever written for an
2176    // incomplete erasure.
2177    if !erasure_pipeline(

2179        erasure_proof_repo,
2180        user_repo,
2181        tenant_id,
2182        user_id,
2183        pseudonym,
2184        email_hash,
2185        sink,
2186        Some(work),
2187    )
2188    .await?
2189    {
2190        return Ok(false);
2191    }
2192    // #689: the anonymization stamped the account's erasure-record marker in
2193    // its own transaction. Hold it from here: cleared with the record below,
2194    // released by the guard if this unit is cut before, so the next sweep
2195    // writes the record at once.
2196    let record = match account_deletion_repo
2197        .erasure_record_stamp(tenant_id, user_id)
2198        .await
2199    {

2201            marker: PendingMarker::Erasure {
2202                deletions: account_deletion_repo.clone(),
2203                tenant_id,
2204                user_id,
2205            },
2206            stamp,
2207            held: true,
2208        }),
2209        // Taken over already, which only a unit held past the lease sees.
2210        Ok(None) => None,
2211        // The marker stays set: a sweep past the lease finds the record
2212        // written and clears it.
2213        Err(e) => {
2214            tracing::warn!(
2215                error = %e,
2216                %tenant_id,
2217                %user_id,

2219            );
2220            None
2221        }
2222    };
2223    work.step(UnitStep::ErasureAnonymized(user_id)).await;
2224
2225    // (#668) A second revocation, by binding only, outside the pinned pipeline
2226    // order: a certificate issued for the account between (b4) and the
2227    // anonymization is still active. The account's names are gone by now, so
2228    // empty names leave only the `user_id` binding (and `metadata.user_id`) to
2229    // match. Best effort and idempotent: past the point of no return a failure
2230    // is logged and never un-records the erasure.
2231    match cert_repo
2232        .revoke_user_certificates(tenant_id, user_id, "", "", Utc::now())
2233        .await
2234    {
2235        Ok(0) => {}
2236        Ok(n) => tracing::info!(
2237            %tenant_id,
2238            %user_id,
2239            revoked = n,
2240            "cleanup: revoked certificates issued for the erased account after its first revocation"
2241        ),
2242        Err(e) => tracing::warn!(
2243            error = %e,
2244            %tenant_id,
2245            %user_id,
2246            "cleanup: second certificate revocation after anonymization failed"
2247        ),
2248    }
2249
2250    // (h) Emit gdpr.user_pseudonymized audit event. actor_id is the
2251    // erased subject's own (now-anonymized) row id — a real, resolvable
2252    // identifier rather than a fabricated nil UUID (Task 1 fix); the row
2253    // still exists (anonymize-in-place preserves referential integrity),
2254    // so this is a legitimate audit-trail reference, not PII leakage.
2255    // A DB-write failure here is dead-lettered to BOTH an append-only
2256    // file AND a structured audit event (SECHRD-12 / D-02, T-24-61) —
2257    // this legally-significant record must never be silently lost.
2258    //
2259    // #689 (i): written BEFORE the request's completion. The completion
2260    // used to come first, so a cut between the two left a completed request
2261    // and no record; now a cut leaves the request pending with the account
2262    // anonymized, which the next sweep completes and records.
2263    write_audit_with_dead_letter(
2264        audit_repo,
2265        CreateAuditLogEntry {
2266            tenant_id,
2267            actor_id: user_id,
2268            actor_type: ActorType::System,
2269            action: USER_PSEUDONYMIZED_ACTION.into(),
2270            resource_id: None,
2271            outcome: AuditOutcome::Success,
2272            ip_address: None,
2273            metadata: Some(serde_json::json!({
2274                "pseudonym": pseudonym,
2275                // #533: the account's certificates this purge revoked; a
2276                // count, as directory linking records its own.
2277                "certificates_revoked": certificates_revoked,
2278            })),
2279        },
2280    )
2281    .await;
2282    work.step(UnitStep::Recorded(user_id)).await;
2283
2284    // (f) Complete the account_deletion row, and clear the record marker,
2285    // in one transaction (#689). Without the marker in hand — its read
2286    // failed above — the request is completed on its own and the marker is
2287    // left for a later sweep, which finds the record written.
2288    match record {
2289        Some(record) => record.settle().await,
2290        None => complete_deletion_request(account_deletion_repo, tenant_id, user_id).await,
2291    }
2292
2293    tracing::info!(
2294        pseudonym = %pseudonym,
2295        tenant_id = %tenant_id,
2296        "user purged and pseudonymized"
2297    );
2298    Ok(true)
2299}

2337/// Record the erasures whose record is still owed — a unit cut between the
2338/// anonymization and the clearing of its marker (#689) — and settle each: the
2339/// first step of every GDPR purge sweep; none starts once the process is
2340/// stopping.
2341///
2342/// [`SurrealAccountDeletionRepository::claim_owed_erasure_records`] claims
2343/// each owed erasure, conditionally on the marker read: a marker released by a
2344/// cut unit or older than `lease`, and (#689 (i)) an anonymized account whose
2345/// deletion request is still pending with no marker at all — an erasure cut
2346/// before schema v108 between its anonymization and its completion. The
2347/// `gdpr.user_pseudonymized` row is written unless the tenant's trail already
2348/// holds one for the account (a unit cut after its record and before the
2349/// clearing): an account is erased once, so that row is this erasure's. The
2350/// row carries the pseudonym the anonymization wrote, `certificates_revoked`
2351/// as `null` — the count lived only in the cut unit — and `recovered: true`.
2352/// Then the marker is cleared and the request completed, in one transaction.
2353/// The claim is held by the same guard the unit holds.
2354///
2355/// `Ok(n)` is the number of records this call wrote.
2356///
2357/// # Errors
2358///
2359/// The datastore error that kept the owed erasures from being read or
2360/// claimed, or [`AxiamError::Internal`] naming how many could not be checked
2361/// (each of those is released for the next sweep).
2362pub async fn record_owed_erasures<A, C>(
2363    audit_repo: &A,
2364    account_deletion_repo: &SurrealAccountDeletionRepository<C>,
2365    lease: chrono::TimeDelta,
2366    work: &AuditedWork,
2367) -> Result<u64, AxiamError>
2368where
2369    A: AuditWriteSink,
2370    C: Connection,
2371{
2372    if work.stopping() {
2373        return Ok(0);
2374    }
2375    let owed = account_deletion_repo
2376        .claim_owed_erasure_records(Utc::now(), lease)
2377        .await?;
2378    let (mut recorded, mut failed) = (0u64, 0u64);
2379    for erasure in owed {
2380        let (tenant_id, user_id) = (erasure.tenant_id, erasure.user_id);
2381        let claim = RecordClaim {
2382            marker: PendingMarker::Erasure {
2383                deletions: account_deletion_repo.clone(),
2384                tenant_id,
2385                user_id,
2386            },
2387            stamp: erasure.claimed_at,
2388            held: true,
2389        };
2390        if work.stopping() {
2391            claim.release().await;
2392            continue;
2393        }
2394        match account_deletion_repo
2395            .erasure_recorded(tenant_id, user_id, USER_PSEUDONYMIZED_ACTION)
2396            .await
2397        {
2398            Ok(true) => {}
2399            Ok(false) => {
2400                write_audit_with_dead_letter(
2401                    audit_repo,
2402                    CreateAuditLogEntry {
2403                        tenant_id,
2404                        actor_id: user_id,
2405                        actor_type: ActorType::System,
2406                        action: USER_PSEUDONYMIZED_ACTION.into(),
2407                        resource_id: None,
2408                        outcome: AuditOutcome::Success,
2409                        ip_address: None,
2410                        metadata: Some(serde_json::json!({
2411                            "pseudonym": erasure.pseudonym,
2412                            "certificates_revoked": null,
2413                            "recovered": true,
2414                        })),
2415                    },
2416                )
2417                .await;
2418                recorded += 1;
2419                tracing::warn!(
2420                    %tenant_id,
2421                    %user_id,
2422                    "recorded an erasure whose unit was cut before its record (#689)"
2423                );
2424                work.step(UnitStep::Recorded(user_id)).await;
2425            }
2426            Err(e) => {
2427                failed += 1;
2428                tracing::warn!(
2429                    error = %e,
2430                    %tenant_id,
2431                    %user_id,
2432                    "could not check an owed erasure record; the next sweep retries it"
2433                );
2434                claim.release().await;
2435                continue;
2436            }
2437        }
2438        claim.settle().await;
2439    }
2440    if failed > 0 {
2441        return Err(AxiamError::Internal(format!(
2442            "{failed} owed erasure record(s) could not be checked"
2443        )));
2444    }
2445    Ok(recorded)
2446}

2465impl<C: Connection> ErasureUnit<C> {
2466    /// The purge of `user` (steps listed at `CleanupTask::sweep_pending_purges`):
2467    /// `Ok(true)` once erased and recorded, `Ok(false)` when given up at the
2468    /// stop before the point of no return — the account stays due — or when the
2469    /// account is no longer due at the claim (its deletion was cancelled).
2470    async fn purge(
2471        self,
2472        user: axiam_core::models::user::User,
2473        pepper: [u8; 32],
2474        now: chrono::DateTime<Utc>,
2475        work: AuditedWork,
2476    ) -> Result<bool, AxiamError> {
2477        // Item 11 of the 1.1.0 atomicity audit: the erasure is claimed before
2478        // its first destructive step, in a transaction that re-checks the
2479        // account is still due at the sweep's `now`. A deletion cancelled since
2480        // the sweep's read is skipped here, before anything is revoked; once
2481        // claimed, a cancel is refused (`409`), so the subject is never told
2482        // "cancelled" for an account this purge goes on to erase.
2483        if !self
2484            .account_deletion_repo
2485            .claim_for_purge(user.tenant_id, user.id, now)
2486            .await?
2487        {
2488            tracing::info!(
2489                user_id = %user.id,
2490                tenant_id = %user.tenant_id,
2491                "purge skipped: the account's deletion was cancelled since the sweep read it"
2492            );
2493            return Ok(false);
2494        }
2495
2496        // Steps (a)…(b4) are idempotent and leave the account due: given up
2497        // where they are once the process is stopping.
2498        let Some(certificates_revoked) = work.until_stopped(self.prepare(&user)).await else {
2499            return Ok(false);
2500        };
2501        let certificates_revoked = certificates_revoked?;
2502        if work.stopping() {
2503            return Ok(false);
2504        }
2505
2506        let (user_id, tenant_id) = (user.id, user.tenant_id);
2507        // (c) Compute deterministic pseudonym (keyed HMAC-SHA256, D-02).
2508        let pseudonym = gdpr_pseudonym(&pepper, tenant_id, user_id);
2509
2510        // Derive email hash for anonymize_user from tenant+user IDs (original email
2511        // is no longer accessible at purge time — user already marked deletion_pending).
2512        use sha2::{Digest, Sha256};
2513        let mut h = Sha256::new();
2514        h.update(tenant_id.as_bytes());
2515        h.update(user_id.as_bytes());
2516        let email_hash = hex::encode(h.finalize());
2517
2518        // The anonymization inside is the point of no return; it is never
2519        // started once the process is stopping (#624).
2520        erase_and_record(
2521            self.audit_repo.as_ref(),
2522            self.erasure_proof_repo.as_ref(),
2523            self.user_repo.as_ref(),
2524            self.cert_repo.as_ref(),
2525            self.account_deletion_repo.as_ref(),
2526            self.ssf_sink.as_deref(),
2527            ErasureSubject {
2528                tenant_id,
2529                user_id,
2530                pseudonym: &pseudonym,
2531                email_hash: &email_hash,
2532                certificates_revoked,
2533            },
2534            &work,
2535        )
2536        .await
2537    }

2539    /// Steps (a)…(b4): sessions, federation links, credentials, the
2540    /// authorization graph and the account's certificates. Returns how many
2541    /// certificates were revoked.
2542    async fn prepare(&self, user: &axiam_core::models::user::User) -> Result<u64, AxiamError> {
2543        let (user_id, tenant_id) = (user.id, user.tenant_id);
2544        // (a) Revoke all sessions and OAuth2 refresh tokens.
2545        self.auth_svc
2546            .revoke_all_sessions(tenant_id, user_id)
2547            .await?;
2548
2549        // (b) Hard-delete federation identity links.
2550        match self
2551            .federation_link_repo
2552            .get_by_user_id(tenant_id, user_id)
2553            .await
2554        {
2555            Ok(links) => {
2556                for link in links {
2557                    if let Err(e) = self.federation_link_repo.delete(tenant_id, link.id).await {
2558                        tracing::warn!(
2559                            error = %e,
2560                            %tenant_id,
2561                            link_id = %link.id,
2562                            "cleanup: failed to delete expired federation link; will retry next cycle"
2563                        );
2564                    }
2565                }
2566            }
2567            Err(e) => {
2568                tracing::warn!(error = ?e, user_id = %user_id, "failed to list federation links for purge");
2569            }
2570        }
2571
2572        // (b2) Hard-delete stored credential material (SEC-056). Done with `?`
2573        // (fail-closed): erasure must not be certified while a user's passkeys or
2574        // password hashes remain. A transient failure aborts the purge and it is
2575        // retried next sweep — every step here is idempotent.
2576        let webauthn_creds = self.webauthn_repo.list_by_user(tenant_id, user_id).await?;
2577        for cred in webauthn_creds {
2578            self.webauthn_repo.delete(tenant_id, cred.id).await?;
2579        }
2580        // Prune the entire password-history chain (keep_count = 0 deletes all).
2581        self.password_history_repo
2582            .prune(tenant_id, user_id, 0)
2583            .await?;
2584
2585        // (b3) Prune authorization-graph edges so the anonymized row is left as
2586        // an isolated tombstone — no dangling `member_of` / `has_role` relations
2587        // skewing group-member queries or leaving live authorization paths.
2588        // Remove group memberships first so the subsequent role pass only sees
2589        // the user's *direct* has_role edges (inherited ones vanish with the
2590        // membership). Fail-closed (`?`): the tombstone must not be created while
2591        // graph edges survive; every delete is idempotent and retried on failure.
2592        let groups = self.group_repo.get_user_groups(tenant_id, user_id).await?;
2593        for group in groups {
2594            self.group_repo
2595                .remove_member(tenant_id, user_id, group.id)
2596                .await?;
2597        }
2598        let assignments = self
2599            .role_repo
2600            .get_user_role_assignments(tenant_id, user_id)
2601            .await?;
2602        for a in assignments {
2603            self.role_repo
2604                .unassign_from_user(tenant_id, user_id, a.role.id, a.resource_id)
2605                .await?;
2606        }
2607
2608        // (b4) Revoke the account's `User` certificates (#533) and clear their
2609        // `metadata` (R101W1-05), while its username and email — which the
2610        // convention for a certificate issued without a `user_id` reads — are
2611        // still on the row. Fail-closed, like the steps above; idempotent (an
2612        // already revoked row is not matched, a cleared one is not rewritten).
2613        revoke_erased_users_certificates(self.cert_repo.as_ref(), user).await
2614    }
2615}

2858    /// Run the cleanup loop until a shutdown signal is received.
2859    ///
2860    /// Never returns `Err` — all sweep errors are logged at `warn` level and
2861    /// the loop continues (T-04-36).
2862    pub async fn run(mut self) -> Result<(), AxiamError> {
2863        let mut ticker = tokio::time::interval(self.interval);
2864        // Skip ticks that were missed while the sweep was running to prevent
2865        // catch-up storms after a pause (T-04-35).
2866        ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
2867
2868        loop {
2869            // A stop that arrived during a tick ends the loop here, whatever the
2870            // ticker says: the select below may otherwise pick a tick that is
2871            // already due over the shutdown change.
2872            if self.stop_requested() {
2873                tracing::info!("cleanup task received shutdown signal");
2874                return Ok(());
2875            }
2876            tokio::select! {
2877                _ = ticker.tick() => {
2878                    self.tick().await;
2879                }
2880                changed = self.shutdown.changed() => {
2881                    if changed.is_ok() && *self.shutdown.borrow() {
2882                        tracing::info!("cleanup task received shutdown signal");
2883                        return Ok(());
2884                    }
2885                }
2886            }
2887        }
2888    }

3308    /// held as a fixed instant, so a long-running process prunes a moving
3309    /// window rather than freezing the boundary at whenever it started.
3310    ///
3311    /// Runs deployment-wide, not per tenant: retention here is a property of
3312    /// the datastore an operator is responsible for, and a per-tenant override
3313    /// would let one tenant's settings decide how long another tenant's
3314    /// records survive on shared storage. Per-tenant retention is a real
3315    /// requirement, but it needs a settings surface (org baseline + tenant
3316    /// override + API + UI) rather than a field on this sweep.
3317    async fn sweep_audit_retention(&self) -> Result<u64, AxiamError> {
3318        let Some(retention) = self.audit_retention else {
3319            return Ok(0);

3676    /// Process a single queued export job.
3677    async fn process_export_job(
3678        &self,
3679        job_id: Uuid,
3680        tenant_id: Uuid,
3681        user_id: Uuid,
3682        key: [u8; 32],
3683    ) -> Result<(), AxiamError> {
3684        // (a)/(b) Aggregate Art. 15 inventory into one sectioned JSON.
3685        let export_json = self.aggregate_export_data(tenant_id, user_id).await?;
3686        let export_bytes = export_json.to_string().into_bytes();
3687
3688        // (c) Encrypt the blob (D-12).
3689        let (nonce_b64, ct_b64) = encrypt_separate(&key, &export_bytes)
3690            .map_err(|e| AxiamError::Internal(format!("export encrypt failed: {e}")))?;

3692        // (d) Generate single-use 24h download token (D-13).
3693        //
3694        // Deliberately v4, not `new_id()`: this token is the sole bearer
3695        // credential for the export download, so it must stay maximally random.
3696        // See `axiam_core::id` — v7 is for identifiers, never for secrets.
3697        let raw_download_token = Uuid::new_v4().to_string();

3699            use sha2::{Digest, Sha256};
3700            let mut h = Sha256::new();
3701            h.update(raw_download_token.as_bytes());
3702            hex::encode(h.finalize())
3703        };
3704        let expires_at = Utc::now() + chrono::Duration::hours(24);
3705

3707            .set_ready(
3708                job_id,
3709                token_hash,
3710                Some(ct_b64),
3711                None, // file_path: stored as DB blob
3712                Some(nonce_b64),
3713                expires_at,
3714            )
3715            .await?;

3722            Ok(tenant) => tenant.organization_id,
3723            Err(e) => {

3726                    %tenant_id,
3727                    "failed to resolve org_id for ExportReady mail — falling back to nil (mail may be undeliverable)"
3728                );
3729                Uuid::nil()
3730            }
3731        };

3736            mail_type: MailType::ExportReady,
3737            tenant_id,
3738            org_id,
3739            user_id,
3740            to_address: String::new(), // mail consumer resolves from user_id
