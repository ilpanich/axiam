1//! Periodic cleanup task for expired federation rows, GDPR purges, and export jobs.
2//!
3//! SurrealDB v3 does not support native TTL on rows (RESEARCH §7), so this task
4//! periodically sweeps various tables:
5//! - `saml_assertion_replay` and `federation_login_state`: expired rows

8//! - every tenant-scoped table: the rows of a deleted (tombstoned) tenant, then
9//!   its tenant row (#523, D-4)

11//! The task shuts down cleanly when the caller sends `true` through the watch

209/// How often the `tenant_purge` sweep also looks for orphaned tenant ids.
210///
211/// The look reads every tenant-scoped table (one `GROUP BY` each), which is not
212/// something to do every minute for a residue that only an upgrade can have: a
213/// deletion made by this version tombstones, so it never leaves an orphan. Once
214/// at process start, which is when an upgraded deployment first runs this
215/// code, and once a day after that.
216pub const ORPHAN_TENANT_SCAN_INTERVAL: Duration = Duration::from_secs(24 * 60 * 60);
217
218/// How long a replica's claim on a tombstoned tenant's purge keeps the other
219/// replicas off it (#610); a claim older than this is taken over.
220///
221/// The purge is not batched: it is one `DELETE … WHERE tenant_id = $id` per
222/// tenant-scoped table, about seventy statements in sequence, each removing
223/// all of the tenant's rows of its table at once (the certificate tables three
224/// statements each), so even a large tenant's purge is expected to take
225/// minutes; an hour leaves an order of magnitude of headroom before a purge
226/// still running could be overlapped — and an overlap is safe in any case: the
227/// steps are idempotent and only one purge's final `DELETE` finds the tenant
228/// row, so the purge is recorded once. What the lease costs is the delay
229/// after a replica dies holding a claim: the tenant waits at most an hour for
230/// another replica, an erasure of data already out of every read path. Clock
231/// skew between replicas, which the stamp is written by, is seconds against an
232/// hour. A purge that does not finish releases its claim — one that fails, one
233/// given up because the process is stopping, and (#655) one aborted or
234/// panicking, from a drop guard — so the next tick, or the next start, resumes
235/// it at once; only a process that ends before the release is written (a
236/// `SIGKILL`, or a stop that runs out of time) leaves the tenant to the lease.
237pub const TENANT_PURGE_LEASE: Duration = Duration::from_secs(60 * 60);

243const fn longest(a: u64, b: u64) -> u64 {
244    if a > b { a } else { b }
245}

247/// The longest fixed window any writer of `rate_limit_bucket` counts in, in
248/// seconds. Derived from the writers, not chosen:
249///
250/// - the REST and gRPC shared counters, whose window is
251///   [`axiam_db::rate_limit_counter::DEFAULT_WINDOW_SECS`] (the REST layer's
252///   private `WINDOW_SECS` and the gRPC layer's are each asserted equal to it
253///   by their own tests; the gRPC one is read here as well);
254/// - the CIBA hint cap, [`axiam_oauth2::ciba::HINT_CAP_WINDOW_SECS`].
255///
256/// `rg -n "increment_by" crates/` finds no other writer; a new one with a
257/// longer window must be added to this expression, or the sweep deletes its
258/// rows early (see [`RATE_LIMIT_BUCKET_RETENTION`] for what that costs).
259pub const RATE_LIMIT_BUCKET_LONGEST_WINDOW_SECS: u64 = longest(
260    axiam_oauth2::ciba::HINT_CAP_WINDOW_SECS,
261    longest(
262        axiam_db::rate_limit_counter::DEFAULT_WINDOW_SECS,
263        axiam_api_grpc::middleware::rate_limit::WINDOW_SECS as u64,
264    ),
265);

267/// Slack on top of two windows, covering the clock offset between a writer, a
268/// reader and the datastore: one minute, far beyond what NTP leaves.
269pub const RATE_LIMIT_BUCKET_CLOCK_MARGIN: Duration = Duration::from_secs(60);
270
271/// How long after its last write a `rate_limit_bucket` row is kept: twice
272/// [`RATE_LIMIT_BUCKET_LONGEST_WINDOW_SECS`], plus
273/// [`RATE_LIMIT_BUCKET_CLOCK_MARGIN`].
274///
275/// # Why no live limiter reads a row this old
276///
277/// A bucket row holds the count of one fixed window `[s, s + W)`, `W` its
278/// writer's window. A write lands at some `u >= s`: it is made while the
279/// window is open, and a write-behind flush of the window comes later still.
280/// A limiter reads a stored count in two places only:
281///
282/// 1. the window it belongs to, at a time `t < s + W` (the shared counters'
283///    flush returns it, and the hint cap adds to it);
284/// 2. the hint cap's *previous* window, at `t < s + 2W` (the one later window
285///    that adds it to its own count).
286///
287/// From `t >= s + 2W` on, nothing reads the stored count as a count: an
288/// upsert for any later window finds `window_start` older than its own and
289/// starts from its delta, and a peek finds the same and answers `0` — both
290/// exactly what an absent row gives. So a row is dead once `s + 2W` has
291/// passed, and `s <= u`.
292///
293/// The sweep deletes a row only when `updated_at` (the datastore's clock at
294/// the last write, `u`) is more than `2L + margin` in the past, `L >= W`
295/// being the longest window in use. A reader at `t` (its own clock) after the
296/// sweep has `t >= u + 2L + margin - offset >= s + 2W` as long as the offset
297/// between the clocks is under the margin. No row a limiter still reads is
298/// ever deleted.
299///
300/// If a clock were off by more than the margin, the cost is a count read as
301/// `0` for the rest of one window — the same thing a window boundary does —
302/// never a refusal that should not happen.
303///
304/// The table is bounded by what was written in the last `RETENTION`; the
305/// sweep makes the growth of a table that gained a row per hint, per client
306/// IP and per endpoint, forever, a steady state.
307pub const RATE_LIMIT_BUCKET_RETENTION: Duration = Duration::from_secs(
308    2 * RATE_LIMIT_BUCKET_LONGEST_WINDOW_SECS + RATE_LIMIT_BUCKET_CLOCK_MARGIN.as_secs(),
309);

311/// Rows one `rate_limit_bucket` delete removes at most.
312pub const RATE_LIMIT_BUCKET_SWEEP_BATCH: u32 = 500;

314/// Deletes in one tick: with [`RATE_LIMIT_BUCKET_SWEEP_BATCH`], 50 000 rows,
315/// after which the next tick continues. A tick is never held open by a table
316/// that grew for months before this sweep existed.
317pub const RATE_LIMIT_BUCKET_SWEEP_MAX_BATCHES: u32 = 100;

319/// One pass of the `rate_limit_bucket` job (#638): delete the buckets whose
320/// last write is older than [`RATE_LIMIT_BUCKET_RETENTION`] at `now`, in
321/// batches; returns how many were removed.
322///
323/// Safe for any number of replicas sweeping at once: each delete re-checks
324/// `updated_at` on the row it removes, so a row is removed once, and a row a
325/// limiter has just written is kept. A public free function, for the reason
326/// [`run_erasure_pipeline`] is.
327pub async fn sweep_rate_limit_buckets<C: Connection + Send + Sync + 'static>(
328    repo: &axiam_db::SurrealRateLimitBucketRepository<C>,
329    now: chrono::DateTime<Utc>,
330) -> Result<u64, AxiamError> {
331    let retention = chrono::Duration::from_std(RATE_LIMIT_BUCKET_RETENTION)
332        .map_err(|e| AxiamError::Internal(e.to_string()))?;
333    let cutoff = now - retention;
334    let mut removed = 0u64;
335    for _ in 0..RATE_LIMIT_BUCKET_SWEEP_MAX_BATCHES {
336        let n = repo
337            .delete_stale(cutoff, RATE_LIMIT_BUCKET_SWEEP_BATCH)
338            .await?;
339        removed += n;
340        if n < u64::from(RATE_LIMIT_BUCKET_SWEEP_BATCH) {
341            break;
342        }
343    }
344    Ok(removed)
345}

355/// One pass of the `tenant_purge` job: purge every tombstoned tenant and, when
356/// `scan_orphans`, every orphaned tenant id.
357///
358/// A free function, and public, for the reason [`run_erasure_pipeline`] is
359/// one. Each tenant is purged by
360/// [`SurrealTenantRepository::purge_tombstoned_rows`] — every tenant-scoped
361/// table in the order user erasure uses, of its audit trail the rows at or
362/// before the time of the export the deletion required (T-118, R1W1-04) — and
363/// then [`SurrealTenantRepository::remove_purged_tenant`], its tenant row; or,
364/// for an orphan, under its claim marker, by
365/// [`SurrealTenantRepository::purge_orphan_rows`], which leaves the audit
366/// trail to the retention sweep, and then by the same removal of the marker.
367/// Audit rows the purge keeps name a tenant that no longer exists and are
368/// deleted by the audit retention sweep (T-119) like any other. Each completed purge is recorded in the **system**
369/// log as `tenants.purged`, beside the `tenants.deleted` record the deletion
370/// wrote, with the export time used and the audit rows deleted and kept;
371/// neither record is ever purged.
372///
373/// Every replica runs this sweep (#610). A tombstoned tenant is purged only by
374/// the replica whose [`SurrealTenantRepository::claim_purge`] wins it; the
375/// others skip it for [`TENANT_PURGE_LEASE`]. A tenant the purge finds already
376/// purged — gone before the claim, or removed by an overlapping purge after a
377/// lease ran out — is success, not a failure, and is not recorded again: only
378/// the purge that removed the tenant row writes `tenants.purged`. An orphaned
379/// id is claimed the same way, by a marker row
380/// ([`SurrealTenantRepository::claim_orphan_purge`], R101W1-09 (a)): replicas
381/// whose daily scans coincide purge and record it once, and a scan that
382/// predates another replica's finished purge finds nothing left and records
383/// nothing.
384///
385/// Each tenant's purge, from its claim to its record, is one unit of
386/// [`AuditedWork`] (#655): a task of its own, which aborting the cleanup task
387/// does not cut. The sweep starts no purge once the process is stopping
388/// (checked between tenants); a purge under way gives itself up at the stop as
389/// long as its tenant row is still there, releasing its claim, and once the row
390/// is removed it runs on to its record.
391///
392/// * `Ok(n)` is the number of tenants (and orphaned ids) this call purged.
393/// * One tenant's failure does not stop the others; any failure makes the sweep
394///   fail, naming how many failed and never a tenant's data. The failed tenant
395///   stays tombstoned, its claim is released, and the next pass resumes it
396///   (every step is idempotent).
397/// * A stop ends the pass early with `Ok`: the tenants left stay tombstoned and
398///   unclaimed, and the next start purges them.
399///
400/// # Errors
401///
402/// [`AxiamError::Internal`] as above, or the datastore error that kept the
403/// tombstoned tenants or the orphans from being listed (the tombstoned tenants
404/// are still purged when only the orphans could not be).
405pub async fn sweep_tenant_purge<C, S>(
406    tenant_repo: &Arc<SurrealTenantRepository<C>>,
407    audit: &Arc<S>,
408    scan_orphans: bool,
409    work: &AuditedWork,
410) -> Result<u64, AxiamError>
411where
412    C: Connection,
413    S: AuditWriteSink + 'static,
414{
415    let mut due: Vec<(Uuid, bool)> = tenant_repo
416        .list_tombstoned()
417        .await?
418        .into_iter()
419        .map(|id| (id, false))
420        .collect();
421    // A failed look for orphans does not hold back the tombstoned tenants: it
422    // is reported after them, and the next pass looks again.
423    let mut scan_error = None;
424    if scan_orphans {
425        match tenant_repo.orphaned_tenant_ids().await {
426            Ok(ids) => due.extend(ids.into_iter().map(|id| (id, true))),
427            Err(e) => scan_error = Some(e),
428        }
429    }
430
431    let lease = chrono::TimeDelta::from_std(TENANT_PURGE_LEASE)
432        .map_err(|e| AxiamError::Internal(format!("tenant purge lease: {e}")))?;
433    let (mut purged, mut failed) = (0u64, 0u64);
434    let due_count = due.len();
435    for (started, (tenant_id, orphan)) in due.into_iter().enumerate() {
436        // #655: no purge starts once the process is stopping; the tenants left
437        // stay tombstoned, and the next start purges them.
438        if work.stopping() {
439            tracing::info!(
440                purged,
441                left = due_count - started,
442                "tenant purge sweep stopped: the process is stopping (the rest stay tombstoned)"
443            );
444            break;
445        }
446        let (tenants, sink, unit) = (Arc::clone(tenant_repo), Arc::clone(audit), work.clone());
447        let outcome = if orphan {
448            work.run(purge_orphan_unit(tenants, sink, tenant_id, unit))
449                .await
450        } else {
451            work.run(purge_tombstoned_unit(tenants, sink, tenant_id, lease, unit))
452                .await
453        };
454        match outcome {
455            Some(Ok(PurgeOutcome::Purged)) => purged += 1,
456            // Another replica holds it, or it is already gone.
457            Some(Ok(PurgeOutcome::Skipped)) => {}
458            // Given up at the stop (or still running, and left to the
459            // teardown's join): the loop's check above ends the pass.
460            Some(Ok(PurgeOutcome::Abandoned)) | None if work.stopping() => {}
461            Some(Ok(PurgeOutcome::Abandoned)) => {}
462            // The unit panicked; its claim guard released the claim.
463            None => {
464                failed += 1;
465                tracing::warn!(
466                    %tenant_id,
467                    orphan,
468                    "tenant purge ended without an outcome; the tenant stays tombstoned and the \
469                     next sweep resumes it"
470                );
471            }
472            Some(Err(e)) => {
473                failed += 1;
474                tracing::warn!(
475                    error = %e,
476                    %tenant_id,
477                    orphan,
478                    "tenant purge incomplete; the tenant stays tombstoned and the next sweep resumes it"
479                );
480            }
481        }
482    }
483
484    if failed > 0 {
485        return Err(AxiamError::Internal(format!(
486            "tenant purge was incomplete for {failed} of {} tenant(s)",
487            failed + purged
488        )));
489    }
490    if let Some(e) = scan_error {
491        return Err(e);
492    }
493    Ok(purged)
494}

496/// How one tenant's purge unit ended, when it did not fail.
497enum PurgeOutcome {
498    /// This unit purged the tenant and wrote `tenants.purged`.
499    Purged,
500    /// Another replica holds the claim, or the tenant is already purged (for
501    /// an orphaned id: a scan that predates its finished purge).
502    Skipped,
503    /// Given up because the process is stopping, before the point of no
504    /// return; the tenant stays tombstoned and unclaimed.
505    Abandoned,
506}

508/// One tombstoned tenant's purge, from its claim to its `tenants.purged`
509/// record: one unit of [`AuditedWork`] (#655).
510///
511/// The point of no return is the removal of the tenant row: before it every
512/// step is idempotent and the tenant stays tombstoned, so the unit gives up at
513/// the stop and releases its claim; it never removes the row once the process
514/// is stopping. After it, the record is the only thing left, and the unit
515/// writes it whatever the stop says.
516///
517/// The tombstoned row may be an orphan purge's claim marker whose claimant
518/// died or gave it up (R101W1-09 (a)): its rows are then purged as an orphan's
519/// and the record says `orphan: true` (the report's `orphan`).
520async fn purge_tombstoned_unit<C, S>(
521    tenants: Arc<SurrealTenantRepository<C>>,
522    audit: Arc<S>,
523    tenant_id: Uuid,
524    lease: chrono::TimeDelta,
525    work: AuditedWork,
526) -> Result<PurgeOutcome, AxiamError>
527where
528    C: Connection,
529    S: AuditWriteSink,
530{
531    let claimed_at = Utc::now();
532    if !tenants.claim_purge(tenant_id, claimed_at, lease).await? {
533        return Ok(PurgeOutcome::Skipped);
534    }
535    // From here every exit but a removed tenant row releases the claim — the
536    // awaited release below, or, for an abort or a panic, the guard's drop.
537    let claim = PurgeClaim {
538        tenants: Arc::clone(&tenants),
539        tenant_id,
540        claimed_at,
541        held: true,
542    };
543    let rows = work
544        .until_stopped(async {
545            let report = tenants.purge_tombstoned_rows(tenant_id).await?;
546            work.step(UnitStep::TenantRowsPurged(tenant_id)).await;
547            Ok::<_, AxiamError>(report)
548        })
549        .await;
550    let report = match rows {
551        Some(Ok(report)) => report,
552        // Purged by an overlapping replica: done, and recorded by the purge
553        // that removed the row.
554        Some(Err(AxiamError::NotFound { .. })) => {
555            claim.settle();
556            return Ok(PurgeOutcome::Skipped);
557        }
558        // Let the next tick (any replica's) retry it now rather than after
559        // the lease.
560        Some(Err(e)) => {
561            claim.release().await;
562            return Err(e);
563        }
564        None => {
565            claim.release().await;
566            return Ok(PurgeOutcome::Abandoned);
567        }
568    };
569    // The point of no return is never crossed once the process is stopping.
570    if work.stopping() {
571        claim.release().await;
572        return Ok(PurgeOutcome::Abandoned);
573    }
574    match tenants.remove_purged_tenant(tenant_id).await {
575        Ok(()) => claim.settle(),
576        Err(AxiamError::NotFound { .. }) => {
577            claim.settle();
578            return Ok(PurgeOutcome::Skipped);
579        }
580        Err(e) => {
581            claim.release().await;
582            return Err(e);
583        }
584    }
585    record_tenant_purged(audit.as_ref(), tenant_id, false, &report).await;
586    Ok(PurgeOutcome::Purged)
587}

589/// One orphaned tenant id's purge and its `tenants.purged` record: one unit of
590/// [`AuditedWork`] (#655), claimed by a marker row (R101W1-09 (a)).
591///
592/// [`SurrealTenantRepository::begin_orphan_purge`] claims the id by writing
593/// its marker, a tombstoned tenant row, and finds whether it still has rows:
594/// another replica's claim, or a scan that predates a finished purge, is
595/// skipped and records nothing. Then the unit runs as
596/// [`purge_tombstoned_unit`] does: the rows
597/// ([`SurrealTenantRepository::purge_orphan_rows`]), then the removal of the
598/// marker ([`SurrealTenantRepository::remove_purged_tenant`]) as the point of
599/// no return, and the record only when this unit removed it. Given up at the
600/// stop, or cut, the marker stays tombstoned with its claim released, and the
601/// tombstoned-tenant sweep purges it as the orphan it is.
602async fn purge_orphan_unit<C, S>(
603    tenants: Arc<SurrealTenantRepository<C>>,
604    audit: Arc<S>,
605    tenant_id: Uuid,
606    work: AuditedWork,
607) -> Result<PurgeOutcome, AxiamError>
608where
609    C: Connection,
610    S: AuditWriteSink,
611{
612    let Some(claimed_at) = tenants.begin_orphan_purge(tenant_id).await? else {
613        return Ok(PurgeOutcome::Skipped);
614    };
615    // As in `purge_tombstoned_unit`: every exit but a removed marker releases
616    // the claim, so the marker is the tombstoned-tenant sweep's at once.
617    let claim = PurgeClaim {
618        tenants: Arc::clone(&tenants),
619        tenant_id,
620        claimed_at,
621        held: true,
622    };
623    let report = match work
624        .until_stopped(tenants.purge_orphan_rows(tenant_id, claimed_at))
625        .await
626    {
627        Some(Ok(report)) => report,
628        // The failed purge dropped its marker (R101W1-09 (a)): the next scan
629        // finds the id again.
630        Some(Err(e)) => {
631            claim.settle();
632            return Err(e);
633        }
634        None => {
635            claim.release().await;
636            return Ok(PurgeOutcome::Abandoned);
637        }
638    };
639    // The point of no return is never crossed once the process is stopping.
640    if work.stopping() {
641        claim.release().await;
642        return Ok(PurgeOutcome::Abandoned);
643    }
644    match tenants.remove_purged_tenant(tenant_id).await {
645        Ok(()) => claim.settle(),
646        // The marker was taken over and removed by another purge, which
647        // records it.
648        Err(AxiamError::NotFound { .. }) => {
649            claim.settle();
650            return Ok(PurgeOutcome::Skipped);
651        }
652        Err(e) => {
653            claim.release().await;
654            return Err(e);
655        }
656    }
657    record_tenant_purged(audit.as_ref(), tenant_id, true, &report).await;
658    Ok(PurgeOutcome::Purged)
659}

698/// A replica's claim on a tombstoned tenant's purge (#610), released on every
699/// exit but a removed tenant row (#655).
700///
701/// The purge releases it itself, awaited, where it can — a failure, or giving
702/// up at the stop. Every other exit — the unit aborted at the teardown's bound,
703/// a panic, a release that was itself cut — drops the guard still holding the
704/// claim, and the drop spawns the release on the runtime. Without that, the
705/// next start's sweep found a claim younger than [`TENANT_PURGE_LEASE`] and
706/// skipped the tenant for up to an hour. A drop during the runtime's own
707/// shutdown cannot write anything; the claim then lapses with its lease.
708struct PurgeClaim<C: Connection> {
709    tenants: Arc<SurrealTenantRepository<C>>,
710    tenant_id: Uuid,
711    claimed_at: chrono::DateTime<Utc>,
712    held: bool,
713}
714
715impl<C: Connection> PurgeClaim<C> {
716    /// Release the claim, awaited. A release cut mid-way is retried by the
717    /// drop, since the claim counts as held until it has answered.
718    async fn release(mut self) {
719        if let Err(e) = self
720            .tenants
721            .release_purge_claim(self.tenant_id, self.claimed_at)
722            .await
723        {
724            tracing::warn!(
725                error = %e,
726                tenant_id = %self.tenant_id,
727                "could not release the tenant purge claim; it lapses with its lease"
728            );
729        }
730        self.held = false;
731    }
732
733    /// The tenant row is gone (this purge removed it, or another did): there is
734    /// nothing left to claim.
735    fn settle(mut self) {
736        self.held = false;
737    }
738}
739
740impl<C: Connection> Drop for PurgeClaim<C> {
741    fn drop(&mut self) {
742        if !self.held {
743            return;
744        }
745        let (tenants, tenant_id, claimed_at) =
746            (Arc::clone(&self.tenants), self.tenant_id, self.claimed_at);
747        match tokio::runtime::Handle::try_current() {
748            Ok(runtime) => {
749                runtime.spawn(async move {
750                    if let Err(e) = tenants.release_purge_claim(tenant_id, claimed_at).await {
751                        tracing::warn!(
752                            error = %e,
753                            %tenant_id,
754                            "could not release the claim of a tenant purge that was cut; it \
755                             lapses with its lease"
756                        );
757                    }
758                });
759            }
760            Err(_) => tracing::warn!(
761                %tenant_id,
762                "the claim of a tenant purge that was cut outside a runtime lapses with its lease"
763            ),
764        }
765    }
766}

774#[doc(hidden)]
775#[derive(Debug, Clone, Copy, PartialEq, Eq)]
776pub enum UnitStep {

778    /// its tenant row: before the purge's point of no return.
779    TenantRowsPurged(Uuid),
780    /// An erasure has anonymized the account (and written its proof) and has

786/// A step [`AuditedWork::with_step_for_tests`] runs at each [`UnitStep`].
787#[doc(hidden)]
788pub type UnitStepForTests = Arc<
789    dyn Fn(UnitStep) -> std::pin::Pin<Box<dyn std::future::Future<Output = ()> + Send>>
790        + Send
791        + Sync,

794/// The cleanup task's units of work that must end in an audit row (#655,
795/// #624): a tombstoned tenant's purge through its `tenants.purged` record, an
796/// orphaned tenant id's purge through its record, and one account's erasure
797/// through its `account_deletion` completion and its `gdpr.user_pseudonymized`
798/// row.
799///
800/// The cleanup task runs them one at a time, each as a `tokio::spawn`ed task of
801/// its own ([`Self::run`]), so aborting the cleanup task — the composition
802/// root's bounded join — does not cut one. The root then waits for the unit in
803/// flight with [`Self::join`], within the same bound, and aborts it past that,
804/// so a replica that lost its lease writes nothing after its stop's budget.
805///
806/// Each unit has a **point of no return** — the write after which no later
807/// sweep finds the work again: the removal of the tenant row; the anonymization
808/// of the account. Before it, the unit gives itself up as soon as the process
809/// is stopping (the tenant stays tombstoned, its claim released; the account
810/// stays due), and it never crosses it once the stop is requested. So the only
811/// unit a stop can find past its point of no return crossed it before the stop
812/// was requested, and it has the whole of the teardown's window for its last
813/// statements — from the stop signal through the gRPC stop, the lease release
814/// and the outbound and audit drains to the end of the cleanup bound: 5 to 18 s
815/// for `SIGTERM`, 2 to 13 s after a lost lease, whose stop signal comes at the
816/// loss (#628). One that outlasts even that is cut, and its record is lost
817/// (T-444).
818#[derive(Clone)]
819pub struct AuditedWork {
820    stop: watch::Receiver<bool>,
821    in_flight: Arc<std::sync::Mutex<Option<tokio::task::JoinHandle<()>>>>,
822    step_for_tests: Option<UnitStepForTests>,
823}
824
825impl AuditedWork {
826    /// Units that give themselves up when `stop` reads `true` — the cleanup
827    /// task's shutdown signal. A `stop` whose sender is gone while it reads
828    /// `false` never stops anything.
829    #[must_use]
830    pub fn new(stop: watch::Receiver<bool>) -> Self {
831        Self {
832            stop,
833            in_flight: Arc::default(),
834            step_for_tests: None,
835        }
836    }
837
838    /// **Test seam, never set in production** (#655, #624): run `step` at each
839    /// [`UnitStep`], so a test can hold a unit there across a stop.
840    #[doc(hidden)]
841    #[must_use]
842    pub fn with_step_for_tests(mut self, step: Option<UnitStepForTests>) -> Self {
843        self.step_for_tests = step;
844        self
845    }
846
847    /// Whether the process has asked the cleanup task to stop.
848    #[must_use]
849    pub fn stopping(&self) -> bool {
850        *self.stop.borrow()
851    }
852
853    /// Resolves once the process is stopping; never, when it cannot be.
854    async fn stopped(&self) {
855        let mut stop = self.stop.clone();
856        if stop.wait_for(|stopping| *stopping).await.is_err() {
857            std::future::pending::<()>().await;
858        }
859    }
860
861    /// `work`'s output, or `None` when the process is stopping first: `work`
862    /// is then dropped where it is. For the steps before a unit's point of no
863    /// return only.
864    async fn until_stopped<F: std::future::Future>(&self, work: F) -> Option<F::Output> {
865        tokio::select! {
866            biased;
867            output = work => Some(output),
868            () = self.stopped() => None,
869        }
870    }
871
872    /// The test step at `at`, if one is set.
873    async fn step(&self, at: UnitStep) {
874        if let Some(step) = &self.step_for_tests {
875            step(at).await;
876        }
877    }
878
879    fn slot(&self) -> std::sync::MutexGuard<'_, Option<tokio::task::JoinHandle<()>>> {
880        self.in_flight
881            .lock()
882            .unwrap_or_else(std::sync::PoisonError::into_inner)
883    }
884
885    /// Spawn `unit` as a task of its own and wait for its output.
886    ///
887    /// `None` when the process is stopping — before the unit was started, or
888    /// while it runs: it is then left running for [`Self::join`] — or when the
889    /// unit panicked (logged here). One unit at a time: the caller awaits this
890    /// before it starts the next.
891    pub async fn run<T: Send + 'static>(
892        &self,
893        unit: impl std::future::Future<Output = T> + Send + 'static,
894    ) -> Option<T> {
895        if self.stopping() {
896            return None;
897        }
898        let (tx, rx) = tokio::sync::oneshot::channel();
899        *self.slot() = Some(tokio::spawn(async move {
900            let _ = tx.send(unit.await);
901        }));
902        tokio::select! {
903            biased;
904            output = rx => {
905                let handle = self.slot().take();
906                match output {
907                    Ok(output) => Some(output),
908                    Err(_) => {
909                        if let Some(handle) = handle
910                            && let Err(e) = handle.await
911                        {
912                            tracing::error!(error = %e, "a cleanup unit ended without an outcome");
913                        }
914                        None
915                    }
916                }
917            }
918            // Left to the teardown, which waits for it.
919            () = self.stopped() => None,
920        }
921    }
922
923    /// Wait until `deadline` for the unit in flight, if any; past it, abort the
924    /// unit — it is cut where it is — and say so with `false`. The composition
925    /// root calls this once the cleanup task itself has ended or been aborted
926    /// (#655, #624).
927    pub async fn join(&self, deadline: tokio::time::Instant) -> bool {
928        let Some(mut handle) = self.slot().take() else {
929            return true;
930        };
931        match tokio::time::timeout_at(deadline, &mut handle).await {
932            Ok(Ok(())) => true,
933            Ok(Err(e)) => {
934                tracing::error!(error = %e, "a cleanup unit ended without an outcome");
935                true
936            }
937            Err(_) => {
938                handle.abort();
939                // Once this returns the unit has been dropped: it writes
940                // nothing more, and its claim guard has run.
941                let _ = handle.await;
942                false
943            }
944        }
945    }
946}

953///
954/// A free function, and public, for the reason [`sweep_directories`] is one.

958/// else is a skipped target and costs a conditional write that matches nothing.
959///
960/// * `Ok(n)` is the number of targets whose run this pass made.
961/// * A target whose run could not do all of its work (the broker refused a
962///   reference, the downstream could not be read, AXIAM's datastore failed)

965/// * A shutdown signal stops the pass between targets.

968///
969/// [`AxiamError::Internal`] when the enabled targets could not be listed or a
970/// run was incomplete, as described above.
971pub async fn sweep_scim_reconciliation(

973    shutdown: &watch::Receiver<bool>,
974) -> Result<u64, AxiamError> {
975    let should_stop = || *shutdown.borrow();
976    let sweep = reconciliation.run_due(&should_stop).await.map_err(|_| {
977        AxiamError::Internal("SCIM reconciliation could not list the enabled targets".into())
978    })?;
979    if sweep.failed > 0 {
980        return Err(AxiamError::Internal(format!(
981            "SCIM reconciliation was incomplete for {} of {} target(s) run",
982            sweep.failed, sweep.reconciled
983        )));
984    }
985    Ok(sweep.reconciled)
986}

988/// The `certificate_retention` job (R101W1-05, #618): delete certificate rows
989/// of live tenants more than eight days past their `notAfter`, in bounded
990/// batches, and return how many.
991///
992/// A revoked row is kept for as long as its issuer's revocation list can name
993/// it (the tenant purge's `KEPT_REVOCATION`, whose grace this sweep shares);
994/// past it the CRL is the same without the row. Nothing deleted a certificate
995/// row in a live tenant before, so a revoked one — and the PEM, subject and
996/// `metadata` in it, of an account the Art. 17 purge had erased — outlived any
997/// purpose for the tenant's lifetime. A deleted tenant's rows are the tenant
998/// purge's, not this sweep's. Two replicas may run it at once: each batch is
999/// chosen and deleted by id, and a row the other replica removed first is not
1000/// counted. Touches no audit row.
1001///
1002/// Runs at most [`CERTIFICATE_RETENTION_MAX_BATCHES`] batches of
1003/// [`CERTIFICATE_RETENTION_BATCH`] rows, and stops between batches on a
1004/// shutdown signal; the next tick continues. A free function, and public, for
1005/// the reason [`run_erasure_pipeline`] is.
1006///
1007/// # Errors
1008///
1009/// The repository's error from the batch that failed; batches already deleted
1010/// stay deleted.
1011pub async fn sweep_certificate_retention<CR>(
1012    cert_repo: &CR,
1013    shutdown: &watch::Receiver<bool>,
1014) -> Result<u64, AxiamError>
1015where
1016    CR: axiam_core::repository::CertificateRepository,
1017{
1018    let mut removed = 0_u64;
1019    for _ in 0..CERTIFICATE_RETENTION_MAX_BATCHES {
1020        if *shutdown.borrow() {
1021            break;
1022        }
1023        let batch = cert_repo
1024            .delete_past_retention(CERTIFICATE_RETENTION_BATCH)
1025            .await?;
1026        removed += batch;
1027        if batch < u64::from(CERTIFICATE_RETENTION_BATCH) {
1028            break;
1029        }
1030    }
1031    Ok(removed)
1032}

1038/// One pass of the directory sync job, as a sweep the scheduler can record.
1039///
1040/// A free function, and public, for the reason [`run_erasure_pipeline`] is one:
1041/// the loop that calls it is awkward to construct in a test, and the conversion
1042/// of the job's outcome into job health is the part worth testing directly.
1043///
1044/// * `Ok(n)` is the number of accounts the pass changed (deactivated, updated or
1045///   re-mapped); a pass that found nothing to do is `Ok(0)`.
1046/// * **Any tenant's failure makes the sweep fail** — an unreachable directory, a
1047///   refused search, the safety valve — with a message that names how many
1048///   tenants failed and the fixed tag of each, never a tenant's data. The other
1049///   tenants still ran; one tenant's outage is not another's.
1050/// * A shutdown signal abandons the pass (`Ok(0)`): every step the job takes is
1051///   idempotent or a compare-and-set, and the next process starts it again.
1052///
1053/// # Errors
1054///
1055/// [`AxiamError::Internal`] when the pass failed, as described above.
1056pub async fn sweep_directories<C: Connection + Send + Sync + 'static>(
1057    sync: &DirectorySyncJob<C>,
1058    mut shutdown: watch::Receiver<bool>,
1059) -> Result<u64, AxiamError> {
1060    let outcome = tokio::select! {
1061        // A shutdown request wins over starting work: polled first.
1062        biased;
1063        _ = async {
1064            // Resolves on a shutdown request; a dropped sender never does.
1065            loop {
1066                if *shutdown.borrow() {
1067                    return;
1068                }
1069                if shutdown.changed().await.is_err() {
1070                    std::future::pending::<()>().await;
1071                }
1072            }
1073        } => {
1074            tracing::info!("directory sync abandoned: the process is shutting down");
1075            return Ok(0);
1076        }
1077        outcome = sync.run_due() => outcome,
1078    };
1079    match outcome {
1080        Ok(summary) => match summary.failure_message() {
1081            Some(message) => Err(AxiamError::Internal(message)),
1082            None => Ok(summary.changed()),
1083        },
1084        Err(error) => Err(AxiamError::Internal(format!(
1085            "directory sync could not list the tenants' directories ({})",
1086            error.tag()
1087        ))),
1088    }

1095/// Whether a self-registered client is due to be swept (T21.4).
1096///
1097/// A free function, and public, for the reason [`run_erasure_pipeline`] is
1098/// one: the loop that calls it is awkward to construct in a test, and the
1099/// decision it makes — *delete somebody's client registration* — is the part
1100/// worth testing directly.
1101///
1102/// The clock it reads is `last_authorized_at` when the client has ever been
1103/// authorized and `created_at` when it has not.
1104///
1105/// `DcrSweepWindow::days == 0` is never due. Zero means "never sweep", which
1106/// an operator who prunes out of band may legitimately want, and reading it as
1107/// "sweep everything immediately" would delete a tenant's whole client table
1108/// on the next tick.
1109///
1110/// # The second clock (T21.8 / MCP-05)
1111///
1112/// A row that has **never** been authorized, in a tenant whose effective
1113/// `dynamic_registration` is `anonymous`, is measured against
1114/// [`axiam_core::models::settings::DCR_UNAUTHORIZED_CLIENT_TTL_SECS`] instead
1115/// — an hour. That constant's own documentation says why it is a constant and
1116/// what promoting it to a tenant setting would cost.
1117///
1118/// Those two conditions are the whole of the change, and each is load-bearing.
1119///
1120/// **Never authorized.** The thirty-day window is sized, in
1121/// `DEFAULT_DCR_UNUSED_CLIENT_TTL_DAYS`'s own doc comment, for "a client
1122/// somebody uses monthly". A client registered and never authorized is not
1123/// that client: every MCP client this phase exists to serve authorizes within
1124/// seconds of registering, because registration is the first step of the same
1125/// flow. One TTL served two situations that have nothing in common, which is
1126/// what made `dcr_max_clients` an availability budget a stranger could spend
1127/// for a month. The sweeper could always tell the two apart — the row carries
1128/// `last_authorized_at: None` — so the fix is a second clock, not a bigger
1129/// quota.
1130///
1131/// **`anonymous` only.** In `initial_access_token` mode the row exists because
1132/// an administrator minted a handle for it and somebody redeemed it. There is
1133/// no unauthenticated exposure to bound, and the thirty-day clock is the right
1134/// one: an operator who hands somebody a registration token on Friday should
1135/// not find the registration gone on Monday.
1136///
1137/// Every tenant on `disabled` or `initial_access_token` — which is every
1138/// tenant that did not opt into `anonymous` behind D3 — takes byte for byte
1139/// the path it took before T21.8.
1140pub fn dcr_client_is_due_for_sweep(
1141    client: &axiam_core::models::oauth2_client::OAuth2Client,
1142    ttl: DcrSweepWindow,
1143    now: chrono::DateTime<Utc>,
1144) -> bool {
1145    if client.last_authorized_at.is_none() && ttl.mode == DynamicRegistrationMode::Anonymous {
1146        return now - client.created_at
1147            > chrono::Duration::seconds(i64::from(DCR_UNAUTHORIZED_CLIENT_TTL_SECS));
1148    }
1149    if ttl.days == 0 {
1150        return false;
1151    }
1152    now - dcr_client_last_seen(client) > chrono::Duration::days(i64::from(ttl.days))
1153}

1293/// Delete externally registered clients that have gone unseen for longer than
1294/// their tenant's `dcr_unused_client_ttl_days` (T21.4, widened to `cimd` by
1295/// T21.8 / MCP-04).
1296///
1297/// One `managed_by` per call, so `/health/jobs` can distinguish the two
1298/// sweeps: they delete different things for different reasons and an operator
1299/// debugging one should not have to read the other's counter.
1300///
1301/// # What it will not touch
1302///
1303/// Never `admin`. An administrator's client is never swept, however long it
1304/// sits unused, because somebody decided it should exist and nothing here is
1305/// entitled to reverse that. The repository query filters on the value rather
1306/// than on "not admin", so a fourth provenance added later is opted **in** by
1307/// somebody writing it down.
1308///
1309/// # Why `cimd` is swept now, when T21.4 argued it should not be
1310///
1311/// T21.4's reasoning, which this function used to carry: a CIMD shadow row is
1312/// a cache of a document the client publishes, so deleting it would be
1313/// re-materialised on the next request and the TTL would mean nothing.
1314///
1315/// That is right about TTL semantics and says nothing about storage, which is
1316/// what MCP-04 is about: a cache that is never evicted is not a cache. An
1317/// inert row is still a row — listed on the OAuth2 clients page, counted in
1318/// every `list_all_by_managed_by` this function runs, and a permanent write a
1319/// stranger made at the cost of one unauthenticated request. And eviction on
1320/// last-seen is *consistent* with the old argument rather than against it: a
1321/// row deleted while its document is still published is re-materialised on the
1322/// next request, which is exactly what a cache should do. What it must not do
1323/// is delete a row that is still in use, which is what
1324/// [`cimd_client_last_seen`] is for — every resolve moves `updated_at`, so a
1325/// document presented once a day is never due under a 30-day TTL.
1326///
1327/// # Per-tenant TTL, resolved per row
1328///
1329/// The window is a tenant setting, so each row's tenant is resolved and its
1330/// effective settings read. Deployment-wide sweeps elsewhere in this file
1331/// (audit retention) deliberately do **not** work this way, and the difference
1332/// is who owns the decision: audit retention is a property of the datastore an
1333/// operator is responsible for, where this is a property of the tenant's
1334/// relationship with the clients it lets register.
1335///
1336/// The settings read is cached per tenant for the duration of one sweep —
1337/// `dcr_max_clients` bounds the rows per tenant, so a hundred clients in one
1338/// tenant is one read rather than a hundred.
1339///
1340/// A tenant whose settings cannot be read is skipped: the fail-closed
1341/// direction for a sweep that **deletes** is to delete nothing.
1342pub async fn sweep_unused_external_clients<CR, TR, SR, RT, AC, PR, DG, CB>(
1343    client_repo: &CR,
1344    tenant_repo: &TR,
1345    settings_repo: &SR,
1346    grants: &ClientGrantStores<RT, AC, PR, DG, CB>,
1347    managed_by: axiam_core::models::oauth2_client::ManagedBy,
1348    now: chrono::DateTime<Utc>,
1349) -> Result<u64, AxiamError>
1350where
1351    CR: axiam_core::repository::OAuth2ClientRepository,
1352    TR: TenantRepository,
1353    SR: axiam_core::repository::SettingsRepository,
1354    RT: axiam_core::repository::RefreshTokenRepository,
1355    AC: axiam_core::repository::AuthorizationCodeRepository,
1356    PR: axiam_core::repository::PushedAuthRequestRepository,
1357    DG: axiam_core::repository::DeviceGrantRepository,
1358    CB: axiam_core::repository::CibaRequestRepository,
1359{
1360    use axiam_core::models::oauth2_client::ManagedBy;
1361    use axiam_core::repository::SettingsRepository;
1362
1363    type DuePredicate = fn(
1364        &axiam_core::models::oauth2_client::OAuth2Client,
1365        DcrSweepWindow,
1366        chrono::DateTime<Utc>,
1367    ) -> bool;
1368    let (job, is_due): (&str, DuePredicate) = match managed_by {
1369        ManagedBy::Dcr => ("dcr_unused_clients", dcr_client_is_due_for_sweep),
1370        ManagedBy::Cimd => ("cimd_unused_clients", cimd_client_is_due_for_sweep),
1371        // An administrator's client is never swept. Answered here rather than
1372        // by trusting every caller, because the argument for it is a security
1373        // one and belongs next to the query.
1374        ManagedBy::Admin => return Ok(0),
1375    };
1376
1377    let clients = client_repo.list_all_by_managed_by(managed_by).await?;
1378    if clients.is_empty() {
1379        return Ok(0);
1380    }
1381
1382    let mut ttl_by_tenant: std::collections::HashMap<Uuid, Option<DcrSweepWindow>> =
1383        std::collections::HashMap::new();
1384    let mut removed = 0u64;
1385
1386    for client in clients {
1387        let ttl = match ttl_by_tenant.get(&client.tenant_id) {
1388            Some(cached) => *cached,
1389            None => {
1390                let resolved = match tenant_repo.get_by_id(client.tenant_id).await {
1391                    Ok(tenant) => SettingsRepository::get_effective_settings(
1392                        settings_repo,
1393                        tenant.organization_id,
1394                        client.tenant_id,
1395                    )
1396                    .await
1397                    .ok()
1398                    .map(|s| DcrSweepWindow {
1399                        days: s.oidc.dcr_unused_client_ttl_days,
1400                        mode: s.oidc.dynamic_registration,
1401                    }),
1402                    Err(_) => None,
1403                };
1404                ttl_by_tenant.insert(client.tenant_id, resolved);
1405                resolved
1406            }
1407        };
1408        // `None` is an unreadable tenant or an unreadable settings row.
1409        let Some(ttl) = ttl else {
1410            continue;
1411        };
1412        if !is_due(&client, ttl, now) {
1413            continue;
1414        }
1415
1416        // #517 — the client's grants go before the row, as on an
1417        // administrator's delete: a `cimd` row re-materialises under the same
1418        // `client_id`, and a refresh token, code or pushed request left behind
1419        // would work again against it. A revocation that fails keeps the row
1420        // for the next sweep rather than leave a row-less client with live
1421        // grants.
1422        if let Err(e) = grants.revoke(client.tenant_id, &client.client_id).await {
1423            tracing::warn!(
1424                job,
1425                error = %e,
1426                client_id = %client.client_id,
1427                "could not revoke an unused externally registered client's grants; not \
1428                 deleting it this sweep"
1429            );
1430            continue;
1431        }
1432        if let Err(e) = client_repo.delete(client.tenant_id, client.id).await {
1433            // One failure must not stop the sweep: the next row may be a
1434            // different tenant entirely.
1435            tracing::warn!(
1436                job,
1437                error = %e,
1438                client_id = %client.client_id,
1439                "could not delete an unused externally registered client"
1440            );
1441            continue;
1442        }
1443        // A request that read the row before the delete may have written a
1444        // grant between the two writes above; one more pass leaves none.
1445        if let Err(e) = grants.revoke(client.tenant_id, &client.client_id).await {
1446            tracing::warn!(
1447                job,
1448                error = %e,
1449                client_id = %client.client_id,
1450                "a swept client's grants could not be revoked a second time"
1451            );
1452        }
1453        removed += 1;
1454        tracing::info!(
1455            job,
1456            tenant_id = %client.tenant_id,
1457            client_id = %client.client_id,
1458            ttl_days = ttl.days,
1459            ever_authorized = client.last_authorized_at.is_some(),
1460            mode = %ttl.mode,
1461            "deleted an externally registered client that has gone unseen within its \
1462             tenant's TTL"
1463        );
1464    }
1465
1466    Ok(removed)
1467}

1595/// account gone — whether or not the erasure proof is then written. A proof that
1596/// fails to write leaves an anonymized account that no sweep selects again, so a
1597/// report that waited for the proof would never be sent.
1598///
1599/// With `work` (#624), the steps give up before the anonymization — the point of
1600/// no return — once the process is stopping, and answer `Ok(false)`: the audit
1601/// scrub is idempotent and the account stays due. `Ok(true)` once the account
1602/// is anonymized and its proof written.
1603#[allow(clippy::too_many_arguments)]
1604async fn erasure_steps<A, EP, U>(
1605    audit_repo: &A,
1606    erasure_proof_repo: &EP,
1607    user_repo: &U,
1608    tenant_id: Uuid,
1609    user_id: Uuid,
1610    pseudonym: &str,
1611    email_hash: &str,
1612    report: Option<(&dyn SsfSystemAccountSink, axiam_core::models::user::User)>,
1613    work: Option<&AuditedWork>,
1614) -> Result<bool, AxiamError>
1615where
1616    A: AuditLogRepository,
1617    EP: ErasureProofRepository,
1618    U: UserRepository,
1619{
1620    // FATAL now (was: `if let Err(e) = ... { tracing::warn!(...) }`).
1621    let scrub = audit_repo.pseudonymize_actor(tenant_id, user_id, pseudonym);
1622    match work {
1623        Some(work) => match work.until_stopped(scrub).await {
1624            Some(scrubbed) => {
1625                scrubbed?;
1626            }
1627            None => return Ok(false),
1628        },
1629        None => {
1630            scrub.await?;
1631        }
1632    }
1633    // #624: the anonymization is never started once the process is stopping.
1634    if work.is_some_and(AuditedWork::stopping) {
1635        return Ok(false);
1636    }
1637
1638    // Clears `deletion_pending` — the re-selection anchor. Must run before
1639    // the proof so a failure here (unreachable in practice since this is
1640    // now the second of three fallible steps) still leaves the user due.
1641    user_repo
1642        .anonymize_user(tenant_id, user_id, email_hash, pseudonym)
1643        .await?;
1644
1645    // G-5 (D-52): the account is gone; tell the receivers what it was.
1646    if let Some((sink, user)) = report {
1647        sink.account_purged(tenant_id, &user).await;
1648    }
1649
1650    // Written STRICTLY LAST — only reached once every step above succeeded.
1651    erasure_proof_repo
1652        .create(CreateErasureProof {
1653            pseudonym: pseudonym.to_string(),
1654            tenant_id,
1655            user_id,
1656            erased_at: Utc::now(),
1657        })
1658        .await?;
1659
1660    Ok(true)
1661}

1754    /// The account.

1757    pub pseudonym: &'a str,
1758    /// What its address is overwritten with.

1765/// An erasure through its point of no return to its record (#624): the erasure
1766/// pipeline ([`run_erasure_pipeline_reporting`]: audit scrub, anonymization,
1767/// proof), then the `account_deletion` row marked completed, then the
1768/// `gdpr.user_pseudonymized` audit row. Run inside a unit of [`AuditedWork`],
1769/// so a cut of the cleanup task does not reach it.
1770///
1771/// The anonymization is the point of no return: it clears `deletion_pending`,
1772/// so no later sweep selects the account again, and nothing but this call
1773/// writes the completion and the audit row. Until then — the audit scrub —
1774/// the call gives up once `work` is stopping and answers `Ok(false)`: the
1775/// account stays due. Past it, it runs on to the audit row whatever the stop
1776/// says, and answers `Ok(true)`. A free function, and public, for the reason
1777/// [`run_erasure_pipeline`] is one.
1778///
1779/// # Errors
1780///
1781/// Those of [`run_erasure_pipeline_reporting`], before anything is recorded.
1782/// The completion and the audit row are not fallible here: a failed completion
1783/// is logged, and a failed audit write is dead-lettered (SECHRD-12).
1784pub async fn erase_and_record<A, EP, U, C>(
1785    audit_repo: &A,
1786    erasure_proof_repo: &EP,
1787    user_repo: &U,
1788    account_deletion_repo: &SurrealAccountDeletionRepository<C>,
1789    sink: Option<&dyn SsfSystemAccountSink>,
1790    subject: ErasureSubject<'_>,
1791    work: &AuditedWork,
1792) -> Result<bool, AxiamError>
1793where
1794    A: AuditLogRepository + AuditWriteSink,
1795    EP: ErasureProofRepository,
1796    U: UserRepository,
1797    C: Connection,
1798{
1799    let ErasureSubject {
1800        tenant_id,
1801        user_id,
1802        pseudonym,
1803        email_hash,
1804        certificates_revoked,
1805    } = subject;
1806    // (d)/(e)/(g) Run the erasure pipeline: pseudonymize audit actor refs
1807    // (now FATAL, was swallowed) -> anonymize the user row -> write the
1808    // erasure proof STRICTLY LAST (SECHRD-06 / D-03a, T-25-13). A failure
1809    // anywhere in this call propagates with `?` and aborts the purge:
1810    // `anonymize_user` (the only step that clears `deletion_pending`)
1811    // never ran, so the user stays due for a retry via
1812    // `find_due_for_purge`, and NO erasure proof is ever written for an
1813    // incomplete erasure.
1814    if !erasure_pipeline(
1815        audit_repo,
1816        erasure_proof_repo,
1817        user_repo,
1818        tenant_id,
1819        user_id,
1820        pseudonym,
1821        email_hash,
1822        sink,
1823        Some(work),
1824    )
1825    .await?
1826    {
1827        return Ok(false);
1828    }
1829    work.step(UnitStep::ErasureAnonymized(user_id)).await;
1830
1831    // (f) Mark the account_deletion row as completed (lookup by user_id).
1832    // Run AFTER the erasure pipeline succeeds — if the pipeline failed
1833    // above, this is never reached and the deletion row stays pending
1834    // for the next retry too.
1835    match account_deletion_repo
1836        .find_pending_by_user_id(tenant_id, user_id)
1837        .await
1838    {
1839        Ok(Some(deletion)) => {
1840            if let Err(e) = account_deletion_repo
1841                .mark_completed(tenant_id, deletion.id)
1842                .await
1843            {
1844                tracing::warn!(
1845                    error = %e,
1846                    %tenant_id,
1847                    deletion_id = %deletion.id,
1848                    "cleanup: failed to mark account_deletion completed"
1849                );
1850            }
1851        }
1852        Ok(None) => {
1853            tracing::debug!(user_id = %user_id, "no pending account_deletion row found at purge time");
1854        }
1855        Err(e) => {
1856            tracing::warn!(error = ?e, user_id = %user_id, "failed to find account_deletion row");
1857        }
1858    }
1859
1860    // (h) Emit gdpr.user_pseudonymized audit event. actor_id is the
1861    // erased subject's own (now-anonymized) row id — a real, resolvable
1862    // identifier rather than a fabricated nil UUID (Task 1 fix); the row
1863    // still exists (anonymize-in-place preserves referential integrity),
1864    // so this is a legitimate audit-trail reference, not PII leakage.
1865    // A DB-write failure here is dead-lettered to BOTH an append-only
1866    // file AND a structured audit event (SECHRD-12 / D-02, T-24-61) —
1867    // this legally-significant record must never be silently lost.
1868    write_audit_with_dead_letter(
1869        audit_repo,
1870        CreateAuditLogEntry {
1871            tenant_id,
1872            actor_id: user_id,
1873            actor_type: ActorType::System,
1874            action: "gdpr.user_pseudonymized".into(),
1875            resource_id: None,
1876            outcome: AuditOutcome::Success,
1877            ip_address: None,
1878            metadata: Some(serde_json::json!({
1879                "pseudonym": pseudonym,
1880                // #533: the account's certificates this purge revoked; a
1881                // count, as directory linking records its own.
1882                "certificates_revoked": certificates_revoked,
1883            })),
1884        },
1885    )
1886    .await;
1887
1888    tracing::info!(
1889        pseudonym = %pseudonym,
1890        tenant_id = %tenant_id,
1891        "user purged and pseudonymized"
1892    );
1893    Ok(true)
1894}

1906    audit_repo: Arc<SurrealAuditLogRepository<C>>,
1907    erasure_proof_repo: Arc<SurrealErasureProofRepository<C>>,
1908    user_repo: Arc<SurrealUserRepository<C>>,

1912
1913impl<C: Connection> ErasureUnit<C> {
1914    /// The purge of `user` (steps listed at `CleanupTask::sweep_pending_purges`):
1915    /// `Ok(true)` once erased and recorded, `Ok(false)` when given up at the
1916    /// stop before the point of no return — the account stays due — or when the
1917    /// account is no longer due at the claim (its deletion was cancelled).
1918    async fn purge(
1919        self,
1920        user: axiam_core::models::user::User,
1921        pepper: [u8; 32],
1922        now: chrono::DateTime<Utc>,
1923        work: AuditedWork,
1924    ) -> Result<bool, AxiamError> {
1925        // Item 11 of the 1.1.0 atomicity audit: the erasure is claimed before
1926        // its first destructive step, in a transaction that re-checks the
1927        // account is still due at the sweep's `now`. A deletion cancelled since
1928        // the sweep's read is skipped here, before anything is revoked; once
1929        // claimed, a cancel is refused (`409`), so the subject is never told
1930        // "cancelled" for an account this purge goes on to erase.
1931        if !self
1932            .account_deletion_repo
1933            .claim_for_purge(user.tenant_id, user.id, now)
1934            .await?
1935        {
1936            tracing::info!(
1937                user_id = %user.id,
1938                tenant_id = %user.tenant_id,
1939                "purge skipped: the account's deletion was cancelled since the sweep read it"
1940            );
1941            return Ok(false);
1942        }
1943
1944        // Steps (a)…(b4) are idempotent and leave the account due: given up
1945        // where they are once the process is stopping.
1946        let Some(certificates_revoked) = work.until_stopped(self.prepare(&user)).await else {
1947            return Ok(false);
1948        };
1949        let certificates_revoked = certificates_revoked?;
1950        if work.stopping() {
1951            return Ok(false);
1952        }
1953
1954        let (user_id, tenant_id) = (user.id, user.tenant_id);
1955        // (c) Compute deterministic pseudonym (keyed HMAC-SHA256, D-02).
1956        let pseudonym = gdpr_pseudonym(&pepper, tenant_id, user_id);
1957
1958        // Derive email hash for anonymize_user from tenant+user IDs (original email
1959        // is no longer accessible at purge time — user already marked deletion_pending).
1960        use sha2::{Digest, Sha256};
1961        let mut h = Sha256::new();
1962        h.update(tenant_id.as_bytes());
1963        h.update(user_id.as_bytes());
1964        let email_hash = hex::encode(h.finalize());
1965
1966        // The anonymization inside is the point of no return; it is never
1967        // started once the process is stopping (#624).
1968        erase_and_record(
1969            self.audit_repo.as_ref(),
1970            self.erasure_proof_repo.as_ref(),
1971            self.user_repo.as_ref(),
1972            self.account_deletion_repo.as_ref(),
1973            self.ssf_sink.as_deref(),
1974            ErasureSubject {
1975                tenant_id,
1976                user_id,
1977                pseudonym: &pseudonym,
1978                email_hash: &email_hash,
1979                certificates_revoked,
1980            },
1981            &work,
1982        )
1983        .await
1984    }
1985
1986    /// Steps (a)…(b4): sessions, federation links, credentials, the
1987    /// authorization graph and the account's certificates. Returns how many
1988    /// certificates were revoked.
1989    async fn prepare(&self, user: &axiam_core::models::user::User) -> Result<u64, AxiamError> {
1990        let (user_id, tenant_id) = (user.id, user.tenant_id);
1991        // (a) Revoke all sessions and OAuth2 refresh tokens.
1992        self.auth_svc
1993            .revoke_all_sessions(tenant_id, user_id)
1994            .await?;
1995
1996        // (b) Hard-delete federation identity links.
1997        match self
1998            .federation_link_repo
1999            .get_by_user_id(tenant_id, user_id)
2000            .await
2001        {
2002            Ok(links) => {
2003                for link in links {
2004                    if let Err(e) = self.federation_link_repo.delete(tenant_id, link.id).await {
2005                        tracing::warn!(
2006                            error = %e,
2007                            %tenant_id,
2008                            link_id = %link.id,
2009                            "cleanup: failed to delete expired federation link; will retry next cycle"
2010                        );
2011                    }
2012                }
2013            }
2014            Err(e) => {
2015                tracing::warn!(error = ?e, user_id = %user_id, "failed to list federation links for purge");
2016            }
2017        }
2018
2019        // (b2) Hard-delete stored credential material (SEC-056). Done with `?`
2020        // (fail-closed): erasure must not be certified while a user's passkeys or
2021        // password hashes remain. A transient failure aborts the purge and it is
2022        // retried next sweep — every step here is idempotent.
2023        let webauthn_creds = self.webauthn_repo.list_by_user(tenant_id, user_id).await?;
2024        for cred in webauthn_creds {
2025            self.webauthn_repo.delete(tenant_id, cred.id).await?;
2026        }
2027        // Prune the entire password-history chain (keep_count = 0 deletes all).
2028        self.password_history_repo
2029            .prune(tenant_id, user_id, 0)
2030            .await?;
2031
2032        // (b3) Prune authorization-graph edges so the anonymized row is left as
2033        // an isolated tombstone — no dangling `member_of` / `has_role` relations
2034        // skewing group-member queries or leaving live authorization paths.
2035        // Remove group memberships first so the subsequent role pass only sees
2036        // the user's *direct* has_role edges (inherited ones vanish with the
2037        // membership). Fail-closed (`?`): the tombstone must not be created while
2038        // graph edges survive; every delete is idempotent and retried on failure.
2039        let groups = self.group_repo.get_user_groups(tenant_id, user_id).await?;
2040        for group in groups {
2041            self.group_repo
2042                .remove_member(tenant_id, user_id, group.id)
2043                .await?;
2044        }
2045        let assignments = self
2046            .role_repo
2047            .get_user_role_assignments(tenant_id, user_id)
2048            .await?;
2049        for a in assignments {
2050            self.role_repo
2051                .unassign_from_user(tenant_id, user_id, a.role.id, a.resource_id)
2052                .await?;
2053        }
2054
2055        // (b4) Revoke the account's `User` certificates (#533) and clear their
2056        // `metadata` (R101W1-05), while its username and email — which the
2057        // convention for a certificate issued without a `user_id` reads — are
2058        // still on the row. Fail-closed, like the steps above; idempotent (an
2059        // already revoked row is not matched, a cleared one is not rewritten).
2060        revoke_erased_users_certificates(self.cert_repo.as_ref(), user).await
2061    }
2062}

2305    /// Run the cleanup loop until a shutdown signal is received.
2306    ///
2307    /// Never returns `Err` — all sweep errors are logged at `warn` level and
2308    /// the loop continues (T-04-36).
2309    pub async fn run(mut self) -> Result<(), AxiamError> {
2310        let mut ticker = tokio::time::interval(self.interval);
2311        // Skip ticks that were missed while the sweep was running to prevent
2312        // catch-up storms after a pause (T-04-35).
2313        ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
2314
2315        loop {
2316            // A stop that arrived during a tick ends the loop here, whatever the
2317            // ticker says: the select below may otherwise pick a tick that is
2318            // already due over the shutdown change.
2319            if self.stop_requested() {
2320                tracing::info!("cleanup task received shutdown signal");
2321                return Ok(());
2322            }
2323            tokio::select! {
2324                _ = ticker.tick() => {
2325                    self.tick().await;
2326                }
2327                changed = self.shutdown.changed() => {
2328                    if changed.is_ok() && *self.shutdown.borrow() {
2329                        tracing::info!("cleanup task received shutdown signal");
2330                        return Ok(());
2331                    }
2332                }
2333            }
2334        }
2335    }
