1//! Periodic cleanup task for expired federation rows, GDPR purges, and export jobs.
2//!
3//! SurrealDB v3 does not support native TTL on rows (RESEARCH §7), so this task
4//! periodically sweeps various tables:
5//! - `saml_assertion_replay` and `federation_login_state`: expired rows
6//! - `user`: accounts past their scheduled purge date (D-05/D-06/D-08)
7//! - `export_job`: queued jobs waiting to have their encrypted blob generated (D-12)
8//! - every tenant-scoped table: the rows of a deleted (tombstoned) tenant, then
9//!   its tenant row (#523, D-4)
10//!
11//! The task shuts down cleanly when the caller sends `true` through the watch
12//! channel (D-09, D-24).

219/// How often the `tenant_purge` sweep also looks for orphaned tenant ids.
220///
221/// The look reads every tenant-scoped table (one `GROUP BY` each), which is not
222/// something to do every minute for a residue that only an upgrade can have: a
223/// deletion made by this version tombstones, so it never leaves an orphan. Once
224/// at process start, which is when an upgraded deployment first runs this
225/// code, and once a day after that.
226pub const ORPHAN_TENANT_SCAN_INTERVAL: Duration = Duration::from_secs(24 * 60 * 60);
227
228/// How long a replica's claim on a tombstoned tenant's purge keeps the other
229/// replicas off it (#610); a claim older than this is taken over.
230///
231/// The purge is not batched: it is one `DELETE … WHERE tenant_id = $id` per
232/// tenant-scoped table, about seventy statements in sequence, each removing
233/// all of the tenant's rows of its table at once (the certificate tables three
234/// statements each), so even a large tenant's purge is expected to take
235/// minutes; an hour leaves an order of magnitude of headroom before a purge
236/// still running could be overlapped — and an overlap is safe in any case: the
237/// steps are idempotent and only one purge's final `DELETE` finds the tenant
238/// row, so the purge is recorded once. What the lease costs is the delay
239/// after a replica dies holding a claim: the tenant waits at most an hour for
240/// another replica, an erasure of data already out of every read path. Clock
241/// skew between replicas, which the stamp is written by, is seconds against an
242/// hour. A purge that does not finish releases its claim — one that fails, one
243/// given up because the process is stopping, and (#655) one aborted or
244/// panicking, from a drop guard — so the next tick, or the next start, resumes
245/// it at once; only a process that ends before the release is written leaves
246/// the tenant to the lease: a `SIGKILL`, or a unit cut at the stop's bound,
247/// whose guard releases the claim from a spawned task that the process exit may
248/// cancel (#687).
249pub const TENANT_PURGE_LEASE: Duration = Duration::from_secs(60 * 60);
250
251/// How long a unit's claim on its pending-record marker (#689) keeps a sweep
252/// from recording the work itself: the tenant purge's lease, for the same
253/// reasons. A unit holds the claim for the handful of statements between its
254/// point of no return and its record, which the teardown bounds at seconds; a
255/// unit that is cut releases it from a drop guard, so the next start records
256/// the work at once, and only a process that ends before the release is
257/// written — a `SIGKILL`, or a unit cut at the stop's bound, whose spawned
258/// release the process exit may cancel (#687) — leaves the marker to this lease.
259pub const PENDING_RECORD_LEASE: Duration = TENANT_PURGE_LEASE;
260
261/// How long a unit's claim on an account's erasure (#738) keeps the other
262/// replicas' sweeps off the account: the tenant purge's lease, for the same
263/// reasons. Every replica runs the GDPR sweep, and two that read one due
264/// account used to run its erasure twice — the record stayed single, but the
265/// SSF `account_purged` went out twice and the later unit failed its tick on
266/// the proof. Now the first claim is exclusive: a second claimant skips the
267/// account without an error. A unit holds the claim from before its first
268/// destructive step to the anonymization — its point of no return, which
269/// clears `deletion_pending`, so no sweep selects the account again — and
270/// releases it on every exit before (a failure, a stop, a cut or a panic, from
271/// a drop guard), so the next sweep takes the account at once. Only a process
272/// that ends before the release is written — a `SIGKILL`, or a unit cut at the
273/// stop's bound, whose spawned release the process exit may cancel (#687) —
274/// leaves the account to this lease.
275pub const ERASURE_CLAIM_LEASE: Duration = TENANT_PURGE_LEASE;

281const fn longest(a: u64, b: u64) -> u64 {
282    if a > b { a } else { b }
283}

285/// The longest fixed window any writer of `rate_limit_bucket` counts in, in
286/// seconds. Derived from the writers, not chosen:
287///
288/// - the REST and gRPC shared counters, whose window is
289///   [`axiam_db::rate_limit_counter::DEFAULT_WINDOW_SECS`] (the REST layer's
290///   private `WINDOW_SECS` and the gRPC layer's are each asserted equal to it
291///   by their own tests; the gRPC one is read here as well);
292/// - the CIBA hint cap, [`axiam_oauth2::ciba::HINT_CAP_WINDOW_SECS`].
293///
294/// `rg -n "increment_by" crates/` finds no other writer; a new one with a
295/// longer window must be added to this expression, or the sweep deletes its
296/// rows early (see [`RATE_LIMIT_BUCKET_RETENTION`] for what that costs).
297pub const RATE_LIMIT_BUCKET_LONGEST_WINDOW_SECS: u64 = longest(
298    axiam_oauth2::ciba::HINT_CAP_WINDOW_SECS,
299    longest(
300        axiam_db::rate_limit_counter::DEFAULT_WINDOW_SECS,
301        axiam_api_grpc::middleware::rate_limit::WINDOW_SECS as u64,
302    ),
303);

305/// Slack on top of two windows, covering the clock offset between a writer, a
306/// reader and the datastore: one minute, far beyond what NTP leaves.
307pub const RATE_LIMIT_BUCKET_CLOCK_MARGIN: Duration = Duration::from_secs(60);
308
309/// How long after its last write a `rate_limit_bucket` row is kept: twice
310/// [`RATE_LIMIT_BUCKET_LONGEST_WINDOW_SECS`], plus
311/// [`RATE_LIMIT_BUCKET_CLOCK_MARGIN`].
312///
313/// # Why no live limiter reads a row this old
314///
315/// A bucket row holds the count of one fixed window `[s, s + W)`, `W` its
316/// writer's window. A write lands at some `u >= s`: it is made while the
317/// window is open, and a write-behind flush of the window comes later still.
318/// A limiter reads a stored count in two places only:
319///
320/// 1. the window it belongs to, at a time `t < s + W` (the shared counters'
321///    flush returns it, and the hint cap adds to it);
322/// 2. the hint cap's *previous* window, at `t < s + 2W` (the one later window
323///    that adds it to its own count).
324///
325/// From `t >= s + 2W` on, nothing reads the stored count as a count: an
326/// upsert for any later window finds `window_start` older than its own and
327/// starts from its delta, and a peek finds the same and answers `0` — both
328/// exactly what an absent row gives. So a row is dead once `s + 2W` has
329/// passed, and `s <= u`.
330///
331/// The sweep deletes a row only when `updated_at` (the datastore's clock at
332/// the last write, `u`) is more than `2L + margin` in the past of the
333/// *datastore's* clock (`time::now()`, evaluated in the selection and in the
334/// `DELETE`, #676), `L >= W` being the longest window in use. The sweeping
335/// replica's own clock takes no part in the comparison, so its offset from the
336/// datastore is not an input. A reader at `t` (its own clock) after the sweep
337/// has `t >= u + 2L + margin - offset >= s + 2W` as long as the offset between
338/// the *reader's* clock and the datastore's is under the margin. No row a
339/// limiter still reads is ever deleted.
340///
341/// If a reader's clock were off by more than the margin, the cost is a count
342/// read as `0` for the rest of one window — the same thing a window boundary
343/// does. For the hint cap that is a stretch that may admit up to twice the
344/// allowance, so its "exactly" holds under the assumption that readers and
345/// the datastore agree to within the margin (T-424).
346///
347/// "The datastore's clock" is one clock for the shipped single-node SurrealDB
348/// (`k8s/surrealdb/statefulset.yml`, `replicas: 1`). A clustered SurrealDB
349/// evaluates `time::now()` on whichever compute node runs the statement, so the
350/// margin must also cover the skew between SurrealDB nodes.
351///
352/// The table is bounded by what was written in the last `RETENTION`; the
353/// sweep makes the growth of a table that gained a row per hint, per client
354/// IP and per endpoint, forever, a steady state.
355pub const RATE_LIMIT_BUCKET_RETENTION: Duration = Duration::from_secs(
356    2 * RATE_LIMIT_BUCKET_LONGEST_WINDOW_SECS + RATE_LIMIT_BUCKET_CLOCK_MARGIN.as_secs(),
357);

359/// Rows one `rate_limit_bucket` delete removes at most.
360pub const RATE_LIMIT_BUCKET_SWEEP_BATCH: u32 = 500;

362/// Deletes in one tick: with [`RATE_LIMIT_BUCKET_SWEEP_BATCH`], 50 000 rows,
363/// after which the next tick continues. A tick is never held open by a table
364/// that grew for months before this sweep existed.
365pub const RATE_LIMIT_BUCKET_SWEEP_MAX_BATCHES: u32 = 100;
366
367/// One pass of the `rate_limit_bucket` job (#638): delete the buckets whose
368/// last write is older than [`RATE_LIMIT_BUCKET_RETENTION`] on the datastore's
369/// clock, in batches; returns how many were removed.
370///
371/// Safe for any number of replicas sweeping at once: each delete re-checks
372/// `updated_at` on the row it removes, so a row is removed once, and a row a
373/// limiter has just written is kept. A public free function, for the reason
374/// [`run_erasure_pipeline`] is.
375pub async fn sweep_rate_limit_buckets<C: Connection + Send + Sync + 'static>(
376    repo: &axiam_db::SurrealRateLimitBucketRepository<C>,
377) -> Result<u64, AxiamError> {
378    let mut removed = 0u64;
379    for _ in 0..RATE_LIMIT_BUCKET_SWEEP_MAX_BATCHES {
380        let n = repo
381            .delete_stale(RATE_LIMIT_BUCKET_RETENTION, RATE_LIMIT_BUCKET_SWEEP_BATCH)
382            .await?;
383        removed += n;
384        if n < u64::from(RATE_LIMIT_BUCKET_SWEEP_BATCH) {
385            break;
386        }
387    }
388    Ok(removed)
389}

391/// Erasure step (a), for the OAuth2 refresh tokens (#753): revoke every row
392/// of the user, then delete them.
393///
394/// The rows carry `user_id`, `client_id`, the scopes and, since v109, the
395/// device key's `dpop_jkt`; revoking left them in the table until the tenant

398/// idempotent. An erasure cut before its point of no return re-runs it; one cut

414}

426/// Step (b3)'s role pass (#758 (e)): unassign every direct role assignment of
427/// the erased account, and write one `network_role.unassigned` row (the
428/// [`axiam_authz::network_gate::AUDIT_ROLE_UNASSIGNED`] action) for each
429/// **network role** it removes.
430///
431/// A network role here is what the gate calls a network-sensitive role: one
432/// carrying at least one `network:` grant, allow or deny
433/// ([`axiam_core::models::network::is_network_action`] over the role's
434/// permission grants, read before the unassignment). The row has the shape the
435/// REST gate writes for an administrator's unassignment (`role_id`, `role`,
436/// `grantee {kind, id}`, `node`, `permissions`), tenant of the account,
437/// resource the node, actor the platform (`System`, nil id) — the erasure is
438/// the platform's act, and the row says so. The grantee's id is cleared again
439/// by `pseudonymize_actor` when the account is anonymized, so no row names
440/// the erased subject.
441///
442/// Each row is written after its unassignment landed, like the gate's, and only
443/// when that unassignment is the one that removed the edge (#792): a re-entered
444/// erasure finds no assignment left and writes none, and of two units that both
445/// read one assignment (a stalled unit and its successor) one writes the row. A public free
446/// function, for the reason [`run_erasure_pipeline`] is.
447///
448/// Last, `flush` (when the deployment caches decisions) drops the account's
449/// cached decisions, so a decision evaluated before the unassignments is not
450/// served, nor inserted behind them, until the TTL (#802, R112-09 (b)). It

528/// The `oauth2_refresh_token` sweep (#753): delete the rows whose token
529/// expired more than [`REFRESH_REPLAY_RETENTION`] before `now`, revoked or not,
530/// in bounded batches, and return how many.
531///
532/// The cutoff is `expires_at`, never `revoked`: a rotated or revoked token is
533/// recognised as a replay by its row until it is deleted (T-254), so a revoked
534/// row whose expiry is still ahead stays.
535///
536/// Runs at most [`REFRESH_SWEEP_MAX_BATCHES`] batches of
537/// [`REFRESH_SWEEP_BATCH`] rows, and stops between batches on a shutdown
538/// signal; the next tick continues (R112W1-02). The first run after the
539/// upgrade, which finds every row expired since the deployment began, thus
540/// takes as many ticks as it needs and never one statement over the whole
541/// backlog. Two replicas may run it at once: each batch is chosen and deleted
542/// by id, and a row the other replica removed first is not counted. A public
543/// free function, for the reason [`run_erasure_pipeline`] is.
544///
545/// # Errors
546///
547/// The repository's error from the batch that failed; batches already deleted
548/// stay deleted.
549pub async fn sweep_refresh_tokens<RT: axiam_core::repository::RefreshTokenRepository>(
550    repo: &RT,
551    now: chrono::DateTime<Utc>,
552    shutdown: &watch::Receiver<bool>,
553) -> Result<u64, AxiamError> {
554    let before = now - REFRESH_REPLAY_RETENTION;
555    let mut removed = 0_u64;
556    for _ in 0..REFRESH_SWEEP_MAX_BATCHES {
557        if *shutdown.borrow() {
558            break;
559        }
560        let batch = repo.delete_expired(before, REFRESH_SWEEP_BATCH).await?;
561        removed += batch;
562        if batch < u64::from(REFRESH_SWEEP_BATCH) {
563            break;
564        }
565    }
566    Ok(removed)
567}

591/// One pass of the `tenant_purge` job: purge every tombstoned tenant and, when
592/// `scan_orphans`, every orphaned tenant id.
593///
594/// A free function, and public, for the reason [`run_erasure_pipeline`] is
595/// one. Each tenant is purged by
596/// [`SurrealTenantRepository::purge_tombstoned_rows`] — every tenant-scoped
597/// table in the order user erasure uses, of its audit trail the rows at or
598/// before the time of the export the deletion required (T-118, R1W1-04) — and
599/// then [`SurrealTenantRepository::remove_purged_tenant`], its tenant row; or,
600/// for an orphan, under its claim marker, by
601/// [`SurrealTenantRepository::purge_orphan_rows`], which leaves the audit
602/// trail to the retention sweep, and then by the same removal of the marker.
603/// Audit rows the purge keeps name a tenant that no longer exists and are
604/// deleted by the audit retention sweep (T-119) like any other. Each completed purge is recorded in the **system**
605/// log as `tenants.purged`, beside the `tenants.deleted` record the deletion
606/// wrote, with the export time used and the audit rows deleted and kept;
607/// neither record is ever purged.
608///
609/// Every replica runs this sweep (#610). A tombstoned tenant is purged only by
610/// the replica whose [`SurrealTenantRepository::claim_purge`] wins it; the
611/// others skip it for [`TENANT_PURGE_LEASE`]. A tenant the purge finds already
612/// purged — gone before the claim, or removed by an overlapping purge after a
613/// lease ran out — is success, not a failure, and is not recorded again: only
614/// the purge that removed the tenant row writes `tenants.purged`. An orphaned
615/// id is claimed the same way, by a marker row
616/// ([`SurrealTenantRepository::claim_orphan_purge`], R101W1-09 (a)): replicas
617/// whose daily scans coincide purge and record it once, and a scan that
618/// predates another replica's finished purge finds nothing left, writes no
619/// marker and records nothing (#714).
620///
621/// Each tenant's purge, from its claim to its record, is one unit of
622/// [`AuditedWork`] (#655): a task of its own, which aborting the cleanup task
623/// does not cut. The sweep starts no purge once the process is stopping
624/// (checked between tenants); a purge under way gives itself up at the stop as
625/// long as its tenant row is still there, releasing its claim, and once the row
626/// is removed it runs on to its record.
627///
628/// The removal leaves a `tenant_purge_pending` marker in its own transaction,
629/// which the unit clears once it has recorded the purge (#689). A unit cut in
630/// between leaves the marker, and every pass starts by recording what the
631/// markers left standing ([`record_pending_purges`]), so the next start writes
632/// the `tenants.purged` the cut unit did not, once.
633///
634/// * `Ok(n)` is the number of tenants (and orphaned ids) this call purged.
635/// * One tenant's failure does not stop the others; any failure makes the sweep
636///   fail, naming how many failed and never a tenant's data. The failed tenant
637///   stays tombstoned, its claim is released, and the next pass resumes it
638///   (every step is idempotent).
639/// * A stop ends the pass early with `Ok`: the tenants left stay tombstoned and
640///   unclaimed, and the next start purges them.
641///
642/// # Errors
643///
644/// [`AxiamError::Internal`] as above, or the datastore error that kept the
645/// tombstoned tenants or the orphans from being listed (the tombstoned tenants
646/// are still purged when only the orphans could not be).
647pub async fn sweep_tenant_purge<C, S>(
648    tenant_repo: &Arc<SurrealTenantRepository<C>>,
649    audit: &Arc<S>,
650    scan_orphans: bool,
651    work: &AuditedWork,
652) -> Result<u64, AxiamError>
653where
654    C: Connection,
655    S: AuditWriteSink + 'static,
656{
657    let lease = chrono::TimeDelta::from_std(TENANT_PURGE_LEASE)
658        .map_err(|e| AxiamError::Internal(format!("tenant purge lease: {e}")))?;
659    // #689: first, the purges a unit removed and was cut before recording. A
660    // failure here does not hold back the purges; it is reported after them.
661    let record_lease = chrono::TimeDelta::from_std(PENDING_RECORD_LEASE)
662        .map_err(|e| AxiamError::Internal(format!("pending record lease: {e}")))?;
663    let pending_error = record_pending_purges(tenant_repo, audit.as_ref(), record_lease, work)
664        .await
665        .err();
666
667    let mut due: Vec<(Uuid, bool)> = tenant_repo
668        .list_tombstoned()
669        .await?
670        .into_iter()
671        .map(|id| (id, false))
672        .collect();
673    // A failed look for orphans does not hold back the tombstoned tenants: it
674    // is reported after them, and the next pass looks again.
675    let mut scan_error = None;
676    if scan_orphans {
677        match tenant_repo.orphaned_tenant_ids().await {
678            Ok(ids) => due.extend(ids.into_iter().map(|id| (id, true))),
679            Err(e) => scan_error = Some(e),
680        }
681    }
682
683    let (mut purged, mut failed) = (0u64, 0u64);
684    let due_count = due.len();
685    for (started, (tenant_id, orphan)) in due.into_iter().enumerate() {
686        // #655: no purge starts once the process is stopping; the tenants left
687        // stay tombstoned, and the next start purges them.
688        if work.stopping() {
689            tracing::info!(
690                purged,
691                left = due_count - started,
692                "tenant purge sweep stopped: the process is stopping (the rest stay tombstoned)"
693            );
694            break;
695        }
696        let (tenants, sink, unit) = (Arc::clone(tenant_repo), Arc::clone(audit), work.clone());
697        let outcome = if orphan {
698            work.run(purge_orphan_unit(tenants, sink, tenant_id, unit))
699                .await
700        } else {
701            work.run(purge_tombstoned_unit(tenants, sink, tenant_id, lease, unit))
702                .await
703        };
704        match outcome {
705            Some(Ok(PurgeOutcome::Purged)) => purged += 1,
706            // Another replica holds it, or it is already gone.
707            Some(Ok(PurgeOutcome::Skipped)) => {}
708            // Given up at the stop (or still running, and left to the
709            // teardown's join): the loop's check above ends the pass.
710            Some(Ok(PurgeOutcome::Abandoned)) | None if work.stopping() => {}
711            Some(Ok(PurgeOutcome::Abandoned)) => {}
712            // The unit panicked; its claim guard released the claim.
713            None => {
714                failed += 1;
715                tracing::warn!(
716                    %tenant_id,
717                    orphan,
718                    "tenant purge ended without an outcome; the tenant stays tombstoned and the \
719                     next sweep resumes it"
720                );
721            }
722            Some(Err(e)) => {
723                failed += 1;
724                tracing::warn!(
725                    error = %e,
726                    %tenant_id,
727                    orphan,
728                    "tenant purge incomplete; the tenant stays tombstoned and the next sweep resumes it"
729                );
730            }
731        }
732    }
733
734    if failed > 0 {
735        return Err(AxiamError::Internal(format!(
736            "tenant purge was incomplete for {failed} of {} tenant(s)",
737            failed + purged
738        )));
739    }
740    if let Some(e) = scan_error.or(pending_error) {
741        return Err(e);
742    }
743    Ok(purged)
744}

746/// How one tenant's purge unit ended, when it did not fail.
747enum PurgeOutcome {
748    /// This unit purged the tenant and wrote `tenants.purged`.
749    Purged,
750    /// Another replica holds the claim, or the tenant is already purged (for
751    /// an orphaned id: a scan that predates its finished purge).
752    Skipped,
753    /// Given up because the process is stopping, before the point of no
754    /// return; the tenant stays tombstoned and unclaimed.
755    Abandoned,
756}

758/// One tombstoned tenant's purge, from its claim to its `tenants.purged`
759/// record: one unit of [`AuditedWork`] (#655).
760///
761/// The point of no return is the removal of the tenant row: before it every
762/// step is idempotent and the tenant stays tombstoned, so the unit gives up at
763/// the stop and releases its claim; it never removes the row once the process
764/// is stopping. After it, the record is the only thing left, and the unit
765/// writes it whatever the stop says.
766///
767/// The tombstoned row may be an orphan purge's claim marker whose claimant
768/// died or gave it up (R101W1-09 (a)): its rows are then purged as an orphan's
769/// and the record says `orphan: true` (the report's `orphan`).
770async fn purge_tombstoned_unit<C, S>(
771    tenants: Arc<SurrealTenantRepository<C>>,
772    audit: Arc<S>,
773    tenant_id: Uuid,
774    lease: chrono::TimeDelta,
775    work: AuditedWork,
776) -> Result<PurgeOutcome, AxiamError>
777where
778    C: Connection,
779    S: AuditWriteSink,
780{
781    let claimed_at = Utc::now();
782    if !tenants.claim_purge(tenant_id, claimed_at, lease).await? {
783        return Ok(PurgeOutcome::Skipped);
784    }
785    // From here every exit but a removed tenant row releases the claim — the
786    // awaited release below, or, for an abort or a panic, the guard's drop.
787    let claim = PurgeClaim {
788        tenants: Arc::clone(&tenants),
789        tenant_id,
790        claimed_at,
791        held: true,
792    };
793    let rows = work
794        .until_stopped(async {
795            let report = tenants.purge_tombstoned_rows(tenant_id).await?;
796            work.step(UnitStep::TenantRowsPurged(tenant_id)).await;
797            Ok::<_, AxiamError>(report)
798        })
799        .await;
800    let report = match rows {
801        Some(Ok(report)) => report,
802        // Purged by an overlapping replica: done, and recorded by the purge
803        // that removed the row.
804        Some(Err(AxiamError::NotFound { .. })) => {
805            claim.settle();
806            return Ok(PurgeOutcome::Skipped);
807        }
808        // Let the next tick (any replica's) retry it now rather than after
809        // the lease.
810        Some(Err(e)) => {
811            claim.release().await;
812            return Err(e);
813        }
814        None => {
815            claim.release().await;
816            return Ok(PurgeOutcome::Abandoned);
817        }
818    };
819    // Once the process is stopping, no new crossing of the point of no return
820    // starts (a crossing already being sent completes).
821    if work.stopping() {
822        claim.release().await;
823        return Ok(PurgeOutcome::Abandoned);
824    }
825    let Some(record) = cross_point_of_no_return(&tenants, tenant_id, &report, claim).await? else {
826        return Ok(PurgeOutcome::Skipped);
827    };
828    work.step(UnitStep::TenantRowRemoved(tenant_id)).await;
829    record_tenant_purged(audit.as_ref(), tenant_id, false, &report).await;
830    work.step(UnitStep::Recorded(tenant_id)).await;
831    record.settle().await;
832    Ok(PurgeOutcome::Purged)
833}

835/// A purge unit's point of no return (#655): remove the tenant row (or the
836/// orphan's marker) and, in the same transaction, write its
837/// `tenant_purge_pending` marker (#689), then settle the purge claim and hold
838/// the marker's in its place. `Ok(None)` when an overlapping purge removed the
839/// row first (it records the purge).
840///
841/// The marker carries a stamp of its own, taken here, rather than the purge
842/// claim's: the claim may be as old as the purge, and a stale stamp would let
843/// another replica's sweep take the marker while this unit records it.
844async fn cross_point_of_no_return<C: Connection>(
845    tenants: &Arc<SurrealTenantRepository<C>>,
846    tenant_id: Uuid,
847    report: &axiam_db::repository::tenant_purge::TenantPurgeReport,
848    claim: PurgeClaim<C>,
849) -> Result<Option<RecordClaim<C>>, AxiamError> {
850    let stamp = Utc::now();
851    match tenants.remove_purged_tenant(tenant_id, report, stamp).await {
852        Ok(()) => {
853            claim.settle();
854            Ok(Some(RecordClaim {
855                marker: PendingMarker::TenantPurge {
856                    tenants: Arc::clone(tenants),
857                    tenant_id,
858                },
859                stamp,
860                held: true,
861            }))
862        }
863        Err(AxiamError::NotFound { .. }) => {
864            claim.settle();
865            Ok(None)
866        }
867        Err(e) => {
868            // The transaction may have committed before the error reached us:
869            // release a marker stamped with ours, if there is one, so the next
870            // sweep records the purge at once. Nothing matches otherwise.
871            if let Err(release) = tenants.release_purge_record(tenant_id, stamp).await {
872                tracing::warn!(
873                    error = %release,
874                    %tenant_id,
875                    "could not release a tenant purge's pending record; it lapses with its lease"
876                );
877            }
878            claim.release().await;
879            Err(e)
880        }
881    }
882}

884/// Record the tenant purges whose `tenant_purge_pending` marker (#689) still
885/// stands — a unit cut between the removal of the tenant row and the clearing
886/// of its marker — and clear each marker. The first step of every
887/// [`sweep_tenant_purge`] pass; none starts once the process is stopping.
888///
889/// Each marker is claimed first
890/// ([`SurrealTenantRepository::claim_pending_purge_records`]): only one released
891/// or older than `lease`, and conditionally on the claim read, so a unit still
892/// recording and another replica's sweep are never both its owner. The record
893/// is written unless the system log already holds a `tenants.purged` for the
894/// id written since the removal (a unit cut after its record and before the
895/// clearing), with what the removal stored: the record is the one the unit
896/// would have written. The claim is held by the same guard a unit holds, so a
897/// pass cut here releases it.
898///
899/// `Ok(n)` is the number of records this call wrote.
900///
901/// # Errors
902///
903/// The datastore error that kept the markers from being read or claimed, or
904/// [`AxiamError::Internal`] naming how many could not be checked (each of
905/// those is released for the next pass).
906pub async fn record_pending_purges<C, S>(
907    tenants: &Arc<SurrealTenantRepository<C>>,
908    audit: &S,
909    lease: chrono::TimeDelta,
910    work: &AuditedWork,
911) -> Result<u64, AxiamError>
912where
913    C: Connection,
914    S: AuditWriteSink,
915{
916    if work.stopping() {
917        return Ok(0);
918    }
919    let pending = tenants
920        .claim_pending_purge_records(Utc::now(), lease)
921        .await?;
922    let (mut recorded, mut failed) = (0u64, 0u64);
923    for marker in pending {
924        let claim = RecordClaim {
925            marker: PendingMarker::TenantPurge {
926                tenants: Arc::clone(tenants),
927                tenant_id: marker.tenant_id,
928            },
929            stamp: marker.claimed_at,
930            held: true,
931        };
932        if work.stopping() {
933            claim.release().await;
934            continue;
935        }
936        match tenants
937            .purge_recorded_since(
938                marker.tenant_id,
939                axiam_api_rest::handlers::tenants::TENANT_PURGED_ACTION,
940                marker.purged_at,
941            )
942            .await
943        {
944            Ok(true) => {}
945            Ok(false) => {
946                record_tenant_purged(audit, marker.tenant_id, false, &marker.report).await;
947                recorded += 1;
948                tracing::warn!(
949                    tenant_id = %marker.tenant_id,
950                    "recorded a tenant purge whose unit was cut before its record (#689)"
951                );
952                work.step(UnitStep::Recorded(marker.tenant_id)).await;
953            }
954            Err(e) => {
955                failed += 1;
956                tracing::warn!(
957                    error = %e,
958                    tenant_id = %marker.tenant_id,
959                    "could not check a pending tenant purge record; the next pass retries it"
960                );
961                claim.release().await;
962                continue;
963            }
964        }
965        claim.settle().await;
966    }
967    if failed > 0 {
968        return Err(AxiamError::Internal(format!(
969            "{failed} pending tenant purge record(s) could not be checked"
970        )));
971    }
972    Ok(recorded)
973}

975/// One orphaned tenant id's purge and its `tenants.purged` record: one unit of
976/// [`AuditedWork`] (#655), claimed by a marker row (R101W1-09 (a)).
977///
978/// The unit arms its [`PurgeClaim`] **before**
979/// [`SurrealTenantRepository::claim_orphan_purge`] writes the marker, keyed by
980/// the stamp the marker will carry, so the marker is guarded from the instant
981/// it exists (#714): a cut anywhere from the `CREATE` on releases the claim,
982/// and the release is conditional on the stamp, so it changes nothing when the
983/// `CREATE` did not land. The claim writes the marker only while the id still
984/// owns a row to purge: another replica's claim, or a scan that predates a
985/// finished purge, is skipped, writes nothing and records nothing. Then the
986/// unit runs as [`purge_tombstoned_unit`] does: the rows
987/// ([`SurrealTenantRepository::purge_orphan_rows`]), then the removal of the
988/// marker ([`SurrealTenantRepository::remove_purged_tenant`]) as the point of
989/// no return, and the record only when this unit removed it. Given up at the
990/// stop, failed, or cut, the marker stays tombstoned with its claim released,
991/// and the tombstoned-tenant sweep purges it at once as the orphan it is, and
992/// records it — a marker always stands for a purge nobody has recorded.
993async fn purge_orphan_unit<C, S>(
994    tenants: Arc<SurrealTenantRepository<C>>,
995    audit: Arc<S>,
996    tenant_id: Uuid,
997    work: AuditedWork,
998) -> Result<PurgeOutcome, AxiamError>
999where
1000    C: Connection,
1001    S: AuditWriteSink,
1002{
1003    let claimed_at = Utc::now();
1004    // Armed before the marker exists (#714): every exit from here but a
1005    // removed marker — or a claim that wrote none — releases the claim, so the
1006    // marker is the tombstoned-tenant sweep's at once.
1007    let claim = PurgeClaim {
1008        tenants: Arc::clone(&tenants),
1009        tenant_id,
1010        claimed_at,
1011        held: true,
1012    };
1013    match tenants.claim_orphan_purge(tenant_id, claimed_at).await {
1014        Ok(true) => {}
1015        // No marker of ours: another replica's, nothing left, or no orphan.
1016        Ok(false) => {
1017            claim.settle();
1018            tenants.orphan_claim_refusal(tenant_id).await?;
1019            return Ok(PurgeOutcome::Skipped);
1020        }
1021        // The marker may have landed: release it, by its stamp.
1022        Err(e) => {
1023            claim.release().await;
1024            return Err(e);
1025        }
1026    }
1027    work.step(UnitStep::OrphanClaimed(tenant_id)).await;
1028    let report = match work
1029        .until_stopped(tenants.purge_orphan_rows(tenant_id))
1030        .await
1031    {
1032        Some(Ok(report)) => report,
1033        Some(Err(e)) => {
1034            claim.release().await;
1035            return Err(e);
1036        }
1037        None => {
1038            claim.release().await;
1039            return Ok(PurgeOutcome::Abandoned);
1040        }
1041    };
1042    // Once the process is stopping, no new crossing of the point of no return
1043    // starts (a crossing already being sent completes).
1044    if work.stopping() {
1045        claim.release().await;
1046        return Ok(PurgeOutcome::Abandoned);
1047    }
1048    // `None`: the marker was taken over and removed by another purge, which
1049    // records it.
1050    let Some(record) = cross_point_of_no_return(&tenants, tenant_id, &report, claim).await? else {
1051        return Ok(PurgeOutcome::Skipped);
1052    };
1053    work.step(UnitStep::TenantRowRemoved(tenant_id)).await;
1054    record_tenant_purged(audit.as_ref(), tenant_id, true, &report).await;
1055    work.step(UnitStep::Recorded(tenant_id)).await;
1056    record.settle().await;
1057    Ok(PurgeOutcome::Purged)
1058}

1097/// A replica's claim on a tombstoned tenant's purge (#610), released on every
1098/// exit but a removed tenant row (#655). An orphan purge arms it before its
1099/// claim marker is written, so the marker is guarded from the instant it
1100/// exists (#714).
1101///
1102/// The purge releases it itself, awaited, where it can — a failure, or giving
1103/// up at the stop. Every other exit — the unit aborted at the teardown's bound,
1104/// a panic, a release that was itself cut — drops the guard still holding the
1105/// claim, and the drop spawns the release on the runtime. Without that, the
1106/// next start's sweep found a claim younger than [`TENANT_PURGE_LEASE`] and
1107/// skipped the tenant for up to an hour. That release is a spawned task, and
1108/// nothing waits for it: a unit cut at the teardown's bound (the one exit where
1109/// the guard is the only release, typically a datastore too slow to answer the
1110/// awaited one) is dropped inside [`AuditedWork::join`], and the process exit
1111/// that follows drops the runtime, which cancels the task unless it has already
1112/// landed. The tenant then waits out [`TENANT_PURGE_LEASE`] (#687; awaiting the
1113/// release at the bound was declined, since it needs a budget line of its own).

1119}

1456/// The cleanup task's units of work that must end in an audit row (#655,
1457/// #624): a tombstoned tenant's purge through its `tenants.purged` record, an
1458/// orphaned tenant id's purge through its record, and one account's erasure
1459/// through its `account_deletion` completion and its `gdpr.user_pseudonymized`
1460/// row.
1461///
1462/// The cleanup task runs them one at a time, each as a `tokio::spawn`ed task of
1463/// its own ([`Self::run`]), so aborting the cleanup task — the composition
1464/// root's bounded join — does not cut one. The root then waits for the unit in
1465/// flight with [`Self::join`], within the same bound, and aborts it past that,
1466/// so a replica that lost its lease writes nothing of its own code after its
1467/// stop's budget (a statement already sent may still commit).
1468///
1469/// Each unit has a **point of no return** — the write after which no later
1470/// sweep finds the work again: the removal of the tenant row; the anonymization
1471/// of the account. Before it, the unit gives itself up as soon as the process
1472/// is stopping (the tenant stays tombstoned, its claim released; the account
1473/// stays due), and it starts no new crossing once the stop is requested (a
1474/// crossing already being sent completes). So the only unit a stop can find
1475/// past its point of no return began crossing it before the stop was requested, and it has the whole of the teardown's window for its last
1476/// statements — from the stop signal through the gRPC stop, the lease release
1477/// and the outbound and audit drains to the end of the cleanup bound: 5 to 18 s
1478/// for `SIGTERM`, 2 to 13 s after a lost lease, whose stop signal comes at the
1479/// loss (#628). One that outlasts even that is cut; the marker its point of
1480/// no return left in the same transaction (#689) makes the next start's sweep
1481/// write the record it owed, once ([`record_owed_erasures`],
1482/// [`record_pending_purges`]; T-444).
1483#[derive(Clone)]
1484pub struct AuditedWork {
1485    stop: watch::Receiver<bool>,
1486    in_flight: Arc<std::sync::Mutex<Option<tokio::task::JoinHandle<()>>>>,
1487    step_for_tests: Option<UnitStepForTests>,
1488}

1490impl AuditedWork {
1491    /// Units that give themselves up when `stop` reads `true` — the cleanup
1492    /// task's shutdown signal. A `stop` whose sender is gone while it reads
1493    /// `false` never stops anything.
1494    #[must_use]
1495    pub fn new(stop: watch::Receiver<bool>) -> Self {
1496        Self {
1497            stop,
1498            in_flight: Arc::default(),

1504    /// [`UnitStep`], so a test can hold a unit there across a stop.

1512    /// Whether the process has asked the cleanup task to stop.

1514    pub fn stopping(&self) -> bool {

1518    /// Resolves once the process is stopping; never, when it cannot be.

1522            std::future::pending::<()>().await;
1523        }

1526    /// `work`'s output, or `None` when the process is stopping first: `work`
1527    /// is then dropped where it is. For the steps before a unit's point of no
1528    /// return only.
1529    async fn until_stopped<F: std::future::Future>(&self, work: F) -> Option<F::Output> {
1530        tokio::select! {
1531            biased;
1532            output = work => Some(output),
1533            () = self.stopped() => None,
1534        }
1535    }

1537    /// The test step at `at`, if one is set.

1540            step(at).await;
1541        }

1544    fn slot(&self) -> std::sync::MutexGuard<'_, Option<tokio::task::JoinHandle<()>>> {
1545        self.in_flight
1546            .lock()
1547            .unwrap_or_else(std::sync::PoisonError::into_inner)
1548    }
1549
1550    /// Spawn `unit` as a task of its own and wait for its output.
1551    ///
1552    /// `None` when the process is stopping — before the unit was started, or
1553    /// while it runs: it is then left running for [`Self::join`] — or when the
1554    /// unit panicked (logged here). One unit at a time: the caller awaits this
1555    /// before it starts the next.
1556    pub async fn run<T: Send + 'static>(
1557        &self,
1558        unit: impl std::future::Future<Output = T> + Send + 'static,
1559    ) -> Option<T> {
1560        if self.stopping() {
1561            return None;
1562        }
1563        let (tx, rx) = tokio::sync::oneshot::channel();
1564        *self.slot() = Some(tokio::spawn(async move {
1565            let _ = tx.send(unit.await);
1566        }));
1567        tokio::select! {
1568            biased;
1569            output = rx => {
1570                let handle = self.slot().take();
1571                match output {
1572                    Ok(output) => Some(output),
1573                    Err(_) => {
1574                        if let Some(handle) = handle
1575                            && let Err(e) = handle.await
1576                        {
1577                            tracing::error!(error = %e, "a cleanup unit ended without an outcome");
1578                        }
1579                        None
1580                    }
1581                }
1582            }
1583            // Left to the teardown, which waits for it.
1584            () = self.stopped() => None,
1585        }
1586    }
1587
1588    /// Wait until `deadline` for the unit in flight, if any; past it, abort the
1589    /// unit — it is cut where it is — and say so with `false`. The composition
1590    /// root calls this once the cleanup task itself has ended or been aborted
1591    /// (#655, #624).
1592    pub async fn join(&self, deadline: tokio::time::Instant) -> bool {
1593        let Some(mut handle) = self.slot().take() else {
1594            return true;
1595        };
1596        match tokio::time::timeout_at(deadline, &mut handle).await {
1597            Ok(Ok(())) => true,
1598            Ok(Err(e)) => {
1599                tracing::error!(error = %e, "a cleanup unit ended without an outcome");
1600                true
1601            }
1602            Err(_) => {
1603                handle.abort();
1604                // Once this returns the unit has been dropped: it writes
1605                // nothing more, and its claim guard has run.
1606                let _ = handle.await;
1607                false

1618///
1619/// A free function, and public, for the reason [`sweep_directories`] is one.

1653/// The `certificate_retention` job (R101W1-05, #618): delete certificate rows
1654/// of live tenants more than eight days past their `notAfter`, in bounded
1655/// batches, and return how many.
1656///
1657/// A revoked row is kept for as long as its issuer's revocation list can name
1658/// it (the tenant purge's `KEPT_REVOCATION`, whose grace this sweep shares);
1659/// past it the CRL is the same without the row. Nothing deleted a certificate
1660/// row in a live tenant before, so a revoked one — and the PEM, subject and
1661/// `metadata` in it, of an account the Art. 17 purge had erased — outlived any
1662/// purpose for the tenant's lifetime. A deleted tenant's rows are the tenant
1663/// purge's, not this sweep's. Two replicas may run it at once: each batch is
1664/// chosen and deleted by id, and a row the other replica removed first is not
1665/// counted. Touches no audit row.
1666///
1667/// Runs at most [`CERTIFICATE_RETENTION_MAX_BATCHES`] batches of
1668/// [`CERTIFICATE_RETENTION_BATCH`] rows, and stops between batches on a
1669/// shutdown signal; the next tick continues. A free function, and public, for
1670/// the reason [`run_erasure_pipeline`] is.
1671///
1672/// # Errors
1673///
1674/// The repository's error from the batch that failed; batches already deleted
1675/// stay deleted.
1676pub async fn sweep_certificate_retention<CR>(
1677    cert_repo: &CR,
1678    shutdown: &watch::Receiver<bool>,
1679) -> Result<u64, AxiamError>
1680where
1681    CR: axiam_core::repository::CertificateRepository,
1682{
1683    let mut removed = 0_u64;
1684    for _ in 0..CERTIFICATE_RETENTION_MAX_BATCHES {
1685        if *shutdown.borrow() {
1686            break;
1687        }
1688        let batch = cert_repo
1689            .delete_past_retention(CERTIFICATE_RETENTION_BATCH)
1690            .await?;
1691        removed += batch;
1692        if batch < u64::from(CERTIFICATE_RETENTION_BATCH) {
1693            break;
1694        }
1695    }
1696    Ok(removed)
1697}

1703/// One pass of the directory sync job, as a sweep the scheduler can record.
1704///
1705/// A free function, and public, for the reason [`run_erasure_pipeline`] is one:
1706/// the loop that calls it is awkward to construct in a test, and the conversion
1707/// of the job's outcome into job health is the part worth testing directly.
1708///
1709/// * `Ok(n)` is the number of accounts the pass changed (deactivated, updated or
1710///   re-mapped); a pass that found nothing to do is `Ok(0)`.
1711/// * **Any tenant's failure makes the sweep fail** — an unreachable directory, a
1712///   refused search, the safety valve — with a message that names how many
1713///   tenants failed and the fixed tag of each, never a tenant's data. The other
1714///   tenants still ran; one tenant's outage is not another's.
1715/// * A shutdown signal abandons the pass (`Ok(0)`): every step the job takes is
1716///   idempotent or a compare-and-set, and the next process starts it again.
1717///
1718/// # Errors
1719///
1720/// [`AxiamError::Internal`] when the pass failed, as described above.
1721pub async fn sweep_directories<C: Connection + Send + Sync + 'static>(
1722    sync: &DirectorySyncJob<C>,
1723    mut shutdown: watch::Receiver<bool>,
1724) -> Result<u64, AxiamError> {
1725    let outcome = tokio::select! {
1726        // A shutdown request wins over starting work: polled first.
1727        biased;
1728        _ = async {
1729            // Resolves on a shutdown request; a dropped sender never does.
1730            loop {
1731                if *shutdown.borrow() {
1732                    return;
1733                }
1734                if shutdown.changed().await.is_err() {
1735                    std::future::pending::<()>().await;
1736                }
1737            }
1738        } => {
1739            tracing::info!("directory sync abandoned: the process is shutting down");
1740            return Ok(0);
1741        }
1742        outcome = sync.run_due() => outcome,
1743    };
1744    match outcome {
1745        Ok(summary) => match summary.failure_message() {
1746            Some(message) => Err(AxiamError::Internal(message)),
1747            None => Ok(summary.changed()),
1748        },
1749        Err(error) => Err(AxiamError::Internal(format!(
1750            "directory sync could not list the tenants' directories ({})",
1751            error.tag()
1752        ))),
1753    }
1754}

1760/// Whether a self-registered client is due to be swept (T21.4).
1761///
1762/// A free function, and public, for the reason [`run_erasure_pipeline`] is
1763/// one: the loop that calls it is awkward to construct in a test, and the
1764/// decision it makes — *delete somebody's client registration* — is the part
1765/// worth testing directly.
1766///
1767/// The clock it reads is `last_authorized_at` when the client has ever been
1768/// authorized and `created_at` when it has not.
1769///
1770/// `DcrSweepWindow::days == 0` is never due. Zero means "never sweep", which
1771/// an operator who prunes out of band may legitimately want, and reading it as
1772/// "sweep everything immediately" would delete a tenant's whole client table
1773/// on the next tick.
1774///
1775/// # The second clock (T21.8 / MCP-05)
1776///
1777/// A row that has **never** been authorized, in a tenant whose effective
1778/// `dynamic_registration` is `anonymous`, is measured against
1779/// [`axiam_core::models::settings::DCR_UNAUTHORIZED_CLIENT_TTL_SECS`] instead
1780/// — an hour. That constant's own documentation says why it is a constant and
1781/// what promoting it to a tenant setting would cost.
1782///
1783/// Those two conditions are the whole of the change, and each is load-bearing.
1784///
1785/// **Never authorized.** The thirty-day window is sized, in
1786/// `DEFAULT_DCR_UNUSED_CLIENT_TTL_DAYS`'s own doc comment, for "a client
1787/// somebody uses monthly". A client registered and never authorized is not
1788/// that client: every MCP client this phase exists to serve authorizes within
1789/// seconds of registering, because registration is the first step of the same
1790/// flow. One TTL served two situations that have nothing in common, which is
1791/// what made `dcr_max_clients` an availability budget a stranger could spend
1792/// for a month. The sweeper could always tell the two apart — the row carries
1793/// `last_authorized_at: None` — so the fix is a second clock, not a bigger
1794/// quota.
1795///
1796/// **`anonymous` only.** In `initial_access_token` mode the row exists because
1797/// an administrator minted a handle for it and somebody redeemed it. There is
1798/// no unauthenticated exposure to bound, and the thirty-day clock is the right
1799/// one: an operator who hands somebody a registration token on Friday should
1800/// not find the registration gone on Monday.
1801///
1802/// Every tenant on `disabled` or `initial_access_token` — which is every
1803/// tenant that did not opt into `anonymous` behind D3 — takes byte for byte
1804/// the path it took before T21.8.
1805pub fn dcr_client_is_due_for_sweep(
1806    client: &axiam_core::models::oauth2_client::OAuth2Client,
1807    ttl: DcrSweepWindow,
1808    now: chrono::DateTime<Utc>,
1809) -> bool {
1810    if client.last_authorized_at.is_none() && ttl.mode == DynamicRegistrationMode::Anonymous {
1811        return now - client.created_at
1812            > chrono::Duration::seconds(i64::from(DCR_UNAUTHORIZED_CLIENT_TTL_SECS));
1813    }
1814    if ttl.days == 0 {
1815        return false;
1816    }
1817    now - dcr_client_last_seen(client) > chrono::Duration::days(i64::from(ttl.days))
1818}

1843/// When a `cimd` row was last any use to anybody — T21.8 / MCP-04.
1844///
1845/// `max(updated_at, last_authorized_at, created_at)`, and each of the three is

1958/// Delete externally registered clients that have gone unseen for longer than
1959/// their tenant's `dcr_unused_client_ttl_days` (T21.4, widened to `cimd` by
1960/// T21.8 / MCP-04).
1961///
1962/// One `managed_by` per call, so `/health/jobs` can distinguish the two
1963/// sweeps: they delete different things for different reasons and an operator
1964/// debugging one should not have to read the other's counter.
1965///
1966/// # What it will not touch
1967///
1968/// Never `admin`. An administrator's client is never swept, however long it
1969/// sits unused, because somebody decided it should exist and nothing here is
1970/// entitled to reverse that. The repository query filters on the value rather
1971/// than on "not admin", so a fourth provenance added later is opted **in** by
1972/// somebody writing it down.
1973///
1974/// # Why `cimd` is swept now, when T21.4 argued it should not be
1975///
1976/// T21.4's reasoning, which this function used to carry: a CIMD shadow row is
1977/// a cache of a document the client publishes, so deleting it would be
1978/// re-materialised on the next request and the TTL would mean nothing.
1979///
1980/// That is right about TTL semantics and says nothing about storage, which is
1981/// what MCP-04 is about: a cache that is never evicted is not a cache. An
1982/// inert row is still a row — listed on the OAuth2 clients page, counted in
1983/// every `list_all_by_managed_by` this function runs, and a permanent write a
1984/// stranger made at the cost of one unauthenticated request. And eviction on
1985/// last-seen is *consistent* with the old argument rather than against it: a
1986/// row deleted while its document is still published is re-materialised on the
1987/// next request, which is exactly what a cache should do. What it must not do
1988/// is delete a row that is still in use, which is what
1989/// [`cimd_client_last_seen`] is for — every resolve moves `updated_at`, so a
1990/// document presented once a day is never due under a 30-day TTL.
1991///
1992/// # Per-tenant TTL, resolved per row
1993///
1994/// The window is a tenant setting, so each row's tenant is resolved and its
1995/// effective settings read. Deployment-wide sweeps elsewhere in this file
1996/// (audit retention) deliberately do **not** work this way, and the difference
1997/// is who owns the decision: audit retention is a property of the datastore an
1998/// operator is responsible for, where this is a property of the tenant's
1999/// relationship with the clients it lets register.
2000///
2001/// The settings read is cached per tenant for the duration of one sweep —
2002/// `dcr_max_clients` bounds the rows per tenant, so a hundred clients in one
2003/// tenant is one read rather than a hundred.
2004///
2005/// A tenant whose settings cannot be read is skipped: the fail-closed
2006/// direction for a sweep that **deletes** is to delete nothing.
2007pub async fn sweep_unused_external_clients<CR, TR, SR, RT, AC, PR, DG, CB>(
2008    client_repo: &CR,
2009    tenant_repo: &TR,
2010    settings_repo: &SR,
2011    grants: &ClientGrantStores<RT, AC, PR, DG, CB>,
2012    managed_by: axiam_core::models::oauth2_client::ManagedBy,
2013    now: chrono::DateTime<Utc>,
2014) -> Result<u64, AxiamError>
2015where
2016    CR: axiam_core::repository::OAuth2ClientRepository,
2017    TR: TenantRepository,
2018    SR: axiam_core::repository::SettingsRepository,
2019    RT: axiam_core::repository::RefreshTokenRepository,
2020    AC: axiam_core::repository::AuthorizationCodeRepository,
2021    PR: axiam_core::repository::PushedAuthRequestRepository,
2022    DG: axiam_core::repository::DeviceGrantRepository,
2023    CB: axiam_core::repository::CibaRequestRepository,
2024{
2025    use axiam_core::models::oauth2_client::ManagedBy;
2026    use axiam_core::repository::SettingsRepository;
2027
2028    type DuePredicate = fn(
2029        &axiam_core::models::oauth2_client::OAuth2Client,
2030        DcrSweepWindow,
2031        chrono::DateTime<Utc>,
2032    ) -> bool;
2033    let (job, is_due): (&str, DuePredicate) = match managed_by {
2034        ManagedBy::Dcr => ("dcr_unused_clients", dcr_client_is_due_for_sweep),
2035        ManagedBy::Cimd => ("cimd_unused_clients", cimd_client_is_due_for_sweep),
2036        // An administrator's client is never swept. Answered here rather than
2037        // by trusting every caller, because the argument for it is a security
2038        // one and belongs next to the query.
2039        ManagedBy::Admin => return Ok(0),
2040    };
2041
2042    let clients = client_repo.list_all_by_managed_by(managed_by).await?;
2043    if clients.is_empty() {
2044        return Ok(0);
2045    }
2046
2047    let mut ttl_by_tenant: std::collections::HashMap<Uuid, Option<DcrSweepWindow>> =
2048        std::collections::HashMap::new();
2049    let mut removed = 0u64;
2050
2051    for client in clients {
2052        let ttl = match ttl_by_tenant.get(&client.tenant_id) {
2053            Some(cached) => *cached,
2054            None => {
2055                let resolved = match tenant_repo.get_by_id(client.tenant_id).await {
2056                    Ok(tenant) => SettingsRepository::get_effective_settings(
2057                        settings_repo,
2058                        tenant.organization_id,
2059                        client.tenant_id,
2060                    )
2061                    .await
2062                    .ok()
2063                    .map(|s| DcrSweepWindow {
2064                        days: s.oidc.dcr_unused_client_ttl_days,
2065                        mode: s.oidc.dynamic_registration,
2066                    }),
2067                    Err(_) => None,
2068                };
2069                ttl_by_tenant.insert(client.tenant_id, resolved);
2070                resolved
2071            }
2072        };
2073        // `None` is an unreadable tenant or an unreadable settings row.
2074        let Some(ttl) = ttl else {
2075            continue;
2076        };
2077        if !is_due(&client, ttl, now) {
2078            continue;
2079        }
2080
2081        // #517 — the client's grants go before the row, as on an
2082        // administrator's delete: a `cimd` row re-materialises under the same
2083        // `client_id`, and a refresh token, code or pushed request left behind
2084        // would work again against it. A revocation that fails keeps the row
2085        // for the next sweep rather than leave a row-less client with live
2086        // grants.
2087        if let Err(e) = grants.revoke(client.tenant_id, &client.client_id).await {
2088            tracing::warn!(
2089                job,
2090                error = %e,
2091                client_id = %client.client_id,
2092                "could not revoke an unused externally registered client's grants; not \
2093                 deleting it this sweep"
2094            );
2095            continue;
2096        }
2097        if let Err(e) = client_repo.delete(client.tenant_id, client.id).await {
2098            // One failure must not stop the sweep: the next row may be a
2099            // different tenant entirely.
2100            tracing::warn!(
2101                job,
2102                error = %e,
2103                client_id = %client.client_id,
2104                "could not delete an unused externally registered client"
2105            );
2106            continue;
2107        }
2108        // A request that read the row before the delete may have written a
2109        // grant between the two writes above; one more pass leaves none.
2110        if let Err(e) = grants.revoke(client.tenant_id, &client.client_id).await {
2111            tracing::warn!(
2112                job,
2113                error = %e,
2114                client_id = %client.client_id,
2115                "a swept client's grants could not be revoked a second time"
2116            );
2117        }
2118        removed += 1;
2119        tracing::info!(
2120            job,
2121            tenant_id = %client.tenant_id,
2122            client_id = %client.client_id,
2123            ttl_days = ttl.days,
2124            ever_authorized = client.last_authorized_at.is_some(),
2125            mode = %ttl.mode,
2126            "deleted an externally registered client that has gone unseen within its \
2127             tenant's TTL"
2128        );
2129    }
2130
2131    Ok(removed)
2132}

2412/// An erasure through its point of no return to its record (#624): the steps of
2413/// [`run_erasure_pipeline_reporting`] in its order — audit scrub,
2414/// anonymization, SSF report, proof — then the #668 second revocation, the
2415/// `gdpr.user_pseudonymized` audit row, and the `account_deletion` row marked
2416/// completed (#689 (i): the record first). Run inside a unit of
2417/// [`AuditedWork`], so a cut of the cleanup task does not reach it.
2418///
2419/// The anonymization is the point of no return: it clears `deletion_pending`,
2420/// so no later sweep selects the account again, and in the same transaction
2421/// sets the account's record marker (#689) to a stamp this call chose (#737).
2422/// The marker's guard is armed from that stamp **before** the write — the
2423/// orphan unit's pattern since #714 — so the marker is guarded from the
2424/// instant it exists: cleared with the completion once the audit row is
2425/// written, and released on every other exit, awaited where it can be and by
2426/// the guard's drop on a cut, so the next sweep ([`record_owed_erasures`])
2427/// writes at once what the erasure still owes: the proof, the second
2428/// revocation and the record. A release whose anonymization never landed
2429/// changes nothing, since it is conditional on the stamp. Until the
2430/// anonymization — the audit scrub — the call gives up once `work` is stopping
2431/// and answers `Ok(false)`: the account stays due. Past it, it runs on to the
2432/// audit row whatever the stop says, and answers `Ok(true)`. A free function,
2433/// and public, for the reason [`run_erasure_pipeline`] is one.
2434///
2435/// `Ok(false)` too, with nothing written, when another unit erased the account
2436/// after taking this one's claim over past its lease (#738, #784): the account
2437/// is read again before the scrub and left alone once it is no longer
2438/// deletion-pending, and the anonymization writes only a deletion-pending
2439/// account, so a unit that stalls after that read stops at the anonymization,
2440/// before the SSF report — its scrub having passed over the other unit's
2441/// record, which the scrub never touches. And `Ok(false)` when the erasure
2442/// proof turns out to exist already: the next sweep wrote it for this unit,
2443/// stalled past the record lease after its anonymization; this call records
2444/// nothing, and releases its marker, which that sweep settles.
2445///
2446/// # Errors
2447///
2448/// Those of the account's read, the scrub and the anonymization, before
2449/// anything is recorded; and a proof that failed to write, past the point of
2450/// no return, with the marker released so that the next sweep writes the
2451/// proof and the record. The completion and the audit row are not fallible

2466    EP: ErasureProofRepository,
2467    U: UserRepository,
2468    CR: axiam_core::repository::CertificateRepository,
2469    C: Connection,

2498        .filter(|sink| sink.is_active())
2499        .map(|sink| (sink, account));

2695/// 1. the erasure proof, unless the account has one: the proof's unique index
2696///    refuses a second, which reads as "written already" — the unit's own, or
2697///    a racing sweep's. It carries the pseudonym on the row and is dated from
2698///    the marker the claim read, or now when the marker was released or unset;
2699/// 2. the #668 second revocation, by binding only (empty names), which is
2700///    idempotent: a certificate bound to the account between the unit's (b4)
2701///    and its anonymization is revoked even when the unit was cut before its
2702///    own pass;
2703/// 3. the `gdpr.user_pseudonymized` row, unless the tenant's trail already
2704///    holds one for the account (a unit cut after its record and before the
2705///    clearing), matched by its actor or by its pseudonym — before 1.1.2
2706///    (#784) a unit that stalled past its claim's lease ran its audit scrub
2707///    over the row another unit wrote and nilled its actor, which the scrub
2708///    no longer does: an account is erased once, so that row is this
2709///    erasure's. It
2710///    carries the pseudonym the anonymization wrote, `certificates_revoked` as
2711///    `null` — the unit's count lived only in the cut unit, and the second
2712///    pass's alone would understate it — and `recovered: true`.
2713///
2714/// Then the marker is cleared and the request completed, in one transaction.
2715/// The claim is held by the same guard the unit holds. A failure of step 1 or
2716/// 2 releases the claim, so the next sweep retries the account whole: no
2717/// request is completed without its proof, nor before the second revocation
2718/// ran.
2719///
2720/// `Ok(n)` is the number of records this call wrote.
2721///
2722/// # Errors
2723///
2724/// The datastore error that kept the owed erasures from being read or
2725/// claimed, or [`AxiamError::Internal`] naming how many could not be finished
2726/// (each of those is released for the next sweep).
2727pub async fn record_owed_erasures<A, EP, CR, C>(
2728    audit_repo: &A,
2729    erasure_proof_repo: &EP,
2730    cert_repo: &CR,
2731    account_deletion_repo: &SurrealAccountDeletionRepository<C>,
2732    lease: chrono::TimeDelta,
2733    work: &AuditedWork,
2734) -> Result<u64, AxiamError>
2735where
2736    A: AuditWriteSink,
2737    EP: ErasureProofRepository,
2738    CR: axiam_core::repository::CertificateRepository,
2739    C: Connection,
2740{
2741    if work.stopping() {
2742        return Ok(0);
2743    }
2744    let owed = account_deletion_repo
2745        .claim_owed_erasure_records(Utc::now(), lease)
2746        .await?;
2747    let (mut recorded, mut failed) = (0u64, 0u64);
2748    for erasure in owed {
2749        let (tenant_id, user_id) = (erasure.tenant_id, erasure.user_id);
2750        let claim = RecordClaim {
2751            marker: PendingMarker::Erasure {
2752                deletions: account_deletion_repo.clone(),
2753                tenant_id,
2754                user_id,
2755            },
2756            stamp: erasure.claimed_at,
2757            held: true,
2758        };
2759        if work.stopping() {
2760            claim.release().await;
2761            continue;
2762        }
2763        if let Err(e) = finish_owed_erasure(erasure_proof_repo, cert_repo, &erasure).await {
2764            failed += 1;
2765            tracing::warn!(
2766                error = %e,
2767                %tenant_id,
2768                %user_id,
2769                "could not finish an owed erasure; the next sweep retries it"
2770            );
2771            claim.release().await;
2772            continue;
2773        }
2774        match account_deletion_repo
2775            .erasure_recorded(
2776                tenant_id,
2777                user_id,
2778                &erasure.pseudonym,
2779                USER_PSEUDONYMIZED_ACTION,
2780            )
2781            .await
2782        {
2783            Ok(true) => {}
2784            Ok(false) => {
2785                write_audit_with_dead_letter(
2786                    audit_repo,
2787                    CreateAuditLogEntry {
2788                        tenant_id,
2789                        actor_id: user_id,
2790                        actor_type: ActorType::System,
2791                        action: USER_PSEUDONYMIZED_ACTION.into(),
2792                        resource_id: None,
2793                        outcome: AuditOutcome::Success,
2794                        ip_address: None,
2795                        metadata: Some(serde_json::json!({
2796                            "pseudonym": erasure.pseudonym,
2797                            "certificates_revoked": null,
2798                            "recovered": true,
2799                        })),
2800                    },
2801                )
2802                .await;
2803                recorded += 1;
2804                tracing::warn!(
2805                    %tenant_id,
2806                    %user_id,
2807                    "recorded an erasure whose unit was cut before its record (#689)"
2808                );
2809                work.step(UnitStep::Recorded(user_id)).await;
2810            }
2811            Err(e) => {
2812                failed += 1;
2813                tracing::warn!(
2814                    error = %e,
2815                    %tenant_id,
2816                    %user_id,
2817                    "could not check an owed erasure record; the next sweep retries it"
2818                );
2819                claim.release().await;
2820                continue;
2821            }
2822        }
2823        claim.settle().await;
2824    }
2825    if failed > 0 {
2826        return Err(AxiamError::Internal(format!(
2827            "{failed} owed erasure record(s) could not be finished"
2828        )));
2829    }
2830    Ok(recorded)
2831}

2834/// its proof unless it has one, then the second revocation.
2835async fn finish_owed_erasure<EP, CR>(

2837    cert_repo: &CR,

2846        erasure_proof_repo,

2848        user_id,

2876/// One due account's erasure under an exclusive claim (#738): claim it at the
2877/// claimant's own `now`, run `erasure` — the unit's steps from the first
2878/// destructive one to the record — and release the claim unless the account
2879/// was erased.
2880///
2881/// `Ok(false)`, with nothing run, when the account is no longer due (its
2882/// deletion was cancelled since the sweep's read, or another unit erased it)
2883/// or another unit holds a claim on it younger than `lease`: two sweeps that
2884/// read one due account erase it once, and the one that skips it reports no
2885/// error. Otherwise the outcome is `erasure`'s. The claim is released, awaited,
2886/// on every outcome but `Ok(true)`, and by its guard's drop if this future is
2887/// dropped — a unit cut at the teardown's bound, or a panic — so the next sweep
2888/// takes the account at once; the release changes nothing once the
2889/// anonymization has run.
2890///
2891/// A free function, and public, for the reason [`run_erasure_pipeline`] is:
2892/// `axiam-server`'s integration tests run two sweeps' units against one store.
2893///
2894/// # Errors
2895///
2896/// A datastore failure of the claim (nothing ran), or `erasure`'s.
2897pub async fn purge_under_claim<C, F>(
2898    deletions: &SurrealAccountDeletionRepository<C>,
2899    tenant_id: Uuid,
2900    user_id: Uuid,
2901    sweep_now: chrono::DateTime<Utc>,
2902    lease: chrono::TimeDelta,
2903    erasure: F,
2904) -> Result<bool, AxiamError>
2905where
2906    C: Connection,
2907    F: std::future::Future<Output = Result<bool, AxiamError>>,
2908{
2909    // The claimant's `now`, not the sweep's: the stamp starts the lease, and a
2910    // sweep that reaches its last account long after it read them must not
2911    // stamp a claim that is already old. Never earlier than the sweep's, so an
2912    // account due at the sweep's read is due here too.
2913    let stamp = Utc::now().max(sweep_now);
2914    match deletions
2915        .claim_for_purge(tenant_id, user_id, stamp, lease)
2916        .await?
2917    {
2918        PurgeClaimOutcome::Claimed => {}
2919        PurgeClaimOutcome::NotDue => {
2920            tracing::info!(
2921                %user_id,
2922                %tenant_id,
2923                "purge skipped: the account is no longer due (its deletion was cancelled, or \
2924                 another unit erased it)"
2925            );
2926            return Ok(false);
2927        }
2928        PurgeClaimOutcome::Held => {
2929            tracing::info!(
2930                %user_id,
2931                %tenant_id,
2932                "purge skipped: another unit holds the account's erasure"
2933            );
2934            return Ok(false);
2935        }
2936    }
2937    let claim = ErasureClaim {
2938        deletions: deletions.clone(),
2939        tenant_id,
2940        user_id,
2941        stamp,
2942        held: true,
2943    };
2944    let erased = erasure.await;
2945    match erased {
2946        Ok(true) => claim.settle(),
2947        _ => claim.release().await,
2948    }
2949    erased
2950}

3009    /// The purge of a claimed account: steps (a)…(h), as
3010    /// [`Self::purge`] lists them.

3012        self,
3013        user: axiam_core::models::user::User,
3014        pepper: [u8; 32],

3016    ) -> Result<bool, AxiamError> {
3017        // Steps (a)…(b4) are idempotent and leave the account due: given up
3018        // where they are once the process is stopping.
3019        let Some(certificates_revoked) = work.until_stopped(self.prepare(&user)).await else {
3020            return Ok(false);

3023        if work.stopping() {

3060    /// Steps (a)…(b4): sessions, federation links, credentials, the
3061    /// authorization graph and the account's certificates. Returns how many
3062    /// certificates were revoked.
3063    async fn prepare(&self, user: &axiam_core::models::user::User) -> Result<u64, AxiamError> {
3064        let (user_id, tenant_id) = (user.id, user.tenant_id);
3065        // (a) Revoke all sessions and OAuth2 refresh tokens, then delete the
3066        // refresh-token rows (#753): they name the user, the client and the
3067        // device key. Both are idempotent, so a re-entered erasure is safe.
3068        self.auth_svc
3069            .revoke_all_sessions(tenant_id, user_id)
3070            .await?;
3071        erase_oauth2_refresh_tokens(self.refresh_token_repo.as_ref(), tenant_id, user_id).await?;
3072
3073        // (b) Hard-delete federation identity links.
3074        match self
3075            .federation_link_repo
3076            .get_by_user_id(tenant_id, user_id)
3077            .await
3078        {
3079            Ok(links) => {
3080                for link in links {
3081                    if let Err(e) = self.federation_link_repo.delete(tenant_id, link.id).await {
3082                        tracing::warn!(
3083                            error = %e,
3084                            %tenant_id,
3085                            link_id = %link.id,
3086                            "cleanup: failed to delete expired federation link; will retry next cycle"
3087                        );
3088                    }
3089                }
3090            }
3091            Err(e) => {
3092                tracing::warn!(error = ?e, user_id = %user_id, "failed to list federation links for purge");
3093            }
3094        }
3095
3096        // (b2) Hard-delete stored credential material (SEC-056). Done with `?`
3097        // (fail-closed): erasure must not be certified while a user's passkeys or
3098        // password hashes remain. A transient failure aborts the purge and it is
3099        // retried next sweep — every step here is idempotent.
3100        let webauthn_creds = self.webauthn_repo.list_by_user(tenant_id, user_id).await?;
3101        for cred in webauthn_creds {
3102            self.webauthn_repo.delete(tenant_id, cred.id).await?;
3103        }
3104        // Prune the entire password-history chain (keep_count = 0 deletes all).
3105        self.password_history_repo
3106            .prune(tenant_id, user_id, 0)
3107            .await?;
3108
3109        // (b3) Prune authorization-graph edges so the anonymized row is left as
3110        // an isolated tombstone — no dangling `member_of` / `has_role` relations
3111        // skewing group-member queries or leaving live authorization paths.
3112        // Remove group memberships first so the subsequent role pass only sees
3113        // the user's *direct* has_role edges (inherited ones vanish with the
3114        // membership). Fail-closed (`?`): the tombstone must not be created while
3115        // graph edges survive; every delete is idempotent and retried on failure.
3116        let groups = self.group_repo.get_user_groups(tenant_id, user_id).await?;
3117        for group in groups {
3118            self.group_repo
3119                .remove_member(tenant_id, user_id, group.id)
3120                .await?;
3121        }
3122        remove_erased_users_roles(
3123            self.role_repo.as_ref(),
3124            self.permission_repo.as_ref(),
3125            self.audit_repo.as_ref(),
3126            self.subject_flush.as_ref(),
3127            tenant_id,
3128            user_id,
3129        )
3130        .await?;
3131
3132        // (b4) Revoke the account's `User` certificates (#533) and clear their
3133        // `metadata` (R101W1-05), while its username and email — which the
3134        // convention for a certificate issued without a `user_id` reads — are
3135        // still on the row. Fail-closed, like the steps above; idempotent (an
3136        // already revoked row is not matched, a cleared one is not rewritten).
3137        revoke_erased_users_certificates(self.cert_repo.as_ref(), user).await
3138    }

3399    /// Run the cleanup loop until a shutdown signal is received.
3400    ///
3401    /// Never returns `Err` — all sweep errors are logged at `warn` level and
3402    /// the loop continues (T-04-36).
3403    pub async fn run(mut self) -> Result<(), AxiamError> {
3404        let mut ticker = tokio::time::interval(self.interval);
3405        // Skip ticks that were missed while the sweep was running to prevent
3406        // catch-up storms after a pause (T-04-35).
3407        ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
3408
3409        loop {
3410            // A stop that arrived during a tick ends the loop here, whatever the
3411            // ticker says: the select below may otherwise pick a tick that is
3412            // already due over the shutdown change.
3413            if self.stop_requested() {
3414                tracing::info!("cleanup task received shutdown signal");
3415                return Ok(());
3416            }
3417            tokio::select! {
3418                _ = ticker.tick() => {
3419                    self.tick().await;
3420                }
3421                changed = self.shutdown.changed() => {
3422                    if changed.is_ok() && *self.shutdown.borrow() {
3423                        tracing::info!("cleanup task received shutdown signal");
3424                        return Ok(());
3425                    }
3426                }
3427            }
3428        }
3429    }
