161/// One pass of the `scim_reconcile` job, as a sweep the scheduler can record.
162///
163/// A free function, and public, for the reason [`sweep_directories`] is one.
164/// The scheduler ticks far more often than a target is due: at every tick this
165/// walks the enabled targets and tries the claim, which succeeds for a target
166/// whose last run is older than 24 hours and only on one replica; everything
167/// else is a skipped target and costs a conditional write that matches nothing.
168///
169/// * `Ok(n)` is the number of targets whose run this pass made.
170/// * A target whose run could not do all of its work (the broker refused a
171///   reference, the downstream could not be read, AXIAM's datastore failed)
172///   makes the sweep fail, with a count and no target's data. The other targets
173///   still ran.
174/// * A shutdown signal stops the pass between targets.
175///
176/// # Errors
177///
178/// [`AxiamError::Internal`] when the enabled targets could not be listed or a
179/// run was incomplete, as described above.
180pub async fn sweep_scim_reconciliation(
181    reconciliation: &dyn axiam_scim::outbound::ScimReconciliation,
182    shutdown: &watch::Receiver<bool>,
183) -> Result<u64, AxiamError> {
184    let should_stop = || *shutdown.borrow();
185    let sweep = reconciliation.run_due(&should_stop).await.map_err(|_| {
186        AxiamError::Internal("SCIM reconciliation could not list the enabled targets".into())
187    })?;
188    if sweep.failed > 0 {
189        return Err(AxiamError::Internal(format!(
190            "SCIM reconciliation was incomplete for {} of {} target(s) run",
191            sweep.failed, sweep.reconciled
192        )));
193    }
194    Ok(sweep.reconciled)
195}

201/// One pass of the directory sync job, as a sweep the scheduler can record.
202///
203/// A free function, and public, for the reason [`run_erasure_pipeline`] is one:
204/// the loop that calls it is awkward to construct in a test, and the conversion
205/// of the job's outcome into job health is the part worth testing directly.
206///
207/// * `Ok(n)` is the number of accounts the pass changed (deactivated, updated or
208///   re-mapped); a pass that found nothing to do is `Ok(0)`.
209/// * **Any tenant's failure makes the sweep fail** — an unreachable directory, a
210///   refused search, the safety valve — with a message that names how many
211///   tenants failed and the fixed tag of each, never a tenant's data. The other
212///   tenants still ran; one tenant's outage is not another's.
213/// * A shutdown signal abandons the pass (`Ok(0)`): every step the job takes is
214///   idempotent or a compare-and-set, and the next process starts it again.
215///
216/// # Errors
217///
218/// [`AxiamError::Internal`] when the pass failed, as described above.
219pub async fn sweep_directories<C: Connection + Send + Sync + 'static>(
220    sync: &DirectorySyncJob<C>,
221    mut shutdown: watch::Receiver<bool>,
222) -> Result<u64, AxiamError> {
223    let outcome = tokio::select! {
224        // A shutdown request wins over starting work: polled first.
225        biased;
226        _ = async {
227            // Resolves on a shutdown request; a dropped sender never does.
228            loop {
229                if *shutdown.borrow() {
230                    return;
231                }
232                if shutdown.changed().await.is_err() {
233                    std::future::pending::<()>().await;
234                }
235            }
236        } => {
237            tracing::info!("directory sync abandoned: the process is shutting down");
238            return Ok(0);
239        }
240        outcome = sync.run_due() => outcome,
241    };
242    match outcome {
243        Ok(summary) => match summary.failure_message() {
244            Some(message) => Err(AxiamError::Internal(message)),
245            None => Ok(summary.changed()),
246        },
247        Err(error) => Err(AxiamError::Internal(format!(
248            "directory sync could not list the tenants' directories ({})",
249            error.tag()
250        ))),
251    }
252}

258/// Whether a self-registered client is due to be swept (T21.4).
259///
260/// A free function, and public, for the reason [`run_erasure_pipeline`] is
261/// one: the loop that calls it is awkward to construct in a test, and the
262/// decision it makes — *delete somebody's client registration* — is the part
263/// worth testing directly.
264///
265/// The clock it reads is `last_authorized_at` when the client has ever been
266/// authorized and `created_at` when it has not.
267///
268/// `DcrSweepWindow::days == 0` is never due. Zero means "never sweep", which
269/// an operator who prunes out of band may legitimately want, and reading it as
270/// "sweep everything immediately" would delete a tenant's whole client table
271/// on the next tick.
272///
273/// # The second clock (T21.8 / MCP-05)
274///
275/// A row that has **never** been authorized, in a tenant whose effective
276/// `dynamic_registration` is `anonymous`, is measured against
277/// [`axiam_core::models::settings::DCR_UNAUTHORIZED_CLIENT_TTL_SECS`] instead
278/// — an hour. That constant's own documentation says why it is a constant and
279/// what promoting it to a tenant setting would cost.
280///
281/// Those two conditions are the whole of the change, and each is load-bearing.
282///
283/// **Never authorized.** The thirty-day window is sized, in
284/// `DEFAULT_DCR_UNUSED_CLIENT_TTL_DAYS`'s own doc comment, for "a client
285/// somebody uses monthly". A client registered and never authorized is not
286/// that client: every MCP client this phase exists to serve authorizes within
287/// seconds of registering, because registration is the first step of the same
288/// flow. One TTL served two situations that have nothing in common, which is
289/// what made `dcr_max_clients` an availability budget a stranger could spend
290/// for a month. The sweeper could always tell the two apart — the row carries
291/// `last_authorized_at: None` — so the fix is a second clock, not a bigger
292/// quota.
293///
294/// **`anonymous` only.** In `initial_access_token` mode the row exists because
295/// an administrator minted a handle for it and somebody redeemed it. There is
296/// no unauthenticated exposure to bound, and the thirty-day clock is the right
297/// one: an operator who hands somebody a registration token on Friday should
298/// not find the registration gone on Monday.
299///
300/// Every tenant on `disabled` or `initial_access_token` — which is every
301/// tenant that did not opt into `anonymous` behind D3 — takes byte for byte
302/// the path it took before T21.8.
303pub fn dcr_client_is_due_for_sweep(
304    client: &axiam_core::models::oauth2_client::OAuth2Client,
305    ttl: DcrSweepWindow,
306    now: chrono::DateTime<Utc>,
307) -> bool {
308    if client.last_authorized_at.is_none() && ttl.mode == DynamicRegistrationMode::Anonymous {
309        return now - client.created_at
310            > chrono::Duration::seconds(i64::from(DCR_UNAUTHORIZED_CLIENT_TTL_SECS));
311    }
312    if ttl.days == 0 {
313        return false;
314    }
315    now - dcr_client_last_seen(client) > chrono::Duration::days(i64::from(ttl.days))
316}

341/// When a `cimd` row was last any use to anybody — T21.8 / MCP-04.
342///
343/// `max(updated_at, last_authorized_at, created_at)`, and each of the three is
344/// load-bearing.
345///
346/// **`updated_at` is the one that matters**, and it is already the stamp
347/// #470 proposed adding a column for. `materialise_if_cimd` calls
348/// `upsert_cimd_client` after *every* successful resolve, and a resolve
349/// returns from the in-memory document cache on a hit — so the upsert runs
350/// whether or not a fetch happened, on authorize, on token and on PAR. The
351/// `UPDATE` arm sets `updated_at = time::now()`. A `cimd` row's `updated_at`
352/// is therefore "last presented", with a resolution of one request, and no
353/// migration is needed to read it.
354///
355/// `last_authorized_at` is read beside it because `touch_last_authorized`
356/// guards on `managed_by != 'admin'` rather than `== 'dcr'`, so it is stamped
357/// on `cimd` rows too. Taking the maximum of the three cannot pick a stamp
358/// older than the row's real last use, whichever of them the write path
359/// happened to move.
360pub fn cimd_client_last_seen(
361    client: &axiam_core::models::oauth2_client::OAuth2Client,
362) -> chrono::DateTime<Utc> {
363    client
364        .updated_at
365        .max(client.last_authorized_at.unwrap_or(client.created_at))
366        .max(client.created_at)
367}

369/// Whether a `cimd` shadow row has gone unseen for longer than its tenant's
370/// TTL (T21.8 / MCP-04).
371///
372/// Shares `dcr_unused_client_ttl_days` with the `dcr` sweep, on T21.5
373/// amendment 4's precedent: `dcr_allowed_scopes` governs both mechanisms and

378///
379/// `0` is never due, for the same reason it is not in the `dcr` arm.
380/// The second clock does **not** apply here, and that is deliberate: a `cimd`
381/// row is never "registered and never authorized" in the sense T21.8's window
382/// is about. It exists because somebody presented a document AXIAM fetched, so

388    ttl: DcrSweepWindow,
389    now: chrono::DateTime<Utc>,
390) -> bool {
391    if ttl.days == 0 {
392        return false;
393    }
394    now - cimd_client_last_seen(client) > chrono::Duration::days(i64::from(ttl.days))
395}

442/// Delete externally registered clients that have gone unseen for longer than
443/// their tenant's `dcr_unused_client_ttl_days` (T21.4, widened to `cimd` by
444/// T21.8 / MCP-04).
445///
446/// One `managed_by` per call, so `/health/jobs` can distinguish the two
447/// sweeps: they delete different things for different reasons and an operator
448/// debugging one should not have to read the other's counter.
449///
450/// # What it will not touch
451///
452/// Never `admin`. An administrator's client is never swept, however long it
453/// sits unused, because somebody decided it should exist and nothing here is
454/// entitled to reverse that. The repository query filters on the value rather
455/// than on "not admin", so a fourth provenance added later is opted **in** by
456/// somebody writing it down.
457///
458/// # Why `cimd` is swept now, when T21.4 argued it should not be
459///
460/// T21.4's reasoning, which this function used to carry: a CIMD shadow row is
461/// a cache of a document the client publishes, so deleting it would be
462/// re-materialised on the next request and the TTL would mean nothing.
463///
464/// That is right about TTL semantics and says nothing about storage, which is
465/// what MCP-04 is about: a cache that is never evicted is not a cache. An
466/// inert row is still a row — listed on the OAuth2 clients page, counted in
467/// every `list_all_by_managed_by` this function runs, and a permanent write a
468/// stranger made at the cost of one unauthenticated request. And eviction on
469/// last-seen is *consistent* with the old argument rather than against it: a
470/// row deleted while its document is still published is re-materialised on the
471/// next request, which is exactly what a cache should do. What it must not do
472/// is delete a row that is still in use, which is what
473/// [`cimd_client_last_seen`] is for — every resolve moves `updated_at`, so a
474/// document presented once a day is never due under a 30-day TTL.
475///
476/// # Per-tenant TTL, resolved per row
477///
478/// The window is a tenant setting, so each row's tenant is resolved and its
479/// effective settings read. Deployment-wide sweeps elsewhere in this file
480/// (audit retention) deliberately do **not** work this way, and the difference
481/// is who owns the decision: audit retention is a property of the datastore an
482/// operator is responsible for, where this is a property of the tenant's
483/// relationship with the clients it lets register.
484///
485/// The settings read is cached per tenant for the duration of one sweep —
486/// `dcr_max_clients` bounds the rows per tenant, so a hundred clients in one
487/// tenant is one read rather than a hundred.
488///
489/// A tenant whose settings cannot be read is skipped: the fail-closed
490/// direction for a sweep that **deletes** is to delete nothing.
491pub async fn sweep_unused_external_clients<CR, TR, SR>(
492    client_repo: &CR,
493    tenant_repo: &TR,
494    settings_repo: &SR,
495    managed_by: axiam_core::models::oauth2_client::ManagedBy,
496    now: chrono::DateTime<Utc>,
497) -> Result<u64, AxiamError>
498where
499    CR: axiam_core::repository::OAuth2ClientRepository,
500    TR: TenantRepository,
501    SR: axiam_core::repository::SettingsRepository,
502{
503    use axiam_core::models::oauth2_client::ManagedBy;
504    use axiam_core::repository::SettingsRepository;
505
506    type DuePredicate = fn(
507        &axiam_core::models::oauth2_client::OAuth2Client,
508        DcrSweepWindow,
509        chrono::DateTime<Utc>,
510    ) -> bool;
511    let (job, is_due): (&str, DuePredicate) = match managed_by {
512        ManagedBy::Dcr => ("dcr_unused_clients", dcr_client_is_due_for_sweep),
513        ManagedBy::Cimd => ("cimd_unused_clients", cimd_client_is_due_for_sweep),
514        // An administrator's client is never swept. Answered here rather than
515        // by trusting every caller, because the argument for it is a security
516        // one and belongs next to the query.
517        ManagedBy::Admin => return Ok(0),
518    };
519
520    let clients = client_repo.list_all_by_managed_by(managed_by).await?;
521    if clients.is_empty() {
522        return Ok(0);
523    }
524
525    let mut ttl_by_tenant: std::collections::HashMap<Uuid, Option<DcrSweepWindow>> =
526        std::collections::HashMap::new();
527    let mut removed = 0u64;
528
529    for client in clients {
530        let ttl = match ttl_by_tenant.get(&client.tenant_id) {
531            Some(cached) => *cached,
532            None => {
533                let resolved = match tenant_repo.get_by_id(client.tenant_id).await {
534                    Ok(tenant) => SettingsRepository::get_effective_settings(
535                        settings_repo,
536                        tenant.organization_id,
537                        client.tenant_id,
538                    )
539                    .await
540                    .ok()
541                    .map(|s| DcrSweepWindow {
542                        days: s.oidc.dcr_unused_client_ttl_days,
543                        mode: s.oidc.dynamic_registration,
544                    }),
545                    Err(_) => None,
546                };
547                ttl_by_tenant.insert(client.tenant_id, resolved);
548                resolved
549            }
550        };
551        // `None` is an unreadable tenant or an unreadable settings row.
552        let Some(ttl) = ttl else {
553            continue;
554        };
555        if !is_due(&client, ttl, now) {
556            continue;
557        }
558
559        if let Err(e) = client_repo.delete(client.tenant_id, client.id).await {
560            // One failure must not stop the sweep: the next row may be a
561            // different tenant entirely.
562            tracing::warn!(
563                job,
564                error = %e,
565                client_id = %client.client_id,
566                "could not delete an unused externally registered client"
567            );
568            continue;
569        }
570        removed += 1;
571        tracing::info!(
572            job,
573            tenant_id = %client.tenant_id,
574            client_id = %client.client_id,
575            ttl_days = ttl.days,
576            ever_authorized = client.last_authorized_at.is_some(),
577            mode = %ttl.mode,
578            "deleted an externally registered client that has gone unseen within its \
579             tenant's TTL"
580        );
581    }
582
583    Ok(removed)
584}

590/// Run the GDPR erasure pipeline for a single user: pseudonymize audit actor
591/// references, anonymize the user row, then write the erasure proof
592/// STRICTLY LAST.
593///
594/// Extracted as a free function generic over the three repo traits it needs
595/// (rather than a `CleanupTask` method) so a unit test can inject a
596/// synthetic failing `AuditLogRepository` double without depending on
597/// `CleanupTask`'s concrete `Arc<SurrealXxxRepository<C>>` fields — `pub` so
598/// `axiam-server`'s integration tests (which link this crate's library
599/// target) can call it directly.
600///
601/// Ordering is a hard security invariant (D-03a, SECHRD-06 / T-25-13):
602/// - `pseudonymize_actor` is now FATAL (`?`, no swallow-and-continue). A
603///   failed audit-actor scrub must abort the erasure — no PII-bearing step
604///   may ever be silently skipped (RESEARCH Pitfall 2).
605/// - `anonymize_user` runs BEFORE the proof (not after). It is the ONLY
606///   step that clears the user's `deletion_pending` flag that
607///   `find_due_for_purge` selects on, so if it never runs (an earlier step
608///   failed and aborted via `?`), the user remains re-selectable for a
609///   retry (RESEARCH Assumption A3).
610/// - `erasure_proof_repo.create` is the LITERAL LAST statement. It only
611///   fires once every PII-bearing step above has succeeded — a proof must
612///   never certify an erasure that did not fully happen (Pitfall 3). The DB
613///   UNIQUE index on `(tenant_id, user_id)` (plan 25-04) makes a retried
614///   erasure's duplicate proof insert an idempotent rejection (D-03b), not
615///   a silent overwrite.
616pub async fn run_erasure_pipeline<A, EP, U>(
617    audit_repo: &A,
618    erasure_proof_repo: &EP,
619    user_repo: &U,
620    tenant_id: Uuid,
621    user_id: Uuid,
622    pseudonym: &str,
623    email_hash: &str,
624) -> Result<(), AxiamError>
625where
626    A: AuditLogRepository,
627    EP: ErasureProofRepository,
628    U: UserRepository,
629{
630    erasure_steps(
631        audit_repo,
632        erasure_proof_repo,
633        user_repo,
634        tenant_id,
635        user_id,
636        pseudonym,
637        email_hash,
638        None,
639    )
640    .await
641}

643/// The three steps of [`run_erasure_pipeline`], with the SSF report (G-5, D-52)
644/// between the second and the third.
645///
646/// The report follows the anonymization because that is the write that makes the
647/// account gone — whether or not the erasure proof is then written. A proof that
648/// fails to write leaves an anonymized account that no sweep selects again, so a
649/// report that waited for the proof would never be sent.
650#[allow(clippy::too_many_arguments)]
651async fn erasure_steps<A, EP, U>(
652    audit_repo: &A,
653    erasure_proof_repo: &EP,
654    user_repo: &U,
655    tenant_id: Uuid,
656    user_id: Uuid,
657    pseudonym: &str,
658    email_hash: &str,
659    report: Option<(&dyn SsfSystemAccountSink, axiam_core::models::user::User)>,
660) -> Result<(), AxiamError>
661where
662    A: AuditLogRepository,
663    EP: ErasureProofRepository,
664    U: UserRepository,
665{
666    // FATAL now (was: `if let Err(e) = ... { tracing::warn!(...) }`).
667    audit_repo
668        .pseudonymize_actor(tenant_id, user_id, pseudonym)
669        .await?;
670
671    // Clears `deletion_pending` — the re-selection anchor. Must run before
672    // the proof so a failure here (unreachable in practice since this is
673    // now the second of three fallible steps) still leaves the user due.
674    user_repo
675        .anonymize_user(tenant_id, user_id, email_hash, pseudonym)
676        .await?;
677
678    // G-5 (D-52): the account is gone; tell the receivers what it was.
679    if let Some((sink, user)) = report {
680        sink.account_purged(tenant_id, &user).await;
681    }
682
683    // Written STRICTLY LAST — only reached once every step above succeeded.
684    erasure_proof_repo
685        .create(CreateErasureProof {
686            pseudonym: pseudonym.to_string(),
687            tenant_id,
688            user_id,
689            erased_at: Utc::now(),
690        })
691        .await?;
692
693    Ok(())
694}

696/// [`run_erasure_pipeline`] and an SSF `account-purged` for the erased account
697/// (G-5, D-52), sent once the account has been anonymized.
698///
699/// The account is read **before** the pipeline writes anything: the erasure
700/// replaces the address, and the event's subject (`iss_sub`, or the address on
701/// an `email` stream) is resolved from what the account was. Nothing is read
702/// when no sink is attached or it is inactive. A failed erasure that did not get
703/// as far as anonymizing reports nothing: the account is still there and still
704/// due, and the next sweep tries again.
705///
706/// # Errors
707///
708/// Those of [`run_erasure_pipeline`]; the sink never adds one.
709#[allow(clippy::too_many_arguments)]
710pub async fn run_erasure_pipeline_reporting<A, EP, U>(
711    audit_repo: &A,
712    erasure_proof_repo: &EP,
713    user_repo: &U,
714    tenant_id: Uuid,
715    user_id: Uuid,
716    pseudonym: &str,
717    email_hash: &str,
718    sink: Option<&dyn SsfSystemAccountSink>,
719) -> Result<(), AxiamError>
720where
721    A: AuditLogRepository,
722    EP: ErasureProofRepository,
723    U: UserRepository,
724{
725    let report = match sink {
726        Some(sink) if sink.is_active() => user_repo
727            .get_by_id(tenant_id, user_id)
728            .await
729            .ok()
730            .map(|user| (sink, user)),
731        _ => None,
732    };
733    erasure_steps(
734        audit_repo,
735        erasure_proof_repo,
736        user_repo,
737        tenant_id,
738        user_id,
739        pseudonym,
740        email_hash,
741        report,
742    )
743    .await
744}

1403    async fn purge_single_user_inner(
1404        &self,
1405        user_id: Uuid,
1406        tenant_id: Uuid,
1407        pepper: [u8; 32],
1408    ) -> Result<(), AxiamError> {
1409        // (a) Revoke all sessions and OAuth2 refresh tokens.
1410        self.auth_svc
1411            .revoke_all_sessions(tenant_id, user_id)
1412            .await?;
1413
1414        // (b) Hard-delete federation identity links.
1415        match self
1416            .federation_link_repo
1417            .get_by_user_id(tenant_id, user_id)
1418            .await
1419        {
1420            Ok(links) => {
1421                for link in links {
1422                    if let Err(e) = self.federation_link_repo.delete(tenant_id, link.id).await {
1423                        tracing::warn!(
1424                            error = %e,
1425                            %tenant_id,
1426                            link_id = %link.id,
1427                            "cleanup: failed to delete expired federation link; will retry next cycle"
1428                        );
1429                    }
1430                }
1431            }
1432            Err(e) => {
1433                tracing::warn!(error = ?e, user_id = %user_id, "failed to list federation links for purge");
1434            }
1435        }
1436
1437        // (b2) Hard-delete stored credential material (SEC-056). Done with `?`
1438        // (fail-closed): erasure must not be certified while a user's passkeys or
1439        // password hashes remain. A transient failure aborts the purge and it is
1440        // retried next sweep — every step here is idempotent.
1441        let webauthn_creds = self.webauthn_repo.list_by_user(tenant_id, user_id).await?;
1442        for cred in webauthn_creds {
1443            self.webauthn_repo.delete(tenant_id, cred.id).await?;
1444        }
1445        // Prune the entire password-history chain (keep_count = 0 deletes all).
1446        self.password_history_repo
1447            .prune(tenant_id, user_id, 0)
1448            .await?;
1449
1450        // (b3) Prune authorization-graph edges so the anonymized row is left as
1451        // an isolated tombstone — no dangling `member_of` / `has_role` relations
1452        // skewing group-member queries or leaving live authorization paths.
1453        // Remove group memberships first so the subsequent role pass only sees
1454        // the user's *direct* has_role edges (inherited ones vanish with the
1455        // membership). Fail-closed (`?`): the tombstone must not be created while
1456        // graph edges survive; every delete is idempotent and retried on failure.
1457        let groups = self.group_repo.get_user_groups(tenant_id, user_id).await?;
1458        for group in groups {
1459            self.group_repo
1460                .remove_member(tenant_id, user_id, group.id)
1461                .await?;
1462        }
1463        let assignments = self
1464            .role_repo
1465            .get_user_role_assignments(tenant_id, user_id)
1466            .await?;
1467        for a in assignments {
1468            self.role_repo
1469                .unassign_from_user(tenant_id, user_id, a.role.id, a.resource_id)
1470                .await?;
1471        }
1472
1473        // (c) Compute deterministic pseudonym (keyed HMAC-SHA256, D-02).
1474        let pseudonym = gdpr_pseudonym(&pepper, tenant_id, user_id);
1475
1476        // Derive email hash for anonymize_user from tenant+user IDs (original email
1477        // is no longer accessible at purge time — user already marked deletion_pending).
1478        use sha2::{Digest, Sha256};
1479        let mut h = Sha256::new();
1480        h.update(tenant_id.as_bytes());
1481        h.update(user_id.as_bytes());
1482        let email_hash = hex::encode(h.finalize());
1483
1484        // (d)/(e)/(g) Run the erasure pipeline: pseudonymize audit actor refs
1485        // (now FATAL, was swallowed) -> anonymize the user row -> write the
1486        // erasure proof STRICTLY LAST (SECHRD-06 / D-03a, T-25-13). A failure
1487        // anywhere in this call propagates with `?` and aborts the purge:
1488        // `anonymize_user` (the only step that clears `deletion_pending`)
1489        // never ran, so the user stays due for a retry via
1490        // `find_due_for_purge`, and NO erasure proof is ever written for an
1491        // incomplete erasure.
1492        run_erasure_pipeline_reporting(
1493            self.audit_repo.as_ref(),
1494            self.erasure_proof_repo.as_ref(),
1495            self.user_repo.as_ref(),
1496            tenant_id,
1497            user_id,
1498            &pseudonym,
1499            &email_hash,
1500            self.ssf_sink.as_deref(),
1501        )
1502        .await?;
1503
1504        // (f) Mark the account_deletion row as completed (lookup by user_id).
1505        // Run AFTER the erasure pipeline succeeds — if the pipeline failed
1506        // above, this is never reached and the deletion row stays pending
1507        // for the next retry too.
1508        match self
1509            .account_deletion_repo
1510            .find_pending_by_user_id(tenant_id, user_id)
1511            .await
1512        {
1513            Ok(Some(deletion)) => {
1514                if let Err(e) = self
1515                    .account_deletion_repo
1516                    .mark_completed(tenant_id, deletion.id)
1517                    .await
1518                {
1519                    tracing::warn!(
1520                        error = %e,
1521                        %tenant_id,
1522                        deletion_id = %deletion.id,
1523                        "cleanup: failed to mark account_deletion completed"
1524                    );
1525                }
1526            }
1527            Ok(None) => {
1528                tracing::debug!(user_id = %user_id, "no pending account_deletion row found at purge time");
1529            }
1530            Err(e) => {
1531                tracing::warn!(error = ?e, user_id = %user_id, "failed to find account_deletion row");
1532            }
1533        }
1534
1535        // (h) Emit gdpr.user_pseudonymized audit event. actor_id is the
1536        // erased subject's own (now-anonymized) row id — a real, resolvable
1537        // identifier rather than a fabricated nil UUID (Task 1 fix); the row
1538        // still exists (anonymize-in-place preserves referential integrity),
1539        // so this is a legitimate audit-trail reference, not PII leakage.
1540        // A DB-write failure here is dead-lettered to BOTH an append-only
1541        // file AND a structured audit event (SECHRD-12 / D-02, T-24-61) —
1542        // this legally-significant record must never be silently lost.
1543        write_erasure_audit_with_dlq(
1544            self.audit_repo.as_ref(),
1545            CreateAuditLogEntry {
1546                tenant_id,
1547                actor_id: user_id,
1548                actor_type: ActorType::System,
1549                action: "gdpr.user_pseudonymized".into(),
1550                resource_id: None,
1551                outcome: AuditOutcome::Success,
1552                ip_address: None,
1553                metadata: Some(serde_json::json!({
1554                    "pseudonym": pseudonym,
1555                })),
1556            },
1557        )
1558        .await;
1559
1560        tracing::info!(
1561            pseudonym = %pseudonym,
1562            tenant_id = %tenant_id,
1563            "user purged and pseudonymized"
1564        );
1565
1566        Ok(())
1567    }

1624    /// Process a single queued export job.
1625    async fn process_export_job(
1626        &self,
1627        job_id: Uuid,
1628        tenant_id: Uuid,
1629        user_id: Uuid,
1630        key: [u8; 32],
1631    ) -> Result<(), AxiamError> {
1632        // (a)/(b) Aggregate Art. 15 inventory into one sectioned JSON.
1633        let export_json = self.aggregate_export_data(tenant_id, user_id).await?;
1634        let export_bytes = export_json.to_string().into_bytes();
1635
1636        // (c) Encrypt the blob (D-12).
1637        let (nonce_b64, ct_b64) = encrypt_separate(&key, &export_bytes)
1638            .map_err(|e| AxiamError::Internal(format!("export encrypt failed: {e}")))?;
1639
1640        // (d) Generate single-use 24h download token (D-13).
1641        //
1642        // Deliberately v4, not `new_id()`: this token is the sole bearer
1643        // credential for the export download, so it must stay maximally random.
1644        // See `axiam_core::id` — v7 is for identifiers, never for secrets.
1645        let raw_download_token = Uuid::new_v4().to_string();
1646        let token_hash = {
1647            use sha2::{Digest, Sha256};
1648            let mut h = Sha256::new();
1649            h.update(raw_download_token.as_bytes());
1650            hex::encode(h.finalize())
1651        };
1652        let expires_at = Utc::now() + chrono::Duration::hours(24);
1653
1654        self.export_job_repo
1655            .set_ready(
1656                job_id,
1657                token_hash,
1658                Some(ct_b64),
1659                None, // file_path: stored as DB blob
1660                Some(nonce_b64),
1661                expires_at,
1662            )
1663            .await?;
1664
1665        // Resolve the real org_id from the tenant before enqueuing the mail
1666        // (SECHRD-08 / D-05d) — a Uuid::nil() org_id leaves the mail
1667        // consumer unable to resolve the sending organization, producing
1668        // undeliverable ExportReady mail.
1669        let org_id = match self.tenant_repo.get_by_id(tenant_id).await {
1670            Ok(tenant) => tenant.organization_id,
1671            Err(e) => {
1672                tracing::warn!(
1673                    error = %e,
1674                    %tenant_id,
1675                    "failed to resolve org_id for ExportReady mail — falling back to nil (mail may be undeliverable)"
1676                );
1677                Uuid::nil()
1678            }
1679        };
1680
1681        // (e) Enqueue ExportReady email.
1682        let download_url = format!("/api/v1/account/export/{}", raw_download_token);
1683        let msg = OutboundMailMessage {
1684            mail_type: MailType::ExportReady,
1685            tenant_id,
1686            org_id,
1687            user_id,
1688            to_address: String::new(), // mail consumer resolves from user_id
1689            template_context: serde_json::json!({
1690                "action_url": download_url,
1691                "expiry_time": expires_at.to_rfc3339(),
1692            }),
1693            attempt_count: 0,
1694            enqueued_at: Utc::now(),
1695        };
1696        if let Err(e) = self.mail_publisher.publish(msg).await {
1697            tracing::warn!(error = %e, job_id = %job_id, "failed to enqueue ExportReady mail");
1698        }
1699
1700        tracing::info!(job_id = %job_id, "export job completed");
1701        Ok(())
1702    }

1704    /// Aggregate Art. 15 personal-data inventory for a user.
1705    ///
1706    /// EXCLUDED (D-10): `password_hash`, `mfa_secret`, any token_hash values.
1707    async fn aggregate_export_data(
1708        &self,
1709        tenant_id: Uuid,
1710        user_id: Uuid,
1711    ) -> Result<serde_json::Value, AxiamError> {
1712        // Profile — no password_hash or mfa_secret.
1713        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
1714        let profile = profile_section(&user);
1715
1716        // Consents — propagate errors (CQ-B38 / SEC-056): a section that fails to
1717        // query must fail the whole export job rather than emit a legally
1718        // incomplete Art. 15 inventory. `process_export_job` marks the job Failed.
1719        let consents = self.consent_repo.list_by_user(tenant_id, user_id).await?;
1720        let consents_json: Vec<_> = consents
1721            .iter()
1722            .map(|c| {
1723                serde_json::json!({
1724                    "consent_type": c.consent_type,
1725                    "version": c.version,
1726                    "accepted_at": c.accepted_at,
1727                    "ip_address": c.ip_address,
1728                })
1729            })
1730            .collect();
1731
1732        // Audit entries where this user was the actor — paginated to collect ALL
1733        // entries regardless of volume (CQ-B38 / REQ-14 AC-5).
1734        const AUDIT_PAGE_SIZE: u64 = 1_000;
1735        let mut audit_items = Vec::new();
1736        let mut offset: u64 = 0;
1737        loop {
1738            let page = self
1739                .audit_repo
1740                .list(
1741                    tenant_id,
1742                    AuditLogFilter {
1743                        actor_id: Some(user_id),
1744                        action: None,
1745                        outcome: None,
1746                        resource_id: None,
1747                        from: None,
1748                        to: None,
1749                    },
1750                    Pagination {
1751                        offset,
1752                        limit: AUDIT_PAGE_SIZE,
1753                        search: None,
1754                    },
1755                )
1756                .await?;
1757            let fetched = page.items.len() as u64;
1758            audit_items.extend(page.items);
1759            offset += fetched;
1760            if fetched < AUDIT_PAGE_SIZE {
1761                break;
1762            }
1763        }
1764        let audit_json: Vec<_> = audit_items
1765            .iter()
1766            .map(|e| {
1767                serde_json::json!({
1768                    "action": e.action,
1769                    "outcome": e.outcome,
1770                    "timestamp": e.timestamp,
1771                    "resource_id": e.resource_id,
1772                })
1773            })
1774            .collect();
1775
1776        // Sessions — real, metadata-only rows (D-03c, SECHRD-06). Token/hash
1777        // material is live credential data and must NEVER appear in a GDPR
1778        // export, so `token_hash` is deliberately excluded from the
1779        // projection below. Propagate errors (CQ-B38 / SEC-056 convention):
1780        // a section that fails to query must fail the whole export job.
1781        let sessions = self.session_repo.list_by_user(tenant_id, user_id).await?;
1782        let sessions_json: Vec<_> = sessions
1783            .iter()
1784            .map(|s| {
1785                serde_json::json!({
1786                    "id": s.id,
1787                    "created_at": s.created_at,
1788                    "expires_at": s.expires_at,
1789                    "ip_address": s.ip_address,
1790                    "user_agent": s.user_agent,
1791                })
1792            })
1793            .collect();
1794
1795        // Federation identities — propagate errors (CQ-B38 / SEC-056).
1796        let fed_links = self
1797            .federation_link_repo
1798            .get_by_user_id(tenant_id, user_id)
1799            .await?;
1800        let fed_json: Vec<_> = fed_links
1801            .iter()
1802            .map(|l| {
1803                serde_json::json!({
1804                    "federation_config_id": l.federation_config_id,
1805                    "external_subject": l.external_subject,
1806                    "created_at": l.created_at,
1807                })
1808            })
1809            .collect();
1810
1811        // Role assignments (direct + inherited via groups), incl. resource scope.
1812        let assignments = self
1813            .role_repo
1814            .get_user_role_assignments(tenant_id, user_id)
1815            .await?;
1816        let assignments_json: Vec<_> = assignments
1817            .iter()
1818            .map(|a| {
1819                serde_json::json!({
1820                    "role_id": a.role.id,
1821                    "role_name": a.role.name,
1822                    "is_global": a.role.is_global,
1823                    "resource_id": a.resource_id,
1824                })
1825            })
1826            .collect();
1827
1828        // Group memberships.
1829        let groups = self.group_repo.get_user_groups(tenant_id, user_id).await?;
1830        let groups_json: Vec<_> = groups
1831            .iter()
1832            .map(|g| {
1833                serde_json::json!({
1834                    "group_id": g.id,
1835                    "name": g.name,
1836                    "description": g.description,
1837                })
1838            })
1839            .collect();
1840
1841        // WebAuthn credentials — metadata only; the encrypted `passkey_json`
1842        // secret material is intentionally EXCLUDED (D-10).
1843        let webauthn_creds = self.webauthn_repo.list_by_user(tenant_id, user_id).await?;
1844        let webauthn_json: Vec<_> = webauthn_creds
1845            .iter()
1846            .map(|c| {
1847                serde_json::json!({
1848                    "id": c.id,
1849                    "credential_id": c.credential_id,
1850                    "name": c.name,
1851                    "credential_type": c.credential_type,
1852                    "created_at": c.created_at,
1853                    "last_used_at": c.last_used_at,
1854                })
1855            })
1856            .collect();
1857
1858        let export = serde_json::json!({
1859            "export_metadata": {
1860                "generated_at": Utc::now(),
1861                "tenant_id": tenant_id,
1862                "subject_id": user_id,
1863                "schema_version": "1.0",
1864            },
1865            "profile": profile,
1866            "consents": consents_json,
1867            "sessions": sessions_json, // metadata only; NO token_hash (D-03c)
1868            "mfa": { "enabled": user.mfa_enabled }, // NO mfa_secret
1869            "federation_identities": fed_json,
1870            "assignments": assignments_json,
1871            "group_memberships": groups_json,
1872            "audit_entries": audit_json,
1873            "webauthn_credentials": webauthn_json,
1874        });
1875
1876        Ok(export)
1877    }
1878}
