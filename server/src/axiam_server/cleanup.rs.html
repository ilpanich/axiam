1//! Periodic cleanup task for expired federation rows, GDPR purges, and export jobs.
2//!
3//! SurrealDB v3 does not support native TTL on rows (RESEARCH §7), so this task
4//! periodically sweeps various tables:
5//! - `saml_assertion_replay` and `federation_login_state`: expired rows
6//! - `user`: accounts past their scheduled purge date (D-05/D-06/D-08)
7//! - `export_job`: queued jobs waiting to have their encrypted blob generated (D-12)
8//!
9//! The task shuts down cleanly when the caller sends `true` through the watch
10//! channel (D-09, D-24).

74    federation_link_repo: Arc<SurrealFederationLinkRepository<C>>,

77    role_repo: Arc<SurrealRoleRepository<C>>,

91    export_encryption_key: Option<[u8; 32]>,
92    interval: Duration,
93    /// Audit retention (T-119). `None` = never prune.
94    audit_retention: Option<chrono::Duration>,
95    /// T-39/T-143: the revocation feed's table, when the deployment runs one.
96    ///
97    /// `None` — the default — means the feed is off, no row is ever written,
98    /// and this sweep is a no-op. A `Some` here is not optional in the way
99    /// `audit_retention`'s is: the entries have their own `expires_at` and the
100    /// read path filters on it, so a sweep that never ran would publish a
101    /// truthful document over a table that grows forever. It is a size bound,
102    /// not a correctness one.

104    /// T-129: records each sweep's outcome for `GET /health/jobs`.
105    job_health: crate::job_health::JobHealth,
106    shutdown: watch::Receiver<bool>,
107}

113/// Run the GDPR erasure pipeline for a single user: pseudonymize audit actor
114/// references, anonymize the user row, then write the erasure proof
115/// STRICTLY LAST.
116///
117/// Extracted as a free function generic over the three repo traits it needs
118/// (rather than a `CleanupTask` method) so a unit test can inject a
119/// synthetic failing `AuditLogRepository` double without depending on
120/// `CleanupTask`'s concrete `Arc<SurrealXxxRepository<C>>` fields — `pub` so
121/// `axiam-server`'s integration tests (which link this crate's library
122/// target) can call it directly.
123///
124/// Ordering is a hard security invariant (D-03a, SECHRD-06 / T-25-13):
125/// - `pseudonymize_actor` is now FATAL (`?`, no swallow-and-continue). A
126///   failed audit-actor scrub must abort the erasure — no PII-bearing step
127///   may ever be silently skipped (RESEARCH Pitfall 2).
128/// - `anonymize_user` runs BEFORE the proof (not after). It is the ONLY
129///   step that clears the user's `deletion_pending` flag that
130///   `find_due_for_purge` selects on, so if it never runs (an earlier step
131///   failed and aborted via `?`), the user remains re-selectable for a
132///   retry (RESEARCH Assumption A3).
133/// - `erasure_proof_repo.create` is the LITERAL LAST statement. It only
134///   fires once every PII-bearing step above has succeeded — a proof must
135///   never certify an erasure that did not fully happen (Pitfall 3). The DB
136///   UNIQUE index on `(tenant_id, user_id)` (plan 25-04) makes a retried
137///   erasure's duplicate proof insert an idempotent rejection (D-03b), not
138///   a silent overwrite.
139pub async fn run_erasure_pipeline<A, EP, U>(
140    audit_repo: &A,
141    erasure_proof_repo: &EP,
142    user_repo: &U,
143    tenant_id: Uuid,
144    user_id: Uuid,
145    pseudonym: &str,
146    email_hash: &str,
147) -> Result<(), AxiamError>
148where
149    A: AuditLogRepository,
150    EP: ErasureProofRepository,
151    U: UserRepository,
152{
153    // FATAL now (was: `if let Err(e) = ... { tracing::warn!(...) }`).
154    audit_repo
155        .pseudonymize_actor(tenant_id, user_id, pseudonym)
156        .await?;
157
158    // Clears `deletion_pending` — the re-selection anchor. Must run before
159    // the proof so a failure here (unreachable in practice since this is
160    // now the second of three fallible steps) still leaves the user due.
161    user_repo
162        .anonymize_user(tenant_id, user_id, email_hash, pseudonym)
163        .await?;
164
165    // Written STRICTLY LAST — only reached once every step above succeeded.
166    erasure_proof_repo
167        .create(CreateErasureProof {
168            pseudonym: pseudonym.to_string(),
169            tenant_id,
170            user_id,
171            erased_at: Utc::now(),
172        })
173        .await?;
174
175    Ok(())
176}

178/// The Art. 15 `profile` section of one user's export.
179///
180/// A free function, and tested as one, because its **key set** is half of
181/// T-261's gate: `axiam_core::personal_data::export_keys()` is the declared
182/// answer to "what does a subject get to see", and
183/// `the_profile_section_shows_exactly_the_declared_export_keys` requires this
184/// literal to match it. A column classified as exported and missing here is a
185/// column the subject is never shown — the defect that nearly stranded
186/// `phone_number` and `address`, whose own entry warned that the path is an
187/// explicit field list and that a column not named in it is never exported.
188///
189/// The literal is deliberately **not** derived from the inventory. Two of its
190/// entries are not plain column reads: `id` is the record identifier rather
191/// than a `DEFINE FIELD`, and `phone_number_verified` is a derived boolean
192/// rather than the `phone_number_verified_at` timestamp — matching the claim
193/// the subject would have seen released, since exporting the internal column
194/// name would describe AXIAM's storage rather than the subject's data.
195/// Deriving the section would have to special-case both, which is a worse
196/// thing to maintain than a checked list.
197///
198/// EXCLUDED (D-10): `password_hash`, `mfa_secret`, any token_hash values.
199fn profile_section(user: &axiam_core::models::user::User) -> serde_json::Value {
200    serde_json::json!({
201        "id": user.id,
202        "username": user.username,
203        "email": user.email,
204        "status": user.status,
205        "mfa_enabled": user.mfa_enabled,
206        "phone_number": user.phone_number,
207        "phone_number_verified": user.phone_number_verified_at.is_some(),
208        "address": user.address,
209        "metadata": user.metadata,
210        "created_at": user.created_at,
211        "updated_at": user.updated_at,
212    })
213}

285    /// Run the cleanup loop until a shutdown signal is received.
286    ///
287    /// Never returns `Err` — all sweep errors are logged at `warn` level and
288    /// the loop continues (T-04-36).
289    pub async fn run(mut self) -> Result<(), AxiamError> {
290        let mut ticker = tokio::time::interval(self.interval);
291        // Skip ticks that were missed while the sweep was running to prevent
292        // catch-up storms after a pause (T-04-35).
293        ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
294
295        loop {
296            tokio::select! {
297                _ = ticker.tick() => {
298                    // Existing federation cleanup sweeps.
299                    Self::record(
300                        &self.job_health,
301                        "saml_assertion_replay",
302                        self.replay_repo.cleanup_expired().await,
303                        tracing::Level::DEBUG,
304                    );
305
306                    Self::record(
307                        &self.job_health,
308                        "federation_login_state",
309                        self.state_repo.cleanup_expired().await,
310                        tracing::Level::DEBUG,
311                    );
312
313                    Self::record(
314                        &self.job_health,
315                        "sso_handoff_code",
316                        self.sso_handoff_code_repo.cleanup_expired().await,
317                        tracing::Level::DEBUG,
318                    );
319
320                    // NEW-4: sweep expired AMQP replay-protection nonces.
321                    Self::record(
322                        &self.job_health,
323                        "amqp_nonce_replay",
324                        self.amqp_nonce_repo.cleanup_expired().await,
325                        tracing::Level::DEBUG,
326                    );
327
328                    // GDPR purge sweep (D-01..D-06, D-08).
329                    Self::record(
330                        &self.job_health,
331                        "gdpr_purge",
332                        self.sweep_pending_purges().await,
333                        tracing::Level::INFO,
334                    );
335
336                    // GDPR export sweep (D-10..D-12).
337                    Self::record(
338                        &self.job_health,
339                        "gdpr_export",
340                        self.sweep_pending_exports().await,
341                        tracing::Level::INFO,
342                    );
343
344                    // Audit retention sweep (T-119).
345                    // INFO because this is the one sweep that destroys records
346                    // nothing else can reconstruct: that it ran, and how much it
347                    // removed, belongs in the operational log by default.
348                    Self::record(
349                        &self.job_health,
350                        "audit_retention",
351                        self.sweep_audit_retention().await,
352                        tracing::Level::INFO,
353                    );
354
355                    // T-39/T-143: drop revocation entries whose access tokens
356                    // have all expired. DEBUG, not INFO: unlike the audit
357                    // sweep this destroys nothing anyone could want back —
358                    // an expired entry describes only tokens that have
359                    // expired on their own `exp`.
360                    Self::record(
361                        &self.job_health,
362                        "revocation_feed",
363                        self.sweep_revocation_feed().await,
364                        tracing::Level::DEBUG,
365                    );
366                }
367                changed = self.shutdown.changed() => {
368                    if changed.is_ok() && *self.shutdown.borrow() {
369                        tracing::info!("cleanup task received shutdown signal");
370                        return Ok(());
371                    }
372                }
373            }
374        }
375    }

377    /// Log one sweep's outcome and record it for `GET /health/jobs` (T-129).
378    ///
379    /// Every sweep in [`Self::run`] goes through here rather than logging for
380    /// itself. That is the point: the previous shape repeated a six-line
381    /// `match` per sweep, and a new sweep added by copying one of them would
382    /// have logged correctly while reporting its liveness nowhere — which is
383    /// the exact failure T-129 describes, reintroduced one sweep at a time.
384    ///
385    /// Errors are recorded and swallowed, never propagated: the loop must
386    /// survive a failing sweep (T-04-36), and now the failure is visible on
387    /// the health endpoint instead of only in the log.
388    fn record(
389        health: &crate::job_health::JobHealth,
390        job: &'static str,
391        outcome: Result<u64, AxiamError>,
392        level: tracing::Level,
393    ) {
394        match outcome {
395            Ok(n) => {
396                if n > 0 {
397                    // `tracing`'s macros need a literal level, so the two cases
398                    // are spelled out rather than computed.
399                    if level == tracing::Level::INFO {
400                        tracing::info!(job, affected = n, "sweep completed");
401                    } else {
402                        tracing::debug!(job, affected = n, "sweep completed");
403                    }
404                }
405                health.record(job, Ok(()));
406            }
407            Err(e) => {
408                tracing::warn!(job, error = ?e, "sweep failed");
409                health.record(job, Err(e.to_string()));
410            }
411        }
412    }

416    // -----------------------------------------------------------------------

418    /// Delete audit entries older than the configured retention window.
419    ///
420    /// A no-op returning `Ok(0)` when retention is `None` (`0` days), which is
421    /// the explicit opt-out for deployments that archive out-of-band.
422    ///
423    /// The cutoff is recomputed from `Utc::now()` on every tick rather than
424    /// held as a fixed instant, so a long-running process prunes a moving
425    /// window rather than freezing the boundary at whenever it started.
426    ///
427    /// Runs deployment-wide, not per tenant: retention here is a property of
428    /// the datastore an operator is responsible for, and a per-tenant override
429    /// would let one tenant's settings decide how long another tenant's
430    /// records survive on shared storage. Per-tenant retention is a real
431    /// requirement, but it needs a settings surface (org baseline + tenant
432    /// override + API + UI) rather than a field on this sweep.
433    async fn sweep_audit_retention(&self) -> Result<u64, AxiamError> {
434        let Some(retention) = self.audit_retention else {
435            return Ok(0);
436        };
437        let cutoff = Utc::now() - retention;
438        self.audit_repo.prune_older_than(cutoff).await
439    }

441    /// Drop revocation-feed entries that have expired (T-39/T-143).
442    ///
443    /// A no-op returning `Ok(0)` when the deployment does not run the feed —
444    /// in which case there are no rows to drop, because nothing wrote any.
445    async fn sweep_revocation_feed(&self) -> Result<u64, AxiamError> {
446        let Some(repo) = &self.revoked_session_repo else {
447            return Ok(0);
448        };
449        repo.prune_expired(Utc::now()).await
450    }

456    /// Sweep users past their scheduled purge date and run the full purge pipeline.
457    ///
458    /// For each user:
459    /// (a) Revoke all sessions (auth artifact cascade via `AuthService`).
460    /// (b) Hard-delete federation links.
461    /// (b2) Hard-delete WebAuthn credentials and password history (SEC-056).
462    /// (b3) Prune `member_of` and `has_role` graph edges (isolated tombstone).
463    /// (c) Compute deterministic GDPR pseudonym via `gdpr_pseudonym`.
464    /// (d)/(e)/(g) Run [`run_erasure_pipeline`]: pseudonymize all audit
465    ///     entries for this user (now FATAL, D-01/D-03/D-04) -> anonymize the
466    ///     user row in-place (D-05) -> insert a PII-free erasure-proof
467    ///     record STRICTLY LAST (D-06, D-03a/D-03b — SECHRD-06).
468    /// (f) Mark the account_deletion row as completed.
469    /// (h) Emit `gdpr.user_pseudonymized` audit event (actor = System).
470    ///
471    /// Returns the count of users purged.
472    async fn sweep_pending_purges(&self) -> Result<u64, AxiamError> {
473        let pepper = match self.gdpr_pepper {
474            Some(p) => p,
475            None => {
476                tracing::warn!("GDPR purge sweep skipped — AXIAM__GDPR_PSEUDONYM_PEPPER not set");
477                return Ok(0);
478            }
479        };
480
481        let now = Utc::now();
482        let due = self.user_repo.find_due_for_purge(now).await?;
483        let mut purged: u64 = 0;
484
485        for user in due {
486            if let Err(e) = self
487                .purge_single_user(user.id, user.tenant_id, pepper)
488                .await
489            {
490                tracing::warn!(
491                    error = ?e,
492                    user_id = %user.id,
493                    tenant_id = %user.tenant_id,
494                    "purge failed for user — skipping"
495                );
496            } else {
497                purged += 1;
498            }
499        }
500
501        Ok(purged)
502    }

504    /// Run the full purge pipeline for a single user.
505    async fn purge_single_user(
506        &self,
507        user_id: Uuid,
508        tenant_id: Uuid,
509        pepper: [u8; 32],
510    ) -> Result<(), AxiamError> {
511        // (a) Revoke all sessions and OAuth2 refresh tokens.
512        self.auth_svc
513            .revoke_all_sessions(tenant_id, user_id)
514            .await?;
515
516        // (b) Hard-delete federation identity links.
517        match self
518            .federation_link_repo
519            .get_by_user_id(tenant_id, user_id)
520            .await
521        {
522            Ok(links) => {
523                for link in links {
524                    if let Err(e) = self.federation_link_repo.delete(tenant_id, link.id).await {
525                        tracing::warn!(
526                            error = %e,
527                            %tenant_id,
528                            link_id = %link.id,
529                            "cleanup: failed to delete expired federation link; will retry next cycle"
530                        );
531                    }
532                }
533            }
534            Err(e) => {
535                tracing::warn!(error = ?e, user_id = %user_id, "failed to list federation links for purge");
536            }
537        }
538
539        // (b2) Hard-delete stored credential material (SEC-056). Done with `?`
540        // (fail-closed): erasure must not be certified while a user's passkeys or
541        // password hashes remain. A transient failure aborts the purge and it is
542        // retried next sweep — every step here is idempotent.
543        let webauthn_creds = self.webauthn_repo.list_by_user(tenant_id, user_id).await?;
544        for cred in webauthn_creds {
545            self.webauthn_repo.delete(tenant_id, cred.id).await?;
546        }
547        // Prune the entire password-history chain (keep_count = 0 deletes all).
548        self.password_history_repo
549            .prune(tenant_id, user_id, 0)
550            .await?;
551
552        // (b3) Prune authorization-graph edges so the anonymized row is left as
553        // an isolated tombstone — no dangling `member_of` / `has_role` relations
554        // skewing group-member queries or leaving live authorization paths.
555        // Remove group memberships first so the subsequent role pass only sees
556        // the user's *direct* has_role edges (inherited ones vanish with the
557        // membership). Fail-closed (`?`): the tombstone must not be created while
558        // graph edges survive; every delete is idempotent and retried on failure.
559        let groups = self.group_repo.get_user_groups(tenant_id, user_id).await?;
560        for group in groups {
561            self.group_repo
562                .remove_member(tenant_id, user_id, group.id)
563                .await?;
564        }
565        let assignments = self
566            .role_repo
567            .get_user_role_assignments(tenant_id, user_id)
568            .await?;
569        for a in assignments {
570            self.role_repo
571                .unassign_from_user(tenant_id, user_id, a.role.id, a.resource_id)
572                .await?;
573        }
574
575        // (c) Compute deterministic pseudonym (keyed HMAC-SHA256, D-02).
576        let pseudonym = gdpr_pseudonym(&pepper, tenant_id, user_id);
577
578        // Derive email hash for anonymize_user from tenant+user IDs (original email
579        // is no longer accessible at purge time — user already marked deletion_pending).
580        use sha2::{Digest, Sha256};
581        let mut h = Sha256::new();
582        h.update(tenant_id.as_bytes());
583        h.update(user_id.as_bytes());
584        let email_hash = hex::encode(h.finalize());
585
586        // (d)/(e)/(g) Run the erasure pipeline: pseudonymize audit actor refs
587        // (now FATAL, was swallowed) -> anonymize the user row -> write the
588        // erasure proof STRICTLY LAST (SECHRD-06 / D-03a, T-25-13). A failure
589        // anywhere in this call propagates with `?` and aborts the purge:
590        // `anonymize_user` (the only step that clears `deletion_pending`)
591        // never ran, so the user stays due for a retry via
592        // `find_due_for_purge`, and NO erasure proof is ever written for an
593        // incomplete erasure.
594        run_erasure_pipeline(
595            self.audit_repo.as_ref(),
596            self.erasure_proof_repo.as_ref(),
597            self.user_repo.as_ref(),
598            tenant_id,
599            user_id,
600            &pseudonym,
601            &email_hash,
602        )
603        .await?;
604
605        // (f) Mark the account_deletion row as completed (lookup by user_id).
606        // Run AFTER the erasure pipeline succeeds — if the pipeline failed
607        // above, this is never reached and the deletion row stays pending
608        // for the next retry too.
609        match self
610            .account_deletion_repo
611            .find_pending_by_user_id(tenant_id, user_id)
612            .await
613        {
614            Ok(Some(deletion)) => {
615                if let Err(e) = self
616                    .account_deletion_repo
617                    .mark_completed(tenant_id, deletion.id)
618                    .await
619                {
620                    tracing::warn!(
621                        error = %e,
622                        %tenant_id,
623                        deletion_id = %deletion.id,
624                        "cleanup: failed to mark account_deletion completed"
625                    );
626                }
627            }
628            Ok(None) => {
629                tracing::debug!(user_id = %user_id, "no pending account_deletion row found at purge time");
630            }
631            Err(e) => {
632                tracing::warn!(error = ?e, user_id = %user_id, "failed to find account_deletion row");
633            }
634        }
635
636        // (h) Emit gdpr.user_pseudonymized audit event. actor_id is the
637        // erased subject's own (now-anonymized) row id — a real, resolvable
638        // identifier rather than a fabricated nil UUID (Task 1 fix); the row
639        // still exists (anonymize-in-place preserves referential integrity),
640        // so this is a legitimate audit-trail reference, not PII leakage.
641        // A DB-write failure here is dead-lettered to BOTH an append-only
642        // file AND a structured audit event (SECHRD-12 / D-02, T-24-61) —
643        // this legally-significant record must never be silently lost.
644        write_erasure_audit_with_dlq(
645            self.audit_repo.as_ref(),
646            CreateAuditLogEntry {
647                tenant_id,
648                actor_id: user_id,
649                actor_type: ActorType::System,
650                action: "gdpr.user_pseudonymized".into(),
651                resource_id: None,
652                outcome: AuditOutcome::Success,
653                ip_address: None,
654                metadata: Some(serde_json::json!({
655                    "pseudonym": pseudonym,
656                })),
657            },
658        )
659        .await;
660
661        tracing::info!(
662            pseudonym = %pseudonym,
663            tenant_id = %tenant_id,
664            "user purged and pseudonymized"
665        );
666
667        Ok(())
668    }
669
670    // -----------------------------------------------------------------------
671    // GDPR export sweep (D-10..D-13)
672    // -----------------------------------------------------------------------
673
674    /// Sweep queued export jobs and generate the encrypted blobs.
675    ///
676    /// For each job:
677    /// (a) Aggregate all Art. 15 sections from the DB (secrets excluded, D-10).
678    /// (b) Serialize to a single sectioned JSON object (D-11).
679    /// (c) Encrypt with `export_encryption_key` (AES-256-GCM, D-12).
680    /// (d) Call `set_ready` with a SHA-256-hashed 24h single-use download token.
681    /// (e) Enqueue `ExportReady` mail with the raw token (D-12).
682    ///
683    /// Returns the count of jobs processed.
684    async fn sweep_pending_exports(&self) -> Result<u64, AxiamError> {
685        let key = match self.export_encryption_key {
686            Some(k) => k,
687            None => {
688                tracing::warn!("GDPR export sweep skipped — AXIAM__EMAIL_ENCRYPTION_KEY not set");
689                return Ok(0);
690            }
691        };
692
693        let queued = self.export_job_repo.find_queued().await?;
694        let mut processed: u64 = 0;
695
696        for job in queued {
697            if let Err(e) = self
698                .process_export_job(job.id, job.tenant_id, job.user_id, key)
699                .await
700            {
701                tracing::warn!(
702                    error = ?e,
703                    job_id = %job.id,
704                    "export job processing failed — marking Failed (CQ-B38)"
705                );
706                // Mark as Failed so the job does not stay stuck as Queued
707                // (CQ-B38 / REQ-14 AC-5).
708                if let Err(mark_err) = self.export_job_repo.mark_failed(job.id).await {
709                    tracing::warn!(
710                        error = ?mark_err,
711                        job_id = %job.id,
712                        "failed to mark export job as Failed"
713                    );
714                }
715            } else {
716                processed += 1;
717            }
718        }
719
720        Ok(processed)
721    }

723    /// Process a single queued export job.
724    async fn process_export_job(
725        &self,
726        job_id: Uuid,
727        tenant_id: Uuid,
728        user_id: Uuid,
729        key: [u8; 32],
730    ) -> Result<(), AxiamError> {
731        // (a)/(b) Aggregate Art. 15 inventory into one sectioned JSON.
732        let export_json = self.aggregate_export_data(tenant_id, user_id).await?;
733        let export_bytes = export_json.to_string().into_bytes();
734
735        // (c) Encrypt the blob (D-12).
736        let (nonce_b64, ct_b64) = encrypt_separate(&key, &export_bytes)
737            .map_err(|e| AxiamError::Internal(format!("export encrypt failed: {e}")))?;
738
739        // (d) Generate single-use 24h download token (D-13).
740        //
741        // Deliberately v4, not `new_id()`: this token is the sole bearer
742        // credential for the export download, so it must stay maximally random.
743        // See `axiam_core::id` — v7 is for identifiers, never for secrets.
744        let raw_download_token = Uuid::new_v4().to_string();
745        let token_hash = {
746            use sha2::{Digest, Sha256};
747            let mut h = Sha256::new();
748            h.update(raw_download_token.as_bytes());
749            hex::encode(h.finalize())
750        };
751        let expires_at = Utc::now() + chrono::Duration::hours(24);
752
753        self.export_job_repo
754            .set_ready(
755                job_id,
756                token_hash,
757                Some(ct_b64),
758                None, // file_path: stored as DB blob
759                Some(nonce_b64),
760                expires_at,
761            )
762            .await?;
763
764        // Resolve the real org_id from the tenant before enqueuing the mail
765        // (SECHRD-08 / D-05d) — a Uuid::nil() org_id leaves the mail
766        // consumer unable to resolve the sending organization, producing
767        // undeliverable ExportReady mail.
768        let org_id = match self.tenant_repo.get_by_id(tenant_id).await {
769            Ok(tenant) => tenant.organization_id,
770            Err(e) => {
771                tracing::warn!(
772                    error = %e,
773                    %tenant_id,
774                    "failed to resolve org_id for ExportReady mail — falling back to nil (mail may be undeliverable)"
775                );
776                Uuid::nil()
777            }
778        };
779
780        // (e) Enqueue ExportReady email.
781        let download_url = format!("/api/v1/account/export/{}", raw_download_token);
782        let msg = OutboundMailMessage {
783            mail_type: MailType::ExportReady,
784            tenant_id,
785            org_id,
786            user_id,
787            to_address: String::new(), // mail consumer resolves from user_id
788            template_context: serde_json::json!({
789                "action_url": download_url,
790                "expiry_time": expires_at.to_rfc3339(),
791            }),
792            attempt_count: 0,
793            enqueued_at: Utc::now(),
794        };
795        if let Err(e) = self.mail_publisher.publish(msg).await {
796            tracing::warn!(error = %e, job_id = %job_id, "failed to enqueue ExportReady mail");
797        }
798
799        tracing::info!(job_id = %job_id, "export job completed");
800        Ok(())
801    }

803    /// Aggregate Art. 15 personal-data inventory for a user.
804    ///
805    /// EXCLUDED (D-10): `password_hash`, `mfa_secret`, any token_hash values.
806    async fn aggregate_export_data(
807        &self,
808        tenant_id: Uuid,
809        user_id: Uuid,
810    ) -> Result<serde_json::Value, AxiamError> {
811        // Profile — no password_hash or mfa_secret.
812        let user = self.user_repo.get_by_id(tenant_id, user_id).await?;
813        let profile = profile_section(&user);
814
815        // Consents — propagate errors (CQ-B38 / SEC-056): a section that fails to
816        // query must fail the whole export job rather than emit a legally
817        // incomplete Art. 15 inventory. `process_export_job` marks the job Failed.
818        let consents = self.consent_repo.list_by_user(tenant_id, user_id).await?;
819        let consents_json: Vec<_> = consents
820            .iter()
821            .map(|c| {
822                serde_json::json!({
823                    "consent_type": c.consent_type,
824                    "version": c.version,
825                    "accepted_at": c.accepted_at,
826                    "ip_address": c.ip_address,
827                })
828            })
829            .collect();
830
831        // Audit entries where this user was the actor — paginated to collect ALL
832        // entries regardless of volume (CQ-B38 / REQ-14 AC-5).
833        const AUDIT_PAGE_SIZE: u64 = 1_000;
834        let mut audit_items = Vec::new();
835        let mut offset: u64 = 0;
836        loop {
837            let page = self
838                .audit_repo
839                .list(
840                    tenant_id,
841                    AuditLogFilter {
842                        actor_id: Some(user_id),
843                        action: None,
844                        outcome: None,
845                        resource_id: None,
846                        from: None,
847                        to: None,
848                    },
849                    Pagination {
850                        offset,
851                        limit: AUDIT_PAGE_SIZE,
852                        search: None,
853                    },
854                )
855                .await?;
856            let fetched = page.items.len() as u64;
857            audit_items.extend(page.items);
858            offset += fetched;
859            if fetched < AUDIT_PAGE_SIZE {
860                break;
861            }
862        }
863        let audit_json: Vec<_> = audit_items
864            .iter()
865            .map(|e| {
866                serde_json::json!({
867                    "action": e.action,
868                    "outcome": e.outcome,
869                    "timestamp": e.timestamp,
870                    "resource_id": e.resource_id,
871                })
872            })
873            .collect();
874
875        // Sessions — real, metadata-only rows (D-03c, SECHRD-06). Token/hash
876        // material is live credential data and must NEVER appear in a GDPR
877        // export, so `token_hash` is deliberately excluded from the
878        // projection below. Propagate errors (CQ-B38 / SEC-056 convention):
879        // a section that fails to query must fail the whole export job.
880        let sessions = self.session_repo.list_by_user(tenant_id, user_id).await?;
881        let sessions_json: Vec<_> = sessions
882            .iter()
883            .map(|s| {
884                serde_json::json!({
885                    "id": s.id,
886                    "created_at": s.created_at,
887                    "expires_at": s.expires_at,
888                    "ip_address": s.ip_address,
889                    "user_agent": s.user_agent,
890                })
891            })
892            .collect();
893
894        // Federation identities — propagate errors (CQ-B38 / SEC-056).
895        let fed_links = self
896            .federation_link_repo
897            .get_by_user_id(tenant_id, user_id)
898            .await?;
899        let fed_json: Vec<_> = fed_links
900            .iter()
901            .map(|l| {
902                serde_json::json!({
903                    "federation_config_id": l.federation_config_id,
904                    "external_subject": l.external_subject,
905                    "created_at": l.created_at,
906                })
907            })
908            .collect();
909
910        // Role assignments (direct + inherited via groups), incl. resource scope.
911        let assignments = self
912            .role_repo
913            .get_user_role_assignments(tenant_id, user_id)
914            .await?;
915        let assignments_json: Vec<_> = assignments
916            .iter()
917            .map(|a| {
918                serde_json::json!({
919                    "role_id": a.role.id,
920                    "role_name": a.role.name,
921                    "is_global": a.role.is_global,
922                    "resource_id": a.resource_id,
923                })
924            })
925            .collect();
926
927        // Group memberships.
928        let groups = self.group_repo.get_user_groups(tenant_id, user_id).await?;
929        let groups_json: Vec<_> = groups
930            .iter()
931            .map(|g| {
932                serde_json::json!({
933                    "group_id": g.id,
934                    "name": g.name,
935                    "description": g.description,
936                })
937            })
938            .collect();
939
940        // WebAuthn credentials — metadata only; the encrypted `passkey_json`
941        // secret material is intentionally EXCLUDED (D-10).
942        let webauthn_creds = self.webauthn_repo.list_by_user(tenant_id, user_id).await?;
943        let webauthn_json: Vec<_> = webauthn_creds
944            .iter()
945            .map(|c| {
946                serde_json::json!({
947                    "id": c.id,
948                    "credential_id": c.credential_id,
949                    "name": c.name,
950                    "credential_type": c.credential_type,
951                    "created_at": c.created_at,
952                    "last_used_at": c.last_used_at,
953                })
954            })
955            .collect();
956
957        let export = serde_json::json!({
958            "export_metadata": {
959                "generated_at": Utc::now(),
960                "tenant_id": tenant_id,
961                "subject_id": user_id,
962                "schema_version": "1.0",
963            },
964            "profile": profile,
965            "consents": consents_json,
966            "sessions": sessions_json, // metadata only; NO token_hash (D-03c)
967            "mfa": { "enabled": user.mfa_enabled }, // NO mfa_secret
968            "federation_identities": fed_json,
969            "assignments": assignments_json,
970            "group_memberships": groups_json,
971            "audit_entries": audit_json,
972            "webauthn_credentials": webauthn_json,
973        });
974
975        Ok(export)
976    }
977}
