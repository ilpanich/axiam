1//! The binary's subcommands, parsed as data rather than matched inline.
2//!
3//! `main.rs` cannot be linked from an integration test, so argv handling that
4//! lives there is untestable by construction — which is how
5//! `tests/healthcheck.rs` ended up re-implementing the probe it meant to test.
6//! This module is the parse, as a pure function over the arguments, so the
7//! table of what the binary answers to is checkable.
8//!
9//! Deliberately not `clap`. The binary answers to three subcommands and
10//! otherwise serves; a dependency that renders help for an argument surface
11//! this small would be more code than it replaces, and the one thing this has
12//! to get right — `setup-token` never silently starting a server — is one
13//! branch.
14
15/// What the binary was asked to do.
16#[derive(Debug, Clone, PartialEq, Eq)]
17pub enum Command {
18    /// Run the server. The default, and what an unrecognised argument still
19    /// means: a deployment that passes an extra flag keeps booting.
20    Serve,
21    /// `healthcheck` — probe `/health` and exit 0 on 2xx (D-09).
22    Healthcheck,
23    /// `--dump-openapi` — print the OpenAPI document and exit (FND-01).
24    DumpOpenApi,
25    /// `setup-token --remint` — replace the bootstrap setup token on a
26    /// deployment nobody has bootstrapped yet (DF-019).
27    RemintSetupToken,
28    /// A subcommand that was recognised but not completed. Carries the line to
29    /// print; the binary exits 2 rather than booting.
30    Usage(&'static str),
31}
32
33/// Parse `args` — the whole of `std::env::args()`, program name included.
34#[must_use]
35pub fn parse(args: &[String]) -> Command {
36    match args.get(1).map(String::as_str) {
37        Some("healthcheck") => Command::Healthcheck,
38        Some("--dump-openapi") => Command::DumpOpenApi,
39        // `setup-token` alone must not fall through to `Serve`: an operator
40        // who mistypes the flag would otherwise start a second server against
41        // the production datastore and read it as the command having done
42        // nothing.
43        Some("setup-token") => match args.get(2).map(String::as_str) {
44            Some("--remint") => Command::RemintSetupToken,
45            _ => Command::Usage("usage: axiam-server setup-token --remint"),
46        },
47        _ => Command::Serve,
48    }
49}

51#[cfg(test)]
52mod tests {
53    use super::*;
54
55    fn argv(rest: &[&str]) -> Vec<String> {
56        std::iter::once("axiam-server")
57            .chain(rest.iter().copied())
58            .map(str::to_owned)
59            .collect()
60    }
61
62    #[test]
63    fn no_arguments_means_serve() {
64        assert_eq!(parse(&argv(&[])), Command::Serve);
65    }
66
67    #[test]
68    fn the_existing_two_subcommands_are_unchanged() {
69        assert_eq!(parse(&argv(&["healthcheck"])), Command::Healthcheck);
70        assert_eq!(parse(&argv(&["--dump-openapi"])), Command::DumpOpenApi);
71    }
72
73    #[test]
74    fn setup_token_remint_is_recognised() {
75        assert_eq!(
76            parse(&argv(&["setup-token", "--remint"])),
77            Command::RemintSetupToken
78        );
79    }
80
81    /// The branch that matters: `setup-token` with the flag missing or
82    /// mistyped must not boot a server.
83    #[test]
84    fn setup_token_without_the_flag_is_usage_not_serve() {
85        for rest in [
86            vec!["setup-token"],
87            vec!["setup-token", "--print"],
88            vec!["setup-token", "remint"],
89            vec!["setup-token", "--remint=yes"],
90        ] {
91            assert!(
92                matches!(parse(&argv(&rest)), Command::Usage(_)),
93                "{rest:?} must not start a server"
94            );
95        }
96    }
97
98    /// The I4 twin: an argument the binary does not know still serves, which
99    /// is what it did before this module existed.
100    #[test]
101    fn an_unrecognised_argument_still_serves() {
102        assert_eq!(parse(&argv(&["--verbose"])), Command::Serve);
103        assert_eq!(parse(&argv(&["serve"])), Command::Serve);
104    }
105}
