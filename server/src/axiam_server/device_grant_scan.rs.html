1//! The boot-time warning for confidential clients that hold the device grant
2//! (R110W3-02, #675).
3//!
4//! Since 1.1.0 (#635) the device grant authenticates its client at both
5//! endpoints by the method the client was registered with. A client registered
6//! without `token_endpoint_auth_method` is confidential, and the 1.0.x
7//! quickstart registered device clients that way; its devices send only a
8//! `client_id`, so after the upgrade every one of them is answered
9//! `401 invalid_client` and the server's only trace is a per-request audit row.
10//! This is the one line that names the cause once, at startup.
11//!
12//! What it promises, and how:
13//!
14//! * **Counts only.** The line carries two numbers. No client id, client name
15//!   or tenant name reaches the log.
16//! * **Bounded.** One round trip, two aggregates; no row of any client leaves
17//!   the datastore (`count_confidential_with_grant`).
18//! * **Never fails or delays boot.** [`spawn_boot_scan`] runs on its own task
19//!   and the scan has a deadline; an error or a timeout is a `warn!` and the
20//!   server carries on.
21//! * **Once per process.** The task is spawned at most once, however often the
22//!   composition root is entered.
23
24use std::future::Future;
25use std::sync::atomic::{AtomicBool, Ordering};
26use std::time::Duration;
27
28use axiam_core::error::AxiamResult;
29use axiam_core::models::oauth2_client::ConfidentialGrantCount;
30use axiam_core::repository::OAuth2ClientRepository;
31use axiam_oauth2::device_service::DEVICE_CODE_GRANT_SPELLINGS;
32
33/// How long the scan may take before boot gives up on it. It runs beside
34/// boot, not in front of it, so this bounds a task, not startup.
35pub const SCAN_TIMEOUT: Duration = Duration::from_secs(5);
36
37static SPAWNED: AtomicBool = AtomicBool::new(false);
38
39/// Start the scan on its own task, unless this process already did.
40///
41/// Returns the task's handle when it started one, so a test can wait for it;
42/// the composition root drops it.
43pub fn spawn_boot_scan<R>(repo: R) -> Option<tokio::task::JoinHandle<()>>
44where
45    R: OAuth2ClientRepository + 'static,
46{
47    if SPAWNED.swap(true, Ordering::SeqCst) {
48        return None;
49    }
50    Some(tokio::spawn(async move {
51        scan_with(
52            repo.count_confidential_with_grant(&DEVICE_CODE_GRANT_SPELLINGS),
53            SCAN_TIMEOUT,
54        )
55        .await;
56    }))
57}
58
59/// Run one scan under `timeout` and log its outcome: one `warn!` when there is
60/// something to say or when the scan could not finish, nothing when the count
61/// is zero.
62///
63/// The scan is a parameter so a test can hand it a failing or a never-ending
64/// one.
65pub async fn scan_with<F>(scan: F, timeout: Duration)
66where
67    F: Future<Output = AxiamResult<ConfidentialGrantCount>>,
68{
69    match tokio::time::timeout(timeout, scan).await {
70        Ok(Ok(count)) if count.clients == 0 => {
71            tracing::debug!("no confidential OAuth2 client holds the device-code grant");
72        }
73        Ok(Ok(count)) => tracing::warn!(
74            clients = count.clients,
75            tenants = count.tenants,
76            "{} confidential OAuth2 client(s) in {} tenant(s) hold the device-code grant. Since 1.1.0 \
77             such a client must send its credential at both device endpoints \
78             (/oauth2/device_authorization and the token poll) or the flow answers 401 \
79             invalid_client. A device that sends only a client_id needs a public client \
80             (token_endpoint_auth_method: none); a client cannot be changed between public \
81             and confidential. See the upgrade note in CHANGELOG.md [1.1.0] and \
82             docs/api/device-flow.md",
83            count.clients,
84            count.tenants,
85        ),
86        Ok(Err(e)) => tracing::warn!(
87            error = %e,
88            "could not count the confidential OAuth2 clients that hold the device-code grant; \
89             startup continues"
90        ),
91        Err(_) => tracing::warn!(
92            timeout_secs = timeout.as_secs_f32(),
93            "counting the confidential OAuth2 clients that hold the device-code grant timed \
94             out; startup continues"
95        ),
96    }
97}
