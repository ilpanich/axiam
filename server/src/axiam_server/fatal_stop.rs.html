1//! The orderly stop for a component that died (P23W5-A11, A12; D-72).
2//!
3//! The full profile used to end the process with `std::process::exit(1)` when
4//! the authz consumer, the audit-ingestion consumer, the mail consumer or the
5//! gRPC server stopped: wherever it was, with the audit rows still queued, the
6//! requests in flight and a GDPR purge between its erasure and its audit row
7//! all lost — the hazard the minimal profile's lost lease had until T23.8.2.
8//!
9//! A dying component now [`raise`](FatalStop::raise)s the signal instead.
10//! [`spawn_fatal_stop`] waits on it and runs the lost-lease path's stop (the
11//! REST listener stops accepting and finishes what is in flight, the teardown
12//! after it stops gRPC, drains the audit queue and the cleanup task, and
13//! `serve` returns an error, so `main` exits non-zero), with a backstop if that
14//! does not finish in time. The deadline is the caller's: it covers the whole
15//! stop, which is longer than the lease's.
16
17use std::sync::Arc;
18use std::time::Duration;
19
20use tokio::sync::watch;
21use tokio::task::JoinHandle;
22use tracing::error;
23
24use crate::profile::OnLeaseLost;
25
26/// The signal a dying component raises. Cheap to clone into each task.
27#[derive(Clone)]
28pub struct FatalStop(Arc<watch::Sender<Option<&'static str>>>);
29
30impl FatalStop {
31    /// A signal and the receiver [`spawn_fatal_stop`] and the teardown read.
32    pub fn new() -> (Self, watch::Receiver<Option<&'static str>>) {
33        let (tx, rx) = watch::channel(None);
34        (Self(Arc::new(tx)), rx)
35    }
36
37    /// `component` has stopped and the process cannot go on without it. The
38    /// first component to say so is the one reported.
39    pub fn raise(&self, component: &'static str) {
40        error!(component, "a component the server needs has stopped");
41        self.0.send_if_modified(|first| {
42            // Later deaths are consequences of the first.
43            if first.is_some() {
44                return false;
45            }
46            *first = Some(component);
47            true
48        });
49    }
50}
51
52/// Once the signal is raised: call `stop` (the composition root's orderly
53/// stop, as for a lost lease) and, if the process is still here `deadline`
54/// later, run `backstop`. The composition root aborts the returned task when
55/// its teardown has finished, which disarms the backstop.
56pub fn spawn_fatal_stop(
57    mut died: watch::Receiver<Option<&'static str>>,
58    stop: impl FnOnce() + Send + 'static,
59    deadline: Duration,
60    backstop: OnLeaseLost,
61) -> JoinHandle<()> {
62    tokio::spawn(async move {
63        // The `Ref` is copied out of and dropped here: it is not `Send`.
64        let component = match died.wait_for(Option::is_some).await {
65            Ok(raised) => (*raised).unwrap_or_default(),
66            Err(_) => return,
67        };
68        error!(
69            component,
70            deadline_secs = deadline.as_secs_f64(),
71            "stopping in order (no new connections, in-flight work finished, queued audit \
72             rows written) and then exiting non-zero"
73        );
74        stop();
75        tokio::time::sleep(deadline).await;
76        error!("the orderly stop after a component died did not finish in time — exiting now");
77        backstop();
78    })
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84    use std::sync::atomic::{AtomicU32, Ordering};
85
86    const DEADLINE: Duration = Duration::from_millis(150);
87
88    fn counter() -> (Arc<AtomicU32>, OnLeaseLost) {
89        let n = Arc::new(AtomicU32::new(0));
90        let m = Arc::clone(&n);
91        (
92            n,
93            Arc::new(move || {
94                m.fetch_add(1, Ordering::SeqCst);
95            }),
96        )
97    }
98
99    #[test]
100    fn the_first_component_to_die_is_the_one_reported() {
101        let (fatal, rx) = FatalStop::new();
102        assert_eq!(*rx.borrow(), None);
103        fatal.raise("AMQP mail consumer");
104        fatal.raise("gRPC server");
105        assert_eq!(*rx.borrow(), Some("AMQP mail consumer"));
106    }
107
108    #[tokio::test]
109    async fn a_dead_consumer_starts_the_orderly_stop_at_once_and_the_backstop_only_after_the_deadline()
110     {
111        let (fatal, rx) = FatalStop::new();
112        let (stops, stop) = counter();
113        let (backstops, backstop) = counter();
114        let task = spawn_fatal_stop(rx, move || stop(), DEADLINE, backstop);
115
116        tokio::time::sleep(Duration::from_millis(50)).await;
117        assert_eq!(stops.load(Ordering::SeqCst), 0, "nothing while all run");
118
119        // A consumer task that returns is what raises it, as in `boot`.
120        tokio::spawn(async move {
121            fatal.raise("AMQP authz consumer");
122        })
123        .await
124        .unwrap();
125        tokio::time::sleep(Duration::from_millis(50)).await;
126        assert_eq!(stops.load(Ordering::SeqCst), 1, "the orderly stop starts");
127        assert_eq!(backstops.load(Ordering::SeqCst), 0, "the backstop waits");
128
129        tokio::time::timeout(Duration::from_secs(2), task)
130            .await
131            .expect("the task ends after the backstop")
132            .unwrap();
133        assert_eq!(backstops.load(Ordering::SeqCst), 1, "an overrun ends in it");
134    }
135
136    #[tokio::test]
137    async fn an_orderly_stop_that_finishes_in_time_disarms_the_backstop() {
138        let (fatal, rx) = FatalStop::new();
139        let (stops, stop) = counter();
140        let (backstops, backstop) = counter();
141        let task = spawn_fatal_stop(rx, move || stop(), DEADLINE, backstop);
142        fatal.raise("gRPC server");
143        tokio::time::sleep(Duration::from_millis(20)).await;
144        assert_eq!(stops.load(Ordering::SeqCst), 1);
145        task.abort(); // the teardown finished
146        tokio::time::sleep(DEADLINE * 2).await;
147        assert_eq!(backstops.load(Ordering::SeqCst), 0);
148    }
149
150    #[tokio::test]
151    async fn two_deaths_stop_once() {
152        let (fatal, rx) = FatalStop::new();
153        let (stops, stop) = counter();
154        let (_, backstop) = counter();
155        let task = spawn_fatal_stop(rx, move || stop(), Duration::from_secs(60), backstop);
156        fatal.raise("AMQP audit consumer");
157        fatal.raise("gRPC server");
158        tokio::time::sleep(Duration::from_millis(50)).await;
159        assert_eq!(stops.load(Ordering::SeqCst), 1);
160        task.abort();
161    }
162
163    #[tokio::test]
164    async fn nothing_dying_never_stops() {
165        let (fatal, rx) = FatalStop::new();
166        let (stops, stop) = counter();
167        let (backstops, backstop) = counter();
168        let task = spawn_fatal_stop(rx, move || stop(), DEADLINE, backstop);
169        drop(fatal); // the sender goes with the server: an ordinary stop
170        tokio::time::timeout(Duration::from_secs(2), task)
171            .await
172            .expect("the watch ends with its sender")
173            .unwrap();
174        assert_eq!(stops.load(Ordering::SeqCst), 0);
175        assert_eq!(backstops.load(Ordering::SeqCst), 0);
176    }
