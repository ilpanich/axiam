1//! Where `axiam-server healthcheck` probes, and what it trusts (D-09, DF-016).
2//!
3//! The probe was `reqwest::blocking::get("http://127.0.0.1:8090/health")` with
4//! `AXIAM_HEALTHCHECK_URL` as its only knob. On a deployment that terminates
5//! TLS in the server process — `AXIAM__SERVER__TLS__ENABLED`, which is how the
6//! shipped Kubernetes ConfigMap runs — that is a plaintext request to a TLS
7//! listener, so the container healthcheck fails forever and the operator's only
8//! recourse was an `AXIAM_HEALTHCHECK_URL` pointing at an `https://` address the
9//! probe then could not verify.
10//!
11//! Two things were needed and one thing was not. The two: a default that
12//! follows the listener, and a way to name the trust anchor. The one that was
13//! not: an "insecure" switch. A probe that skips verification is a probe that
14//! answers "healthy" to anything listening on the port, which is worse than no
15//! probe, because a deployment stops looking.
16
17use std::path::PathBuf;
18
19/// The default REST port, matching `ServerConfig`'s own default.
20const DEFAULT_PORT: u16 = 8090;
21
22/// What to connect to, and what to trust when connecting.
23#[derive(Debug, Clone, PartialEq, Eq)]
24pub struct Probe {
25    /// The URL to request.
26    pub url: String,
27    /// A PEM bundle whose certificates are added as trust anchors.
28    ///
29    /// `None` means the platform trust store alone — right for a plaintext
30    /// probe and for a publicly-issued certificate, and wrong for the private
31    /// CA or self-signed certificate a direct-TLS deployment usually carries,
32    /// which is what the resolution below supplies.
33    pub trust_anchors: Option<PathBuf>,
34}
35
36/// Resolve the probe from the environment.
37///
38/// `var` reads an environment variable; a variable that is set but empty is
39/// treated as unset, because that is what a Compose file writes when an
40/// interpolated variable is missing.
41///
42/// - `AXIAM_HEALTHCHECK_URL` still wins outright, unchanged.
43/// - Otherwise the scheme follows the listener: `https` when the server
44///   terminates TLS itself, `http` when it does not.
45/// - `AXIAM_HEALTHCHECK_CA_FILE` names the trust anchors.
46/// - With no CA file and an `https` probe of a TLS-terminating server, the
47///   anchors are the server's **own** `AXIAM__SERVER__TLS__CERT_PATH` chain:
48///   a process verifying the certificate it is itself serving adds no trust it
49///   does not already have.
50#[must_use]
51pub fn resolve<F>(var: F) -> Probe
52where
53    F: Fn(&str) -> Option<String>,
54{
55    let read = |name: &str| var(name).filter(|v| !v.trim().is_empty());
56
57    // Both, not just the path. `docker-compose.prod.yml` sets
58    // `AXIAM__SERVER__TLS__CERT_PATH` unconditionally and gates the listener on
59    // `AXIAM__SERVER__TLS__ENABLED`, so reading the path alone would switch
60    // every Compose deployment's probe to `https` against a plaintext listener
61    // — the present defect, in the opposite direction.
62    let cert_path = read("AXIAM__SERVER__TLS__CERT_PATH").map(PathBuf::from);
63    let serving_tls =
64        cert_path.is_some() && is_true(read("AXIAM__SERVER__TLS__ENABLED").as_deref());
65
66    let url = read("AXIAM_HEALTHCHECK_URL").unwrap_or_else(|| {
67        let port = read("AXIAM__SERVER__PORT")
68            .and_then(|p| p.trim().parse::<u16>().ok())
69            .unwrap_or(DEFAULT_PORT);
70        let scheme = if serving_tls { "https" } else { "http" };
71        format!("{scheme}://127.0.0.1:{port}/health")
72    });
73
74    let trust_anchors = match read("AXIAM_HEALTHCHECK_CA_FILE") {
75        Some(path) => Some(PathBuf::from(path)),
76        // Only for an `https` probe: adding anchors to a plaintext request is
77        // meaningless, and doing it would hide a misconfigured CA file until
78        // the day TLS was turned on.
79        None if serving_tls && url.starts_with("https://") => cert_path,
80        None => None,
81    };
82
83    Probe { url, trust_anchors }
84}

86/// The truthiness `config-rs` gives a boolean field, plus the spellings the
87/// `AXIAM__PKI__MDS_*` variables already accept in `main.rs`.
88fn is_true(value: Option<&str>) -> bool {
89    matches!(
90        value.map(|v| v.trim().to_ascii_lowercase()).as_deref(),
91        Some("true" | "1" | "yes" | "on")
92    )
93}
94
95/// Run the probe. `true` iff the endpoint answered 2xx.
96///
97/// Every failure is `false` — unreachable, a TLS handshake that did not verify,
98/// an unreadable or empty anchor bundle. A healthcheck reports health; it is
99/// not the place to learn why a deployment is broken, and a probe that
100/// distinguished "could not verify" from "not listening" by exiting differently
101/// would make an orchestrator treat a misconfiguration as a healthy pod.
102/// The reason goes to stderr, where `docker inspect` and `kubectl describe`
103/// surface it.
104#[must_use]
105pub fn run(probe: &Probe) -> bool {
106    let mut builder = reqwest::blocking::Client::builder();
107
108    if let Some(path) = &probe.trust_anchors {
109        let pem = match std::fs::read(path) {
110            Ok(pem) => pem,
111            Err(e) => {
112                eprintln!("healthcheck: cannot read {}: {e}", path.display());
113                return false;
114            }
115        };
116        let anchors = match reqwest::Certificate::from_pem_bundle(&pem) {
117            Ok(anchors) => anchors,
118            Err(e) => {
119                eprintln!("healthcheck: {} is not a PEM bundle: {e}", path.display());
120                return false;
121            }
122        };
123        // `from_pem_bundle` answers `Ok` with an empty list for a file that
124        // holds no certificate, which would silently leave the probe on the
125        // platform trust store — the pinning the operator asked for, absent,
126        // with nothing to say so.
127        if anchors.is_empty() {
128            eprintln!("healthcheck: {} contains no certificates", path.display());
129            return false;
130        }
131        for anchor in anchors {
132            builder = builder.add_root_certificate(anchor);
133        }
134    }
135
136    let client = match builder.build() {
137        Ok(client) => client,
138        Err(e) => {
139            eprintln!("healthcheck: could not build the HTTP client: {e}");
140            return false;
141        }
142    };
143
144    match client.get(&probe.url).send() {
145        Ok(response) => response.status().is_success(),
146        Err(e) => {
147            eprintln!("healthcheck: {} did not answer: {e}", probe.url);
148            false
149        }
150    }
151}

153#[cfg(test)]
154mod tests {
155    use super::*;
156    use std::collections::HashMap;

158    fn env(pairs: &[(&str, &str)]) -> HashMap<String, String> {
159        pairs
160            .iter()
161            .map(|(k, v)| ((*k).to_owned(), (*v).to_owned()))
162            .collect()
163    }
164
165    fn resolve_with(pairs: &[(&str, &str)]) -> Probe {
166        let map = env(pairs);
167        resolve(|name| map.get(name).cloned())
168    }

170    /// The I4 twin: a plaintext deployment probes exactly what it probed
171    /// before, with the platform trust store and nothing added.
172    #[test]
173    fn a_plaintext_deployment_is_unchanged() {
174        assert_eq!(
175            resolve_with(&[]),
176            Probe {
177                url: "http://127.0.0.1:8090/health".to_owned(),
178                trust_anchors: None,
179            }
180        );
181    }

183    /// `docker-compose.prod.yml` sets the certificate path unconditionally and
184    /// gates the listener on `ENABLED`. Reading the path alone would move every
185    /// Compose deployment's probe to `https` against a plaintext listener.
186    #[test]
187    fn a_certificate_path_without_enabled_stays_plaintext() {
188        assert_eq!(
189            resolve_with(&[(
190                "AXIAM__SERVER__TLS__CERT_PATH",
191                "/etc/axiam/server-tls/fullchain.pem"
192            )]),
193            Probe {
194                url: "http://127.0.0.1:8090/health".to_owned(),
195                trust_anchors: None,
196            }
197        );
198    }

200    #[test]
201    fn a_tls_deployment_probes_https_and_trusts_its_own_chain() {
202        assert_eq!(
203            resolve_with(&[
204                ("AXIAM__SERVER__TLS__ENABLED", "true"),
205                ("AXIAM__SERVER__TLS__CERT_PATH", "/etc/axiam/tls.crt"),
206            ]),
207            Probe {
208                url: "https://127.0.0.1:8090/health".to_owned(),
209                trust_anchors: Some(PathBuf::from("/etc/axiam/tls.crt")),
