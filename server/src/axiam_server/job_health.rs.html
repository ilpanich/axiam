1//! Scheduled-job liveness tracking (T-129).
2//!
3//! Every background sweep records the outcome of each run here, and
4//! `GET /health/jobs` reads it back. The threat this closes is not a job that
5//! errors — those were already logged — but a job that stops running at all:
6//! GDPR erasure and certificate expiry fail silently by simply not happening,
7//! and nothing in an error-driven monitoring setup ever fires.
8//!
9//! Deliberately in-memory and per-process. A restart resets the history, and
10//! two replicas each report their own. That is the correct scope: this answers
11//! "is *this* process's scheduler alive", which is what a per-pod alert needs.
12//! Aggregating across replicas is the monitoring system's job, and persisting
13//! it would make the health of the sweep depend on the datastore the sweep is
14//! there to maintain.
15
16use std::collections::BTreeMap;
17use std::sync::{Arc, Mutex};
18use std::time::Duration;

20use axiam_api_rest::health::{JobHealthReporter, JobStatus};

23/// How many expected runs a job may miss before it is reported as stalled.
24///
25/// Three rather than one: a sweep that overruns its interval, or a tick
26/// skipped under load, is normal and must not page anyone. Three consecutive
27/// missed intervals is not noise.
28const STALL_INTERVALS: u32 = 3;
29
30/// The jobs registered at start-up, so a job that has never run once still
31/// appears in `GET /health/jobs` (T-129). One list, read by `main`; a test pins
32/// that the single-logout sweeps are in it and in the cleanup loop.
33pub const SWEEP_JOBS: &[&str] = &[
34    "saml_assertion_replay",
35    "federation_login_state",
36    "saml_authn_request",
37    // G-2 (T23.2.4): the single-logout participant rows and logout runs.
38    "saml_sp_session",
39    "saml_logout_run",
40    // G-3 (T23.3.5): the directory sync job.
41    "directory_sync",
42    // G-6 (T23.6.3, D-58): the outbound SCIM reconciliation.
43    "scim_reconcile",
44    // G-5 (T23.5.3): the SSF poll/hold buffer's seven-day expiry.
45    "ssf_event_buffer",
46    // G-5 (T23.5.3, D-53 (1)): the step-up record's ten-minute expiry.
47    "ssf_step_up",
48    // G-7 (T23.7.1): the CIBA pending-request store's expiry.
49    "ciba_request",
50    "amqp_nonce_replay",
51    "gdpr_purge",
52    "gdpr_export",
53    "audit_retention",
54];

56#[derive(Default, Clone)]
57struct Entry {
58    last_success_at: Option<DateTime<Utc>>,
59    last_failure_at: Option<DateTime<Utc>>,
60    last_error: Option<String>,
61    consecutive_failures: u32,
62}
63
64/// Shared, cloneable handle to the job-status table.
65#[derive(Clone)]
66pub struct JobHealth {
67    inner: Arc<Mutex<BTreeMap<&'static str, Entry>>>,
68    /// The sweep interval, used to decide what counts as stalled.
69    interval: Duration,
70    /// When the process started, so a job that has never run once can be
71    /// distinguished from one that has stopped running. Without this, every
72    /// job reads as stalled for the first few seconds after boot.
73    started_at: DateTime<Utc>,
74}
75
76impl JobHealth {
77    /// Create a tracker for sweeps running every `interval`.
78    pub fn new(interval: Duration) -> Self {
79        Self {
80            inner: Arc::new(Mutex::new(BTreeMap::new())),
81            interval,
82            started_at: Utc::now(),
83        }
84    }
85
86    /// Register a job so it appears in the snapshot before its first run.
87    ///
88    /// Without this a job that has never once succeeded would be absent
89    /// entirely, and "not in the list" is indistinguishable from "not
90    /// deployed" — precisely the silence T-129 is about.
91    pub fn register(&self, name: &'static str) {
92        self.lock().entry(name).or_default();
93    }
94
95    /// Record a completed run.
96    pub fn record(&self, name: &'static str, outcome: Result<(), String>) {
97        let mut guard = self.lock();
98        let entry = guard.entry(name).or_default();
99        match outcome {
100            Ok(()) => {
101                entry.last_success_at = Some(Utc::now());
102                entry.consecutive_failures = 0;
103                entry.last_error = None;
104            }
105            Err(e) => {
106                entry.last_failure_at = Some(Utc::now());
107                entry.consecutive_failures = entry.consecutive_failures.saturating_add(1);
108                entry.last_error = Some(e);
109            }
110        }
111    }
112
113    /// A poisoned mutex must not take the server down: this table is
114    /// diagnostics, and panicking here would turn a reporting bug into an
115    /// outage. Recovering the guard keeps whatever data survived.
116    fn lock(&self) -> std::sync::MutexGuard<'_, BTreeMap<&'static str, Entry>> {
117        self.inner.lock().unwrap_or_else(|e| e.into_inner())
118    }
119
120    /// Whether `entry` has gone longer than [`STALL_INTERVALS`] without a
121    /// successful run, measured from process start when it has never had one.
122    fn is_stalled(&self, entry: &Entry, now: DateTime<Utc>) -> bool {
123        let budget = match chrono::Duration::from_std(self.interval * STALL_INTERVALS) {
124            Ok(d) => d,
125            // Only reachable with an absurd configured interval; treating that
126            // as "never stalled" is the safe direction — a false alarm on a
127            // liveness signal is how alerts get muted.
128            Err(_) => return false,
129        };
130        let reference = entry.last_success_at.unwrap_or(self.started_at);
131        now.signed_duration_since(reference) > budget
132    }
133}
134
135impl JobHealthReporter for JobHealth {
136    fn snapshot(&self) -> Vec<JobStatus> {
137        let now = Utc::now();
138        self.lock()
139            .iter()
140            .map(|(name, entry)| JobStatus {
141                name: (*name).to_string(),
142                last_success_at: entry.last_success_at.map(|t| t.to_rfc3339()),
143                last_failure_at: entry.last_failure_at.map(|t| t.to_rfc3339()),
144                last_error: entry.last_error.clone(),
145                consecutive_failures: entry.consecutive_failures,
146                stalled: self.is_stalled(entry, now),
147            })
148            .collect()
149    }
150}

156    fn tracker() -> JobHealth {

160    /// T-384: the single-logout stores are swept by the cleanup scheduler and
161    /// reported on `/health/jobs` — both sweeps are recorded by the loop in
162    /// `cleanup.rs`, both are registered (so listed before their first run), and
163    /// both appear in a tracker built the way `main` builds it.
164    #[test]
165    fn the_slo_sweeps_are_recorded_by_the_cleanup_loop_and_registered() {
166        let source = include_str!("cleanup.rs");
167        // T23.5.3 (T-395): the SSF buffer's expiry sweep is in the same loop and
168        // the same list, so `/health/jobs` shows it from boot.
169        for job in [
170            "saml_sp_session",
171            "saml_logout_run",
172            "ssf_event_buffer",
173            "ssf_step_up",
174            // T23.6.3 (P23W4-06): a job the loop records and `SWEEP_JOBS` forgets
175            // reads as "not deployed" on `/health/jobs`.
176            "scim_reconcile",
177            // T23.7.1 (G-7): the CIBA pending-request sweep.
178            "ciba_request",
179        ] {
180            let recorded = source.match_indices("&self.job_health,").any(|(at, _)| {
181                let rest = &source[at + "&self.job_health,".len()..];
182                rest.trim_start().starts_with(&format!("\"{job}\""))
183            });
184            assert!(recorded, "{job} is swept by the loop");
185            assert!(SWEEP_JOBS.contains(&job), "{job} is registered");
186        }
187        let tracker = tracker();
188        for job in SWEEP_JOBS {
189            tracker.register(job);
190        }
191        let names: Vec<_> = tracker.snapshot().into_iter().map(|s| s.name).collect();
192        assert!(names.iter().any(|n| n == "saml_sp_session"));
193        assert!(names.iter().any(|n| n == "saml_logout_run"));
194        assert!(names.iter().any(|n| n == "ssf_event_buffer"));
195        assert!(names.iter().any(|n| n == "ssf_step_up"));
196        assert!(names.iter().any(|n| n == "scim_reconcile"));
197        assert!(names.iter().any(|n| n == "ciba_request"));
198    }

200    #[test]
201    fn a_registered_job_appears_before_it_has_ever_run() {
202        let h = tracker();
203        h.register("gdpr_purge");
204        let snap = h.snapshot();
205        assert_eq!(snap.len(), 1);
206        assert_eq!(snap[0].name, "gdpr_purge");
207        assert!(snap[0].last_success_at.is_none());
208        assert!(!snap[0].stalled, "a job that just started is not stalled");
209    }

211    #[test]
212    fn success_clears_the_failure_streak_and_the_error_text() {
213        let h = tracker();
214        h.record("audit_retention", Err("db down".into()));
215        h.record("audit_retention", Err("db down".into()));
216        assert_eq!(h.snapshot()[0].consecutive_failures, 2);
217        assert_eq!(h.snapshot()[0].last_error.as_deref(), Some("db down"));
218
219        h.record("audit_retention", Ok(()));
220        let s = &h.snapshot()[0];
221        assert_eq!(s.consecutive_failures, 0);
222        assert!(s.last_error.is_none(), "a stale error reads as a live one");
223        // The failure timestamp is kept on purpose: "it is working now, and it
224        // last broke at T" is more useful than pretending it never broke.
225        assert!(s.last_failure_at.is_some());
226    }

228    #[test]
229    fn failures_alone_do_not_mark_a_job_stalled() {
230        // Stalled means "not running". A job that runs and fails every time is
231        // a different condition, reported by consecutive_failures, and
232        // conflating them would hide whichever one you were not looking for.
233        let h = tracker();
234        for _ in 0..10 {
235            h.record("cert_expiry", Err("boom".into()));
236        }
237        let s = &h.snapshot()[0];
238        assert!(s.consecutive_failures >= 10);
239        assert!(!s.stalled);
240    }

242    #[test]
243    fn a_job_is_stalled_once_it_misses_the_configured_number_of_intervals() {
244        let h = tracker();
245        h.register("gdpr_purge");
246        let now = Utc::now();
247        let entry = Entry {
248            last_success_at: Some(now - chrono::Duration::seconds(61 * 3)),
249            ..Default::default()
250        };
251        assert!(h.is_stalled(&entry, now));
252
253        let recent = Entry {
254            last_success_at: Some(now - chrono::Duration::seconds(61)),
255            ..Default::default()
256        };
257        assert!(
258            !h.is_stalled(&recent, now),
259            "one missed tick is normal under load and must not alert"
260        );
261    }

263    #[test]
264    fn a_job_that_never_succeeds_becomes_stalled_relative_to_process_start() {
265        // The case that matters most: a sweep whose very first run never
266        // happens. Measuring from process start is what makes that visible,
267        // since there is no last_success_at to measure from.
268        let h = tracker();
269        h.register("gdpr_purge");
270        let entry = Entry::default();
271        let much_later = h.started_at + chrono::Duration::seconds(61 * 4);
272        assert!(h.is_stalled(&entry, much_later));
273    }
