1//! Environment variables that look like a secret's name and are read by nothing.
2//!
3//! Four variables were documented, set in two compose files and named in a
4//! dozen error messages, and none of them ever reached the server
5//! (DF-018/DF-022). `AXIAM__EMAIL_ENCRYPTION_KEY` and
6//! `AXIAM__GDPR_PSEUDONYM_PEPPER` map to `AppConfig` fields that are
7//! `#[serde(skip)]`; `AXIAM__PKI__ENCRYPTION_KEY` and
8//! `AXIAM__FEDERATION_ENCRYPTION_KEY` have no `AppConfig` field at all. All
9//! four are supplied by the secret provider instead, under the names
10//! [`axiam_core::secrets::env_var_name`] resolves.
11//!
12//! The fix is the rename (decision D-1: one name per secret, no alias). This
13//! module is the other half — the deployment that is already wrong learns so
14//! at startup, in one line, naming both spellings.
15//!
16//! **The value is never read.** [`legacy_secret_env_warnings`] is handed a
17//! predicate that answers *is this variable set*, not a lookup that returns
18//! what it holds, so there is no path by which a key could reach a log line.
19
20use axiam_core::secrets as keys;
21
22/// A variable an operator may have set, and the logical secret it looks like.
23///
24/// Deliberately not a table of aliases: nothing here is ever read. `AXIAM`
25/// resolves each logical key to exactly one variable, and this is the list of
26/// spellings that were *documented* as that variable and were not.
27///
28/// `AXIAM__AMQP__SIGNING_KEY` is **not** in this list although it looks like a
29/// sibling. It is a real, honoured variable — `load_config` deserialises it
30/// into `AmqpConfig::signing_key`, and the provider's
31/// `AXIAM__AUTH__AMQP_SIGNING_KEY` merely takes precedence when both are set.
32/// Warning about it would tell an operator with a working deployment that it
33/// is broken.
34const LEGACY_SPELLINGS: &[(&str, &str)] = &[
35    ("AXIAM__PKI__ENCRYPTION_KEY", keys::PKI_ENCRYPTION_KEY),
36    ("AXIAM__EMAIL_ENCRYPTION_KEY", keys::EMAIL_ENCRYPTION_KEY),
37    ("AXIAM__GDPR_PSEUDONYM_PEPPER", keys::GDPR_PSEUDONYM_PEPPER),
38    (
39        "AXIAM__FEDERATION_ENCRYPTION_KEY",
40        keys::FEDERATION_ENCRYPTION_KEY,
41    ),
42];
43
44/// One finding: a variable that is set and read by nothing, and the variable
45/// that would have been read instead.
46#[derive(Debug, Clone, PartialEq, Eq)]
47pub struct LegacyEnvWarning {
48    /// The spelling the operator set.
49    pub legacy: &'static str,
50    /// The variable the secret provider actually reads for the same secret.
51    pub resolved: String,
52}
53
54/// Every legacy spelling that is set while the variable AXIAM reads is not.
55///
56/// `is_set` answers whether a variable is present in the environment. It is a
57/// predicate rather than a lookup on purpose: a function that cannot obtain a
58/// value cannot leak one.
59///
60/// Both-set is silent. That is the shape of a migration in progress — the
61/// deployment already works, and the stale variable is about to go — and a
62/// warning there would be noise on a correct configuration.
63pub fn legacy_secret_env_warnings<F>(is_set: F) -> Vec<LegacyEnvWarning>
64where
65    F: Fn(&str) -> bool,
66{
67    LEGACY_SPELLINGS
68        .iter()
69        .filter_map(|(legacy, logical)| {
70            let resolved = keys::env_var_name(logical);
71            (is_set(legacy) && !is_set(&resolved)).then_some(LegacyEnvWarning { legacy, resolved })
72        })
73        .collect()
74}

76#[cfg(test)]
77mod tests {
78    use super::*;
79    use std::collections::HashSet;
80
81    fn env(vars: &[&str]) -> HashSet<String> {
82        vars.iter().map(|v| (*v).to_string()).collect()
83    }
84
85    #[test]
86    fn a_legacy_spelling_alone_is_reported_with_the_variable_that_is_read() {
87        let set = env(&["AXIAM__PKI__ENCRYPTION_KEY"]);
88        let found = legacy_secret_env_warnings(|v| set.contains(v));
89
90        assert_eq!(
91            found,
92            vec![LegacyEnvWarning {
93                legacy: "AXIAM__PKI__ENCRYPTION_KEY",
94                resolved: "AXIAM__AUTH__PKI_ENCRYPTION_KEY".to_owned(),
95            }]
96        );
97    }
98
99    #[test]
100    fn all_four_are_reported() {
101        let set = env(&[
102            "AXIAM__PKI__ENCRYPTION_KEY",
103            "AXIAM__EMAIL_ENCRYPTION_KEY",
104            "AXIAM__GDPR_PSEUDONYM_PEPPER",
105            "AXIAM__FEDERATION_ENCRYPTION_KEY",
106        ]);
107        let found = legacy_secret_env_warnings(|v| set.contains(v));
108
109        assert_eq!(found.len(), 4);
110        // The message interpolates nothing. Both fields are variable *names*
111        // and never values — `legacy_secret_env_warnings` takes an is-set
112        // predicate, so no value is reachable from here — but a panic message
113        // reaches stderr and a CI log that outlives the run, so a test is not
114        // exempt from the rule this repository already settled once: don't
115        // format something that reads as secret material into an assertion.
116        // CodeQL reported exactly this shape (`rust/cleartext-logging`, high)
117        // and is right about the construct even though nothing leaks. The four
118        // inputs are the four literals directly above; a failure is
119        // reproducible without them being printed back.
120        assert!(
121            found
122                .iter()
123                .all(|w| w.resolved.starts_with("AXIAM__AUTH__")),
124            "every legacy spelling must resolve to an AXIAM__AUTH__ variable"
125        );
126    }
127
128    /// The I4 twin: a deployment that set nothing, or set the right variable,
129    /// says nothing at startup. A warning an operator cannot act on is one
130    /// they learn to scroll past.
131    #[test]
132    fn a_correct_deployment_is_silent() {
133        let nothing = env(&[]);
134        assert!(legacy_secret_env_warnings(|v| nothing.contains(v)).is_empty());
135
136        let correct = env(&[
137            "AXIAM__AUTH__PKI_ENCRYPTION_KEY",
138            "AXIAM__AUTH__EMAIL_ENCRYPTION_KEY",
139            "AXIAM__AUTH__GDPR_PSEUDONYM_PEPPER",
140            "AXIAM__AUTH__FEDERATION_ENCRYPTION_KEY",
141        ]);
142        assert!(legacy_secret_env_warnings(|v| correct.contains(v)).is_empty());
143    }
144
145    /// A migration in progress is not a misconfiguration.
146    #[test]
147    fn both_spellings_set_is_silent() {
148        let both = env(&[
149            "AXIAM__PKI__ENCRYPTION_KEY",
150            "AXIAM__AUTH__PKI_ENCRYPTION_KEY",
151        ]);
152        assert!(legacy_secret_env_warnings(|v| both.contains(v)).is_empty());
153    }
154
155    /// `AXIAM__AMQP__SIGNING_KEY` is honoured by `load_config`. Telling an
156    /// operator who set it that it does nothing would be false.
157    #[test]
158    fn the_amqp_signing_key_is_not_treated_as_legacy() {
159        let set = env(&["AXIAM__AMQP__SIGNING_KEY"]);
160        assert!(legacy_secret_env_warnings(|v| set.contains(v)).is_empty());
161    }
