1//! The interval task that sweeps the in-process rate-limiter maps (#673).
2//!
3//! Every governor of the REST and gRPC crates registers its map with
4//! `axiam_core::limiter_sweep` when it is built; this task calls the sweep once
5//! a minute, the time a cell takes to replenish. It stops with the cleanup
6//! task's shutdown signal, so it ends with the server.
7
8use std::time::Duration;
9
10use tokio::sync::watch;
11use tokio::task::JoinHandle;
12
13/// How often the maps are swept: one replenish period of the slowest quota.
14pub const SWEEP_EVERY: Duration = Duration::from_secs(60);
15
16/// Spawn the sweep; it returns when `shutdown` turns `true` (or is dropped).
17pub fn spawn(mut shutdown: watch::Receiver<bool>, every: Duration) -> JoinHandle<()> {
18    tokio::spawn(async move {
19        let mut tick = tokio::time::interval(every);
20        tick.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);
21        tick.tick().await; // the first tick is immediate; nothing to sweep yet
22        loop {
23            tokio::select! {
24                _ = tick.tick() => {
25                    let report = axiam_core::limiter_sweep::sweep_all();
26                    tracing::debug!(
27                        limiters = report.limiters,
28                        cells = report.cells,
29                        "in-process rate-limiter maps swept"
30                    );
31                }
32                changed = shutdown.changed() => {
33                    if changed.is_err() || *shutdown.borrow() {
34                        return;
35                    }
36                }
37            }
38        }
39    })
40}

42#[cfg(test)]
43mod tests {
44    use super::*;
45
46    #[tokio::test(start_paused = true)]
47    async fn the_task_sweeps_on_its_interval_and_stops_on_shutdown() {
48        let (tx, rx) = watch::channel(false);
49        let handle = spawn(rx, Duration::from_secs(60));
50        tokio::time::sleep(Duration::from_secs(130)).await;
51        assert!(!handle.is_finished());
52        tx.send_replace(true);
53        handle.await.expect("the sweep task ends cleanly");
54    }
