1//! FIDO MDS3 background refresh job (X3 wave 3, D10) — pure decision/jitter
2//! helpers factored out of `main.rs`.
3//!
4//! `main.rs` is a binary entry point and cannot be linked from `tests/`
5//! (Rust integration tests only link a package's *library* crate), so the
6//! one piece of this job's logic worth asserting in a test — "is the
7//! disabled-by-default posture actually disabled" (the wave spec's "Default
8//! is disabled ⇒ zero outbound calls" requirement) — is pulled out here as a
9//! pure function `main.rs` calls as its single spawn gate. See this module's
10//! tests for the assertion.
11
12use axiam_pki::PkiConfig;
13
14/// Whether the weekly background refresh job should be spawned at all.
15///
16/// `mds_enabled: false` is [`PkiConfig::default`]'s shipped value — "off
17/// means zero outbound calls" (D10) — so this returns `false` whenever the
18/// master switch is off, AND whenever the refresh interval is `0` (D10:
19/// "`0` disables the background job", while leaving the admin-triggered
20/// `POST /api/v1/mds/refresh` endpoint working). `main.rs`'s `tokio::spawn`
21/// for this job is gated on this function and nothing else, so a default
22/// (unconfigured) deployment makes zero outbound MDS calls — see
23/// `disabled_by_default_never_spawns` below.
24pub fn should_spawn_refresh_job(cfg: &PkiConfig) -> bool {
25    cfg.mds_enabled && cfg.mds_refresh_interval_secs > 0
26}
27
28/// Compute the startup jitter (seconds) for the job's first fire, so
29/// replicas that start at the same instant do not all hit the MDS BLOB
30/// source in the same second (D10: "jittered"). Bounded to 10% of the
31/// refresh interval, capped at one hour, and clamped to at least 1 second so
32/// the modulus below is never a divide-by-zero.
33///
34/// `entropy` is injected so this stays a pure, deterministically-testable
35/// function — the production caller passes `Uuid::new_v4().as_u128()`
36/// (`uuid` is already a dependency of this binary; no new crate needed for
37/// randomness).
38pub fn jitter_secs(interval_secs: u64, entropy: u128) -> u64 {
39    let cap = interval_secs.saturating_div(10).clamp(1, 3600);
40    (entropy % cap as u128) as u64
41}

43#[cfg(test)]
44mod tests {
45    use super::*;
46
47    fn cfg(mds_enabled: bool, interval: u64) -> PkiConfig {
48        PkiConfig {
49            mds_enabled,
50            mds_refresh_interval_secs: interval,
51            ..Default::default()
52        }
53    }
54
55    #[test]
56    fn disabled_by_default_never_spawns() {
57        // The exact posture the wave spec requires: PkiConfig::default()
58        // (what an unconfigured deployment gets) must gate the background
59        // job off, so main.rs makes zero outbound MDS calls out of the box.
60        assert!(!should_spawn_refresh_job(&PkiConfig::default()));
61    }
62
63    #[test]
64    fn enabled_with_positive_interval_spawns() {
65        assert!(should_spawn_refresh_job(&cfg(true, 604_800)));
66    }
67
68    #[test]
69    fn enabled_with_zero_interval_does_not_spawn() {
70        // D10: "0 disables the background job" even with the master switch on.
71        assert!(!should_spawn_refresh_job(&cfg(true, 0)));
72    }
73
74    #[test]
75    fn disabled_with_positive_interval_does_not_spawn() {
76        assert!(!should_spawn_refresh_job(&cfg(false, 604_800)));
77    }
78
79    #[test]
80    fn jitter_is_bounded_by_the_one_hour_cap() {
81        for entropy in [0u128, 1, 999_999, u128::MAX] {
82            let j = jitter_secs(604_800, entropy);
83            assert!(j < 3600, "jitter must stay within the 1h cap, got {j}");
84        }
85    }
86
87    #[test]
88    fn jitter_cap_clamps_to_at_least_one_second() {
89        // A pathologically small interval (only realistic in a test) must
90        // not divide-by-zero or produce an empty modulus range.
91        let j = jitter_secs(1, 42);
92        assert_eq!(j, 0, "cap clamps to 1, so entropy % 1 == 0");
93    }
94
95    #[test]
96    fn jitter_is_a_pure_function_of_its_inputs() {
97        assert_eq!(jitter_secs(604_800, 12345), jitter_secs(604_800, 12345));
98    }
99}
