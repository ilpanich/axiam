1//! Choosing the transport for outbound delivery and transactional mail
2//! (T23.8.1, G-8, D-59).
3//!
4//! With the broker (`AXIAM__AMQP__ENABLED=true`, the default) the four outbound
5//! kinds ride durable AMQP queues and mail rides `axiam.mail.outbound`. Without
6//! it they ride the in-process dispatcher and the in-process mail channel of
7//! `axiam-amqp`. The composition root picks **once, at start-up**; every
8//! producer keeps the same port (`Arc<dyn OutboundPublisher>`, `MailPublisher`)
9//! and every deliverer is the same one, so nothing downstream branches on the
10//! profile.
11
12use std::collections::HashMap;
13use std::sync::Arc;
14use std::time::Duration;

18    InProcessShutdown, MailOutboundPublisher, OutboundDeliverers, OutboundRetryConfig,
19    spawn_in_process_consumer, spawn_outbound_consumer,
20};
21use axiam_core::error::AxiamResult;
22use axiam_core::models::mail::OutboundMailMessage;
23use axiam_core::outbound::{OutboundKind, OutboundPublisher};
24use axiam_core::repository::{AuditLogRepository, MailPublisher};

26/// The transport behind the four outbound kinds.
27pub enum OutboundTransport {
28    /// Durable AMQP queues: one publisher channel per kind, a supervised
29    /// consumer per kind.
30    Amqp {
31        /// The shared broker connection.
32        manager: Arc<AmqpManager>,
33        /// The publisher per kind, kept so the kind's consumer can use it for
34        /// its TTL-delayed retry republish.
35        publishers: HashMap<OutboundKind, AmqpOutboundPublisher>,
36    },
37    /// The minimal profile: bounded in-process queues, lost on restart.
38    InProcess(InProcessOutbound),
39}
40
41impl OutboundTransport {
42    /// The broker-backed transport.
43    pub fn amqp(manager: Arc<AmqpManager>) -> Self {
44        Self::Amqp {
45            manager,
46            publishers: HashMap::new(),
47        }
48    }
49
50    /// The broker-less transport.
51    pub fn in_process() -> Self {
52        Self::InProcess(InProcessOutbound::new())
53    }
54
55    /// The handle the teardown stops the consumers with. Call it once every
56    /// consumer has been spawned.
57    pub fn shutdown(&mut self) -> OutboundShutdown {
58        match self {
59            Self::Amqp { .. } => OutboundShutdown::Broker,
60            Self::InProcess(hub) => OutboundShutdown::InProcess(hub.shutdown()),
61        }
62    }
63
64    /// The publisher producers of `kind` hold. Call it once per kind, before
65    /// [`Self::spawn_consumer`]: with AMQP it opens that kind's confirm-enabled
66    /// publisher channel (a failure here is a boot failure, as it always was).
67    pub async fn publisher(&mut self, kind: OutboundKind) -> Arc<dyn OutboundPublisher> {
68        match self {
69            Self::Amqp {
70                manager,
71                publishers,
72            } => {
73                let channel = manager
74                    .create_publisher_channel()
75                    .await
76                    .unwrap_or_else(|e| {
77                        panic!("Failed to create the AMQP {kind} publisher channel: {e}")
78                    });
79                let publisher = AmqpOutboundPublisher::new(channel);
80                publishers.insert(kind, publisher.clone());
81                Arc::new(publisher)
82            }
83            Self::InProcess(hub) => Arc::new(hub.publisher()),
84        }
85    }
86
87    /// Start the consumer of `kind`: the supervised AMQP loop, or the in-process
88    /// one. `audit_repo` receives the same `<slug>.delivery_*` rows either way.
89    pub fn spawn_consumer<A>(
90        &mut self,
91        kind: OutboundKind,
92        deliverers: OutboundDeliverers,
93        audit_repo: A,
94        cfg: OutboundRetryConfig,
95    ) where
96        A: AuditLogRepository + 'static,
97    {
98        match self {
99            Self::Amqp {
100                manager,
101                publishers,
102            } => {
103                let publisher = publishers
104                    .get(&kind)
105                    .unwrap_or_else(|| panic!("no AMQP publisher was created for kind {kind}"))
106                    .clone();
107                // CQ-B53: the supervisor never exits the process; see
108                // `spawn_outbound_consumer` for the reconnect backoff.
109                spawn_outbound_consumer(
110                    Arc::clone(manager),
111                    kind,
112                    deliverers,
113                    publisher,
114                    audit_repo,
115                    cfg,
116                );
117            }
118            Self::InProcess(hub) => {
119                let end = hub
120                    .take_consumer_end(kind)
121                    .unwrap_or_else(|| panic!("the in-process {kind} queue was already consumed"));
122                spawn_in_process_consumer(end, &deliverers, audit_repo, cfg)
123                    .unwrap_or_else(|e| panic!("cannot start the in-process {kind} consumer: {e}"));
124            }
125        }
126    }
127}

129/// What the teardown calls to account for the outbound deliveries still held
130/// in memory (P23W5-A4).
131pub enum OutboundShutdown {
132    /// The broker holds the queues durably; nothing is in memory to account for.
133    Broker,
134    /// The minimal profile's in-process dispatcher.
135    InProcess(InProcessShutdown),
136}
137
138impl OutboundShutdown {
139    /// Stop taking outbound work and write a `delivery_abandoned` audit row for
140    /// every delivery still queued or waiting for a retry, waiting at most
141    /// `deadline`. Returns the kinds that had not finished by then.
142    pub async fn stop(self, deadline: Duration) -> Vec<OutboundKind> {
143        match self {
144            Self::Broker => Vec::new(),
145            Self::InProcess(shutdown) => shutdown.stop(deadline).await,
146        }
147    }
148}

150/// The mail publisher every producer holds: the broker's, or the in-process
151/// channel's. One concrete, cheaply cloned type, so the REST state, the audit
152/// notification sink, the CIBA notifier and the cleanup task need no generics
153/// of their own.
154#[derive(Clone)]
155pub enum MailTransportPublisher {
156    /// Publishes to `axiam.mail.outbound`.
157    Amqp(MailOutboundPublisher),
158    /// Sends on the in-process bounded channel (minimal profile).
159    InProcess(InProcessMailPublisher),
160}
161
162impl MailPublisher for MailTransportPublisher {
163    async fn publish(&self, msg: OutboundMailMessage) -> AxiamResult<()> {
164        match self {
165            Self::Amqp(p) => p.publish(msg).await,
166            Self::InProcess(p) => p.publish(msg).await,
167        }
168    }
169}

171#[cfg(test)]
172mod tests {
173    use super::*;
174    use axiam_amqp::in_process_mail_channel;
175    use axiam_core::models::mail::MailType;
176
177    fn mail() -> OutboundMailMessage {
178        OutboundMailMessage {
179            mail_type: MailType::PasswordReset,
180            tenant_id: uuid::Uuid::new_v4(),
181            org_id: uuid::Uuid::new_v4(),
182            user_id: uuid::Uuid::new_v4(),
183            to_address: String::new(),
184            template_context: serde_json::json!({}),
185            attempt_count: 0,
186            enqueued_at: chrono::Utc::now(),
187        }
188    }
189
190    /// The in-process arm of the mail publisher reaches the channel's queue.
191    #[tokio::test]
192    async fn the_in_process_mail_publisher_delivers_to_the_channel() {
193        let (publisher, _queue) = in_process_mail_channel();
194        let publisher = MailTransportPublisher::InProcess(publisher);
195        publisher.publish(mail()).await.unwrap();
196        assert!(
197            MailTransportPublisher::InProcess(InProcessMailPublisher::disabled())
198                .publish(mail())
199                .await
200                .is_err()
201        );
202    }
203
204    /// Every kind gets a publisher and a consumer end from the in-process
205    /// transport, and each kind's end is handed out once.
206    #[tokio::test]
207    async fn the_in_process_transport_serves_every_kind() {
208        let mut transport = OutboundTransport::in_process();
209        for kind in OutboundKind::ALL {
210            let _ = transport.publisher(*kind).await;
211        }
212        let OutboundTransport::InProcess(hub) = &mut transport else {
213            unreachable!()
214        };
215        for kind in OutboundKind::ALL {
216            assert!(hub.take_consumer_end(*kind).is_some());
217            assert!(hub.take_consumer_end(*kind).is_none());
218        }
219    }
