1//! Building the mTLS client trust store out of the organization CAs an operator
2//! flagged in the admin UI.
3//!
4//! # What this is for
5//!
6//! `AXIAM__SERVER__TLS__CLIENT_AUTH` and `AXIAM__SERVER__TLS__CLIENT_CA_PATH`
7//! are how the rustls listener is told to verify client certificates. Setting
8//! them by hand means an operator has to get a copy of their organization CA's
9//! certificate onto the server volume themselves, keep it in step with the CA
10//! they actually issue from, and remember to remove it when they rotate — for a
11//! CA that AXIAM generated, holds, and already shows them on a page.
12//!
13//! Flagging a CA
14//! ([`axiam_core::models::certificate::CaCertificate::mtls_trust_anchor`]) does
15//! that work instead: the certificate is exported at startup and the listener is
16//! pointed at it.
17//!
18//! # The private key is not involved
19//!
20//! Only `public_cert_pem` is written. The signing key stays wherever its
21//! custodian put it — Vault, under the deployment's own policy — and never
22//! reaches the volume. Nothing is weakened by the copy: a trust anchor is
23//! public by construction. It is what the server offers every client during the
24//! handshake, and every device that validates an AXIAM-issued chain already
25//! holds it.
26//!
27//! # Applying a change without a restart
28//!
29//! rustls builds its `RootCertStore` once, when the `ServerConfig` is
30//! constructed, and actix-web binds that config for the life of the process —
31//! so flagging a CA used to change what the *next* boot trusted, and the API
32//! said so.
33//!
34//! What rustls does consult per handshake is the **verifier**. A verifier that
35//! delegates to a swappable anchor set therefore gives the config something
36//! permanent to hold while what it trusts changes underneath;
37//! [`crate::tls::ReloadableClientCertVerifier`] is that, and
38//! [`TrustAnchorReload`] here is what drives it from the handler that flags a
39//! CA. The bundle on disk is still written and still read at every boot, so a
40//! restart reaches the same state by the same path.
41//!
42//! Connections already established keep the verifier they handshook with. That
43//! is correct rather than a limitation: a certificate accepted a moment ago
44//! does not become invalid mid-connection, and ending a session that is already
45//! authenticated is what session revocation is for.
46
47use std::path::{Path, PathBuf};

49use axiam_api_rest::config::{ClientAuth, TlsConfig};
50use axiam_core::models::certificate::CaCertificate;

52/// The bundle filename used when only [`TlsConfig::cert_path`] is known.
53const DEFAULT_BUNDLE_FILENAME: &str = "client-ca-bundle.pem";
54
55/// What [`plan`] decided to do.
56#[derive(Debug, Clone, PartialEq, Eq)]
57pub enum AnchorPlan {
58    /// Nothing to do. No CA is flagged, so the deployment's TLS posture is
59    /// untouched — the case every existing deployment is in.
60    NoAnchors,
61    /// CAs are flagged but there is nowhere to put the bundle: neither
62    /// `client_ca_bundle_path` nor `cert_path` is set, so no path can be
63    /// derived. Reported to the operator rather than guessed at.
64    NoBundlePath,
65    /// Write `pem` to `path`, and apply `client_auth` / `client_ca_path` as
66    /// described by the two booleans.
67    Write {
68        /// Where the bundle goes.
69        path: PathBuf,
70        /// The concatenated PEM certificates.
71        pem: String,
72        /// How many CAs went in, for the startup log line.
73        anchor_count: usize,
74        /// Whether to turn `client_auth` on. False when the operator already
75        /// chose a mode, which is never overridden.
76        set_client_auth: bool,
77        /// Whether to point `client_ca_path` at the bundle. False when the
78        /// operator named their own bundle, which is never overridden.
79        set_client_ca_path: bool,
80    },
81}
82
83/// Where the bundle should be written, given the TLS configuration.
84///
85/// An explicit `client_ca_bundle_path` wins. Otherwise the bundle goes beside
86/// the server's certificate, which is the directory an operator has already had
87/// to make readable-and-writable for TLS material — inventing a path under
88/// `/var/lib` would work on one packaging and not another.
89pub fn bundle_path(tls: &TlsConfig) -> Option<PathBuf> {
90    if let Some(ref explicit) = tls.client_ca_bundle_path {
91        return Some(explicit.clone());
92    }
93    tls.cert_path
94        .as_ref()
95        .and_then(|p| p.parent())
96        .map(|dir| dir.join(DEFAULT_BUNDLE_FILENAME))
97}

99/// Concatenate the anchors' public certificates into one PEM bundle.
100///
101/// Each certificate is newline-terminated so two PEMs cannot run together —
102/// `-----END CERTIFICATE----------BEGIN CERTIFICATE-----` parses as neither,
103/// and the resulting trust store would silently be empty.
104///
105/// A CA's `chain_pem` is **not** included. Under `vault_pki` custody
106/// `public_cert_pem` is the signing intermediate and `chain_pem` carries the
107/// root above it; adding the root to a *client* trust store would extend trust
108/// to everything else that root ever signs, which is broader than the operator
109/// flagged.
110fn bundle_pem(anchors: &[CaCertificate]) -> String {
111    let mut out = String::new();
112    for anchor in anchors {
113        let pem = anchor.public_cert_pem.trim_end();
114        if pem.is_empty() {
115            continue;
116        }
117        out.push_str(pem);
118        out.push('\n');
119    }
120    out
121}

123/// Decide what to do with the flagged CAs, without touching the filesystem.
124///
125/// Split from the doing so the decision — especially "does an operator's
126/// explicit setting survive?" — is testable without a disk or a database.
127///
128/// # Examples
129///
130/// ```
131/// use axiam_api_rest::config::{ClientAuth, TlsConfig};
132/// use axiam_server::mtls_anchors::{plan, AnchorPlan};
133///
134/// // No flagged CAs: nothing changes, which is every existing deployment.
135/// let tls = TlsConfig::default();
136/// assert_eq!(plan(&tls, &[]), AnchorPlan::NoAnchors);
137/// ```
138pub fn plan(tls: &TlsConfig, anchors: &[CaCertificate]) -> AnchorPlan {
139    if anchors.is_empty() {
140        return AnchorPlan::NoAnchors;
141    }
142    let Some(path) = bundle_path(tls) else {
143        return AnchorPlan::NoBundlePath;
144    };
145    AnchorPlan::Write {
146        path,
147        pem: bundle_pem(anchors),
148        anchor_count: anchors.len(),
149        // An operator who set `client_auth` made a decision — including the
150        // decision to leave it `off` while keeping a CA flagged for a later
151        // change. `Off` is the default, so "still Off" is the only reading
152        // available here that does not override a deliberate `required`.
153        set_client_auth: tls.client_auth == ClientAuth::Off,
154        // Likewise: a bundle the operator named is theirs. Pointing at ours
155        // instead would replace a trust store they curated with one assembled
156        // from database rows.
157        set_client_ca_path: tls.client_ca_path.is_none(),
158    }
159}

161/// Apply a [`AnchorPlan::Write`] to `tls`, writing the bundle to disk.
162///
163/// Returns the path written on success.
164///
165/// # Errors
166///
167/// Any I/O failure creating the directory or writing the file.
168pub fn apply(tls: &mut TlsConfig, plan: &AnchorPlan) -> std::io::Result<Option<PathBuf>> {
169    let AnchorPlan::Write {
170        path,
171        pem,
172        set_client_auth,
173        set_client_ca_path,
174        ..
175    } = plan
176    else {
177        return Ok(None);
178    };
179
180    if let Some(dir) = path.parent() {
181        std::fs::create_dir_all(dir)?;
182    }
183    write_bundle(path, pem)?;
184
185    if *set_client_auth {
186        // `Optional`, never `Required`. A server that suddenly refuses every
187        // client without a certificate would lock every browser out of the
188        // admin UI the moment an operator flagged a CA — including the operator
189        // who flagged it. `Optional` verifies a certificate when one is offered
190        // and lets password and passkey logins carry on working, which is the
191        // posture an IoT deployment actually wants: devices authenticate by
192        // certificate, humans do not.
193        tls.client_auth = ClientAuth::Optional;
194    }
195    if *set_client_ca_path {
196        tls.client_ca_path = Some(path.clone());
197    }
198    Ok(Some(path.clone()))
199}
200
201/// Write `pem` to `path`, replacing whatever was there.
202///
203/// The bundle is rebuilt from the database on every boot, so a partial write
204/// from a previous crash is not something to preserve. It is also not secret —
205/// these are the certificates the server hands to every client — so no special
206/// permissions are set beyond the process umask.
207fn write_bundle(path: &Path, pem: &str) -> std::io::Result<()> {
208    std::fs::write(path, pem)
209}

215/// Rebuilds the trust anchor bundle from the database and installs it on the
216/// running listener.
217///
218/// Registered in `AppState` at startup as the implementation of
219/// [`axiam_api_rest::TrustAnchorReloader`]. It holds the CA repository and the
220/// resolved bundle path — the same path the boot sequence wrote, so a reload
221/// and a restart converge on the same file.
222pub struct TrustAnchorReload<C: surrealdb::Connection> {
223    ca_repo: axiam_db::SurrealCaCertificateRepository<C>,
224    /// Where the bundle is written. `None` when no path could be derived, in
225    /// which case there is nothing on disk for a restart to read and the reload
226    /// reports that a restart is required.
227    bundle_path: Option<PathBuf>,
228}
229
230impl<C: surrealdb::Connection> TrustAnchorReload<C> {
231    pub fn new(
232        ca_repo: axiam_db::SurrealCaCertificateRepository<C>,
233        bundle_path: Option<PathBuf>,
234    ) -> Self {
235        Self {
236            ca_repo,
237            bundle_path,
238        }
239    }
240}
241
242impl<C: surrealdb::Connection> axiam_api_rest::TrustAnchorReloader for TrustAnchorReload<C> {
243    fn reload<'a>(
244        &'a self,
245    ) -> std::pin::Pin<
246        Box<
247            dyn std::future::Future<Output = axiam_core::error::AxiamResult<Option<usize>>>
248                + Send
249                + 'a,
250        >,
251    > {
252        Box::pin(async move {
253            use axiam_core::repository::CaCertificateRepository as _;
254
255            // Re-read rather than accept a delta from the caller. The listener's
256            // trust store is a whole set, the database is what a restart would
257            // read, and rebuilding from it is what keeps the two from drifting
258            // — including when two administrators toggle two CAs at once.
259            let anchors = self.ca_repo.list_mtls_trust_anchors().await?;
260            let pem = bundle_pem(&anchors);
261
262            // Write the bundle first. If the process dies between the write and
263            // the swap, the next boot reads the file and reaches the same state;
264            // the other order would leave a listener trusting a set that no
265            // restart could reproduce.
266            if let Some(path) = self.bundle_path.as_ref() {
267                if let Some(dir) = path.parent() {
268                    std::fs::create_dir_all(dir).map_err(|e| {
269                        axiam_core::error::AxiamError::Internal(format!(
270                            "failed to create the trust anchor bundle directory {}: {e}",
271                            dir.display()
272                        ))
273                    })?;
274                }
275                write_bundle(path, &pem).map_err(|e| {
276                    axiam_core::error::AxiamError::Internal(format!(
277                        "failed to write the trust anchor bundle {}: {e}",
278                        path.display()
279                    ))
280                })?;
281            }
282
283            crate::tls::reload_trust_anchors(&pem).map_err(|e| {
284                axiam_core::error::AxiamError::Internal(format!(
285                    "failed to install the trust anchor bundle on the listener: {e}"
286                ))
287            })
288        })
289    }
290}

300    fn anchor(pem: &str) -> CaCertificate {
301        CaCertificate {
302            id: Uuid::new_v4(),
303            organization_id: Uuid::new_v4(),
304            tenant_id: None,
305            parent_ca_id: None,
306            subject: "CN=Test Root".into(),
307            public_cert_pem: pem.into(),
308            chain_pem: None,
309            fingerprint: "ff".into(),
310            key_algorithm: KeyAlgorithm::Ed25519,
311            not_before: Utc::now() - Duration::days(1),
312            not_after: Utc::now() + Duration::days(365),
313            status: CertificateStatus::Active,
314            encrypted_private_key: None,
315            key_custody: CaKeyCustody::Vault,
316            key_locator: Some("axiam/ca/x".into()),
317            mtls_trust_anchor: true,
318            created_at: Utc::now(),
319        }
320    }

322    fn tls_with_cert(dir: &Path) -> TlsConfig {
323        TlsConfig {
324            enabled: true,
325            cert_path: Some(dir.join("server.pem")),
326            key_path: Some(dir.join("server.key")),
327            ..TlsConfig::default()
328        }

364    #[test]

366        // The important one. Someone who set `required` chose a stricter posture
367        // than this feature would install, and silently relaxing it to
368        // `optional` would be a security regression performed by a convenience.
369        let mut tls = tls_with_cert(Path::new("/etc/axiam/tls"));

391    #[test]
392    fn certificates_are_newline_separated_so_they_cannot_run_together() {
393        // Without the separator the concatenation reads
394        // `-----END CERTIFICATE----------BEGIN CERTIFICATE-----`, which parses
395        // as neither certificate and yields a silently empty trust store.
396        let pem = bundle_pem(&[anchor("-----A-----"), anchor("-----B-----")]);
397        assert_eq!(pem, "-----A-----\n-----B-----\n");
398    }

412    #[test]
413    fn apply_writes_the_bundle_and_points_the_listener_at_it() {
414        let dir = std::env::temp_dir().join(format!("axiam-mtls-{}", Uuid::new_v4()));
415        let mut tls = tls_with_cert(&dir);
416        let p = plan(&tls, &[anchor("-----A-----")]);
417        let written = apply(&mut tls, &p).expect("write must succeed");
418
419        let path = written.expect("a write plan returns its path");
420        assert_eq!(std::fs::read_to_string(&path).unwrap(), "-----A-----\n");
421        // Optional, not Required — see `apply`.
422        assert_eq!(tls.client_auth, ClientAuth::Optional);
423        assert_eq!(tls.client_ca_path, Some(path));
424
425        std::fs::remove_dir_all(&dir).ok();
426    }
