1//! Building the mTLS client trust store out of the organization CAs an operator
2//! flagged in the admin UI.
3//!
4//! # What this is for
5//!
6//! `AXIAM__SERVER__TLS__CLIENT_AUTH` and `AXIAM__SERVER__TLS__CLIENT_CA_PATH`
7//! are how the rustls listener is told to verify client certificates. Setting
8//! them by hand means an operator has to get a copy of their organization CA's
9//! certificate onto the server volume themselves, keep it in step with the CA
10//! they actually issue from, and remember to remove it when they rotate — for a
11//! CA that AXIAM generated, holds, and already shows them on a page.
12//!
13//! Flagging a CA
14//! ([`axiam_core::models::certificate::CaCertificate::mtls_trust_anchor`]) does
15//! that work instead: the certificate is exported at startup and the listener is
16//! pointed at it.
17//!
18//! # The private key is not involved
19//!
20//! Only `public_cert_pem` is written. The signing key stays wherever its
21//! custodian put it — Vault, under the deployment's own policy — and never
22//! reaches the volume. Nothing is weakened by the copy: a trust anchor is
23//! public by construction. It is what the server offers every client during the
24//! handshake, and every device that validates an AXIAM-issued chain already
25//! holds it.
26//!
27//! # Why a restart is needed
28//!
29//! rustls builds its `RootCertStore` once, when the `ServerConfig` is
30//! constructed, and actix-web binds that config for the life of the process.
31//! There is no supported way to add a root to a listening server, so flagging a
32//! CA changes what the *next* boot trusts. The API says so in its response
33//! rather than letting an operator believe otherwise.
34
35use std::path::{Path, PathBuf};
36
37use axiam_api_rest::config::{ClientAuth, TlsConfig};
38use axiam_core::models::certificate::CaCertificate;
39
40/// The bundle filename used when only [`TlsConfig::cert_path`] is known.
41const DEFAULT_BUNDLE_FILENAME: &str = "client-ca-bundle.pem";
42
43/// What [`plan`] decided to do.
44#[derive(Debug, Clone, PartialEq, Eq)]
45pub enum AnchorPlan {
46    /// Nothing to do. No CA is flagged, so the deployment's TLS posture is
47    /// untouched — the case every existing deployment is in.
48    NoAnchors,
49    /// CAs are flagged but there is nowhere to put the bundle: neither
50    /// `client_ca_bundle_path` nor `cert_path` is set, so no path can be
51    /// derived. Reported to the operator rather than guessed at.
52    NoBundlePath,
53    /// Write `pem` to `path`, and apply `client_auth` / `client_ca_path` as
54    /// described by the two booleans.
55    Write {
56        /// Where the bundle goes.
57        path: PathBuf,
58        /// The concatenated PEM certificates.
59        pem: String,
60        /// How many CAs went in, for the startup log line.
61        anchor_count: usize,
62        /// Whether to turn `client_auth` on. False when the operator already
63        /// chose a mode, which is never overridden.
64        set_client_auth: bool,
65        /// Whether to point `client_ca_path` at the bundle. False when the
66        /// operator named their own bundle, which is never overridden.
67        set_client_ca_path: bool,
68    },
69}
70
71/// Where the bundle should be written, given the TLS configuration.
72///
73/// An explicit `client_ca_bundle_path` wins. Otherwise the bundle goes beside
74/// the server's certificate, which is the directory an operator has already had
75/// to make readable-and-writable for TLS material — inventing a path under
76/// `/var/lib` would work on one packaging and not another.
77fn bundle_path(tls: &TlsConfig) -> Option<PathBuf> {
78    if let Some(ref explicit) = tls.client_ca_bundle_path {
79        return Some(explicit.clone());
80    }
81    tls.cert_path
82        .as_ref()
83        .and_then(|p| p.parent())
84        .map(|dir| dir.join(DEFAULT_BUNDLE_FILENAME))
85}

87/// Concatenate the anchors' public certificates into one PEM bundle.
88///
89/// Each certificate is newline-terminated so two PEMs cannot run together —
90/// `-----END CERTIFICATE----------BEGIN CERTIFICATE-----` parses as neither,
91/// and the resulting trust store would silently be empty.
92///
93/// A CA's `chain_pem` is **not** included. Under `vault_pki` custody
94/// `public_cert_pem` is the signing intermediate and `chain_pem` carries the
95/// root above it; adding the root to a *client* trust store would extend trust
96/// to everything else that root ever signs, which is broader than the operator
97/// flagged.
98fn bundle_pem(anchors: &[CaCertificate]) -> String {
99    let mut out = String::new();
100    for anchor in anchors {
101        let pem = anchor.public_cert_pem.trim_end();
102        if pem.is_empty() {
103            continue;
104        }
105        out.push_str(pem);
106        out.push('\n');
107    }
108    out
109}

111/// Decide what to do with the flagged CAs, without touching the filesystem.
112///
113/// Split from the doing so the decision — especially "does an operator's
114/// explicit setting survive?" — is testable without a disk or a database.
115///
116/// # Examples
117///
118/// ```
119/// use axiam_api_rest::config::{ClientAuth, TlsConfig};
120/// use axiam_server::mtls_anchors::{plan, AnchorPlan};
121///
122/// // No flagged CAs: nothing changes, which is every existing deployment.
123/// let tls = TlsConfig::default();
124/// assert_eq!(plan(&tls, &[]), AnchorPlan::NoAnchors);
125/// ```
126pub fn plan(tls: &TlsConfig, anchors: &[CaCertificate]) -> AnchorPlan {
127    if anchors.is_empty() {
128        return AnchorPlan::NoAnchors;
129    }
130    let Some(path) = bundle_path(tls) else {
131        return AnchorPlan::NoBundlePath;
132    };
133    AnchorPlan::Write {
134        path,
135        pem: bundle_pem(anchors),
136        anchor_count: anchors.len(),
137        // An operator who set `client_auth` made a decision — including the
138        // decision to leave it `off` while keeping a CA flagged for a later
139        // change. `Off` is the default, so "still Off" is the only reading
140        // available here that does not override a deliberate `required`.
141        set_client_auth: tls.client_auth == ClientAuth::Off,
142        // Likewise: a bundle the operator named is theirs. Pointing at ours
143        // instead would replace a trust store they curated with one assembled
144        // from database rows.
145        set_client_ca_path: tls.client_ca_path.is_none(),
146    }
147}

149/// Apply a [`AnchorPlan::Write`] to `tls`, writing the bundle to disk.
150///
151/// Returns the path written on success.
152///
153/// # Errors
154///
155/// Any I/O failure creating the directory or writing the file.
156pub fn apply(tls: &mut TlsConfig, plan: &AnchorPlan) -> std::io::Result<Option<PathBuf>> {
157    let AnchorPlan::Write {
158        path,
159        pem,
160        set_client_auth,
161        set_client_ca_path,
162        ..
163    } = plan
164    else {
165        return Ok(None);
166    };
167
168    if let Some(dir) = path.parent() {
169        std::fs::create_dir_all(dir)?;
170    }
171    write_bundle(path, pem)?;
172
173    if *set_client_auth {
174        // `Optional`, never `Required`. A server that suddenly refuses every
175        // client without a certificate would lock every browser out of the
176        // admin UI the moment an operator flagged a CA — including the operator
177        // who flagged it. `Optional` verifies a certificate when one is offered
178        // and lets password and passkey logins carry on working, which is the
179        // posture an IoT deployment actually wants: devices authenticate by
180        // certificate, humans do not.
181        tls.client_auth = ClientAuth::Optional;
182    }
183    if *set_client_ca_path {
184        tls.client_ca_path = Some(path.clone());
185    }
186    Ok(Some(path.clone()))
187}
188
189/// Write `pem` to `path`, replacing whatever was there.
190///
191/// The bundle is rebuilt from the database on every boot, so a partial write
192/// from a previous crash is not something to preserve. It is also not secret —
193/// these are the certificates the server hands to every client — so no special
194/// permissions are set beyond the process umask.
195fn write_bundle(path: &Path, pem: &str) -> std::io::Result<()> {
196    std::fs::write(path, pem)
197}

201    use super::*;
202    use axiam_core::ca_keys::CaKeyCustody;
203    use axiam_core::models::certificate::{CertificateStatus, KeyAlgorithm};
204    use chrono::{Duration, Utc};
205    use uuid::Uuid;

207    fn anchor(pem: &str) -> CaCertificate {
208        CaCertificate {
209            id: Uuid::new_v4(),
210            organization_id: Uuid::new_v4(),
211            tenant_id: None,
212            parent_ca_id: None,
213            subject: "CN=Test Root".into(),
214            public_cert_pem: pem.into(),
215            chain_pem: None,
216            fingerprint: "ff".into(),
217            key_algorithm: KeyAlgorithm::Ed25519,
218            not_before: Utc::now() - Duration::days(1),
219            not_after: Utc::now() + Duration::days(365),
220            status: CertificateStatus::Active,
221            encrypted_private_key: None,
222            key_custody: CaKeyCustody::Vault,
223            key_locator: Some("axiam/ca/x".into()),
224            mtls_trust_anchor: true,
225            created_at: Utc::now(),
226        }
227    }
228
229    fn tls_with_cert(dir: &Path) -> TlsConfig {
230        TlsConfig {
231            enabled: true,
232            cert_path: Some(dir.join("server.pem")),
233            key_path: Some(dir.join("server.key")),
234            ..TlsConfig::default()
235        }
236    }

238    #[test]
239    fn no_flagged_cas_changes_nothing() {
240        // The property every existing deployment depends on.
241        assert_eq!(plan(&TlsConfig::default(), &[]), AnchorPlan::NoAnchors);
242    }
243
244    #[test]
245    fn flagged_cas_with_nowhere_to_write_are_reported_not_guessed() {
246        let tls = TlsConfig::default(); // no cert_path, no bundle path
247        assert_eq!(plan(&tls, &[anchor("PEM")]), AnchorPlan::NoBundlePath);
248    }
249
250    #[test]
251    fn the_bundle_lands_beside_the_servers_own_certificate_by_default() {
252        let tls = tls_with_cert(Path::new("/etc/axiam/tls"));
253        match plan(&tls, &[anchor("PEM")]) {
254            AnchorPlan::Write { path, .. } => {
255                assert_eq!(path, Path::new("/etc/axiam/tls/client-ca-bundle.pem"));
256            }
257            other => panic!("expected a write plan, got {other:?}"),
258        }
259    }

261    #[test]
262    fn an_explicit_bundle_path_wins_over_the_derived_one() {
263        let mut tls = tls_with_cert(Path::new("/etc/axiam/tls"));
264        tls.client_ca_bundle_path = Some(PathBuf::from("/srv/anchors.pem"));
265        match plan(&tls, &[anchor("PEM")]) {
266            AnchorPlan::Write { path, .. } => assert_eq!(path, Path::new("/srv/anchors.pem")),
267            other => panic!("expected a write plan, got {other:?}"),

271    #[test]
272    fn an_operators_explicit_client_auth_is_never_overridden() {
273        // The important one. Someone who set `required` chose a stricter posture
274        // than this feature would install, and silently relaxing it to
275        // `optional` would be a security regression performed by a convenience.
276        let mut tls = tls_with_cert(Path::new("/etc/axiam/tls"));
277        tls.client_auth = ClientAuth::Required;
278        match plan(&tls, &[anchor("PEM")]) {
279            AnchorPlan::Write {
280                set_client_auth, ..
281            } => assert!(!set_client_auth),
282            other => panic!("expected a write plan, got {other:?}"),
283        }
284    }

298    #[test]
299    fn certificates_are_newline_separated_so_they_cannot_run_together() {
300        // Without the separator the concatenation reads
301        // `-----END CERTIFICATE----------BEGIN CERTIFICATE-----`, which parses
302        // as neither certificate and yields a silently empty trust store.
303        let pem = bundle_pem(&[anchor("-----A-----"), anchor("-----B-----")]);
304        assert_eq!(pem, "-----A-----\n-----B-----\n");
305    }

307    #[test]
308    fn a_certificate_already_ending_in_a_newline_does_not_get_a_second() {
309        let pem = bundle_pem(&[anchor("-----A-----\n")]);
310        assert_eq!(pem, "-----A-----\n");
311    }

313    #[test]
314    fn an_empty_certificate_is_skipped_rather_than_writing_a_blank_line() {

316        assert_eq!(pem, "-----B-----\n");
317    }

319    #[test]
320    fn apply_writes_the_bundle_and_points_the_listener_at_it() {
321        let dir = std::env::temp_dir().join(format!("axiam-mtls-{}", Uuid::new_v4()));
322        let mut tls = tls_with_cert(&dir);
323        let p = plan(&tls, &[anchor("-----A-----")]);
324        let written = apply(&mut tls, &p).expect("write must succeed");
325
326        let path = written.expect("a write plan returns its path");
327        assert_eq!(std::fs::read_to_string(&path).unwrap(), "-----A-----\n");
328        // Optional, not Required — see `apply`.
329        assert_eq!(tls.client_auth, ClientAuth::Optional);
330        assert_eq!(tls.client_ca_path, Some(path));
331
332        std::fs::remove_dir_all(&dir).ok();
333    }

335    #[test]
336    fn apply_is_a_no_op_for_a_plan_that_is_not_a_write() {
337        let mut tls = TlsConfig::default();

339        assert_eq!(apply(&mut tls, &AnchorPlan::NoBundlePath).unwrap(), None);
340        assert_eq!(tls.client_auth, ClientAuth::Off);
341        assert!(tls.client_ca_path.is_none());
