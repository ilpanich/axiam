1//! The minimal profile's boot guards (T23.8.1, G-8, D-59).
2//!
3//! `AXIAM__AMQP__ENABLED=false` runs AXIAM with SurrealDB only. That is safe
4//! only under three conditions the process cannot take on trust, so each is
5//! checked at boot and refused with an error that names the profile and the
6//! fix:
7//!
8//! 1. **No broadcast.** `decision_cache_broadcast_enabled = true` asks for
9//!    cross-replica cache invalidation over AMQP; there is no transport to carry
10//!    it. ([`refuse_broadcast`])
11//! 2. **No enabled reactor registration** in the datastore, in any tenant. A
12//!    `fail_closed` reactor with no transport would deny logins in every tenant
13//!    that registered one. ([`refuse_enabled_reactors`])
14//! 3. **No second live instance.** Without the broker nothing tells replicas
15//!    about each other's mutations, so the profile is single-instance by
16//!    definition — and a replica count is the orchestrator's knowledge, not the
17//!    process's. A **singleton lease** row in the shared datastore is the only
18//!    thing that can tell ([`acquire_lease`], [`spawn_lease_renewal`]); an
19//!    instance that loses it exits rather than run beside another — through
20//!    the orderly stop, so that the audit rows it still holds are written
21//!    first ([`signal_on_lease_lost`], [`spawn_lease_loss_stop`]; T23.8.2,
22//!    P23W5-A1), with a backstop if that stop does not finish in time.
23//!
24//! Every constant is a field of [`LeaseTiming`] so the logic is tested with
25//! short values; [`LeaseTiming::PRODUCTION`] is what the server uses.
26

42/// Timing of the singleton lease.
43#[derive(Debug, Clone, Copy, PartialEq, Eq)]
44pub struct LeaseTiming {
45    /// How long a claim or renewal keeps the lease alive.
46    pub ttl: Duration,
47    /// How often the holder renews. Well under `ttl`, so one or two missed
48    /// renewals do not lose the lease.
49    pub renew_every: Duration,
50    /// How long a boot that finds another instance's live lease waits for it to
51    /// expire before refusing — a rolling update's old pod stops renewing when
52    /// it stops, and its lease expires within `ttl`.
53    pub boot_wait: Duration,
54    /// How often that wait re-tries the claim.
55    pub boot_poll: Duration,
56    /// How long an instance that lost the lease may take to stop in order —
57    /// stop accepting, finish what is in flight, write its queued audit rows —
58    /// before the backstop ends the process anyway. It runs beside the new
59    /// holder for that long at most, accepting no new connection.
60    pub lost_stop_deadline: Duration,
61}
62
63impl LeaseTiming {
64    /// TTL 30 s, renewed every 10 s, a boot waits up to 45 s (D-59); a lost
65    /// lease stops the instance within 15 s (T23.8.2) — the REST listener's
66    /// idle keep-alive (5 s) and the audit drain's bound (5 s) fit inside it.
67    pub const PRODUCTION: Self = Self {
68        ttl: Duration::from_secs(30),
69        renew_every: Duration::from_secs(10),
70        boot_wait: Duration::from_secs(45),
71        boot_poll: Duration::from_secs(1),
72        lost_stop_deadline: Duration::from_secs(15),
73    };
74
75    fn ttl_chrono(&self) -> chrono::Duration {
76        chrono::Duration::from_std(self.ttl).unwrap_or_else(|_| chrono::Duration::seconds(30))
77    }
78}
79
80impl Default for LeaseTiming {
81    fn default() -> Self {
82        Self::PRODUCTION
83    }
84}

86/// Why the minimal profile refuses to boot.
87#[derive(Debug, Clone, PartialEq, Eq)]
88pub enum ProfileRefusal {
89    /// `decision_cache_broadcast_enabled = true`.
90    BroadcastEnabled,
91    /// This many reactor registrations are enabled, across all tenants.
92    EnabledReactors(u64),
93    /// Another instance holds a live lease and did not release it in time.
94    SecondInstance {
95        /// The other instance's id.
96        holder: String,
97        /// When its lease expires unless renewed.
98        expires_at: chrono::DateTime<Utc>,
99    },
100    /// The datastore could not be asked (the lease or the registration count).
101    Datastore(String),
102}
103
104impl fmt::Display for ProfileRefusal {
105    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
106        match self {
107            ProfileRefusal::BroadcastEnabled => write!(
108                f,
109                "{SWITCH} is set together with the decision-cache broadcast \
110                 (AXIAM__AUTHZ__DECISION_CACHE_BROADCAST_ENABLED=true), which fans cache \
111                 invalidations out to other replicas over AMQP. The minimal profile has no \
112                 broker and is single-instance, so there is nothing to broadcast to. Unset \
113                 the broadcast, or run the full profile (AXIAM__AMQP__ENABLED=true)."
114            ),
115            ProfileRefusal::EnabledReactors(count) => write!(
116                f,
117                "{SWITCH} (the minimal profile has no reactor transport) but the datastore \
118                 holds {count} enabled reactor registration(s). A fail_closed reactor with no \
119                 transport denies login.post_auth, user.pre_create, user.pre_update and \
120                 grant.pre_assign in every tenant that registered one. Disable or delete those \
121                 registrations (PUT /api/v1/reactors/{{id}} with enabled: false) and start \
122                 again, or run the full profile (AXIAM__AMQP__ENABLED=true)."
123            ),
124            ProfileRefusal::SecondInstance { holder, expires_at } => write!(
125                f,
126                "{SWITCH} (the minimal profile is single-instance) but another instance \
127                 ({holder}) holds the datastore lease until {expires_at} and did not release \
128                 it while this one waited. Stop the other instance, or run the full profile \
129                 (AXIAM__AMQP__ENABLED=true) with a broker to scale beyond one."
130            ),
131            ProfileRefusal::Datastore(reason) => write!(
132                f,
133                "{SWITCH}: the minimal profile's start-up checks could not read the \
134                 datastore: {reason}"
135            ),
136        }
137    }
138}

142/// Refuse the broadcast: it has no transport in the minimal profile.
143pub fn refuse_broadcast(broadcast_enabled: bool) -> Result<(), ProfileRefusal> {
144    if broadcast_enabled {
145        Err(ProfileRefusal::BroadcastEnabled)
146    } else {
147        Ok(())
148    }
149}

151/// Refuse while any reactor registration is enabled, in any tenant.
152pub async fn refuse_enabled_reactors<R: ReactorRepository>(
153    reactors: &R,
154) -> Result<(), ProfileRefusal> {
155    match reactors.count_enabled().await {
156        Ok(0) => Ok(()),
157        Ok(count) => Err(ProfileRefusal::EnabledReactors(count)),
158        Err(e) => Err(ProfileRefusal::Datastore(e.to_string())),
159    }
160}

162/// Claim the singleton lease for `instance_id`, waiting up to
163/// [`LeaseTiming::boot_wait`] for another instance's live lease to expire.
164pub async fn acquire_lease<C: Connection>(
165    leases: &SurrealMinimalProfileLeaseRepository<C>,
166    instance_id: &str,
167    timing: LeaseTiming,
168) -> Result<(), ProfileRefusal> {
169    let started = tokio::time::Instant::now();
170    let mut warned = false;
171    loop {
172        match leases
173            .claim(instance_id, Utc::now(), timing.ttl_chrono())
174            .await
175        {
176            Ok(LeaseClaim::Acquired) => return Ok(()),
177            Ok(LeaseClaim::Held { holder, expires_at }) => {
178                let waited = started.elapsed();
179                if waited >= timing.boot_wait {
180                    return Err(ProfileRefusal::SecondInstance { holder, expires_at });
181                }
182                if !warned {
183                    // Once per boot, not once per poll.
184                    warn!(
185                        %holder,
186                        %expires_at,
187                        wait_secs = timing.boot_wait.as_secs_f64(),
188                        "another instance holds the minimal-profile lease; waiting for it to \
189                         expire (a rolling update's old instance stops renewing when it stops)"
190                    );
191                    warned = true;
192                }
193                tokio::time::sleep(timing.boot_poll.min(timing.boot_wait - waited)).await;
194            }
195            Err(e) => return Err(ProfileRefusal::Datastore(e.to_string())),
196        }
197    }
198}

200/// A reaction to a lost lease. [`spawn_lease_renewal`] calls one when its
201/// renewal finds the lease taken; the composition root passes
202/// [`signal_on_lease_lost`], and keeps [`exit_on_lease_lost`] as the backstop
203/// of [`spawn_lease_loss_stop`].
204pub type OnLeaseLost = Arc<dyn Fn() + Send + Sync>;
205
206/// The backstop: exit the process non-zero at once (the orchestrator restarts
207/// the instance, which then waits for the other to go).
208///
209/// Until T23.8.2 this was the reaction itself, called from the renewal task:
210/// the process ended wherever it was, and with it every audit row the
211/// middleware still had queued, a request between its write and its audit row,
212/// and a GDPR purge between the erasure and `gdpr.user_pseudonymized`
213/// (P23W5-A1). It now runs only if the orderly stop has not finished within
214/// [`LeaseTiming::lost_stop_deadline`].
215pub fn exit_on_lease_lost() -> OnLeaseLost {
216    Arc::new(|| std::process::exit(1))
217}
218
219/// The reaction the composition root gives the renewal task: raise `lost`.
220/// Nothing stops here; [`spawn_lease_loss_stop`] waits on the flag.
221pub fn signal_on_lease_lost(lost: watch::Sender<bool>) -> OnLeaseLost {
222    Arc::new(move || {
223        lost.send_replace(true);
224    })
225}
226
227/// Once `lost` is raised: log it, call `stop` (the composition root's orderly
228/// stop: the REST listener stops accepting and finishes what is in flight, the
229/// teardown after it drains the audit queue and the cleanup task, and `serve`
230/// returns an error, so the process exits non-zero), and, if the process is
231/// still here `deadline` later, run `backstop`. The composition root aborts the
232/// returned task when its teardown has finished, which disarms the backstop. A
233/// dropped `lost` sender without the flag raised ends the task quietly.
234pub fn spawn_lease_loss_stop(
235    mut lost: watch::Receiver<bool>,
236    stop: impl FnOnce() + Send + 'static,
237    deadline: Duration,
238    backstop: OnLeaseLost,
239) -> JoinHandle<()> {
240    tokio::spawn(async move {
241        if lost.wait_for(|lost| *lost).await.is_err() {
242            return;
243        }
244        error!(
245            deadline_secs = deadline.as_secs_f64(),
246            "the minimal-profile lease is lost — stopping in order (no new connections, \
247             in-flight work finished, queued audit rows written) and then exiting non-zero"
248        );
249        stop();
250        tokio::time::sleep(deadline).await;
251        error!(
252            "the orderly stop after a lost minimal-profile lease did not finish in time — \
253             exiting now"
254        );
255        backstop();
256    })
257}

259/// Renew the lease every [`LeaseTiming::renew_every`]. When a renewal reports
260/// the lease is no longer this instance's, log once at ERROR and call
261/// `on_lost`; the task then ends. A datastore error is logged and retried — the
262/// instance cannot serve without the datastore either, and the next successful
263/// renewal tells it whether it still holds the lease.
264pub fn spawn_lease_renewal<C: Connection>(
265    leases: SurrealMinimalProfileLeaseRepository<C>,
266    instance_id: String,
267    timing: LeaseTiming,
268    on_lost: OnLeaseLost,
269) -> JoinHandle<()> {
270    tokio::spawn(async move {
271        let mut ticker = tokio::time::interval(timing.renew_every);
272        ticker.tick().await; // the first tick fires at once: the claim just renewed it
273        let mut failing = false;
274        loop {
275            ticker.tick().await;
276            match leases
277                .renew(&instance_id, Utc::now(), timing.ttl_chrono())
278                .await
279            {
280                Ok(true) => {
281                    if failing {
282                        info!("minimal-profile lease renewal recovered");
283                        failing = false;
284                    }
285                }
286                Ok(false) => {
287                    error!(
288                        %instance_id,
289                        "the minimal-profile lease was taken by another instance — this instance \
290                         is no longer the only one and exits rather than run beside it \
291                         ({SWITCH} is single-instance by definition)"
292                    );
293                    on_lost();
294                    return;
295                }
296                Err(e) => {
297                    if !failing {
298                        warn!(error = %e, "minimal-profile lease renewal failed; retrying");
299                        failing = true;
300                    }
301                }
302            }
303        }
304    })
305}

307/// Everything the minimal profile checks at boot, in order: the broadcast
308/// switch, the reactor registrations, then the lease. On success the lease is
309/// held and its renewal task is returned.
310pub async fn enforce_minimal_profile<C, R>(
311    broadcast_enabled: bool,
312    reactors: &R,
313    leases: SurrealMinimalProfileLeaseRepository<C>,
314    instance_id: &str,
315    timing: LeaseTiming,
316    on_lost: OnLeaseLost,
317) -> Result<JoinHandle<()>, ProfileRefusal>
318where
319    C: Connection,
320    R: ReactorRepository,
321{
322    refuse_broadcast(broadcast_enabled)?;
323    refuse_enabled_reactors(reactors).await?;
324    acquire_lease(&leases, instance_id, timing).await?;
325    info!(
326        %instance_id,
327        ttl_secs = timing.ttl.as_secs_f64(),
328        renew_secs = timing.renew_every.as_secs_f64(),
329        "minimal profile (AXIAM__AMQP__ENABLED=false): instance lease held"
330    );
331    Ok(spawn_lease_renewal(
332        leases,
333        instance_id.to_owned(),
334        timing,
335        on_lost,
336    ))
337}
