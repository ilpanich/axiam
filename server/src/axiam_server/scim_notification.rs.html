1//! One `scim_delivery_failed` notification per target per hour (W5 F4 review,
2//! T-418, D-73).
3//!
4//! The outbound dispatcher writes `scim_push.delivery_failed` for every
5//! dead-lettered SCIM delivery, and the consumer's audit log is a
6//! [`NotifyingAuditLog`], so a tenant's notification rule for
7//! `scim_delivery_failed` sees each of those rows (D-58). A downstream that is
8//! down past its retry budget, or that refuses AXIAM's credential, dead-letters
9//! **every** reference — the whole tenant at the next reconciliation — and the
10//! rule used to mail each recipient once per row.
11//!
12//! [`ScimFailureNotificationGate`] lets one row per target per
13//! [`FAILURE_NOTIFICATION_INTERVAL_SECS`] through, claimed in the datastore
14//! with a conditional write on `scim_target_state.failure_notified_at`
15//! ([`ScimTargetStateRepository::claim_failure_notification`]), so replicas
16//! agree. Every row is still appended, and `dead_lettered_total` still counts
17//! every dead letter: the console's `state` shows the size of an outage the
18//! one mail announces.
19//!
20//! [`scim_dead_letter_audit`] builds the wrapper the composition root gives the
21//! `scim_push` consumer, so that the composition and its test are one thing.
22

34/// The [`NotificationGate`] of the `scim_push` consumer's audit log.
35pub struct ScimFailureNotificationGate<S> {
36    states: S,
37    interval_secs: i64,
38}
39
40impl<S> ScimFailureNotificationGate<S> {
41    /// A gate over the targets' delivery state, one notification per target
42    /// per [`FAILURE_NOTIFICATION_INTERVAL_SECS`].
43    pub fn new(states: S) -> Self {
44        Self {
45            states,
46            interval_secs: FAILURE_NOTIFICATION_INTERVAL_SECS,
47        }
48    }
49}
50
51impl<S: ScimTargetStateRepository + 'static> NotificationGate for ScimFailureNotificationGate<S> {
52    fn admit<'a>(
53        &'a self,
54        entry: &'a CreateAuditLogEntry,
55    ) -> Pin<Box<dyn Future<Output = bool> + Send + 'a>> {
56        Box::pin(async move {
57            // Only the dead letter is coalesced; any other notifiable row this
58            // log ever writes goes through as it did.
59            if entry.action != SCIM_DEAD_LETTER_AUDIT_ACTION {
60                return true;
61            }
62            // The dispatcher's row names the target as its resource. One that
63            // does not cannot be coalesced, and is not a dead letter this
64            // dispatcher wrote: no notification.
65            let Some(target_id) = entry.resource_id else {
66                return false;
67            };
68            match self
69                .states
70                .claim_failure_notification(
71                    entry.tenant_id,
72                    target_id,
73                    chrono::Utc::now(),
74                    self.interval_secs,
75                )
76                .await
77            {
78                Ok(claimed) => claimed,
79                Err(error) => {
80                    // Fail closed into silence, not into a flood: the row is
81                    // appended either way and `state` still counts it.
82                    tracing::warn!(
83                        %error,
84                        tenant_id = %entry.tenant_id,
85                        %target_id,
86                        "the SCIM failure notification could not be claimed; not notifying"
87                    );
88                    false
89                }
90            }
91        })
92    }
93}

95/// The audit log the `scim_push` consumer writes through: every row appended
96/// to `audit`, a dead letter handed to `sink` (the tenant's notification rules)
97/// at most once per target per hour.
98pub fn scim_dead_letter_audit<A, T, S>(
99    audit: A,
100    sink: Arc<dyn AuditEventSink>,
101    tenants: T,
102    states: S,
103) -> NotifyingAuditLog<A, T>
104where
105    S: ScimTargetStateRepository + 'static,
106{
107    NotifyingAuditLog::new(
108        audit,
109        sink,
110        tenants,
111        Arc::new(ScimFailureNotificationGate::new(states)),
112    )
113}
