1//! Optional direct-TLS support for the REST API listener (ASVS V9.1.2/V9.1.3).
2//!
3//! TLS termination at a proxy/load balancer remains the recommended deployment
4//! pattern (D-06). This module provides an *opt-in* alternative for deployments
5//! that terminate TLS in-process: when `server.tls.enabled` is set, the server
6//! builds a rustls [`ServerConfig`] restricted to **TLS 1.3 only**. Restricting
7//! to TLS 1.3 also satisfies V9.1.3 — every TLS 1.3 cipher suite is
8//! ASVS-approved, so no manual cipher-suite filtering is required.
9//!
10//! The `ring` crypto provider is selected explicitly (rather than relying on a
11//! process-default provider) so `build_rustls_server_config` is self-contained
12//! and deterministic regardless of what other crates in the tree pull in.
13

38/// The ALPN protocol list the rustls listener is built with.
39///
40/// This is exactly what the actix-web rustls bind ends up advertising, and it
41/// is **not** configurable — see [`reject_unsupported_http2_knob`] for the
42/// source-level proof.
43fn alpn_protocols() -> Vec<Vec<u8>> {

47/// Reject `server.tls.http2 = false`, which the native listener **cannot**
48/// honour.
49///
50/// # Why this is a hard error rather than a warning (G8)
51///
52/// The knob used to narrow the rustls `ServerConfig`'s ALPN list to
53/// `http/1.1`, log a warning, and then serve h2 anyway. That is the worst of
54/// both worlds: an operator (or a benchmark cell) can *believe* the listener is
55/// HTTP/1.1-only while every client still negotiates h2. Startup now fails, in
56/// the same fail-fast style as every other misconfiguration in this module.
57///
58/// The knob is unimplementable on this bind, verified against the exact
59/// versions in `Cargo.lock` (actix-web 4.14.0 / actix-http 3.13.1 /
60/// rustls 0.23.42):
61///
62/// * `actix_web::HttpServer::bind_rustls_0_23` (actix-web-4.14.0
63///   `src/server.rs:587`) delegates to `listen_rustls_0_23_inner`
64///   (`src/server.rs:1006`), which finishes with
65///   `.rustls_0_23_with_config(config, acceptor_config)`.
66/// * `actix_http::HttpService::rustls_0_23_with_config` (actix-http-3.13.1
67///   `src/service.rs:735`) then does, unconditionally:
68///
69///   ```text
70///   let mut protos = vec![b"h2".to_vec(), b"http/1.1".to_vec()];   // :747
71///   protos.extend_from_slice(&config.alpn_protocols);              // :748
72///   config.alpn_protocols = protos;                                // :749
73///   ```
74///
75///   so `h2` always lands at index 0 of the effective list. Every actix TLS
76///   bind documents this ("ALPN protocols "h2" and "http/1.1" are added to any
77///   configured ones") and every one of them shares the prepend.
78/// * rustls picks the ALPN protocol by **server** preference order —
79///   `our_protocols.iter().find(|ours| their_protocols.contains(ours))` in
80///   rustls-0.23.42 `src/server/hs.rs:99-108` — so index 0 wins for any client
81///   offering h2. Appending to the list can never outrank the prepended `h2`.
82/// * The h2 support cannot be compiled out either: actix-web's `rustls-0_23`
83///   feature *implies* `http2` (actix-web-4.14.0 `Cargo.toml`,
84///   `rustls-0_23 = ["__tls", "http2", ...]`), and the ALPN prepend lives in a
85///   module gated only on `rustls-0_23`, not on `http2`.
86///
87/// A genuinely `http/1.1`-only TLS 1.3 listener is therefore available only by:
88/// 1. fronting the plaintext bind with the `tls13-h1` nginx edge
89///    (`benchmarks/targets/axiam/tls/tls13-h1.conf`) — what the G8 conviction
90///    cell uses; or
91/// 2. abandoning `actix_web::HttpServer` for this bind and driving
92///    `actix_http::H1Service::rustls_0_23` (actix-http-3.13.1
93///    `src/h1/service.rs:371`) directly on an `actix_server::Server` — that
94///    constructor is the one rustls service factory in the tree that does
95///    **not** touch `alpn_protocols`. See
96///    `claude_dev/b2-tls-h2-investigation.md` for why that rewrite is not
97///    justified today.
98fn reject_unsupported_http2_knob(tls: &TlsConfig) -> io::Result<()> {
99    if tls.http2 {
100        return Ok(());
101    }
102    Err(io::Error::new(
103        io::ErrorKind::Unsupported,
104        "server.tls.http2=false is not supported by the native TLS listener: \
105         actix-web's rustls bind unconditionally prepends \"h2\" to the ALPN \
106         list (actix-http HttpService::rustls_0_23_with_config), and rustls \
107         selects by server preference, so h2 would still be negotiated. \
108         Refusing to start rather than silently serving the opposite of what \
109         was configured. For an http/1.1-only TLS 1.3 endpoint, leave \
110         server.tls disabled and terminate TLS at an edge that does not enable \
111         HTTP/2 (see benchmarks/targets/axiam/tls/tls13-h1.conf and \
112         docs/security-profiles.md).",
113    ))
114}

129/// HTTP/2 tuning surface for the native TLS listener (G8/B2).
130///
131/// h2 is only ever negotiated on the rustls bind (the plaintext bind is served
132/// as HTTP/1.1 — `HttpServer::bind` never calls `listen_auto_h2c`), so these
133/// keys live next to the TLS ones and are a no-op when `server.tls.enabled` is
134/// false.
135///
136/// **Every field defaults to `None`, and `None` means "do not call the actix
137/// setter at all"** — an unset config reproduces today's behaviour byte for
138/// byte, because `actix_web::HttpServer` itself stores these as `Option`s and
139/// only forwards them to `HttpService` when `Some`
140/// (actix-web-4.14.0 `src/server.rs:1041-1048`).
141///
142/// | Env var | Type | Unset ⇒ actix default |
143/// |---|---|---|
144/// | `AXIAM__SERVER__H2__INITIAL_STREAM_WINDOW_SIZE` | `u32` bytes | 1 MiB |
145/// | `AXIAM__SERVER__H2__INITIAL_CONNECTION_WINDOW_SIZE` | `u32` bytes | 2 MiB |
146///
147/// # What is deliberately *not* exposed
148///
149/// * **`max_concurrent_streams`** — actix-http 3.13.1 never sets it. Its h2
150///   handshake builder (`src/h2/mod.rs:60-71`) calls only
151///   `initial_window_size` and `initial_connection_window_size`; the underlying
152///   `h2::server::Builder::max_concurrent_streams` (h2-0.4.15
153///   `src/server.rs:858`) exists but is unreachable, and neither
154///   `actix_http::HttpServiceBuilder` nor `actix_web::HttpServer` re-exports
155///   it. Because actix never sends `SETTINGS_MAX_CONCURRENT_STREAMS`
156///   (`h2::frame::Settings` derives `Default`, i.e. `None`), the server
157///   advertises *no* stream limit — so a stream cap is provably **not** the
158///   cause of the B2 concurrency ceiling, and inventing a knob for it here
159///   would be a lie.
160/// * **keep-alive** — `HttpServer::keep_alive` exists but is shared by h1 and
161///   h2 and does not bear on a multiplexed-connection ceiling: the single h2
162///   connection stays open for the whole run either way. Changing it would
163///   alter p0 and p2 alike, so it is not a B2 lever.
164#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, serde::Deserialize)]
165#[serde(default, deny_unknown_fields)]
166pub struct Http2Tuning {
167    /// `SETTINGS_INITIAL_WINDOW_SIZE` advertised per stream, in bytes.
168    pub initial_stream_window_size: Option<u32>,
169    /// Connection-level receive window, in bytes.
170    pub initial_connection_window_size: Option<u32>,
171}
172
173impl Http2Tuning {
174    /// Load the `server.h2` subtree using the same sources and idiom as the
175    /// main `AppConfig` loader (`config/default.toml` + `AXIAM__*` env with a
176    /// `__` separator).
177    ///
178    /// An absent subtree yields [`Self::default`] (all `None`).
179    pub fn load() -> io::Result<Self> {
180        let sources = config::Config::builder()
181            .add_source(config::File::with_name("config/default").required(false))
182            .add_source(config::Environment::with_prefix("AXIAM").separator("__"))
183            .build()
184            .map_err(|e| io::Error::other(format!("failed to load configuration: {e}")))?;
185
186        let tuning = match sources.get::<Self>("server.h2") {
187            Ok(t) => t,
188            // Nothing configured at all — keep actix's defaults.
189            Err(config::ConfigError::NotFound(_)) => Self::default(),
190            Err(e) => {
191                return Err(io::Error::new(
192                    io::ErrorKind::InvalidInput,
193                    format!("invalid server.h2 configuration: {e}"),
194                ));
195            }
196        };
197        tuning.validate()?;
198        Ok(tuning)
199    }
200
201    /// Reject window sizes HTTP/2 cannot express, so a typo fails at startup
202    /// rather than at the first h2 handshake (`h2::server::Builder` asserts
203    /// `size <= MAX_WINDOW_SIZE` and would panic inside a worker).
204    fn validate(&self) -> io::Result<()> {
205        for (name, value) in [
206            (
207                "initial_stream_window_size",
208                self.initial_stream_window_size,
209            ),
210            (
211                "initial_connection_window_size",
212                self.initial_connection_window_size,
213            ),
214        ] {
215            let Some(v) = value else { continue };
216            if v == 0 || v > MAX_H2_WINDOW_SIZE {
217                return Err(io::Error::new(
218                    io::ErrorKind::InvalidInput,
219                    format!(
220                        "server.h2.{name} must be between 1 and {MAX_H2_WINDOW_SIZE} \
221                         bytes (RFC 9113 §6.5.2); got {v}"
222                    ),
223                ));
224            }
225        }
226        Ok(())
227    }
228
229    /// True when nothing is configured, i.e. actix's defaults are untouched.
230    pub fn is_default(&self) -> bool {
231        *self == Self::default()
232    }
233
234    /// The window actix will actually use per stream (configured or default) —
235    /// for the startup log line only.
236    pub fn effective_stream_window(&self) -> u32 {
237        self.initial_stream_window_size
238            .unwrap_or(ACTIX_DEFAULT_H2_STREAM_WINDOW)
239    }
240
241    /// The window actix will actually use per connection (configured or
242    /// default) — for the startup log line only.
243    pub fn effective_connection_window(&self) -> u32 {
244        self.initial_connection_window_size
245            .unwrap_or(ACTIX_DEFAULT_H2_CONNECTION_WINDOW)
246    }
247}

249/// Build a rustls [`ClientCertVerifier`] from the configured client-CA bundle
250/// (D3 native mTLS).
251///
252/// Loads the PEM bundle at `client_ca_path` into a [`RootCertStore`] and builds
253/// a [`WebPkiClientVerifier`] over it, using the same [`CryptoProvider`] as the
254/// server config. For [`ClientAuth::Optional`] the verifier still *offers* and
255/// *verifies* client certs but permits anonymous clients
256/// (`allow_unauthenticated`); for [`ClientAuth::Required`] a verified client
257/// cert is mandatory.
258///
259/// Fails fast (aborting startup) when the CA path is unset, missing/unreadable,
260/// empty, or malformed — matching this file's existing `io::Error` style so a
261/// misconfigured mTLS server never starts.
262fn read_client_ca_roots(tls: &TlsConfig) -> io::Result<RootCertStore> {
263    let ca_path = tls.client_ca_path.as_ref().ok_or_else(|| {
264        io::Error::new(
265            io::ErrorKind::InvalidInput,
266            "server.tls.client_auth is optional/required but \
267             server.tls.client_ca_path is not set",
268        )
269    })?;
270
271    let pem = std::fs::read_to_string(ca_path).map_err(|e| {
272        io::Error::new(
273            e.kind(),
274            format!(
275                "failed to open TLS client CA bundle {}: {e}",
276                ca_path.display()
277            ),
278        )
279    })?;
280    let roots = roots_from_pem(&pem, &ca_path.display().to_string())?;
281    if roots.is_empty() {
282        return Err(io::Error::new(
283            io::ErrorKind::InvalidData,
284            format!("no client CA certificates found in {}", ca_path.display()),
285        ));
286    }
287    Ok(roots)
288}

290/// The verifier the live listener is using, and the provider to rebuild with.
291///
292/// A process-global because there is one TLS listener per process and the
293/// handler that flags a CA has no path to the `ServerConfig` actix built — it
294/// runs inside it. Set once, during `build_rustls_server_config`.
295static LIVE_VERIFIER: std::sync::OnceLock<(
296    Arc<ReloadableClientCertVerifier>,
297    Arc<rustls::crypto::CryptoProvider>,
298)> = std::sync::OnceLock::new();
299
300fn install_reloadable_verifier(
301    verifier: Arc<ReloadableClientCertVerifier>,
302    provider: Arc<rustls::crypto::CryptoProvider>,
303) {
304    // `set` fails only if called twice, which means a second listener was
305    // built. The first one is the one actix is serving on.
306    let _ = LIVE_VERIFIER.set((verifier, provider));
307}

309/// Classify a peer certificate the live listener accepted (RFC 8705 §2.2).
310///
311/// Called from the `on_connect` hook, which is the first place after the
312/// handshake where the peer chain and the live verifier are both reachable.
313/// Returns [`CertTrust::ChainedToAnchor`] without doing any work under every
314/// client-auth policy except [`ClientAuth::OptionalSelfSigned`], because under
315/// those a peer certificate exists only if webpki built a chain for it.
316///
317/// # The no-verifier case
318///
319/// Unreachable: a peer certificate implies a TLS listener, and every TLS
320/// listener this binary builds installs its verifier here. If it is ever
321/// reached anyway the answer is [`CertTrust::SelfAsserted`], the *less*
322/// privileged of the two — a certificate classified that way can authenticate
323/// only an RFC 8705 §2.2 client whose thumbprint is registered, where one
324/// wrongly classified `ChainedToAnchor` would be admitted to device
325/// authentication and to `tls_client_auth`.
326pub fn peer_certificate_trust(
327    leaf: &CertificateDer<'_>,
328    intermediates: &[CertificateDer<'_>],
329) -> CertTrust {
330    match LIVE_VERIFIER.get() {
331        Some((verifier, _)) => verifier.trust_of(leaf, intermediates),
332        None => {
333            tracing::warn!(
334                "a TLS connection presented a peer certificate but no client-certificate \
335                 verifier is installed in this process; treating it as self-asserted"
336            );
337            CertTrust::SelfAsserted
338        }
339    }
340}

342/// Install `pem` as the live client trust anchor set, without a restart.
343///
344/// Returns the number of anchors now trusted, or `None` when this process has
345/// no TLS listener with client authentication enabled — a plaintext deployment,
346/// or one whose operator set `client_auth = off`. That is not a failure: there
347/// is nothing to reload into, and saying so lets the caller report "saved, and
348/// it will apply when TLS is enabled" rather than a spurious error.
349///
350/// # Errors
351///
352/// A bundle that does not parse, or a certificate webpki refuses as a trust
353/// anchor. The live anchor set is left **unchanged** in that case: the new
354/// bundle is fully parsed and the verifier fully built before anything is
355/// swapped, so a bad reload cannot leave the listener trusting nothing.
356pub fn reload_trust_anchors(pem: &str) -> io::Result<Option<usize>> {
357    let Some((verifier, provider)) = LIVE_VERIFIER.get() else {
358        return Ok(None);
359    };
360    let roots = roots_from_pem(pem, "the mTLS trust anchor bundle")?;
361    let count = verifier.replace(roots, provider)?;
362    Ok(Some(count))
363}

369/// Read the configured certificate chain and private key and prove they match.
370///
371/// Shared by the boot path and every reload, so a certificate that would be
372/// rejected at startup is rejected identically at 3am when certbot replaces it
373/// — the alternative being two subtly different parsers and a renewal that
374/// installs something the next boot refuses.
375///
376/// # Errors
377///
378/// A missing or unreadable file (`NotFound`/`PermissionDenied`), a PEM the
379/// parser rejects, an empty chain, or a key that does not match the leaf's
380/// public key.
381fn read_certified_key(
382    cert_path: &std::path::Path,
383    key_path: &std::path::Path,
384    provider: &rustls::crypto::CryptoProvider,
385) -> io::Result<CertifiedKey> {
386    // Open the files explicitly so a missing/unreadable path yields a clean
387    // io::Error (NotFound / PermissionDenied) before any PEM parsing. Cert and
388    // key are parsed via rustls-pki-types' `PemObject` trait directly —
389    // rustls-pemfile is unmaintained (RUSTSEC-2025-0134) and is a thin wrapper
390    // over this same code.
391    let cert_file = File::open(cert_path).map_err(|e| {
392        io::Error::new(
393            e.kind(),
394            format!("failed to open TLS cert file {}: {e}", cert_path.display()),
395        )
396    })?;
397    let cert_chain: Vec<CertificateDer<'static>> = CertificateDer::pem_reader_iter(cert_file)
398        .collect::<Result<Vec<_>, _>>()
399        .map_err(|e| {
400            io::Error::new(
401                io::ErrorKind::InvalidData,
402                format!(
403                    "failed to parse TLS certificates from {}: {e}",
404                    cert_path.display()
405                ),
406            )
407        })?;
408    if cert_chain.is_empty() {
409        return Err(io::Error::new(
410            io::ErrorKind::InvalidData,
411            format!("no certificates found in {}", cert_path.display()),
412        ));
413    }
414
415    let key_file = File::open(key_path).map_err(|e| {
416        io::Error::new(
417            e.kind(),
418            format!("failed to open TLS key file {}: {e}", key_path.display()),
419        )
420    })?;
421    let key = PrivateKeyDer::from_pem_reader(key_file).map_err(|e| {
422        io::Error::new(
423            io::ErrorKind::InvalidData,
424            format!(
425                "failed to read a TLS private key from {}: {e}",
426                key_path.display()
427            ),
428        )
429    })?;
430
431    // `from_der` loads the key through the provider and compares its SPKI with
432    // the leaf's, which is the check that makes a half-written renewal — the
433    // new chain beside the old key — fail here rather than at a client's
434    // handshake.
435    CertifiedKey::from_der(cert_chain, key, provider).map_err(|e| {
436        io::Error::new(
437            io::ErrorKind::InvalidData,
438            format!("invalid TLS certificate/key pair: {e}"),
439        )
440    })
441}

443/// A [`ResolvesServerCert`] whose certificate can be replaced while the server
444/// is listening.
445///
446/// # Why this exists
447///
448/// rustls consults the resolver on every handshake but reads nothing from disk;
449/// `with_single_cert` installs an immutable `SingleCertAndKey` and actix binds
450/// the resulting config for the process's life. So the leaf a server started
451/// with is the leaf it serves forever.
452///
453/// That is fine for a certificate an operator installs by hand and lives with
454/// for a year. It is not fine for ACME: Let's Encrypt issues for 90 days,
455/// clients renew at 60, and the only remedy without this type is to restart the
456/// process every couple of months — which for an identity provider means
457/// dropping in-flight requests and re-reading every secret out of Vault on a
458/// schedule, to work around a `Vec<u8>` that could have been swapped.
459///
460/// The shape deliberately mirrors [`ReloadableClientCertVerifier`]: an
461/// `ArcSwap` consulted on the hot path, replaced wholesale by a reload that has
462/// already done all of its fallible work. There is one mechanism for "TLS
463/// material changed while we were running", not two.
464#[derive(Debug)]
465pub struct ReloadableCertResolver {
466    current: ArcSwap<CertifiedKey>,
467}
468
469impl ReloadableCertResolver {
470    /// Start serving `initial`.
471    pub fn new(initial: CertifiedKey) -> Self {
472        Self {
473            current: ArcSwap::from_pointee(initial),
474        }
475    }
476
477    /// Swap in `next`, returning the certificate that was replaced.
478    ///
479    /// Infallible by construction: every way this can fail — unreadable file,
480    /// malformed PEM, mismatched pair — has already happened (or not) in
481    /// [`read_certified_key`]. A caller that reaches this point holds a
482    /// certificate rustls has already accepted, so the listener cannot be left
483    /// serving nothing.
484    pub fn replace(&self, next: CertifiedKey) -> Arc<CertifiedKey> {
485        self.current.swap(Arc::new(next))
486    }
487
488    /// The certificate currently being served.
489    pub fn current(&self) -> Arc<CertifiedKey> {
490        self.current.load_full()
491    }
492}
493
494impl ResolvesServerCert for ReloadableCertResolver {
495    fn resolve(&self, _client_hello: ClientHello<'_>) -> Option<Arc<CertifiedKey>> {
496        // Deliberately ignores SNI, exactly as `SingleCertAndKey` does: this
497        // listener serves one deployment on one name. Returning the certificate
498        // regardless of the name offered is what lets a client connecting to
499        // `127.0.0.1` with an overridden SNI (how the edge proxy reaches the
500        // backend on the loopback bind) complete the handshake against the
501        // public leaf.
502        Some(self.current.load_full())
503    }
504}

516/// Every leaf a listener in this process is serving.
517///
518/// A process-global for the same reason [`LIVE_VERIFIER`] is one: the signal
519/// handler and poll task that trigger a reload run outside anything holding a
520/// `ServerConfig`.
521///
522/// # Why a list, and not the single slot this used to be (R-1)
523///
524/// Until R-1 there was one TLS listener per process — REST — and this was a
525/// `OnceLock` holding it. The gRPC listener now terminates TLS too, so a
526/// process can have two, and a `OnceLock` would have silently kept whichever
527/// was built first: the second listener's leaf would never be reloaded, which
528/// is the exact failure (a certificate that expires in place) the reloader
529/// exists to prevent, moved rather than fixed.
530///
531/// Registration goes through [`shared_resolver`], which returns the **existing**
532/// resolver when a leaf with the same cert and key paths is already registered.
533/// So the documented topology — both listeners on one leaf, "there is no second
534/// certificate and there must not be" (Pi runbook §14.4) — ends up with one
535/// entry and one resolver instance shared by both listeners, and a deployment
536/// that really does point the two listeners at different files ends up with two
537/// entries, both reloaded on the same `SIGHUP` and the same poll.
538static LIVE_CERT_RESOLVERS: std::sync::Mutex<Vec<LiveLeaf>> = std::sync::Mutex::new(Vec::new());

540/// The resolver serving `cert_path`/`key_path`, creating and registering one if

542///
543/// Sharing the instance (rather than building a second resolver over the same
544/// files) is what makes one reload cover both listeners: a swap on this
545/// `ArcSwap` is observed by every `ServerConfig` that holds it.
546///
547/// # Errors
548///
549/// Anything [`read_certified_key`] rejects, and only when the pair is new —
550/// a caller that finds an already-registered resolver does no file IO at all.
551fn shared_resolver(
552    cert_path: &std::path::Path,
553    key_path: &std::path::Path,
554    provider: &Arc<rustls::crypto::CryptoProvider>,
555) -> io::Result<Arc<ReloadableCertResolver>> {
556    // Poisoning would mean a panic while holding this lock; the data behind it
557    // is a plain list of handles and cannot be left half-updated, so recovering
558    // is strictly better than propagating a panic into a listener's boot.
559    let mut live = LIVE_CERT_RESOLVERS
560        .lock()
561        .unwrap_or_else(|poisoned| poisoned.into_inner());
562
563    if let Some(existing) = live
564        .iter()
565        .find(|leaf| leaf.cert_path == cert_path && leaf.key_path == key_path)
566    {
567        return Ok(Arc::clone(&existing.resolver));
568    }
569
570    let certified = read_certified_key(cert_path, key_path, provider)?;
571    let resolver = Arc::new(ReloadableCertResolver::new(certified));
572    live.push(LiveLeaf {
573        resolver: Arc::clone(&resolver),
574        provider: Arc::clone(provider),
575        cert_path: cert_path.to_path_buf(),
576        key_path: key_path.to_path_buf(),
577    });
578    Ok(resolver)
579}

581/// Re-read every configured certificate and key and install them on the live
582/// listeners, without a restart.
583///
584/// Since R-1 this covers **all** of them — the REST listener and, when it
585/// terminates TLS in-process, the gRPC one. On the documented topology the two
586/// share one leaf and therefore one registry entry, so this is one pair of file
587/// reads either way; a deployment that points them at different files gets both
588/// reloaded here, on the same trigger, rather than one of them silently never.
589///
590/// Returns `Ok(None)` when this process has no TLS listener at all — a
591/// plaintext deployment behind a terminating proxy. That is not a failure:
592/// there is nothing to reload into, and saying so lets a `SIGHUP` handler log
593/// "no TLS listener" rather than an error an operator would go looking for.
594///
595/// Returns `Ok(Some(false))` when every registered pair parsed but is
596/// byte-identical to what is already being served, so a poll that fires between
597/// renewals stays silent instead of logging a reload an hour, forever;
598/// `Ok(Some(true))` when at least one leaf actually changed.
599///
600/// # Errors
601///
602/// Anything [`read_certified_key`] rejects, for any registered leaf. **Every
603/// live certificate is left unchanged in that case** — each pair is fully read,
604/// parsed and consistency-checked before anything is swapped, so a renewal
605/// caught half-written (certbot writes the chain and the key as two separate
606/// operations) fails this call and is retried on the next tick rather than
607/// taking a listener down. The other leaves are still attempted before the
608/// error is returned: one unreadable pair must not stop the leaf that *was*
609/// renewed from being installed.
610pub fn reload_leaf_certificate() -> io::Result<Option<bool>> {
611    // Cloned out from under the lock: `reload_into` does file IO, and holding
612    // a process-global mutex across it would let a slow or hung disk block
613    // every listener's boot path.
614    let live: Vec<LiveLeaf> = LIVE_CERT_RESOLVERS
615        .lock()
616        .unwrap_or_else(|poisoned| poisoned.into_inner())
617        .clone();
618
619    if live.is_empty() {
620        return Ok(None);
621    }
622
623    let mut changed = false;
624    let mut first_error = None;
625    for leaf in &live {
626        match reload_into(
627            &leaf.resolver,
628            &leaf.provider,
629            &leaf.cert_path,
630            &leaf.key_path,
631        ) {
632            Ok(leaf_changed) => changed |= leaf_changed,
633            Err(e) if first_error.is_none() => first_error = Some(e),
634            Err(_) => {}
635        }
636    }
637
638    match first_error {
639        Some(e) => Err(e),
640        None => Ok(Some(changed)),
641    }
642}

644/// The reload itself, against an explicit resolver rather than the process
645/// global.
646///
647/// Split out so the behaviour that matters — swap on change, no-op when
648/// unchanged, leave the old certificate serving on a bad pair — is testable
649/// directly. Going through [`reload_leaf_certificate`] would mean reloading
650/// every leaf every other test in the binary happened to register; a mechanism
651/// whose failure mode is "serves the wrong certificate" should not be tested
652/// order-dependently.
653///
654/// # Errors
655///
656/// Anything [`read_certified_key`] rejects. The resolver is left untouched.
657fn reload_into(
658    resolver: &ReloadableCertResolver,
659    provider: &rustls::crypto::CryptoProvider,
660    cert_path: &std::path::Path,
661    key_path: &std::path::Path,
662) -> io::Result<bool> {
663    let next = read_certified_key(cert_path, key_path, provider)?;
664    // Compare the DER chain rather than file bytes: PEM whitespace, comment
665    // lines and the ordering certbot happens to write are not differences a
666    // handshake can see, and a poll that reloaded on them would churn.
667    if resolver.current().cert == next.cert {
668        return Ok(false);
669    }
670    resolver.replace(next);
671    Ok(true)
672}

674/// Report the outcome of one reload attempt.
675///
676/// A free function rather than a closure inside [`spawn_leaf_reloader`] so the
677/// mapping from outcome to log level is testable: the WARN-not-ERROR choice on
678/// a failed reload is deliberate (the previous certificate is still serving)
679/// and is exactly the kind of decision that gets "tidied" into an ERROR by
680/// someone who has not read why.
681fn log_reload_outcome(trigger: &'static str, outcome: io::Result<Option<bool>>) {
682    match outcome {
683        Ok(Some(true)) => tracing::info!(
684            trigger,
685            "TLS leaf certificate reloaded; new connections use it immediately"
686        ),
687        Ok(Some(false)) => tracing::debug!(
688            trigger,
689            "TLS leaf certificate unchanged on disk; nothing to reload"
690        ),
691        Ok(None) => tracing::debug!(
692            trigger,
693            "no direct-TLS listener in this process; nothing to reload"
694        ),
695        // WARN, not ERROR, and deliberately not fatal: the previous
696        // certificate is still serving. The likeliest cause is a renewal
697        // observed mid-write, which the next trigger resolves on its own.
698        Err(e) => tracing::warn!(
699            trigger,
700            error = %e,
701            "TLS leaf certificate reload failed; continuing to serve the \
702             previous certificate"
703        ),
704    }
705}

707/// Start the background triggers that keep the leaf certificate current.
708///
709/// Spawns, on the ambient tokio runtime:
710///
711/// * a `SIGHUP` listener — the conventional operator hook, and what an ACME
712///   `--deploy-hook` sends. Immediate.
713/// * a `stat` poll every `interval_secs`, skipped entirely when that is `0`.
714///
715/// Both funnel into [`reload_leaf_certificate`], which is a no-op when the
716/// certificate on disk is the one already being served. Calling this on a
717/// plaintext deployment is harmless: the reload reports "no TLS listener" and
718/// the tasks idle.
719///
720/// # Why both, and why the poll is not the only one
721///
722/// The signal is the correct mechanism and takes effect within milliseconds.
723/// It is also the one that silently does not happen: a deploy hook nobody
724/// wired up, a container runtime that does not forward signals to PID 1, an
725/// operator who renews by hand. The poll is what turns "the certificate
726/// expired in production" into "the certificate was replaced within the hour",
727/// and it costs one read of two small files an hour.
728pub fn spawn_leaf_reloader(interval_secs: u64) {
729    // Idempotent since R-1: both listeners' boot paths call this, because
730    // either can be the only one with TLS on. The reloader is process-wide (it
731    // walks every registered leaf), so the second call must be a no-op rather
732    // than a second SIGHUP handler and a second poll doing the same reads
733    // twice — and logging every outcome twice.
734    static SPAWNED: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
735    if SPAWNED.swap(true, std::sync::atomic::Ordering::SeqCst) {
736        tracing::debug!("TLS certificate reloader already running; not spawning a second one");
737        return;
738    }
739
740    tokio::spawn(async move {
741        let mut hangup = match tokio::signal::unix::signal(tokio::signal::unix::SignalKind::hangup())
742        {
743            Ok(s) => s,
744            Err(e) => {
745                tracing::warn!(
746                    error = %e,
747                    "could not install a SIGHUP handler; TLS certificate reload                      is available only through the periodic poll"
748                );
749                return;
750            }
751        };
752        while hangup.recv().await.is_some() {
753            log_reload_outcome("sighup", reload_leaf_certificate());
754        }
755    });
756
757    if interval_secs == 0 {
758        tracing::info!(
759            "TLS certificate reload polling is disabled              (server.tls.reload_interval_secs = 0); SIGHUP still reloads"
760        );
761        return;
762    }
763
764    tokio::spawn(async move {
765        let mut ticker = tokio::time::interval(std::time::Duration::from_secs(interval_secs));
766        // The first tick fires immediately and would re-read the certificate
767        // the boot path just read. Burn it.
768        ticker.tick().await;
769        loop {
770            ticker.tick().await;
771            log_reload_outcome("poll", reload_leaf_certificate());
772        }
773    });
774}

780/// A [`ClientCertVerifier`] whose trust anchors can be replaced while the
781/// server is listening.
782///
783/// # Why this exists
784///
785/// rustls builds its client trust store when the `ServerConfig` is constructed,
786/// and actix binds that config for the process's life. Flagging a CA as an mTLS
787/// trust anchor therefore used to take effect at the *next* boot, and the API
788/// said so rather than pretending otherwise — which is honest and still means
789/// an operator adding a device CA has to restart the server every browser
790/// session and every IoT device is currently connected to.
791///
792/// rustls consults the verifier **per handshake**, not once at construction. So
793/// a verifier that delegates to a swappable inner one gives the config
794/// something permanent to hold while the anchors behind it change. Connections
795/// already established keep the verifier they handshook with, which is correct:
796/// a certificate accepted a moment ago does not become invalid mid-connection,
797/// and revocation of an established session is what session invalidation is
798/// for.
799///
800/// # The empty state
801///
802/// [`Anchors::None`] is not "trust nothing" — it is "do not ask for a client
803/// certificate at all", the same posture as `with_no_client_auth()`. A
804/// deployment that boots with no flagged CA must behave exactly as it does
805/// today, *and* must be able to reach the anchored state without a restart.
806/// Installing this verifier unconditionally is what makes both true;
807/// `WebPkiClientVerifier` cannot express it, because it refuses to build over
808/// an empty root store.
809#[derive(Debug)]
810pub struct ReloadableClientCertVerifier {
811    anchors: ArcSwap<Anchors>,
812    /// The operator's client-authentication policy, fixed at construction.
813    ///
814    /// It is a policy decision, not a property of the anchor set: it decides
815    /// whether unauthenticated clients can connect at all
816    /// ([`ClientAuth::Required`]) and whether a certificate that chains to
817    /// nothing may complete the handshake
818    /// ([`ClientAuth::OptionalSelfSigned`]). Reloading anchors must not be
819    /// able to change either.
820    policy: ClientAuth,
821}
822
823/// What the verifier currently trusts.
824#[derive(Debug)]
825enum Anchors {
826    /// No trust anchors: no client certificate is requested.
827    None,
828    /// A webpki verifier over the current anchor set.
829    Some(Arc<dyn ClientCertVerifier>),
830}
831
832impl ReloadableClientCertVerifier {
833    /// An empty verifier that offers no client authentication, governed by
834    /// `policy` once anchors are installed.
835    ///
836    /// Takes the [`ClientAuth`] value rather than the `mandatory` boolean it
837    /// used to, because there are now two independent things the policy
838    /// decides — whether a certificate is *required* and whether an unchained
839    /// one is *accepted* — and passing them separately would admit the pair
840    /// (required, accept-self-asserted), which is not a policy any variant of
841    /// [`ClientAuth`] can express and not one this code is willing to
842    /// implement.
843    pub fn empty(policy: ClientAuth) -> Self {
844        Self {
845            anchors: ArcSwap::from_pointee(Anchors::None),
846            policy,
847        }
848    }
849
850    /// Replace the trust anchors with a verifier built over `roots`.
851    ///
852    /// An empty `roots` returns to [`Anchors::None`] rather than building a
853    /// verifier that trusts nothing: a verifier that requests a certificate and
854    /// then rejects every one is strictly worse than not requesting one, and it
855    /// is what un-flagging the last CA should produce.
856    pub fn replace(
857        &self,
858        roots: RootCertStore,
859        provider: &Arc<rustls::crypto::CryptoProvider>,
860    ) -> io::Result<usize> {
861        if roots.is_empty() {
862            self.anchors.store(Arc::new(Anchors::None));
863            return Ok(0);
864        }
865        let count = roots.len();
866        let builder =
867            WebPkiClientVerifier::builder_with_provider(Arc::new(roots), provider.clone());
868        let verifier = if self.policy == ClientAuth::Required {
869            builder.build()
870        } else {
871            builder.allow_unauthenticated().build()
872        }
873        .map_err(|e| io::Error::other(format!("failed to build client cert verifier: {e}")))?;
874
875        self.anchors.store(Arc::new(Anchors::Some(verifier)));
876        Ok(count)
877    }
878
879    /// How many anchors are currently installed.
880    pub fn anchor_count(&self) -> usize {
881        match &**self.anchors.load() {
882            Anchors::None => 0,
883            Anchors::Some(v) => v.root_hint_subjects().len(),
884        }
885    }
886
887    /// The current inner verifier, if any.
888    fn current(&self) -> Option<Arc<dyn ClientCertVerifier>> {
889        match &**self.anchors.load() {
890            Anchors::None => None,
891            Anchors::Some(v) => Some(Arc::clone(v)),
892        }
893    }
894
895    /// Would this leaf have chained to a currently-installed trust anchor?
896    ///
897    /// Answers the question `verify_client_cert` deliberately stops asking
898    /// under [`ClientAuth::OptionalSelfSigned`]. Under every other policy the
899    /// answer is known without asking — rustls only let the certificate
900    /// through *because* it chained — so this runs the extra chain validation
901    /// only in the mode that needs it.
902    ///
903    /// # Why the answer is re-derived rather than returned from the verifier
904    ///
905    /// It cannot be smuggled out of `verify_client_cert`. rustls's
906    /// `ClientCertVerified` is an opaque token carrying no payload, and the
907    /// verifier is handed no connection handle to key a side channel on — no
908    /// session id, no peer address, nothing that distinguishes one concurrent
909    /// handshake from another. So the trust level is recomputed where the peer
910    /// chain is next in hand: the `on_connect` hook, through
911    /// [`peer_certificate_trust`].
912    ///
913    /// Cost is one additional chain validation per mTLS connection (not per
914    /// request), and only under the new policy.
915    pub fn chains_to_anchor(
916        &self,
917        leaf: &CertificateDer<'_>,
918        intermediates: &[CertificateDer<'_>],
919    ) -> bool {
920        match self.current() {
921            Some(v) => v
922                .verify_client_cert(leaf, intermediates, rustls::pki_types::UnixTime::now())
923                .is_ok(),
924            // No anchors installed: nothing to chain to. Reporting `false` is
925            // both the truth and the fail-closed answer.
926            None => false,
927        }
928    }
929
930    /// Classify a peer certificate rustls has already accepted.
931    ///
932    /// Under [`ClientAuth::Off`], [`ClientAuth::Optional`] and
933    /// [`ClientAuth::Required`] a peer certificate exists only if it chained,
934    /// so the answer is [`CertTrust::ChainedToAnchor`] with no work done at
935    /// all. Under [`ClientAuth::OptionalSelfSigned`] the chain is re-checked.
936    pub fn trust_of(
937        &self,
938        leaf: &CertificateDer<'_>,
939        intermediates: &[CertificateDer<'_>],
940    ) -> CertTrust {
941        if !self.policy.accepts_self_asserted() {
942            return CertTrust::ChainedToAnchor;
943        }
944        if self.chains_to_anchor(leaf, intermediates) {
945            CertTrust::ChainedToAnchor
946        } else {
947            CertTrust::SelfAsserted
948        }
949    }
950}

952/// Reject a self-asserted certificate that is outside its validity window.
953///
954/// webpki performs this check as part of building a chain, so the branch that
955/// accepts a certificate *without* building one silently loses it. That would
956/// quietly turn "the certificate expired, rotate it" from an enforced rule into
957/// advice, and only for the clients whose certificates nobody else vouches for.
958///
959/// # What is deliberately not checked here
960///
961/// The **self-signature**. Under RFC 8705 §2.2 the certificate's identity is
962/// the SHA-256 of its DER, and possession of the matching private key is proven
963/// by TLS 1.3's `CertificateVerify` over the handshake transcript — which
964/// rustls checks regardless. A certificate whose self-signature did not verify
965/// would still have to be byte-identical to one an administrator registered
966/// before it authenticated anything, so verifying it would refuse a strictly
967/// empty set of attacks. Validity dates are different: they are the one
968/// property of the certificate body that an operator relies on being enforced.
969fn self_asserted_is_within_validity(
970    leaf: &CertificateDer<'_>,
971    now: rustls::pki_types::UnixTime,
972) -> Result<(), rustls::Error> {
973    use rustls::CertificateError;
974
975    let (_, cert) = x509_parser::prelude::parse_x509_certificate(leaf.as_ref())
976        .map_err(|_| rustls::Error::InvalidCertificate(CertificateError::BadEncoding))?;
977
978    // `UnixTime::as_secs` is seconds since the epoch; `ASN1Time::timestamp` is
979    // the same scale, signed because it can predate 1970.
980    let now = i64::try_from(now.as_secs())
981        .map_err(|_| rustls::Error::InvalidCertificate(CertificateError::BadEncoding))?;
982    let validity = cert.validity();
983    if now < validity.not_before.timestamp() {
984        return Err(rustls::Error::InvalidCertificate(
985            CertificateError::NotValidYet,
986        ));
987    }
988    if now > validity.not_after.timestamp() {
989        return Err(rustls::Error::InvalidCertificate(CertificateError::Expired));
990    }
991    Ok(())
992}

994impl ClientCertVerifier for ReloadableClientCertVerifier {
995    fn offer_client_auth(&self) -> bool {
996        // Consulted per handshake, which is the whole mechanism: the answer
997        // changes the moment `replace` installs an anchor set.
998        self.current().is_some()
999    }
1000
1001    fn client_auth_mandatory(&self) -> bool {
1002        // Only meaningful while anchors exist. Answering `true` with none
1003        // installed would refuse every connection to a server that cannot
1004        // verify anybody — a self-inflicted outage on a deployment that
1005        // un-flagged its last CA.
1006        self.policy == ClientAuth::Required && self.current().is_some()
1007    }
1008
1009    fn root_hint_subjects(&self) -> &[rustls::DistinguishedName] {
1010        // Cannot borrow through the ArcSwap guard, and the hint is advisory:
1011        // it tells a client which issuers the server would accept, and a client
1012        // that offers a certificate anyway is verified normally by
1013        // `verify_client_cert`. Returning nothing costs a round trip in the
1014        // worst case and never accepts anything it should not.
1015        &[]
1016    }
1017
1018    fn verify_client_cert(
1019        &self,
1020        end_entity: &CertificateDer<'_>,
1021        intermediates: &[CertificateDer<'_>],
1022        now: rustls::pki_types::UnixTime,
1023    ) -> Result<rustls::server::danger::ClientCertVerified, rustls::Error> {
1024        let Some(v) = self.current() else {
1025            // Unreachable while rustls honours `offer_client_auth`, and a
1026            // refusal rather than an acceptance if it ever does not.
1027            return Err(rustls::Error::General(
1028                "no client trust anchors are configured".into(),
1029            ));
1030        };
1031
1032        let chained = v.verify_client_cert(end_entity, intermediates, now);
1033        if chained.is_ok() || !self.policy.accepts_self_asserted() {
1034            // Every policy that existed before `OptionalSelfSigned` takes this
1035            // path and only this path: webpki's verdict, unmodified.
1036            return chained;
1037        }
1038
1039        // RFC 8705 §2.2 (`ClientAuth::OptionalSelfSigned` only). The
1040        // certificate chains to nothing, which for a self-signed client
1041        // credential is not a defect but the design. Accept it so the request
1042        // reaches the application, which is the only layer that can decide
1043        // whether *this* client is one whose thumbprint was registered —
1044        // `axiam_oauth2::mtls::authenticate_mtls_client`. Everything else
1045        // treats it as if no certificate had been presented; see
1046        // [`CertTrust::SelfAsserted`].
1047        self_asserted_is_within_validity(end_entity, now)?;
1048        tracing::debug!(
1049            "accepting a client certificate that chains to no configured anchor \
1050             (client_auth = optional_self_signed); it can authenticate only an \
1051             RFC 8705 self_signed_tls_client_auth client with this exact thumbprint"
1052        );
1053        Ok(rustls::server::danger::ClientCertVerified::assertion())
1054    }
1055
1056    fn verify_tls12_signature(
1057        &self,
1058        message: &[u8],
1059        cert: &CertificateDer<'_>,
1060        dss: &rustls::DigitallySignedStruct,
1061    ) -> Result<rustls::client::danger::HandshakeSignatureValid, rustls::Error> {
1062        match self.current() {
1063            Some(v) => v.verify_tls12_signature(message, cert, dss),
1064            None => Err(rustls::Error::General(
1065                "no client trust anchors are configured".into(),
1066            )),
1067        }
1068    }
1069
1070    fn verify_tls13_signature(
1071        &self,
1072        message: &[u8],
1073        cert: &CertificateDer<'_>,
1074        dss: &rustls::DigitallySignedStruct,
1075    ) -> Result<rustls::client::danger::HandshakeSignatureValid, rustls::Error> {
1076        match self.current() {
1077            Some(v) => v.verify_tls13_signature(message, cert, dss),
1078            None => Err(rustls::Error::General(
1079                "no client trust anchors are configured".into(),
1080            )),
1081        }
1082    }
1083
1084    fn supported_verify_schemes(&self) -> Vec<rustls::SignatureScheme> {
1085        match self.current() {
1086            Some(v) => v.supported_verify_schemes(),
1087            // The provider's full set: this is a capability advertisement, and
1088            // an empty list would make a handshake fail to negotiate rather
1089            // than fall through to the "no anchors" path above.
1090            None => rustls::crypto::ring::default_provider()
1091                .signature_verification_algorithms
1092                .supported_schemes(),
1093        }
1094    }
1095}

1097/// Build a [`RootCertStore`] from concatenated PEM.
1098///
1099/// Shared by the boot path and the reload path so a hot-reloaded anchor set is
1100/// parsed by exactly the same code as one read at startup — two parsers is how
1101/// a bundle comes to be accepted at boot and rejected on reload.
1102pub fn roots_from_pem(pem: &str, source: &str) -> io::Result<RootCertStore> {
1103    let certs: Vec<CertificateDer<'static>> = CertificateDer::pem_slice_iter(pem.as_bytes())
1104        .collect::<Result<Vec<_>, _>>()
1105        .map_err(|e| {
1106            io::Error::new(
1107                io::ErrorKind::InvalidData,
1108                format!("failed to parse client CA certificates from {source}: {e}"),
1109            )
1110        })?;
1111
1112    let mut roots = RootCertStore::empty();
1113    for cert in certs {
1114        roots.add(cert).map_err(|e| {
1115            io::Error::new(
1116                io::ErrorKind::InvalidData,
1117                format!("invalid client CA certificate in {source}: {e}"),
1118            )
1119        })?;
1120    }
1121    Ok(roots)
1122}

1124/// Build a TLS 1.3-only rustls [`ServerConfig`] from the configured PEM files.
1125///
1126/// Fails fast (returning an `io::Error` that aborts startup) when TLS is enabled
1127/// but misconfigured: a missing `cert_path`/`key_path`, an unreadable or
1128/// malformed PEM file, an empty cert chain, a cert/key mismatch, or an
1129/// unsatisfiable `http2 = false` request (see [`reject_unsupported_http2_knob`]).
1130pub fn build_rustls_server_config(tls: &TlsConfig) -> io::Result<ServerConfig> {
1131    // Checked before any file IO: an unsatisfiable ALPN request is a config
1132    // error regardless of whether the cert/key happen to be readable.
1133    reject_unsupported_http2_knob(tls)?;
1134
1135    let cert_path = tls.cert_path.as_ref().ok_or_else(|| {
1136        io::Error::new(
1137            io::ErrorKind::InvalidInput,
1138            "server.tls.enabled is true but server.tls.cert_path is not set",
1139        )
1140    })?;
1141    let key_path = tls.key_path.as_ref().ok_or_else(|| {
1142        io::Error::new(
1143            io::ErrorKind::InvalidInput,
1144            "server.tls.enabled is true but server.tls.key_path is not set",
1145        )
1146    })?;
1147
1148    // TLS 1.3 only (ASVS V9.1.2). ring provider selected explicitly.
1149    let provider = Arc::new(rustls::crypto::ring::default_provider());
1150    let builder = ServerConfig::builder_with_provider(provider.clone())
1151        .with_protocol_versions(&[&rustls::version::TLS13])
1152        .map_err(|e| io::Error::other(format!("rustls TLS 1.3 configuration failed: {e}")))?;
1153
1154    // Client-certificate (mTLS) policy (D3). `off` keeps the server-auth-only
1155    // behaviour (`with_no_client_auth`); `optional`/`required` install a
1156    // WebPkiClientVerifier over the configured CA bundle so rustls verifies the
1157    // client cert during the handshake and the *verified* cert (not a header)
1158    // drives certificate-based identity.
1159    // Always a `ReloadableClientCertVerifier`, even when `client_auth` is off.
1160    //
1161    // rustls binds whatever it is given here for the process's life, so the
1162    // choice made at this line decides whether flagging a CA later can take
1163    // effect without a restart. `with_no_client_auth()` decides "no", forever.
1164    // The reloadable verifier starts in exactly that posture — it offers no
1165    // client authentication until anchors are installed — and can leave it.
1166    //
1167    // A listener that boots with `Off` therefore starts by requesting no
1168    // client certificate and can still be given anchors later — which matches
1169    // what `mtls_anchors::apply` already does at boot, where flagging a CA
1170    // upgrades an unset `Off` to `Optional`.
1171    let reloadable = Arc::new(ReloadableClientCertVerifier::empty(tls.client_auth));
1172    if tls.client_auth.requests_client_certificate() {
1173        // The bundle written by `mtls_anchors::apply` at boot, or one the
1174        // operator curated themselves.
1175        let roots = read_client_ca_roots(tls)?;
1176        let count = reloadable.replace(roots, &provider)?;
1177        tracing::info!(anchors = count, "client trust anchors loaded");
1178    }
1179    let builder =
1180        builder.with_client_cert_verifier(reloadable.clone() as Arc<dyn ClientCertVerifier>);
1181    install_reloadable_verifier(reloadable, Arc::clone(&provider));
1182
1183    // The leaf. Read, parsed and proven to match its key in `shared_resolver`,
1184    // then installed behind an `ArcSwap` so a renewal can replace it without a
1185    // restart — and shared with the gRPC listener when that one serves the same
1186    // pair, so one reload covers both (R-1).
1187    //
1188    // This is what `with_single_cert` does internally — `CertifiedKey::from_der`
1189    // followed by `with_cert_resolver(SingleCertAndKey::from(..))` (rustls
1190    // 0.23.43 `src/server/builder.rs:65-72`) — with the immutable
1191    // `SingleCertAndKey` swapped for a resolver that can be reloaded. The error
1192    // text is unchanged so the failure an operator sees for a mismatched pair
1193    // is the one it has always been.
1194    let resolver = shared_resolver(cert_path, key_path, &provider)?;
1195    let mut config = builder.with_cert_resolver(resolver as Arc<dyn ResolvesServerCert>);
1196
1197    // ALPN (B2/G8): h2 + http/1.1. This is not configurable — `http2 = false`
1198    // was rejected above because actix re-prepends h2 regardless.
1199    config.alpn_protocols = alpn_protocols();
1200
1201    // TLS 1.3 session resumption (B2). Without a ticketer + session store a
1202    // rustls server does a *full* handshake on every connection; k6 opens many
1203    // short-lived connections per VU, so a full ECDHE handshake per request is a
1204    // per-request fixed cost that shows up as the ~2× p50 inflation on the token
1205    // endpoints. Enabling stateless TLS 1.3 tickets lets repeat connections
1206    // resume (PSK) instead. We deliberately do NOT enable 0-RTT/early-data:
1207    // rustls' `max_early_data_size` stays at its default 0 because the token
1208    // endpoints are non-idempotent POSTs and early data is replayable
1209    // (see docs/security-profiles.md).
1210    config.session_storage = ServerSessionMemoryCache::new(RESUMPTION_CACHE_SIZE);
1211    match rustls::crypto::ring::Ticketer::new() {
1212        Ok(ticketer) => config.ticketer = ticketer,
1213        Err(e) => tracing::warn!(
1214            error = %e,
1215            "failed to construct a TLS ticketer; session resumption disabled \
1216             (falling back to full handshakes)"
1217        ),
1218    }
1219
1220    Ok(config)
1221}

1248/// Decide, from the environment, how the gRPC listener should terminate TLS,
1249/// and build the configuration for it.
1250///
1251/// Returns `None` when neither variable is set — the shipped in-mesh posture,
1252/// where a sidecar or a terminating proxy owns transport security. Setting only
1253/// one of the two is treated as "off" for the same reason it always was: the
1254/// runbook says both or neither, and half a TLS configuration is more likely a
1255/// typo than an intention. (The typo *within* a set variable is what panics —
1256/// see below.)
1257///
1258/// # What this shares with the REST listener
1259///
1260/// The configuration resolves its leaf through [`shared_resolver`], so when the
1261/// two listeners are pointed at the same cert and key — "there is no second
1262/// certificate and there must not be", Pi runbook §14.4 — they hold the **same**
1263/// [`ReloadableCertResolver`] instance and one `SIGHUP` (or one poll tick)
1264/// renews both. When they are pointed at different files, the gRPC pair is
1265/// registered as a second entry and [`reload_leaf_certificate`] reloads it on
1266/// the same triggers; what is not possible any more is a second, unreloaded
1267/// path.
1268///
1269/// # What it deliberately does not share
1270///
1271/// Client certificate policy. The REST listener installs a
1272/// [`ReloadableClientCertVerifier`]; this one calls `with_no_client_auth`,
1273/// which is exactly what tonic's `ServerTlsConfig` did with no `client_ca_root`
1274/// configured. Requesting a client certificate here would change what every
1275/// existing in-mesh gRPC client is asked for during the handshake, which is a
1276/// deployment decision and not part of closing T-234.
1277///
1278/// Session resumption, likewise: tonic set no ticketer, and gRPC connections
1279/// are long-lived by design, so a full handshake per *connection* is not the
1280/// per-request cost it was for REST (B2).
1281///
1282/// # Panics
1283///
1284/// If a variable is set but the file behind it cannot be read or does not parse
1285/// as a certificate/key pair. T-233 rests on "a typo is a failed boot": a
1286/// listener that fell back to plaintext because a path was misspelled would be
1287/// serving unencrypted traffic on a port an operator believes is TLS, and would
1288/// say so only in a log line nobody greps until the incident.
1289pub fn grpc_tls_from_env() -> Option<Arc<ServerConfig>> {
1290    let cert_path = std::env::var(ENV_GRPC_TLS_CERT_PATH).ok()?;
1291    let key_path = std::env::var(ENV_GRPC_TLS_KEY_PATH).ok()?;
1292
1293    let config = build_grpc_rustls_server_config(
1294        std::path::Path::new(&cert_path),
1295        std::path::Path::new(&key_path),
1296    )
1297    .unwrap_or_else(|e| {
1298        panic!(
1299            "{ENV_GRPC_TLS_CERT_PATH}/{ENV_GRPC_TLS_KEY_PATH} set but the pair at \
1300             '{cert_path}' + '{key_path}' is unusable: {e}"
1301        )
1302    });
1303
1304    Some(Arc::new(config))
1305}

1307/// Build the gRPC listener's TLS 1.3-only rustls [`ServerConfig`] over the
1308/// given certificate and key.
1309///
1310/// Separate from [`grpc_tls_from_env`] so the behaviour that matters — TLS 1.3
1311/// only, ALPN `h2`, and a resolver shared with the REST listener when the paths
1312/// match — is testable without mutating process-global environment state.
1313///
1314/// # Example
1315///
1316/// ```no_run
1317/// use std::path::Path;
1318/// use std::sync::Arc;
1319///
1320/// # fn main() -> std::io::Result<()> {
1321/// let config = axiam_server::tls::build_grpc_rustls_server_config(
1322///     Path::new("/etc/letsencrypt/live/example.org/fullchain.pem"),
1323///     Path::new("/etc/letsencrypt/live/example.org/privkey.pem"),
1324/// )?;
1325///
1326/// // TLS 1.3 only, and HTTP/2 and nothing else — gRPC speaks no other
1327/// // protocol, so offering `http/1.1` would only let a client negotiate one
1328/// // no service on this listener answers.
1329/// assert_eq!(config.alpn_protocols, vec![b"h2".to_vec()]);
1330///
1331/// // Hand it to the listener. Pointing the REST listener at the same two
1332/// // files makes both serve from one resolver, so one SIGHUP renews both.
1333/// let tls = axiam_api_grpc::GrpcTls::Rustls(Arc::new(config));
1334/// axiam_server::tls::spawn_leaf_reloader(3600);
1335/// # let _ = tls;
1336/// # Ok(())
1337/// # }
1338/// ```
1339///
1340/// # Errors
1341///
1342/// A missing or unreadable file, a PEM the parser rejects, an empty chain, or a
1343/// key that does not match the leaf's public key — everything
1344/// [`read_certified_key`] rejects.
1345pub fn build_grpc_rustls_server_config(
1346    cert_path: &std::path::Path,
1347    key_path: &std::path::Path,
1348) -> io::Result<ServerConfig> {
1349    // The same explicit `ring` selection the REST listener makes, and for the
1350    // same reason: this build links both `ring` and `aws-lc-rs`, so anything
1351    // that resolves the *process-default* provider is one transitive dependency
1352    // away from panicking on every handshake (the failure
1353    // tests/grpc_tls_crypto_provider.rs documents).
1354    let provider = Arc::new(rustls::crypto::ring::default_provider());
1355
1356    // TLS 1.3 only (ASVS V9.1.2) — the pin tonic's `ServerTlsConfig` had no
1357    // knob for, which is half of what T-234 was about.
1358    let mut config = ServerConfig::builder_with_provider(Arc::clone(&provider))
1359        .with_protocol_versions(&[&rustls::version::TLS13])
1360        .map_err(|e| io::Error::other(format!("rustls TLS 1.3 configuration failed: {e}")))?
1361        .with_no_client_auth()
1362        .with_cert_resolver(
1363            shared_resolver(cert_path, key_path, &provider)? as Arc<dyn ResolvesServerCert>
1364        );
1365
1366    config.alpn_protocols = vec![GRPC_ALPN_H2.to_vec()];
1367    Ok(config)
1368}

2318    /// Pump handshake records between two in-memory rustls connections until
2319    /// both finish handshaking or one errors. Returns the first processing
2320    /// error (e.g. the server rejecting a missing required client cert).
2321    fn drive_handshake(
2322        client: &mut rustls::Connection,
2323        server: &mut rustls::Connection,
2324    ) -> Result<(), rustls::Error> {
2325        for _ in 0..16 {
2326            let mut buf = Vec::new();
2327            while client.wants_write() {
2328                client.write_tls(&mut buf).unwrap();
2329            }
2330            let mut rd: &[u8] = &buf;
2331            while !rd.is_empty() {
2332                server.read_tls(&mut rd).unwrap();
2333            }
2334            server.process_new_packets()?;
2335
2336            let mut buf = Vec::new();
2337            while server.wants_write() {
2338                server.write_tls(&mut buf).unwrap();
2339            }
2340            let mut rd: &[u8] = &buf;
2341            while !rd.is_empty() {
2342                client.read_tls(&mut rd).unwrap();
2343            }
2344            client.process_new_packets()?;
2345
2346            if !client.is_handshaking() && !server.is_handshaking() {
2347                return Ok(());
2348            }
2349        }
2350        Ok(())
2351    }
