1//! Optional direct-TLS support for the REST API listener (ASVS V9.1.2/V9.1.3).
2//!
3//! TLS termination at a proxy/load balancer remains the recommended deployment
4//! pattern (D-06). This module provides an *opt-in* alternative for deployments
5//! that terminate TLS in-process: when `server.tls.enabled` is set, the server
6//! builds a rustls [`ServerConfig`] restricted to **TLS 1.3 only**. Restricting
7//! to TLS 1.3 also satisfies V9.1.3 — every TLS 1.3 cipher suite is
8//! ASVS-approved, so no manual cipher-suite filtering is required.
9//!
10//! The `ring` crypto provider is selected explicitly (rather than relying on a
11//! process-default provider) so `build_rustls_server_config` is self-contained
12//! and deterministic regardless of what other crates in the tree pull in.
13

27/// Number of resumption entries kept in the in-process session cache.
28///
29/// TLS 1.3 resumption is ticket-based (stateless — the ticket travels with the
30/// client), so this cache mainly bounds any non-ticket session state; a small
31/// value is plenty for a benchmark/service workload and caps memory.
32const RESUMPTION_CACHE_SIZE: usize = 512;

34/// The ALPN protocol list the rustls listener is built with.
35///
36/// This is exactly what the actix-web rustls bind ends up advertising, and it
37/// is **not** configurable — see [`reject_unsupported_http2_knob`] for the
38/// source-level proof.
39fn alpn_protocols() -> Vec<Vec<u8>> {
40    vec![b"h2".to_vec(), b"http/1.1".to_vec()]
41}

43/// Reject `server.tls.http2 = false`, which the native listener **cannot**
44/// honour.
45///
46/// # Why this is a hard error rather than a warning (G8)
47///
48/// The knob used to narrow the rustls `ServerConfig`'s ALPN list to
49/// `http/1.1`, log a warning, and then serve h2 anyway. That is the worst of
50/// both worlds: an operator (or a benchmark cell) can *believe* the listener is
51/// HTTP/1.1-only while every client still negotiates h2. Startup now fails, in
52/// the same fail-fast style as every other misconfiguration in this module.
53///
54/// The knob is unimplementable on this bind, verified against the exact
55/// versions in `Cargo.lock` (actix-web 4.14.0 / actix-http 3.13.1 /
56/// rustls 0.23.42):
57///
58/// * `actix_web::HttpServer::bind_rustls_0_23` (actix-web-4.14.0
59///   `src/server.rs:587`) delegates to `listen_rustls_0_23_inner`
60///   (`src/server.rs:1006`), which finishes with
61///   `.rustls_0_23_with_config(config, acceptor_config)`.
62/// * `actix_http::HttpService::rustls_0_23_with_config` (actix-http-3.13.1
63///   `src/service.rs:735`) then does, unconditionally:
64///
65///   ```text
66///   let mut protos = vec![b"h2".to_vec(), b"http/1.1".to_vec()];   // :747
67///   protos.extend_from_slice(&config.alpn_protocols);              // :748
68///   config.alpn_protocols = protos;                                // :749
69///   ```
70///
71///   so `h2` always lands at index 0 of the effective list. Every actix TLS
72///   bind documents this ("ALPN protocols "h2" and "http/1.1" are added to any
73///   configured ones") and every one of them shares the prepend.
74/// * rustls picks the ALPN protocol by **server** preference order —
75///   `our_protocols.iter().find(|ours| their_protocols.contains(ours))` in
76///   rustls-0.23.42 `src/server/hs.rs:99-108` — so index 0 wins for any client
77///   offering h2. Appending to the list can never outrank the prepended `h2`.
78/// * The h2 support cannot be compiled out either: actix-web's `rustls-0_23`
79///   feature *implies* `http2` (actix-web-4.14.0 `Cargo.toml`,
80///   `rustls-0_23 = ["__tls", "http2", ...]`), and the ALPN prepend lives in a
81///   module gated only on `rustls-0_23`, not on `http2`.
82///
83/// A genuinely `http/1.1`-only TLS 1.3 listener is therefore available only by:
84/// 1. fronting the plaintext bind with the `tls13-h1` nginx edge
85///    (`benchmarks/targets/axiam/tls/tls13-h1.conf`) — what the G8 conviction
86///    cell uses; or
87/// 2. abandoning `actix_web::HttpServer` for this bind and driving
88///    `actix_http::H1Service::rustls_0_23` (actix-http-3.13.1
89///    `src/h1/service.rs:371`) directly on an `actix_server::Server` — that
90///    constructor is the one rustls service factory in the tree that does
91///    **not** touch `alpn_protocols`. See
92///    `claude_dev/b2-tls-h2-investigation.md` for why that rewrite is not
93///    justified today.
94fn reject_unsupported_http2_knob(tls: &TlsConfig) -> io::Result<()> {
95    if tls.http2 {
96        return Ok(());
97    }
98    Err(io::Error::new(
99        io::ErrorKind::Unsupported,
100        "server.tls.http2=false is not supported by the native TLS listener: \
101         actix-web's rustls bind unconditionally prepends \"h2\" to the ALPN \
102         list (actix-http HttpService::rustls_0_23_with_config), and rustls \
103         selects by server preference, so h2 would still be negotiated. \
104         Refusing to start rather than silently serving the opposite of what \
105         was configured. For an http/1.1-only TLS 1.3 endpoint, leave \
106         server.tls disabled and terminate TLS at an edge that does not enable \
107         HTTP/2 (see benchmarks/targets/axiam/tls/tls13-h1.conf and \
108         docs/security-profiles.md).",
109    ))
110}

112/// Maximum legal HTTP/2 flow-control window, `2^31 - 1` (RFC 9113 §6.5.2;
113/// `h2::frame::settings::MAX_INITIAL_WINDOW_SIZE` in h2 0.4.15).
114const MAX_H2_WINDOW_SIZE: u32 = (1 << 31) - 1;

116/// actix-http 3.13.1's default initial **stream** window (`src/config.rs:19`,
117/// `DEFAULT_H2_STREAM_WINDOW_SIZE`). Recorded so operators can see what an
118/// unset knob means; never written into the builder.
119pub const ACTIX_DEFAULT_H2_STREAM_WINDOW: u32 = 1024 * 1024;

125/// HTTP/2 tuning surface for the native TLS listener (G8/B2).
126///
127/// h2 is only ever negotiated on the rustls bind (the plaintext bind is served
128/// as HTTP/1.1 — `HttpServer::bind` never calls `listen_auto_h2c`), so these
129/// keys live next to the TLS ones and are a no-op when `server.tls.enabled` is
130/// false.
131///
132/// **Every field defaults to `None`, and `None` means "do not call the actix
133/// setter at all"** — an unset config reproduces today's behaviour byte for
134/// byte, because `actix_web::HttpServer` itself stores these as `Option`s and
135/// only forwards them to `HttpService` when `Some`
136/// (actix-web-4.14.0 `src/server.rs:1041-1048`).
137///
138/// | Env var | Type | Unset ⇒ actix default |
139/// |---|---|---|
140/// | `AXIAM__SERVER__H2__INITIAL_STREAM_WINDOW_SIZE` | `u32` bytes | 1 MiB |
141/// | `AXIAM__SERVER__H2__INITIAL_CONNECTION_WINDOW_SIZE` | `u32` bytes | 2 MiB |
142///
143/// # What is deliberately *not* exposed
144///
145/// * **`max_concurrent_streams`** — actix-http 3.13.1 never sets it. Its h2
146///   handshake builder (`src/h2/mod.rs:60-71`) calls only
147///   `initial_window_size` and `initial_connection_window_size`; the underlying
148///   `h2::server::Builder::max_concurrent_streams` (h2-0.4.15
149///   `src/server.rs:858`) exists but is unreachable, and neither
150///   `actix_http::HttpServiceBuilder` nor `actix_web::HttpServer` re-exports
151///   it. Because actix never sends `SETTINGS_MAX_CONCURRENT_STREAMS`
152///   (`h2::frame::Settings` derives `Default`, i.e. `None`), the server
153///   advertises *no* stream limit — so a stream cap is provably **not** the
154///   cause of the B2 concurrency ceiling, and inventing a knob for it here
155///   would be a lie.
156/// * **keep-alive** — `HttpServer::keep_alive` exists but is shared by h1 and
157///   h2 and does not bear on a multiplexed-connection ceiling: the single h2
158///   connection stays open for the whole run either way. Changing it would
159///   alter p0 and p2 alike, so it is not a B2 lever.
160#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, serde::Deserialize)]
161#[serde(default, deny_unknown_fields)]
162pub struct Http2Tuning {
163    /// `SETTINGS_INITIAL_WINDOW_SIZE` advertised per stream, in bytes.
164    pub initial_stream_window_size: Option<u32>,
165    /// Connection-level receive window, in bytes.
166    pub initial_connection_window_size: Option<u32>,
167}
168
169impl Http2Tuning {
170    /// Load the `server.h2` subtree using the same sources and idiom as the
171    /// main `AppConfig` loader (`config/default.toml` + `AXIAM__*` env with a
172    /// `__` separator).
173    ///
174    /// An absent subtree yields [`Self::default`] (all `None`).
175    pub fn load() -> io::Result<Self> {
176        let sources = config::Config::builder()
177            .add_source(config::File::with_name("config/default").required(false))
178            .add_source(config::Environment::with_prefix("AXIAM").separator("__"))
179            .build()
180            .map_err(|e| io::Error::other(format!("failed to load configuration: {e}")))?;
181
182        let tuning = match sources.get::<Self>("server.h2") {
183            Ok(t) => t,
184            // Nothing configured at all — keep actix's defaults.
185            Err(config::ConfigError::NotFound(_)) => Self::default(),
186            Err(e) => {
187                return Err(io::Error::new(
188                    io::ErrorKind::InvalidInput,
189                    format!("invalid server.h2 configuration: {e}"),
190                ));
191            }
192        };
193        tuning.validate()?;
194        Ok(tuning)
195    }
196
197    /// Reject window sizes HTTP/2 cannot express, so a typo fails at startup
198    /// rather than at the first h2 handshake (`h2::server::Builder` asserts
199    /// `size <= MAX_WINDOW_SIZE` and would panic inside a worker).
200    fn validate(&self) -> io::Result<()> {
201        for (name, value) in [
202            (
203                "initial_stream_window_size",
204                self.initial_stream_window_size,
205            ),
206            (
207                "initial_connection_window_size",
208                self.initial_connection_window_size,
209            ),
210        ] {
211            let Some(v) = value else { continue };
212            if v == 0 || v > MAX_H2_WINDOW_SIZE {
213                return Err(io::Error::new(
214                    io::ErrorKind::InvalidInput,
215                    format!(
216                        "server.h2.{name} must be between 1 and {MAX_H2_WINDOW_SIZE} \
217                         bytes (RFC 9113 §6.5.2); got {v}"
218                    ),
219                ));
220            }
221        }
222        Ok(())
223    }
224
225    /// True when nothing is configured, i.e. actix's defaults are untouched.
226    pub fn is_default(&self) -> bool {
227        *self == Self::default()
228    }
229
230    /// The window actix will actually use per stream (configured or default) —
231    /// for the startup log line only.
232    pub fn effective_stream_window(&self) -> u32 {
233        self.initial_stream_window_size
234            .unwrap_or(ACTIX_DEFAULT_H2_STREAM_WINDOW)
235    }
236
237    /// The window actix will actually use per connection (configured or
238    /// default) — for the startup log line only.
239    pub fn effective_connection_window(&self) -> u32 {
240        self.initial_connection_window_size
241            .unwrap_or(ACTIX_DEFAULT_H2_CONNECTION_WINDOW)
242    }
243}

245/// Build a rustls [`ClientCertVerifier`] from the configured client-CA bundle
246/// (D3 native mTLS).
247///
248/// Loads the PEM bundle at `client_ca_path` into a [`RootCertStore`] and builds
249/// a [`WebPkiClientVerifier`] over it, using the same [`CryptoProvider`] as the
250/// server config. For [`ClientAuth::Optional`] the verifier still *offers* and
251/// *verifies* client certs but permits anonymous clients
252/// (`allow_unauthenticated`); for [`ClientAuth::Required`] a verified client
253/// cert is mandatory.
254///
255/// Fails fast (aborting startup) when the CA path is unset, missing/unreadable,
256/// empty, or malformed — matching this file's existing `io::Error` style so a
257/// misconfigured mTLS server never starts.
258fn read_client_ca_roots(tls: &TlsConfig) -> io::Result<RootCertStore> {
259    let ca_path = tls.client_ca_path.as_ref().ok_or_else(|| {
260        io::Error::new(
261            io::ErrorKind::InvalidInput,
262            "server.tls.client_auth is optional/required but \
263             server.tls.client_ca_path is not set",
264        )
265    })?;
266
267    let pem = std::fs::read_to_string(ca_path).map_err(|e| {
268        io::Error::new(
269            e.kind(),
270            format!(
271                "failed to open TLS client CA bundle {}: {e}",
272                ca_path.display()
273            ),
274        )
275    })?;
276    let roots = roots_from_pem(&pem, &ca_path.display().to_string())?;
277    if roots.is_empty() {
278        return Err(io::Error::new(
279            io::ErrorKind::InvalidData,
280            format!("no client CA certificates found in {}", ca_path.display()),
281        ));
282    }
283    Ok(roots)
284}

286/// The verifier the live listener is using, and the provider to rebuild with.
287///
288/// A process-global because there is one TLS listener per process and the
289/// handler that flags a CA has no path to the `ServerConfig` actix built — it
290/// runs inside it. Set once, during `build_rustls_server_config`.
291static LIVE_VERIFIER: std::sync::OnceLock<(
292    Arc<ReloadableClientCertVerifier>,
293    Arc<rustls::crypto::CryptoProvider>,
294)> = std::sync::OnceLock::new();
295
296fn install_reloadable_verifier(
297    verifier: Arc<ReloadableClientCertVerifier>,
298    provider: Arc<rustls::crypto::CryptoProvider>,
299) {
300    // `set` fails only if called twice, which means a second listener was
301    // built. The first one is the one actix is serving on.
302    let _ = LIVE_VERIFIER.set((verifier, provider));
303}

305/// Install `pem` as the live client trust anchor set, without a restart.
306///
307/// Returns the number of anchors now trusted, or `None` when this process has
308/// no TLS listener with client authentication enabled — a plaintext deployment,
309/// or one whose operator set `client_auth = off`. That is not a failure: there
310/// is nothing to reload into, and saying so lets the caller report "saved, and
311/// it will apply when TLS is enabled" rather than a spurious error.
312///
313/// # Errors
314///
315/// A bundle that does not parse, or a certificate webpki refuses as a trust
316/// anchor. The live anchor set is left **unchanged** in that case: the new
317/// bundle is fully parsed and the verifier fully built before anything is
318/// swapped, so a bad reload cannot leave the listener trusting nothing.
319pub fn reload_trust_anchors(pem: &str) -> io::Result<Option<usize>> {
320    let Some((verifier, provider)) = LIVE_VERIFIER.get() else {
321        return Ok(None);
322    };
323    let roots = roots_from_pem(pem, "the mTLS trust anchor bundle")?;
324    let count = verifier.replace(roots, provider)?;
325    Ok(Some(count))
326}

332/// A [`ClientCertVerifier`] whose trust anchors can be replaced while the
333/// server is listening.
334///
335/// # Why this exists
336///
337/// rustls builds its client trust store when the `ServerConfig` is constructed,
338/// and actix binds that config for the process's life. Flagging a CA as an mTLS
339/// trust anchor therefore used to take effect at the *next* boot, and the API
340/// said so rather than pretending otherwise — which is honest and still means
341/// an operator adding a device CA has to restart the server every browser
342/// session and every IoT device is currently connected to.
343///
344/// rustls consults the verifier **per handshake**, not once at construction. So
345/// a verifier that delegates to a swappable inner one gives the config
346/// something permanent to hold while the anchors behind it change. Connections
347/// already established keep the verifier they handshook with, which is correct:
348/// a certificate accepted a moment ago does not become invalid mid-connection,
349/// and revocation of an established session is what session invalidation is
350/// for.
351///
352/// # The empty state
353///
354/// [`Anchors::None`] is not "trust nothing" — it is "do not ask for a client
355/// certificate at all", the same posture as `with_no_client_auth()`. A
356/// deployment that boots with no flagged CA must behave exactly as it does
357/// today, *and* must be able to reach the anchored state without a restart.
358/// Installing this verifier unconditionally is what makes both true;
359/// `WebPkiClientVerifier` cannot express it, because it refuses to build over
360/// an empty root store.
361#[derive(Debug)]
362pub struct ReloadableClientCertVerifier {
363    anchors: ArcSwap<Anchors>,
364    /// Whether a verified client certificate is required once anchors exist.
365    ///
366    /// Fixed at construction from `client_auth`: it is an operator's policy
367    /// decision, not a property of the anchor set, and changing it changes
368    /// whether unauthenticated clients can connect at all.
369    mandatory: bool,
370}
371
372/// What the verifier currently trusts.
373#[derive(Debug)]
374enum Anchors {
375    /// No trust anchors: no client certificate is requested.
376    None,
377    /// A webpki verifier over the current anchor set.
378    Some(Arc<dyn ClientCertVerifier>),
379}
380
381impl ReloadableClientCertVerifier {
382    /// An empty verifier that offers no client authentication.
383    pub fn empty(mandatory: bool) -> Self {
384        Self {
385            anchors: ArcSwap::from_pointee(Anchors::None),
386            mandatory,
387        }
388    }
389
390    /// Replace the trust anchors with a verifier built over `roots`.
391    ///
392    /// An empty `roots` returns to [`Anchors::None`] rather than building a
393    /// verifier that trusts nothing: a verifier that requests a certificate and
394    /// then rejects every one is strictly worse than not requesting one, and it
395    /// is what un-flagging the last CA should produce.
396    pub fn replace(
397        &self,
398        roots: RootCertStore,
399        provider: &Arc<rustls::crypto::CryptoProvider>,
400    ) -> io::Result<usize> {
401        if roots.is_empty() {
402            self.anchors.store(Arc::new(Anchors::None));
403            return Ok(0);
404        }
405        let count = roots.len();
406        let builder =
407            WebPkiClientVerifier::builder_with_provider(Arc::new(roots), provider.clone());
408        let verifier = if self.mandatory {
409            builder.build()
410        } else {
411            builder.allow_unauthenticated().build()
412        }
413        .map_err(|e| io::Error::other(format!("failed to build client cert verifier: {e}")))?;
414
415        self.anchors.store(Arc::new(Anchors::Some(verifier)));
416        Ok(count)
417    }
418
419    /// How many anchors are currently installed.
420    pub fn anchor_count(&self) -> usize {
421        match &**self.anchors.load() {
422            Anchors::None => 0,
423            Anchors::Some(v) => v.root_hint_subjects().len(),
424        }
425    }
426
427    /// The current inner verifier, if any.
428    fn current(&self) -> Option<Arc<dyn ClientCertVerifier>> {
429        match &**self.anchors.load() {
430            Anchors::None => None,
431            Anchors::Some(v) => Some(Arc::clone(v)),
432        }
433    }
434}

436impl ClientCertVerifier for ReloadableClientCertVerifier {
437    fn offer_client_auth(&self) -> bool {
438        // Consulted per handshake, which is the whole mechanism: the answer
439        // changes the moment `replace` installs an anchor set.
440        self.current().is_some()
441    }
442
443    fn client_auth_mandatory(&self) -> bool {
444        // Only meaningful while anchors exist. Answering `true` with none
445        // installed would refuse every connection to a server that cannot
446        // verify anybody — a self-inflicted outage on a deployment that
447        // un-flagged its last CA.
448        self.mandatory && self.current().is_some()
449    }
450
451    fn root_hint_subjects(&self) -> &[rustls::DistinguishedName] {
452        // Cannot borrow through the ArcSwap guard, and the hint is advisory:
453        // it tells a client which issuers the server would accept, and a client
454        // that offers a certificate anyway is verified normally by
455        // `verify_client_cert`. Returning nothing costs a round trip in the
456        // worst case and never accepts anything it should not.
457        &[]
458    }
459
460    fn verify_client_cert(
461        &self,
462        end_entity: &CertificateDer<'_>,
463        intermediates: &[CertificateDer<'_>],
464        now: rustls::pki_types::UnixTime,
465    ) -> Result<rustls::server::danger::ClientCertVerified, rustls::Error> {
466        match self.current() {
467            Some(v) => v.verify_client_cert(end_entity, intermediates, now),
468            // Unreachable while rustls honours `offer_client_auth`, and a
469            // refusal rather than an acceptance if it ever does not.
470            None => Err(rustls::Error::General(
471                "no client trust anchors are configured".into(),
472            )),
473        }
474    }
475
476    fn verify_tls12_signature(
477        &self,
478        message: &[u8],
479        cert: &CertificateDer<'_>,
480        dss: &rustls::DigitallySignedStruct,
481    ) -> Result<rustls::client::danger::HandshakeSignatureValid, rustls::Error> {
482        match self.current() {
483            Some(v) => v.verify_tls12_signature(message, cert, dss),
484            None => Err(rustls::Error::General(
485                "no client trust anchors are configured".into(),
486            )),
487        }
488    }
489
490    fn verify_tls13_signature(
491        &self,
492        message: &[u8],
493        cert: &CertificateDer<'_>,
494        dss: &rustls::DigitallySignedStruct,
495    ) -> Result<rustls::client::danger::HandshakeSignatureValid, rustls::Error> {
496        match self.current() {
497            Some(v) => v.verify_tls13_signature(message, cert, dss),
498            None => Err(rustls::Error::General(
499                "no client trust anchors are configured".into(),
500            )),
501        }
502    }
503
504    fn supported_verify_schemes(&self) -> Vec<rustls::SignatureScheme> {
505        match self.current() {
506            Some(v) => v.supported_verify_schemes(),
507            // The provider's full set: this is a capability advertisement, and
508            // an empty list would make a handshake fail to negotiate rather
509            // than fall through to the "no anchors" path above.
510            None => rustls::crypto::ring::default_provider()
511                .signature_verification_algorithms
512                .supported_schemes(),
513        }
514    }
515}

517/// Build a [`RootCertStore`] from concatenated PEM.
518///
519/// Shared by the boot path and the reload path so a hot-reloaded anchor set is
520/// parsed by exactly the same code as one read at startup — two parsers is how
521/// a bundle comes to be accepted at boot and rejected on reload.
522pub fn roots_from_pem(pem: &str, source: &str) -> io::Result<RootCertStore> {
523    let certs: Vec<CertificateDer<'static>> = CertificateDer::pem_slice_iter(pem.as_bytes())
524        .collect::<Result<Vec<_>, _>>()
525        .map_err(|e| {
526            io::Error::new(
527                io::ErrorKind::InvalidData,
528                format!("failed to parse client CA certificates from {source}: {e}"),
529            )
530        })?;
531
532    let mut roots = RootCertStore::empty();
533    for cert in certs {
534        roots.add(cert).map_err(|e| {
535            io::Error::new(
536                io::ErrorKind::InvalidData,
537                format!("invalid client CA certificate in {source}: {e}"),
538            )
539        })?;
540    }
541    Ok(roots)
542}

544/// Build a TLS 1.3-only rustls [`ServerConfig`] from the configured PEM files.
545///
546/// Fails fast (returning an `io::Error` that aborts startup) when TLS is enabled
547/// but misconfigured: a missing `cert_path`/`key_path`, an unreadable or
548/// malformed PEM file, an empty cert chain, a cert/key mismatch, or an
549/// unsatisfiable `http2 = false` request (see [`reject_unsupported_http2_knob`]).
550pub fn build_rustls_server_config(tls: &TlsConfig) -> io::Result<ServerConfig> {
551    // Checked before any file IO: an unsatisfiable ALPN request is a config
552    // error regardless of whether the cert/key happen to be readable.
553    reject_unsupported_http2_knob(tls)?;
554
555    let cert_path = tls.cert_path.as_ref().ok_or_else(|| {
556        io::Error::new(
557            io::ErrorKind::InvalidInput,
558            "server.tls.enabled is true but server.tls.cert_path is not set",
559        )
560    })?;
561    let key_path = tls.key_path.as_ref().ok_or_else(|| {
562        io::Error::new(
563            io::ErrorKind::InvalidInput,
564            "server.tls.enabled is true but server.tls.key_path is not set",
565        )
566    })?;
567
568    // Open the files explicitly so a missing/unreadable path yields a clean
569    // io::Error (NotFound / PermissionDenied) before any PEM parsing. Cert and
570    // key are parsed via rustls-pki-types' `PemObject` trait directly —
571    // rustls-pemfile is unmaintained (RUSTSEC-2025-0134) and is a thin wrapper
572    // over this same code.
573    let cert_file = File::open(cert_path).map_err(|e| {
574        io::Error::new(
575            e.kind(),
576            format!("failed to open TLS cert file {}: {e}", cert_path.display()),
577        )
578    })?;
579    let cert_chain: Vec<CertificateDer<'static>> = CertificateDer::pem_reader_iter(cert_file)
580        .collect::<Result<Vec<_>, _>>()
581        .map_err(|e| {
582            io::Error::new(
583                io::ErrorKind::InvalidData,
584                format!(
585                    "failed to parse TLS certificates from {}: {e}",
586                    cert_path.display()
587                ),
588            )
589        })?;
590    if cert_chain.is_empty() {
591        return Err(io::Error::new(
592            io::ErrorKind::InvalidData,
593            format!("no certificates found in {}", cert_path.display()),
594        ));
595    }
596
597    let key_file = File::open(key_path).map_err(|e| {
598        io::Error::new(
599            e.kind(),
600            format!("failed to open TLS key file {}: {e}", key_path.display()),
601        )
602    })?;
603    let key = PrivateKeyDer::from_pem_reader(key_file).map_err(|e| {
604        io::Error::new(
605            io::ErrorKind::InvalidData,
606            format!(
607                "failed to read a TLS private key from {}: {e}",
608                key_path.display()
609            ),
610        )
611    })?;
612
613    // TLS 1.3 only (ASVS V9.1.2). ring provider selected explicitly.
614    let provider = Arc::new(rustls::crypto::ring::default_provider());
615    let builder = ServerConfig::builder_with_provider(provider.clone())
616        .with_protocol_versions(&[&rustls::version::TLS13])
617        .map_err(|e| io::Error::other(format!("rustls TLS 1.3 configuration failed: {e}")))?;
618
619    // Client-certificate (mTLS) policy (D3). `off` keeps the server-auth-only
620    // behaviour (`with_no_client_auth`); `optional`/`required` install a
621    // WebPkiClientVerifier over the configured CA bundle so rustls verifies the
622    // client cert during the handshake and the *verified* cert (not a header)
623    // drives certificate-based identity.
624    // Always a `ReloadableClientCertVerifier`, even when `client_auth` is off.
625    //
626    // rustls binds whatever it is given here for the process's life, so the
627    // choice made at this line decides whether flagging a CA later can take
628    // effect without a restart. `with_no_client_auth()` decides "no", forever.
629    // The reloadable verifier starts in exactly that posture — it offers no
630    // client authentication until anchors are installed — and can leave it.
631    //
632    // A listener that boots with `Off` therefore starts by requesting no
633    // client certificate and can still be given anchors later — which matches
634    // what `mtls_anchors::apply` already does at boot, where flagging a CA
635    // upgrades an unset `Off` to `Optional`.
636    let reloadable = Arc::new(ReloadableClientCertVerifier::empty(
637        tls.client_auth == ClientAuth::Required,
638    ));
639    if tls.client_auth != ClientAuth::Off {
640        // The bundle written by `mtls_anchors::apply` at boot, or one the
641        // operator curated themselves.
642        let roots = read_client_ca_roots(tls)?;
643        let count = reloadable.replace(roots, &provider)?;
644        tracing::info!(anchors = count, "client trust anchors loaded");
645    }
646    let builder =
647        builder.with_client_cert_verifier(reloadable.clone() as Arc<dyn ClientCertVerifier>);
648    install_reloadable_verifier(reloadable, Arc::clone(&provider));
649
650    let mut config = builder.with_single_cert(cert_chain, key).map_err(|e| {
651        io::Error::new(
652            io::ErrorKind::InvalidData,
653            format!("invalid TLS certificate/key pair: {e}"),
654        )
655    })?;
656
657    // ALPN (B2/G8): h2 + http/1.1. This is not configurable — `http2 = false`
658    // was rejected above because actix re-prepends h2 regardless.
659    config.alpn_protocols = alpn_protocols();
660
661    // TLS 1.3 session resumption (B2). Without a ticketer + session store a
662    // rustls server does a *full* handshake on every connection; k6 opens many
663    // short-lived connections per VU, so a full ECDHE handshake per request is a
664    // per-request fixed cost that shows up as the ~2× p50 inflation on the token
665    // endpoints. Enabling stateless TLS 1.3 tickets lets repeat connections
666    // resume (PSK) instead. We deliberately do NOT enable 0-RTT/early-data:
667    // rustls' `max_early_data_size` stays at its default 0 because the token
668    // endpoints are non-idempotent POSTs and early data is replayable
669    // (see docs/security-profiles.md).
670    config.session_storage = ServerSessionMemoryCache::new(RESUMPTION_CACHE_SIZE);
671    match rustls::crypto::ring::Ticketer::new() {
672        Ok(ticketer) => config.ticketer = ticketer,
673        Err(e) => tracing::warn!(
674            error = %e,
675            "failed to construct a TLS ticketer; session resumption disabled \
676             (falling back to full handshakes)"
677        ),
678    }
679
680    Ok(config)
681}

888    /// Write `contents` to a unique temp file and return its path.
889    fn write_tmp(tag: &str, contents: &str) -> PathBuf {
890        let n = TMP_COUNTER.fetch_add(1, Ordering::Relaxed);
891        let path =
892            std::env::temp_dir().join(format!("axiam-d3-{}-{n}-{tag}.pem", std::process::id()));
893        let mut f = File::create(&path).expect("create temp pem");
894        f.write_all(contents.as_bytes()).expect("write temp pem");
895        path
896    }

1226    /// Pump handshake records between two in-memory rustls connections until
1227    /// both finish handshaking or one errors. Returns the first processing
1228    /// error (e.g. the server rejecting a missing required client cert).
1229    fn drive_handshake(
1230        client: &mut rustls::Connection,
1231        server: &mut rustls::Connection,
1232    ) -> Result<(), rustls::Error> {
1233        for _ in 0..16 {
1234            let mut buf = Vec::new();
1235            while client.wants_write() {
1236                client.write_tls(&mut buf).unwrap();
1237            }
1238            let mut rd: &[u8] = &buf;
1239            while !rd.is_empty() {
1240                server.read_tls(&mut rd).unwrap();
1241            }
1242            server.process_new_packets()?;
1243
1244            let mut buf = Vec::new();
1245            while server.wants_write() {
1246                server.write_tls(&mut buf).unwrap();
1247            }
1248            let mut rd: &[u8] = &buf;
1249            while !rd.is_empty() {
1250                client.read_tls(&mut rd).unwrap();
1251            }
1252            client.process_new_packets()?;
1253
1254            if !client.is_handshaking() && !server.is_handshaking() {
1255                return Ok(());
1256            }
1257        }
1258        Ok(())
1259    }

1274    /// Client config trusting the CA; `client_cert` toggles whether it presents
1275    /// its own certificate.
1276    fn make_client(pki: &TestPki, present_cert: bool) -> rustls::ClientConnection {
1277        use rustls::pki_types::pem::PemObject;
1278        use rustls::pki_types::{CertificateDer, PrivateKeyDer, ServerName};
1279
1280        let mut roots = RootCertStore::empty();
1281        roots
1282            .add(CertificateDer::from_pem_slice(pki.ca_pem.as_bytes()).unwrap())
1283            .unwrap();
1284
1285        let provider = Arc::new(rustls::crypto::ring::default_provider());
1286        let builder = rustls::ClientConfig::builder_with_provider(provider)
1287            .with_protocol_versions(&[&rustls::version::TLS13])
1288            .unwrap()
1289            .with_root_certificates(roots);
1290
1291        let config = if present_cert {
1292            let chain =
1293                vec![CertificateDer::from_pem_slice(pki.client_cert_pem.as_bytes()).unwrap()];
1294            let key = PrivateKeyDer::from_pem_slice(pki.client_key_pem.as_bytes()).unwrap();
1295            builder.with_client_auth_cert(chain, key).unwrap()
1296        } else {
1297            builder.with_no_client_auth()
1298        };
1299
1300        let name = ServerName::try_from("localhost").unwrap();
1301        rustls::ClientConnection::new(Arc::new(config), name).expect("client connection")
1302    }
