1//! Optional direct-TLS support for the REST API listener (ASVS V9.1.2/V9.1.3).
2//!
3//! TLS termination at a proxy/load balancer remains the recommended deployment
4//! pattern (D-06). This module provides an *opt-in* alternative for deployments
5//! that terminate TLS in-process: when `server.tls.enabled` is set, the server
6//! builds a rustls [`ServerConfig`] restricted to **TLS 1.3 only**. Restricting
7//! to TLS 1.3 also satisfies V9.1.3 — every TLS 1.3 cipher suite is
8//! ASVS-approved, so no manual cipher-suite filtering is required.
9//!
10//! The `ring` crypto provider is selected explicitly (rather than relying on a
11//! process-default provider) so `build_rustls_server_config` is self-contained
12//! and deterministic regardless of what other crates in the tree pull in.

37/// The ALPN protocol list the rustls listener is built with.
38///
39/// This is exactly what the actix-web rustls bind ends up advertising, and it
40/// is **not** configurable — see [`reject_unsupported_http2_knob`] for the

46/// Reject `server.tls.http2 = false`, which the native listener **cannot**
47/// honour.
48///
49/// # Why this is a hard error rather than a warning (G8)
50///
51/// The knob used to narrow the rustls `ServerConfig`'s ALPN list to
52/// `http/1.1`, log a warning, and then serve h2 anyway. That is the worst of
53/// both worlds: an operator (or a benchmark cell) can *believe* the listener is
54/// HTTP/1.1-only while every client still negotiates h2. Startup now fails, in
55/// the same fail-fast style as every other misconfiguration in this module.
56///
57/// The knob is unimplementable on this bind, verified against the exact
58/// versions in `Cargo.lock` (actix-web 4.14.0 / actix-http 3.13.1 /
59/// rustls 0.23.42):
60///
61/// * `actix_web::HttpServer::bind_rustls_0_23` (actix-web-4.14.0
62///   `src/server.rs:587`) delegates to `listen_rustls_0_23_inner`
63///   (`src/server.rs:1006`), which finishes with
64///   `.rustls_0_23_with_config(config, acceptor_config)`.
65/// * `actix_http::HttpService::rustls_0_23_with_config` (actix-http-3.13.1
66///   `src/service.rs:735`) then does, unconditionally:
67///
68///   ```text
69///   let mut protos = vec![b"h2".to_vec(), b"http/1.1".to_vec()];   // :747
70///   protos.extend_from_slice(&config.alpn_protocols);              // :748
71///   config.alpn_protocols = protos;                                // :749
72///   ```
73///
74///   so `h2` always lands at index 0 of the effective list. Every actix TLS
75///   bind documents this ("ALPN protocols "h2" and "http/1.1" are added to any
76///   configured ones") and every one of them shares the prepend.
77/// * rustls picks the ALPN protocol by **server** preference order —
78///   `our_protocols.iter().find(|ours| their_protocols.contains(ours))` in
79///   rustls-0.23.42 `src/server/hs.rs:99-108` — so index 0 wins for any client
80///   offering h2. Appending to the list can never outrank the prepended `h2`.
81/// * The h2 support cannot be compiled out either: actix-web's `rustls-0_23`
82///   feature *implies* `http2` (actix-web-4.14.0 `Cargo.toml`,
83///   `rustls-0_23 = ["__tls", "http2", ...]`), and the ALPN prepend lives in a
84///   module gated only on `rustls-0_23`, not on `http2`.
85///
86/// A genuinely `http/1.1`-only TLS 1.3 listener is therefore available only by:
87/// 1. fronting the plaintext bind with the `tls13-h1` nginx edge
88///    (`benchmarks/targets/axiam/tls/tls13-h1.conf`) — what the G8 conviction
89///    cell uses; or
90/// 2. abandoning `actix_web::HttpServer` for this bind and driving
91///    `actix_http::H1Service::rustls_0_23` (actix-http-3.13.1
92///    `src/h1/service.rs:371`) directly on an `actix_server::Server` — that
93///    constructor is the one rustls service factory in the tree that does
94///    **not** touch `alpn_protocols`. See
95///    `claude_dev/b2-tls-h2-investigation.md` for why that rewrite is not
96///    justified today.
97fn reject_unsupported_http2_knob(tls: &TlsConfig) -> io::Result<()> {
98    if tls.http2 {
99        return Ok(());
100    }
101    Err(io::Error::new(
102        io::ErrorKind::Unsupported,
103        "server.tls.http2=false is not supported by the native TLS listener: \
104         actix-web's rustls bind unconditionally prepends \"h2\" to the ALPN \
105         list (actix-http HttpService::rustls_0_23_with_config), and rustls \
106         selects by server preference, so h2 would still be negotiated. \
107         Refusing to start rather than silently serving the opposite of what \
108         was configured. For an http/1.1-only TLS 1.3 endpoint, leave \
109         server.tls disabled and terminate TLS at an edge that does not enable \
110         HTTP/2 (see benchmarks/targets/axiam/tls/tls13-h1.conf and \
111         docs/security-profiles.md).",
112    ))
113}

128/// HTTP/2 tuning surface for the native TLS listener (G8/B2).
129///
130/// h2 is only ever negotiated on the rustls bind (the plaintext bind is served
131/// as HTTP/1.1 — `HttpServer::bind` never calls `listen_auto_h2c`), so these
132/// keys live next to the TLS ones and are a no-op when `server.tls.enabled` is
133/// false.
134///
135/// **Every field defaults to `None`, and `None` means "do not call the actix
136/// setter at all"** — an unset config reproduces today's behaviour byte for
137/// byte, because `actix_web::HttpServer` itself stores these as `Option`s and
138/// only forwards them to `HttpService` when `Some`
139/// (actix-web-4.14.0 `src/server.rs:1041-1048`).
140///
141/// | Env var | Type | Unset ⇒ actix default |
142/// |---|---|---|
143/// | `AXIAM__SERVER__H2__INITIAL_STREAM_WINDOW_SIZE` | `u32` bytes | 1 MiB |
144/// | `AXIAM__SERVER__H2__INITIAL_CONNECTION_WINDOW_SIZE` | `u32` bytes | 2 MiB |
145///
146/// # What is deliberately *not* exposed
147///
148/// * **`max_concurrent_streams`** — actix-http 3.13.1 never sets it. Its h2
149///   handshake builder (`src/h2/mod.rs:60-71`) calls only
150///   `initial_window_size` and `initial_connection_window_size`; the underlying
151///   `h2::server::Builder::max_concurrent_streams` (h2-0.4.15
152///   `src/server.rs:858`) exists but is unreachable, and neither
153///   `actix_http::HttpServiceBuilder` nor `actix_web::HttpServer` re-exports
154///   it. Because actix never sends `SETTINGS_MAX_CONCURRENT_STREAMS`
155///   (`h2::frame::Settings` derives `Default`, i.e. `None`), the server
156///   advertises *no* stream limit — so a stream cap is provably **not** the
157///   cause of the B2 concurrency ceiling, and inventing a knob for it here
158///   would be a lie.
159/// * **keep-alive** — `HttpServer::keep_alive` exists but is shared by h1 and
160///   h2 and does not bear on a multiplexed-connection ceiling: the single h2
161///   connection stays open for the whole run either way. Changing it would
162///   alter p0 and p2 alike, so it is not a B2 lever.
163#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, serde::Deserialize)]
164#[serde(default, deny_unknown_fields)]
165pub struct Http2Tuning {
166    /// `SETTINGS_INITIAL_WINDOW_SIZE` advertised per stream, in bytes.
167    pub initial_stream_window_size: Option<u32>,
168    /// Connection-level receive window, in bytes.
169    pub initial_connection_window_size: Option<u32>,
170}
171
172impl Http2Tuning {
173    /// Load the `server.h2` subtree using the same sources and idiom as the
174    /// main `AppConfig` loader (`config/default.toml` + `AXIAM__*` env with a
175    /// `__` separator).
176    ///
177    /// An absent subtree yields [`Self::default`] (all `None`).
178    pub fn load() -> io::Result<Self> {
179        let sources = config::Config::builder()
180            .add_source(config::File::with_name("config/default").required(false))
181            .add_source(config::Environment::with_prefix("AXIAM").separator("__"))
182            .build()
183            .map_err(|e| io::Error::other(format!("failed to load configuration: {e}")))?;
184
185        let tuning = match sources.get::<Self>("server.h2") {
186            Ok(t) => t,
187            // Nothing configured at all — keep actix's defaults.
188            Err(config::ConfigError::NotFound(_)) => Self::default(),
189            Err(e) => {
190                return Err(io::Error::new(
191                    io::ErrorKind::InvalidInput,
192                    format!("invalid server.h2 configuration: {e}"),
193                ));
194            }
195        };
196        tuning.validate()?;
197        Ok(tuning)
198    }
199
200    /// Reject window sizes HTTP/2 cannot express, so a typo fails at startup
201    /// rather than at the first h2 handshake (`h2::server::Builder` asserts
202    /// `size <= MAX_WINDOW_SIZE` and would panic inside a worker).
203    fn validate(&self) -> io::Result<()> {
204        for (name, value) in [
205            (
206                "initial_stream_window_size",
207                self.initial_stream_window_size,
208            ),
209            (
210                "initial_connection_window_size",
211                self.initial_connection_window_size,
212            ),
213        ] {
214            let Some(v) = value else { continue };
215            if v == 0 || v > MAX_H2_WINDOW_SIZE {
216                return Err(io::Error::new(
217                    io::ErrorKind::InvalidInput,
218                    format!(
219                        "server.h2.{name} must be between 1 and {MAX_H2_WINDOW_SIZE} \
220                         bytes (RFC 9113 §6.5.2); got {v}"
221                    ),
222                ));
223            }
224        }
225        Ok(())
226    }
227
228    /// True when nothing is configured, i.e. actix's defaults are untouched.
229    pub fn is_default(&self) -> bool {
230        *self == Self::default()
231    }
232
233    /// The window actix will actually use per stream (configured or default) —
234    /// for the startup log line only.
235    pub fn effective_stream_window(&self) -> u32 {
236        self.initial_stream_window_size
237            .unwrap_or(ACTIX_DEFAULT_H2_STREAM_WINDOW)
238    }
239
240    /// The window actix will actually use per connection (configured or
241    /// default) — for the startup log line only.
242    pub fn effective_connection_window(&self) -> u32 {
243        self.initial_connection_window_size
244            .unwrap_or(ACTIX_DEFAULT_H2_CONNECTION_WINDOW)
245    }
246}

248/// Build a rustls [`ClientCertVerifier`] from the configured client-CA bundle
249/// (D3 native mTLS).
250///
251/// Loads the PEM bundle at `client_ca_path` into a [`RootCertStore`] and builds
252/// a [`WebPkiClientVerifier`] over it, using the same [`CryptoProvider`] as the
253/// server config. For [`ClientAuth::Optional`] the verifier still *offers* and
254/// *verifies* client certs but permits anonymous clients
255/// (`allow_unauthenticated`); for [`ClientAuth::Required`] a verified client
256/// cert is mandatory.
257///
258/// Fails fast (aborting startup) when the CA path is unset, missing/unreadable,
259/// empty, or malformed — matching this file's existing `io::Error` style so a
260/// misconfigured mTLS server never starts.
261fn read_client_ca_roots(tls: &TlsConfig) -> io::Result<RootCertStore> {
262    let ca_path = tls.client_ca_path.as_ref().ok_or_else(|| {
263        io::Error::new(
264            io::ErrorKind::InvalidInput,
265            "server.tls.client_auth is optional/required but \
266             server.tls.client_ca_path is not set",
267        )
268    })?;
269
270    let pem = std::fs::read_to_string(ca_path).map_err(|e| {
271        io::Error::new(
272            e.kind(),
273            format!(
274                "failed to open TLS client CA bundle {}: {e}",
275                ca_path.display()
276            ),
277        )
278    })?;
279    let roots = roots_from_pem(&pem, &ca_path.display().to_string())?;
280    if roots.is_empty() {
281        return Err(io::Error::new(
282            io::ErrorKind::InvalidData,
283            format!("no client CA certificates found in {}", ca_path.display()),
284        ));
285    }
286    Ok(roots)
287}

308/// Install `pem` as the live client trust anchor set, without a restart.
309///
310/// Returns the number of anchors now trusted, or `None` when this process has
311/// no TLS listener with client authentication enabled — a plaintext deployment,
312/// or one whose operator set `client_auth = off`. That is not a failure: there
313/// is nothing to reload into, and saying so lets the caller report "saved, and
314/// it will apply when TLS is enabled" rather than a spurious error.
315///
316/// # Errors
317///
318/// A bundle that does not parse, or a certificate webpki refuses as a trust
319/// anchor. The live anchor set is left **unchanged** in that case: the new
320/// bundle is fully parsed and the verifier fully built before anything is
321/// swapped, so a bad reload cannot leave the listener trusting nothing.
322pub fn reload_trust_anchors(pem: &str) -> io::Result<Option<usize>> {
323    let Some((verifier, provider)) = LIVE_VERIFIER.get() else {
324        return Ok(None);
325    };
326    let roots = roots_from_pem(pem, "the mTLS trust anchor bundle")?;
327    let count = verifier.replace(roots, provider)?;
328    Ok(Some(count))
329}

335/// Read the configured certificate chain and private key and prove they match.
336///
337/// Shared by the boot path and every reload, so a certificate that would be
338/// rejected at startup is rejected identically at 3am when certbot replaces it
339/// — the alternative being two subtly different parsers and a renewal that
340/// installs something the next boot refuses.
341///
342/// # Errors
343///
344/// A missing or unreadable file (`NotFound`/`PermissionDenied`), a PEM the
345/// parser rejects, an empty chain, or a key that does not match the leaf's
346/// public key.
347fn read_certified_key(
348    cert_path: &std::path::Path,
349    key_path: &std::path::Path,
350    provider: &rustls::crypto::CryptoProvider,
351) -> io::Result<CertifiedKey> {
352    // Open the files explicitly so a missing/unreadable path yields a clean
353    // io::Error (NotFound / PermissionDenied) before any PEM parsing. Cert and
354    // key are parsed via rustls-pki-types' `PemObject` trait directly —
355    // rustls-pemfile is unmaintained (RUSTSEC-2025-0134) and is a thin wrapper
356    // over this same code.
357    let cert_file = File::open(cert_path).map_err(|e| {
358        io::Error::new(
359            e.kind(),
360            format!("failed to open TLS cert file {}: {e}", cert_path.display()),
361        )
362    })?;
363    let cert_chain: Vec<CertificateDer<'static>> = CertificateDer::pem_reader_iter(cert_file)
364        .collect::<Result<Vec<_>, _>>()
365        .map_err(|e| {
366            io::Error::new(
367                io::ErrorKind::InvalidData,
368                format!(
369                    "failed to parse TLS certificates from {}: {e}",
370                    cert_path.display()
371                ),
372            )
373        })?;
374    if cert_chain.is_empty() {
375        return Err(io::Error::new(
376            io::ErrorKind::InvalidData,
377            format!("no certificates found in {}", cert_path.display()),
378        ));
379    }
380
381    let key_file = File::open(key_path).map_err(|e| {
382        io::Error::new(
383            e.kind(),
384            format!("failed to open TLS key file {}: {e}", key_path.display()),
385        )
386    })?;
387    let key = PrivateKeyDer::from_pem_reader(key_file).map_err(|e| {
388        io::Error::new(
389            io::ErrorKind::InvalidData,
390            format!(
391                "failed to read a TLS private key from {}: {e}",
392                key_path.display()
393            ),
394        )
395    })?;
396
397    // `from_der` loads the key through the provider and compares its SPKI with
398    // the leaf's, which is the check that makes a half-written renewal — the
399    // new chain beside the old key — fail here rather than at a client's
400    // handshake.
401    CertifiedKey::from_der(cert_chain, key, provider).map_err(|e| {
402        io::Error::new(
403            io::ErrorKind::InvalidData,
404            format!("invalid TLS certificate/key pair: {e}"),
405        )
406    })
407}

409/// A [`ResolvesServerCert`] whose certificate can be replaced while the server
410/// is listening.
411///
412/// # Why this exists
413///
414/// rustls consults the resolver on every handshake but reads nothing from disk;
415/// `with_single_cert` installs an immutable `SingleCertAndKey` and actix binds
416/// the resulting config for the process's life. So the leaf a server started
417/// with is the leaf it serves forever.
418///
419/// That is fine for a certificate an operator installs by hand and lives with
420/// for a year. It is not fine for ACME: Let's Encrypt issues for 90 days,
421/// clients renew at 60, and the only remedy without this type is to restart the
422/// process every couple of months — which for an identity provider means
423/// dropping in-flight requests and re-reading every secret out of Vault on a
424/// schedule, to work around a `Vec<u8>` that could have been swapped.
425///
426/// The shape deliberately mirrors [`ReloadableClientCertVerifier`]: an
427/// `ArcSwap` consulted on the hot path, replaced wholesale by a reload that has
428/// already done all of its fallible work. There is one mechanism for "TLS
429/// material changed while we were running", not two.
430#[derive(Debug)]
431pub struct ReloadableCertResolver {
432    current: ArcSwap<CertifiedKey>,
433}
434
435impl ReloadableCertResolver {
436    /// Start serving `initial`.
437    pub fn new(initial: CertifiedKey) -> Self {
438        Self {
439            current: ArcSwap::from_pointee(initial),
440        }
441    }
442
443    /// Swap in `next`, returning the certificate that was replaced.
444    ///
445    /// Infallible by construction: every way this can fail — unreadable file,
446    /// malformed PEM, mismatched pair — has already happened (or not) in
447    /// [`read_certified_key`]. A caller that reaches this point holds a
448    /// certificate rustls has already accepted, so the listener cannot be left
449    /// serving nothing.
450    pub fn replace(&self, next: CertifiedKey) -> Arc<CertifiedKey> {
451        self.current.swap(Arc::new(next))
452    }
453
454    /// The certificate currently being served.
455    pub fn current(&self) -> Arc<CertifiedKey> {
456        self.current.load_full()
457    }
458}
459
460impl ResolvesServerCert for ReloadableCertResolver {
461    fn resolve(&self, _client_hello: ClientHello<'_>) -> Option<Arc<CertifiedKey>> {
462        // Deliberately ignores SNI, exactly as `SingleCertAndKey` does: this
463        // listener serves one deployment on one name. Returning the certificate
464        // regardless of the name offered is what lets a client connecting to
465        // `127.0.0.1` with an overridden SNI (how the edge proxy reaches the
466        // backend on the loopback bind) complete the handshake against the
467        // public leaf.
468        Some(self.current.load_full())
469    }
470}

472/// The live resolver, the provider to rebuild with, and where to re-read from.

474/// A process-global for the same reason [`LIVE_VERIFIER`] is one: there is a

476/// trigger a reload run outside anything holding the `ServerConfig` actix
477/// built. Set once, during [`build_rustls_server_config`].
478#[allow(clippy::type_complexity)]
479static LIVE_CERT_RESOLVER: std::sync::OnceLock<(
480    Arc<ReloadableCertResolver>,
481    Arc<rustls::crypto::CryptoProvider>,
482    std::path::PathBuf,
483    std::path::PathBuf,
484)> = std::sync::OnceLock::new();

486fn install_reloadable_resolver(
487    resolver: Arc<ReloadableCertResolver>,
488    provider: Arc<rustls::crypto::CryptoProvider>,
489    cert_path: std::path::PathBuf,
490    key_path: std::path::PathBuf,

495}

497/// Re-read the configured certificate and key and install them on the live
498/// listener, without a restart.
499///
500/// Returns `Ok(None)` when this process has no TLS listener — a plaintext
501/// deployment behind a terminating proxy. That is not a failure: there is
502/// nothing to reload into, and saying so lets a `SIGHUP` handler log "no TLS
503/// listener" rather than an error an operator would go looking for.
504///
505/// Returns `Ok(Some(false))` when the files parsed but are byte-identical to
506/// what is already being served, so a poll that fires between renewals stays
507/// silent instead of logging a reload an hour, forever.
508///
509/// # Errors
510///
511/// Anything [`read_certified_key`] rejects. **The live certificate is left
512/// unchanged in that case** — the new pair is fully read, parsed and
513/// consistency-checked before anything is swapped, so a renewal caught
514/// half-written (certbot writes the chain and the key as two separate
515/// operations) fails this call and is retried on the next tick rather than
516/// taking the listener down.
517pub fn reload_leaf_certificate() -> io::Result<Option<bool>> {
518    let Some((resolver, provider, cert_path, key_path)) = LIVE_CERT_RESOLVER.get() else {
519        return Ok(None);
520    };
521    reload_into(resolver, provider, cert_path, key_path).map(Some)
522}

524/// The reload itself, against an explicit resolver rather than the process
525/// global.
526///
527/// Split out so the behaviour that matters — swap on change, no-op when
528/// unchanged, leave the old certificate serving on a bad pair — is testable
529/// directly. Going through [`reload_leaf_certificate`] would mean racing for
530/// [`LIVE_CERT_RESOLVER`]'s single `set`, which whichever test ran first would
531/// win; a mechanism whose failure mode is "serves the wrong certificate" should
532/// not be tested order-dependently.
533///
534/// # Errors
535///
536/// Anything [`read_certified_key`] rejects. The resolver is left untouched.
537fn reload_into(
538    resolver: &ReloadableCertResolver,
539    provider: &rustls::crypto::CryptoProvider,
540    cert_path: &std::path::Path,
541    key_path: &std::path::Path,
542) -> io::Result<bool> {
543    let next = read_certified_key(cert_path, key_path, provider)?;
544    // Compare the DER chain rather than file bytes: PEM whitespace, comment
545    // lines and the ordering certbot happens to write are not differences a
546    // handshake can see, and a poll that reloaded on them would churn.
547    if resolver.current().cert == next.cert {
548        return Ok(false);
549    }
550    resolver.replace(next);
551    Ok(true)
552}

554/// Report the outcome of one reload attempt.
555///
556/// A free function rather than a closure inside [`spawn_leaf_reloader`] so the
557/// mapping from outcome to log level is testable: the WARN-not-ERROR choice on
558/// a failed reload is deliberate (the previous certificate is still serving)
559/// and is exactly the kind of decision that gets "tidied" into an ERROR by
560/// someone who has not read why.
561fn log_reload_outcome(trigger: &'static str, outcome: io::Result<Option<bool>>) {
562    match outcome {
563        Ok(Some(true)) => tracing::info!(
564            trigger,
565            "TLS leaf certificate reloaded; new connections use it immediately"
566        ),
567        Ok(Some(false)) => tracing::debug!(
568            trigger,
569            "TLS leaf certificate unchanged on disk; nothing to reload"
570        ),
571        Ok(None) => tracing::debug!(
572            trigger,
573            "no direct-TLS listener in this process; nothing to reload"
574        ),
575        // WARN, not ERROR, and deliberately not fatal: the previous
576        // certificate is still serving. The likeliest cause is a renewal
577        // observed mid-write, which the next trigger resolves on its own.
578        Err(e) => tracing::warn!(
579            trigger,
580            error = %e,
581            "TLS leaf certificate reload failed; continuing to serve the \
582             previous certificate"
583        ),
584    }
585}

587/// Start the background triggers that keep the leaf certificate current.
588///
589/// Spawns, on the ambient tokio runtime:
590///
591/// * a `SIGHUP` listener — the conventional operator hook, and what an ACME
592///   `--deploy-hook` sends. Immediate.
593/// * a `stat` poll every `interval_secs`, skipped entirely when that is `0`.
594///
595/// Both funnel into [`reload_leaf_certificate`], which is a no-op when the
596/// certificate on disk is the one already being served. Calling this on a
597/// plaintext deployment is harmless: the reload reports "no TLS listener" and
598/// the tasks idle.
599///
600/// # Why both, and why the poll is not the only one
601///
602/// The signal is the correct mechanism and takes effect within milliseconds.
603/// It is also the one that silently does not happen: a deploy hook nobody
604/// wired up, a container runtime that does not forward signals to PID 1, an
605/// operator who renews by hand. The poll is what turns "the certificate
606/// expired in production" into "the certificate was replaced within the hour",
607/// and it costs one read of two small files an hour.
608pub fn spawn_leaf_reloader(interval_secs: u64) {
609    tokio::spawn(async move {
610        let mut hangup = match tokio::signal::unix::signal(tokio::signal::unix::SignalKind::hangup())
611        {
612            Ok(s) => s,
613            Err(e) => {
614                tracing::warn!(
615                    error = %e,
616                    "could not install a SIGHUP handler; TLS certificate reload                      is available only through the periodic poll"
617                );
618                return;
619            }
620        };
621        while hangup.recv().await.is_some() {
622            log_reload_outcome("sighup", reload_leaf_certificate());
623        }
624    });
625
626    if interval_secs == 0 {
627        tracing::info!(
628            "TLS certificate reload polling is disabled              (server.tls.reload_interval_secs = 0); SIGHUP still reloads"
629        );
630        return;
631    }
632
633    tokio::spawn(async move {
634        let mut ticker = tokio::time::interval(std::time::Duration::from_secs(interval_secs));
635        // The first tick fires immediately and would re-read the certificate
636        // the boot path just read. Burn it.
637        ticker.tick().await;
638        loop {
639            ticker.tick().await;
640            log_reload_outcome("poll", reload_leaf_certificate());
641        }
642    });
643}

649/// A [`ClientCertVerifier`] whose trust anchors can be replaced while the
650/// server is listening.
651///
652/// # Why this exists
653///
654/// rustls builds its client trust store when the `ServerConfig` is constructed,
655/// and actix binds that config for the process's life. Flagging a CA as an mTLS
656/// trust anchor therefore used to take effect at the *next* boot, and the API
657/// said so rather than pretending otherwise — which is honest and still means
658/// an operator adding a device CA has to restart the server every browser
659/// session and every IoT device is currently connected to.
660///
661/// rustls consults the verifier **per handshake**, not once at construction. So
662/// a verifier that delegates to a swappable inner one gives the config
663/// something permanent to hold while the anchors behind it change. Connections
664/// already established keep the verifier they handshook with, which is correct:
665/// a certificate accepted a moment ago does not become invalid mid-connection,
666/// and revocation of an established session is what session invalidation is
667/// for.
668///
669/// # The empty state
670///
671/// [`Anchors::None`] is not "trust nothing" — it is "do not ask for a client
672/// certificate at all", the same posture as `with_no_client_auth()`. A
673/// deployment that boots with no flagged CA must behave exactly as it does
674/// today, *and* must be able to reach the anchored state without a restart.
675/// Installing this verifier unconditionally is what makes both true;
676/// `WebPkiClientVerifier` cannot express it, because it refuses to build over
677/// an empty root store.
678#[derive(Debug)]
679pub struct ReloadableClientCertVerifier {
680    anchors: ArcSwap<Anchors>,
681    /// Whether a verified client certificate is required once anchors exist.
682    ///
683    /// Fixed at construction from `client_auth`: it is an operator's policy
684    /// decision, not a property of the anchor set, and changing it changes
685    /// whether unauthenticated clients can connect at all.
686    mandatory: bool,
687}
688
689/// What the verifier currently trusts.
690#[derive(Debug)]
691enum Anchors {
692    /// No trust anchors: no client certificate is requested.
693    None,
694    /// A webpki verifier over the current anchor set.
695    Some(Arc<dyn ClientCertVerifier>),
696}
697
698impl ReloadableClientCertVerifier {
699    /// An empty verifier that offers no client authentication.
700    pub fn empty(mandatory: bool) -> Self {
701        Self {
702            anchors: ArcSwap::from_pointee(Anchors::None),
703            mandatory,
704        }
705    }
706
707    /// Replace the trust anchors with a verifier built over `roots`.
708    ///
709    /// An empty `roots` returns to [`Anchors::None`] rather than building a
710    /// verifier that trusts nothing: a verifier that requests a certificate and
711    /// then rejects every one is strictly worse than not requesting one, and it
712    /// is what un-flagging the last CA should produce.
713    pub fn replace(
714        &self,
715        roots: RootCertStore,
716        provider: &Arc<rustls::crypto::CryptoProvider>,
717    ) -> io::Result<usize> {
718        if roots.is_empty() {
719            self.anchors.store(Arc::new(Anchors::None));
720            return Ok(0);
721        }
722        let count = roots.len();
723        let builder =
724            WebPkiClientVerifier::builder_with_provider(Arc::new(roots), provider.clone());
725        let verifier = if self.mandatory {
726            builder.build()
727        } else {
728            builder.allow_unauthenticated().build()
729        }
730        .map_err(|e| io::Error::other(format!("failed to build client cert verifier: {e}")))?;
731
732        self.anchors.store(Arc::new(Anchors::Some(verifier)));
733        Ok(count)
734    }
735
736    /// How many anchors are currently installed.
737    pub fn anchor_count(&self) -> usize {
738        match &**self.anchors.load() {
739            Anchors::None => 0,
740            Anchors::Some(v) => v.root_hint_subjects().len(),
741        }
742    }
743
744    /// The current inner verifier, if any.
745    fn current(&self) -> Option<Arc<dyn ClientCertVerifier>> {
746        match &**self.anchors.load() {
747            Anchors::None => None,
748            Anchors::Some(v) => Some(Arc::clone(v)),
749        }
750    }
751}
752
753impl ClientCertVerifier for ReloadableClientCertVerifier {
754    fn offer_client_auth(&self) -> bool {
755        // Consulted per handshake, which is the whole mechanism: the answer
756        // changes the moment `replace` installs an anchor set.
757        self.current().is_some()
758    }
759
760    fn client_auth_mandatory(&self) -> bool {
761        // Only meaningful while anchors exist. Answering `true` with none
762        // installed would refuse every connection to a server that cannot
763        // verify anybody — a self-inflicted outage on a deployment that
764        // un-flagged its last CA.
765        self.mandatory && self.current().is_some()
766    }
767
768    fn root_hint_subjects(&self) -> &[rustls::DistinguishedName] {
769        // Cannot borrow through the ArcSwap guard, and the hint is advisory:
770        // it tells a client which issuers the server would accept, and a client
771        // that offers a certificate anyway is verified normally by
772        // `verify_client_cert`. Returning nothing costs a round trip in the
773        // worst case and never accepts anything it should not.
774        &[]
775    }
776
777    fn verify_client_cert(
778        &self,
779        end_entity: &CertificateDer<'_>,
780        intermediates: &[CertificateDer<'_>],
781        now: rustls::pki_types::UnixTime,
782    ) -> Result<rustls::server::danger::ClientCertVerified, rustls::Error> {
783        match self.current() {
784            Some(v) => v.verify_client_cert(end_entity, intermediates, now),
785            // Unreachable while rustls honours `offer_client_auth`, and a
786            // refusal rather than an acceptance if it ever does not.
787            None => Err(rustls::Error::General(
788                "no client trust anchors are configured".into(),
789            )),
790        }
791    }
792
793    fn verify_tls12_signature(
794        &self,
795        message: &[u8],
796        cert: &CertificateDer<'_>,
797        dss: &rustls::DigitallySignedStruct,
798    ) -> Result<rustls::client::danger::HandshakeSignatureValid, rustls::Error> {
799        match self.current() {
800            Some(v) => v.verify_tls12_signature(message, cert, dss),
801            None => Err(rustls::Error::General(
802                "no client trust anchors are configured".into(),
803            )),
804        }
805    }
806
807    fn verify_tls13_signature(
808        &self,
809        message: &[u8],
810        cert: &CertificateDer<'_>,
811        dss: &rustls::DigitallySignedStruct,
812    ) -> Result<rustls::client::danger::HandshakeSignatureValid, rustls::Error> {
813        match self.current() {
814            Some(v) => v.verify_tls13_signature(message, cert, dss),
815            None => Err(rustls::Error::General(
816                "no client trust anchors are configured".into(),
817            )),
818        }
819    }
820
821    fn supported_verify_schemes(&self) -> Vec<rustls::SignatureScheme> {
822        match self.current() {
823            Some(v) => v.supported_verify_schemes(),
824            // The provider's full set: this is a capability advertisement, and
825            // an empty list would make a handshake fail to negotiate rather
826            // than fall through to the "no anchors" path above.
827            None => rustls::crypto::ring::default_provider()
828                .signature_verification_algorithms
829                .supported_schemes(),
830        }
831    }
832}

834/// Build a [`RootCertStore`] from concatenated PEM.
835///
836/// Shared by the boot path and the reload path so a hot-reloaded anchor set is
837/// parsed by exactly the same code as one read at startup — two parsers is how
838/// a bundle comes to be accepted at boot and rejected on reload.
839pub fn roots_from_pem(pem: &str, source: &str) -> io::Result<RootCertStore> {
840    let certs: Vec<CertificateDer<'static>> = CertificateDer::pem_slice_iter(pem.as_bytes())
841        .collect::<Result<Vec<_>, _>>()
842        .map_err(|e| {
843            io::Error::new(
844                io::ErrorKind::InvalidData,
845                format!("failed to parse client CA certificates from {source}: {e}"),
846            )
847        })?;
848
849    let mut roots = RootCertStore::empty();
850    for cert in certs {
851        roots.add(cert).map_err(|e| {
852            io::Error::new(
853                io::ErrorKind::InvalidData,
854                format!("invalid client CA certificate in {source}: {e}"),
855            )
856        })?;
857    }
858    Ok(roots)
859}

861/// Build a TLS 1.3-only rustls [`ServerConfig`] from the configured PEM files.
862///
863/// Fails fast (returning an `io::Error` that aborts startup) when TLS is enabled
864/// but misconfigured: a missing `cert_path`/`key_path`, an unreadable or
865/// malformed PEM file, an empty cert chain, a cert/key mismatch, or an
866/// unsatisfiable `http2 = false` request (see [`reject_unsupported_http2_knob`]).
867pub fn build_rustls_server_config(tls: &TlsConfig) -> io::Result<ServerConfig> {
868    // Checked before any file IO: an unsatisfiable ALPN request is a config
869    // error regardless of whether the cert/key happen to be readable.
870    reject_unsupported_http2_knob(tls)?;
871
872    let cert_path = tls.cert_path.as_ref().ok_or_else(|| {
873        io::Error::new(
874            io::ErrorKind::InvalidInput,
875            "server.tls.enabled is true but server.tls.cert_path is not set",
876        )
877    })?;
878    let key_path = tls.key_path.as_ref().ok_or_else(|| {
879        io::Error::new(
880            io::ErrorKind::InvalidInput,
881            "server.tls.enabled is true but server.tls.key_path is not set",
882        )
883    })?;
884
885    // TLS 1.3 only (ASVS V9.1.2). ring provider selected explicitly.
886    let provider = Arc::new(rustls::crypto::ring::default_provider());
887    let builder = ServerConfig::builder_with_provider(provider.clone())
888        .with_protocol_versions(&[&rustls::version::TLS13])
889        .map_err(|e| io::Error::other(format!("rustls TLS 1.3 configuration failed: {e}")))?;
890
891    // Client-certificate (mTLS) policy (D3). `off` keeps the server-auth-only
892    // behaviour (`with_no_client_auth`); `optional`/`required` install a
893    // WebPkiClientVerifier over the configured CA bundle so rustls verifies the
894    // client cert during the handshake and the *verified* cert (not a header)
895    // drives certificate-based identity.
896    // Always a `ReloadableClientCertVerifier`, even when `client_auth` is off.
897    //
898    // rustls binds whatever it is given here for the process's life, so the
899    // choice made at this line decides whether flagging a CA later can take
900    // effect without a restart. `with_no_client_auth()` decides "no", forever.
901    // The reloadable verifier starts in exactly that posture — it offers no
902    // client authentication until anchors are installed — and can leave it.
903    //
904    // A listener that boots with `Off` therefore starts by requesting no
905    // client certificate and can still be given anchors later — which matches
906    // what `mtls_anchors::apply` already does at boot, where flagging a CA
907    // upgrades an unset `Off` to `Optional`.
908    let reloadable = Arc::new(ReloadableClientCertVerifier::empty(
909        tls.client_auth == ClientAuth::Required,
910    ));
911    if tls.client_auth != ClientAuth::Off {
912        // The bundle written by `mtls_anchors::apply` at boot, or one the
913        // operator curated themselves.
914        let roots = read_client_ca_roots(tls)?;
915        let count = reloadable.replace(roots, &provider)?;
916        tracing::info!(anchors = count, "client trust anchors loaded");
917    }
918    let builder =
919        builder.with_client_cert_verifier(reloadable.clone() as Arc<dyn ClientCertVerifier>);
920    install_reloadable_verifier(reloadable, Arc::clone(&provider));
921
922    // The leaf. Read, parsed and proven to match its key here, then installed
923    // behind an `ArcSwap` so a renewal can replace it without a restart.
924    //
925    // This is what `with_single_cert` does internally — `CertifiedKey::from_der`
926    // followed by `with_cert_resolver(SingleCertAndKey::from(..))` (rustls
927    // 0.23.43 `src/server/builder.rs:65-72`) — with the immutable
928    // `SingleCertAndKey` swapped for a resolver that can be reloaded. The error
929    // text is unchanged so the failure an operator sees for a mismatched pair
930    // is the one it has always been.
931    let certified = read_certified_key(cert_path, key_path, &provider)?;
932    let resolver = Arc::new(ReloadableCertResolver::new(certified));
933    install_reloadable_resolver(
934        Arc::clone(&resolver),
935        Arc::clone(&provider),
936        cert_path.clone(),
937        key_path.clone(),
938    );
939    let mut config = builder.with_cert_resolver(resolver as Arc<dyn ResolvesServerCert>);
940
941    // ALPN (B2/G8): h2 + http/1.1. This is not configurable — `http2 = false`
942    // was rejected above because actix re-prepends h2 regardless.
943    config.alpn_protocols = alpn_protocols();
944
945    // TLS 1.3 session resumption (B2). Without a ticketer + session store a
946    // rustls server does a *full* handshake on every connection; k6 opens many
947    // short-lived connections per VU, so a full ECDHE handshake per request is a
948    // per-request fixed cost that shows up as the ~2× p50 inflation on the token
949    // endpoints. Enabling stateless TLS 1.3 tickets lets repeat connections
950    // resume (PSK) instead. We deliberately do NOT enable 0-RTT/early-data:
951    // rustls' `max_early_data_size` stays at its default 0 because the token
952    // endpoints are non-idempotent POSTs and early data is replayable
953    // (see docs/security-profiles.md).
954    config.session_storage = ServerSessionMemoryCache::new(RESUMPTION_CACHE_SIZE);
955    match rustls::crypto::ring::Ticketer::new() {
956        Ok(ticketer) => config.ticketer = ticketer,
957        Err(e) => tracing::warn!(
958            error = %e,
959            "failed to construct a TLS ticketer; session resumption disabled \
960             (falling back to full handshakes)"
961        ),
962    }
963
964    Ok(config)
965}

1514    /// Pump handshake records between two in-memory rustls connections until
1515    /// both finish handshaking or one errors. Returns the first processing
1516    /// error (e.g. the server rejecting a missing required client cert).
1517    fn drive_handshake(
1518        client: &mut rustls::Connection,
1519        server: &mut rustls::Connection,
1520    ) -> Result<(), rustls::Error> {
1521        for _ in 0..16 {
1522            let mut buf = Vec::new();
1523            while client.wants_write() {
1524                client.write_tls(&mut buf).unwrap();
1525            }
1526            let mut rd: &[u8] = &buf;
1527            while !rd.is_empty() {
1528                server.read_tls(&mut rd).unwrap();
1529            }
1530            server.process_new_packets()?;
1531
1532            let mut buf = Vec::new();
1533            while server.wants_write() {
1534                server.write_tls(&mut buf).unwrap();
1535            }
1536            let mut rd: &[u8] = &buf;
1537            while !rd.is_empty() {
1538                client.read_tls(&mut rd).unwrap();
1539            }
1540            client.process_new_packets()?;
1541
1542            if !client.is_handshaking() && !server.is_handshaking() {
1543                return Ok(());
1544            }
1545        }
1546        Ok(())
1547    }
