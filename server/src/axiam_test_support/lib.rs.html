1//! Shared fixtures for AXIAM's test suites.
2//!
3//! Dev-dependency only, and deliberately a leaf: it pulls in `uuid` and
4//! nothing else, so no crate under test can ever cycle back through it.
5
6use std::sync::OnceLock;
7
8use uuid::Uuid;
9
10static TEST_PASSWORD: OnceLock<String> = OnceLock::new();
11
12/// The password every test fixture user is created with.
13///
14/// # Why this exists
15///
16/// This helper was previously re-declared in fourteen test files across six
17/// crates, with **four** different implementations — a per-call random, a
18/// process-stable env-var lookup, and two different split-string literals.
19/// Nothing reconciled them, so which semantics a given test got depended on
20/// which file it lived in. Alongside them sat raw literals, and one of those —
21/// a four-word passphrase repeated 21 times across two suites — was what a
22/// secret scanner eventually flagged as a critical finding on a pull request.
23///
24/// The finding was a false positive — the value guards nothing but an
25/// in-memory database that is created and dropped inside a single test — but
26/// the underlying problem was real: the convention that avoided it was
27/// followable only from memory, so each new test file re-decided it, and some
28/// decided wrong.
29///
30/// The offending literal is deliberately not reproduced above: quoting it,
31/// even in a doc comment, would re-trip the very scanner this helper exists to
32/// satisfy.
33///
34/// # Semantics
35///
36/// **Random per process, stable within one.** Both halves are load-bearing:
37///
38/// - *Random per process* means no credential-shaped literal appears in the
39///   source at all, so the scanner finding cannot recur by construction rather
40///   than by everyone remembering a rule.
41/// - *Stable within a process* is what makes this a safe replacement for the
42///   deterministic helpers it subsumes. Many callers create a user with this
43///   password and then authenticate as them, calling this function once per
44///   step and relying on the two agreeing. A naive per-call random would
45///   compile everywhere and fail those tests at runtime.
46///
47/// Callers that were previously per-call random now share one value within a
48/// binary. That is harmless: fixture users may share a password, and Argon2id
49/// salts per hash regardless, so no two stored hashes collide.
50///
51/// `AXIAM_TEST_PASSWORD` overrides it, preserving the escape hatch six of the
52/// previous helpers offered — useful when reproducing a failure against a
53/// database seeded by an earlier run.
54///
55/// # Composition
56///
57/// `Fx1!` prefixes a UUID's simple form, which satisfies every password policy
58/// in the codebase — upper, lower, digit, symbol, and comfortably past any
59/// length floor — so this is usable for the strict registration paths and not
60/// only for direct repository inserts.
61pub fn test_password() -> String {
62    TEST_PASSWORD
63        .get_or_init(|| {
64            std::env::var("AXIAM_TEST_PASSWORD")
65                .unwrap_or_else(|_| format!("Fx1!{}", Uuid::new_v4().simple()))
66        })
67        .clone()
68}
69
70/// A policy-valid password that is *not* [`test_password`], fresh per call.
71///
72/// Two uses, both wanting the same thing:
73///
74/// - the wrong-credentials half of an authentication test, where staying
75///   policy-valid is what keeps the negative case honest — a rejection then
76///   proves the credential check ran, rather than a length or complexity rule
77///   tripping first and passing the test for the wrong reason;
78/// - the replacement value in a change-password flow.
79///
80/// Unlike [`test_password`] this is *not* memoised: a caller rotating a
81/// password twice needs two distinct values, and a caller wanting one stable
82/// value can bind it to a variable.
83pub fn other_password() -> String {
84    format!("Nx9?{}", Uuid::new_v4().simple())
85}

87#[cfg(test)]
88mod tests {
89    use super::*;
90
91    #[test]
92    fn is_stable_within_a_process() {
93        // The property the deterministic call sites depend on.
94        assert_eq!(test_password(), test_password());
95    }
96
97    #[test]
98    fn other_password_differs_from_the_fixture_and_from_itself() {
99        assert_ne!(test_password(), other_password());
100        // Fresh per call — a double rotation needs two distinct values.
101        assert_ne!(other_password(), other_password());
102    }
103
104    #[test]
105    fn other_password_also_satisfies_the_policy() {
106        // Load-bearing for negative tests: a wrong password that is *invalid*
107        // proves nothing, because the policy check would reject it before the
108        // credential check ever ran.
109        let pw = other_password();
110        assert!(pw.len() >= 12);
111        assert!(pw.chars().any(|c| c.is_ascii_uppercase()));
112        assert!(pw.chars().any(|c| c.is_ascii_lowercase()));
113        assert!(pw.chars().any(|c| c.is_ascii_digit()));
114        assert!(pw.chars().any(|c| !c.is_ascii_alphanumeric()));
115    }
116
117    #[test]
118    fn satisfies_the_strictest_policy_in_the_codebase() {
119        let pw = test_password();
120        assert!(pw.len() >= 12, "length floor: {pw}");
121        assert!(pw.chars().any(|c| c.is_ascii_uppercase()), "upper: {pw}");
122        assert!(pw.chars().any(|c| c.is_ascii_lowercase()), "lower: {pw}");
123        assert!(pw.chars().any(|c| c.is_ascii_digit()), "digit: {pw}");
124        assert!(
125            pw.chars().any(|c| !c.is_ascii_alphanumeric()),
126            "symbol: {pw}"
127        );
128    }
129}
